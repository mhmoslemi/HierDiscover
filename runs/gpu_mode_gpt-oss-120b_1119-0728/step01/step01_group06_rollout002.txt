"""
TriMul (outgoing) forward pass – Triton‐accelerated contraction.

The heavy operation of the outgoing TriMul module is:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension `d` (dim‑wise we use `hidden_dim`) this is a
matrix multiplication `C = A @ Bᵀ` with
    A = left[b,…,d]   (shape N×N)
    B = right[b,…,d]  (shape N×N)

We fuse all hidden dimensions and the batch dimension into a single batch
(`B*hidden_dim`) and run a batched GEMM in a single Triton kernel.
Inputs are cast to FP16, the multiplication uses Tensor‑Core‑friendly
`tl.dot` (FP16×FP16→FP32 accumulation) and the results are written back as
FP32.  The rest of the TriMul pipeline (LayerNorms, linear projections,
gates) is kept in PyTorch – they are cheap compared to the N³ contraction.

The kernel works for any sequence length up to 1024, batch size ≤ 2 and
hidden dimension 128 (as required by the tests).  Block sizes are chosen
adaptively to keep occupancy high.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,                           # matrix sizes (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes, for each batch element:
        C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, k, j]
    A, B are FP16; the accumulator is FP32; C is stored as FP32.
    """
    pid_batch = tl.program_id(0)   # which (batch, hidden) pair
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # --------------------------------------------------------------
    # Offsets inside the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch entry
    a_base = a_ptr + pid_batch * a_batch_stride
    b_base = b_ptr + pid_batch * b_batch_stride
    c_base = c_ptr + pid_batch * c_batch_stride

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (blocked)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- Load A tile (M × K) -------------------------------------------------
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                # FP16 loads

        # ---- Load B tile (K × N) -------------------------------------------------
        # B is already stored in the transposed layout (K as rows, N as cols)
        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                # FP16 loads

        # ---- Compute tile --------------------------------------------------------
        acc += tl.dot(a, b)                # FP16×FP16 → FP32 accumulation

    # --------------------------------------------------------------
    # Write back the tile
    # --------------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to pick good block sizes based on sequence length
# ----------------------------------------------------------------------
def _choose_block_params(seqlen: int):
    """Select BLOCK_M/N/K to keep tiles well‑filled."""
    if seqlen <= 256:
        return 64, 64, 32
    elif seqlen <= 512:
        return 128, 128, 32
    else:  # up to 1024 in test suite
        return 128, 128, 64


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, C]  (float32)
        mask         : torch.Tensor [B, N, N] or None
        weights      : dict of model parameters (see problem statement)
        config       : dict with keys:
                         - "dim"         : C
                         - "hidden_dim" : H
                         - "nomask"      : bool (True → ignore mask)
    Returns
    -------
    torch.Tensor [B, N, N, C]  (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(dtype=dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the batched GEMM
    #    left: (B, N, N, H) -> (B, H, N, N)
    #    right: transpose k↔j -> (B, H, N, N)
    #    flatten batch & hidden -> (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left_h.shape
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) (k↔j)

    BH = B * hidden_dim
    left_flat = left_perm.reshape(BH, N, N)                  # (BH, N, N)
    right_flat = right_perm.reshape(BH, N, N)                # (BH, N, N)

    # Output buffer in FP32 (accumulated result)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 7. Gather stride information (element strides, not byte strides)
    # --------------------------------------------------------------
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)
    a_stride_k = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_stride_k = right_flat.stride(1)   # K dimension (row after transpose)
    b_stride_n = right_flat.stride(2)   # N dimension (col)

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # --------------------------------------------------------------
    # 8. Launch Triton kernel
    # --------------------------------------------------------------
    BLOCK_M, BLOCK_N, BLOCK_K = _choose_block_params(N)

    grid = (
        BH,                                 # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # tile rows
        (N + BLOCK_N - 1) // BLOCK_N,       # tile cols
    )

    _batched_gemm_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,               # M
        N,               # N
        N,               # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 9. Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 10. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, C]

    return out
