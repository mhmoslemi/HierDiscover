import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_gemm_kernel( a_ptr, b_ptr, c_ptr,
                          a_batch_stride, a_stride_m, a_stride_k,
                          b_batch_stride, b_stride_k, b_stride_n,
                          c_batch_stride, c_stride_m, c_stride_n,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Batched GEMM: out = A @ B.T with A of shape (batch, M, K), B of shape (batch, N, K) transposed.
    A and B are stored as half, accumulator is fp32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # strides in elements
    # offsets within tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * a_batch_stride
    b_base = b_ptr + batch_id * b_batch_stride
    c_base = c_ptr + batch_id * c_batch_stride

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # load B tile: (BLOCK_K, BLOCK_N) - B is already transposed (K dim first)
        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)

   becomes a batched GEMM:  out = left @ rightᵀ.
5. Run a custom Triton kernel that computes the batched GEMM
   with fp16 operands and fp32 accumulation (tensor‑core friendly).
6. Reshape the result back to (B, N, N, H), apply a second
   LayerNorm, multiply by the out‑gate, and finish with a linear
   projection to the original feature dimension.
The heavy O(N³) contraction is completely offloaded to Triton,
while the cheap linear / LayerNorm / sigmoid ops use highly
optimised PyTorch kernels.  All arithmetic ends in float32 as
required by the model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.
    A : (batch, M, K)   – fp16
    B : (batch, K, N)   – fp16 (already the transposed right‑matrix)
    C : (batch, M, N)   – fp32 accumulator, stored as fp32
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * a_batch_stride
    b_base = b_ptr + batch_id * b_batch_stride
    c_base = c_ptr + batch_id * c_batch_stride

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile : (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # B tile : (BLOCK_K, BLOCK_N)   (K is first dim because B is already transposed)
        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

        # matrix multiply – accumulates in FP32
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a fused Triton GEMM.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, C] (float32)
        mask         : torch.Tensor [B, N, N] or None
        weights      : dict with the module parameters
        config       : dict with "dim", "hidden_dim", "nomask" (bool) etc.

    Returns
    -------
    torch.Tensor  [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]          # C
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]

    # ------------------------------------------------------------------ #
    # 2) Fuse projection + gate for left/right sides
    # ------------------------------------------------------------------ #
    B, N, _, _ = x.shape

    # flatten spatial dims for a single large mat‑mul
    x_flat = x.reshape(B * N * N, dim)                     # (B·N·N, C)

    # left_proj + left_gate
    left_cat_w = torch.cat([weights["left_proj.weight"],
                            weights["left_gate.weight"]], dim=0)   # (2·H, C)
    left_cat = F.linear(x_flat, left_cat_w)               # (B·N·N, 2·H)
    left_cat = left_cat.view(B, N, N, 2 * hidden_dim)
    left_proj = left_cat[..., :hidden_dim]                # (B, N, N, H)
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])  # (B, N, N, H)

    # right_proj + right_gate
    right_cat_w = torch.cat([weights["right_proj.weight"],
                             weights["right_gate.weight"]], dim=0)  # (2·H, C)
    right_cat = F.linear(x_flat, right_cat_w)
    right_cat = right_cat.view(B, N, N, 2 * hidden_dim)
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # out_gate (separate linear)
    out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"]))
    out_gate = out_gate.view(B, N, N, hidden_dim)          # (B, N, N, H)

    # ------------------------------------------------------------------ #
    # 3) Apply mask (if any) and gating, cast to fp16 for the kernel
    # ------------------------------------------------------------------ #
    if mask is not None and not nomask:
        mask_fp = mask.to(dtype=torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left_proj.to(torch.float16) * left_gate.to(torch.float16) * mask_fp
        right = right_proj.to(torch.float16) * right_gate.to(torch.float16) * mask_fp
    else:
        left = left_proj.to(torch.float16) * left_gate.to(torch.float16)
        right = right_proj.to(torch.float16) * right_gate.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 4) Prepare tensors for the batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------ #
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs to be transposed on the last two axes before flattening
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim                     # merged batch dimension
    left_flat = left_perm.reshape(BH, N, N)   # (BH, M=N, K=N)
    right_flat = right_perm.reshape(BH, N, N) # (BH, K=N, N)

    # output buffer (fp32 accumulator)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # strides in element units (required by Triton)
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)   # stride for M (row)
    a_stride_k = left_flat.stride(2)   # stride for K (col)

    b_batch_stride = right_flat.stride(0)
    b_stride_k = right_flat.stride(1)  # K dimension stride
    b_stride_n = right_flat.stride(2)  # N dimension stride

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # ------------------------------------------------------------------ #
    # 5) Launch the Triton kernel
    # ------------------------------------------------------------------ #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,            # M
        N,            # N
        N,            # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------ #
    # 6) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------ #
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 7) Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # (B, N, N, H)

    out = out * out_gate                               # gated output
    out = F.linear(out, weights["to_out.weight"])      # (B, N, N, C)

    return out
```
