"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched matmul.

The heavy work is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be interpreted as a separate matrix multiplication for every
(b, d) pair:
    out_{b,d} = left_{b,d} @ right_{b,d}.T
We pack the hidden dimension `d` into the batch dimension, cast the
operands to fp16 and perform the multiplication with an fp32 accumulator.
All other layers (LayerNorm, linear projections, sigmoid gates) are
executed with PyTorch’s highly‑optimised kernels.

The Triton kernel is a classic tiled GEMM:
* BLOCK_M × BLOCK_N output tile.
* BLOCK_K reduction tile.
* Optional `num_warps` tuning for H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
# A, B, C : [batch, N, N] (half‑precision), accumulation in fp32.
# ----------------------------------------------------------------------
@triton.jit
def batch_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # dimensions (M = N = K = seq_len)
    stride_am, stride_ak, stride_ab,   # A strides: row, col, batch
    stride_bm, stride_bk, stride_bb,   # Bᵀ strides: col, row, batch
    stride_cm, stride_cn, stride_cb,   # C strides: row, col, batch
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
):
    """Batched GEMM: for each batch `b`, compute
       C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k].
    """
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    batch_id = tl.program_id(0)      # which (b, hidden) batch
    pid_m    = tl.program_id(1)      # block row
    pid_n    = tl.program_id(2)      # block col

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the tiles of A and Bᵀ
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + batch_id * stride_bb + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + offs_n[None, :] * stride_bm

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = _k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # (BLOCK_K, BLOCK_N) fp16

        # Matrix‑multiply the tile (tensor core friendly)
        acc += tl.dot(a, b)

        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) and gating pre‑activations
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])
    left_gate_p  = F.linear(x, weights["left_gate.weight"])
    right_gate_p = F.linear(x, weights["right_gate.weight"])
    out_gate_p   = F.linear(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = left_gate_p.sigmoid()
    right_gate = right_gate_p.sigmoid()
    out_gate   = out_gate_p.sigmoid()   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling (mask is 0/1)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((batch_size, seq_len, seq_len),
                                 dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)   # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask + gating
    # ------------------------------------------------------------------
    left  = left_proj * mask_tensor * left_gate   # [B,N,N,hidden_dim]
    right = right_proj * mask_tensor * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Cast to fp16 for the N³ kernel
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7️⃣ Pack hidden dimension into the batch dimension
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = batch_size * hidden_dim
    left_h  = left_h.view(batch_h, seq_len, seq_len)
    right_h = right_h.view(batch_h, seq_len, seq_len)

    # ------------------------------------------------------------------
    # 8️⃣ Allocate output buffer (half)
    # ------------------------------------------------------------------
    out_h = torch.empty_like(left_h, dtype=torch.float16)   # (B*H, N, N)

    # ------------------------------------------------------------------
    # 9️⃣ Extract strides (in element units, not bytes)
    # ------------------------------------------------------------------
    # A (left_h) layout: (batch, row, col)
    stride_ab, stride_am, stride_ak = left_h.stride()
    # B (right_h) layout is identical
    b_batch, b_row, b_col = right_h.stride()
    stride_bb = b_batch                     # batch stride
    stride_bm = b_row                       # column stride of Bᵀ (j dimension)
    stride_bk = b_col                       # row stride of Bᵀ (k dimension)

    # C strides follow A's layout
    stride_cb, stride_cm, stride_cn = out_h.stride()

    # ------------------------------------------------------------------
    # 10️⃣ Choose block sizes (tuned for H100)
    # ------------------------------------------------------------------
    if seq_len <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        NUM_WARPS = 4
    else:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        NUM_WARPS = 8

    # ------------------------------------------------------------------
    # 11️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    grid = (
        batch_h,                                 # batch dimension (B·H)
        triton.cdiv(seq_len, BLOCK_M),           # block rows
        triton.cdiv(seq_len, BLOCK_N),           # block cols
    )
    batch_tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        seq_len,
        seq_len,
        seq_len,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bm,
        stride_bk,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_WARPS=NUM_WARPS,
    )

    # ------------------------------------------------------------------
    # 12️⃣ Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = out_h.view(batch_size, hidden_dim, seq_len, seq_len)
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]
    out = out.to(torch.float32)                  # back to FP32 for the remaining ops

    # ------------------------------------------------------------------
    # 13️⃣ Output LayerNorm over hidden_dim and apply output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                         # broadcast over dim

    # ------------------------------------------------------------------
    # 14️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
