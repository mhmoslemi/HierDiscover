"""
TriMul (outgoing) – fast forward implementation.

The heavy “TriMul” part computes, for each batch and hidden channel d,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with left/right derived from a normalized input, a linear projection and
an element‑wise gate (sigmoid).  This is a batched matrix‑multiply
(L @ Rᵀ) of shape (B·H, N, N).  We implement this multiplication with a
single Triton kernel that processes many (B·H) slices in parallel and
uses Tensor‑Core‐friendly fp16·fp16→fp32 accumulation.  The surrounding
layer‑norms, linear layers and gating are performed with PyTorch (cuBLAS)
because they are already highly optimized.

Key optimisations:
 * fp16 data for the GEMM, fp32 accumulation.
 * Block sizes (64 × 64 × 64) tuned for H100.
 * A single launch with a 3‑D grid (M‑blocks, N‑blocks, batch·hidden) –
   no host‑side loops.
 * Optional mask is fused into the kernel (the mask is broadcast over the
   hidden dimension before launching).
 * All tensors stay on GPU; minimal copies – only the necessary
   permutes for the matmul layout.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) for many (batch*hidden) slices.
# A : (B·H, M, K) – fp16
# B : (B·H, K, N) – fp16  (already transposed, i.e. rightᵀ)
# C : (B·H, M, N) – fp32 accumulator, final result
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,   # A strides  (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # B strides  (batch, row(K), col)
    stride_Cb, stride_Ci, stride_Cn,   # C strides  (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    mask_ptr: tl.constexpr,            # 0 = no mask, 1 = mask present
    mask_stride_b: tl.constexpr,
    mask_stride_i: tl.constexpr,
    mask_stride_j: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)   # block row
    pid_n = tl.program_id(axis=1)   # block col
    pid_b = tl.program_id(axis=2)   # batch*hidden index

    # --------------------------------------------------------------
    # Offsets in M, N, K dimensions (static within a block)
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current tile
    # ------------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Ai
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk
        + offs_n[None, :] * stride_Bn
    )

    # ------------------------------------------------------------------
    # Optional mask (same for all hidden channels).  The mask is a
    # tensor of shape [B, N, N] (bool/float) – we broadcast it over the
    # hidden dimension inside the kernel.
    # ------------------------------------------------------------------
    if mask_ptr != 0:
        # compute batch index from pid_b // hidden_per_batch (actually we need original batch)
        # We rely on the caller to embed the mask as a flat view matching the
        # (B·H, N, N) layout, i.e. the same mask repeated for each hidden
        # channel (so mask_stride_b = N*N, mask_stride_i = N, mask_stride_j = 1)
        mask_offs_b = pid_b * mask_stride_b
        mask_a_ptrs = mask_ptr + mask_offs_b + offs_m[:, None] * mask_stride_i + offs_k[None, :] * mask_stride_j
        mask_b_ptrs = mask_ptr + mask_offs_b + offs_k[:, None] * mask_stride_i + offs_n[None, :] * mask_stride_j
    # ------------------------------------------------------------------

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles (fp16).  Zero‑pad out‑of‑bounds entries.
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )

        # Apply mask if present – the mask is 0/1 (float) and already
        # broadcasted over the hidden dimension.
        if mask_ptr != 0:
            m_a = tl.load(
                mask_a_ptrs,
                mask=mask_m[:, None] & (offs_k < K),
                other=0.0,
            )
            m_b = tl.load(
                mask_b_ptrs,
                mask=(offs_k[:, None] < K) & mask_n[None, :],
                other=0.0,
            )
            a = a * m_a
            b = b * m_b

        # Accumulate: fp16×fp16 → fp32 (uses Tensor Cores on H100)
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk
        if mask_ptr != 0:
            mask_a_ptrs += BLOCK_K * mask_stride_j
            mask_b_ptrs += BLOCK_K * mask_stride_i

    # --------------------------------------------------------------
    # Store the result (fp32)
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Ci
        + offs_n[None, :] * stride_Cn
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights:     dict of model parameters (all on CUDA)
            - config:      dict with keys ['dim', 'hidden_dim', 'nomask']

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # --------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # Linear projections (no bias) – shape [..., hidden_dim]
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # --------------------------------------------------------------
    # Apply mask (if any) and gating *before the heavy matmul.
    # Everything is kept in fp16 for the GEMM.
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → broadcast over hidden_dim
        mask_f = mask.to(dtype=x.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 – this is the dtype used by the Triton kernel.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) and transpose right to (B, hidden, K, J)
    # --------------------------------------------------------------
    # left:  (B, N, N, H) → (B, H, N, N)  rows i, cols k
    left = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) → (B, H, N, N) then transpose last two dims → (B, H, K, J)
    right = right.permute(0, 3, 2, 1).contiguous()

    # Flatten batch & hidden into a single GEMM batch dimension.
    batch_hidden = bs * hidden_dim
    left_flat = left.view(batch_hidden, seq_len, seq_len)
    right_flat = right.view(batch_hidden, seq_len, seq_len)

    # Output buffer – accumulator is fp32 (required for later LayerNorm)
    out_flat = torch.empty(
        (batch_hidden, seq_len, seq_len), dtype=torch.float32, device=device
    )

    # --------------------------------------------------------------
    # Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,      # blocks along M
        (seq_len + BLOCK_N - 1) // BLOCK_N,      # blocks along N
        batch_hidden,                            # batch*hidden slices
    )

    # Strides in element units (not bytes)
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)   # K‑dimension stride
    stride_Bn = right_flat.stride(2)   # N‑dimension stride

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # ------------------------------------------------------------------
    # Optional mask handling inside the kernel.
    # We broadcast the (B,N,N) mask across the hidden dimension by
    # repeating it `hidden_dim` times.  The kernel expects a pointer
    # to the flattened mask with strides:
    #   mask_stride_b = N*N      (jump to next batch)
    #   mask_stride_i = N        (row stride)
    #   mask_stride_j = 1        (col stride)
    # If there is no mask we pass mask_ptr = 0.
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp = mask.to(torch.float16)
        mask_ptr = mask_fp
        mask_stride_b = mask_fp.stride(0)
        mask_stride_i = mask_fp.stride(1)
        mask_stride_j = mask_fp.stride(2)
        mask_int = mask_ptr.data_ptr()
    else:
        # Encode “no mask” as a null pointer (value 0).  Triton treats
        # a pointer value of 0 specially when mask_ptr is a compile‑time
        # constant.
        mask_int = 0
        mask_stride_b = 0
        mask_stride_i = 0
        mask_stride_j = 0

    # ------------------------------------------------------------------
    # Launch the kernel
    # ------------------------------------------------------------------
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,
        seq_len,
        seq_len,
        stride_Ab,
        stride_Ai,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Ci,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        mask_ptr=mask_int,
        mask_stride_b=mask_stride_b,
        mask_stride_i=mask_stride_i,
        mask_stride_j=mask_stride_j,
    )

    # --------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # Final LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply the output gate (element‑wise) and project back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    result = torch.nn.functional.linear(out, weights["to_out.weight"])

    return result
