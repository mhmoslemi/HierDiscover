def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = input_tensor.shape

    # Cast everything to half for compute
    dtype_compute = torch.float16

    # Cast input + layernorm weight+bias
    weight_norm = weights["norm.weight"].to(dtype_compute)
    bias_norm = weights["norm.bias"].to(dtype_compute)
    x = torch.nn.functional.layer_norm(
        input_tensor.to(dtype_compute),
        normalized_shape=(dim,),
        weight=weight_norm,
        bias=bias_norm,
        eps=eps,
    )  # shape [B,N,N,dim] fp16

    # Prepare concatenated weight matrices for left and right
    left_concat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(dtype_compute)  # shape [2H, dim]
    right_concat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(dtype_compute)  # shape [2H, dim]

    # Linear projections + gates combined (no bias)
    left_concat = torch.nn.functional.linear(x, left_concat_weight)  # [B,N,N,2H]
    right_concat = torch.nn.functional.linear(x, right_concat_weight)  # [B,N,N,2H]

    # Split
    left_proj = left_concat[..., :hidden_dim]   # [B,N,N,H]
    left_gate_logit = left_concat[..., hidden_dim:]  # [B,N,N,H]
    right_proj = right_concat[..., :hidden_dim]
    right_gate_logit = right_concat[..., hidden_dim:]

    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)

    # Apply mask if present
    if not config.get("nomask", False) and mask is not None:
        mask_fp = mask.to(dtype_compute).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Now we have left and right in fp16 shape [B,N,N,H]. Fuse them into groups.
    # Rearrange to [G,N,N] where G = B*H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # shape [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # shape [B, H, N, N]

    G = B * hidden_dim
    left_flat = left_perm.view(G, N, N)
    right_flat = right_perm.view(G, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # accumulate in fp32

    # Define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides
    stride_ag = N * N
    stride_am = N
    stride_ak = 1

    stride_bg = N * N
    stride_bk = 1   # because we treat right as transposed
    stride_bn = N

    stride_cg = N * N
    stride_cm = N
    stride_cn = 1

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), G)

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        N=N,
        stride_ag=stride_ag,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bg=stride_bg,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cg=stride_cg,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original shape
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H] float32

    # Compute out_gate
    out_gate_weight = weights["out_gate.weight"].to(dtype_compute)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Second layer norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float32)  # keep FP32 for final
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"].to(dtype_compute)
    output = torch.nn.functional.linear(out, to_out_weight)

    # Cast to float32
    output = output.to(torch.float32)

    return output

"""
Optimized Triton implementation of the *outgoing* TriMul (Triangle‑Multiplicative‑Update)
operator.

Algorithm
---------
1. Layer‑norm the input (fp16 for speed).
2. Fuse the two linear projections (proj + gate) for the *left* and *right* paths
   by concatenating the weight matrices and performing a single matmul per path.
3. Apply the sigmoid gates and (optional) mask – still in fp16.
4. Compute the cubic update  
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]  
   using a custom Triton batched‑GEMM kernel that evaluates A @ Bᵀ for every
   (batch, hidden) pair in parallel.  The kernel loads fp16 data, uses
   `tl.dot` (tensor‑core friendly) and accumulates in fp32.
5. Second LayerNorm (fp32), output gate and final linear projection.
6. Return a float32 tensor of shape [B, N, N, C].

Only the heavy N³ multiplication is performed in Triton; the remaining
operations stay in PyTorch but are heavily fused and run in half‑precision.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) over the (batch * hidden) dimension.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,               # *float16   left  tensor, shape (G, N, N)
    b_ptr,               # *float16   right tensor, shape (G, N, N)
    c_ptr,               # *float32   output tensor, shape (G, N, N)
    N: tl.constexpr,     # sequence length (compile‑time constant)
    stride_ag, stride_am, stride_ak,   # strides for A  (group, row, col)
    stride_bg, stride_bk, stride_bn,   # strides for B  (group, row‑of‑Bᵀ, col‑of‑Bᵀ)
    stride_cg, stride_cm, stride_cn,   # strides for C  (group, row, col)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # tile over output rows  (i)
    pid_j = tl.program_id(1)          # tile over output cols  (j)
    pid_g = tl.program_id(2)          # group index = batch * hidden

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this group
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_g * stride_ag
    b_base = b_ptr + pid_g * stride_bg
    c_base = c_ptr + pid_g * stride_cg

    # Pointers to the start of the current K‑tile (k = 0)
    a_ptrs = a_base + offs_i[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_j[None, :] * stride_bn

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (blocked)
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        # k‑valid mask (handles the last partial tile)
        k_valid = (tl.arange(0, BLOCK_K) + _ * BLOCK_K) < N

        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & k_valid[None, :],
            other=0.0,
        )  # fp16

        b = tl.load(
            b_ptrs,
            mask=k_valid[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # tf32‑compatible: a,b fp16 -> fp32 accumulator via tensor cores
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]   (float32)
        - mask         : torch.Tensor [B, N, N]   (float32 or bool) or None
        - weights      : dict of model weights (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask" …

    Returns
    -------
    torch.Tensor : [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    dtype_compute = torch.float16

    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1) Input layer‑norm (fp16)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(dtype_compute, device=device)
    norm_b = weights["norm.bias"].to(dtype_compute, device=device)
    x = torch.nn.functional.layer_norm(
        input_tensor.to(dtype_compute),
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )                               # [B, N, N, C]  fp16

    # --------------------------------------------------------------
    # 2) fused linear proj + gate for *left* and *right*
    # --------------------------------------------------------------
    # concatenate projection and gate weight matrices (out_features stacked)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype_compute, device=device)          # (2H, C)

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype_compute, device=device)          # (2H, C)

    # one matmul per side
    left_concat = torch.nn.functional.linear(x, left_concat_w)   # [B,N,N,2H] fp16
    right_concat = torch.nn.functional.linear(x, right_concat_w)  # [B,N,N,2H] fp16

    # split projection / gate logits
    left_proj = left_concat[..., :hidden_dim]          # [B,N,N,H] fp16
    left_gate_logit = left_concat[..., hidden_dim:]    # [B,N,N,H] fp16
    right_proj = right_concat[..., :hidden_dim]        # [B,N,N,H] fp16
    right_gate_logit = right_concat[..., hidden_dim:]  # [B,N,N,H] fp16

    left_gate = torch.sigmoid(left_gate_logit)         # fp16
    right_gate = torch.sigmoid(right_gate_logit)       # fp16

    # --------------------------------------------------------------
    # 3) Mask & gate fusion (still in fp16, simple broadcasting)
    # --------------------------------------------------------------
    if (not config.get("nomask", False)) and (mask is not None):
        mask_fp = mask.to(dtype_compute, device=device).unsqueeze(-1)   # [B,N,N,1]
        left = left_proj * left_gate * mask_fp
        right = right_proj * right_gate * mask_fp
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------------
    # 4) Core TriMul: batched matmul A @ Bᵀ over the hidden dimension
    #    left/right : [B, N, N, H]  (fp16)
    #    out        : [B, N, N, H]  (fp32)
    # --------------------------------------------------------------
    # bring hidden dim to axis‑1 and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    G = B * hidden_dim                         # total number of independent matrices
    left_flat = left_perm.view(G, N, N)        # [G, N, N]   fp16
    right_flat = right_perm.view(G, N, N)      # [G, N, N]   fp16

    # output buffer (fp32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides for a contiguous (G,N,N) tensor (row‑major)
    stride_ag = N * N                 # distance (in elements) between groups
    stride_am = N                     # distance between rows (i)
    stride_ak = 1                     # distance between columns (k)

    # for Bᵀ we swap the two inner strides
    stride_bg = N * N                 # group stride
    stride_bk = 1                     # stride along k  (col of original B)
    stride_bn = N                     # stride along n  (row of original B)

    stride_cg = N * N                 # group stride for output
    stride_cm = N                     # row stride
    stride_cn = 1                     # col stride

    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
        G,                            # groups = batch * hidden
    )
    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        N=N,
        stride_ag=stride_ag,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bg=stride_bg,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cg=stride_cg,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H] (fp32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 5) Second LayerNorm + output gate
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(dtype_compute, device=device)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))   # [B,N,N,H] fp16
    out_gate = out_gate.to(torch.float32)                                 # cast for fp32 mult

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float32, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float32, device=device)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                               # fp32
    out = out * out_gate           # fp32

    # --------------------------------------------------------------
    # 6) Final linear projection (fp16 for speed, back to fp32)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(dtype_compute, device=device)
    output = torch.nn.functional.linear(out.to(dtype_compute), to_out_w)  # fp16
    output = output.to(torch.float32)                                    # final fp32

    return output
