import torch
import torch.nn.functional as F
import triton
import triton.ops as ops

def custom_kernel(data):
    inp, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5
    device = inp.device

    # Cast weights to half
    norm_weight = weights['norm.weight'].to(torch.float16)
    norm_bias = weights['norm.bias'].to(torch.float16)
    left_proj_weight = weights['left_proj.weight'].to(torch.float16)
    right_proj_weight = weights['right_proj.weight'].to(torch.float16)
    left_gate_weight = weights['left_gate.weight'].to(torch.float16)
    right_gate_weight = weights['right_gate.weight'].to(torch.float16)
    out_gate_weight = weights['out_gate.weight'].to(torch.float16)
    to_out_norm_weight = weights['to_out_norm.weight'].to(torch.float16)
    to_out_norm_bias = weights['to_out_norm.bias'].to(torch.float16)
    to_out_weight = weights['to_out.weight'].to(torch.float16)

    # LayerNorm on input (float16)
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    ).to(torch.float16)   # Cast after LN

    # Linear projections
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,H]
    right = F.linear(x, right_proj_weight)

    # Apply mask if present
    if mask is not None and not config.get('nomask', False):
        mask = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Reshape for batched GEMM
    B, N, _, H = left.shape

    # prepare A (left) shape (B*H, N, N) as rows: i, cols: k
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # B (right) needs to be (B*H, N, N) where rows: k, cols: j
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Batched matmul with Triton
    out_flat = ops.matmul(left_flat, right_flat)  # Returns half resolution

    # Restore shape
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # still half

    # multiply by out_gate (still half)
    out = out * out_gate

    # final linear to dim
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    # Cast back to float32 for output
    return out.to(torch.float32)
```

"""
TriMul (outgoing) forward pass – highly‑optimized version.

* All heavy tensor algebra is carried out in float16 to maximise H100
  bandwidth and compute utilisation.
* The O(N³) contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is expressed as a batched matrix‑multiply
      out[b,d] = left[b,d] @ right[b,d].T
  and performed with Triton's fused `matmul` kernel (which compiles a
  custom tiled kernel on the fly).  This kernel runs on the combined
  batch‑hidden dimension (B·hidden_dim) and therefore exploits the full
  parallelism of the GPU.
* LayerNorm, Linear and sigmoid gates are executed with PyTorch but in
  float16, keeping the whole forward pass in the fast datatype.
* The final output is cast back to float32 to match the reference
  implementation.

The implementation supports optional masking, arbitrary batch size (1 or 2)
and sequence lengths up to 1024, as required by the test suite.
"""

import torch
import torch.nn.functional as F
import triton.ops as ops


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, C] (float32)
            - mask:        [B, N, N] (bool/float) or None
            - weights: dict with the following keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least:
                "dim" (C), "hidden_dim" (H), "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, C] (float32) – the TriMul forward result.
    """
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dtype = torch.float16  # we work entirely in FP16 until the very end
    eps = 1e-5

    dim = cfg["dim"]                # C
    hidden_dim = cfg["hidden_dim"]  # H

    # -----------------------------------------------------------------
    # cast all model parameters to FP16 once (minimal overhead)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype).to(device)
    norm_bias = weights["norm.bias"].to(dtype).to(device)

    left_proj_w = weights["left_proj.weight"].to(dtype).to(device)
    right_proj_w = weights["right_proj.weight"].to(dtype).to(device)

    left_gate_w = weights["left_gate.weight"].to(dtype).to(device)
    right_gate_w = weights["right_gate.weight"].to(dtype).to(device)
    out_gate_w = weights["out_gate.weight"].to(dtype).to(device)

    to_out_norm_w = weights["to_out_norm.weight"].to(dtype).to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype).to(device)

    to_out_w = weights["to_out.weight"].to(dtype).to(device)

    # -----------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (still in FP32, then cast)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    ).to(dtype)          # → FP16

    # -----------------------------------------------------------------
    # 2️⃣ Linear projections (left / right) – FP16 matmul
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w) # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over the hidden dim)
    # -----------------------------------------------------------------
    if mask is not None and not cfg.get("nomask", False):
        # mask: [B,N,N] → [B,N,N,1] (FP16) for broadcasting
        mask = mask.to(dtype=dtype, device=device).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear transforms)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣ Batched GEMM for the N³ contraction
    #    (out[b,d] = left[b,d] @ right[b,d].T)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                     # B, N, N, H
    # bring the hidden dimension to the second axis so we can fuse batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) → row=i, col=k
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) → row=k, col=j

    # collapse batch and hidden into a single leading dimension
    left_flat = left_perm.view(B * H, N, N)   # (B·H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B·H, N, N)

    # Triton‑powered batched matrix multiply (FP16 inputs → FP16 output)
    out_flat = ops.matmul(left_flat, right_flat)  # (B·H, N, N) in FP16

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 6️⃣ Output LayerNorm + gating + final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                                # still FP16

    out = out * out_gate               # element‑wise gate (FP16)

    out = F.linear(out, to_out_w)      # [B, N, N, C] in FP16

    # -----------------------------------------------------------------
    # 7️⃣ Cast back to FP32 to match the reference implementation
    # -----------------------------------------------------------------
    return out.to(torch.float32)
