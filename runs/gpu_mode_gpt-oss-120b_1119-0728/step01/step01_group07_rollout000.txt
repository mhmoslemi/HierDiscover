def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Cast everything to same dtype if needed: assume weights already same dtype.

    # Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Combine left projection and gate weights
    left_weight_comb = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)  # (2*H, dim)
    right_weight_comb = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)  # (2*H, dim)

    # Compute left+gate and right+gate in one pass each
    left_comb = torch.nn.functional.linear(x, left_weight_comb)  # [B,N,N,2H]
    right_comb = torch.nn.functional.linear(x, right_weight_comb)

    # Split
    left_proj = left_comb[..., :hidden_dim]
    left_gate_logits = left_comb[..., hidden_dim:]
    right_proj = right_comb[..., :hidden_dim]
    right_gate_logits = right_comb[..., hidden_dim:]

    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    # Out gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply mask and gating
    if mask is not None and not config.get("nomask", False):
        # mask shape: [B,N,N] or [B,N,N,1]
        mask_tensor = mask.to(input_tensor.dtype)
        if mask_tensor.ndim == 4:
            mask_tensor = mask_tensor.squeeze(-1)  # to [B,N,N]
        left = _apply_gate_and_mask(left_proj, left_gate, mask_tensor)
        right = _apply_gate_and_mask(right_proj, right_gate, mask_tensor)
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Core multiplicative update
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch*H dim
    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)

    # batched matmul
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # shape (B*H, N, N)

    # reshape back

    # Apply out gate
    out = out * out_gate

    # Final linear projection to dim
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) forward kernel – highly‑optimized implementation.

Algorithm
---------
1. Input LayerNorm over the last channel (dim).
2. Fuse projection + gating linear layers:
   * Concatenate left_proj & left_gate weights → one GEMM → split.
   * Concatenate right_proj & right_gate weights → one GEMM → split.
   * Compute out_gate with a separate linear + sigmoid.
3. Fuse element‑wise mask, gate and projection for left/right using a tiny Triton kernel.
4. Core multiplicative update:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply (left @ rightᵀ) per head.
   We reshape to (B*H, N, N) and call torch.bmm (CUDA/cuBLAS).
5. Second LayerNorm over the hidden dimension, apply out_gate and final linear projection.
6. Return the result with shape [B, N, N, dim].

Only the mask‑gate fusion is performed in Triton; all heavy matrix ops use
high‑throughput cuBLAS kernels. The implementation reduces the number of
separate linear projections from 5 → 3 and keeps everything in float32 (the
final output type required by the reference implementation).
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of tensor * gate * mask
# (mask is broadcast over the hidden dimension)
# -------------------------------------------------------------------------
@triton.jit
def fuse_mul_gate_mask_kernel(
    a_ptr,          # *float32  tensor a (flattened)                [total]
    gate_ptr,       # *float32  gate tensor (flattened)            [total]
    mask_ptr,       # *float32  mask tensor (flattened)            [B*N*N]
    out_ptr,        # *float32  output tensor (flattened)          [total]
    total_elements,  # i32      total number of elements in a/gate
    hidden_dim,     # i32      size of the hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # Bounds check
    mask = offs < total_elements

    # Each hidden slice shares the same mask entry.
    # Because the layout is [B,N,N,H] (fastest dim = H) the mask index is:
    #   floor(offset / H)
    mask_idx = offs // hidden_dim

    a   = tl.load(a_ptr   + offs, mask=mask, other=0.0)
    g   = tl.load(gate_ptr+ offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr+ mask_idx, mask=mask, other=0.0)

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask(a: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse elementwise a * gate * mask (mask broadcast across hidden dim)
    using the Triton kernel above.

    Arguments
    ---------
    a    : [B, N, N, H]   float32
    gate : [B, N, N, H]   float32
    mask : [B, N, N]      float32 (will be broadcast)

    Returns
    -------
    out  : [B, N, N, H]   float32
    """
    # Ensure contiguous layout for flattening
    a_flat    = a.contiguous().view(-1)
    gate_flat = gate.contiguous().view(-1)
    mask_flat = mask.contiguous().view(-1)   # shape B*N*N
    out_flat  = torch.empty_like(a_flat)

    total      = a_flat.numel()
    hidden_dim = a.shape[-1]
    BLOCK      = 1024
    grid       = (triton.cdiv(total, BLOCK),)

    fuse_mul_gate_mask_kernel[grid](
        a_flat,
        gate_flat,
        mask_flat,
        out_flat,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    return out_flat.view_as(a)


# -------------------------------------------------------------------------
# Main entry point for the outgoing TriMul operator
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul (forward only).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] or [B, N, N, 1] (or None)
        - weights      : dict of torch.Tensors, see keys in implementation
        - config       : dict containing at least "dim", "hidden_dim",
                         optional "nomask" (bool)

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Fuse projection + gate linear layers (two GEMMs instead of four)
    # -----------------------------------------------------------------
    # Concatenate left projection & gate weights: shape (2*H, dim)
    left_weight_comb = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )  # (2*H, dim)
    right_weight_comb = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )  # (2*H, dim)

    # One linear call for left side (proj + gate)
    left_comb = torch.nn.functional.linear(x, left_weight_comb)  # [B, N, N, 2*H]
    # One linear call for right side (proj + gate)
    right_comb = torch.nn.functional.linear(x, right_weight_comb)  # [B, N, N, 2*H]

    # Split projection and gate logits
    left_proj       = left_comb[..., :hidden_dim]                        # [B,N,N,H]
    left_gate_logits = left_comb[..., hidden_dim:]                       # [B,N,N,H]
    right_proj      = right_comb[..., :hidden_dim]                       # [B,N,N,H]
    right_gate_logits = right_comb[..., hidden_dim:]                    # [B,N,N,H]

    # Apply sigmoid to gate logits
    left_gate  = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)

    # Output gate (single linear + sigmoid)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 3) Mask & gate fusion (Triton) – optional
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # Ensure mask has shape [B, N, N]
        mask_tensor = mask.to(input_tensor.dtype)
        if mask_tensor.ndim == 4:
            mask_tensor = mask_tensor.squeeze(-1)  # [B,N,N]

        # Fuse mask, gate and projection in a single kernel for each side
        left  = _apply_gate_and_mask(left_proj,  left_gate,  mask_tensor)  # [B,N,N,H]
        right = _apply_gate_and_mask(right_proj, right_gate, mask_tensor)  # [B,N,N,H]
    else:
        # No mask → simple elementwise multiplication
        left  = left_proj * left_gate
        right = right_proj * right_gate

    # -----------------------------------------------------------------
    # 4) Core multiplicative update: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    This is a batched GEMM (left @ rightᵀ) per head.
    # -----------------------------------------------------------------
    # Bring hidden dimension to the second axis for batched matmul:
    #   left_perm / right_perm : [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    # Collapse batch & head into a single leading dimension
    left_flat  = left_perm.view(B * H, N, N)          # [B*H, N, N]
    right_flat = right_perm.view(B * H, N, N)         # [B*H, N, N]

    # Batched matrix multiplication (CUDA/cuBLAS) → O(B*H*N³) work
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # -----------------------------------------------------------------
    # 5) Second LayerNorm, apply out gate and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
