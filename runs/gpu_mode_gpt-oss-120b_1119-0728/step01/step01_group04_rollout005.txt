def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # linear projections
    left_proj = F.linear(x, weights["left_proj.weight"])   # no bias
    right_proj = F.linear(x, weights["right_proj.weight"])

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # mask
    if mask is None or config.get("nomask", False):
        mask_f = torch.ones((batch_size, seq_len, seq_len), dtype=x.dtype, device=device)
    else:
        mask_f = mask.to(x.dtype)
    mask_f = mask_f.unsqueeze(-1)  # (B,N,N,1)

    # mask + gating
    left = left_proj * left_gate * mask_f
    right = right_proj * right_gate * mask_f

    # cast to half for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # allocate output tensor in half, shape [B,N,N,hidden_dim]
    out_h = torch.empty((batch_size, seq_len, seq_len, hidden_dim), dtype=torch.float16, device=device)

    # compute strides for kernel
    stride_Ab, stride_Ai, stride_Ak, stride_Ah = left_h.stride()
    stride_Bb, stride_Bi, stride_Bk, stride_Bh = right_h.stride()
    stride_Cb, stride_Ci, stride_Cj, stride_Ch = out_h.stride()

    # block sizes (choose good values)
    BLOCK_M = 64  # along i dimension
    BLOCK_N = 64  # along j dimension
    BLOCK_K = 64

    grid = (batch_size * hidden_dim,
            triton.cdiv(seq_len, BLOCK_M),
            triton.cdiv(seq_len, BLOCK_N))

    outgoing_tri_mul_kernel[grid](
        left_h,
        right_h,
        out_h,
        batch_size,
        seq_len,
        hidden_dim,
        stride_Ab,
        stride_Ai,
        stride_Ak,
        stride_Bb,
        stride_Bi,
        stride_Bk,
        stride_Cb,
        stride_Ci,
        stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # after kernel, cast back to float32 for downstream ops
    out = out_h.to(torch.float32)

    # to_out_norm layernorm
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    # apply out_gate
    out = out * out_gate

    # final linear projection
    out = F.linear(out, weights["to_out.weight"])

    return out

""" 
Custom Triton kernel for the outgoing TriMul operation.

The kernel computes, for each batch b and hidden channel h,
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
where left and right are the gated, masked linear projections of the input.
The computation is performed as a batched half‑precision matrix multiplication
with an FP32 accumulator. All surrounding operations (layernorms, linear
projections, gating) are executed with PyTorch. The kernel works directly on
the original [B, N, N, H] layout, avoiding costly transposes or copies.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def outgoing_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,    # pointers to left, right, output (float16)
    N, H,                    # sequence length, hidden dimension
    stride_Ab, stride_Ai, stride_Ak,  # left strides (batch, i, k)
    stride_Bb, stride_Bi, stride_Bk,  # right strides (batch, j, k)
    stride_Cb, stride_Ci, stride_Cj,  # output strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication for TriMul:
    For each (batch, hidden) pair compute C = A @ Bᵀ where
    A : [M, K] = left[batch, :, :, hidden]
    B : [N, K] = right[batch, :, :, hidden]
    Output C : [M, N]
    """
    # program ids
    batch_h = tl.program_id(0)      # combined batch * hidden index
    pid_m   = tl.program_id(1)      # block row (i)
    pid_n   = tl.program_id(2)      # block column (j)

    # recover batch and hidden index
    batch = batch_h // H
    h     = batch_h % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (batch, hidden) pair
    a_base = a_ptr + batch * stride_Ab + h
    b_base = b_ptr + batch * stride_Bb + h
    c_base = c_ptr + batch * stride_Cb + h

    # pointers that will be advanced in the K loop
    a_ptrs = a_base + offs_m[:, None] * stride_Ai + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bi

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)           # (BLOCK_M, BLOCK_K) half
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)           # (BLOCK_K, BLOCK_N) half

        # promote to FP32 and accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # store result (cast back to FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask : torch.Tensor of shape [B, N, N] or None
        - weights : dict of model parameters (torch.Tensors)
        - config : dict containing at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating pre‑activations + sigmoid
    # ------------------------------------------------------------------
    left_gate_pre  = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate_pre   = F.linear(x, weights["out_gate.weight"])

    left_gate  = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate   = out_gate_pre.sigmoid()          # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((batch_size, seq_len, seq_len),
                                 dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)     # [B, N, N, 1]

    # ------------------------------------------------------------------
    # 5. Apply mask and gating
    # ------------------------------------------------------------------
    left  = left_proj * left_gate * mask_tensor   # [B, N, N, hidden_dim]
    right = right_proj * right_gate * mask_tensor

    # ------------------------------------------------------------------
    # 6. Cast to half‑precision for the cubic term
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Allocate output tensor (half) in the original layout to avoid
    #    costly transposes.
    # ------------------------------------------------------------------
    out_h = torch.empty((batch_size, seq_len, seq_len, hidden_dim),
                        dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 8. Extract strides for the Triton kernel
    # ------------------------------------------------------------------
    stride_Ab, stride_Ai, stride_Ak, stride_Ah = left_h.stride()
    stride_Bb, stride_Bi, stride_Bk, stride_Bh = right_h.stride()
    stride_Cb, stride_Ci, stride_Cj, stride_Ch = out_h.stride()

    # ------------------------------------------------------------------
    # 9. Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_size * hidden_dim,                     # (batch * hidden)
        triton.cdiv(seq_len, BLOCK_M),               # M tiles
        triton.cdiv(seq_len, BLOCK_N),               # N tiles
    )

    outgoing_tri_mul_kernel[grid](
        left_h,               # a_ptr
        right_h,              # b_ptr
        out_h,                # c_ptr
        seq_len,              # N
        hidden_dim,           # H
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bi, stride_Bk,
        stride_Cb, stride_Ci, stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Convert back to FP32 for the remaining linear layers
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 11. Output LayerNorm (hidden_dim) and output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""Outgoing TriMul forward pass accelerated with Triton.

The core operation is the 4‑D einsum

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be expressed as a batched matrix multiplication
C = A @ Bᵀ for each hidden channel h.  The kernel treats every
(batch, hidden) pair as an independent batch and computes the
multiplication in half‑precision with an FP32 accumulator.
All surrounding LayerNorm, Linear and gating operations are performed
with PyTorch, while the cubic N³ term is fused into a single Triton
kernel that works directly on the original [B, N, N, H] layout,
avoiding costly transposes or copies.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def outgoing_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers to left, right, output (float16)
    N, H,                         # sequence length, hidden dimension
    stride_Ab, stride_Ai, stride_Ak,   # left strides (batch, i, k)
    stride_Bb, stride_Bi, stride_Bk,   # right strides (batch, j, k)
    stride_Cb, stride_Ci, stride_Cj,   # output strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul for TriMul:
    for each (batch, hidden) pair compute
        C = A @ Bᵀ   where
        A = left[batch, :, :, hidden]    (M×K)
        B = right[batch, :, :, hidden]   (N×K)
    """
    # --------------------------------------------------------------
    # Program IDs
    # --------------------------------------------------------------
    batch_h = tl.program_id(0)       # combined (batch * hidden) index
    pid_m   = tl.program_id(1)       # block row   (i)
    pid_n   = tl.program_id(2)       # block col   (j)

    # --------------------------------------------------------------
    # Recover original batch and hidden indices
    # --------------------------------------------------------------
    batch = batch_h // H
    h     = batch_h % H

    # --------------------------------------------------------------
    # Tile offsets within the M×N output matrix
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # --------------------------------------------------------------
    a_base = a_ptr + batch * stride_Ab + h          # left
    b_base = b_ptr + batch * stride_Bb + h          # right
    c_base = c_ptr + batch * stride_Cb + h          # output

    # --------------------------------------------------------------
    # Pointers for the current K‑tile
    # --------------------------------------------------------------
    a_ptrs = a_base + offs_m[:, None] * stride_Ai + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Bi

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # (BLOCK_M, BLOCK_K) half
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # (BLOCK_K, BLOCK_N) half

        # Promote to FP32 and accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        # Advance tile pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # --------------------------------------------------------------
    # Store the result (cast back to half)
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (torch.Tensors)
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating pre‑activations + sigmoid
    # ------------------------------------------------------------------
    left_gate_pre  = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate_pre   = F.linear(x, weights["out_gate.weight"])

    left_gate  = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate   = out_gate_pre.sigmoid()          # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((batch_size, seq_len, seq_len),
                                 dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)      # [B, N, N, 1]

    # ------------------------------------------------------------------
    # 5. Apply mask and gating
    # ------------------------------------------------------------------
    left  = left_proj * left_gate * mask_tensor   # [B, N, N, hidden_dim]
    right = right_proj * right_gate * mask_tensor

    # ------------------------------------------------------------------
    # 6. Cast to half‑precision for the cubic term
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Allocate output tensor (half) in the original layout
    # ------------------------------------------------------------------
    out_h = torch.empty((batch_size, seq_len, seq_len, hidden_dim),
                        dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 8. Strides for the Triton kernel
    # ------------------------------------------------------------------
    stride_Ab, stride_Ai, stride_Ak, stride_Ah = left_h.stride()
    stride_Bb, stride_Bi, stride_Bk, stride_Bh = right_h.stride()
    stride_Cb, stride_Ci, stride_Cj, stride_Ch = out_h.stride()

    # ------------------------------------------------------------------
    # 9. Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_size * hidden_dim,                # combined (batch * hidden)
        triton.cdiv(seq_len, BLOCK_M),          # tiles along i
        triton.cdiv(seq_len, BLOCK_N),          # tiles along j
    )

    outgoing_tri_mul_kernel[grid](
        left_h,               # a_ptr
        right_h,              # b_ptr
        out_h,                # c_ptr
        seq_len,              # N
        hidden_dim,           # H
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bi, stride_Bk,
        stride_Cb, stride_Ci, stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Cast back to FP32 for the remaining linear layers
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 11. Output LayerNorm (over hidden_dim) and output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12. Final linear projection back to the original dimension `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
