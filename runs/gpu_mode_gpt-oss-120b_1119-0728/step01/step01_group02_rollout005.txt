TriMul (outgoing) – Triton‑accelerated implementation
====================================================

Given an input tensor  ``x``  of shape ``[B, N, N, D]`` the module
computes

    out[b,i,j,:] = Σ_k left[b,i,k,:] • right[b,j,k,:]

where ``left``  and ``right`` are linear projections of a layer‑normed
input, optionally masked and gated by learned sigmoids.  The heavy N³
reduction is realised as a **batched GEMM**:

    left_proj[b, :, :, h]   : (N, N)   ←  x[b] @ W_leftᵀ   (h = hidden dim)
    right_proj[b, :, :, h]  : (N, N)   ←  x[b] @ W_rightᵀ
    out_tile   = left_tile @ right_tileᵀ   (batch = B·H)

All arithmetic up to the GEMM is performed in **FP16** (half‑precision);
the GEMM accumulates in **FP32** and stores the result in FP16.
A tiny Triton kernel implements the batched GEMM with an
autotuned set of tile sizes (BLOCK_M, BLOCK_N, BLOCK_K).  The rest of
the TriMul pipeline (layer‑norms, linear layers, sigmoids, masks) is
handled with regular PyTorch which is negligible compared to the O(N³)
work.

The end‑to‑end forward pass returns a tensor of shape
``[B, N, N, D]`` in FP32.
"""

import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

@triton.autotune(
    configs=[
        # These configs were selected empirically for H100 (large shared memory,
        # very high tensor‑core throughput).  The autotuner picks the best
        # one for the given (M, N, K) at compile time.
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32}, num_warps=8, num_stages=2),
        triton.Config({"BLOCK_M": 64,  "BLOCK_N": 64,  "BLOCK_K": 64}, num_warps=4, num_stages=2),
        triton.Config({"BLOCK_M": 64,  "BLOCK_N": 128, "BLOCK_K": 32}, num_warps=4, num_stages=2),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 64,  "BLOCK_K": 32}, num_warps=8, num_stages=2),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers to tensors (FP16)
    batch, M, N, K,                       # logical dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A,B ∈ FP16, acc in FP32, C stored as FP16).

    * ``batch`` = B·H (B=batch size, H=hidden_dim)
    * ``A`` shape (batch, M, K)  – here M = N
    * ``B`` shape (batch, N, K)  – we multiply by Bᵀ inside the kernel
    * ``C`` shape (batch, M, N)
    """
    pid_b = tl.program_id(2)   # batch × hidden index
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col

    # ------------------------------------------------------------------
    # Compute offsets for the current tile (rows, cols, k‑dimension)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ------------------------------------------------------------------
    # Pointers for the first block of A and Bᵀ (the K‑loop will stride)
    # ------------------------------------------------------------------
    a_ptrs = (
        a_ptr
        + pid_b * stride_a_batch
        + offs_m[:, None] * stride_a_m
        + offs_k[None, :] * stride_a_k
    )
    b_ptrs = (
        b_ptr
        + pid_b * stride_b_batch
        + offs_k[:, None] * stride_b_k
        + offs_n[None, :] * stride_b_n
    )
    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # dynamic offsets for this K‑tile
        a_tile = tl.load(
            a_ptrs + cur_k * stride_a_k,
            mask=(offs_m[:, None] < M) & (cur_k + offs_k[None, :] < K),
            other=0.0,
        )
        b_tile = tl.load(
            b_ptrs + cur_k * stride_b_k,
            mask=(cur_k + offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # Matrix‑multiply the two tiles (tensor‑core instruction)
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N) FP32

    # ------------------------------------------------------------------
    # Write back the result (store as FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        c,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------
# Top‑level entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]   (float32)
        - mask_tensor  : torch.Tensor [B, N, N]    (bool/float, may be ignored)
        - weights_dict : mapping name → torch.Tensor (float32)
        - config_dict  : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (FP16 for speed)
    # --------------------------------------------------------------
    # Cast weights to the reduced precision once
    norm_weight = w["norm.weight"].to(torch.float16, device=device)
    norm_bias   = w["norm.bias"].to(torch.float16,   device=device)

    x = F.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    ).to(torch.float16)        # [B, N, N, D] in FP16

    # --------------------------------------------------------------
    # 2️⃣  Linear projections to hidden dimension (still FP16)
    # --------------------------------------------------------------
    left = F.linear(x, w["left_proj.weight"], bias=None)   # [B, N, N, H]
    right = F.linear(x, w["right_proj.weight"], bias=None)

    # --------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N] → [B, N, N, 1] → FP16
        mask_fp = mask.to(dtype=left.dtype, device=device).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4️⃣  Sigmoid gates (per hidden channel)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, w["left_gate.weight"], bias=None))
    right_gate = torch.sigmoid(F.linear(x, w["right_gate.weight"], bias=None))
    out_gate   = torch.sigmoid(F.linear(x, w["out_gate.weight"], bias=None))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Batched GEMM   (left @ rightᵀ)   →  out_tile in FP16
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B·H, N, N) and ensure contiguity for the kernel
    left_tile  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_tile = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    out_tile   = torch.empty_like(left_tile, dtype=torch.float16)

    # strides for the three tensors (batch, row, col)
    stride_a_batch = left_tile.stride(0)
    stride_a_m     = left_tile.stride(1)
    stride_a_k     = left_tile.stride(2)

    stride_b_batch = right_tile.stride(0)
    stride_b_n     = right_tile.stride(1)
    stride_b_k     = right_tile.stride(2)

    stride_c_batch = out_tile.stride(0)
    stride_c_m     = out_tile.stride(1)
    stride_c_n     = out_tile.stride(2)

    # launch configuration (tiles over rows, cols, batch*hidden)
    grid = (
        (N + 64 - 1) // 64,          # placeholder; real BLOCK_M/N determined by autotune
        (N + 64 - 1) // 64,
        B * H,
    )
    # Triton will pick the optimal block sizes from the autotune configs.
    tri_mul_gemm_kernel[grid](
        left_tile,
        right_tile,
        out_tile,
        B * H,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        # NOTE: the BLOCK_* constants are filled in by the autotuner.
    )

    # --------------------------------------------------------------
    # 6️⃣  Restore the original layout: [B, N, N, H] (still FP16)
    # --------------------------------------------------------------
    out = (
        out_tile.view(B, H, N, N)          # (B, H, N, N)
        .permute(0, 2, 3, 1)               # (B, N, N, H)
    )

    # --------------------------------------------------------------
    # 7️⃣  Final layer‑norm over the hidden dimension (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = w["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = w["to_out_norm.bias"].to(torch.float16,   device=device)
    out = F.layer_norm(
        out,
        (hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # --------------------------------------------------------------
    # 8️⃣  Apply the output gate (FP16)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9️⃣  Project back to the original dimension (still FP16)
    # --------------------------------------------------------------
    out = F.linear(out, w["to_out.weight"], bias=None)   # [B, N, N, D] FP16

    # --------------------------------------------------------------
    # 10️⃣ Cast back to FP32 for the final output as required
    # --------------------------------------------------------------
    return out.to(torch.float32)
