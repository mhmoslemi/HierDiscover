def fuse_mul_gate_mask_half_kernel(
    a_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE, dtype=tl.int64)
    mask = offs < total_elements

    mask_idx = offs // hidden_dim

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=mask)

Return output.

def custom_kernel(data):
    """
    Outgoing TriMul (forward only) implemented with mixed‑precision.
    Steps:
    1. FP32 LayerNorm on input.
    2. Cast to FP16 for all subsequent linear / gating ops.
    3. Compute left/right projections and gates (FP16).
    4. Fuse mask, gate and projection with a Triton elementwise kernel.
    5. Batched FP16 GEMM per hidden channel (torch.matmul uses tensor cores).
    6. FP16 LayerNorm, output gate and final linear projection.
    7. Cast result back to FP32.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1. FP32 layernorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # float32

    # 2. Cast to FP16 for all heavy ops
    x_half = x.half()

    # Cast weights to FP16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # 3. Linear projections (FP16)
    left = torch.nn.functional.linear(x_half, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_half, right_proj_w)

    # 4. Gating (FP16)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_half, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_half, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_half, out_gate_w))

    # 5. Apply mask + gate fusion (or simple multiplication)
    if config.get("nomask", False) or mask is None:
        left = left * left_gate
        right = right * right_gate
    else:
        mask_half = mask.to(torch.float16)
        if mask_half.dim() == 4:
            mask_half = mask_half.squeeze(-1)
        left = _apply_gate_and_mask_half(left, left_gate, mask_half)
        right = _apply_gate_and_mask_half(right, right_gate, mask_half)

    # 6. Core multiplicative update – batched GEMM per hidden channel
    # left: [B,N,N,H] -> [B,H,N,N]
    left_perm = left.permute(0, 3, 1, 2)   # B, H, N, N

"""
Outgoing TriMul (forward only) – mixed‑precision Triton implementation.

Algorithm
---------
1. FP32 LayerNorm on the input tensor.
2. Cast the normalized tensor to FP16 (bfloat16 could also be used) – the
   remaining heavy work (linear projections, gating, the O(N³) multiplicative
   update and the final projection) is performed in FP16 and thus benefits
   from Tensor‑Core acceleration on H100.
3. Linear projections (left / right) and three gating linear layers are
   computed with torch.nn.functional.linear (FP16 weights).
4. When a mask is present we fuse the element‑wise multiplication
   `proj * gate * mask` with a small Triton kernel that streams the three
   tensors and writes the result back.  The mask is broadcast over the hidden
   dimension inside the kernel.
5. Core multiplicative update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batched matrix multiplication per hidden channel.
   Using torch.matmul on FP16 tensors triggers Tensor‑Core GEMM kernels.
6. Second LayerNorm (FP16), output‑gate multiplication and final linear
   projection (FP16).  The final tensor is cast back to FP32 before returning.
   The implementation works for the full range of test configurations
   (varying batch, sequence length, dimensions and optional mask).
"""

import torch
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: fuse projection * gate * mask (FP16)
# ---------------------------------------------------------------------------
@triton.jit
def fuse_mul_gate_mask_half_kernel(
    a_ptr,          # *float16  – projection tensor (flattened)
    g_ptr,          # *float16  – gate tensor      (flattened)
    m_ptr,          # *float16  – mask tensor      (flattened, size B*N*N)
    out_ptr,        # *float16  – output tensor    (flattened)
    total_elements,  # i32      – total number of elements in a/g
    hidden_dim,      # i32      – hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE, dtype=tl.int64)
    mask = offs < total_elements

    # Index of the mask entry for each hidden slice.
    #   flat_index = ((b*N + i)*N + j) * H + h
    #   mask_index = flat_index // H
    mask_idx = offs // tl.int64(hidden_dim)

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    g = tl.load(g_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(m_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=mask)


def _apply_gate_and_mask_half(proj: torch.Tensor,
                              gate: torch.Tensor,
                              mask: torch.Tensor,
                              block_size: int = 1024) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` (mask broadcast over the hidden dim) using Triton.
    All tensors must be FP16 and contiguous.
    """
    proj_f = proj.contiguous().view(-1)
    gate_f = gate.contiguous().view(-1)
    mask_f = mask.contiguous().view(-1)          # shape B*N*N, FP16

    out_f = torch.empty_like(proj_f)

    total = proj_f.numel()
    hidden_dim = proj.shape[-1]

    grid = (triton.cdiv(total, block_size),)

    fuse_mul_gate_mask_half_kernel[grid](
        proj_f,
        gate_f,
        mask_f,
        out_f,
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
    )
    return out_f.view_as(proj)

    Output:

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    x = torch.nn.functional.layer_norm(

        normalized_shape=(dim,),

        eps=eps,
    )  # [B, N, N, dim] float32

    # --------------------------------------------------------------------
    # 2) Cast to FP16 for all heavy ops
    # --------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] float16

    # 3) Cast all weight tensors to FP16 once
    # --------------------------------------------------------------------
    left_proj_w    = weights["left_proj.weight"].to(torch.float16)
    right_proj_w   = weights["right_proj.weight"].to(torch.float16)
    left_gate_w    = weights["left_gate.weight"].to(torch.float16)
    right_gate_w   = weights["right_gate.weight"].to(torch.float16)
    out_gate_w     = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w  = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b  = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w       = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # --------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_h, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_h, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------------
    # 5) Gating (FP16)
    # --------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # --------------------------------------------------------------------
    # 6) Apply mask + gate (fuse with Triton if mask present)
    # --------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        left  = left * left_gate
        right = right * right_gate
    else:
        # mask: [B, N, N] (float32/float16)
        mask_h = mask.to(torch.float16)
        if mask_h.dim() == 4:          # possibly [B, N, N, 1]
            mask_h = mask_h.squeeze(-1)
        left  = _apply_gate_and_mask_half(left, left_gate, mask_h)
        right = _apply_gate_and_mask_half(right, right_gate, mask_h)

    # --------------------------------------------------------------------
    # 7) Core multiplicative update (batched GEMM per hidden channel)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # --------------------------------------------------------------------
    # reshape to [B, H, N, N] for a single batched matmul
    left_perm  = left.permute(0, 3, 1, 2)   # [B, H, N, N]  FP16
    right_perm = right.permute(0, 3, 1, 2)  # [B, H, N, N]  FP16

    # batched matmul: (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, H, N, N] FP16
    out = out.permute(0, 2, 3, 1)                               # [B, N, N, H] FP16

    # --------------------------------------------------------------------
    # 8) Second LayerNorm (FP16)
    # --------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # --------------------------------------------------------------------
    # 9) Output gate and final linear projection (FP16)
    # --------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] FP16

    # --------------------------------------------------------------------
    # 10) Cast back to FP32
    # --------------------------------------------------------------------
    return out.float()
