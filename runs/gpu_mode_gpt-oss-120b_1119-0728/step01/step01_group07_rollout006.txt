```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    bid = pid // (tl.cdiv(N, BLOCK_N) * tl.cdiv(M, BLOCK_M))
    pid_mn = pid % (tl.cdiv(N, BLOCK_N) * tl.cdiv(M, BLOCK_M))
    pid_m = pid_mn // tl.cdiv(N, BLOCK_N)
    pid_n = pid_mn % tl.cdiv(N, BLOCK_N)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for OOB
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + batch_stride_a * bid + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_stride_b * bid + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + batch_stride_c * bid + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None,:])
```

    a_ptr, b_ptr, c_ptr,
    stride_ab, stride_ah, stride_am, stride_ak,
    stride_bb, stride_bh, stride_bk, stride_bn,
    stride_cb, stride_ch, stride_cm, stride_cn,
    B, H, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)   # combined batch hidden id
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # derive b and h
    b_idx = pid // H
    h_idx = pid % H

    # offsets within dimension i, j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # calculate offsets for a and b base pointers
    a_base = a_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = b_ptr + b_idx * stride_bb + h_idx * stride_bh
    c_base = c_ptr + b_idx * stride_cb + h_idx * stride_ch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):

        # compute pointer for A block
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # dot product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

"""
tri_mul_outgoing.py

Highly‑optimized Triton implementation of the “outgoing” TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1.  Layer‑norm the 4‑D input tensor (B, N, N, C).
2.  Linear projections (left/right) and three gating linear layers (no bias).
3.  Fuse element‑wise multiplication of the projections with their
    respective gates and the optional mask using a tiny Triton kernel.
4.  Core multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched GEMM (B·H independent N×N products) implemented
    in Triton by tiling the M, N and K dimensions.  The kernel works on
    the tensors in shape (B, H, N, N) without any extra copies.
5.  Second LayerNorm, apply the output gate and a final linear projection
    back to the original channel dimension.

All heavy work (step 3 and step 4) runs on the GPU in custom Triton
kernels, while the cheap LayerNorm / Linear calls stay in PyTorch.
The implementation supports both masked and un‑masked inputs and works
for the full range of test configurations (B∈{1,2}, N∈{128…1024},
C∈{128}, H∈{128,384,768}).

The final output is a float‑32 tensor of shape [B, N, N, C].
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ------------------------------------------------------------------------
#  Kernel 1 : fuse element‑wise multiplication (proj * gate * mask)
# ------------------------------------------------------------------------
@triton.jit
def fuse_mul_gate_mask_kernel(
    a_ptr,      # *float32  projection tensor (flattened)
    gate_ptr,   # *float32  gate tensor (flattened)
    mask_ptr,   # *float32  mask tensor (flattened, size B*N*N)
    out_ptr,    # *float32  output tensor (flattened)
    total,      # i32       total number of elements in a / gate
    hidden_dim, # i32       hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # mask index = element index // hidden_dim  (broadcast mask over H)
    mask_idx = offs // hidden_dim

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    tl.store(out_ptr + offs, a * g * m, mask=mask)


def _apply_gate_and_mask(proj: torch.Tensor,
                         gate: torch.Tensor,
                         mask: torch.Tensor) -> torch.Tensor:
    """
    Fuse `proj * gate * mask` where ``mask`` has shape [B, N, N] and
    is broadcast over the hidden dimension.  All tensors are expected
    to be contiguous and float32.
    """
    proj_f = proj.contiguous().view(-1)
    gate_f = gate.contiguous().view(-1)
    mask_f = mask.contiguous().view(-1)               # [B,N,N]
    out_f = torch.empty_like(proj_f)

    total = proj_f.numel()
    hidden = proj.shape[-1]
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    fuse_mul_gate_mask_kernel[grid](
        proj_f,
        gate_f,
        mask_f,
        out_f,
        total,
        hidden,
        BLOCK_SIZE=BLOCK,
    )
    return out_f.view_as(proj)


# ------------------------------------------------------------------------
#  Kernel 2 : batched matrix multiplication for the core TriMul update
#             out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ------------------------------------------------------------------------
@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                      # *float32 pointers
    stride_ab, stride_ah, stride_am, stride_ak,
    stride_bb, stride_bh, stride_bk, stride_bn,
    stride_cb, stride_ch, stride_cm, stride_cn,
    B, H,                                      # runtime batch / hidden sizes
    N: tl.constexpr,                           # sequence length (also K)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid = tl.program_id(0)          # collapsed (b, h)
    pid_m = tl.program_id(1)        # tile over rows (i)
    pid_n = tl.program_id(2)        # tile over cols (j)

    # derive concrete batch / hidden indices
    b_idx = pid // H
    h_idx = pid % H

    # offsets for the current output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this (b,h) pair
    a_base = a_ptr + b_idx * stride_ab + h_idx * stride_ah
    b_base = b_ptr + b_idx * stride_bb + h_idx * stride_bh
    c_base = c_ptr + b_idx * stride_cb + h_idx * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K‑dimension (size N)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointer arithmetic for A and B tiles
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # guarded loads (avoid OOB reads)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < N) & (offs_k[None, :] < N),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < N) & (offs_n[None, :] < N),
                    other=0.0)

        acc += tl.dot(a, b)

    # write the result tile
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_store)

#  Main entry point

    """

    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N]  (or None)
        - weights      : dict of model weights (see code)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` flag.

    Returns
    -------
    output : torch.Tensor [B, N, N, C]
    """
    # --------------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------
    # 2) Linear projections (no bias) + gating logits
    # --------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------------
    # 3) Fuse mask and gates (if a mask is present)
    # --------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        # Simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
    else:
        # mask is [B, N, N] – broadcast over hidden_dim
        mask_f = mask.to(input_tensor.dtype)
        if mask_f.dim() == 4:
            mask_f = mask_f.squeeze(-1)       # -> [B, N, N]
        left = _apply_gate_and_mask(left, left_gate, mask_f)
        right = _apply_gate_and_mask(right, right_gate, mask_f)

    # --------------------------------------------------------------------
    # 4) Core multiplicative update – batched GEMM per hidden channel
    # --------------------------------------------------------------------
    # (B,N,N,hidden) --> (B,hidden,N,N)
    left_perm = left.permute(0, 3, 1, 2)       # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1)     # transposed on the last two dims
    B, H, N, _ = left_perm.shape

    # allocate output buffer in the same layout as left_perm
    out_perm = torch.empty_like(left_perm)

    # tile sizes – tuned for H100 (fit in shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # collapsed batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_outgoing_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        # strides for A (left)
        left_perm.stride(0), left_perm.stride(1),
        left_perm.stride(2), left_perm.stride(3),
        # strides for B (right, already transposed)
        right_perm.stride(0), right_perm.stride(1),
        right_perm.stride(2), right_perm.stride(3),
        # strides for C (output)
        out_perm.stride(0), out_perm.stride(1),
        out_perm.stride(2), out_perm.stride(3),
        B, H, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: [B, N, N, hidden_dim]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------
    # 5) Second LayerNorm, output gate, final projection
    # --------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
```
