"""
TriMul (outgoing) – fast forward pass
-------------------------------------

We implement the forward of the “outgoing” TriMul operator (AlphaFold‑3) in a
single entry point ``custom_kernel``.  The heavy O(N³) work is a batched
matrix‑multiply:

    C[g,i,j] = Σₖ  left[g,i,k] · right[g,k,j]      (g = batch × hidden_dim)

where *left* and *right* are the projected pairwise tensors of shape
(B, H, N, N).  The product is performed by a custom Triton kernel that
tiles the (M,N,K) dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates
in float32 while the inputs/outputs are fp16.  All point‑wise operations
(layer‑norms, linear projections, sigmoid gates and the final projection)
are done with PyTorch (mostly fp32, the GEMM tensors are cast to fp16).

The implementation supports:
  - optional masking,
  - arbitrary batch size (1‑2) and sequence length (up to 1024),
  - any hidden‑dim (default 128) and model dim (128/384),
  - both normal and heavy‑tailed input distributions.

The returned tensor is always ``float32`` with shape ``[B, N, N, D]``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – batched FP16 GEMM (C[g,i,j] = Σₖ A[g,i,k]·B[g,k,j])
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Arguments
    ---------
    a_ptr, b_ptr, c_ptr : pointers to fp16 tensors
    M, N, K           : matrix dimensions (all int64)
    stride_*          : element strides for batch, row, column
    BLOCK_*           : compile‑time constants (tile sizes)

    Grid: (ceil(M/BLOCK_M), ceil(N/BLOCK_N), groups)
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_g = tl.program_id(2)          # group = batch * hidden_dim

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ------------------------------------------------------------------
    # Pointers for the current tile
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_g * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k

    b_ptrs = b_ptr + pid_g * stride_b_batch \
                   + offs_k[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        # Load tiles with out‑of‑bounds masking
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Accumulate dot‑product (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

        # Advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # Write back result (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_g * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n

    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# ----------------------------------------------------------------------
# Entrypoint used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

        * ``input_tensor`` – (B, N, N, D) torch.Tensor, float32 or float16
        * ``mask_tensor``  – (B, N, N)  torch.Tensor or ``None``
        * ``weights``      – dict with all linear / layer‑norm parameters
        * ``config``       – dict containing ``dim``, ``hidden_dim`` and
          optional ``nomask`` flag

    Returns
    -------
    torch.Tensor
        (B, N, N, D) float32
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                     # D
    hidden_dim = config["hidden_dim"]       # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper – fetch weight tensors and move them to the proper device
    # ------------------------------------------------------------------
    def get(name):
        return weights[name].to(device)

    # LayerNorm over the model dimension D
    norm_weight = get("norm.weight")
    norm_bias   = get("norm.bias")

    # Linear weights (bias‑free)
    left_proj_weight   = get("left_proj.weight")    # (H, D)
    right_proj_weight  = get("right_proj.weight")   # (H, D)
    left_gate_weight   = get("left_gate.weight")    # (H, D)
    right_gate_weight  = get("right_gate.weight")   # (H, D)
    out_gate_weight    = get("out_gate.weight")     # (H, D)

    # Output norm & projection
    to_out_norm_weight = get("to_out_norm.weight")  # (H,)
    to_out_norm_bias   = get("to_out_norm.bias")    # (H,)
    to_out_weight      = get("to_out.weight")       # (D, H)

    # ------------------------------------------------------------------
    # 1️⃣  Input layer‑norm  (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣  Project to hidden dimension (float32)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: (B,N,N) → (B,N,N,1)
        mask_fp = mask_tensor.to(dtype).unsqueeze(-1)
        left  = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4️⃣  Compute sigmoid gates (float32)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣  Cast to fp16 and apply per‑channel gates
    # ------------------------------------------------------------------
    left  = left.to(torch.float16) * left_gate.to(torch.float16)
    right = right.to(torch.float16) * right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣  Rearrange for the batched GEMM:
    #     left  : (B,N,N,H) → (B,H,N,N) with axes (i,k)
    #     right : (B,N,N,H) → (B,H,N,N) with axes (k,j)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N) – i,k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) – k,j

    B, H, N, _ = left_perm.shape
    groups = B * H                     # total number of independent matmuls

    # Collapse the (B,H) axes → (G,N,N)
    A = left_perm.view(groups, N, N)
    B_mat = right_perm.view(groups, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7️⃣  Launch the Triton batched GEMM
    # ------------------------------------------------------------------
    # Tile sizes – tuned for H100 (shared‑mem friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        groups                     # one block per (batch×hidden)
    )

    _batched_matmul_kernel[grid](
        A, B_mat, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # batch, row, col strides
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣  Reshape back to (B,N,N,H) and promote to fp32 for the rest
    # ------------------------------------------------------------------
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 9️⃣  Output layer‑norm, out‑gate, final linear projection (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(out_hidden,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                       # element‑wise gating
    out = F.linear(out, to_out_weight)        # (B,N,N,D)

    return out
