"""
high‑performance Triton implementation of the *outgoing* TriMul operator.

Algorithm
---------
1. **Layer‑Norm** on the input tensor (FP16 when possible).
2. **Linear projections** (`left_proj`, `right_proj`) to the hidden dimension
   and optional **mask** multiplication.
3. **Gating**: three sigmoids (`left_gate`, `right_gate`, `out_gate`).
   The first two gates are fused with the projections (element‑wise product).
4. **Batched GEMM** (the N³ contraction):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is realised as a batched matrix‑multiply `C = A @ Bᵀ` where each
   `(batch, hidden)` pair is treated as an independent GEMM.
   The heavy work is performed by a single Triton kernel
   (`_batched_gemm_kernel`).  The kernel works on **fp16** data
   (inputs) and accumulates in **fp32** to keep numerical stability.
5. **Output Layer‑Norm**, multiplication with `out_gate` and a final
   linear projection back to the original `dim` (float32 output).

The implementation keeps all memory on‑device, uses mixed precision,
and fuses the N³ contraction into one highly‑tuned Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------
# Triton kernel: batched GEMM computing C = A @ Bᵀ
# A : (batch, M, K)   – fp16
# B : (batch, K, N)   – fp16 (already transposed for the contract)
# C : (batch, M, N)   – fp32 accumulator, stored as fp32
# ---------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    a_batch_stride, a_stride_m, a_stride_k,  # strides of A (elements)
    b_batch_stride, b_stride_k, b_stride_n,  # strides of B (elements)
    c_batch_stride, c_stride_m, c_stride_n,  # strides of C (elements)
    M, N, K,                                 # problem size
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each batch (including hidden dim) compute
    C = A @ Bᵀ   where A∈(M,K), B∈(N,K) (B is stored transposed as K×N).
    """
    batch_id = tl.program_id(0)          # batch = B * hidden_dim
    pid_m = tl.program_id(1)             # tile row
    pid_n = tl.program_id(2)             # tile col

    # -------------------- tile offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * a_batch_stride
    b_base = b_ptr + batch_id * b_batch_stride
    c_base = c_ptr + batch_id * c_batch_stride

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- K‑loop --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (M,K)
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # B tile (already transposed): (K,N)
        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # FP16·FP16 → FP32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # -------------------- write back --------------------
    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ---------------------------------------------------------------------
# Entry point (Python side)
# ---------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, C]   (float32)
        - mask         : torch.Tensor[B, N, N] or None (bool/float)
        - weights      : dict with the following keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
        - config : dict containing at least
                "dim" (C), "hidden_dim" (H), "nomask" (bool)

    Returns
    -------
    torch.Tensor[B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣  Layer‑Norm + Linear projections (mixed‑precision)
    # --------------------------------------------------------------
    # Autocast gives us fp16 for everything that can benefit from it.
    # The final output will be cast back to fp32.
    with torch.autocast(device_type="cuda", dtype=torch.float16):
        # Layer‑Norm on the input (no bias‑fusion needed)
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps,
        )  # [B, N, N, C]  (fp16)

        # Linear projections to hidden_dim
        left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
        right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

        # Optional mask (broadcast)
        if mask is not None and not nomask:
            # mask is float32 / bool → cast to fp16
            mask_fp16 = mask.to(dtype=torch.float16).unsqueeze(-1)  # [B, N, N, 1]
            left = left * mask_fp16
            right = right * mask_fp16

        # Gating logits → sigmoid (still fp16)
        left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
        right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
        out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,H]

        # Apply first two gates (still fp16)
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 2️⃣  Batched GEMM (the N³ contraction)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = [B, N, N, H]

    # -----------------------------------------------------------------
    # Re‑order tensors so that each (batch, hidden) pair becomes a
    # separate batch for the Triton kernel:
    #   left  : (B, H, N, N)    where dims are (batch, hidden, i, k)
    #   right : (B, H, N, N)    where dims are (batch, hidden, k, j)
    # -----------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()       # (B, H, N, N)

    BH = B * hidden_dim                                        # total "batch" for kernel

    left_flat = left_perm.reshape(BH, N, N)                    # (BH, M, K)
    right_flat = right_perm.reshape(BH, N, N)                  # (BH, K, N) – already transposed

    # Output buffer (fp32 accumulation)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Strides in *elements* (not bytes)
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)
    a_stride_k = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_stride_k = right_flat.stride(1)   # K dimension (row after transpose)
    b_stride_n = right_flat.stride(2)   # N dimension (col)

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # Tile sizes – chosen to give good occupancy on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # 3‑D grid: (batch, tile_m, tile_n)
    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,                 # M
        N,                 # N
        N,                 # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original layout: (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # --------------------------------------------------------------
    # 3️⃣  Output LayerNorm, gating, final linear (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]  (fp32)

    # Multiply by the previously computed out_gate (cast to fp32)
    out = out * out_gate.to(out.dtype)

    # Final projection back to `dim` (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
```
