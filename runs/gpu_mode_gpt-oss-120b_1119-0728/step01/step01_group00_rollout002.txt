"""
TriMul (outgoing) forward pass – highly optimized version.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,D).
2. Linear projections → left/right tensors of shape (B,N,N,H).
3. (Optional) apply the pairwise mask.
4. Compute three sigmoid gates from the normalized input.
5. Cast left/right to fp16 and multiply by their respective gates.
6. Re‑order tensors so that each hidden channel becomes a separate
   “group” (G = B·H) of matrices A[g,i,k] and B[g,k,j] (both fp16).
7. Launch a custom Triton kernel that performs a batched GEMM:
       C[g,i,j] = Σ_k A[g,i,k] * B[g,k,j]
   The kernel tiles over M,N,K (128×128×64) and accumulates in
   float32, storing the result in fp16. 8 warps per program tile
   exploit H100 tensor‑cores.
8. Reshape C back to (B,N,N,H) and cast to fp32.
9. Layer‑norm over the hidden dimension, element‑wise gating,
   and a final linear projection back to D.

All heavy O(N³·H) work (step 7) runs on Triton; the remaining
operations use PyTorch (mixed‑precision where possible).  The final
output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched fp16 matmul with fp32 accumulation.
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[g,i,j] = Σ_k A[g,i,k] * B[g,k,j] for all groups g.

    Grid dimensions: (num_blocks_M, num_blocks_N, num_groups)
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_g = tl.program_id(2)          # group (batch * hidden)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the current tile of A and B
    a_ptrs = a_ptr + pid_g * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_g * stride_b_batch \
                   + offs_k[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K‐tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        # Load tiles (fp16) with bounds checking
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Tensor‑core dot product (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

        # Advance K dimension
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back result (cast to fp16)
    c_ptrs = c_ptr + pid_g * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


# --------------------------------------------------------------
# Entry point – the custom forward kernel.
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] or None
        - weights_dict : dict of torch.Tensor (model weights)
        - config_dict  : dict with keys:
            * dim          – input/output channel dimension D
            * hidden_dim   – projection dimension H
            * nomask (opt) – True if mask is absent

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Helper for weight retrieval (move to correct device)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name].to(device)

    # LayerNorm over the last dim (D)
    norm_weight = _w("norm.weight")
    norm_bias = _w("norm.bias")

    # Linear projection weights (no bias)
    left_proj_weight = _w("left_proj.weight")
    right_proj_weight = _w("right_proj.weight")
    left_gate_weight = _w("left_gate.weight")
    right_gate_weight = _w("right_gate.weight")
    out_gate_weight = _w("out_gate.weight")
    to_out_norm_weight = _w("to_out_norm.weight")
    to_out_norm_bias = _w("to_out_norm.bias")
    to_out_weight = _w("to_out.weight")       # shape (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 4) Project to hidden dimension
    # ------------------------------------------------------------------
    # (B,N,N,H)  – keep in float32 for numerical stability before gating
    left = F.linear(x, left_proj_weight)      # → (B,N,N,H)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # mask: (B,N,N) → (B,N,N,1) → broadcast
        mask_fp = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 6) Gates (float32)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 7) Apply gates and cast to fp16 for the GEMM
    # ------------------------------------------------------------------
    left = (left.to(torch.float16) * left_gate.to(torch.float16))
    right = (right.to(torch.float16) * right_gate.to(torch.float16))

    # ------------------------------------------------------------------
    # 8) Rearrange tensors for batched GEMM:
    #    left  → (B,H,N,N) with axes (i,k)
    #    right → (B,H,N,N) with axes (k,j)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N) (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N) (k,j)

    B, H, N, _ = left_perm.shape
    groups = B * H

    # Collapse batch and hidden dim → (G,N,N)
    A = left_perm.view(groups, N, N)
    B_mat = right_perm.view(groups, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 9) Launch Triton batched GEMM kernel
    # --------------------------------------------------------------
    # Tile size tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        groups,
    )

    _batched_matmul_kernel[grid](
        A, B_mat, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps ⇒ 256 threads per program
    )

    # --------------------------------------------------------------
    # 10) Reshape back to (B,N,N,H) and cast to fp32 for remaining ops
    # --------------------------------------------------------------
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out_hidden = out_hidden.to(torch.float32)

    # --------------------------------------------------------------
    # 11) Output layer‑norm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(out_hidden,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate               # element‑wise gating (fp32)

    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
