"""
Highly‑optimized forward pass of the *outgoing* TriMul block (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the 4‑D input (float16 for speed).
2. Fuse the two linear projections (left/right) into a single matmul.
3. Fuse the three gate‑linear projections (left_gate, right_gate, out_gate)
   into a single matmul and apply sigmoid.
4. Apply optional pairwise mask, then multiply left/right by their gates.
5. Reshape to (B·H, N, N) and run a custom Triton GEMM:
       C = A @ Bᵀ   (A = left, B = right)
   – each program instance computes a tile (BLOCK_M×BLOCK_N) and
   – accumulation is FP32 while all loads/stores are FP16.
   – BLOCK_M = BLOCK_N = 128, BLOCK_K = 64 are tuned for H100 tensor‑cores.
6. Reshape the result back to (B, N, N, H) (still FP16).
7. Layer‑norm over the hidden dimension, multiply by the out‑gate,
   and a final FP16 linear projection back to the original channel dimension.
8. Return the result as FP32.

Only the O(N³) reduction (step 5) is performed in Triton; the surrounding
operations are fused and run in mixed‑precision PyTorch to minimise memory
traffic and maximize throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                    # left  tile     (batch, M, K)
    b_ptr,                    # right tile     (batch, N, K) – we read it as (K, N)
    c_ptr,                    # output tile    (batch, M, N)
    batch, M, N, K,           # logical sizes
    stride_ab, stride_am, stride_ak,   # strides of A (batch, M, K)
    stride_bb, stride_bn, stride_bk,   # strides of B (batch, N, K)
    stride_cb, stride_cm, stride_cn,   # strides of C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where each program instance computes a (BLOCK_M, BLOCK_N)
    tile of the output for a specific (batch, hidden) slice.
    Accumulation uses FP32 for numeric stability; the final result is stored
    as FP16 to keep memory bandwidth low.
    """
    pid_b = tl.program_id(2)                # batch * hidden index
    pid_m = tl.program_id(0)                # tile row
    pid_n = tl.program_id(1)                # tile col

    # ----- tile coordinates ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ----- accumulator ----------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑loop ---------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # masks for tail handling
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # A : (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )
        # Bᵀ : (BLOCK_K, BLOCK_N)  (we read B as (K,N) and treat it as transposed)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
        )
        # dot product over the K‑dimension
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)   FP32 += FP16·FP16 → FP32

    # ----- write back ------------------------------------------------------
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_c,
    )

        eps=eps,
    )
    x = x.to(torch.float16)   # mixed‑precision path

    # ------------------------------------------------------------------
    # 2. Fused projection: left & right (dim → hidden_dim)
    # ------------------------------------------------------------------
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, D)
    proj = F.linear(x, proj_weight)         # (B, N, N, 2*H)
    left, right = proj.split(hidden_dim, dim=-1)   # each (B, N, N, H)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask may be bool or float; cast once to FP16
        mask_ = mask.to(dtype=torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Fused gating linear + sigmoid
    # ------------------------------------------------------------------
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # (3*H, D)
    gates = F.linear(x, gate_weight)        # (B, N, N, 3*H)
    left_gate, right_gate, out_gate = gates.split(hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5. Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Batched GEMM: out = left @ rightᵀ  (per hidden channel)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B·H, N, N) and keep FP16
    left_tile = (
        left.permute(0, 3, 1, 2)                 # (B, H, N, N)
        .contiguous()
        .view(B * H, N, N)
    )
    right_tile = (
        right.permute(0, 3, 1, 2)
        .contiguous()
        .view(B * H, N, N)
    )
    out_tile = torch.empty_like(left_tile, dtype=torch.float16)

    # strides for the kernel (batch, M, K) / (batch, N, K) / (batch, M, N)
    stride_ab = left_tile.stride(0)
    stride_am = left_tile.stride(1)
    stride_ak = left_tile.stride(2)

    stride_bb = right_tile.stride(0)
    stride_bn = right_tile.stride(1)
    stride_bk = right_tile.stride(2)

    stride_cb = out_tile.stride(0)
    stride_cm = out_tile.stride(1)
    stride_cn = out_tile.stride(2)

    # tile sizes – empirically good for H100 (fits in shared memory,
    # allows Tensor‑Core usage)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along M (rows)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along N (cols)
        B * H,                          # one program per (batch, hidden)
    )

    tri_mul_gemm_kernel[grid](
        left_tile,
        right_tile,
        out_tile,
        B * H,
        N,
        N,
        N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bn,
        stride_bk,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Restore shape: (B, N, N, H)  – keep FP16 for the next ops
    # ------------------------------------------------------------------
    out = (
        out_tile.view(B, H, N, N)           # (B, H, N, N)
        .permute(0, 2, 3, 1)                # (B, N, N, H)
    )  # still FP16

    # ------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm over hidden dim, out‑gate, final proj
    # ------------------------------------------------------------------
    # to_out_norm (hidden_dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # multiply by out‑gate (broadcasted)
    out = out * out_gate

    # final projection back to the original dim
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 9. Cast back to FP32 for the public API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
