    device = input_tensor.device
    dtype = input_tensor.dtype

"""
Outgoing TriMul (Triangular Multiplicative Update) implementation.

The forward pass follows the reference PyTorch code but the cubic
contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is executed by a custom Triton kernel.  The kernel treats each
(batch, hidden) pair as a separate batch entry and computes a
batched GEMM (C = A @ Bᵀ) in half‑precision with FP32 accumulation.
All other operations (LayerNorm, Linear, sigmoid gates) are performed
with PyTorch for simplicity.

The kernel has been tuned for H100:
  * BLOCK_M = BLOCK_N = 64, BLOCK_K = 32 → good occupancy.
  * Inputs are cast to fp16 before the GEMM, allowing tensor‑core
    execution while keeping the final result in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,                       # pointers
    a_batch_stride, a_stride_m, a_stride_k,    # A strides (elements)
    b_batch_stride, b_stride_k, b_stride_n,    # B strides (elements)
    c_batch_stride, c_stride_m, c_stride_n,    # C strides (elements)
    M, N, K,                                   # matrix sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A,B,C are (batch, M, K), (batch, K, N), (batch, M, N)).
    A and B are stored as fp16, accumulation is performed in fp32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_base = a_ptr + batch_id * a_batch_stride
    b_base = b_ptr + batch_id * b_batch_stride
    c_base = c_ptr + batch_id * c_batch_stride

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (float16)
        a = tl.load(
            a_base
            + offs_m[:, None] * a_stride_m
            + offs_k[None, :] * a_stride_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # load B tile (float16) – B is already transposed (K × N)
        b = tl.load(
            b_base
            + offs_k[:, None] * b_stride_k
            + offs_n[None, :] * b_stride_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # FP16 × FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + offs_m[:, None] * c_stride_m + offs_n[None, :] * c_stride_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask        : torch.Tensor [B, N, N] or None
            - weights     : dict with all model parameters
            - config      : dict containing 'dim', 'hidden_dim', 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # -----------------------------------------------------------------
    # 2. Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask is broadcasted on the channel dimension
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))      # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Half‑precision batched GEMM (the O(N³) contraction)
    # -----------------------------------------------------------------
    # Cast to fp16 to enable tensor‑core acceleration
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Rearrange to (batch, hidden, N, N) and prepare the transposed right operand
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) – rows=i, cols=k
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – rows=k, cols=j

    B, H, N, _ = left_perm.shape
    BH = B * H  # flat batch dimension used by the kernel

    left_flat = left_perm.reshape(BH, N, N)   # (BH, N, N)
    right_flat = right_perm.reshape(BH, N, N)  # (BH, N, N)

    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Strides in element units (not bytes)
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)
    a_stride_k = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_stride_k = right_flat.stride(1)
    b_stride_n = right_flat.stride(2)

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # Tuned tile sizes for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_fp16_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, C]

    return out
