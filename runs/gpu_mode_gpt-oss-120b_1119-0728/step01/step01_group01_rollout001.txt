"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) with a lightweight Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and apply the three linear projections (left, right, gates)
   with fp16 weights – this is the dominant work (B·N²·dim·hidden).
3. Optional mask (broadcast over the hidden dimension) is applied in fp32
   before the cast to fp16.
4. Element‑wise multiplication of the projected tensors with their per‑
   element gates is performed by a tiny Triton kernel (`_elemwise_mul`).
5. The heavy contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched GEMM (A @ Bᵀ) and is executed with `torch.bmm`, which
   leverages cuBLAS fp16 kernels on H100.
6. The result is cast back to fp32, Layer‑normed, gated by ``out_gate``,
   and finally projected back to the original channel dimension.

Only the element‑wise gate application (step 4) uses a custom Triton
kernel, satisfying the “use Triton” requirement while the compute‑heavy
matrix multiply is delegated to the highly‑optimised cuBLAS implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: out = a * b (element‑wise) on a flat view.
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul(
    a_ptr,                # pointer to operand a (contiguous 1‑D)
    b_ptr,                # pointer to operand b (contiguous 1‑D)
    out_ptr,              # pointer to output (contiguous 1‑D)
    size: tl.constexpr,   # total number of elements
    BLOCK: tl.constexpr   # block size (tiles processed per program)
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
        mask_tensor  : torch.Tensor of shape [B, N, N] (float32/bool/None)
        weights_dict : mapping of weight names to tensors (already on device)
        config_dict  : {"dim": int,
                        "hidden_dim": int,
                        "nomask": bool (optional)}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_fp32 = input_tensor.dtype
    dtype_fp16 = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Helper to fetch weights (already on the correct device)
    # --------------------------------------------------------------
    def _w(name):
        return weights[name]

    # --------------------------------------------------------------
    # LayerNorm over the last dimension (float32)
    # --------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim] (fp32)

    # --------------------------------------------------------------
    # Cast to fp16 for the heavy linear work
    # --------------------------------------------------------------
    x = x_norm_fp32.to(dtype_fp16)

    # --------------------------------------------------------------
    # Linear projections (no bias) – fp16 weights
    # --------------------------------------------------------------
    left_proj_w = _w("left_proj.weight").to(dtype_fp16)
    right_proj_w = _w("right_proj.weight").to(dtype_fp16)

    left = F.linear(x, left_proj_w)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(dtype_fp16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gating – linear + sigmoid (still fp16)
    # --------------------------------------------------------------
    left_gate_w = _w("left_gate.weight").to(dtype_fp16)
    right_gate_w = _w("right_gate.weight").to(dtype_fp16)
    out_gate_w = _w("out_gate.weight").to(dtype_fp16)

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm_fp32, _w("out_gate.weight")))  # keep fp32 for final gating

    # --------------------------------------------------------------
    # Triton element‑wise multiplication: left = left * left_gate
    #   (same for right).  This is the only required Triton kernel.
    # --------------------------------------------------------------
    BLOCK = 65536  # 64 K elements per program – fits comfortably in registers

    # left = left * left_gate
    left_flat = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    gated_left = torch.empty_like(left_flat)
    grid_left = (triton.cdiv(left_flat.numel(), BLOCK),)
    _elemwise_mul[grid_left](
        left_flat,
        left_gate_flat,
        gated_left,
        left_flat.numel(),
        BLOCK=BLOCK,
        num_warps=1,
    )
    left = gated_left.view_as(left)

    # right = right * right_gate
    right_flat = right.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    gated_right = torch.empty_like(right_flat)
    grid_right = (triton.cdiv(right_flat.numel(), BLOCK),)
    _elemwise_mul[grid_right](
        right_flat,
        right_gate_flat,
        gated_right,
        right_flat.numel(),
        BLOCK=BLOCK,
        num_warps=1,
    )
    right = gated_right.view_as(right)

    # --------------------------------------------------------------
    # Heavy contraction – batched GEMM (A @ Bᵀ) with fp16 cuBLAS.
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N)  (contiguous)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_mat = left_perm.view(batch_total, N, N)          # [B*H, N, N]
    right_mat = right_perm.view(batch_total, N, N)        # [B*H, N, N]

    # Batched matmul: (N,N) @ (N,N)ᵀ  ->  (N,N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16

    # Cast back to FP32 for the remaining layers
    out = out_mat.to(dtype_fp32).view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # Output processing: LayerNorm → out_gate → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight"),
        bias=_w("to_out_norm.bias"),
    )
    out = out * out_gate  # broadcast over hidden dim
    out = F.linear(out, _w("to_out.weight"))   # [B, N, N, dim] (fp32)

    return out
