import torch
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    c = a * b
    tl.store(c_ptr + offs, c, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    The heavy N³ matrix multiplication is performed with a batched
    torch.bmm (leveraging cuBLAS/Tensor‑Core kernels).  A tiny Triton
    kernel is used to fuse the final element‑wise multiplication with the
    output gate.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ---------- Input layer‑norm (float32) ----------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ---------- Cast to half for the bulk of the work ----------
    x_h = x_norm.half()

    # ---------- Concatenate projection + gate weights ----------
    left_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    out_gate_w = weights["out_gate.weight"].half()

    # ---------- Linear transforms producing proj & gate ----------
    left_cat = torch.nn.functional.linear(x_h, left_w)   # [B,N,N,2*hidden] half
    right_cat = torch.nn.functional.linear(x_h, right_w)

    # ---------- Split, apply sigmoid to gates ----------
    left_proj = left_cat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])

    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))  # [B,N,N,hidden] half

    # ---------- Optional mask ----------
    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)  # broadcast
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ---------- Apply element‑wise gates ----------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ---------- Batched matmul (B*hidden GEMMs) ----------
    B, N, _, _ = left.shape
    # reshape to [B*hidden, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm uses Tensor Cores for FP16 → fast
    out_flat = torch.bmm(left_flat, right_flat)  # half

    # ---------- Reshape back to [B, N, N, hidden] ----------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).float()  # float32

    # ---------- Output layer‑norm ----------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # ---------- Fuse out_gate with out_norm via Triton ----------
    # out_gate is half → cast to float32
    out_gate_f = out_gate.float()
    out_gated = torch.empty_like(out_norm)
    total = out_norm.numel()
    BLOCK = 8192
    grid = (triton.cdiv(total, BLOCK),)
    elementwise_mul_kernel[grid](
        out_norm,
        out_gate_f,
        out_gated,
        total,
        BLOCK_SIZE=BLOCK,
    )

    # ---------- Final linear projection ----------
    result = torch.nn.functional.linear(out_gated, weights["to_out.weight"])
    return result

import torch
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(
    a_ptr,  # float32*   – first operand
    b_ptr,  # float32*   – second operand (out_gate)
    c_ptr,  # float32*   – output
    size,   # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """
    Very small Element‑wise kernel used to fuse the final
    out_norm * out_gate multiplication.  It loads a 1‑D
    slice of the tensors, multiplies them and writes the result
    back, handling boundary conditions with a mask.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul module
    (the heavy N³ operation is performed with a batched torch.bmm,
    which on H100 uses Tensor‑Core kernels).  A tiny Triton kernel
    fuses the final element‑wise multiplication with the output gate.
    All other operations stay in PyTorch.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑Norm on the input (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                            # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all heavy linear / matmul work
    # ------------------------------------------------------------------
    x_h = x_norm.half()                          # [B, N, N, dim] (float16)

    # ------------------------------------------------------------------
    # 3) Fuse projection + gate linear layers by concatenating weights
    # ------------------------------------------------------------------
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()                                      # [2*hidden, dim] (fp16)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()
    out_gate_w = weights["out_gate.weight"].half()  # [hidden, dim] (fp16)

    # ------------------------------------------------------------------
    # 4) One linear per side → (proj , gate) packed in the last dim
    # ------------------------------------------------------------------
    left_cat = torch.nn.functional.linear(x_h, left_w)   # [B,N,N,2*hidden] fp16
    right_cat = torch.nn.functional.linear(x_h, right_w)

    # split
    left_proj = left_cat[..., :hidden_dim]               # [B,N,N,hidden] fp16
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])  # fp16
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # output gate (used after the output LayerNorm)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))  # fp16

    # ------------------------------------------------------------------
    # 5) Optional mask (applied before gating, exactly as reference)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)           # [B,N,N,1] fp16
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # [B,N,N,hidden] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Batched matrix multiplication  (B*hidden independent GEMMs)
    #    Each GEMM is:  (N x N) @ (N x N)ᵀ  →  (N x N)
    #    Implemented with torch.bmm which on H100 runs on Tensor‑Cores.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to [B*hidden, N, N]; for the right side we need the (k,j) order → transpose last two dims
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # fp16
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)  # fp16

    out_flat = torch.bmm(left_flat, right_flat)           # fp16, shape [B*hidden, N, N]

    # ------------------------------------------------------------------
    # 8) Reshape back to the original layout and cast to float32
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .float()
    )                                                       # [B,N,N,hidden] fp32

    # ------------------------------------------------------------------
    # 9) Layer‑Norm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                       # fp32

    # ------------------------------------------------------------------
    # 10) Fuse out_norm * out_gate using a tiny Triton kernel
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.float()
    gated = torch.empty_like(out_norm)                     # fp32 output buffer
    total_elems = out_norm.numel()
    BLOCK = 8192                                            # 8 K elements per program
    grid = (triton.cdiv(total_elems, BLOCK),)
    elementwise_mul_kernel[grid](
        out_norm,
        out_gate_f32,
        gated,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 11) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    result = torch.nn.functional.linear(gated, weights["to_out.weight"])

    return result
