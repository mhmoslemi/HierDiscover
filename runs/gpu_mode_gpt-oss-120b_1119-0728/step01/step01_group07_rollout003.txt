"""
TriMul (outgoing) – fast Triton implementation.

The forward pass consists of
1️⃣ Layer‑Norm (fp16)
2️⃣ Linear projections  x → left, right               (B·N²×dim → B·N²×hidden)
3️⃣ Gating + optional mask (element‑wise, fp16)
4️⃣ Core multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply per (b,h).  The kernel below
   builds on the classic tiled GEMM (BLOCK_M×BLOCK_N×BLOCK_K)
   and multiplies left (M×K) with the *transpose* of right (K×N) so
   we obtain the required (M×N) block.  All arithmetic is in fp16
   with fp32 accumulation for numeric stability.
5️⃣ Second Layer‑Norm + output‑gate + final linear (fp16 → fp32).

The heavy N³ work is done entirely in Triton, while the cheap
operations (norms, linear layers, sigmoids) use torch kernels that
already exploit cuBLAS/cuDNN.  Half‑precision cuts memory traffic
and enables tensor‑core utilisation on H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr,          # *fp16   left tensor  [S, N, N]  (S = B*H)
    right_ptr,         # *fp16   right tensor [S, N, N]
    out_ptr,           # *fp16   output tensor[S, N, N]
    total_batches,     # i32     = B*H
    N,                 # i32
    # strides (in elements) of the three tensors
    stride_left_batch, stride_left_row, stride_left_col,
    stride_right_batch, stride_right_row, stride_right_col,
    stride_out_batch, stride_out_row, stride_out_col,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) Map program id to (batch, tile_m, tile_n)
    # --------------------------------------------------------------
    tiles_m = tl.cdiv(N, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = tiles_m * tiles_n

    batch_idx = pid // tiles_per_batch
    tile_idx  = pid % tiles_per_batch
    pid_m = tile_idx // tiles_n
    pid_n = tile_idx % tiles_n

    # --------------------------------------------------------------
    # 2) Compute the coordinates of the output block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # (M,)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # (N,)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over K dimension (tiles of size BLOCK_K)
    # --------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)               # (K,)
        mask_k = cur_k < N

        # left block:  (M, K)
        left_offs = (
            batch_idx * stride_left_batch
            + offs_m[:, None] * stride_left_row
            + cur_k[None, :] * stride_left_col
        )
        left = tl.load(left_ptr + left_offs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # right block *transposed*: (K, N)
        # we load right[j,k] as right_T[k,j] to match tl.dot(left, right_T)
        right_offs = (
            batch_idx * stride_right_batch
            + cur_k[:, None] * stride_right_row
            + offs_n[None, :] * stride_right_col
        )
        right = tl.load(right_ptr + right_offs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        # dot product – fp16 inputs, fp32 accumulation
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # 4) Write back the result
    # --------------------------------------------------------------
    out_offs = (
        batch_idx * stride_out_batch
        + offs_m[:, None] * stride_out_row
        + offs_n[None, :] * stride_out_col
    )
    tl.store(out_ptr + out_offs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# Helper: fuse element‑wise projection * gate * mask (optional)
# ----------------------------------------------------------------------
def _apply_gate_and_mask(tensor, gate, mask):
    """
    tensor, gate : [B,N,N,H] (fp16)
    mask         : [B,N,N]   (fp16) – broadcast over H
    Returns tensor * gate * mask  (still fp16)
    """
    # torch broadcasting is already very fast; we keep it simple.
    return tensor * gate * mask.unsqueeze(-1)
