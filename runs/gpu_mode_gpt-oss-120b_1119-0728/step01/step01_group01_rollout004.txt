```python
def trimmul_outgoing_gemm_kernel(a_ptr, b_ptr, c_ptr,
                                 stride_a_batch, stride_a_m, stride_a_k,
                                 stride_b_batch, stride_b_k, stride_b_n,
                                 stride_c_batch, stride_c_m, stride_c_n,
                                 N,
                                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0) # block row
    pj = tl.program_id(1) # block col
    pb = tl.program_id(2) # batch dim (b*hidden)

    off_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pj * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # a: shape [M, K]
        a_ptrs = a_ptr + pb * stride_a_batch + off_m[:, None] * stride_a_m + off_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=(off_m[:, None] < N) & (off_k[None, :] < N), other=0.0).to(tl.float32)
```

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _trimmul_outgoing_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    N,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block
    pid_b = tl.program_id(2)   # batch*hidden

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        cur_k = kk * BLOCK_K + offs_k

        # A block (M x K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + cur_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & (cur_k[None, :] < N)), other=0.0)
        a = a.to(tl.float32)

        # B block (K x N)
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + cur_k[:, None] * stride_b_i
            + offs_n[None, :] * stride_b_k
        )
        b = tl.load(b_ptrs, mask=((cur_k[:, None] < N) & mask_n[None, :]), other=0.0)
        b = b.to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # write-back
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc, mask=(mask_m[:, None] & mask_n[None, :]))
```

def custom_kernel(data):
    """TriMul outgoing implementation using a fused Triton GEMM.

    The heavy contraction out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is executed as a batched matrix multiplication A @ B^T where A and B
    are the left/right projections after gating and optional masking.
    All other operations (LayerNorm, gating, final projection) are done
    with regular PyTorch calls.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Shortcut for weights
    _w = lambda name: weights[name]

    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=_w("norm.weight"),
                     bias=_w("norm.bias"))

    # Linear projections (no bias)
    left = F.linear(x, _w("left_proj.weight"))   # [B,N,N,hidden]
    right = F.linear(x, _w("right_proj.weight"))

    # mask handling (broadcast over hidden)
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating values (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, _w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, _w("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x, _w("out_gate.weight")))

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the GEMM.
    # ------------------------------------------------------------------
    # left: [B,N,N,hidden]  -> (B,hidden,N,N)  (row=i, col=k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: need to transpose the last two dims so that we get (B,hidden,k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape

"""
TriMul “outgoing” implementation (AlphaFold3).

The costly part of the module is the 4‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be seen as a batched matrix‑multiply
    A @ Bᵀ   with   A = left[...,i,k,d]   (shape B×H×N×N)
                B = right[...,j,k,d]  (shape B×H×N×N)

We (1) apply LayerNorm, linear projections and sigmoid gates with PyTorch,
(2) reorder the tensors so that each (batch,head) pair is an N×N matrix,
(3) launch a Triton kernel that computes a FP16 matrix multiply with
    FP32 accumulation (tensor‑core friendly on H100),
(4) reshape the result back to [B,N,N,H] and finish with a second
    LayerNorm, an output gate and a final linear projection to the model
    dimension.

All tensors are kept on the device; the kernel works for any batch size
(1 – 2) and sequence length up to 1024.  The code follows the signature
required by the evaluation harness.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _trimmul_outgoing_gemm_kernel(
    a_ptr,                     # left  (B*H,N,N)  FP16
    b_ptr,                     # right (B*H,N,N)  FP16  (already transposed)
    c_ptr,                     # out   (B*H,N,N)  FP32
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    N,                         # sequence length
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_b = tl.program_id(2)          # batch*head index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds loads
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for block_k in range(num_k_blocks):
        cur_k = block_k * BLOCK_K + offs_k

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + cur_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (cur_k[None, :] < N),
            other=0.0,
        )
        a = a.to(tl.float32)

        # Load B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + pid_b * stride_b_batch
            + cur_k[:, None] * stride_b_i   # row‑stride = K‑dimension
            + offs_n[None, :] * stride_b_k   # col‑stride = N‑dimension
        )
        b = tl.load(
            b_ptrs,
            mask=(cur_k[:, None] < N) & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)

        # Accumulate matmul
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result (FP32)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor  [B, N, N, dim]   (float32, CUDA)
        mask_tensor  : torch.Tensor  [B, N, N]       (float32/bool/None)
        weights_dict : dict(name -> torch.Tensor)    (already on device)
        config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Helper to fetch a weight tensor
    _w = lambda name: weights[name]

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the last model dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    left = F.linear(x, _w("left_proj.weight"))
    right = F.linear(x, _w("right_proj.weight"))

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid over another linear projection)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, _w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, _w("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x, _w("out_gate.weight")))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM
    # --------------------------------------------------------------
    # left: (B, N, N, H) → (B, H, N, N)  rows=i, cols=k
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs a transpose of the last two dims: (B, H, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # Cast to FP16 so the kernel can exploit tensor‑cores.
    # Accumulation stays in FP32 inside the kernel.
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Collapse batch & head dimensions → [batch_total, N, N]
    left_flat = left_fp16.view(batch_total, N, N)
    right_flat = right_fp16.view(batch_total, N, N)

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty((batch_total, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for H100, FP16 tensor cores)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_total,               # batch*head
    )

    _trimmul_outgoing_gemm_kernel[grid](
        # pointers
        left_flat,
        right_flat,
        out_flat,
        # strides for A (left)
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for B (right, already transposed)
        right_flat.stride(0),
        right_flat.stride(1),   # row‑stride (K)
        right_flat.stride(2),   # col‑stride (N)
        # strides for C (output)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        # problem size
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6️⃣ Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Post‑processing: LayerNorm → output gate → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight"),
        bias=_w("to_out_norm.bias"),
    )
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, _w("to_out.weight"))  # → [B, N, N, dim]

    return out
