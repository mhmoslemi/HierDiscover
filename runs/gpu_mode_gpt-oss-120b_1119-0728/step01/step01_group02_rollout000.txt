"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The module works on a 4‑D tensor `x` of shape [B, N, N, D] (D = dim) and
computes

    out[b,i,j,d] = Σ_k left[b,i,k,d] · right[b,j,k,d]

where ``left`` and ``right`` are linear projections of a layer‑normed input,
optionally masked and modulated by learned sigmoid gates.  The costly
N³ reduction is realised as a *batched* matrix multiplication (A @ Bᵀ) for
every hidden channel (H = hidden_dim).  The implementation fuses this
multiplication in a Triton kernel:

*   The input is layer‑normalised (FP32) and projected to the hidden
    dimension (FP32).
*   Masking and gate multiplications are performed in PyTorch (they are
    element‑wise and inexpensive compared with the N³ step).
*   ``left`` and ``right`` are reshaped to (B·H, N, N) and cast to FP16.
*   A custom Triton GEMM kernel (`tri_mul_gemm_kernel`) computes
    C = A @ Bᵀ for each batch·hidden pair, accumulating in FP32 and writing
    FP16.
*   The result is reshaped back to [B, N, N, H] (FP32), layer‑normalised,
    modulated by the output gate and finally projected back to ``dim``.
*   The final tensor has shape [B, N, N, dim] and dtype ``float32``.

Only the O(N³) part lives on the GPU; everything else uses regular
PyTorch (negligible runtime).  The kernel uses a 64×64×64 tiling which
fits well in the shared memory of an H100 and provides high occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                                    # A = left  (batch, M, K)
    b_ptr,                                    # B = right (batch, N, K)  (K‑dimension is transposed inside the kernel)
    c_ptr,                                    # C = out   (batch, M, N)
    batch, M, N, K,                           # dimensions
    stride_ab, stride_am, stride_ak,          # strides for A   (batch, M, K)
    stride_bb, stride_bn, stride_bk,          # strides for B   (batch, N, K)
    stride_cb, stride_cm, stride_cn,          # strides for C   (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:  C = A @ Bᵀ
    A shape: (batch, M, K)
    B shape: (batch, N, K)  (treated as K×N after loading)
    C shape: (batch, M, N)
    Accumulation is performed in FP32, while the output is stored as FP16.
    """
    pid_b = tl.program_id(2)               # batch * hidden index
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile column

    # 1) Compute the absolute offsets of the tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                     # reused inside the loop

    # 2) Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # 3) Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        cur_k_offs = cur_k + offs_k

        # Masks for out‑of‑bounds loads (necessary for partial tiles)
        mask_a = (offs_m[:, None] < M) & (cur_k_offs[None, :] < K)
        mask_b = (cur_k_offs[:, None] < K) & (offs_n[None, :] < N)

        # Load tiles of A and Bᵀ
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + cur_k_offs[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + cur_k_offs[:, None] * stride_bk    # K‑dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bn,       # N‑dimension (column of Bᵀ)
            mask=mask_b,
            other=0.0,
        )
        # Accumulate the partial result
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # 4) Write the result (convert to FP16)
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_c,
    )

        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Apply the output gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
