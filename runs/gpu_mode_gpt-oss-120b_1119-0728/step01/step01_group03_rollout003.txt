def _gate_and_mask_kernel(
    L_ptr, G_ptr, M_ptr, L_out_ptr,
    N_total,  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N_total

    # load
    l = tl.load(L_ptr + offsets, mask=mask, other=0.0)
    g = tl.load(G_ptr + offsets, mask=mask, other=0.0)
    m = tl.load(M_ptr + offsets, mask=mask, other=1.0)  # default 1

    out = l * g * m
    tl.store(L_out_ptr + offsets, out, mask=mask)

"""
Optimized TriMul outgoing forward.

- Compute input layernorm.
- Project to hidden using FP16 linear layers.
- Compute element‑wise gates with sigmoid.
- Fuse gate (and optional mask) multiplication with a small Triton kernel.
- Perform the N³ batched matrix multiplication with `torch.bmm`, leveraging
  Tensor‑Core GEMM on the H100.
- Apply the output layer‑norm, gating and final linear projection.
"""
import torch
import triton
import triton.language as tl

@triton.jit
def _gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    total_elements,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        m = tl.load(mask_ptr + offs, mask=mask, other=1.0)
        l = l * lg * m
        r = r * rg * m
    else:
        l = l * lg
        r = r * rg

    tl.store(left_out_ptr + offs, l, mask=mask)
    tl.store(right_out_ptr + offs, r, mask=mask)

def custom_kernel(data):
    """Triton‑accelerated forward for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors residing on CUDA
        - config       : dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5
    # -------------------------------------------------
    # 1. Input LayerNorm
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2. Linear projections (FP16) – left & right
    # -------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]).to(torch.float16)
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"]).to(torch.float16)

    # -------------------------------------------------
    # 3. Gates (sigmoid) – also FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    ).to(torch.float16)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    ).to(torch.float16)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    ).to(torch.float32)   # keep as FP32 for later multiplication

    # -------------------------------------------------
    # 4. Fuse gate (and mask) with a tiny Triton kernel
    # -------------------------------------------------
    # Prepare mask if it exists and nomask=False
    if not nomask and mask is not None:
        mask_fp16 = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        has_mask = 1
    else:
        mask_fp16 = torch.empty(0, device=device, dtype=torch.float16)  # dummy
        has_mask = 0

    # Allocate output buffers
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total = left.numel()
    BLOCK = 65536  # 64k elements per program (should fit L2)
    grid = ( (total + BLOCK - 1) // BLOCK, )
    _gate_mask_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask_fp16,
        left_gated, right_gated,
        total,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )

    # -------------------------------------------------
    # 5. Batched matrix multiplication via torch.bmm (TensorCores)
    # -------------------------------------------------
    # Reshape to (B*hidden, N, N) and perform L @ Rᴛ
    bh = bs * hidden_dim
    # Bring hidden dimension to the second axis
    left_mat = left_gated.permute(0, 3, 1, 2).contiguous().view(bh, seq_len, seq_len)      # [B*H, N, N]
    right_mat = right_gated.permute(0, 3, 1, 2).contiguous().view(bh, seq_len, seq_len)    # [B*H, N, N]

    # GEMM: (N,N) @ (N,N)ᵀ → (N,N) for each slice
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16 output

    # Cast to FP32 for the following LayerNorm
    out_fp32 = out_mat.to(torch.float32)

    # Restore the original layout: [B, N, N, hidden]
    out = out_fp32.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6. Output LayerNorm, gating and final linear
    # -------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden] float32

    # Apply output gate (broadcasted on last dim)
    out_norm = out_norm * out_gate

    # Final projection back to `dim`
    result = torch.nn.functional.linear(out_norm, weights["to_out.weight"])  # float32

    return result
```

"""
Triton‑accelerated outgoing TriMul forward.

The heavy N³ computation is done with a batched GEMM using
`torch.bmm`, which on an H100 runs on Tensor‑Core FP16.
A tiny Triton kernel fuses the element‑wise gating
(left_gate/right_gate) for all hidden channels in a single pass.
Masking (when present) is applied afterwards with a cheap
broadcasted multiplication.

Input tensors are FP32, intermediate projections are cast to FP16
to maximise throughput, and the final output is returned in FP32.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def _gate_fuse_kernel(
    left_ptr,   # FP16 input [B,N,N,H]
    left_gate_ptr,  # FP16 gate   [B,N,N,H]
    right_ptr,  # FP16 input [B,N,N,H]
    right_gate_ptr, # FP16 gate   [B,N,N,H]
    left_out_ptr,   # FP16 output
    right_out_ptr,  # FP16 output
    total_elems,    # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuses   out_left = left * left_gate   and   out_right = right * right_gate."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    l  = tl.load(left_ptr + offs,      mask=mask, other=0.0)  # FP16
    lg = tl.load(left_gate_ptr + offs,mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    tl.store(left_out_ptr  + offs, l * lg, mask=mask)
    tl.store(right_out_ptr + offs, r * rg, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (already on CUDA)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden – keep everything in FP16
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"]).to(torch.float16)
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid) – also FP16, except out_gate which stays FP32
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    ).to(torch.float16)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    ).to(torch.float16)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    ).to(torch.float32)   # used later after the output LayerNorm

    # ------------------------------------------------------------------
    # 4) Fuse gate multiplication with a tiny Triton kernel
    # ------------------------------------------------------------------
    left_gated  = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    total = left.numel()                     # B * N * N * H
    BLOCK = 65536                               # 64 Ki elements per program
    grid = ((total + BLOCK - 1) // BLOCK,)

    _gate_fuse_kernel[grid](
        left, left_gate,
        right, right_gate,
        left_gated, right_gated,
        total,
        BLOCK_SIZE=BLOCK,
    )

    # Apply mask if required (broadcasted over the hidden dimension)
    if not nomask and mask is not None:
        mask_fp = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left_gated  = left_gated * mask_fp
        right_gated = right_gated * mask_fp

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Perform it as a batch of GEMMs (B*H, N, N) @ (B*H, N, N)^T using torch.bmm
    # ------------------------------------------------------------------
    bh = bs * hidden_dim  # total batch * hidden slices

    # Move hidden to the second axis so we can view as (bh, N, N)
    left_mat  = left_gated.permute(0, 3, 1, 2).contiguous().view(bh, seq_len, seq_len)
    right_mat = right_gated.permute(0, 3, 1, 2).contiguous().view(bh, seq_len, seq_len)

    # FP16 GEMM – Tensor‑Core accelerated
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))      # [bh, N, N] FP16

    # Cast to FP32 for the following LayerNorm
    out_fp32 = out_mat.to(torch.float32)

    # Restore original layout: [B, N, N, hidden]
    out = out_fp32.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden]  (float32)

    # Apply output gate (broadcasted on the hidden axis)
    out_norm = out_norm * out_gate

    # Final projection back to the original `dim`
    result = torch.nn.functional.linear(out_norm, weights["to_out.weight"])   # float32

    return result
