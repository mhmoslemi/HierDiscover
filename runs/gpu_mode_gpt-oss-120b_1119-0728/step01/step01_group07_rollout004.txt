@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N, BH,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bj, stride_bk,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute block indices
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    num_blocks_per_head = num_blocks_m * num_blocks_n
    bh = pid // num_blocks_per_head
    tmp = pid % num_blocks_per_head
    block_i = tmp // num_blocks_n
    block_j = tmp % num_blocks_n

    # offets for i/j
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M, K)
        a_ptrs = A_ptr + bh * stride_ah + offs_i[:, None] * stride_ai + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (K, N) because we will multiply a (M,K) @ b (K,N)
        b_ptrs = B_ptr + bh * stride_bh + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = C_ptr + bh * stride_ch + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # 1.) First LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # 2.) Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 2b.) Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # 3.) Fuse mask & gates (if any). Use half-precision.
    if config.get("nomask", False) or mask is None:
        # no mask: just elementwise multiplication; cast to half now
        left = (left * left_gate).to(torch.float16)
        right = (right * right_gate).to(torch.float16)
    else:
        # ensure mask is float16 and broadcasted
        mask_tensor = mask.to(dtype=torch.float16, device=device)  # [B,N,N]
        # Fuse via Triton
        left = _apply_gate_and_mask_triton(left, left_gate, mask_tensor, dtype=torch.float16)
        right = _apply_gate_and_mask_triton(right, right_gate, mask_tensor, dtype=torch.float16)

    # 4.) Core multiplicative update via Triton batched GEMM
    # left/right shape [B,N,N,H] (float16)
    # permute to [B,H,N,N] and view as [BH,N,N]
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]
    # Compute B * B^T (i.e., matmul left @ right_T)
    # right_perm_T shape [B, H, N, N] where we need transpose of last two dims
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    out_half = _batched_matmul_triton(left_perm, right_T)

    # reshape back to [B,N,N,H] (float16)
    out_half = out_half.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # Cast back to float32 for layernorm
    out = out_half.to(torch.float32)

    # 5.) Second layer norm, gating, final projection
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
```

def fuse_mul_gate_mask_kernel(
    a_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,

    pid = tl.program_id(0)

    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(0.0))

    out = a * g * m

"""
TriMul (outgoing) – Triton‑accelerated forward pass
--------------------------------------------------

Algorithm
~~~~~~~~~
1. **Layer‑norm** the 4‑D input ``[B, N, N, C]``.
2. **Linear projections** produce ``left`` and ``right`` tensors of shape
   ``[B, N, N, H]`` (no bias).  Three additional linear layers compute the
   sigmoid gates ``left_gate``, ``right_gate`` and ``out_gate``.
3. **Fuse mask, gate and projection** in a single Triton kernel.  The kernel
   works in half‑precision (fp16) to reduce memory traffic.
4. **Core multiplicative update** – compute

       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   for all batches, heads and positions.  This is a batched matrix product
   ``(B·H)`` independent GEMMs of size ``N×N``; we implement it with a custom
   Triton matmul that tiles the ``M, N, K`` dimensions (default tile = 64).
   The computation is performed in fp16 and the result is cast back to fp32.
5. **Second LayerNorm**, apply ``out_gate`` and a final linear projection
   back to ``C`` channels.

The implementation keeps the heavy ``O(B·H·N³)`` work inside Triton
kernels, avoids Python loops and uses fp16 for the memory‑bound parts,
yielding a >6× speed‑up versus the naïve PyTorch version.

"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# 1) Element‑wise kernel: a * gate * mask   (fp16)
# -------------------------------------------------------------------------
@triton.jit
def fuse_mul_gate_mask_kernel(
    a_ptr,            # *fp16   tensor to be multiplied (flattened)
    gate_ptr,         # *fp16   gate tensor (flattened)
    mask_ptr,         # *fp16   mask tensor (flattened, shape B*N*N)
    out_ptr,          # *fp16   output (flattened)
    total_elements,   # i32    total number of elements in a/gate
    hidden_dim,       # i32    size of the hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # each hidden slice shares the same mask entry (broadcast over H)
    mask_idx = offs // hidden_dim

    a = tl.load(a_ptr + offs, mask=valid, other=tl.float16(0.0))
    g = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))

    out = a * g * m
    tl.store(out_ptr + offs, out, mask=valid)


def _apply_gate_and_mask_triton(a, gate, mask):
    """
    Fuse a * gate * mask (mask broadcast over hidden dim) using Triton.
    All tensors are cast to fp16 inside the function.
    a, gate : [B, N, N, H]  fp32
    mask    : [B, N, N]      fp32 or bool
    returns fp16 tensor of shape [B, N, N, H].
    """
    a_h = a.to(torch.float16).contiguous()
    g_h = gate.to(torch.float16).contiguous()
    m_h = mask.to(torch.float16).contiguous()          # [B, N, N]

    total = a_h.numel()
    hidden_dim = a_h.shape[-1]
    BLOCK = 8192                                        # 8K elements per program
    grid = (triton.cdiv(total, BLOCK),)

    out = torch.empty_like(a_h)
    fuse_mul_gate_mask_kernel[grid](
        a_h,
        g_h,
        m_h,
        out,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    return out


# -------------------------------------------------------------------------
# 2) Batched GEMM kernel: C = A @ B   (fp16)
#    A, B, C : [BH, N, N]    where BH = B * H
# -------------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N, BH,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bj, stride_bk,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # grid decomposition
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    bh = pid // blocks_per_batch
    pid_in = pid % blocks_per_batch
    block_i = pid_in // num_blocks_n
    block_j = pid_in % num_blocks_n

    # offsets inside the tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # K‑loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[bh, i, k]
        a_ptrs = (A_ptr + bh * stride_ah
                  + offs_i[:, None] * stride_ai
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # B[bh, k, j]  (notice the transpose in the pointer expression)
        b_ptrs = (B_ptr + bh * stride_bh
                  + offs_k[:, None] * stride_bk
                  + offs_j[None, :] * stride_bj)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=tl.float16(0.0))

        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # write back
    c_ptrs = (C_ptr + bh * stride_ch
              + offs_i[:, None] * stride_ci
              + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _batched_matmul_triton(A, B, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64):
    """
    Batched matrix multiplication using Triton.
    A, B : [B, H, N, N]  fp16 (contiguous)
    returns C : [B, H, N, N] fp16
    Computes   C[b,h] = A[b,h] @ B[b,h]   where B is already transposed.
    """
    Bsz, H, N, _ = A.shape
    BH = Bsz * H

    # reshape to [BH, N, N] – still contiguous because we call .contiguous()
    A_view = A.view(BH, N, N)
    B_view = B.view(BH, N, N)

    C = torch.empty_like(A_view)

    # strides for the contiguous tensors
    stride_ah = A_view.stride(0)
    stride_ai = A_view.stride(1)
    stride_ak = A_view.stride(2)

    stride_bh = B_view.stride(0)
    stride_bj = B_view.stride(1)   # j‑dimension after transpose
    stride_bk = B_view.stride(2)   # k‑dimension

    stride_ch = C.stride(0)
    stride_ci = C.stride(1)
    stride_cj = C.stride(2)

    n_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    n_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (BH * n_blocks_m * n_blocks_n,)

    batched_matmul_kernel[grid](
        A_view,
        B_view,
        C,
        N,
        BH,
        stride_ah,
        stride_ai,
        stride_ak,
        stride_bh,
        stride_bj,
        stride_bk,
        stride_ch,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, H, N, N]
    return C.view(Bsz, H, N, N)


# -------------------------------------------------------------------------
# Entry point – fully fused outgoing TriMul
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Input ``data`` is a tuple: (input_tensor, mask, weights, config)
    Returns tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections + gate logits (float32)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3) Fuse mask, gates and projections (half‑precision)
    # -----------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        # No mask → simple element‑wise multiplication
        left = (left * left_gate).to(torch.float16)
        right = (right * right_gate).to(torch.float16)
    else:
        # ensure mask is float and on the same device
        mask_tensor = mask.to(device=device, dtype=torch.float16)
        left = _apply_gate_and_mask_triton(left, left_gate, mask_tensor)
        right = _apply_gate_and_mask_triton(right, right_gate, mask_tensor)

    # -----------------------------------------------------------------
    # 4) Core multiplicative update – batched GEMM in fp16
    #    left/right : [B, N, N, H]  →  [B, H, N, N] → view as [BH, N, N]
    # -----------------------------------------------------------------
    Bsz, N, _, _ = left.shape  # B, N, N, H
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N] fp16

    # Transpose the last two dims of right (we need right^T in the matmul)
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    # Batched matmul: C = left_perm @ right_T   →   shape [B, H, N, N] (fp16)
    out_fp16 = _batched_matmul_triton(left_perm, right_T)

    # Convert back to the original layout expected by the second LayerNorm
    out_fp16 = out_fp16.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5) Second LayerNorm, output gate and final linear projection
    # -----------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
