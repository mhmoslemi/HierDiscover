"""
TriMul (outgoing) forward pass – Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×dim}.
2. Project to hidden:  left = x·W_left , right = x·W_right   (no bias).
3. (Optional) mask the two projections.
4. Compute element‑wise gates with sigmoid:
       left_gate  = σ(x·W_left_gate)
       right_gate = σ(x·W_right_gate)
       out_gate   = σ(x·W_out_gate)
   Apply left/right gates:  left  ← left⋅left_gate,
                            right ← right⋅right_gate.
5. Cast left/right to FP16 and reshape to (B·H, N, N) where H=hidden_dim:
       left  : rows i, cols k   (A)
       right : rows k, cols j   (B) – a transpose on the last two dims.
   The heavy part is the batched matrix multiplication
       C[b] = A[b] @ B[b]      (C ∈ ℝ^{B·H×N×N})  accumulated in FP32.
   This is implemented with a custom Triton kernel.
6. Reshape C back to (B, N, N, H) and apply a second LayerNorm over H.
7. Multiply by out_gate (broadcasted over H) and linearly project back
   to the original dim.
All non‑trivial work (the N³ GEMM) runs on the GPU in FP16 with FP32
accumulation, which is the main performance bottleneck.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ B   (A: [M,K], B: [K,N]) for many batches
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_Ab, stride_Ai, stride_Ak,   # strides for A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, row(K), col(N))
    stride_Cb, stride_Ci, stride_Cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)   # block row
    pid_n = tl.program_id(axis=1)   # block column
    pid_b = tl.program_id(axis=2)   # batch*hidden index

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main loop over K dimension
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)   # FP16
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0) # FP16
        acc += tl.dot(a, b)    # FP16×FP16 → FP32

        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # --------------------------------------------------------------
    # Write back result
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all on CUDA)
            - config: dict with keys:
                * "dim" (int)                – input feature dimension
                * "hidden_dim" (int)         – hidden dimension for the mul
                * "nomask" (bool, optional) – if True, mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5  # matches torch.nn.LayerNorm default

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)     # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply elementwise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 and reshape for Triton GEMM
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to (B, hidden, N, N) and transpose right so that its
    # inner dimension is the K‑index of the multiplication.
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N] (i,k)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (k,j)

    # Collapse batch and hidden into a single GEMM batch dimension.
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)     # [BH, M, K]
    right_flat = right_perm.view(BH, N, N)   # [BH, K, N]

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # [BH, M, N]

    # ------------------------------------------------------------------
    # 6️⃣ Launch Triton kernel
    # ------------------------------------------------------------------
    # Tile sizes – these values give good occupancy on H100.
    BLOCK_M = 64   # rows per block
    BLOCK_N = 64   # cols per block
    BLOCK_K = 32   # reduction tile

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        BH,                             # batch*hidden dimension
    )

    # Strides in number of elements (not bytes)
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)   # K dimension
    stride_Bn = right_flat.stride(2)   # N dimension

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Number of warps – 8 is a sweet spot for H100 FP16 GEMM.
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Ci, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate (broadcast over the hidden dim)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    result = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return result
```
