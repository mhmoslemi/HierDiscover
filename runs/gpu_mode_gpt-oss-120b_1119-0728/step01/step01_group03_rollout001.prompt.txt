```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
TriMul forward (outgoing) implementation with a custom Triton kernel.

The operation can be written as:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

where left/right are linear projections of the normalized input and are
gated element‑wise.  For each batch and hidden channel this is a matrix
multiply L @ Rᵀ (size N×N), repeated `hidden_dim` times.  The heavy N³
computation is offloaded to a fused Triton GEMM kernel that processes a
batch*hidden slice at a time (B·hidden GEMMs in parallel).  The rest of
the pipeline (layer‑norms, linear projections, gating) stays in PyTorch.
The kernel works on FP16 inputs (to save memory and increase throughput)
and accumulates in FP32, producing an FP32 output that is later used by
the following LayerNorm and final linear projection.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr,                 # FP16 pointer, shape [batch*hidden, M, K]
    B_ptr,                 # FP16 pointer, shape [batch*hidden, K, N]  (Rᵀ)
    C_ptr,                 # FP32 pointer, shape [batch*hidden, M, N]
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,   # strides for A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, row(K), col(N))
    stride_Cb, stride_Ci, stride_Cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: [M,K], B: [K,N]) for many batches."""
    pid_m = tl.program_id(axis=0)          # block row
    pid_n = tl.program_id(axis=1)          # block column
    pid_b = tl.program_id(axis=2)          # batch*hidden identifier

    # ------------------- offsets in each dimension -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                     # FP16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                     # FP16
        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Store result (FP32)
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model weights (all torch.Tensors on CUDA)
            - config: dict with keys ['dim', 'hidden_dim', 'nomask' (bool)]

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------- unpack inputs ---------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------- layer‑norm on the input ---------------------
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------- linear projections (no bias) ---------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])   # [B,N,N, hidden]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # --------------------- optional mask ---------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)      # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------- gating (sigmoid) ---------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # --------------------- cast to FP16 for the heavy multiplication ---------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Re‑order to (B, hidden, N, N) and make contiguous
    # left  : (i, k)   -> rows i, cols k
    # right : (j, k)   -> we need (k, j) i.e. transpose the last two dims
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()          # [B, hidden, N, N]
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()        # [B, hidden, N, N] (k, j)

    # Flatten batch*hidden so that we can launch one GEMM per slice
    bh = bs * hidden_dim
    left_flat = left_perm.view(bh, seq_len, seq_len)                # [B*H, N, N]
    right_flat = right_perm.view(bh, seq_len, seq_len)              # [B*H, N, N]

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)     # [B*H, N, N]

    # --------------------- launch Triton kernel ---------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # number of blocks along M
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # number of blocks along N
        bh,                                   # batch*hidden dimension
    )

    # Strides (in elements) for the flattened tensors
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)    # K dimension stride
    stride_Bn = right_flat.stride(2)    # N dimension stride

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len, seq_len, seq_len,
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Ci, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------- reshape back to original layout ---------------------
    # out_flat  -> [B, hidden, N, N] -> permute to [B, N, N, hidden]
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # --------------------- final LayerNorm, gating, and linear ---------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate (element‑wise)
    out_norm = out_norm * out_gate

    # Final projection back to `dim`
    result = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return result
```
