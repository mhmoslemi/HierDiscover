import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward pass implemented with a mixed‑precision Triton batched GEMM.

Algorithm
---------
1. Layer‑norm the 4‑D input (B,N,N,D).
2. Linear projections → left/right tensors of shape (B,N,N,H) (H = hidden_dim).
3. (Optional) apply the pairwise mask.
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the same
   normalized input.
5. Cast left/right to fp16 and multiply by their respective gates (still fp16).
6. Permute tensors to shape (B,H,N,N) and flatten the batch/channel dimension
   → A, B of shape (G,N,N) where G = B·H.
7. Call a Triton kernel that performs a batched GEMM:
       C[g,i,j] = Σ_k A[g,i,k] * B[g,k,j]
   The kernel tiles over M,N,K (BLOCK_M×BLOCK_N×BLOCK_K) and accumulates in
   float32, storing the result in fp16.
8. Reshape C back to (B,H,N,N) → (B,N,N,H).
9. Layer‑norm over the hidden dimension, multiply by out_gate, and a final
   linear projection back to D.
All heavy O(N³) work (step 7) runs on Triton; the remaining point‑wise
operations use PyTorch (fp16 where possible). The final output is float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matrix multiplication:
        C[g,i,j] = Σ_k A[g,i,k] * B[g,k,j]

    Grid:
        (num_blocks_M, num_blocks_N, num_groups)
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_g = tl.program_id(2)          # group (batch * hidden)

    # Offsets for the tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for the current tile of A and B
    a_ptrs = a_ptr + pid_g * stride_a_batch \
                   + offs_m[:, None] * stride_a_m \
                   + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_g * stride_b_batch \
                   + offs_k[:, None] * stride_b_k \
                   + offs_n[None, :] * stride_b_n

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        # Load tiles (fp16) and mask out‑of‑bounds elements
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        # Dot‑product accumulation
        acc += tl.dot(a, b)          # a,b are fp16 → acc is fp32

        # Advance k‑dimension pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back result (cast to fp16)
    c_ptrs = c_ptr + pid_g * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # config
    dim = config["dim"]                     # input/ output channel dimension D
    hidden_dim = config["hidden_dim"]       # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name].to(device)

    # LayerNorm over last dimension (D)
    norm_weight = _w("norm.weight")
    norm_bias = _w("norm.bias")

    # Linear weights (no bias)
    left_proj_weight = _w("left_proj.weight")
    right_proj_weight = _w("right_proj.weight")
    left_gate_weight = _w("left_gate.weight")
    right_gate_weight = _w("right_gate.weight")
    out_gate_weight = _w("out_gate.weight")
    to_out_norm_weight = _w("to_out_norm.weight")
    to_out_norm_bias = _w("to_out_norm.bias")
    to_out_weight = _w("to_out.weight")   # shape (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: (B,N,N) → (B,N,N,1)
        mask_fp = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------
    # 4) Gates (float32)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Cast to fp16 and apply gates
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Prepare tensors for batched GEMM
    #    left: (B,N,N,H) → (B,H,N,N)  (i   k axes)
    #    right: (B,N,N,H) → (B,H,N,N) where we need (k j) ordering
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B,H,N,N)  (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B,H,N,N)  (k,j)

    B, H, N, _ = left_perm.shape
    groups = B * H

    # Flatten batch&channel → (G,N,N)
    A = left_perm.view(groups, N, N)
    B_mat = right_perm.view(groups, N, N)

    # Output buffer (fp16)
    C = torch.empty_like(A, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7) Triton batched GEMM kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        groups,
    )

    _batched_matmul_kernel[grid](
        A, B_mat, C,
        N, N, N,
        A.stride(0), A.stride(1), A.stride(2),   # a batch, m, k
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),   # b batch, k, n
        C.stride(0), C.stride(1), C.stride(2),   # c batch, m, n
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to (B,N,N,H) and cast to fp32 for remaining ops
    # ------------------------------------------------------------------
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    out_hidden = out_hidden.to(torch.float32)

    # ------------------------------------------------------------------
    # 9) Output layer‑norm, gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out_hidden,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate               # element‑wise gating (fp32)

    # Final projection back to original dim D
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
