import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HEAD_DIM: tl.constexpr
):
    """Batched GEMM for TriMul.

    Computes for each (batch, head) pair:
        C[i, j] = Σ_k A[i, k] * B[k, j]
    where A and B are views of the left/right projected tensors.
    """
    pid_m = tl.program_id(0)  # rows (i)
    pid_n = tl.program_id(1)  # cols (j)
    pid_g = tl.program_id(2)  # combined (batch, head)

    # Decompose pid_g into batch and head indices.
    batch_idx = pid_g // HEAD_DIM
    head_idx = pid_g % HEAD_DIM

    # Tile offsets.
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute pointer base for this tile.
    a_ptrs = a_ptr + batch_idx * stride_a_batch \
                   + head_idx * stride_a_head \
                   + offs_i[:, None] * stride_a_i \
                   + offs_k[None, :] * stride_a_k

    b_ptrs = b_ptr + batch_idx * stride_b_batch \
                   + head_idx * stride_b_head \
                   + offs_k[:, None] * stride_b_k \
                   + offs_j[None, :] * stride_b_j

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        a = tl.load(a_ptrs,
                    mask=(offs_i < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_j < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)   # fp16 * fp16 -> fp32

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back result.
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
                   + head_idx * stride_c_head \
                   + offs_i[:, None] * stride_c_i \
                   + offs_j[None, :] * stride_c_j

    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(offs_i < M)[:, None] & (offs_j < N)[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 style).

    The heavy N³ work (∑ₖ left[i,k]·right[j,k]) is performed by a
    custom Triton batched‑GEMM kernel that works directly on the
    (B,N,N,hidden_dim) layout, thus avoiding costly permutes/copies.
    All remaining point‑wise ops (layer‑norm, gating, final projection)
    are executed with PyTorch in mixed precision.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config values.
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Helper to fetch weight tensors on the right device.
    def _w(name):
        return weights[name].to(device)

    # LayerNorm over input channel dimension D.
    norm_weight = _w("norm.weight")
    norm_bias = _w("norm.bias")

    # Linear projection weights (no bias).
    left_proj_weight = _w("left_proj.weight")
    right_proj_weight = _w("right_proj.weight")
    left_gate_weight = _w("left_gate.weight")
    right_gate_weight = _w("right_gate.weight")
    out_gate_weight = _w("out_gate.weight")

    # Output projection / post‑norm.
    to_out_norm_weight = _w("to_out_norm.weight")
    to_out_norm_bias = _w("to_out_norm.bias")
    to_out_weight = _w("to_out.weight")   # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1. Input LayerNorm.
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension.
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Optional pairwise mask.
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_fp = mask_tensor.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_fp
        right = right * mask_fp

    # --------------------------------------------------------------
    # 4. Compute sigmoid gates.
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 5. Cast to fp16 and apply per‑head gates.
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6. Batched GEMM (the O(N³) core).
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # (B,N,N,H)
    # Output buffer – same layout as the inputs.
    out_hidden = torch.empty_like(left, dtype=torch.float16, device=device)

    # Block sizes – tuned for H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (tiles_M, tiles_N, groups)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    # Strides (in elements).
    a_batch = left.stride(0)
    a_head = left.stride(3)
    a_i = left.stride(1)
    a_k = left.stride(2)

    b_batch = right.stride(0)
    b_head = right.stride(3)
    b_k = right.stride(2)   # row index (k)
    b_j = right.stride(1)   # column index (j)

    c_batch = out_hidden.stride(0)
    c_head = out_hidden.stride(3)
    c_i = out_hidden.stride(1)
    c_j = out_hidden.stride(2)

    _batched_matmul_kernel[grid](
        left,
        right,
        out_hidden,
        N,
        N,
        N,
        a_batch,
        a_head,
        a_i,
        a_k,
        b_batch,
        b_head,
        b_k,
        b_j,
        c_batch,
        c_head,
        c_i,
        c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HEAD_DIM=H,
    )

    # --------------------------------------------------------------
    # 7. Cast back to float32, apply post‑GEMM LayerNorm, gate and
    #    final projection.
    # --------------------------------------------------------------
    out_hidden = out_hidden.to(torch.float32)

    out = F.layer_norm(out_hidden,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate          # elementwise gating (float32)

    # Final linear projection back to original dimension D.
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out

"""
TriMul (outgoing) forward pass, highly optimized for H100.

Algorithm
---------
1. Layer‑norm the input 4‑D tensor (B,N,N,D).
2. Two linear projections → left/right tensors of shape (B,N,N,H).
3. (Optional) apply the pairwise mask.
4. Three sigmoid gates (left, right, out) are computed from the same
   normalized input.
5. Cast left/right to fp16, multiply by their respective gates (still fp16).
6. A custom Triton kernel performs a *batched* matrix‑multiply:
       C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   The kernel works directly on the (B,N,N,H) layout – no permutes,
   no copies – and accumulates in float32 before storing fp16.
7. Cast the result back to float32, apply a second LayerNorm over H,
   multiply by the out‑gate, and finally project back to D with a
   linear layer.
All heavy O(N³) work is done on the GPU; point‑wise ops stay in PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_head, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_head, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_head, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HEAD_DIM: tl.constexpr
):
    """Batched GEMM for a single (batch, head) pair.

    Computes
        C[i, j] = Σ_k A[i, k] * B[k, j]
    where A = left[b, i, k, h] and B = right[b, j, k, h].
    """
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_g = tl.program_id(2)          # combined (batch, head)

    # Decompose group id → batch and head.
    batch_idx = pid_g // HEAD_DIM
    head_idx  = pid_g %  HEAD_DIM

    # Tile offsets.
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for the current tile.
    a_ptrs = a_ptr + batch_idx * stride_a_batch \
                     + head_idx  * stride_a_head  \
                     + offs_i[:, None] * stride_a_i \
                     + offs_k[None, :] * stride_a_k

    b_ptrs = b_ptr + batch_idx * stride_b_batch \
                     + head_idx  * stride_b_head  \
                     + offs_k[:, None] * stride_b_k \
                     + offs_j[None, :] * stride_b_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        a = tl.load(a_ptrs,
                    mask=(offs_i < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
