import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy part of the TriMul operator is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix‑multiply:
    out[b,d] = left[b,d] @ right[b,d].T
We fuse this N³ operation across the batch⋅hidden_dim dimension with a
single Triton kernel (batched GEMM).  All other linear / LayerNorm /
sigmoid operations are performed with PyTorch for simplicity.

The kernel works on tensors of shape (B*H, N, N) where
    B – batch size,
    H – hidden_dim (output feature dimension of the projections),
    N – sequence length.
It computes C = A @ Bᵀ for each batch entry using a tiled
BLOCK_M×BLOCK_N×BLOCK_K algorithm and stores the result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A,B,C are (batch, M, K), (batch, K, N), (batch, M, N))."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ----------------------- offsets within a tile -----------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch (note: strides are *element* strides)
    a_base = a_ptr + batch_id * a_batch_stride
    b_base = b_ptr + batch_id * b_batch_stride
    c_base = c_ptr + batch_id * c_batch_stride

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------- loop over K dimension -----------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)     # [BLOCK_K]
        mask_k = offs_k < K

        # A tile: rows = offs_m, cols = offs_k
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B tile (already transposed in memory, i.e. B is K×N): rows = offs_k, cols = offs_n
        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # GEMM on the tile
        acc += tl.dot(a, b)   # acc ∈ FP32

    # ----------------------- write back -----------------------
    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    N³ einsum.  Arguments:

    data = (input_tensor, mask, weights, config)

    * input_tensor : [B, N, N, C] (float32)
    * mask          : [B, N, N] (bool or float; may be None)
    * weights       : dict of model parameters (see keys in the prompt)
    * config        : {'dim': C, 'hidden_dim': H, 'nomask': bool, ...}

    Returns:
        out : [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------- LayerNorm on input -----------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # ----------------------- Linear projections -----------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ----------------------- Optional mask -----------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(dtype=dtype).unsqueeze(-1)                # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ----------------------- Gating (sigmoid) -----------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))      # [B, N, N, H]

    left = left * left_gate
    right = right * right_gate

    # ----------------------- Triton batched GEMM (einsum) -----------------------
    # Re‑order tensors so that each (batch, hidden) pair becomes a separate batch for the kernel.
    B, N, _, _ = left.shape
    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: need k↔j transpose for the matmul -> (B, H, N, N) where dim2=k, dim3=j
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and hidden dimensions
    BH = B * hidden_dim
    left_flat = left_perm.reshape(BH, N, N)
    right_flat = right_perm.reshape(BH, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # Strides (in element units) required by the kernel
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)   # stride for M (row)
    a_stride_k = left_flat.stride(2)   # stride for K (col)

    b_batch_stride = right_flat.stride(0)
    b_stride_k = right_flat.stride(1)  # K dimension (row after transpose)
    b_stride_n = right_flat.stride(2)  # N dimension (col)

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------- Output LayerNorm, gating and final Linear -----------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, C]

    return out
