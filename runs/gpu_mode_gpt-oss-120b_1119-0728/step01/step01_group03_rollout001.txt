"""
TriMul (outgoing) forward pass – highly optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,D]).
2. Linear projections to the hidden dimension (no bias):
       left = x @ W_left   ,  right = x @ W_right
3. (Optional) apply the pair‑wise mask.
4. Compute three sigmoid‑gates from the same normalized input and
   multiply them element‑wise with left, right and later the output.
5. Fuse the N³ “tri‑multiply” as a batched matrix‑multiply:
       out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
   This is exactly a GEMM for every (batch, hidden) pair.
   A custom Triton kernel processes (B·H) independent GEMMs in parallel
   using 256×256 tiles and 64‑wide K‑tiles, leveraging Tensor‑cores
   (FP16 × FP16 → FP32 accumulation).
6. Layer‑norm the intermediate result (hidden dim), apply the output gate,
   and a final linear projection back to the original feature dimension.
All heavy arithmetic stays in FP16/FP32; the final tensor is FP32 as required.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ B   where
#   A : [batch*hidden, M, K]   (FP16)
#   B : [batch*hidden, K, N]   (FP16)   (right already transposed)
#   C : [batch*hidden, M, N]   (FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # problem size
    stride_Ab, stride_Ai, stride_Ak,  # A strides (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,  # B strides (batch, row(K), col(N))
    stride_Cb, stride_Ci, stride_Cn,  # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)    # block row
    pid_n = tl.program_id(1)    # block col
    pid_b = tl.program_id(2)    # batch*hidden index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks (mostly true for the test sizes; kept for safety)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile in A and B
    a_ptrs = A_ptr + pid_b * stride_Ab \
                   + offs_m[:, None] * stride_Ai \
                   + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + pid_b * stride_Bb \
                   + offs_k[:, None] * stride_Bk \
                   + offs_n[None, :] * stride_Bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_tiles):
        # Load a tile of A and B
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)          # FP16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)          # FP16
        # Compute one tile of the GEMM
        acc += tl.dot(a, b)   # FP16×FP16 → FP32

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Store the result tile
    c_ptrs = C_ptr + pid_b * stride_Cb \
                     + offs_m[:, None] * stride_Ci \
                     + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model parameters (all on CUDA)
            - config: dict with keys ['dim', 'hidden_dim', 'nomask' (bool)]

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (over the feature dimension)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → hidden dimension
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid), applied element‑wise
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 for the expensive GEMM
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Rearrange tensors to (batch*hidden, M, K) layout
    # ------------------------------------------------------------------
    # left: (B,N,N,H) → (B,H,N,N)   rows=i  cols=k
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    # right: (B,N,N,H) → (B,H,N,N) and transpose last two dims → (K=j)
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()

    bh = bs * hidden_dim                     # batch * hidden
    left_flat = left_perm.view(bh, seq_len, seq_len)   # [B*H, N, N]
    right_flat = right_perm.view(bh, seq_len, seq_len) # [B*H, N, N]

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty(
        (bh, seq_len, seq_len), dtype=torch.float32, device=device
    )

    # ------------------------------------------------------------------
    # 7️⃣ Launch the Triton batched GEMM
    # ------------------------------------------------------------------
    # Tuned tile sizes for H100 (Tensor‑core friendly)
    BLOCK_M = 256
    BLOCK_N = 256
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # blocks along M
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # blocks along N
        bh,                                   # batch*hidden dimension
    )

    # Strides are expressed in number of elements (not bytes)
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)   # K‑dimension stride
    stride_Bn = right_flat.stride(2)   # N‑dimension stride

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,
        seq_len,
        seq_len,
        stride_Ab,
        stride_Ai,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bn,
        stride_Cb,
        stride_Ci,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to the original layout
    # ------------------------------------------------------------------
    # out_flat : [B*H, N, N] → [B, H, N, N] → [B, N, N, H]
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Final LayerNorm, output gate, and projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    result = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return result
