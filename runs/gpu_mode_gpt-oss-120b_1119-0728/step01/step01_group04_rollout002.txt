C = A @ Bᵀ for each hidden channel d.
We fuse the N³ multiplication in a single Triton kernel; all
other operations (LayerNorms, linear projections, gating and masking)
are performed with Torch’s highly‑optimised kernels.

Key ideas
---------
* The hidden dimension is treated as an extra batch dimension:
      batch_h = B * hidden_dim
  so the kernel sees a tensor of shape (batch_h, N, N) for A and B.
* Inputs to the kernel are half‑precision; accumulation is FP32,
  which maps to Tensor‑Core execution on an H100.
* Several kernel configurations are autotuned; for typical
  sequence lengths (256‑1024) the kernel runs with 64‑128‑32
  tile sizes and 8 warps.
* Masking and gating are applied in FP32 before the cast to
  FP16 to avoid loss of precision.

The function `custom_kernel` receives a tuple
(input_tensor, mask, weights, config) and returns a float32
tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
# A, B and C are (batch_h, N, N) tensors (N = seq_len).  A and B are
# FP16, accumulation is FP32.  Each program processes a tile of (M,N)
# for a single batch element.
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        # Small tiles – good for very short sequences
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32},
            num_warps=4,
            num_stages=2,
        ),
        # Larger tiles – best on H100 for N ≥ 512
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32},
            num_warps=8,
            num_stages=2,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def batch_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B (original), C
    M, N, K,                           # all equal to seq_len
    stride_am, stride_ak, stride_ab,   # A: (row, col, batch) strides
    stride_bm, stride_bk, stride_bb,   # Bᵀ (transposed) strides
    stride_cm, stride_cn, stride_cb,   # C strides (same layout as A)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]
    A, B ∈ FP16, accumulator ∈ FP32.
    """
    batch_id = tl.program_id(0)    # batch index = B * hidden_dim
    pid_m    = tl.program_id(1)    # tile row index
    pid_n    = tl.program_id(2)    # tile col index

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the current A and Bᵀ tiles
    # ------------------------------------------------------------------
    a_ptrs = (
        a_ptr
        + batch_id * stride_ab
        + offs_m[:, None] * stride_am
        + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    )
    b_ptrs = (
        b_ptr
        + batch_id * stride_bb
        + (tl.arange(0, BLOCK_K)[:, None] * stride_bk)
        + offs_n[None, :] * stride_bm
    )

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # (BLOCK_M, BLOCK_K)  fp16
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # (BLOCK_K, BLOCK_N)  fp16

        # Dot product; accumulation happens in FP32 automatically
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back the result (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_id * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
                         "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dim
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim], FP32

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    # Shape after each linear: [B, N, N, hidden_dim]
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Gating pre‑activations (no bias)
    # --------------------------------------------------------------
    left_gate_pre = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate_pre = F.linear(x, weights["out_gate.weight"])

    # --------------------------------------------------------------
    # 4. Sigmoid gates (still FP32)
    # --------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()      # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 5. Mask handling
    # --------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones(
            (batch_size, seq_len, seq_len), dtype=x.dtype, device=device
        )
    else:
        mask_tensor = mask.to(x.dtype)

    mask_tensor = mask_tensor.unsqueeze(-1)  # [B, N, N, 1]

    # --------------------------------------------------------------
    # 6. Apply mask and gating
    # --------------------------------------------------------------
    # These ops are in FP32 – we cast to FP16 only right before the
    # heavy N³ multiplication.
    left = left_proj * mask_tensor * left_gate   # [B,N,N,hidden_dim]
    right = right_proj * mask_tensor * right_gate

    # --------------------------------------------------------------
    # 7. Cast to half‑precision for the N³ matmul
    # --------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # --------------------------------------------------------------
    # 8. Reshape so each hidden channel becomes a separate batch item
    # --------------------------------------------------------------
    # (B, hidden_dim, N, N) -> (batch_h, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = batch_size * hidden_dim
    left_h = left_h.view(batch_h, seq_len, seq_len)
    right_h = right_h.view(batch_h, seq_len, seq_len)

    # --------------------------------------------------------------
    # 9. Allocate output buffer (half) for the matmul result
    # --------------------------------------------------------------
    out_h = torch.empty_like(left_h, dtype=torch.float16)  # (batch_h, N, N)

    # --------------------------------------------------------------
    # 10. Extract strides needed by the kernel
    # --------------------------------------------------------------
    # A (left_h) strides: batch, row(i), col(k)
    stride_ab, stride_am, stride_ak = left_h.stride()
    # B (right_h) strides: batch, row(j), col(k)
    b_batch, b_row, b_col = right_h.stride()
    # For Bᵀ we expose the transposed layout:
    stride_bb = b_batch          # batch stride (unchanged)
    stride_bk = b_col            # stride of k dimension (= original col stride = 1)
    stride_bm = b_row            # stride of j dimension (= original row stride = N)

    # C (out_h) strides are identical to A's layout
    stride_cb, stride_cm, stride_cn = out_h.stride()

    # --------------------------------------------------------------
    # 11. Launch the Triton kernel
    # --------------------------------------------------------------
    grid = (
        batch_h,
        triton.cdiv(seq_len, 128),   # i‑dimension (M) tiles
        triton.cdiv(seq_len, 128),   # j‑dimension (N) tiles
    )
    batch_tri_mul_kernel[
        grid
    ](
        left_h,
        right_h,
        out_h,
        seq_len,
        seq_len,
        seq_len,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bm,
        stride_bk,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )

    # --------------------------------------------------------------
    # 12. Reshape back to [B, N, N, hidden_dim] and cast to FP32
    # --------------------------------------------------------------
    out = out_h.view(batch_size, hidden_dim, seq_len, seq_len)
    out = out.permute(0, 2, 3, 1).contiguous()      # [B, N, N, hidden_dim]
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 13. Output LayerNorm (over hidden_dim) + output gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 14. Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
