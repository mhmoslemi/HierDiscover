@triton.jit
def _batched_gemm_kernel_fp16(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute tile start offsets...
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_base = a_ptr + batch_id * a_batch_stride
    b_base = b_ptr + batch_id * b_batch_stride
    c_base = c_ptr + batch_id * c_batch_stride

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # GEMM using tensor cores
        # tl.dot expects either float16 or bfloat16 for inputs
        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_gemm_fp16(...):
    ... as above

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_fp16 = True

    # Cast tensors and weights to half if fp16
    dtype = torch.float16 if use_fp16 else torch.float32
    # Cast input
    x = input_tensor.to(dtype)

    # LayerNorm
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
        eps=eps,
    )

    # Linear projections (half precision)
    left = F.linear(x, weights["left_proj.weight"].to(dtype), bias=None)
    right = F.linear(x, weights["right_proj.weight"].to(dtype), bias=None)

    # Mask
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates (compute in half)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype), bias=None))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype), bias=None))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype), bias=None))

    left = left * left_gate
    right = right * right_gate

    # Permute for GEMM
    B, N, _, _ = left.shape
    # left: [B,N,N,H] -> [B,H,N,N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: need transpose j<->k
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.reshape(BH, N, N)
    right_flat = right_perm.reshape(BH, N, N)

    # Output buffer
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)   # store fp32

    # Strides
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)
    a_stride_k = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_stride_k = right_flat.stride(1)
    b_stride_n = right_flat.stride(2)

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # Kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64  # K tile aligning with half

    grid = (BH, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    _batched_gemm_fp16[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape output to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm in half
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
        eps=eps,
    )

    # gating
    out = out * out_gate  # out_gate half

    # final linear
    out = F.linear(out, weights["to_out.weight"].to(dtype), bias=None)

    # cast back to float32
    out = out.to(torch.float32)

    return out

"""
TriMul (outgoing) kernel using half‑precision fused GEMM.

The heavy tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is expressed as a batched matrix multiplication
    out = left @ rightᵀ
for every (batch, head) pair.  We cast the intermediate tensors to
float16, use a custom Triton kernel that loads tiles in FP16, performs
the multiplication with `tl.dot` (tensor‑core accelerated) and accumulates
in FP32.  All linear projections, LayerNorms and sigmoids are performed
with PyTorch but in FP16 to keep memory traffic low.  The final output
is returned in FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _batched_gemm_fp16(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ   (A,B: (batch,M,K),(batch,K,N))."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * a_batch_stride
    b_base = b_ptr + batch_id * b_batch_stride
    c_base = c_ptr + batch_id * c_batch_stride

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (FP16)
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (FP16). B is stored as K×N.
        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Tensor‑core GEMM – result is accumulated in FP32
        acc += tl.dot(a, b)

    # Store result (FP32)
    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused FP16
    Triton kernel for the N³ einsum.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict containing model parameters
            - config: dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    use_fp16 = True
    dtype = torch.float16 if use_fp16 else torch.float32

    # ------------------- LayerNorm on input (half) ------------------- #
    x = input_tensor.to(dtype)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
        eps=eps,
    )

    # ------------------- Linear projections (half) ------------------- #
    left = F.linear(x, weights["left_proj.weight"].to(dtype), bias=None)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"].to(dtype), bias=None)  # [B,N,N,H]

    # ------------------- Optional mask ------------------- #
    if mask is not None and not config.get("nomask", False):
        m = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------- Gating (half) ------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype), bias=None))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype), bias=None))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype), bias=None))

    left = left * left_gate
    right = right * right_gate

    # --------------- Prepare tensors for batched GEMM --------------- #
    B, N, _, _ = left.shape

    # left: (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: need (K,N) orientation -> (B, H, N, N) where dim2=k, dim3=j
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.reshape(BH, N, N)   # (BH, M, K)
    right_flat = right_perm.reshape(BH, N, N) # (BH, K, N)

    # Output buffer (FP32 to keep final precision)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Strides in element units
    a_batch_stride = left_flat.stride(0)
    a_stride_m = left_flat.stride(1)
    a_stride_k = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_stride_k = right_flat.stride(1)
    b_stride_n = right_flat.stride(2)

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # ------------ Triton kernel launch (FP16 GEMM) ------------ #
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # multiple of 16 → good for tensor cores

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_gemm_fp16[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------- Output LayerNorm & final linear ------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(dtype),
        bias=weights["to_out_norm.bias"].to(dtype),
        eps=eps,
    )
    out = out * out_gate  # out_gate is half
    out = F.linear(out, weights["to_out.weight"].to(dtype), bias=None)

    # Convert back to float32 as required by the API
    return out.to(torch.float32)
