"""
custom_kernel for the *outgoing* TriMul operation.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to FP16 and apply three linear projections:
      left_proj , right_proj  → hidden_dim
      left_gate , right_gate , out_gate  → hidden_dim   (no bias)
3. Fuse mask, sigmoid‑gate and projection in a single Triton kernel:
      out = proj * mask * sigmoid(gate_pre)
   The kernel works on the flattened tensor, computes the mask index by
   integer‑division and therefore does not need the mask to be expanded.
4. Treat each hidden channel as an independent batch and perform the
   expensive N³ contraction as a batched matrix multiplication
      out_{b,h} = left_{b,h} @ right_{b,h}ᵀ
   using `torch.bmm`, which on an H100 invokes the highly‑optimised
   cuBLAS Tensor‑Core GEMM (FP16 → FP32 accumulation).
5. Reshape back to [B,N,N,hidden_dim] and convert to FP32.
6. Apply a second LayerNorm, the output‑gate (sigmoid of out_gate_pre) and a
   final linear projection back to the original feature dimension.

Only the element‑wise mask‑gate‑projection step is written in Triton; the
major N³ GEMM is delegated to cuBLAS, which on H100 is far faster than a
generic Triton implementation for the large sequence lengths used in the
tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: fused mask, sigmoid‑gate and projection
# -------------------------------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    proj_ptr,          # half*  (B,N,N,hidden)
    gate_ptr,          # half*  (B,N,N,hidden)  – pre‑sigmoid
    mask_ptr,          # half*  (B,N,N)          – broadcasted
    out_ptr,           # half*  (same shape as proj)
    hidden_dim,        # int    – size of hidden dimension
    total_elements,    # int    – B*N*N*hidden
    BLOCK_SIZE: tl.constexpr,
):
    """
    out = proj * mask * sigmoid(gate_pre)

    All tensors are stored in row‑major order.
    `mask` is broadcasted across the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load projection and gate (half precision)
    proj = tl.load(proj_ptr + offs, mask=valid, other=tl.float16(0.0))
    gate_pre = tl.load(gate_ptr + offs, mask=valid, other=tl.float16(0.0))

    # Compute mask index = floor(idx / hidden_dim)  (broadcast)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=tl.float16(0.0))

    # Sigmoid (approximation using exp, works in FP16)
    gate = 1.0 / (1.0 + tl.exp(-gate_pre))

    out = proj * mask_val * gate
    tl.store(out_ptr + offs, out, mask=valid)


# -------------------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None, shape [B, N, N] (float32)
        - weights      : dict of torch.Tensor model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # ------------------------ 1. First LayerNorm ---------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------ 2. Cast to FP16 -----------------------------
    x_h = x.to(torch.float16)

    # ------------------------ 3. Linear projections -----------------------
    left_proj      = F.linear(x_h, weights["left_proj.weight"])      # (B,N,N,hidden)
    right_proj     = F.linear(x_h, weights["right_proj.weight"])
    left_gate_pre  = F.linear(x_h, weights["left_gate.weight"])
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"])
    out_gate_pre   = F.linear(x_h, weights["out_gate.weight"])   # (B,N,N,hidden)

    # ------------------------ 4. Prepare mask (FP16) -----------------------
    if mask is None or nomask:
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(torch.float16)

    # ------------------------ 5. Fuse mask + sigmoid gate (Triton) -------
    total = left_proj.numel()                      # B * N * N * hidden
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # In‑place: write results back into left_proj / right_proj
    proj_gate_mask_kernel[grid](
        left_proj,
        left_gate_pre,
        mask_h,
        left_proj,                       # output = same buffer
        hidden_dim,
        total,
        BLOCK_SIZE=BLOCK,
    )
    proj_gate_mask_kernel[grid](
        right_proj,
        right_gate_pre,
        mask_h,
        right_proj,
        hidden_dim,
        total,
        BLOCK_SIZE=BLOCK,
    )
    # free temporary gate tensors
    del left_gate_pre, right_gate_pre

    # ------------------------ 6. Batched GEMM (cuBLAS) --------------------
    # reshape to (B*hidden, N, N)
    left = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right_proj.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left = left.view(batch_h, N, N)                     # (B*H, N, N)
    right = right.view(batch_h, N, N)

    # C = A @ Bᵀ  (FP16 input, FP32 accumulation inside cuBLAS)
    out = torch.bmm(left, right.transpose(1, 2))       # (B*H, N, N)

    # reshape back to [B, N, N, hidden]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)                         # back to FP32 for LN

    # ------------------------ 7. Output gate ------------------------------
    out_gate = out_gate_pre.sigmoid()                   # (B,N,N,hidden)
    del out_gate_pre

    # ------------------------ 8. Second LayerNorm ------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------ 9. Final linear projection -----------------
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
