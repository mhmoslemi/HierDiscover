"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The dominant cost is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
for each hidden channel *d*.

We therefore:
  1. Apply LayerNorm, the four linear projections and the three sigmoid
     gates with PyTorch (highly‑optimised kernels).
  2. Fuse mask & gating into the projected tensors.
  3. Cast the tensors to FP16 and reshape them to
        (batch_h = B·hidden_dim, N, N)   # each hidden channel is a batch.
  4. Launch a custom Triton kernel that performs a *batched* GEMM
     (C = A @ Bᵀ) using 128×128 tiles and a 64‑wide K‑tile.
     The kernel works on FP16 inputs and accumulates in FP32 for
     numerical stability.
  5. Reshape the result back, apply the output gate, a second LayerNorm
     and the final linear projection.

The kernel is written for Triton 3.3.1 and targets an H100 (shared‑memory
size ≈ 100 KB), so the tile sizes are chosen to maximise occupancy while
keeping the shared‑memory footprint < 64 KB.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B half, C half, acc FP32)
# ----------------------------------------------------------------------
@triton.jit
def batch_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers to the three tensors
    M, N, K,                            # matrix dimensions (M=N=K=seq_len)
    stride_am, stride_ak, stride_ab,    # strides for A: (row, col, batch)
    stride_bm, stride_bk, stride_bb,    # strides for Bᵀ: (col, row, batch)
    stride_cm, stride_cn, stride_cb,    # strides for C: (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM:
        C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, j, k]
    with FP32 accumulation.
    """
    batch_id = tl.program_id(0)   # 0‑th axis -> (B·H) batch
    pid_m    = tl.program_id(1)   # block row
    pid_n    = tl.program_id(2)   # block col

    # ------------------------------------------------------------------
    # Tile offsets (global indices)
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the tiles of A and Bᵀ
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_ab \
                     + offs_m[:, None] * stride_am \
                     + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + batch_id * stride_bb \
                     + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) \
                     + offs_n[None, :] * stride_bm

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over the K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # (BLOCK_M, BLOCK_K) half
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # (BLOCK_K, BLOCK_N) half

        # Matrix‑multiply the tile, promoting operands to FP32
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write‑back
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cb \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gate pre‑activations
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # → [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])
    left_gate_pre = F.linear(x, weights["left_gate.weight"])
    right_gate_pre = F.linear(x, weights["right_gate.weight"])
    out_gate_pre = F.linear(x, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3. Sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate   = out_gate_pre.sigmoid()   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Optional mask (binary 0/1)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((batch_size, seq_len, seq_len),
                                 dtype=x.dtype, device=device)
    else:
        mask_tensor = mask.to(x.dtype)
    mask_tensor = mask_tensor.unsqueeze(-1)          # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5. Apply mask & gating (still in FP32)
    # ------------------------------------------------------------------
    left  = left_proj * mask_tensor * left_gate     # [B,N,N,hidden_dim]
    right = right_proj * mask_tensor * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to half for the GEMM and pack hidden dim into the batch axis
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute to (B, hidden_dim, N, N) and make contiguous so that the inner
    # dimensions are dense (row‑major layout expected by the kernel).
    left_h  = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    batch_h = batch_size * hidden_dim
    left_h  = left_h.view(batch_h, seq_len, seq_len)    # (B·H, N, N)
    right_h = right_h.view(batch_h, seq_len, seq_len)

    # ------------------------------------------------------------------
    # 7. Allocate the output buffer (half)
    # ------------------------------------------------------------------
    out_h = torch.empty_like(left_h, dtype=torch.float16)   # (B·H, N, N)

    # ------------------------------------------------------------------
    # 8. Extract strides for the kernel (all tensors are row‑major)
    # ------------------------------------------------------------------
    stride_ab, stride_am, stride_ak = left_h.stride()      # (batch, row, col)
    # right_h strides: (batch, row, col)
    stride_bb, stride_bm, stride_bk = right_h.stride()
    # For Bᵀ we need the stride for the k‑dimension (original col) and the
    # stride for the j‑dimension (original row).
    # In the kernel we treat B as transposed, i.e. we read B[j,k].
    # Hence:
    #   stride_bk -> stride for k (original col)
    #   stride_bm -> stride for j (original row)
    # The batch stride stays unchanged.
    stride_cb, stride_cm, stride_cn = out_h.stride()

    # ------------------------------------------------------------------
    # 9. Launch the GEMM kernel
    # ------------------------------------------------------------------
    # Tile size selection – 128×128 tiles give good occupancy for
    # seq_len up to 1024 on an H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        batch_h,                              # batch dimension (B·H)
        triton.cdiv(seq_len, BLOCK_M),        # rows
        triton.cdiv(seq_len, BLOCK_N),        # cols
    )
    batch_gemm_kernel[grid](
        left_h,
        right_h,
        out_h,
        seq_len,                               # M
        seq_len,                               # N
        seq_len,                               # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bm,
        stride_bk,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Reshape back to the original 4‑D layout and convert to FP32
    # ------------------------------------------------------------------
    out = out_h.view(batch_size, hidden_dim, seq_len, seq_len)   # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()                  # (B, N, N, H)
    out = out.to(torch.float32)                                 # back to FP32

    # ------------------------------------------------------------------
    # 11. Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate    # broadcast over the hidden dimension

    # ------------------------------------------------------------------
    # 12. Final projection to original model dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
```
