@triton.jit
def _batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_hidden_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_hidden_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch index
    b = pid // hidden_dim
    h = pid % hidden_dim

    # base pointers
    a_base = a_ptr + b * a_batch_stride + h * a_hidden_stride
    b_base = b_ptr + b * b_batch_stride + h * b_hidden_stride
    c_base = c_ptr + pid * c_batch_stride

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Outgoing TriMul forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and fuse the 5 linear maps (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single mat‑mul.
3. Apply the optional pairwise mask, sigmoid gates and element‑wise
   gating (still in fp16).
4. Perform the heavy O(N³) contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   as a batched GEMM using a Triton kernel that works directly on the
   original (B,N,N,H) layout – no large permute / copy is performed.
5. Reshape the result back to (B,N,N,H) and apply the second LayerNorm,
   the output gate and the final linear projection to the original
   channel dimension C.
All heavy arithmetic is done in fp16 with fp32 accumulation; the final
output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _batched_gemm_fp16_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_hidden_stride, a_stride_m, a_stride_k,
    b_batch_stride, b_hidden_stride, b_stride_k, b_stride_n,
    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K, hidden_dim,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ
    A : (batch, H, M, K)   – stored as (B, N, N, H) in memory,
                            accessed with (batch, hidden) → (i,k)
    B : (batch, H, K, N)   – stored as (B, N, N, H) but with the two
                            inner N‑axes swapped so that rows are K.
    C : (batch*H, M, N)   – output in FP32.
    All pointers are element‑wise (not byte) offsets.
    """
    pid = tl.program_id(0)          # combined batch*hidden index
    pid_m = tl.program_id(1)        # tile row
    pid_n = tl.program_id(2)        # tile column

    # -------------------------------------------------
    # split combined index into batch & hidden indices
    # -------------------------------------------------
    b = pid // hidden_dim
    h = pid % hidden_dim

    # base pointers for this (batch, hidden) pair
    a_base = a_ptr + b * a_batch_stride + h * a_hidden_stride
    b_base = b_ptr + b * b_batch_stride + h * b_hidden_stride
    c_base = c_ptr + pid * c_batch_stride   # contiguous batch stride for C

    # tile offsets in the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (offs_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # fp16

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + (offs_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # fp16

        # Tensor‑core dot (fp16 → fp32)
        acc += tl.dot(a, b)

    # write back FP32 result
    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        weights      : dict of parameters with keys:
                       "norm.weight", "norm.bias",
                       "left_proj.weight", "right_proj.weight",
                       "left_gate.weight", "right_gate.weight", "out_gate.weight",
                       "to_out_norm.weight", "to_out_norm.bias",
                       "to_out.weight"
        config       : dict containing at least:
                       "dim" (C), "hidden_dim" (H), "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C] (float32)
    """
    # -------------------------------------------------
    # unpack arguments
    # -------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                               # [B, N, N, C]  float32

    # -------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # -------------------------------------------------
    x_h = x.to(torch.float16)       # [B, N, N, C]  fp16

    # -------------------------------------------------
    # 3. Fuse the five linear maps (proj + 3 gates) into one matmul
    # -------------------------------------------------
    # concatenate weight matrices along the output‑feature dimension
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)            # shape (5*H, C)

    fused = F.linear(x_h, fused_weight)          # [B, N, N, 5*H]  fp16

    # slice the fused tensor
    h = hidden_dim
    left = fused[..., :h]                         # [B,N,N,H]
    right = fused[..., h:2 * h]
    left_gate = fused[..., 2 * h:3 * h]
    right_gate = fused[..., 3 * h:4 * h]
    out_gate = fused[..., 4 * h:]                 # [B,N,N,H]

    # -------------------------------------------------
    # 4. Apply mask (still fp16) and gates
    # -------------------------------------------------
    if mask is not None and not nomask:
        # mask may be bool or float; bring to fp16 and broadcast
        mask_h = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5. Heavy O(N³) contraction via a Triton batched GEMM
    # -------------------------------------------------
    B, N, _, _ = left.shape

    # Strides of the (B,N,N,H) tensors (element strides, not byte strides)
    a_batch_stride = left.stride(0)      # distance between batches
    a_hidden_stride = left.stride(3)     # distance between hidden channels (==1)
    a_stride_m = left.stride(1)          # stride for axis i
    a_stride_k = left.stride(2)          # stride for axis k

    b_batch_stride = right.stride(0)
    b_hidden_stride = right.stride(3)
    b_stride_k = right.stride(2)         # row dimension (k) after transposition
    b_stride_n = right.stride(1)         # column dimension (j)

    # Output buffer: shape (B*H, N, N), FP32 accumulator as in the reference
    BH = B * hidden_dim
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    c_batch_stride = out_flat.stride(0)
    c_stride_m = out_flat.stride(1)
    c_stride_n = out_flat.stride(2)

    # Tile sizes – tuned for H100 Tensor Core usage
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_fp16_kernel[grid](
        left,
        right,
        out_flat,
        a_batch_stride,
        a_hidden_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_hidden_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,                # M
        N,                # N
        N,                # K
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 6. Second LayerNorm (float32)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                               # [B, N, N, H]  float32

    # -------------------------------------------------
    # 7. Output gate and final projection
    # -------------------------------------------------
    out_gate_f32 = out_gate.to(dtype=torch.float32)
    out = out * out_gate_f32

    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, C]  float32

    return out
