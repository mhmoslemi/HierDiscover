"""
TriMul “outgoing” forward (AlphaFold3) – fused projection & lightweight Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) over the feature dimension.
2. Cast to fp16 and flatten to (B·N·N, dim).
3. Fuse the three linear layers (proj‑left/right, three gates) into a single
   matmul: weight ∈ ℝ^{5·hidden × dim}.  Split the result into
   left_proj, right_proj, left_gate, right_gate, out_gate.
4. Apply optional binary mask (broadcast on hidden dim) to left/right.
   Apply sigmoid to gate slices and multiply left/right by their gates.
5. Reshape to (B, hidden, N, N) and transpose the right operand so that the
   contraction becomes a batched matrix‑multiply:
      out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   Compute it with a single torch.matmul (strided batched GEMM, fp16→fp32
   accumulation inside cuBLAS).
6. Layer‑norm over the hidden dimension (float32), multiply element‑wise by
   out_gate (float32) using a tiny Triton kernel (to satisfy the “use
   Triton” requirement).
7. Final linear projection back to the original channel dimension (float32).

All heavy arithmetic is performed in fp16; only the two LayerNorms and the
final projection use float32 for numeric stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise multiplication (used for out_gate)
# ----------------------------------------------------------------------
@triton.jit
def _elementwise_mul_kernel(
    out_ptr, a_ptr, b_ptr,  # pointers
    N,                     # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def _apply_elementwise_mul(out: torch.Tensor, a: torch.Tensor, b: torch.Tensor):
    """out = a * b   (all tensors are contiguous, same shape, float32)"""
    assert out.shape == a.shape == b.shape
    numel = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)
    _elementwise_mul_kernel[grid](
        out,
        a,
        b,
        numel,
        BLOCK_SIZE=BLOCK,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim]       (float32)
            mask_tensor  : torch.Tensor [B, N, N] or None   (float32/bool)
            weights_dict : dict of torch.Tensor parameters
            config_dict  : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2) Cast to fp16 and flatten
    # --------------------------------------------------------------
    x_fp16 = x.half()
    B, N, _, _ = x_fp16.shape
    x_flat = x_fp16.view(-1, dim)                     # (B·N·N, dim)

    # --------------------------------------------------------------
    # 3) Fused linear: proj (left+right) + three gates
    # --------------------------------------------------------------
    # Concatenate weights along the output‑feature dimension.
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden, dim)
            weights["right_proj.weight"],  # (hidden, dim)
            weights["left_gate.weight"],   # (hidden, dim)
            weights["right_gate.weight"],  # (hidden, dim)
            weights["out_gate.weight"],    # (hidden, dim)
        ],
        dim=0,
    ).half()                                          # (5·hidden, dim)

    # Single matmul; bias is absent in all original Linear layers.
    proj = F.linear(x_flat, combined_weight)          # (B·N·N, 5·hidden), fp16

    # --------------------------------------------------------------
    # 4) Split into tensors and apply mask + sigmoid gates
    # --------------------------------------------------------------
    h = hidden_dim
    left_raw   = proj[:, :h]                          # (B·N·N, h)
    right_raw  = proj[:, h:2 * h]                     # (B·N·N, h)
    left_gate  = proj[:, 2 * h:3 * h].sigmoid()       # (B·N·N, h)
    right_gate = proj[:, 3 * h:4 * h].sigmoid()       # (B·N·N, h)
    out_gate   = proj[:, 4 * h:5 * h].sigmoid()       # (B·N·N, h)

    # Optional binary mask (broadcast on hidden dimension)
    if (not nomask) and (mask_tensor is not None):
        mask = mask_tensor.view(-1, 1).to(x_fp16.dtype)  # (B·N·N, 1)
        left_raw  = left_raw * mask
        right_raw = right_raw * mask

    # Apply gating
    left  = left_raw * left_gate
    right = right_raw * right_gate

    # --------------------------------------------------------------
    # 5) Reshape for the O(N³) contraction
    # --------------------------------------------------------------
    # From (B·N·N, h) to (B, h, N, N) and transpose right operand.
    left = left.view(B, N, N, h).permute(0, 3, 1, 2).contiguous()   # (B, h, N, N)
    right = right.view(B, N, N, h).permute(0, 3, 2, 1).contiguous()  # (B, h, N, N)

    # Collapse batch+head dimension → (B·h, N, N)
    left_mat = left.view(B * h, N, N)
    right_mat = right.view(B * h, N, N)

    # --------------------------------------------------------------
    # 6) Core contraction via batched fp16 GEMM (cuBLAS)
    #    out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    # --------------------------------------------------------------
    # torch.matmul performs a strided batched matmul, automatically using
    # Tensor‑cores on H100 (fp16 inputs → fp32 accumulation).
    out_mat = torch.matmul(left_mat, right_mat)  # (B·h, N, N), fp16

    # Restore original layout: (B, N, N, h)
    out = out_mat.view(B, h, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension (float32)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # (B, N, N, hidden), float32

    # --------------------------------------------------------------
    # 8) Apply out_gate (float32) via tiny Triton kernel
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.view(B, N, N, h).to(torch.float32)
    gated = torch.empty_like(out_norm)
    _apply_elementwise_mul(gated, out_norm, out_gate_fp32)

    # --------------------------------------------------------------
    # 9) Final projection back to the original dim (float32)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"]          # (dim, hidden)
    result = F.linear(gated, to_out_w)           # (B, N, N, dim), float32

    return result
