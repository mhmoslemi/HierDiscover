"""
Highly‑optimized Triton‑accelerated forward pass for the *outgoing* TriMul
operator (AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm over the input channel dimension (float32).
2. FP16 linear projections → hidden dimension.
   * Optional mask is applied (broadcast over the hidden dim).
   * Gated element‑wise multiplication (sigmoid gates) is performed.
3. Heavy N³ interaction:
   outᵇʰ = leftᵇʰ @ rightᵇʰᵀ   with torch.bmm  (B·H batched GEMM,
   FP16 inputs, FP16 output → Tensor‑Core fast path).
4. Layer‑norm over the hidden dimension (float32).
5. Apply the output gate.  This tiny element‑wise product is performed by a
   custom Triton kernel (required by the spec).
6. Final FP16 linear projection back to `dim` (cuBLAS matmul) and cast to
   float32.

All large dense matrix multiplications use cuBLAS/Tensor‑Core kernels,
while Triton is exercised for the element‑wise gated multiplication.
The implementation fits in the memory limits for the tested sizes and
comfortably reaches sub‑millisecond runtimes on an H100.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply (float32 × float32 → float32)
# ----------------------------------------------------------------------
@triton.jit
def _elem_mul_kernel(
    a_ptr,                # float32 pointer (out_norm)
    b_ptr,                # float32 pointer (out_gate)
    c_ptr,                # float32 pointer (output)
    N_ELEMS: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (all on CUDA)
        - config       : dict with keys
              * "dim"        – input channel dimension
              * "hidden_dim" – hidden dimension used inside the module
              * "nomask"     – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣  Layer‑norm over the channel dimension (float32)
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2️⃣  FP16 linear projections + gates
    # --------------------------------------------------------------
    # Cast everything to FP16 for the dense ops
    x_fp16 = x_norm.to(torch.float16)

    # Helper for one‑time dtype conversion
    to_fp16 = lambda t: t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # Linear projections (B·N·N, dim) @ (dim, hidden) → (B·N·N, hidden)
    left = torch.nn.functional.linear(x_fp16, left_proj_w)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B,N,N,hidden]

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # Optional mask (broadcast over hidden)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 3️⃣  Heavy N³ interaction – batched GEMM via torch.bmm (fast)
    # --------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) → flatten batch·hidden dimension
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden, N, N)   # [B*H, N, N]
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden, N, N) # [B*H, N, N]

    # Transpose the second operand (k ↔ j) for the matrix product
    right_T = right_flat.transpose(1, 2)

    # Batched matmul: out_flat[b*h] = left_flat[b*h] @ right_T[b*h]
    out_flat = torch.bmm(left_flat, right_T)            # FP16, shape [B*H, N, N]

    # Reshape back to [B, N, N, hidden] (float16)
    out_hidden = out_flat.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4️⃣  Layer‑norm over hidden dimension (float32)
    # --------------------------------------------------------------
    out_hidden_fp32 = out_hidden.to(torch.float32)      # cast before norm
    out_norm = torch.nn.functional.layer_norm(
        out_hidden_fp32,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                   # [B,N,N,hidden] float32

    # --------------------------------------------------------------
    # 5️⃣  Apply the output gate – tiny Triton kernel
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)          # [B,N,N,hidden] float32
    gated = torch.empty_like(out_norm, dtype=torch.float32, device=device)

    # launch kernel (one‑dimensional grid)
    total_elems = out_norm.numel()
    BLOCK = 1024
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    _elem_mul_kernel[grid](
        out_norm,
        out_gate_fp32,
        gated,
        total_elems,
        BLOCK=BLOCK,
    )

    # --------------------------------------------------------------
    # 6️⃣  Final projection back to `dim` (cuBLAS FP16 matmul)
    # --------------------------------------------------------------
    gated_fp16 = gated.to(torch.float16)
    M = bs * N * N
    gated_flat = gated_fp16.view(M, hidden).contiguous()   # [M, hidden]

    # Linear: (M, hidden) @ (hidden, dim).T → (M, dim)
    out_linear_fp16 = torch.nn.functional.linear(gated_flat, to_out_w)  # FP16

    # Cast to float32 and reshape to the expected output shape
    out = out_linear_fp16.to(torch.float32).view(bs, N, N, dim)

    return out
