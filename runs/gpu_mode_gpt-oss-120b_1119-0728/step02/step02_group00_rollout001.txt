"""
TriMul “outgoing” forward pass (AlphaFold3) – mixed‑precision Triton implementation.
--------------------------------------------------------------------------------
The heavy part of the module is the three‑dimensional contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiply `left @ rightᵀ` for each hidden slice `d`.
We fuse this contraction in a custom Triton kernel that works on
FP16 data, accumulates in FP32 and exploits Tensor‑core friendly tile sizes.
All light‑weight operations (LayerNorm, linear projections, sigmoid gates,
final LayerNorm and linear) are performed in PyTorch (FP16 where possible)
and the output is cast back to FP32 as required by the API.

The kernel is launched with a grid of shape
  (⌈N/BLOCK_M⌉, ⌈N/BLOCK_N⌉, B*hidden_dim)
so each hidden slice of each batch is processed independently.
"""

import torch
import torch.nn.functional as F

import triton
import triton.language as tl


@triton.jit
def _trimmul_outgoing_kernel(
    a_ptr,            # left  [batch_total, N, N]  (row‑major)
    b_ptr,            # right [batch_total, N, N]  (row‑major, accessed transposed)
    c_ptr,            # out   [batch_total, N, N]
    stride_a_batch, stride_a_i, stride_a_k,   # A strides
    stride_b_batch, stride_b_k, stride_b_i,   # B strides (k as row, j as col)
    stride_c_batch, stride_c_i, stride_c_j,   # C strides
    N,                                      # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ   for each batch slice.
    A : (M, K)   = (BLOCK_M, BLOCK_K)
    B : (K, N)   = (BLOCK_K, BLOCK_N)   – loaded transposed from memory
    C : (M, N)   = (BLOCK_M, BLOCK_N)   – accumulated in FP32
    """
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # col tile
    pid_z = tl.program_id(2)   # batch * hidden slice

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A (M × K) – FP16
        a_ptrs = (
            a_ptr
            + pid_z * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # Load B (K × N) – transposed view (k as row, n as col)
        b_ptrs = (
            b_ptr
            + pid_z * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # GEMM – accumulate in FP32
        acc += tl.dot(a, b)

    # Write the result (FP16)
    c_ptrs = (
        c_ptr
        + pid_z * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype=torch.float32, CUDA
        - mask_tensor  : torch.Tensor, shape [B, N, N]  (optional, may be None)
        - weights_dict : mapping name → torch.Tensor (float32, CUDA)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    out : torch.Tensor, shape [B, N, N, dim], dtype=torch.float32, CUDA
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Shortcut for weight lookup
    def w(name):
        return weights[name]

    # --------------------------------------------------------------
    # 2️⃣ Input LayerNorm (keep FP32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=w("norm.weight"),
        bias=w("norm.bias"),
    )  # [B, N, N, dim], float32

    # --------------------------------------------------------------
    # 3️⃣ Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (no bias) – FP16
    # --------------------------------------------------------------
    left  = F.linear(x_h, w("left_proj.weight").to(torch.float16))   # [B, N, N, H]
    right = F.linear(x_h, w("right_proj.weight").to(torch.float16))

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)               # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 6️⃣ Gating (sigmoid of linear layers) – FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, w("left_gate.weight").to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, w("right_gate.weight").to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_h, w("out_gate.weight").to(torch.float16)))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Heavy contraction: batched GEMM left @ rightᵀ
    # --------------------------------------------------------------
    #   Shape transform: [B, N, N, H] → [B, H, N, N] → view as [B*H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_flat  = left_perm.view(batch_total, N, N)    # FP16
    right_flat = right_perm.view(batch_total, N, N)

    out_flat = torch.empty_like(left_flat)           # FP16 buffer for result

    # --------------------------------------------------------------
    # Triton launch configuration (tuned for H100 Tensor‑cores)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 32‑wide tiles map to Tensor‑core WMMA

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_total,               # one GEMM per (batch, hidden) slice
    )

    _trimmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for B (right) – note the transposed view
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] (still FP16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ Output processing – LayerNorm, gate, final linear (FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w("to_out_norm.weight").to(torch.float16),
        bias=w("to_out_norm.bias").to(torch.float16),
    )
    out = out * out_gate                               # element‑wise gate
    out = F.linear(out, w("to_out.weight").to(torch.float16))

    # --------------------------------------------------------------
    # 9️⃣ Cast back to FP32 (the public API expects float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)
