    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

The operator works on a tensor `x` of shape [B, N, N, C] (C = dim) and
produces a tensor of the same shape.  The computation is:

1️⃣ Layer‑norm over the channel dimension (float32).
2️⃣ Two linear projections (left/right) followed by element‑wise
   gating.  An optional mask is applied before gating.  All linear
   work is done in FP16 so that Tensor‑Core kernels are used.
3️⃣ Heavy N³ interaction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (B·hidden independent matrix multiplies of
   size N×N) and is implemented with a single Triton kernel that
   fuses the full batch dimension → maximal GPU utilization.
4️⃣ Layer‑norm over the hidden dimension, an output gate and a final
   linear projection back to `dim`.  The final projection is performed
   with Torch’s cuBLAS‑based `nn.functional.linear`, which is faster
   than a custom Triton matmul for this relatively small GEMM.

The kernel runs on H100 (Triton 3.3.1) and reaches sub‑millisecond
runtimes on the benchmark configurations.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul interaction
# ----------------------------------------------------------------------
@triton.jit
def _batched_trimul_kernel(
    A_ptr,                # fp16   [B*H, M, K]   (left)
    B_ptr,                # fp16   [B*H, K, N]   (rightᵀ)
    C_ptr,                # fp32   [B*H, M, N]   (output)
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,   # strides for A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, row(K), col(N))
    stride_Cb, stride_Ci, stride_Cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A:[M,K], B:[K,N]) for many batches."""
    pid_m = tl.program_id(axis=0)          # block row
    pid_n = tl.program_id(axis=1)          # block column
    pid_b = tl.program_id(axis=2)          # flattened (batch*hidden)

    # ------------------------------------------------------------------
    # Offsets in each dimension
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_ptrs = (A_ptr + pid_b * stride_Ab
              + offs_m[:, None] * stride_Ai
              + offs_k[None, :] * stride_Ak)
    b_ptrs = (B_ptr + pid_b * stride_Bb
              + offs_k[:, None] * stride_Bk
              + offs_n[None, :] * stride_Bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    num_blocks_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_blocks_k):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=tl.float16(0.0))
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=tl.float16(0.0))
        acc += tl.dot(a, b)                # fp16×fp16 → fp32
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr + pid_b * stride_Cb
              + offs_m[:, None] * stride_Ci
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

        left_flat,
        right_flat,
        out_flat,
        seq_len, seq_len, seq_len,
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Ci, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] (float32)
    out_hidden = (
        out_flat.view(bs, hidden_dim, seq_len, seq_len)
                .permute(0, 2, 3, 1)
                .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣  Layer‑norm over hidden, output gate, final projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]  (float32)

    # Fuse output gate (cast gate to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear back to `dim` – use cuBLAS which is faster than a custom kernel
    out_norm_fp16 = out_norm.to(torch.float16)                 # [B,N,N,H]
    out_fp16 = torch.nn.functional.linear(out_norm_fp16, to_out_w)  # [B,N,N,dim]

    # Cast back to float32 as required by the specification
    out = out_fp16.to(torch.float32)
    return out
