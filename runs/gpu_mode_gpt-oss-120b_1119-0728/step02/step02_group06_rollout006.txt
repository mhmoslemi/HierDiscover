def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps=1e-5

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"],
                                      bias=weights["norm.bias"], eps=eps)

    # 2. Cast to fp16
    x_h = x.half()

    # 3. Concatenate projection+gate weights
    left_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    out_gate_w = weights["out_gate.weight"].half()
    # linear
    left_cat = torch.nn.functional.linear(x_h, left_w) # shape (B,N,N,2*hidden)
    right_cat = torch.nn.functional.linear(x_h, right_w)
    # split
    left_proj = left_cat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    # 4. Mask
    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # 5. Gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # 6. Compute out = sum_k left[i,k] * right[j,k]
    # allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)
    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    # calculate launch grid
    num_batch = B * hidden_dim
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = num_batch * num_pid_m * num_pid_n
    grid = (total_programs,)

    # stride definitions
    stride_batch = left.stride(0)  # N*N*hidden_dim
    stride_i = left.stride(1)      # N*hidden_dim
    stride_k = left.stride(2)      # hidden_dim
    stride_h = left.stride(3)      # 1

    trimul_gemm_kernel[grid](
        left, right, out,
        stride_batch, stride_i, stride_k, stride_h,
        B, N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # 7. LayerNorm over hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    )

    # 8. Fuse final elementwise multiplication and projection
    # Option A: Use small kernel:
    out_gate_f32 = out_gate.float()
    # result = torch.empty((B,N,N,dim), dtype=torch.float32, device=device)
    # Use fused kernel:

        # strides
        out_norm.stride(0), out_norm.stride(1), out_norm.stride(2), out_norm.stride(3),
        out_gate_f32.stride(0), out_gate_f32.stride(1), out_gate_f32.stride(2), out_gate_f32.stride(3),
        weights["to_out.weight"].stride(0), weights["to_out.weight"].stride(1),
        B, N, dim, hidden_dim,
        # block sizes...
    )
    return result

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_batch, stride_i, stride_k, stride_h,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # number of blocks per dimension
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    # total blocks per (b,h) pair
    pid_bh = pid // (num_pid_m * num_pid_n)  # batch*hidden index
    pid_in = pid % (num_pid_m * num_pid_n)
    pid_m = pid_in // num_pid_n
    pid_n = pid_in % num_pid_n

    # compute batch & hidden indices
    bh = pid_bh
    b = bh // H
    h = bh % H

    # base pointers for this (b,h)
    left_base = left_ptr + b * stride_batch + h * stride_h
    right_base = right_ptr + b * stride_batch + h * stride_h
    out_base = out_ptr + b * stride_batch + h * stride_h

    # coordinate ranges
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        k_valid = (k_offset + k_range) < N

        # Load A tile (M x K)
        a_ptrs = left_base + offs_m[:, None] * stride_i + (k_offset + k_range)[None, :] * stride_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B tile (K x N) where B is transposed version of right
        b_ptrs = right_base + (k_offset + k_range)[:, None] * stride_k + offs_n[None, :] * stride_i
        b = tl.load(b_ptrs, mask=k_valid[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = out_base + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # 1 LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,),
                                      weight=weights["norm.weight"],
                                      bias=weights["norm.bias"],
                                      eps=eps)
    # cast to fp16
    x_h = x.half()

    # 2 Linear projections + gates
    left_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    left_cat = torch.nn.functional.linear(x_h, left_w)  # [B,N,N,2*hidden]
    right_cat = torch.nn.functional.linear(x_h, right_w)

    left_proj = left_cat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, weights["out_gate.weight"].half()))

    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Ensure tensors are contiguous (they are)
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # Allocate output buffer for tri-mul result
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_progs = B * hidden_dim * num_blocks_m * num_blocks_n
    grid = (total_progs,)

    # compute strides
    stride_batch = left.stride(0)   # N*N*H
    stride_i = left.stride(1)       # N*H
    stride_k = left.stride(2)       # H
    stride_h = left.stride(3)       # 1

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        stride_batch,
        stride_i,
        stride_k,
        stride_h,
        B,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 3. Output LayerNorm
    out_norm = torch.nn.functional.layer_norm(out,
                (hidden_dim,),
                weight=weights["to_out_norm.weight"],
                bias=weights["to_out_norm.bias"],
                eps=eps)

    # 4. Fuse elementwise multiplication with output gate
    out_gate_f32 = out_gate.float()
    gated = torch.empty_like(out_norm)  # float32
    total_elems = out_norm.numel()
    BLOCK = 8192
    grid_mul = (triton.cdiv(total_elems, BLOCK),)
    elementwise_mul_kernel[grid_mul](
        out_norm,
        out_gate_f32,
        gated,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # 5. Final linear projection
    result = torch.nn.functional.linear(gated, weights["to_out.weight"])

    return result

"""
TriMul (outgoing) forward pass with Triton‑accelerated N³ reduction.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and compute the two linear projections + their gates in a fused
   fashion (still in PyTorch, but without extra copies).
3. Apply the optional mask, then element‑wise gate the projections.
4. Compute the heavy N³ update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton batched GEMM kernel that works directly on the
   (B, N, N, H) tensors (no permute / view copies).  The kernel loads fp16
   tiles, accumulates in fp32 and writes the result in fp32.
5. Layer‑norm the hidden‑dim result.
6. Fuse the final element‑wise multiplication with the output gate using a tiny
   Triton kernel and then apply the final linear projection back to `dim`
   (implemented with PyTorch’s `linear`).

The Triton kernels replace the previous torch.bmm and the final element‑wise
multiply, providing a large speed‑up while keeping the implementation simple
and fully differentiable‑free (forward‑only).
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny element‑wise kernel (same as the given baseline)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                 # float32* – first operand (out_norm)
    b_ptr,                 # float32* – second operand (out_gate)
    c_ptr,                 # float32* – output
    size,                  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Batched GEMM kernel for the core TriMul reduction.
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,      # fp16* – left tensor (B,N,N,H)
    right_ptr,     # fp16* – right tensor (B,N,N,H)
    out_ptr,       # fp32* – output tensor (B,N,N,H)

    # strides of the *original* (B,N,N,H) layout
    stride_batch,  # B‑stride  (N*N*H)
    stride_i,      # stride over the first N dimension (N*H)
    stride_k,      # stride over the second N dimension (H)
    stride_h,      # stride over the hidden dimension (1)

    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,

    # tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1️⃣  Grid layout
    # --------------------------------------------------------------
    # number of tiles per matrix
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # each program works on one (b,h) pair and one (i‑tile, j‑tile)
    pid_bh = pid // (num_pid_m * num_pid_n)          # batch*hidden index
    pid_in = pid % (num_pid_m * num_pid_n)

    pid_m = pid_in // num_pid_n
    pid_n = pid_in % num_pid_n

    # recover b and h from the combined index
    b = pid_bh // H
    h = pid_bh % H

    # --------------------------------------------------------------
    # 2️⃣  Pointers for this (b,h)
    # --------------------------------------------------------------
    left_base  = left_ptr  + b * stride_batch + h * stride_h
    right_base = right_ptr + b * stride_batch + h * stride_h
    out_base   = out_ptr   + b * stride_batch + h * stride_h

    # --------------------------------------------------------------
    # 3️⃣  Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i dimension
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j dimension

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 4️⃣  Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_range = tl.arange(0, BLOCK_K)
        k_valid = (k_offset + k_range) < N

        # ---- A tile: left[b,i,k,h]  (M×K) -----------------------
        a_ptrs = left_base + offs_m[:, None] * stride_i + (k_offset + k_range)[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0).to(tl.float32)

        # ---- B tile: right_T[b,k,j,h] where right_T[k,j] = right[b,j,k,h]
        #      shape (K×N) ---------------------------------------
        b_ptrs = right_base + (k_offset + k_range)[:, None] * stride_k + offs_n[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ---- Accumulate -------------------------------------------------
        acc += tl.dot(a, b)          # (BLOCK_M,BLOCK_N) float32

    # --------------------------------------------------------------
    # 5️⃣  Write back
    # --------------------------------------------------------------
    c_ptrs = out_base + offs_m[:, None] * stride_i + offs_n[None, :] * stride_k
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point: the fully‑fused forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul module.
    Input:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]                # input channel dimension
    hidden_dim = config["hidden_dim"]  # projection dimension (always 128 in tests)
    nomask = config.get("nomask", False)

    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣  Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                            # [B, N, N, dim] (fp32)

    # ------------------------------------------------------------------
    # 2️⃣  Cast to fp16 – all heavy work lives in fp16
    # ------------------------------------------------------------------
    x_h = x.half()                               # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣  Fuse projection + gate linear layers
    # ------------------------------------------------------------------
    left_w  = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    out_gate_w = weights["out_gate.weight"].half()   # [hidden, dim]

    # Linear with concatenated weights → [B, N, N, 2*hidden]
    left_cat  = torch.nn.functional.linear(x_h, left_w)
    right_cat = torch.nn.functional.linear(x_h, right_w)

    # Split projection & gate
    left_proj = left_cat[..., :hidden_dim]                     # (B,N,N,hidden) fp16
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])     # (B,N,N,hidden) fp16
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # Output gate (used after the output LayerNorm)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))  # (B,N,N,hidden) fp16

    # ------------------------------------------------------------------
    # 4️⃣  Optional mask (applied before the gates, as in the reference)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)              # (B,N,N,1) fp16
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣  Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate          # (B,N,N,hidden) fp16
    right = right_proj * right_gate        # (B,N,N,hidden) fp16

    # Ensure contiguity for the Triton kernel (they are already contiguous,
    # but calling .contiguous() makes the intention explicit)
    left  = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6️⃣  N³ reduction via Triton GEMM kernel
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_progs = B * hidden_dim * num_blocks_m * num_blocks_n
    grid = (total_progs,)

    # Strides of the (B,N,N,hidden) layout
    stride_batch = left.stride(0)   # N*N*hidden
    stride_i     = left.stride(1)   # N*hidden
    stride_k     = left.stride(2)   # hidden
    stride_h     = left.stride(3)   # 1

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        stride_batch,
        stride_i,
        stride_k,
        stride_h,
        B,
        N,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣  Layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                            # (B,N,N,hidden) fp32

    # ------------------------------------------------------------------
    # 8️⃣  Fuse final element‑wise multiplication (out_norm * out_gate)
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.float()
    gated = torch.empty_like(out_norm)           # (B,N,N,hidden) fp32
    total_elements = out_norm.numel()
    BLOCK = 8192
    grid_mul = (triton.cdiv(total_elements, BLOCK),)
    elementwise_mul_kernel[grid_mul](
        out_norm,
        out_gate_f32,
        gated,
        total_elements,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 9️⃣  Final projection back to `dim`
    # ------------------------------------------------------------------
    result = torch.nn.functional.linear(gated, weights["to_out.weight"])

    return result
