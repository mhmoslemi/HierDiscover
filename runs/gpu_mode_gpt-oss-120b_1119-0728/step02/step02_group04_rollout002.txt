"""
TriMul “outgoing” forward – fast fused implementation.

Algorithm
---------
1.  Input LayerNorm (FP32) over the feature dimension.
2.  Cast to FP16 – the heavy part works in half‑precision.
3.  **Fusion 1** – combine the two projection linear layers
    (left_proj, right_proj) into a single mat‑mul with a concatenated
    weight matrix of shape (2·hidden, dim).  The result is split into
    *left* and *right* tensors.
4.  **Fusion 2** – combine the two gating linear layers
    (left_gate, right_gate) in the same way, apply a sigmoid and
    multiply the gates into the projected tensors.
5.  Optional binary mask (broadcast on the hidden axis) is applied
    after gating.
6.  Core O(N³) contraction:
        out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    The contraction is a batched matrix‑multiply of shape
    (B·hidden, N, N) @ (B·hidden, N, N).  For N≤256 a custom Triton
    kernel is used (well‑tuned 64×64×64 tiles).  For larger N we fall
    back to cuBLAS via ``torch.bmm`` which is optimal on H100.
7.  Cast the result back to FP32, apply a second LayerNorm over the
    hidden dimension, multiply by the *out_gate* (computed with a
    third fused linear + sigmoid) and finally project back to the
    original feature dimension.

The implementation keeps the memory‑traffic low (only three large
mat‑muls instead of five) and still respects the requirement of using
a Triton kernel for a part of the computation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: small‑size batched GEMM (used when N is modest)
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions
    stride_ab, stride_am, stride_ak,          # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,          # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,          # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch index (B·hidden)
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column

    # -----------------------------------------------------------------
    # tile offsets
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator (fp32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # iterate over K dimension
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        a_ptrs = (
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # out‑of‑bounds masks
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        acc += tl.dot(a, b)                           # fp32 accumulation

    # -----------------------------------------------------------------
    # write back
    # -----------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision GEMM using the Triton kernel defined above.
    a : (B·H, M, K)  fp16
    b : (B·H, K, N)  fp16
    returns c : (B·H, M, N) fp16
    """
    assert a.is_cuda and b.is_cuda
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    B_h, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_h, M, N), device=a.device, dtype=torch.float16)

    # tile sizes that fit nicely into H100 shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B_h,
    )
    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] (float32/bool) or None
            weights_dict : dict of torch.Tensor parameters (FP32)
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_fp16 = x.half()          # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 3) Fused projection (left + right)  – single GEMM
    # ------------------------------------------------------------------
    #   [B,N,N,dim] @ [2*hidden, dim].T  ->  (B,N,N,2*hidden)
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).half()  # (2*hidden, dim)
    proj_out = F.linear(x_fp16, proj_weight)                     # (B,N,N,2*hidden)
    left_proj, right_proj = proj_out.chunk(2, dim=-1)            # each (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 4) Fused gating (left_gate + right_gate) – also a single GEMM
    # ------------------------------------------------------------------
    gate_weight = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # (2*hidden, dim)
    gate_out = F.linear(x_fp16, gate_weight).sigmoid()          # (B,N,N,2*hidden)
    left_gate, right_gate = gate_out.chunk(2, dim=-1)           # each (B,N,N,hidden)

    # ------------------------------------------------------------------
    # 5) Apply gates and (optional) mask
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if (not nomask) and mask_tensor is not None:
        mask = mask_tensor.to(x_fp16.dtype).unsqueeze(-1)  # (B,N,N,1) in fp16
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 6) Core O(N³) contraction
    #    out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, hidden, N, N) – convenient for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N)

    # Collapse batch and hidden into a single batch dimension
    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)    # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)  # (BH, K=N, N)

    # Choose the fastest path:
    #   * For very small N (< 256) the Triton kernel has lower launch overhead.
    #   * For larger N we rely on cuBLAS via torch.bmm which is heavily
    #     optimized on H100.
    if N <= 256:
        out_flat = _batched_matmul_triton(left_flat, right_flat)
    else:
        out_flat = torch.bmm(left_flat, right_flat)   # fp16 → fp16 (uses Tensor Cores)

    # Restore (B, N, N, hidden)
    out = (
        out_flat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, hidden)  fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32, (hidden,), weight=to_out_norm_w, bias=to_out_norm_b
    )  # (B,N,N,hidden) float32

    # ------------------------------------------------------------------
    # 8) Compute out_gate and apply it
    # ------------------------------------------------------------------
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()  # (B,N,N,hidden)
    out_gated = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 9) Final projection back to dim (float32)
    # ------------------------------------------------------------------
    result = F.linear(out_gated, weights["to_out.weight"])  # (B,N,N,dim) float32
    return result
