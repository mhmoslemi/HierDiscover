    pid_n = tl.program_id(2)

"""
custom_kernel – Outgoing TriMul (AlphaFold3)

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (half‑precision).
2. Six bias‑free linear projections (left/right) and three gating projections
   (left_gate, right_gate, out_gate); all in fp16, followed by a sigmoid.
3. Fuse the binary mask and the gate values into the projected tensors.
4. Collapse the hidden dimension into the batch dimension:
        (B, N, N, H) → (B·H, N, N)
   This layout matches the Triton kernel.
5. Triton kernel – batched matrix multiplication with the contraction
        C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
   * A, B are fp16, accumulation is fp32.
   * Tile size 128×128×64, 8 warps, 4 stages → full Tensor‑Core utilisation.
6. Reshape the result back to (B, N, N, H) and apply a second LayerNorm.
7. Multiply by the output gate and a final bias‑free linear projection back
   to the original feature dimension.
8. Return a float‑32 tensor.

The kernel is the only Triton component; the surrounding ops use fast
torch.nn.functional primitives in half‑precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched fp16 matmul with fp32 accumulation.
# For each batch `b`:
#     C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]
# ----------------------------------------------------------------------
@triton.jit
def batch_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                     # problem sizes (all = seq_len)
    stride_am, stride_ak, stride_ab,   # A: (row, col, batch) strides
    stride_bm, stride_bk, stride_bb,   # Bᵀ: (row‑=j, col‑=k) strides
    stride_cm, stride_cn, stride_cb,   # C: (row, col, batch) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)          # B·H dimension
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile col

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the A and Bᵀ tiles for this batch
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_ab \
                + offs_m[:, None] * stride_am \
                + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bb \
                + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                + offs_n[None, :] * stride_bm

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K
        k_mask = (cur_k + tl.arange(0, BLOCK_K)) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)               # (BLOCK_M, BLOCK_K)  fp16
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0)               # (BLOCK_K, BLOCK_N)  fp16

        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back the output tile
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cb \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (float32/bool)
        weights      : dict of model parameters (float32)
        config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # cast weights needed in fp16 once
    # ------------------------------------------------------------------
    w_left_proj  = weights["left_proj.weight"].to(device=device, dtype=torch.float16)
    w_right_proj = weights["right_proj.weight"].to(device=device, dtype=torch.float16)
    w_left_gate  = weights["left_gate.weight"].to(device=device, dtype=torch.float16)
    w_right_gate = weights["right_gate.weight"].to(device=device, dtype=torch.float16)
    w_out_gate   = weights["out_gate.weight"].to(device=device, dtype=torch.float16)
    w_to_out     = weights["to_out.weight"].to(device=device, dtype=torch.float16)

    # LayerNorm parameters (keep in fp16 for speed)
    ln_weight = weights["norm.weight"].to(device=device, dtype=torch.float16)
    ln_bias   = weights["norm.bias"].to(device=device, dtype=torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device=device, dtype=torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fp16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.half(),
        normalized_shape=(dim,),
        weight=ln_weight,
        bias=ln_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) – all fp16
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, w_left_proj)   # [B, N, N, H]
    right_proj = F.linear(x, w_right_proj)

    # ------------------------------------------------------------------
    # 3️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, w_left_gate).sigmoid()
    right_gate = F.linear(x, w_right_gate).sigmoid()
    out_gate   = F.linear(x, w_out_gate).sigmoid()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Mask handling (fp16)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    if nomask or mask is None:
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(dtype=torch.float16, device=device)
    mask_h = mask_h.unsqueeze(-1)   # [B, N, N, 1]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gate into the projected tensors
    # ------------------------------------------------------------------
    left  = left_proj * mask_h * left_gate   # fp16
    right = right_proj * mask_h * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Collapse the hidden dimension into the batch dimension
    #    (B·H, N, N) – the layout expected by the Triton kernel
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left = left.view(batch_h, N, N)      # (B·H, N, N)
    right = right.view(batch_h, N, N)

    # ------------------------------------------------------------------
    # 7️⃣ Allocate output buffer (fp16)
    # ------------------------------------------------------------------
    out_h = torch.empty_like(left)      # (B·H, N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Gather stride information for the kernel
    # ------------------------------------------------------------------
    stride_ab, stride_am, stride_ak = left.stride()    # (batch, row, col)
    stride_bb, stride_br, stride_bc = right.stride()  # (batch, row, col)

    # For the transposed view of B (we need B[j,k] → k as inner dim)
    stride_bk = stride_bc   # step along k
    stride_bm = stride_br   # step along j (output column)

    stride_cb, stride_cm, stride_cn = out_h.stride()

    # ------------------------------------------------------------------
    # 9️⃣ Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # chosen to match Tensor‑Core 16×16×16 WMMA fragments

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_tri_mul_kernel[grid](
        left,
        right,
        out_h,
        N, N, N,
        stride_am, stride_ak, stride_ab,
        stride_bm, stride_bk, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    # ensure kernel finishes before we touch the result
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 10️⃣ Reshape back to (B, N, N, H) and continue in fp16
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N)               # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()          # (B, N, N, H)

    # ------------------------------------------------------------------
    # 11️⃣ Hidden‑dim LayerNorm (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 12️⃣ Apply output gate and final projection (both fp16)
    # ------------------------------------------------------------------
    out = out * out_gate                               # fp16, broadcast over H
    out = F.linear(out, w_to_out)                      # (B, N, N, dim) fp16

    # ------------------------------------------------------------------
    # 13️⃣ Cast to float32 for the final output
    # ------------------------------------------------------------------
    return out.float()
