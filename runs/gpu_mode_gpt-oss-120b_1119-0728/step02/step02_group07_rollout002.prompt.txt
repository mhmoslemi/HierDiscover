import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------

The module works on a 4‑D tensor `x` of shape [B, N, N, D] and computes

    out[b,i,j,:] = Σ_k left[b,i,k,:] · right[b,j,k,:]

with a per‑channel sigmoid gating and optional masking.
The heavy N³ reduction is executed as a batched half‑precision GEMM
C = A @ Bᵀ, where A and B have shape (B·H, N, N) (H = hidden_dim).
The rest of the network (layer‑norms, linear projections, sigmoids)
is performed in regular PyTorch – but we fuse two linear layers
(left_proj+left_gate, right_proj+right_gate) to cut the number of
matrix‑multiplications in half.  All matrix work is performed in FP16,
while accumulation uses FP32 for numerical stability.

Key speed‑ups
------------
*   Fused linear layers (proj + gate) → 4 matmuls → 6 matmuls.
*   All large matmuls run in half‑precision.
*   Triton GEMM uses larger tiles (128×128×32) → far fewer blocks.
*   Minimal memory traffic: tensors are permuted & made contiguous only
    once before the kernel launch.

The function `custom_kernel` follows the required signature.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ  (A,B : (batch, M, K), (batch, N, K))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                 # Pointers
    batch, M, N, K,                       # Dimensions (batch = B*H)
    stride_ab, stride_am, stride_ak,       # Strides for A
    stride_bb, stride_bn, stride_bk,       # Strides for B
    stride_cb, stride_cm, stride_cn,       # Strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM with FP32 accumulation.

    Each program instance computes a (BLOCK_M × BLOCK_N) tile of C for a
    particular batch index (pid_b).  The K‑dimension is tiled with BLOCK_K.
    """
    pid_b = tl.program_id(2)   # batch * hidden index
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col

    # Starting offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (BLOCK_M × BLOCK_K)
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )

        # Load Bᵀ tile (BLOCK_K × BLOCK_N)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
        )

        # Matrix multiply for this K‑block
        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_N) in FP32

    # Write back in FP16
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_c,
    )


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]   (float32)
            - mask_tensor : torch.Tensor [B, N, N]       (bool/float) – may be ignored
            - weights_dict: dict with all Linear/LayerNorm parameters (float32)
            - config_dict : contains "dim" and "hidden_dim" entries and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                 # [B, N, N, dim]  (fp32)

    # --------------------------------------------------------------
    # 2) Fuse projection + gate for left and right streams (half precision)
    # --------------------------------------------------------------
    # Prepare half‑precision copies of the weights
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                # (2*H, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                # (2*H, dim)

    # Work in fp16 after the first (stable) LayerNorm
    x_h = x.to(torch.float16)           # [B,N,N,dim] (fp16)

    # Left stream:   first H → proj, second H → gate pre‑sigmoid
    left_concat = F.linear(x_h, left_concat_w)   # [B,N,N,2*H] (fp16)
    left_proj = left_concat[..., :hidden_dim]    # [B,N,N,H]   (fp16)
    left_gate_pre = left_concat[..., hidden_dim:]  # [B,N,N,H] (fp16)

    # Right stream:  first H → proj, second H → gate pre‑sigmoid
    right_concat = F.linear(x_h, right_concat_w)   # [B,N,N,2*H] (fp16)
    right_proj = right_concat[..., :hidden_dim]    # [B,N,N,H]
    right_gate_pre = right_concat[..., hidden_dim:]  # [B,N,N,H]

    # --------------------------------------------------------------
    # 3) Out‑gate (keep in fp32 for later multiplication)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"])
    )   # [B,N,N,H] (fp32)

    # --------------------------------------------------------------
    # 4) Optional mask (broadcast on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype=left_proj.dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 5) Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)   # fp16
    right_gate = torch.sigmoid(right_gate_pre) # fp16

    left = left_proj * left_gate   # fp16
    right = right_proj * right_gate   # fp16

    # --------------------------------------------------------------
    # 6) Batched GEMM: out = left @ rightᵀ   (fp16 → fp32 accumulation)
    # --------------------------------------------------------------
    B, N, _, H = left.shape      # H == hidden_dim
    # (B, H, N, N) → (B*H, N, N) and ensure contiguous layout
    left_tile = (
        left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.float16)
    )
    right_tile = (
        right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.float16)
    )
    out_tile = torch.empty_like(left_tile, dtype=torch.float16)

    # Strides for the Triton kernel (in elements, not bytes)
    stride_ab = left_tile.stride(0)
    stride_am = left_tile.stride(1)
    stride_ak = left_tile.stride(2)

    stride_bb = right_tile.stride(0)
    stride_bn = right_tile.stride(1)
    stride_bk = right_tile.stride(2)

    stride_cb = out_tile.stride(0)
    stride_cm = out_tile.stride(1)
    stride_cn = out_tile.stride(2)

    # Tile sizes – tuned for H100 (good occupancy, modest register pressure)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch × hidden
    )

    # Launch kernel
    tri_mul_gemm_kernel[grid](
        left_tile,
        right_tile,
        out_tile,
        B * H,
        N,
        N,
        N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bn,
        stride_bk,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and promote to fp32 for the remaining ops
    out = (
        out_tile.view(B, H, N, N)
        .permute(0, 2, 3, 1)      # [B, N, N, H]
        .to(torch.float32)
    )

    # --------------------------------------------------------------
    # 7) Hidden‑dim LayerNorm
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, H]

    # --------------------------------------------------------------
    # 8) Apply out‑gate (already fp32)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9) Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
