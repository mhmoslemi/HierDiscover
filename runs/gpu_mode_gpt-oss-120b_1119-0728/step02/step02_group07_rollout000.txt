"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The kernel computes  

    out[b,i,j,:] = Σ_k left[b,i,k,:] · right[b,j,k,:]          (1)

where `left` and `right` are linear projections of the input tensor
followed by element‑wise sigmoid gates and an optional mask.
The N³ reduction is performed as a batched half‑precision GEMM  
C = A @ Bᵀ (A,B : (B·H, N, N)) with FP32 accumulation.
All cheap operations (LayerNorm, linear projections, sigmoids, and the final
projection) are kept in PyTorch, while the heavy N³ part is off‑loaded to a
custom Triton kernel.  The kernel uses large 256×256 tiles and 64‑element
K‑tiles, 8 warps per block and FP32 accumulation for numerical stability.
The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ   (A,B : (batch, M, K) , (batch, N, K))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    batch, M, N, K,                      # dimensions
    stride_ab, stride_am, stride_ak,     # A strides
    stride_bb, stride_bn, stride_bk,     # B strides
    stride_cb, stride_cm, stride_cn,     # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM with FP32 accumulation.

    Each program instance computes a (BLOCK_M × BLOCK_N) tile of the output
    matrix C for a particular batch index (pid_b).  The K‑dimension is tiled
    with BLOCK_K.
    """
    pid_b = tl.program_id(2)   # batch index = batch * hidden
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col

    # Tile base offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (BLOCK_M × BLOCK_K)
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )

        # Load Bᵀ tile (BLOCK_K × BLOCK_N)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
        )

        # Multiply‑accumulate (FP32)
        acc += tl.dot(a, b)

    # Write back as FP16
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_c,
    )


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict: dict with all Linear/LayerNorm parameters (float32)
            - config_dict : contains "dim", "hidden_dim" and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 – stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------
    # 2) Fuse proj + gate for left / right streams (FP16)
    # ------------------------------------------------------------------
    # Concatenate projection and gate weights for a single matmul each
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # Convert activations to FP16 after the first stable LayerNorm
    x_h = x.to(torch.float16)  # [B,N,N,dim] (fp16)

    # Linear → (B,N,N,2*H)
    left_concat = F.linear(x_h, left_concat_w)   # fp16
    right_concat = F.linear(x_h, right_concat_w)  # fp16

    # Split into proj and gate pre‑sigmoid parts
    left_proj = left_concat[..., :hidden_dim]          # [B,N,N,H]
    left_gate_pre = left_concat[..., hidden_dim:]       # [B,N,N,H]
    right_proj = right_concat[..., :hidden_dim]        # [B,N,N,H]
    right_gate_pre = right_concat[..., hidden_dim:]     # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Out‑gate (keep in FP32 for later multiplication)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4) Optional masking (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to same dtype as left_proj (fp16)
        mask = mask_tensor.to(dtype=left_proj.dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5) Apply sigmoid gates (FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)   # fp16
    right_gate = torch.sigmoid(right_gate_pre) # fp16

    left = left_proj * left_gate   # fp16
    right = right_proj * right_gate  # fp16

    # ------------------------------------------------------------------
    # 6) Batched GEMM: out = left @ rightᵀ   (FP16 inputs, FP32 accumulation)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Rearrange to (B*H, N, N) for the batched GEMM
    left_tile = (
        left.permute(0, 3, 1, 2)  # B, H, N, N
        .contiguous()
        .view(B * H, N, N)
        .to(torch.float16)
    )
    right_tile = (
        right.permute(0, 3, 1, 2)
        .contiguous()
        .view(B * H, N, N)
        .to(torch.float16)
    )
    out_tile = torch.empty((B * H, N, N), dtype=torch.float16, device=device)

    # Strides (in elements)
    stride_ab = left_tile.stride(0)
    stride_am = left_tile.stride(1)
    stride_ak = left_tile.stride(2)

    stride_bb = right_tile.stride(0)
    stride_bn = right_tile.stride(1)
    stride_bk = right_tile.stride(2)

    stride_cb = out_tile.stride(0)
    stride_cm = out_tile.stride(1)
    stride_cn = out_tile.stride(2)

    # Tile sizes – tuned for H100 (large tiles, 8 warps)
    BLOCK_M = 256
    BLOCK_N = 256
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch × hidden
    )

    # Launch the kernel
    tri_mul_gemm_kernel[
        grid,
        num_warps=8,
    ](
        left_tile,
        right_tile,
        out_tile,
        B * H,
        N,
        N,
        N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bn,
        stride_bk,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and promote to FP32 for the remaining ops
    out = (
        out_tile.view(B, H, N, N)
        .permute(0, 2, 3, 1)      # [B, N, N, H]
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 7) Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8) Apply out‑gate (still FP32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
