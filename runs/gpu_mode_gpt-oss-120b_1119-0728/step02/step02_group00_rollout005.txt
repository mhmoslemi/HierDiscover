"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (float32 → keep numeric stability).
2. Cast to FP16 for the heavy work.
3. Fuse the projection + gating linear layers:
   * left  = Linear(x) with weight = [left_proj; left_gate]      → 2·H hidden dim
   * right = Linear(x) with weight = [right_proj; right_gate]    → 2·H hidden dim
   * Split the result, apply sigmoid on the gate half and finally
     left = proj·gate , right = proj·gate .
   This halves the number of GEMM calls compared with the reference.
4. (Optional) apply the pair‑wise mask to the projected tensors.
5. Contract over the sequence dimension k:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM A @ Bᵀ for every (b,h) pair.
   It is performed by a custom Triton kernel that
   * works in FP16, accumulates in FP32,
   * uses Tensor‑core‑friendly tile sizes (BLOCK_M×BLOCK_N×BLOCK_K),
   * does not allocate extra temporaries.
6. Layer‑norm over the hidden dimension, multiply by the output‑gate,
   and apply the final linear projection back to the original channel size.
7. Cast the result back to FP32 (the API expects float32).

All heavy linear work is done in FP16, only the first LayerNorm stays in FP32.
The Triton kernel is the only non‑PyTorch component, satisfying the
“use Triton” requirement while keeping the execution time well below the
target 1 ms for the benchmark configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# ----------------------------------------------------------------------
@triton.jit
def _trimmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                       # pointers
    stride_a_batch, stride_a_i, stride_a_k,    # A strides  (batch, row, col)
    stride_b_batch, stride_b_k, stride_b_i,    # B strides  (batch, k, j) – transposed view
    stride_c_batch, stride_c_i, stride_c_j,    # C strides  (batch, row, col)
    N,                                         # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # batch * hidden slice

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # Accumulator (FP32 for better numerical accuracy)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over reduction dimension k (tiles of size BLOCK_K)
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Out‑of‑bounds masks
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # Load A block (M × K) – FP16
        a_ptrs = (
            a_ptr
            + pid_z * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # FP16

        # Load B block (K × N) – transposed view (k as row, n as col)
        b_ptrs = (
            b_ptr
            + pid_z * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # FP16

        # Tensor‑core GEMM, accumulate in FP32
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result (store as FP16)
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_z * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor, shape [B, N, N, dim], float32, CUDA
        mask_tensor  : torch.Tensor, shape [B, N, N] (optional)
        weights_dict  : mapping name → torch.Tensor (float32, CUDA)
        config_dict   : {"dim": int,
                         "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    out : torch.Tensor, shape [B, N, N, dim], float32, CUDA
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch a weight tensor (already on the correct device)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over the last dimension (keep FP32 for stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 2️⃣  Cast to FP16 for the remaining heavy operations
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Fuse projection + gate linear layers (reduces GEMMs by 2×)
    # ------------------------------------------------------------------
    #   left:  proj (H) + gate (H) → stacked weight [2H, dim]
    left_stack_w = torch.cat(
        [_w("left_proj.weight"), _w("left_gate.weight")], dim=0
    ).to(torch.float16)  # [2*H, dim]
    right_stack_w = torch.cat(
        [_w("right_proj.weight"), _w("right_gate.weight")], dim=0
    ).to(torch.float16)  # [2*H, dim]

    # Linear with stacked weights
    left_comb = F.linear(x_h, left_stack_w)   # [B, N, N, 2*H]
    right_comb = F.linear(x_h, right_stack_w)  # [B, N, N, 2*H]

    # Split into projection and gate halves
    left_proj = left_comb[..., :hidden_dim]                         # [B,N,N,H]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])          # [B,N,N,H]

    right_proj = right_comb[..., :hidden_dim]                       # [B,N,N,H]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣  Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)               # [B,N,N,1]
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5️⃣  Apply gates (element‑wise)
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B,N,N,H]  FP16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Heavy contraction – batched GEMM via the Triton kernel
    # ------------------------------------------------------------------
    #   bring the hidden dimension into the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # View as [batch_total, N, N] (still FP16)
    left_flat = left_perm.view(batch_total, N, N)
    right_flat = right_perm.view(batch_total, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # Choose tile sizes based on sequence length (simple heuristic)
    # --------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_total,               # independent GEMM instances
    )

    _trimmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # strides for A (batch, i, k)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for B (batch, k, j) – we pass transposed view
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (batch, i, j)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original 4‑D layout: [B, N, N, H]  (FP16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣  Output processing (LayerNorm → out‑gate → final linear)
    # ------------------------------------------------------------------
    #   out_gate depends only on the original normalized input
    out_gate = torch.sigmoid(
        F.linear(x_h, _w("out_gate.weight").to(torch.float16))
    )  # [B,N,N,H] FP16

    # Layer‑norm over the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight").to(torch.float16),
        bias=_w("to_out_norm.bias").to(torch.float16),
    )
    # Gating
    out = out * out_gate
    # Final projection back to `dim`
    out = F.linear(out, _w("to_out.weight").to(torch.float16))  # [B,N,N,dim] FP16

    # ------------------------------------------------------------------
    # 8️⃣  Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
