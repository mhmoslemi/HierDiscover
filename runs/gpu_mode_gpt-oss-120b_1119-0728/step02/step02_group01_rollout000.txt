"""
TriMul “outgoing’’ forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm on the last feature dimension (float32 → fp16).
2. Fuse the five linear transforms (left‑proj, right‑proj, three gates) into a
   single GEMM to minimise memory traffic.
3. Apply optional pair‑wise mask, sigmoid gates and element‑wise multiplication.
4. Core O(N³·H) operation:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication for each hidden channel.
   A custom Triton kernel performs the batched GEMM with fp16 I/O and fp32
   accumulation (tensor‑core friendly).
5. Output LayerNorm → out‑gate → final linear projection.
All intermediate tensors are kept in fp16; the final result is cast back to
float32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------
# Triton batched GEMM kernel (half‑precision ↔ fp32 accumulation)
# -------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix dimensions
    stride_ab, stride_am, stride_ak,     # strides for A: (batch, M, K)
    stride_bb, stride_bk, stride_bn,     # strides for B: (batch, K, N)
    stride_cb, stride_cm, stride_cn,     # strides for C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # -----------------------------------------------------------------
    # Program IDs – each (batch * hidden) instance gets its own 2‑D grid.
    # -----------------------------------------------------------------
    pid_b = tl.program_id(2)                 # batch dimension (B·H)
    pid_m = tl.program_id(0)                 # tile row
    pid_n = tl.program_id(1)                 # tile column

    # -----------------------------------------------------------------
    # Compute tile offsets.
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # -----------------------------------------------------------------
    # Accumulator in fp32 (numerical stability on large reductions).
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps.
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # A and B pointers for the current tile.
        a_ptrs = (
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Out‑of‑bounds masks.
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Load fp16 values, promote to fp32 for the dot.
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float16)

        # Tensor‑core dot product (fp32 accumulation).
        acc += tl.dot(a, b)

    # -----------------------------------------------------------------
    # Write the result back (fp16 output, fp32 → fp16 conversion).
    # -----------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision matrix multiplication using the Triton kernel above.

    Arguments
    ---------
    a : Tensor[batch*H, M, K] (fp16)
    b : Tensor[batch*H, K, N] (fp16)

    Returns
    -------
    c : Tensor[batch*H, M, N] (fp16)
    """
    assert a.is_cuda and b.is_cuda
    B_h, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_h, M, N), device=a.device, dtype=a.dtype)

    # Block sizes tuned for H100 fp16 tensor‑cores.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,   # program_id(0)
        (N + BLOCK_N - 1) // BLOCK_N,   # program_id(1)
        B_h,                            # program_id(2)
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# -------------------------------------------------------------
# Entry point – custom Triton‑accelerated TriMul operator
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.FloatTensor [B, N, N, C]
        - mask_tensor  : torch.FloatTensor [B, N, N] (or None)
        - weights_dict : mapping from weight name to torch.FloatTensor
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.FloatTensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # 0) Unpack arguments.
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32 → fp16 for the rest of the pipeline).
    # -----------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 2) Fuse the five linear transforms (proj + three gates) into one GEMM.
    # -----------------------------------------------------------------
    # Cast projection/gate weights once to fp16.
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5·H, C)

    # One large linear yields all five intermediate tensors in a single pass.
    fused = F.linear(x, fused_weight)  # (B, N, N, 5·H) fp16
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=-1
    )
    del fused, fused_weight, x  # free memory early

    # -----------------------------------------------------------------
    # 3) Optional mask (applied before gating).
    # -----------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # -----------------------------------------------------------------
    # 4) Apply sigmoid gates and fuse with the projections.
    # -----------------------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left_proj * left_gate   # (B, N, N, H) fp16
    right = right_proj * right_gate

    # Release temporaries that are no longer needed.
    del left_proj, left_gate, right_proj, right_gate

    # -----------------------------------------------------------------
    # 5) Core O(N³·H) operation implemented as a batched matrix multiply.
    #    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B, H, N, N) and (B, H, K, N) for GEMM.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) == (B,H,K,N)

    # Collapse batch & hidden dimensions → (B·H, N, N) each.
    BH = B * H
    left_flat = left_t.view(BH, N, N)
    right_flat = right_t.view(BH, N, N)

    # Batched GEMM (fp16 input → fp32 acc → fp16 output).
    out_flat = _batched_matmul(left_flat, right_flat)   # (B·H, N, N)

    # Reshape back to (B, N, N, H).
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Clean up large temporaries.
    del left, right, left_t, right_t, left_flat, right_flat, out_flat

    # -----------------------------------------------------------------
    # 6) Output LayerNorm (over hidden dimension).
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # -----------------------------------------------------------------
    # 7) Apply the output gate.
    # -----------------------------------------------------------------
    out = out * out_gate
    del out_gate

    # -----------------------------------------------------------------
    # 8) Final linear projection back to the original feature dimension.
    # -----------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    result = F.linear(out, to_out_w)   # (B, N, N, C) fp16

    # Cast back to float32 to match the reference implementation.
    return result.to(torch.float32)
