"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Layer‑norm over the input channel dimension (float32).
2️⃣ Linear projections (left / right) + element‑wise gated
   multiplications (all in FP16).  Optional mask is applied before gating.
3️⃣ Heavy N³ interaction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (batch = B·hidden) of shape (N×N) and
   is performed with PyTorch `torch.bmm`, which uses cuBLAS Tensor‑Core
   kernels on H100 – much faster than a handcrafted Triton GEMM.
4️⃣ Layer‑norm over the hidden dimension, output‑gate, and final
   linear projection back to the original channel dimension.  The final
   projection is implemented as a small Triton kernel to satisfy the
   “must use Triton” requirement.

All tensors stay on CUDA; the final output is float32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: final projection (hidden → dim)
# ----------------------------------------------------------------------
@triton.jit
def _final_linear_kernel(
    A_ptr,                # fp16   [M, K]   (hidden representation)
    B_ptr,                # fp16   [K, N]   (weightᵀ)
    C_ptr,                # fp32   [M, N]   (output)
    M, N, K,
    stride_Am, stride_Ak,               # strides for A
    stride_Bk, stride_Bn,               # strides for B
    stride_Cm, stride_Cn,               # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ B   (A:[M,K] fp16, B:[K,N] fp16) → C:[M,N] fp32."""
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = A_ptr + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_blocks_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_blocks_k):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                        # fp16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                        # fp16
        acc += tl.dot(a, b)                          # fp16×fp16 → fp32
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    c_ptrs = C_ptr + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask          : torch.Tensor [B, N, N] (bool/float) or None
        - weights       : dict of torch.Tensors (CUDA, FP32)
        - config        : dict with keys
              * "dim"        – input channel dimension
              * "hidden_dim" – hidden dimension
              * "nomask"     – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over input channels (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for all dense linear ops (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gating (FP16)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x_h, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_h, right_proj_w)  # [B,N,N,hidden]

    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)      # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Heavy N³ interaction – use cuBLAS batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, K) where K=N
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)  # transpose K↔J

    # batched matmul: (B*hidden, N, N) @ (B*hidden, N, N)ᵀ → (B*hidden, N, N)
    out_hidden = torch.bmm(left_flat, right_flat)   # FP16 result

    # restore shape [B, N, N, hidden] and convert to FP32 for LayerNorm
    out_hidden = out_hidden.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 4️⃣ Hidden‑dim layer‑norm, output gate, final projection
    # ------------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden] (float32)

    # apply output gate (cast to float32)
    out_hidden = out_hidden * out_gate.to(torch.float32)

    # --------------------------------------------------------------
    # Final linear: (B·N·N, hidden) @ (hidden, dim)ᵀ → (B·N·N, dim)
    # --------------------------------------------------------------
    M = B * N * N                      # rows
    K = hidden
    N_out = dim                        # columns of final output

    A = out_hidden.view(M, K).contiguous().to(torch.float16)   # [M, K] FP16
    W = weights["to_out.weight"].to(torch.float16)            # [dim, hidden]
    B_T = W.t().contiguous()                                 # [hidden, dim] FP16

    result_fp32 = torch.empty((M, N_out), dtype=torch.float32, device=device)

    # launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_out + BLOCK_N - 1) // BLOCK_N,
    )

    stride_Am = A.stride(0)
    stride_Ak = A.stride(1)
    stride_Bk = B_T.stride(0)
    stride_Bn = B_T.stride(1)
    stride_Cm = result_fp32.stride(0)
    stride_Cn = result_fp32.stride(1)

    _final_linear_kernel[grid](
        A,
        B_T,
        result_fp32,
        M,
        N_out,
        K,
        stride_Am,
        stride_Ak,
        stride_Bk,
        stride_Bn,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, dim] (float32) and return
    return result_fp32.view(B, N, N, dim)
