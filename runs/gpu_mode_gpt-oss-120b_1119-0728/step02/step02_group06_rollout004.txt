@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_abl, stride_am, stride_ak,
    stride_bbl, stride_bk, stride_bn,
    stride_cbl, stride_cm, stride_cn,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for region
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_batch_ptr = a_ptr + batch * stride_abl
    b_batch_ptr = b_ptr + batch * stride_bbl
    c_batch_ptr = c_ptr + batch * stride_cbl

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # offsets for K dimension
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        # mask for K dimension
        mask_k = offs_k < K

        # compute pointers for A tile
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        # compute pointers for B tile
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load with masks: combined masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy N³ part of the module is the Einstein sum

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batched matrix multiplication of
shape (B·D, N, N) @ (B·D, N, N)ᵀ for each hidden dimension *d*.
A custom Triton kernel computes all B·D independent GEMMs in one
launch, using fp16 data and fp32 accumulation.  The remaining
operations (LayerNorms, linear projections, sigmoids, masking and
the final linear projection) are kept in PyTorch, which already uses
high‑performance CUDA kernels.  The element‑wise multiplication of
the output‑norm with the output‑gate is also fused in a tiny Triton
kernel.

The function `custom_kernel` follows the exact semantics of the
reference PyTorch implementation while reducing the number of
memory round‑trips for the N³ bottleneck.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                # fp16*   – left tensor  [C, M, K]
    b_ptr,                # fp16*   – right tensor [C, K, N] (already transposed)
    c_ptr,                # fp16*   – output       [C, M, N]
    M, N, K,              # int32   – matrix dimensions (all equal to seq_len)
    stride_a_batch,       # int32   – stride to next batch in A
    stride_a_m, stride_a_k,
    stride_b_batch,       # int32   – stride to next batch in B
    stride_b_k, stride_b_n,
    stride_c_batch,       # int32   – stride to next batch in C
    stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for the TriMul contraction.
    Each program instance handles one (batch, block_row, block_col)
    tile of the output matrix C.
    """
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # ------------------------------------------------------------------
    # Pointers to the start of the batch
    # ------------------------------------------------------------------
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k   = k * BLOCK_K
        offs_k  = cur_k + tl.arange(0, BLOCK_K)
        mask_k  = offs_k < K

        # A tile: [BLOCK_M, BLOCK_K]
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # B tile: [BLOCK_K, BLOCK_N]
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # GEMM on the tile
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back the result (store as fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


@triton.jit
def elementwise_mul_kernel(
    a_ptr,        # float32* – first operand (out_norm)
    b_ptr,        # float32* – second operand (out_gate)
    c_ptr,        # float32* – output buffer
    size,         # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """
    Fuse the final element‑wise multiplication out_norm * out_gate.
    """
    pid   = tl.program_id(0)
    offs  = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask  = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``
    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
        The forward output of the TriMul block.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → there is no mask

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑Norm on the input (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                            # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all heavy linear / matmul work
    # ------------------------------------------------------------------
    x_h = x_norm.half()                          # [B, N, N, dim] (float16)

    # ------------------------------------------------------------------
    # 3) Fuse projection + gate linear layers (concatenation trick)
    # ------------------------------------------------------------------
    left_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    out_gate_w = weights["out_gate.weight"].half()            # [hidden, dim]

    # ------------------------------------------------------------------
    # 4) Perform the two linear maps (proj + gate) in one call each
    # ------------------------------------------------------------------
    # left_cat: [..., 2*hidden]   →  proj part [:hidden], gate part [hidden:]
    left_cat  = torch.nn.functional.linear(x_h, left_w)       # [B,N,N,2*hidden]
    right_cat = torch.nn.functional.linear(x_h, right_w)      # [B,N,N,2*hidden]

    left_proj = left_cat[..., :hidden_dim]                    # [B,N,N,hidden]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])    # [B,N,N,hidden]

    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # ------------------------------------------------------------------
    # 5) Optional mask (applied before gating, exactly as in reference)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)   # [B,N,N,1]
        left_proj  = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate          # [B,N,N,hidden] (fp16)
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Prepare tensors for the batched GEMM
    #    left  → (C, M, K)  with M = N, K = N
    #    right → (C, K, N)  (already transposed)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    C = B * hidden_dim                                 # total number of independent GEMMs

    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(C, N, N)   # (C, M, K)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(C, N, N)  # (C, K, N)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # 8) Launch the Triton batched‑matmul kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    M = N
    K = N

    # Strides (int64)
    stride_a_batch = left_flat.stride(0)
    stride_a_m     = left_flat.stride(1)
    stride_a_k     = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_k     = right_flat.stride(1)
    stride_b_n     = right_flat.stride(2)

    stride_c_batch = out_flat.stride(0)
    stride_c_m     = out_flat.stride(1)
    stride_c_n     = out_flat.stride(2)

    grid = (
        C,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to the original layout (float32 for LayerNorm)
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .float()
    )  # [B, N, N, hidden] (float32)

    # ------------------------------------------------------------------
    # 10) Layer‑Norm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                            # [B, N, N, hidden] (float32)

    # ------------------------------------------------------------------
    # 11) Output gate (computed once, same shape as out_norm)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))  # [B,N,N,hidden] fp16
    out_gate_f32 = out_gate.float()

    # ------------------------------------------------------------------
    # 12) Fuse out_norm * out_gate with a tiny Triton kernel
    # ------------------------------------------------------------------
    gated = torch.empty_like(out_norm)                     # float32 buffer
    total_elems = out_norm.numel()
    BLOCK = 8192
    grid = (triton.cdiv(total_elems, BLOCK),)
    elementwise_mul_kernel[grid](
        out_norm,
        out_gate_f32,
        gated,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 13) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    result = torch.nn.functional.linear(gated, weights["to_out.weight"])

    return result
