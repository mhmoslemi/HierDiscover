TriMul “outgoing” forward pass – Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the 4‑D input over the feature dimension (float32).
2. Cast to fp16 for all heavy work.
3. Linear projections (no bias) → left, right (shape [B,N,N,H]).
4. Compute three gating vectors (sigmoid of linear projections of x):
   left_gate, right_gate, out_gate (shape [B,N,N,H]).
5. Multiply left/right by their gates and by the optional mask
   (mask broadcast on H, still in fp16).
6. Core O(N³·H) contraction:
      out_{i,j,h} = Σ_k left_{i,k,h} · right_{j,k,h}
   Implemented as a batched half‑precision GEMM with fp32 accumulation.
   Tiles are 128×128×64 (M,N,K) – ideal for H100 tensor cores.
7. Convert result to float32, apply a second LayerNorm over H,
   multiply by out_gate and finally project back to the original
   feature dimension (H → dim) with a linear layer.
All tensor dimensions are validated for the supported test‑cases.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: batched half‑precision GEMM with fp32 accumulation.
#   A : (batch, M, K)  – fp16
#   B : (batch, K, N)  – fp16
#   C : (batch, M, N)  – fp16

    stride_cb, stride_cm, stride_cn,  # C : (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_b = tl.program_id(2)   # batch / hidden‑channel index
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32 (precision‑stable)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers for the current tile of A and B
        a_ptrs = (
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # out‑of‑bounds handling
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # tensor‑core dot (fp16×fp16 → fp32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back the result (cast to fp16 for storage)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)

    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision matrix multiplication using the Triton kernel.
    a : (B*H, M, K)  fp16
    b : (B*H, K, N)  fp16
    returns c : (B*H, M, N) fp16
    """
    assert a.is_cuda and b.is_cuda
    assert a.dtype == torch.float16 and b.dtype == torch.float16

    B_h, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B_h, M, N), device=a.device, dtype=torch.float16)

    # Tile sizes tuned for H100 (fit in shared memory, enable tensor cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B_h,
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 prefers 8 warps per block
        num_stages=4,         # double‑buffering
    )
    return c


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim]      (float32, on CUDA)
            mask_tensor  : torch.Tensor [B, N, N] or None (float32/bool)
            weights_dict : dict of torch.Tensor parameters
            config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------- unpack ---------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------- 1) Input LayerNorm --------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  (float32)

    # --------------------------- 2) Cast to fp16 ----------------------
    x_fp16 = x.half()  # (fp16)

    # --------------------------- 3) Projections ------------------------
    left_proj_w = weights["left_proj.weight"].half()   # (H, dim)
    right_proj_w = weights["right_proj.weight"].half() # (H, dim)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, H] fp16

    # --------------------------- 4) Gates ------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # [B,N,N,H] fp16
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid() # [B,N,N,H] fp16
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()    # [B,N,N,H] fp16

    # Apply gating (mask will be applied afterwards)
    left = left * left_gate
    right = right * right_gate

    # --------------------------- 5) Optional mask -----------------------
    if (not nomask) and (mask_tensor is not None):
        # broadcast mask on the hidden dimension
        mask = mask_tensor.to(x_fp16.dtype).unsqueeze(-1)  # (B,N,N,1) fp16
        left = left * mask
        right = right * mask

    # --------------------------- 6) Core O(N³) contraction ----------
    B, N, _, _ = left.shape

    # Rearrange to (B, H, N, N) – required layout for our batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be transposed on the last two axes so that we multiply
    # (i,k) with (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    # Collapse the leading two dimensions -> (B*H, N, N)
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N) # (BH, K=N, N)

    # Batched half‑precision GEMM
    out_flat = _batched_matmul(left_flat, right_flat)   # (BH, N, N) fp16

    # Restore original layout: (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------- 7) Output LayerNorm -------------------
    out_fp32 = out.to(torch.float32)  # convert for numerically stable LN
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H] float32

    # --------------------------- 8) Apply out gate --------------------
    out_gate_fp32 = out_gate.to(torch.float32)   # broadcastable
    out_gated = out_norm * out_gate_fp32

    # --------------------------- 9) Final linear ----------------------
    to_out_w = weights["to_out.weight"]
    result = F.linear(out_gated, to_out_w)       # [B, N, N, dim] float32

    return result
