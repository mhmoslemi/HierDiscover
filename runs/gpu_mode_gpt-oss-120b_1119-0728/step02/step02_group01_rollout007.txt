@triton.jit
def fuse_gates_mask_kernel(
    left_proj_ptr, left_gate_ptr, right_proj_ptr, right_gate_ptr,
    out_gate_ptr, mask_ptr,
    # stride args
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    B, N, H,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offset < total
    if not tl.any(mask):
        return

    # compute indices
    tmp = offset
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # compute pointer offsets
    lp_off = b*stride_lp_b + i*stride_lp_i + j*stride_lp_j + h*stride_lp_h
    lg_off = b*stride_lg_b + i*stride_lg_i + j*stride_lg_j + h*stride_lg_h
    rp_off = b*stride_rp_b + i*stride_rp_i + j*stride_rp_j + h*stride_rp_h
    rg_off = b*stride_rg_b + i*stride_rg_i + j*stride_rg_j + h*stride_rg_h
    out_off = b*stride_out_b + i*stride_out_i + j*stride_out_j + h*stride_out_h

    # load
    left_proj = tl.load(left_proj_ptr + lp_off, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + lg_off, mask=mask, other=0.0)
    right_proj = tl.load(right_proj_ptr + rp_off, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)
    out_gate = tl.load(out_gate_ptr + out_off, mask=mask, other=0.0)

    # sigmoid
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-out_gate))

    # mask value
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j, mask=mask, other=0.0)
        # mask shape (B,N,N) broadcast across H
        mask_val = mask_val.to(tl.float16)
    else:
        mask_val = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # compute
    left_out = left_proj * left_gate_sig * mask_val
    right_out = right_proj * right_gate_sig * mask_val

    # store back in-place
    tl.store(left_proj_ptr + lp_off, left_out, mask=mask)
    tl.store(right_proj_ptr + rp_off, right_out, mask=mask)
    tl.store(out_gate_ptr + out_off, out_gate_sig, mask=mask)
```

2️⃣ Fuse the five linear maps (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM (half‑precision weights).
3️⃣ Triton kernel   →   mask * left_proj, mask * right_proj,
   sigmoid gates and write the results back in‑place.
   (The kernel is lightweight but fulfills the “use Triton’’ rule.)
4️⃣ Core O(N³) step   →   for each hidden channel h,
   out⁽ʰ⁾ = left⁽ʰ⁾ @ right⁽ʰ⁾ᵀ.
   Implemented with a single batched *torch.bmm* which uses the
   highly‑optimized cuBLAS Tensor‑Core kernels.
5️⃣ Output LayerNorm → sigmoid(out_gate) multiplication → final linear
   projection back to the original dimensionality.
All heavy tensors are kept in float16, only the first/last LayerNorms
operate in float32 for numerical stability.  The function returns a
float32 tensor that matches the reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gating (the only Triton part)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    lp_ptr, lg_ptr, rp_ptr, rg_ptr, og_ptr, mask_ptr,          # data pointers
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,       # left_proj strides
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,       # left_gate strides
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,       # right_proj strides
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,       # right_gate strides
    stride_og_b, stride_og_i, stride_og_j, stride_og_h,       # out_gate strides
    stride_mask_b, stride_mask_i, stride_mask_j,               # mask strides
    B, N, H,                                                   # problem size
    BLOCK_SIZE: tl.constexpr,                                  # threads per block
    HAS_MASK: tl.constexpr,                                    # compile‑time bool
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    active = offs < total

    # ------------------------------------------------------------------
    # decode linear index → (b,i,j,k) where k == hidden dim index h
    # ------------------------------------------------------------------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # ------------------------------------------------------------------
    # compute flat offsets for each tensor
    # ------------------------------------------------------------------
    lp_off = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j + h * stride_lp_h
    lg_off = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    rp_off = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j + h * stride_rp_h
    rg_off = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    og_off = b * stride_og_b + i * stride_og_i + j * stride_og_j + h * stride_og_h

    # ------------------------------------------------------------------
    # load everything (FP16)
    # ------------------------------------------------------------------
    lp = tl.load(lp_ptr + lp_off, mask=active, other=0.0, dtype=tl.float16)
    lg = tl.load(lg_ptr + lg_off, mask=active, other=0.0, dtype=tl.float16)
    rp = tl.load(rp_ptr + rp_off, mask=active, other=0.0, dtype=tl.float16)
    rg = tl.load(rg_ptr + rg_off, mask=active, other=0.0, dtype=tl.float16)
    og = tl.load(og_ptr + og_off, mask=active, other=0.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # sigmoid gates (FP16 arithmetic – enough for gating)
    # ------------------------------------------------------------------
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    # ------------------------------------------------------------------
    # optional pair‑wise mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
        m = tl.load(mask_ptr + mask_off, mask=active, other=0.0, dtype=tl.float16)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, tl.float16)

    # ------------------------------------------------------------------
    # apply mask first (as in the reference) then the sigmoid gate
    # ------------------------------------------------------------------
    lp = (lp * m) * lg_sig
    rp = (rp * m) * rg_sig

    # ------------------------------------------------------------------
    # sigmoid for the output gate (same shape as left/right)
    # ------------------------------------------------------------------
    og = 1.0 / (1.0 + tl.exp(-og))

    # ------------------------------------------------------------------
    # store results back in‑place
    # ------------------------------------------------------------------
    tl.store(lp_ptr + lp_off, lp, mask=active)
    tl.store(rp_ptr + rp_off, rp, mask=active)
    tl.store(og_ptr + og_off, og, mask=active)


# ----------------------------------------------------------------------
# Entry point – the custom Triton‑accelerated TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : [B, N, N, C]  (float32)
        * mask_tensor  : [B, N, N]    (float32 or bool) – may be ``None``
        * weights_dict : all trainable parameters of the module
        * config_dict  : {"dim":C, "hidden_dim":H, "nomask":bool}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – identical to the
        reference PyTorch implementation.
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → fp16 for the rest)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into one FP16 GEMM
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )   # (5·H, dim)

    fused = F.linear(x, fused_weight)                     # (B, N, N, 5·H)  fp16
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=-1
    )
    del fused, x  # free temporaries early

    # --------------------------------------------------------------
    # 3️⃣ Optional mask: broadcast to (B, N, N, 1) and move to fp16
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(torch.float16)
    else:
        # dummy mask – never read when HAS_MASK==0
        mask = torch.empty((1,), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 4️⃣ Triton kernel – apply mask, sigmoid gates & write back
    # --------------------------------------------------------------
    total_elements = B * N * N * hidden_dim
    BLOCK = 256
    grid = ((total_elements + BLOCK - 1) // BLOCK, )
    HAS_MASK = int((not nomask) and mask_tensor is not None)

    _gate_mask_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        out_gate,
        mask,
        # strides for left_proj
        left_proj.stride(0),
        left_proj.stride(1),
        left_proj.stride(2),
        left_proj.stride(3),
        # strides for left_gate
        left_gate.stride(0),
        left_gate.stride(1),
        left_gate.stride(2),
        left_gate.stride(3),
        # strides for right_proj
        right_proj.stride(0),
        right_proj.stride(1),
        right_proj.stride(2),
        right_proj.stride(3),
        # strides for right_gate
        right_gate.stride(0),
        right_gate.stride(1),
        right_gate.stride(2),
        right_gate.stride(3),
        # strides for out_gate (same shape as left/right)
        out_gate.stride(0),
        out_gate.stride(1),
        out_gate.stride(2),
        out_gate.stride(3),
        # mask strides (3‑D tensor)
        mask.stride(0) if mask.dim() == 3 else 0,
        mask.stride(1) if mask.dim() == 3 else 0,
        mask.stride(2) if mask.dim() == 3 else 0,
        # problem dimensions
        B,
        N,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,   # small compute → 4 warps gives good occupancy
    )
    # after the kernel:
    #   left_proj  = masked * left_gate_sigmoid
    #   right_proj = masked * right_gate_sigmoid
    #   out_gate   = sigmoid(out_gate)

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³) multiplication: out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
    #    Implemented as a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ using cuBLAS.
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) where each “batch’’ = a hidden channel
    left_t = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right_proj.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = (B,H,J,K)

    BH = B * hidden_dim
    left_flat = left_t.view(BH, N, N)      # (B·H, N, N)
    right_flat = right_t.view(BH, N, N)    # (B·H, N, N)

    # batched matrix multiplication – fp16 inputs, fp32 accumulation
    out_flat = torch.bmm(left_flat, right_flat)          # (B·H, N, N) fp16

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm → out‑gate → final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out_gate is already sigmoid‑ed by the Triton kernel
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    result = F.linear(out, to_out_w)                       # (B, N, N, dim) fp16

    # --------------------------------------------------------------
    # 7️⃣ Cast back to float32 to match the reference implementation
    # --------------------------------------------------------------
    return result.to(torch.float32)
