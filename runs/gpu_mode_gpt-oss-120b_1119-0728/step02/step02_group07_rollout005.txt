@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, hidden_dim, M, N, K,
    stride_ab, stride_am, stride_ak, stride_ah,
    stride_bb, stride_bn, stride_bk, stride_bh,
    stride_cb, stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr = 4,  # optional
):
    pid = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_z = tl.program_id(2)  # batch * hidden

    # decompose batch and hidden
    b = pid_z // hidden_dim
    h = pid_z - b * hidden_dim

    # compute base pointers
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_bb + h * stride_bh
    c_base = c_ptr + b * stride_cb + h * stride_ch

    # offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A: [BLOCK_M, BLOCK_K]
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )
        # Load B^T: B [BLOCK_K, BLOCK_N] (since we need B^T)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
        )
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

        # pointer increment not needed because we recompute offsets each iteration.

    # Write results
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_base
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_c,
    )

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------

The module works on a 4‑D tensor `x` of shape [B, N, N, D] and computes

    out[b,i,j,:] = Σ_k left[b,i,k,:] · right[b,j,k,:]

where `left` and `right` are linear projections of `x` followed by a
sigmoid gate and an optional mask.  The heavy N³ reduction is executed as a
batched half‑precision GEMM (C = A @ Bᵀ) with FP32 accumulation.  The kernel
operates directly on the original “[B, N, N, H]” layout – no intermediate
contiguous copies are required, which removes a large memory‑traffic bottleneck
present in naïve implementations.

Algorithm
~~~~~~~~~
1. Layer‑norm the input (FP32).
2. Fuse “proj + gate” for the left and right streams:
   a single `nn.Linear` with stacked weights produces a tensor of shape
   `[B, N, N, 2*H]`.  The first H channels are the projection, the second H
   channels are the pre‑sigmoid gate.
3. Apply the (optional) mask, the sigmoid gates and the element‑wise mask.
4. Run a custom Triton GEMM kernel:
   - The kernel receives the tensors `left` and `right` **in the original**
     layout `[B, N, N, H]` (contiguous, hidden dimension innermost).
   - Each program instance works on a tile `(BLOCK_M × BLOCK_N)` of the
     output for a particular `(batch, hidden)` pair.  The hidden dimension is
     folded into the program‑ID dimension, so a single launch covers all
     `B*H` independent matrix multiplications.
   - Loads are strided (`stride_ak = H`, `stride_bk = H`) because `k` steps
     over the hidden channel.  FP16 inputs are multiplied with Tensor‑Core
     `tl.dot`, accumulation is FP32.
5. Layer‑norm the hidden‑dim output, apply the out‑gate and a final linear
   projection back to the original feature dimension.
6. Return a FP32 tensor `[B, N, N, D]`.

The implementation keeps the most expensive part (the N³ reduction) on the
GPU, removes unnecessary copies, and exploits Tensor‑Core FP16 × FP16 → FP32
dot‑products for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B : (B, H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                     # const float16 *
    b_ptr,                     # const float16 *
    c_ptr,                     # float16 *
    batch,                     # int  (original batch size B)
    hidden_dim,                # int  (H)
    M,                         # int  (N)
    N,                         # int  (N)
    K,                         # int  (N)
    # strides for A (shape [B,N,N,H])
    stride_ab, stride_am, stride_ak, stride_ah,
    # strides for B (shape [B,N,N,H])
    stride_bb, stride_bn, stride_bk, stride_bh,
    # strides for C (shape [B,N,N,H])
    stride_cb, stride_cm, stride_cn, stride_ch,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr = 8,
):
    """Batched half‑precision GEMM with fp32 accumulation.

    Each program instance computes a (BLOCK_M × BLOCK_N) tile of C for a
    particular (batch, hidden) pair.
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_z = tl.program_id(2)          # flattened (batch * hidden)

    # Decompose flattened index into (batch, hidden)
    b = pid_z // hidden_dim
    h = pid_z - b * hidden_dim

    # Base pointers that already account for batch and hidden offsets
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_bb + h * stride_bh
    c_base = c_ptr + b * stride_cb + h * stride_ch

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile   (M × K)
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )

        # Load Bᵀ tile  (K × N)  – note the transposition in the addressing
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
        )

        # Matrix‑multiply the tile; tl.dot uses Tensor‑Cores (fp16×fp16→fp32)
        acc += tl.dot(a, b)

    # Write the result back (fp16)
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_base
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_c,
    )


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass (outgoing TriMul)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask_tensor  : torch.Tensor [B, N, N]       (bool/float) – optional
            - weights_dict : dict with LayerNorm/Linear parameters (float32)
            - config_dict  : contains "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (fp32 – stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                 # [B, N, N, dim] (fp32)

    # ------------------------------------------------------------------
    # 2) Fuse projection + gate for both streams (fp16)
    #    We concatenate the projection and gate weights so that a single
    #    linear call produces both outputs at once.
    # ------------------------------------------------------------------
    # Concatenated weights: (2*H, dim) for left and right streams
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # Cast to fp16 after the first (stable) LayerNorm
    x_h = x.to(torch.float16)          # [B, N, N, dim] (fp16)

    # Left stream: first H → proj, second H → gate (pre‑sigmoid)
    left_concat = F.linear(x_h, left_concat_w)            # [B,N,N,2*H] (fp16)
    left_proj = left_concat[..., :hidden_dim]            # [B,N,N,H]
    left_gate_pre = left_concat[..., hidden_dim:]        # [B,N,N,H]

    # Right stream: same pattern
    right_concat = F.linear(x_h, right_concat_w)         # [B,N,N,2*H] (fp16)
    right_proj = right_concat[..., :hidden_dim]          # [B,N,N,H]
    right_gate_pre = right_concat[..., hidden_dim:]      # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Out‑gate (computed in fp32 – used later)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"])           # [B,N,N,H] fp32
    )                                                    # (fp32)

    # ------------------------------------------------------------------
    # 4) Optional input mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype=left_proj.dtype).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5) Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)              # fp16
    right_gate = torch.sigmoid(right_gate_pre)            # fp16

    # ------------------------------------------------------------------
    # 6) Element‑wise gated tensors (fp16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate                           # fp16
    right = right_proj * right_gate                        # fp16

    # ------------------------------------------------------------------
    # 7) Batched GEMM: out_hidden[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    The kernel works directly on the [B,N,N,H] layout – no copies.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                               # H == hidden_dim

    # Output placeholder (fp16) – same layout as inputs
    out_hidden = torch.empty_like(left, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_ab = left.stride(0)    # batch stride
    stride_am = left.stride(1)    # row   stride (i)
    stride_ak = left.stride(2)    # col   stride (k)
    stride_ah = left.stride(3)    # hidden stride (1)

    stride_bb = right.stride(0)
    stride_bn = right.stride(1)
    stride_bk = right.stride(2)
    stride_bh = right.stride(3)

    stride_cb = out_hidden.stride(0)
    stride_cm = out_hidden.stride(1)
    stride_cn = out_hidden.stride(2)
    stride_ch = out_hidden.stride(3)

    # Tile sizes – tuned for H100 (large tiles, good occupancy)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
        B * H,                                 # (batch * hidden)
    )

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        H,
        N,
        N,
        N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bb,
        stride_bn,
        stride_bk,
        stride_bh,
        stride_cb,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Convert the hidden‑dim result back to fp32 for the remaining ops
    out = out_hidden.to(torch.float32)                    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8) Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 9) Apply out‑gate (already fp32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])        # [B,N,N,dim]

    return out
