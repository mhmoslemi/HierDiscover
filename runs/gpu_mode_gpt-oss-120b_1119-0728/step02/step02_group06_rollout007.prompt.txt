import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

import torch
import triton
import triton.language as tl


@triton.jit
def elementwise_mul_kernel(
    a_ptr,  # float32*   – first operand
    b_ptr,  # float32*   – second operand (out_gate)
    c_ptr,  # float32*   – output
    size,   # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """
    Very small Element‑wise kernel used to fuse the final
    out_norm * out_gate multiplication.  It loads a 1‑D
    slice of the tensors, multiplies them and writes the result
    back, handling boundary conditions with a mask.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul module
    (the heavy N³ operation is performed with a batched torch.bmm,
    which on H100 uses Tensor‑Core kernels).  A tiny Triton kernel
    fuses the final element‑wise multiplication with the output gate.
    All other operations stay in PyTorch.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑Norm on the input (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                            # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all heavy linear / matmul work
    # ------------------------------------------------------------------
    x_h = x_norm.half()                          # [B, N, N, dim] (float16)

    # ------------------------------------------------------------------
    # 3) Fuse projection + gate linear layers by concatenating weights
    # ------------------------------------------------------------------
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()                                      # [2*hidden, dim] (fp16)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()
    out_gate_w = weights["out_gate.weight"].half()  # [hidden, dim] (fp16)

    # ------------------------------------------------------------------
    # 4) One linear per side → (proj , gate) packed in the last dim
    # ------------------------------------------------------------------
    left_cat = torch.nn.functional.linear(x_h, left_w)   # [B,N,N,2*hidden] fp16
    right_cat = torch.nn.functional.linear(x_h, right_w)

    # split
    left_proj = left_cat[..., :hidden_dim]               # [B,N,N,hidden] fp16
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])  # fp16
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    # output gate (used after the output LayerNorm)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))  # fp16

    # ------------------------------------------------------------------
    # 5) Optional mask (applied before gating, exactly as reference)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)           # [B,N,N,1] fp16
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # [B,N,N,hidden] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Batched matrix multiplication  (B*hidden independent GEMMs)
    #    Each GEMM is:  (N x N) @ (N x N)ᵀ  →  (N x N)
    #    Implemented with torch.bmm which on H100 runs on Tensor‑Cores.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to [B*hidden, N, N]; for the right side we need the (k,j) order → transpose last two dims
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # fp16
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)  # fp16

    out_flat = torch.bmm(left_flat, right_flat)           # fp16, shape [B*hidden, N, N]

    # ------------------------------------------------------------------
    # 8) Reshape back to the original layout and cast to float32
    # ------------------------------------------------------------------
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .float()
    )                                                       # [B,N,N,hidden] fp32

    # ------------------------------------------------------------------
    # 9) Layer‑Norm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                                       # fp32

    # ------------------------------------------------------------------
    # 10) Fuse out_norm * out_gate using a tiny Triton kernel
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.float()
    gated = torch.empty_like(out_norm)                     # fp32 output buffer
    total_elems = out_norm.numel()
    BLOCK = 8192                                            # 8 K elements per program
    grid = (triton.cdiv(total_elems, BLOCK),)
    elementwise_mul_kernel[grid](
        out_norm,
        out_gate_f32,
        gated,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 11) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    result = torch.nn.functional.linear(gated, weights["to_out.weight"])

    return result
