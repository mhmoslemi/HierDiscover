import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_mul_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elements,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # flatten indices to multi-dim
    # compute batch index
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # offsets for each tensor
    proj_offset = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
    gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    out_offset  = b * stride_out_b  + i * stride_out_i  + j * stride_out_j  + h * stride_out_h
    # mask has no H dimension; broadcast
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    proj = tl.load(proj_ptr + proj_offset, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + gate_offset, mask=mask, other=tl.float16(0.0))
    msk  = tl.load(mask_ptr + mask_offset, mask=mask, other=tl.float16(0.0))

    out = proj * msk * gate
    tl.store(out_ptr + out_offset, out, mask=mask)

def fused_mul(proj, gate, mask):
    # proj, gate: [B,N,N,H], fp16
    # mask: [B,N,N,1] or [B,N,N], fp16 broadcast along H
    B, N, _, H = proj.shape
    total = B * N * N * H
    # strides
    stride_proj = proj.stride()
    stride_gate = gate.stride()
    stride_mask = mask.stride()
    stride_out = proj.stride()  # output same layout as proj
    # allocate output
    out = torch.empty_like(proj)
    BLOCK = 65536  # 64K elements per block
    grid = (triton.cdiv(total, BLOCK),)
    fuse_mul_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        B, N, H,
        stride_proj[0], stride_proj[1], stride_proj[2], stride_proj[3],
        stride_gate[0], stride_gate[1], stride_gate[2], stride_gate[3],
        stride_mask[0], stride_mask[1], stride_mask[2],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=2,
    )
    return out

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fuse_mul_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elements,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Compute indices in the 4‑D tensor
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # Offsets for each tensor (stride in elements)
    proj_offset = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + h * stride_proj_h
    gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    out_offset  = b * stride_out_b  + i * stride_out_i  + j * stride_out_j  + h * stride_out_h
    # mask has no H dimension
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    proj = tl.load(proj_ptr + proj_offset, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + gate_offset, mask=mask, other=tl.float16(0.0))
    msk  = tl.load(mask_ptr + mask_offset, mask=mask, other=tl.float16(0.0))

    out = proj * msk * gate
    tl.store(out_ptr + out_offset, out, mask=mask)

def fused_mul(proj, gate, mask):
    """
    proj, gate : torch.Tensor of shape [B, N, N, H] (fp16)
    mask       : torch.Tensor of shape [B, N, N, 1] (fp16)
    Returns: torch.Tensor of same shape as proj/gate with element‑wise product
             proj * mask * gate computed in a fused Triton kernel.
    """
    B, N, _, H = proj.shape
    total = B * N * N * H
    # allocate output
    out = torch.empty_like(proj)
    # stride information (in elements)
    spb, spi, spj, sph = proj.stride()
    sgb, sgi, sgj, sgh = gate.stride()
    smb, smi, smj = mask.stride()   # mask has no H stride (broadcasted)
    sob, soi, soj, soh = out.stride()
    BLOCK = 65536  # 64 K elements per program instance – a good trade‑off on H100
    grid = (triton.cdiv(total, BLOCK),)
    fuse_mul_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        B, N, H,
        spb, spi, spj, sph,
        sgb, sgi, sgj, sgh,
        smb, smi, smj,
        sob, soi, soj, soh,
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=2,
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor or None [B, N, N] (float32 or bool)
        weights      : dict of model parameters (float32)
        config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 before the heavy linear ops
    # ------------------------------------------------------------------
    x_h = x.half()  # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,hidden_dim] fp16
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4️⃣ Gating pre‑activations
    # ------------------------------------------------------------------
    left_gate_pre  = F.linear(x_h, weights["left_gate.weight"].half())
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())
    out_gate_pre   = F.linear(x_h, weights["out_gate.weight"].half())

    left_gate  = left_gate_pre.sigmoid()   # fp16
    right_gate = right_gate_pre.sigmoid()
    out_gate   = out_gate_pre.sigmoid()    # fp16, shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Mask handling (fp16)
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_h = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask + gate + projection with a Triton kernel
    # ------------------------------------------------------------------
    left  = fused_mul(left_proj, left_gate, mask_h)    # fp16
    right = fused_mul(right_proj, right_gate, mask_h)  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Reshape for batched matrix multiplication
    #    (B·H, N, N) where each hidden channel becomes a separate batch.
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left = left.view(batch_h, N, N)   # (B·H, N, N)
    right = right.view(batch_h, N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Batched matmul C = A @ Bᵀ (fp16 × fp16 → fp16, fp32 accumulate)
    # ------------------------------------------------------------------
    out_h = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to the 4‑D layout and cast to FP32 for the remaining ops
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N)               # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()          # (B, N, N, H)
    out = out.float()                                   # FP32

    # ------------------------------------------------------------------
    # 10️⃣ Hidden‑dim LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.float()   # broadcast over (B,N,N,H)

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass
---------------------------------------------------

The original PyTorch implementation performs:

1. LayerNorm over the feature dimension (float32).
2. Three linear projections (left/right) and three gates (left/right/out) – all
   weight matrices are applied without bias.
3. Element‑wise mask × gate × projection fusion.
4. A cubic “pairwise” product:
       out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   which is equivalent to a batched matrix‑multiply C = A @ Bᵀ where
   A and B have shape (B·H, N, N) (H = hidden_dim).
5. LayerNorm on the hidden dimension, multiplication by the out‑gate,
   and a final linear projection back to the original feature size.

In this version we keep steps 1, 2, 4‑5 in PyTorch (leveraging cuBLAS / Tensor‑Core
bmm for the N³ part) and we **fuse step 3** with a tiny Triton kernel.
Only the element‑wise mask‑gate‑projection multiplication runs in Triton,
while the heavy N³ reduction uses `torch.bmm`, which on H100 is far faster than
the hand‑written Triton matmul used previously.  The kernel processes the
flattened 4‑D tensor in blocks of 64 K elements, broadcasting the mask along
the hidden dimension.

The function `custom_kernel` follows the exact API required by the test harness
and returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask × gate × projection
# ----------------------------------------------------------------------
@triton.jit
def fuse_mul_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    total_elements,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK: tl.constexpr,
):
    """
    out = proj * mask * gate    (all fp16)

    Parameters
    ----------
    proj_ptr, gate_ptr, mask_ptr, out_ptr : pointers to the tensors
    total_elements : B·N·N·H  (flattened size)
    B, N, H        : tensor dimensions
    stride_*       : element‑wise strides for each tensor
    BLOCK          : compile‑time tile size (number of elements per program)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # ----- decode linear index into (b,i,j,h) -----
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # ----- compute byte offsets (strides are in elements) -----
    proj_offset = (
        b * stride_proj_b
        + i * stride_proj_i
        + j * stride_proj_j
        + h * stride_proj_h
    )
    gate_offset = (
        b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h
    )
    out_offset = (
        b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h
    )
    # mask has no H dimension: broadcast over `h`
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # ----- load, compute, store -----
    proj = tl.load(proj_ptr + proj_offset, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + gate_offset, mask=mask, other=tl.float16(0.0))
    msk  = tl.load(mask_ptr + mask_offset, mask=mask, other=tl.float16(0.0))

    out = proj * msk * gate
    tl.store(out_ptr + out_offset, out, mask=mask)


def fused_mul(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise fusion:   out = proj * mask * gate
    All tensors are fp16 and have shape [B, N, N, H] (mask = [B, N, N, 1]).
    The computation is performed by a Triton kernel to avoid three separate
    memory passes.
    """
    B, N, _, H = proj.shape
    total = B * N * N * H

    out = torch.empty_like(proj)

    # strides in elements (torch returns int)
    spb, spi, spj, sph = proj.stride()
    sgb, sgi, sgj, sgh = gate.stride()
    smb, smi, smj = mask.stride()          # mask has no H stride (broadcast)
    sob, soi, soj, soh = out.stride()

    BLOCK = 65536  # 64 K elements per program (good trade‑off on H100)
    grid = (triton.cdiv(total, BLOCK),)

    fuse_mul_kernel[grid](
        proj,
        gate,
        mask,
        out,
        total,
        B,
        N,
        H,
        spb,
        spi,
        spj,
        sph,
        sgb,
        sgi,
        sgj,
        sgh,
        smb,
        smi,
        smj,
        sob,
        soi,
        soj,
        soh,
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask         : torch.Tensor or None, shape [B, N, N] (float32/bool)
        weights      : dict of model parameters (float32)
        config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 before heavy linear ops
    # --------------------------------------------------------------
    x_h = x.half()                     # fp16

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – fp16
    # --------------------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())

    # --------------------------------------------------------------
    # 4️⃣ Gating pre‑activations + sigmoid
    # --------------------------------------------------------------
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].half())
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].half())

    left_gate = left_gate_pre.sigmoid()      # fp16
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()        # fp16, shape [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 5️⃣ Mask handling (fp16)
    # --------------------------------------------------------------
    if mask is None or nomask:
        mask_h = torch.ones((B, N, N, 1), dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]

    # --------------------------------------------------------------
    # 6️⃣ Fuse mask × gate × projection with Triton (one kernel per side)
    # --------------------------------------------------------------
    left = fused_mul(left_proj, left_gate, mask_h)     # fp16, [B,N,N,hidden_dim]
    right = fused_mul(right_proj, right_gate, mask_h)  # fp16

    # --------------------------------------------------------------
    # 7️⃣ Reshape for batched matmul C = A @ Bᵀ
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left = left.view(batch_h, N, N)                # (B·H, N, N)
    right = right.view(batch_h, N, N)

    # --------------------------------------------------------------
    # 8️⃣ Batched matmul (fp16*fp16 → fp16, fp32 accumulation)
    # --------------------------------------------------------------
    out_h = torch.bmm(left, right.transpose(1, 2))   # (B·H, N, N)  fp16

    # --------------------------------------------------------------
    # 9️⃣ Restore 4‑D layout and promote to fp32 for the remaining ops
    # --------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N)            # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()       # (B, N, N, H)
    out = out.float()                                # fp32

    # --------------------------------------------------------------
    # 10️⃣ Hidden‑dim LayerNorm + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.float()                     # broadcast over (B,N,N,H)

    # --------------------------------------------------------------
    # 11️⃣ Final projection back to ``dim``
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])    # [B, N, N, dim]  float32

    return out
