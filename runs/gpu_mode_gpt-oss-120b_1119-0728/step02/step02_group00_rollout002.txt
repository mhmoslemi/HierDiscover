"""
TriMul “outgoing” forward‑pass (AlphaFold3) – Triton + PyTorch implementation.
---------------------------------------------------------------------------
The heavy contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batched matrix‑multiply A @ Bᵀ for each hidden slice `h`.
We fuse this N×N×N³ operation into a single Triton GEMM kernel that
* works on FP16 inputs,
* accumulates in FP32 for numerical safety,
* uses tensor‑core friendly tiles (128×128×32) and a 3‑D launch grid
  (M‑tile, N‑tile, batch×hidden slice).

All remaining primitives (LayerNorm, linear projections, sigmoid gates)
are kept in PyTorch (FP16 where possible) to minimise code‑complexity while
still achieving a large speed‑up over a pure‑Python implementation.
The final result is returned as FP32 as required by the API.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# Input A : [B*H, N, N]   (row‑major, dtype=fp16)
# Input B : [B*H, N, N]   (row‑major, dtype=fp16) – accessed transposed
# Output C: [B*H, N, N]   (dtype=fp16, accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_a_batch, stride_a_i, stride_a_k,  # A strides (batch, row, col)
    stride_b_batch, stride_b_k, stride_b_i,  # B strides (batch, k, j) – transposed view
    stride_c_batch, stride_c_i, stride_c_j,  # C strides (batch, row, col)
    N,                                        # sequence length
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 2‑D grid identifies the tile (m,n); 3‑rd dimension identifies the batch*hidden id.
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_z = tl.program_id(2)   # batch * hidden slice

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # Accumulator (FP32) – Tensor‑core friendly dot accumulates in FP32.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # Load A block (M × K)
        a_ptrs = (
            a_ptr
            + pid_z * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)       # fp16

        # Load B block (K × N) – note the transposed view (k as row)
        b_ptrs = (
            b_ptr
            + pid_z * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)       # fp16

        # Accumulate
        acc += tl.dot(a, b)   # (M×K)·(K×N) → (M×N), acc in fp32

    # ------------------------------------------------------------------
    # Write the result back (convert to fp16 on store)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_z * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operator.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32, CUDA)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (float32) or None
        - weights_dict : mapping name → torch.Tensor (float32, CUDA)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}
    Returns
    -------
    torch.Tensor   : shape [B, N, N, dim] (float32, CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weights (already on the correct device)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (keep FP32 for stability)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # Cast to half for all subsequent heavy work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, _w("left_proj.weight").to(torch.float16))   # [B, N, N, H]
    right = F.linear(x_h, _w("right_proj.weight").to(torch.float16))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Broadcast mask over the hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear layers) – still FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, _w("left_gate.weight").to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, _w("right_gate.weight").to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_h, _w("out_gate.weight").to(torch.float16)))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM on the (B × hidden_dim) dimension
    #    Compute   out = left @ rightᵀ   for each hidden slice.
    # ------------------------------------------------------------------
    #   left, right : [B, N, N, H] → permute → [B, H, N, N] → view as [B*H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # Flatten batch and hidden dim for the kernel
    left_flat  = left_perm.view(batch_total, N, N)   # FP16, contiguous
    right_flat = right_perm.view(batch_total, N, N)  # FP16, contiguous

    # Allocate output buffer (FP16)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for H100 Tensor‑cores)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # 16‑16‑16 tensor‑core friendly (two per block)

    grid = (
        triton.cdiv(N, BLOCK_M),   # number of tiles in M dimension
        triton.cdiv(N, BLOCK_N),   # number of tiles in N dimension
        batch_total,               # independent GEMM instances (B×H)
    )

    _trimmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # A strides (batch, row=i, col=k)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # B strides (batch, k, j) – we treat the original row as ‘k’
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        # C strides (batch, row, col)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] (still FP16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Output processing (LayerNorm → gating → final linear)
    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension (FP16)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight").to(torch.float16),
        bias=_w("to_out_norm.bias").to(torch.float16),
    )
    # Element‑wise output gate (FP16)
    out = out * out_gate

    # Final linear projection back to “dim”
    out = F.linear(out, _w("to_out.weight").to(torch.float16))   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Cast back to FP32 (the API expects float32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
