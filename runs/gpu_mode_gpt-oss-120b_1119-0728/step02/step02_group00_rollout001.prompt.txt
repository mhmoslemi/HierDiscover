import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton implementation.
---------------------------------------------------------------
The heavy contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix‑multiply A @ Bᵀ for each hidden slice `d`.
We cast the intermediate tensors to FP16, perform the batched GEMM with a
custom Triton kernel that uses Tensor‑core friendly tile sizes
(128×128×32) and accumulates in FP32 for numerical safety.
All other operations (LayerNorm, linear projections, gating) are kept
in PyTorch but also run in FP16 to reduce memory traffic.
The final result is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction.
# Input A : [B*H, N, N]   (row‑major, dtype=fp16)
# Input B : [B*H, N, N]   (row‑major, dtype=fp16) – accessed transposed
# Output C: [B*H, N, N]   (dtype=fp16, accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_a_batch, stride_a_i, stride_a_k,  # A strides  (batch, row, col)
    stride_b_batch, stride_b_k, stride_b_i,  # B strides (batch, k, j) – transposed view
    stride_c_batch, stride_c_i, stride_c_j,  # C strides  (batch, row, col)
    N,                                        # sequence length
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # batch * hidden slice

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # Accumulator (FP32 for better accuracy – Tensor‑core accumulation)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over reduction dimension k
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Out‑of‑bounds masks
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # Load A block (M × K) – fp16
        a_ptrs = (
            a_ptr
            + pid_z * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # Load B block (K × N) – transposed view (k as row, n as col)
        b_ptrs = (
            b_ptr
            + pid_z * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Tensor‑core GEMM: accumulate in FP32
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result (still fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_z * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor, shape [B, N, N, dim], float32, CUDA
        mask_tensor  : torch.Tensor, shape [B, N, N]   (optional)
        weights_dict : mapping name → torch.Tensor (float32, CUDA)
        config_dict  : {"dim": int, "hidden_dim": int,
                        "nomask": bool (optional)}
    Returns
    -------
    out : torch.Tensor, shape [B, N, N, dim], float32, CUDA
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper for weight access (all weights are on the correct device)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (float32 → keep precision)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # Cast to half for the remaining heavy work
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – all in FP16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, _w("left_proj.weight").to(torch.float16))   # [B, N, N, H]
    right = F.linear(x_h, _w("right_proj.weight").to(torch.float16))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of linear layers) – still FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, _w("left_gate.weight").to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, _w("right_gate.weight").to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_h, _w("out_gate.weight").to(torch.float16)))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM on the (B×H) dimension
    # ------------------------------------------------------------------
    #   left, right : [B, N, N, H]  →  permute → [B, H, N, N] → view as [B*H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_flat  = left_perm.view(batch_total, N, N)    # fp16
    right_flat = right_perm.view(batch_total, N, N)

    # Output buffer (fp16)
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------
    # Triton launch configuration (tuned for H100 Tensor‑cores)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # Tensor‑core friendly K‑tile

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_total,               # independent GEMM instances
    )

    _trimmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # strides for A
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for B (transposed view)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original 4‑D layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Output processing (LayerNorm → gate → final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight").to(torch.float16),
        bias=_w("to_out_norm.bias").to(torch.float16),
    )
    out = out * out_gate                     # element‑wise gate (fp16)
    out = F.linear(out, _w("to_out.weight").to(torch.float16))   # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # Cast back to float32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
