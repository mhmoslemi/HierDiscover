"""
custom_kernel  –  Triton‑accelerated outgoing TriMul (AlphaFold3)

Algorithm
---------

1. **Layer‑norm (channel)** – float32 LayerNorm over the last dimension *C*.
2. **Linear projections + gates** – All linear layers are performed in fp16
   (Tensor‑Core) using `torch.nn.functional.linear`.  The optional mask is
   applied before the element‑wise gates.
3. **Heavy N³ interaction** – For each hidden channel the operation

       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   is a batched matrix‑multiply `left @ rightᵀ`.  We reshape the tensors to
   `[B·H, N, N]` and invoke `torch.bmm`, which runs on cuBLAS and achieves
   near‑optimal Tensor‑Core throughput on H100.
4. **Layer‑norm (hidden) + output gate** – float32 LayerNorm over the hidden
   dimension, multiplied by a sigmoid gate.
5. **Final projection** – The hidden representation (fp16) is projected back
   to the original channel dimension with a custom Triton GEMM kernel
   (`_final_linear_kernel`).  The kernel accumulates in fp32 and writes the
   result as float32.

Only the final projection is implemented in Triton, satisfying the requirement
while the bulk of the work (large GEMMs) uses cuBLAS for maximal speed.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: final dense projection (hidden → dim)
# ----------------------------------------------------------------------
@triton.jit
def _final_linear_kernel(
    A_ptr,                # fp16   [M, K]   (input hidden representation)
    B_ptr,                # fp16   [K, N]   (to_out weightᵀ)
    C_ptr,                # fp32   [M, N]   (output)
    M, N, K,
    stride_Am, stride_Ak,               # strides for A (row, col)
    stride_Bk, stride_Bn,               # strides for B (row(K), col(N))
    stride_Cm, stride_Cn,               # strides for C (row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ B   where A:[M,K], B:[K,N]; fp16·fp16 → fp32 accumulation."""
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this tile
    a_ptrs = A_ptr + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B_ptr + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K‑loop
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_blocks):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                     # fp16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                     # fp16
        acc += tl.dot(a, b)                       # fp16×fp16 → fp32
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write back
    c_ptrs = C_ptr + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask          : torch.Tensor [B, N, N] (bool/float) or None
        - weights       : dict of torch.Tensors (all on CUDA)
        - config        : dict with keys
              * "dim"        – input channel dimension (C)
              * "hidden_dim" – hidden dimension (H)
              * "nomask"     – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  Layer‑norm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Cast to fp16 for all dense linear ops (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Projections + gates (all fp16)
    # ------------------------------------------------------------------
    # Helper to obtain fp16 view of a weight
    to_fp16 = lambda t: t.to(torch.float16)

    left_proj_w   = to_fp16(weights["left_proj.weight"])
    right_proj_w  = to_fp16(weights["right_proj.weight"])
    left_gate_w   = to_fp16(weights["left_gate.weight"])
    right_gate_w  = to_fp16(weights["right_gate.weight"])
    out_gate_w    = to_fp16(weights["out_gate.weight"])
    to_out_w      = to_fp16(weights["to_out.weight"])

    # Linear projections
    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w) # [B,N,N,H]

    # Optional mask (broadcast over hidden)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)       # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # Apply element‑wise gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣  Heavy N³ interaction – batched GEMM via cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    # Prepare tensors of shape [B·H, N, N] suitable for torch.bmm
    left_flat  = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)   # [B*H, N, N]
    right_flat = right.permute(0, 3, 2, 1).reshape(bs * hidden_dim, seq_len, seq_len)  # [B*H, N, N] (k,j)

    # Batched matrix multiplication: (i,k) @ (k,j) → (i,j)
    out_hidden = torch.bmm(left_flat, right_flat)                                     # fp16 output
    out_hidden = out_hidden.to(torch.float32)                                          # cast for LayerNorm

    # Reshape back to [B, N, N, H]
    out_hidden = out_hidden.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4️⃣  Hidden Layer‑norm, output gate, final projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]  float32

    # Fuse output gate (cast to float32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Cast to fp16 for final projection
    out_norm_fp16 = out_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # Final linear projection using a Triton GEMM kernel
    # ------------------------------------------------------------------
    M = bs * seq_len * seq_len          # total rows in the fused matrix
    N_out = dim
    K = hidden_dim

    A = out_norm_fp16.view(M, K).contiguous()      # [M, K] fp16
    B_T = to_out_w.t().contiguous()                # [K, N_out] fp16 (weightᵀ)

    # Output buffer (fp32 accumulation)
    result_fp32 = torch.empty((M, N_out), dtype=torch.float32, device=device)

    # Kernel launch configuration (tuned for H100 Tensor‑Cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N_out + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides in *elements* (not bytes)
    stride_Am = A.stride(0)
    stride_Ak = A.stride(1)
    stride_Bk = B_T.stride(0)
    stride_Bn = B_T.stride(1)
    stride_Cm = result_fp32.stride(0)
    stride_Cn = result_fp32.stride(1)

    _final_linear_kernel[grid](
        A,
        B_T,
        result_fp32,
        M, N_out, K,
        stride_Am, stride_Ak,
        stride_Bk, stride_Bn,
        stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape to the expected output layout
    result = result_fp32.view(bs, seq_len, seq_len, dim)
    return result
