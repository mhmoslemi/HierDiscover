"""
TriMul “outgoing” forward (AlphaFold3) – Triton‑accelerated contraction.
--------------------------------------------------------------------
The expensive step is the 3‑D contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication A @ Bᵀ for every hidden slice `d`.
We pack the (batch, hidden) dimensions together, treat each slice as a
(N×N) × (N×N) GEMM and compute it with a custom Triton kernel that
uses Tensor‑core‑friendly tiles (128×128×32).  The kernel accumulates in
FP32 for numerical safety and writes back in FP16.  All surrounding
operations (LayerNorm, linear projections, gating, final Linear) are
performed in PyTorch in FP16 to keep memory traffic low.  The final
output is cast back to float32 as required by the API.

The implementation works for any batch size 1‑2, sequence length up to
1024 and hidden dimensions 128/384, with or without an input mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_a_batch, stride_a_i, stride_a_k,  # A strides (batch, row, col)
    stride_b_batch, stride_b_k, stride_b_i,  # B strides (batch, k‑row, j‑col) – transposed view
    stride_c_batch, stride_c_i, stride_c_j,  # C strides (batch, row, col)
    N,                                        # sequence length
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for one hidden slice:
       C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (B is accessed transposed)
    """
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_z = tl.program_id(2)   # batch * hidden slice

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 (Tensor‑core accumulates in FP32 automatically)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # A block: shape (M, K) – FP16
        a_ptrs = (
            a_ptr
            + pid_z * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B block (transposed view): shape (K, N) – FP16
        b_ptrs = (
            b_ptr
            + pid_z * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # GEMM for this tile
        acc += tl.dot(a, b)   # fp16×fp16→fp32 accumulation

    # ------------------------------------------------------------------
    # Write result (convert back to FP16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_z * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], float32, CUDA
        - mask_tensor  : torch.Tensor, shape [B, N, N] (optional, may be None)
        - weights_dict : dict mapping weight names to torch.Tensor (float32, CUDA)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    out : torch.Tensor, shape [B, N, N, dim], float32, CUDA
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Helper to fetch weights
    # --------------------------------------------------------------
    def _w(name):
        return weights[name]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (keep in FP32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # Cast to fp16 for the heavy work
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – fp16
    # --------------------------------------------------------------
    left = F.linear(x_h, _w("left_proj.weight").to(torch.float16))   # [B, N, N, H]
    right = F.linear(x_h, _w("right_proj.weight").to(torch.float16))

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear layers) – fp16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, _w("left_gate.weight").to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, _w("right_gate.weight").to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, _w("out_gate.weight").to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy contraction – batched GEMM via Triton
    # --------------------------------------------------------------
    # Rearrange to [B, H, N, N] and flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # View as (B*H, N, N) – row‑major fp16
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # Allocate output buffer (fp16)
    out_flat = torch.empty_like(left_flat)

    # Triton launch grid
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # friendly for Tensor Cores

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_hidden,              # independent GEMMs
    )

    _trimmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # strides for A (batch, i, k)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for B (batch, k, j) – note the transposed view
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (batch, i, j)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] (fp16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm → gate → final linear
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight").to(torch.float16),
        bias=_w("to_out_norm.bias").to(torch.float16),
    )
    out = out * out_gate          # element‑wise gate (fp16)
    out = F.linear(out, _w("to_out.weight").to(torch.float16))   # [B,N,N,dim] (fp16)

    # --------------------------------------------------------------
    # 7️⃣ Cast back to float32 for the public API
    # --------------------------------------------------------------
    return out.to(torch.float32)
