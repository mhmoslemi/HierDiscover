```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32}, num_warps=4, num_stages=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 32}, num_warps=8, num_stages=3),
        triton.Config({'BLOCK_M': 256, 'BLOCK_N': 256, 'BLOCK_K': 64}, num_warps=16, num_stages=3),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def batch_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bm, stride_bk, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B^T where A,B are (batch, M, K), (batch, N, K).

    The kernel accumulates in FP32 and writes FP16 result.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Coordinates of the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for the current tile of A and B^T
    a_ptrs = a_ptr + batch_id * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bm

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)   # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)   # (BLOCK_K, BLOCK_N) fp16

        # Convert to fp32 for accumulation
        a_fp32 = a.to(tl.float32)
        b_fp32 = b.to(tl.float32)
        acc += tl.dot(a_fp32, b_fp32)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = c_ptr + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        mask         : torch.Tensor of shape [B, N, N] (float32 or bool) or None
        weights      : dict of model parameters (float32)
        config       : dict with keys:
            - "dim"         : feature dimension
            - "hidden_dim"  : hidden dimension for the multiplicative update
            - "nomask"      : bool indicating mask presence

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # -------- 1. LayerNorm over feature dimension (FP32) ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim] FP32

    # -------- 2. Cast to FP16 for heavy ops ----------
    x_h = x.half()  # fp16

    # Linear projections (no bias)
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,hidden] fp16
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())

    # Gating pre‑activations
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].half())
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].half())

    # Sigmoid (still fp16)
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()   # [B,N,N,hidden] fp16

    # -------- 3. Mask handling ----------
    if mask is None or nomask:
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(dtype=torch.float16, device=device)
    mask_h = mask_h.unsqueeze(-1)   # [B,N,N,1]

    # -------- 4. Fuse mask, projection and gate ----------
    left = left_proj * mask_h * left_gate   # fp16, shape [B,N,N,hidden]
    right = right_proj * mask_h * right_gate

    # -------- 5. Rearrange to (B*hidden, N, N) ----------
    # Bring hidden channel to batch dimension
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left = left.view(batch_h, N, N)
    right = right.view(batch_h, N, N)

    # -------- 6. Batched matrix multiply using Triton ----------
    out_h = torch.empty_like(left)   # fp16, shape (B*hidden, N, N)

    # Strides
    stride_ab, stride_am, stride_ak = left.stride()
    stride_bb, stride_brow, stride_bcol = right.stride()
    # For B^T we need row stride (output column) = original row stride
    stride_bm = stride_brow
    stride_bk = stride_bcol
    stride_cb, stride_cm, stride_cn = out_h.stride()

    # Tile size selection is left to the autotuned kernel
    # We still need to provide concrete block sizes for launch; the autotuner will
    # pick the best config among those we declared.
    # Choose a conservative default; the kernel will ignore values that don't match the
    # selected config.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_tri_mul_kernel[grid](
        left,
        right,
        out_h,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bm,
        stride_bk,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------- 7. Reshape back to (B, N, N, hidden) ----------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,hidden) FP16
    out = out.to(torch.float32)   # back to FP32 for the final LayerNorm

    # -------- 8. Hidden‑dim LayerNorm + output gate ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Multiply by output gate (cast to FP32)
    out = out * out_gate.to(torch.float32)

    # -------- 9. Final projection back to `dim` ----------
    out = F.linear(out, weights["to_out.weight"])   # shape [B, N, N, dim] FP32

    return out

"""
Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

The computation consists of:
1. Layer‑Norm over the feature dimension (FP32).
2. Half‑precision linear projections for left/right tensors and three gates.
3. Element‑wise fusion of mask, projection and gate (still FP16).
4. Batched matrix multiplication C = A @ Bᵀ where each hidden channel becomes a
   separate batch: (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N).  The heavy N³ work is
   performed by an autotuned Triton kernel that accumulates in FP32 and writes
   the result in FP16.
5. Reshape back to the original 4‑D layout, FP32 Layer‑Norm over the hidden
   dimension, multiplication with the output gate and a final linear projection
   back to the original feature dimension.

The Triton kernel is autotuned for three tile sizes:
   64×64×32  (4 warps), 128×128×32 (8 warps) and 256×256×64 (16 warps).
This gives good performance across the test configurations (N ∈ {256,512,768,1024},
hidden ∈ {128,384}) on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched fp16 matmul C = A @ Bᵀ  (A,B : (batch, N, N))
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32},
            num_warps=4,
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32},
            num_warps=8,
            num_stages=3,
        ),
        triton.Config(
            {"BLOCK_M": 256, "BLOCK_N": 256, "BLOCK_K": 64},
            num_warps=16,
            num_stages=3,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def batch_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # base pointers
    M, N, K,                      # all equal to seq_len
    stride_am, stride_ak, stride_ab,   # A strides  (row, col, batch)
    stride_bm, stride_bk, stride_bb,   # Bᵀ strides (col, row, batch)
    stride_cm, stride_cn, stride_cb,   # C strides  (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.

    A, B, C are fp16 tensors; accumulation is performed in fp32 for numerical
    stability.  The kernel processes one (batch, M, N) tile per program.
    """
    batch_id = tl.program_id(0)   # batch dimension = B * hidden_dim
    pid_m    = tl.program_id(1)   # tile row
    pid_n    = tl.program_id(2)   # tile column

    # ------------------------------------------------------------------
    # Offsets for the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the current tile of A and Bᵀ
    # ------------------------------------------------------------------
    a_ptrs = (
        a_ptr
        + batch_id * stride_ab
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        b_ptr
        + batch_id * stride_bb
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bm
    )

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # (BLOCK_K, BLOCK_N) fp16

        # Convert to fp32 for the dot product
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write the result tile back to global memory
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_id * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Triton‑accelerated outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask         : torch.Tensor (float32/bool) or None, shape [B, N, N]
        weights      : dict of model parameters (float32 tensors)
        config       : dict with keys
                       - "dim"          : int, feature dimension
                       - "hidden_dim"   : int, hidden dimension for the update
                       - "nomask"       : bool, True if mask should be ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 before the heavy linear ops
    # ------------------------------------------------------------------
    x_h = x.half()  # fp16

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,hidden] fp16
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())

    # --------------------------------------------------------------
    # Gating pre‑activations
    # --------------------------------------------------------------
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].half())
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].half())

    # --------------------------------------------------------------
    # Sigmoid (fp16)
    # --------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()      # [B,N,N,hidden] fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask handling (fp16)
    # --------------------------------------------------------------
    if mask is None or nomask:
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(dtype=torch.float16, device=device)
    mask_h = mask_h.unsqueeze(-1)          # [B,N,N,1]

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask, projection and gate
    # --------------------------------------------------------------
    left = left_proj * mask_h * left_gate      # fp16, shape [B,N,N,hidden]
    right = right_proj * mask_h * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Bring hidden dimension into the batch axis
    #    resulting shape (B·hidden, N, N)
    # --------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left = left.view(batch_h, N, N)
    right = right.view(batch_h, N, N)

    # --------------------------------------------------------------
    # 6️⃣ Allocate output buffer (fp16)
    # --------------------------------------------------------------
    out_h = torch.empty_like(left)   # (B·hidden, N, N) fp16

    # --------------------------------------------------------------
    # 7️⃣ Gather stride information for the Triton kernel
    # --------------------------------------------------------------
    # A (left) strides: (batch, row, col)
    stride_ab, stride_am, stride_ak = left.stride()
    # B (right) strides – we will read it as Bᵀ
    stride_bb, stride_brow, stride_bcol = right.stride()
    stride_bm = stride_brow      # row stride after transpose (output column)
    stride_bk = stride_bcol      # inner (k) stride
    # C strides are identical to A's layout
    stride_cb, stride_cm, stride_cn = out_h.stride()

    # --------------------------------------------------------------
    # 8️⃣ Launch the autotuned Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 128  # dummy values – the autotuner will replace them
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_h,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    batch_tri_mul_kernel[grid](
        left,
        right,
        out_h,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bm,
        stride_bk,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 9️⃣ Reshape back to the original 4‑D layout and promote to fp32
    # --------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden) fp16
    out = out.to(torch.float32)                                              # FP32 for the remaining ops

    # --------------------------------------------------------------
    # 1️⃣0️⃣ Hidden‑dim LayerNorm + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate.to(torch.float32)   # broadcast over (B,N,N,hidden)

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] (FP32)

    return out
