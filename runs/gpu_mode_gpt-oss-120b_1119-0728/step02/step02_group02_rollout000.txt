"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The computation follows the reference PyTorch implementation:

1️⃣  LayerNorm over the input channels (float32).
2️⃣  Two linear projections (left/right) → hidden_dim (FP16) + learned
    element‑wise gates (sigmoid).  An optional mask is applied before gating.
3️⃣  Heavy N³ interaction:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    This is a *batched* GEMM: for every batch‑index and hidden channel we
    multiply two (N×N) matrices.  The operation is implemented in Triton
    (FP16×FP16 → FP32 accumulation) for maximal throughput on H100.
4️⃣  LayerNorm over the hidden dimension, an output gate, and a final
    linear projection back to the original channel dimension (FP16 matmul
    via cuBLAS).  The result is returned as float32.

Only the N³ interaction (step 3) is written in Triton; all other steps use
high‑performance PyTorch ops.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul interaction
#   C = A @ B   where A.shape = [B*H, M, K]   (left)
#                         B.shape = [B*H, K, N]   (rightᵀ)
#   Accumulation is FP32, inputs/outputs are FP16/FP32 respectively.
# ----------------------------------------------------------------------
@triton.jit
def _batched_trimul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,   # A: batch, row, col strides
    stride_Bb, stride_Bk, stride_Bn,   # B: batch, row(K), col(N) strides
    stride_Cb, stride_Ci, stride_Cn,   # C: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)          # block row
    pid_n = tl.program_id(axis=1)          # block column
    pid_b = tl.program_id(axis=2)          # batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_ptrs = (A_ptr + pid_b * stride_Ab
                + offs_m[:, None] * stride_Ai
                + offs_k[None, :] * stride_Ak)
    b_ptrs = (B_ptr + pid_b * stride_Bb
                + offs_k[:, None] * stride_Bk
                + offs_n[None, :] * stride_Bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k_blocks):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                     # FP16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                     # FP16
        acc += tl.dot(a, b)                       # FP16×FP16 → FP32
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    c_ptrs = (C_ptr + pid_b * stride_Cb
                + offs_m[:, None] * stride_Ci
                + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask          : torch.Tensor [B, N, N] (bool/float) or None
        - weights       : dict of torch.Tensors (all on CUDA)
        - config        : dict with keys
              * "dim"        – input channel dimension (C)
              * "hidden_dim" – hidden dimension inside the module
              * "nomask"     – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over input channels (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for all dense linear ops
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections + element‑wise gates (FP16)
    # ------------------------------------------------------------------
    # Cast weights once
    _to_fp16 = lambda t: t.to(torch.float16)
    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])
    to_out_w      = _to_fp16(weights["to_out.weight"])

    # Linear projections (FP16 matrix multiply)
    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w) # [B, N, N, H]

    # Optional mask (broadcast on hidden dim)
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # Apply per‑element gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣  Heavy N³ interaction – batched GEMM in Triton
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) and transpose right to match formula
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (K/J transposed)

    # Merge batch & hidden into a single batch dimension for the kernel
    bh = bs * hidden_dim
    left_flat  = left.view(bh, seq_len, seq_len)   # [B*H, N, N]  (fp16)
    right_flat = right.view(bh, seq_len, seq_len)  # [B*H, N, N]  (fp16)

    # Allocate output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Kernel launch configuration (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # M tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # N tiles
        bh,                                   # one program per (batch*hidden)
    )

    # Strides in element units
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Launch the batched GEMM kernel
    _batched_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len, seq_len, seq_len,
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Ci, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim] (float32)
    out_hidden = out_flat.view(bs, hidden_dim, seq_len, seq_len) \
                         .permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣  Hidden‑dim LayerNorm, output gate, final linear (FP16)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H]  (float32)

    # Apply output gate (gate computed in FP16 → cast to FP32)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Cast to FP16 for the final projection (Tensor‑Core friendly)
    out_norm_fp16 = out_norm.to(torch.float16)

    # Final linear: hidden -> dim  (use cuBLAS via torch.nn.functional.linear)
    # Weight shape expected: (out_features, in_features) = (dim, hidden_dim)
    # Result is kept in FP16 then cast back to FP32 as required
    result_fp16 = torch.nn.functional.linear(out_norm_fp16, to_out_w)   # [B,N,N,dim]
    result = result_fp16.to(torch.float32)  # final output dtype

    return result
