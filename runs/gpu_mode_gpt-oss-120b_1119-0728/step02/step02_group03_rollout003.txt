    c_batch_stride, c_stride_m, c_stride_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ Compute C = A @ Bᵀ  for many (batch, hidden) pairs.

    A: (batch, M, K)  fp16
    B: (batch, K, N)  fp16   (i.e. B is already transposed in memory)
    C: (batch, M, N)  fp32
    """
    batch_id = tl.program_id(0)          # which (batch, hidden) pair
    pid_m    = tl.program_id(1)          # tile row
    pid_n    = tl.program_id(2)          # tile column

    # ---- tile offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- base pointers for this batch ----
    a_base = a_ptr + batch_id * a_batch_stride
    b_base = b_ptr + batch_id * b_batch_stride
    c_base = c_ptr + batch_id * c_batch_stride

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A tile  (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + (offs_m[:, None] * a_stride_m) + (cur_k[None, :] * a_stride_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # fp16

        # B tile  (BLOCK_K, BLOCK_N) – already transposed
        b_ptrs = b_base + (cur_k[:, None] * b_stride_k) + (offs_n[None, :] * b_stride_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # fp16

        # Tensor‑core dot: fp16 → fp32 accumulation
        acc += tl.dot(a, b)              # (BLOCK_M, BLOCK_N), fp32

    # write back
    c_ptrs = c_base + (offs_m[:, None] * c_stride_m) + (offs_n[None, :] * c_stride_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of model parameters (see keys in the prompt)
        - config       : dict with keys 'dim', 'hidden_dim', optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor, shape [B, N, N, C], dtype float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, C]  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse projection + gate linear layers (fp16)
    # ------------------------------------------------------------------
    # left: concat([proj, gate])
    left_w_cat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    # right: concat([proj, gate])
    right_w_cat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # (B, N, N, 2*hidden_dim)
    left_cat = F.linear(x_h, left_w_cat)   # no bias
    right_cat = F.linear(x_h, right_w_cat)

    # split projection / gate
    left_proj = left_cat[..., :hidden_dim]                     # [B,N,N,H]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])     # [B,N,N,H]

    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Optional pairwise mask (still fp16)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_h = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5. Output gate (fp16)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6. Batched GEMM for the N³ contraction (fp16 -> fp32)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # rearrange to (B, H, N, N) where the last two axes are (i,k) and (k,j) respectively
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)  = (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  = (k,j)

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)      # fp16
    right_flat = right_perm.view(BH, N, N)    # fp16 (already transposed)

    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # strides in element units
    a_batch_stride = left_flat.stride(0)
    a_stride_m     = left_flat.stride(1)
    a_stride_k     = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_stride_k     = right_flat.stride(1)   # K dimension (already transposed)
    b_stride_n     = right_flat.stride(2)

    c_batch_stride = out_flat.stride(0)
    c_stride_m     = out_flat.stride(1)
    c_stride_n     = out_flat.stride(2)

    # Tile size tuned for H100 Tensor‑Core
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_fp16_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        a_batch_stride,
        a_stride_m,
        a_stride_k,
        b_batch_stride,
        b_stride_k,
        b_stride_n,
        c_batch_stride,
        c_stride_m,
        c_stride_n,
        N,           # M
        N,           # N
        N,           # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H) – still float32 because kernel accumulated in fp32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, H]  float32

    # ------------------------------------------------------------------
    # 8. Apply output gate and final projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(dtype=torch.float32)       # broadcast multiply
    out = F.linear(out, weights["to_out.weight"])      # [B, N, N, C]  float32

    return out
