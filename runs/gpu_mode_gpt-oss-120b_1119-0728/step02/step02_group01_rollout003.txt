"""
TriMul “outgoing’’ forward (AlphaFold‑3 style)

Algorithm
---------
1. Layer‑norm on the last (feature) axis – fp32 → fp16.
2. Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM (half‑precision).  Split the
   result and apply a sigmoid to the three gate tensors.
3. If a pair‑wise mask is present, fuse *mask*, *gate* and *projection*
   with a tiny Triton kernel (`mask_gate_fuse`).  When no mask is used we
   simply multiply the projections by the gates in PyTorch.
4. Core O(N³) step:
   *   left : (B,N,N,H)  → (B·H,N,N)
   *   right: (B,N,N,H)  → (B·H,N,N)   (the last two axes are swapped)
   *   Use a strided‑batched cuBLAS GEMM via `torch.bmm`, which on an
       H100 runs at >1 TFLOP/s for FP16.
5. Output layer‑norm over the hidden dimension, multiplication by the
   output‑gate, and a final linear projection back to the original `dim`.
6. Return FP32 tensor (the reference model works in FP32).

Only the mask‑gate fusion is written in Triton – all heavy linear algebra
is delegated to highly‑optimised cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton: fuse mask (if any) and gating into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,                       # (B·N·N) or dummy
    left_out_ptr, right_out_ptr,
    total_elems: tl.constexpr,
    hidden_dim: tl.constexpr,
    mask_present: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise computation:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
    mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # index of the mask element (broadcast over H)
    mask_idx = offs // hidden_dim

    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if mask_present:
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    else:
        m = 1.0

    tl.store(left_out_ptr + offs, lp * lg * m, mask=mask)
    tl.store(right_out_ptr + offs, rp * rg * m, mask=mask)


# ----------------------------------------------------------------------
# Helper: batched matrix multiplication via cuBLAS (torch.bmm)
# ----------------------------------------------------------------------
def _batched_matmul_cublas(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : (B·H, N, N)  – half precision
    returns      : (B·H, N, N)  – half precision
    """
    # right is already transposed (K,N) → (N,K) for the matmul
    return torch.bmm(left, right)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul “outgoing’’.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -------------------------- unpack ------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ------------------- 1) input LayerNorm -----------------------
    # LayerNorm in fp32 for stability, then cast to fp16
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # ------------------- 2) fused linear -------------------------
    # Cast all five projection/gate weights to fp16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0
    )  # (5·H, dim)

    # One GEMM → (B, N, N, 5·H)
    fused = F.linear(x, fused_weight)  # fp16
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=-1
    )
    del fused, fused_weight, x  # free memory

    # ------------------- 3) optional mask + gating -----------------
    # out_gate needs sigmoid regardless of mask
    out_gate = out_gate.sigmoid()

    if not nomask and mask_tensor is not None:
        # Triton kernel fuses mask, sigmoid‑gates and projections
        mask = mask_tensor.to(torch.float16)  # (B, N, N)
        total = left_proj.numel()  # B·N·N·H
        BLOCK = 1024  # reasonable atomic block size

        left_out = torch.empty_like(left_proj)
        right_out = torch.empty_like(right_proj)

        grid = (triton.cdiv(total, BLOCK),)
        mask_gate_fuse_kernel[grid](
            left_proj,
            left_gate,
            right_proj,
            right_gate,
            mask,
            left_out,
            right_out,
            total,
            hidden_dim,
            mask_present=1,
            BLOCK_SIZE=BLOCK,
        )
        left = left_out
        right = right_out
        # release unused tensors
        del left_proj, left_gate, right_proj, right_gate, mask
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate
        del left_proj, left_gate, right_proj, right_gate

    # ------------------- 4) core O(N³) multiplication -------------
    B, N, _, H = left.shape  # (B, N, N, H)

    # reshape to (B·H, N, N) for cuBLAS batched GEMM
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with last two axes swapped

    BH = B * H
    left_flat = left_t.view(BH, N, N)   # (B·H, N, K)
    right_flat = right_t.view(BH, N, N)  # (B·H, K, N) because of the previous transpose

    # Batched matmul – this is a strided‑batched cuBLAS call
    out_flat = _batched_matmul_cublas(left_flat, right_flat)  # (B·H, N, N)

    # reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    )  # (B, N, N, H)

    # ------------------- 5) output LayerNorm --------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )  # fp16

    # ------------------- 6) apply out‑gate ----------------------
    out = out * out_gate
    del out_gate

    # ------------------- 7) final projection --------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    result = F.linear(out, to_out_w)  # (B, N, N, dim) fp16

    # ------------------- 8) cast back to fp32 -------------------
    return result.to(torch.float32)
