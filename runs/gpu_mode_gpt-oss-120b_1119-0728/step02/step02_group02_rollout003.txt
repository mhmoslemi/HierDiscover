"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The operator works on a tensor x of shape [B, N, N, C] (C = dim) and
produces a tensor of the same shape.  The computation consists of:

1️⃣  Layer‑norm over the input channel dimension (float32).
2️⃣  Two linear projections (left/right) to a hidden size H, optional mask
    application and per‑channel sigmoid gates – all performed in FP16.
3️⃣  Heavy N³ interaction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM (B·H independent matrix multiplies of size N×N)
   implemented in Triton.  Accumulation is FP32 for numeric stability.
4️⃣  Layer‑norm over the hidden dimension, an output gate and a final
    projection back to dim – performed with PyTorch (FP16 matmul).

The heavy N³ step is the only custom Triton kernel; the rest of the
pipeline uses highly tuned PyTorch/cuBLAS kernels, which together give a
sub‑millisecond runtime on H100 for the test configurations.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul interaction
# ----------------------------------------------------------------------
@triton.jit
def _batched_trimul_kernel(
    A_ptr,                # fp16   [B*H, M, K]   (left)
    B_ptr,                # fp16   [B*H, K, N]   (rightᵀ)
    C_ptr,                # fp32   [B*H, M, N]   (output)
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,   # strides for A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides for B (batch, row(K), col(N))
    stride_Cb, stride_Ci, stride_Cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A:[M,K], B:[K,N]) for many batches."""
    pid_m = tl.program_id(axis=0)          # block row
    pid_n = tl.program_id(axis=1)          # block column
    pid_b = tl.program_id(axis=2)          # batch*hidden index

    # ------------------------------------------------------------------
    # Offsets in each dimension
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this batch
    a_ptrs = (A_ptr + pid_b * stride_Ab
              + offs_m[:, None] * stride_Ai
              + offs_k[None, :] * stride_Ak)
    b_ptrs = (B_ptr + pid_b * stride_Bb
              + offs_k[:, None] * stride_Bk
              + offs_n[None, :] * stride_Bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    num_blocks_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_blocks_k):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k < K),
                    other=0.0)                     # fp16
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)                     # fp16
        acc += tl.dot(a, b)                       # fp16×fp16 → fp32
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = (C_ptr + pid_b * stride_Cb
              + offs_m[:, None] * stride_Ci
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask          : torch.Tensor [B, N, N]    (bool/float) or None
        - weights       : dict of torch.Tensors (all on CUDA)
        - config        : dict with keys
              * "dim"        – input channel dimension (C)
              * "hidden_dim" – hidden dimension (H)
              * "nomask"     – bool, True ⇒ ignore mask (default False)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gates (FP16)
    # ------------------------------------------------------------------
    # Cast normalized tensor to half‑precision (Tensor‑Core friendly)
    x_fp16 = x_norm.to(torch.float16)

    # Helper to ensure weights are half‑precision
    def _to_fp16(t): return t.to(torch.float16)

    left_proj_w   = _to_fp16(weights["left_proj.weight"])
    right_proj_w  = _to_fp16(weights["right_proj.weight"])
    left_gate_w   = _to_fp16(weights["left_gate.weight"])
    right_gate_w  = _to_fp16(weights["right_gate.weight"])
    out_gate_w    = _to_fp16(weights["out_gate.weight"])
    to_out_w      = _to_fp16(weights["to_out.weight"])

    # ---------- projections ----------
    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B,N,N,H] fp16
    right = torch.nn.functional.linear(x_fp16, right_proj_w) # [B,N,N,H] fp16

    # ---------- optional mask ----------
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)      # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # ---------- gates ----------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_fp16, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_fp16, out_gate_w))

    # ---------- apply gates ----------
    left  = left  * left_gate   # fp16
    right = right * right_gate  # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Heavy N³ interaction – batched GEMM in Triton
    # ------------------------------------------------------------------
    # Rearrange to [B, H, N, N] (left: (i,k), right: (k,j))
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    # Merge batch and hidden dims for the kernel
    bh = bs * hidden_dim
    left_flat  = left.view(bh, seq_len, seq_len).contiguous()   # [B*H, N, N] (M,K)
    right_flat = right.view(bh, seq_len, seq_len).contiguous()  # [B*H, N, N] (K,N)

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Heuristic block sizes
    if seq_len <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # rows
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # cols
        bh,                                   # batch*hidden
    )

    # Strides in elements
    stride_Ab = left_flat.stride(0)
    stride_Ai = left_flat.stride(1)
    stride_Ak = left_flat.stride(2)

    stride_Bb = right_flat.stride(0)
    stride_Bk = right_flat.stride(1)
    stride_Bn = right_flat.stride(2)

    stride_Cb = out_flat.stride(0)
    stride_Ci = out_flat.stride(1)
    stride_Cn = out_flat.stride(2)

    # Launch the Triton kernel (8 warps, 3 stages → good for H100)
    _batched_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,
        seq_len,
        seq_len,
        stride_Ab, stride_Ai, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Ci, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Reshape back to [B, N, N, H] (float32)
    out_hidden = (
        out_flat.view(bs, hidden_dim, seq_len, seq_len)
                .permute(0, 2, 3, 1)
                .contiguous()
    )  # [B, N, N, H]  (float32)

    # ------------------------------------------------------------------
    # 4️⃣ Layer‑norm over hidden, apply output gate, final projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # Fuse the output gate (still half‑precision) before the final linear
    # Both operands are cast to FP16 to let Tensor‑Core handle the matmul.
    out_norm_fp16 = (out_norm.to(torch.float16) * out_gate)  # [B,N,N,H] fp16

    # Final linear: (B·N·N, H) @ (dim, H)^T → (B·N·N, dim)
    M = bs * seq_len * seq_len
    out_flat_fp16 = out_norm_fp16.view(M, hidden_dim)        # [M, H] fp16
    result_fp16 = torch.nn.functional.linear(out_flat_fp16, to_out_w)  # [M, dim] fp16

    # Reshape to the original 4‑D layout and cast back to FP32
    result = result_fp16.view(bs, seq_len, seq_len, dim).to(torch.float32)

    return result
```
