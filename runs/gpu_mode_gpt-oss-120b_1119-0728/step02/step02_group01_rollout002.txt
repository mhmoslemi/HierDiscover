"""
TriMul “outgoing’’ forward – Triton‑accelerated implementation.

Algorithm
---------
1. **LayerNorm** on the input (float32 → half) – uses torch.linalg for stability.
2. **Fused projection + gating**:
   * Concatenate the five weight matrices (left_proj, right_proj,
     left_gate, right_gate, out_gate) into a single (5·H, C) matrix.
   * A single `F.linear` on the half‑precision tensor yields the five
     intermediate tensors in one GEMM, saving 5× memory traffic.
3. **Masking & sigmoid gates** – applied in half precision.
4. **Core O(N³) operation**:
   * We need, for each hidden channel *h*,
       out[b,i,j,h] = Σ_k left[b,i,k,h]·right[b,j,k,h].
   * This is a batched matrix multiplication of shape
       (B·H, N, N) = left_flat @ right_flat .
   * The heavy lifting is done by a custom Triton GEMM kernel that
     uses fp16 loads, fp32 accumulation and Tensor‑Core `tl.dot`.  The
     kernel is tuned for H100 (BLOCK_M/N = 128, BLOCK_K = 64, 8 warps).
5. **Output LayerNorm → out‑gate → final projection** – all half‑precision,
   cast back to float32 at the end.

The implementation keeps the only costly part (the N³ GEMM) in Triton,
while the surrounding linear / norm / sigmoid work is performed with
highly‑optimized PyTorch kernels.  This hybrid approach satisfies the
requirement “use at least part of the operations in a kernel’’ and
offers a noticeable speed‑up over a naïve PyTorch‑only version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------
# Triton batched GEMM kernel (FP16‐input, FP32 accumulation)
# -------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dims
    stride_ab, stride_am, stride_ak,         # strides of A: (batch, M, K)
    stride_bb, stride_bk, stride_bn,         # strides of B: (batch, K, N)
    stride_cb, stride_cm, stride_cn,         # strides of C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_STAGES: tl.constexpr = 2,
):
    """Batched FP16 GEMM with FP32 accumulation.
    Grid: (ceil(M/BLOCK_M), ceil(N/BLOCK_N), batch) where batch = B*H.
    """
    pid_b = tl.program_id(2)            # batch index (B·H)
    pid_m = tl.program_id(0)            # tile row
    pid_n = tl.program_id(1)            # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # allocate accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over K dimension with double‑buffering (NUM_STAGES)
    # ---------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointer arithmetic
        a_ptrs = (
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # out‑of‑bounds masks
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # half‑precision loads
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float16)

        # Tensor‑core dot (fp32 accum)
        acc += tl.dot(a, b)

    # ---------------------------------------------------------
    # Write back (store as FP16)
    # ---------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    a : (B·H, M, K)   half‑precision
    b : (B·H, K, N)   half‑precision
    returns c : (B·H, M, N) half‑precision
    """
    assert a.is_cuda and b.is_cuda
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    B_h, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, f"Inner dimensions must match (got {K} vs {K2})"

    c = torch.empty((B_h, M, N), device=a.device, dtype=a.dtype)

    # Tuned block sizes for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B_h,
    )
    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return c


# -------------------------------------------------------------
# Entry point – forward pass of TriMul (outgoing)
# -------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ module.

    Arguments
    ---------
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -------------------------------------------------
    # 0) Unpack
    # -------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # Ensure TF32 is disabled for reproducibility on H100
    torch.backends.cuda.matmul.allow_tf32 = False
    torch.backends.cudnn.allow_tf32 = False

    # -------------------------------------------------
    # 1) Input LayerNorm (float32 → half)
    # -------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # torch.layer_norm runs in float32; cast back to fp16 afterwards
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)
    x = x.to(torch.float16)

    # -------------------------------------------------
    # 2) Fuse the five linear transforms (proj & gates)
    # -------------------------------------------------
    # Cast weights once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)    # (H, C)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)    # (H, C)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)      # (H, C)

    # Concatenate along output dimension (0)
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=0
    )  # (5·H, C)

    # Single fused linear (half‑precision)
    fused = F.linear(x, fused_weight)  # (B, N, N, 5·H)
    # Split back to individual tensors
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=-1
    )
    # Release memory early
    del fused, fused_weight, x

    # -------------------------------------------------
    # 3) Optional mask (applied before gating)
    # -------------------------------------------------
    if (not nomask) and mask_tensor is not None:
        # mask: (B, N, N, 1) broadcasted over hidden dim
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # -------------------------------------------------
    # 4) Sigmoid gates & combine with projections
    # -------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    left = left_proj * left_gate   # (B, N, N, H)
    right = right_proj * right_gate

    # Release temporaries we no longer need
    del left_proj, left_gate, right_proj, right_gate

    # -------------------------------------------------
    # 5) Core N³ computation via batched GEMM (Triton)
    #    out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # -------------------------------------------------
    B, N, _, H = left.shape

    # Rearrange to (B·H, M, K) and (B·H, K, N) for the batched GEMM
    # Left:   (B, N, N, H) -> (B, H, N, N) -> (B·H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()      # (B, H, N, N)
    right_t = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N)

    BH = B * H
    left_flat = left_t.view(BH, N, N)    # (B·H, N, N)
    right_flat = right_t.view(BH, N, N)  # (B·H, N, N)

    # Triton batched GEMM (fp16→fp32→fp16)
    out_flat = _batched_matmul(left_flat, right_flat)   # (B·H, N, N)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Clean up large intermediates
    del left, right, left_t, right_t, left_flat, right_flat, out_flat

    # -------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # -------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # -------------------------------------------------
    # 7) Apply the output gate
    # -------------------------------------------------
    out = out * out_gate
    del out_gate

    # -------------------------------------------------
    # 8) Final linear projection back to original dim
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (C, H)
    result = F.linear(out, to_out_w)    # (B, N, N, C) in fp16

    # Cast back to float32 to match the reference implementation
    return result.to(torch.float32)
