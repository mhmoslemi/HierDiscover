"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton implementation.
-----------------------------------------------------------------
The heavy part of the TriMul module is the 3‑D contraction:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden slice `d` this is a matrix multiplication
    L₍b,d₎ @ R₍b,d₎ᵀ          (L,R ∈ ℝ^{N×N})
with `B·H` independent GEMMs (B = batch, H = hidden_dim).
We implement this batch of GEMMs with a custom Triton kernel that
operates in FP16 (Tensor‑core friendly) and accumulates in FP32 for
numerical stability.  All other inexpensive operations (LayerNorm,
linear projections, gating) are performed in PyTorch in half‑precision.
The final result is returned as FP32 to match the original module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction.
#   A : [B*H, N, N] (row‑major, FP16)
#   B : [B*H, N, N] (row‑major, FP16) – accessed transposed
#   C : [B*H, N, N] (FP16, accumulation in FP32)
# ----------------------------------------------------------------------
@triton.jit
def _trimmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                     # tensor pointers
    stride_a_batch, stride_a_i, stride_a_k,  # A strides (batch, row, col)
    stride_b_batch, stride_b_k, stride_b_i,  # B strides (batch, k, j) – transposed view
    stride_c_batch, stride_c_i, stride_c_j,  # C strides (batch, row, col)
    N,                                        # sequence length
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_z = tl.program_id(2)  # batch * hidden slice

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Mask out‑of‑bounds elements
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # Load A block (M×K) – FP16
        a_ptrs = (
            a_ptr
            + pid_z * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # Load B block (K×N) – transposed view (k as row, n as col)
        b_ptrs = (
            b_ptr
            + pid_z * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Tensor‑core GEMM – accumulate in FP32
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result (cast back to FP16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_z * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], float32, CUDA
        - mask_tensor  : torch.Tensor, shape [B, N, N] (may be None)
        - weights_dict : dict of torch.Tensors (float32, CUDA)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], float32, CUDA
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper for weight lookup (all tensors are already on `device`)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (keep FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # Cast to FP16 before the heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left = F.linear(x_h, _w("left_proj.weight").to(torch.float16))   # [B, N, N, H]
    right = F.linear(x_h, _w("right_proj.weight").to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear layers) – still FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, _w("left_gate.weight").to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_h, _w("right_gate.weight").to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_h, _w("out_gate.weight").to(torch.float16)))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy contraction: batched GEMM (A @ Bᵀ) for each hidden slice
    # ------------------------------------------------------------------
    #   Shape before: left/right = [B, N, N, H]
    #   Permute to [B, H, N, N] and view as [B*H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_flat = left_perm.view(batch_total, N, N)   # FP16
    right_flat = right_perm.view(batch_total, N, N)

    # Allocate output buffer (FP16)
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # Triton launch parameters (tuned for H100 Tensor‑cores)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # Tensor‑core friendly K‑tile

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        batch_total,               # independent GEMM instances
    )

    _trimmul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        # A strides
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # B strides (transposed view)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # C strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to 4‑D layout: [B, N, N, H] (still FP16)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output processing: LayerNorm → gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight").to(torch.float16),
        bias=_w("to_out_norm.bias").to(torch.float16),
    )
    out = out * out_gate                                 # element‑wise gate (FP16)
    out = F.linear(out, _w("to_out.weight").to(torch.float16))   # [B, N, N, dim] (FP16)

    # ------------------------------------------------------------------
    # Cast back to FP32 as required by the original API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
