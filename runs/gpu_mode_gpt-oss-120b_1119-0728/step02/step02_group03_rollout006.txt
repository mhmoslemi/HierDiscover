def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)
    # layernorm fp32
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # cast half
    x_h = x.to(torch.float16)

    # linear not fused: left, right
    left = F.linear(x_h, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,H]
    right= F.linear(x_h, weights["right_proj.weight"].to(torch.float16))
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate= torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate  = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # rearrange
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # B, H, N, N
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = bs * hidden_dim
    left_flat = left_perm.reshape(BH, N, N)
    right_flat = right_perm.reshape(BH, N, N)

    # Batched GEMM using torch.bmm
    out_flat = torch.bmm(left_flat, right_flat)  # shape (BH, N, N), dtype=half

    # reshape back
    out = out_flat.reshape(bs, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # second layernorm half
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"].to(torch.float16), bias=weights["to_out_norm.bias"].to(torch.float16), eps=eps)

    out = out * out_gate

    # final linear: half weight
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # cast output back to float32
    out = out.to(input_tensor.dtype)

    return out

TriMul (outgoing) – fast Triton‑free forward pass.

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32) – this is the only FP32
    operation we keep for numerical stability.
2.  Cast to half (fp16) and apply the four linear projections
    (left, right and three gates) with half‑precision weights.
3.  Optionally mask the left/right tensors (still fp16).
4.  Multiply left/right by their respective sigmoid gates (fp16).
5.  Fuse the O(N³) contraction as a *batched* half‑precision matrix
    multiplication:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    which is equivalent to a batch of GEMMs of shape (N,N)×(N,N)ᵀ
    for each hidden channel d.  Using ``torch.bmm`` lets cuBLAS on H100

6.  Reshape the result back to (B,N,N,H) (still fp16).
7.  Apply the second LayerNorm, the output‑gate and the final
    projection – all in half precision.
8.  Cast the final tensor back to float32.

    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]  float32

    # ------------------------------------------------------------------
    # 3️⃣ cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ half‑precision linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(
        x_h, weights["left_proj.weight"].to(torch.float16), bias=None
    )  # [B, N, N, H]  fp16
    right = F.linear(
        x_h, weights["right_proj.weight"].to(torch.float16), bias=None
    )  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 5️⃣ optional mask (still fp16)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # mask may be bool / float; convert to half and broadcast
        mask_h = mask.to(torch.float16, device=device).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ gating (sigmoid) – keep everything in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].to(torch.float16), bias=None)
    )
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].to(torch.float16), bias=None)
    )
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].to(torch.float16), bias=None)
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ O(N³) contraction – batched GEMM via cuBLAS (torch.bmm)
    # ------------------------------------------------------------------
    #   left  : [B, N, N, H] -> [B, H, N, N] -> (BH, N, N)
    #   right : [B, N, N, H] -> [B, H, N, N] with (j,k) order -> (BH, N, N)
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    BH = B * hidden_dim
    left_flat = left_perm.reshape(BH, N, N)   # fp16
    right_flat = right_perm.reshape(BH, N, N)  # fp16

    # torch.bmm uses the best cuBLAS kernel for fp16 → fp32 accumulation
    out_flat = torch.bmm(left_flat, right_flat)  # (BH, N, N) fp16

    # reshape back to (B, N, N, H) – still fp16
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8️⃣ second LayerNorm (still fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
        eps=eps,
    )  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ output gate & final projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gating, fp16
    out = F.linear(out, weights["to_out.weight"].to(torch.float16), bias=None)

    # ------------------------------------------------------------------
    # 10️⃣ cast back to float32 – matches reference implementation
    # ------------------------------------------------------------------
    out = out.to(input_tensor.dtype)  # float32

    return out
```
