"""
TriMul “outgoing” forward pass (AlphaFold3) – optimized Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) over the feature dimension.
2. Cast to fp16 – all heavy work is performed in half precision.
3. Linear projections (no bias) → left, right (shape [B, N, N, H]) in fp16.
4. (Optional) binary mask is broadcast on the hidden dimension.
5. Gating: three linear layers + sigmoid (all in fp16) → left_gate,
   right_gate, out_gate.  Multiply left/right by their gates.
6. Core contraction (O(N³)):
        out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   Implemented as a batched half‑precision GEMM (left @ rightᵀ) for each
   hidden channel using a Triton kernel with fp32 accumulation.
7. Convert to fp32, apply LayerNorm over the hidden dimension, multiply by
   out_gate, and finally project back to the original feature dimension.
   The final result is float32.

The heavy O(N³) part is fully fused in Triton, while the surrounding
operations (layer‑norm, linear projections, gating, final projection) are
performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM with fp32 accumulation.
#   A : (batch, M, K)  – fp16
#   B : (batch, K, N)  – fp16
#   C : (batch, M, N)  – fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # matrix dims
    stride_ab, stride_am, stride_ak,  # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,  # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,  # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)      # hidden‑batch dimension (B*H)
    pid_m = tl.program_id(0)      # tile‑row
    pid_n = tl.program_id(1)      # tile‑col

    # -------------------------------------------------
    # Offsets of the current tile
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # fp32 accumulator (prevents overflow, keeps precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over K dimension in BLOCK_K chunks
    # -------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers for the current tile
        a_ptrs = (
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Out‑of‑bounds masking
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # Tensor‑core friendly dot (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Write back the result (cast to fp16)
    # -------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision matrix multiplication using the Triton kernel.
    a : (B*H, M, K)  fp16
    b : (B*H, K, N)  fp16
    Returns c : (B*H, M, N) fp16
    """
    assert a.is_cuda and b.is_cuda
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    BH, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((BH, M, N), device=a.device, dtype=torch.float16)

    # Tuned tile sizes for H100 (fit in shared memory + registers)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask_tensor  : torch.Tensor [B, N, N] or None (float32/bool)
            weights_dict : dict of torch.Tensor parameters
            config_dict  : dict with keys
                 "dim", "hidden_dim", optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_fp16 = x.half()                     # [B, N, N, dim]  (fp16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H] (fp16)
    right = F.linear(x_fp16, right_proj_w) # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 4) Optional binary mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        # mask: [B, N, N] -> [B, N, N, 1] -> fp16
        mask = mask_tensor.to(x_fp16.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – stay in fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = F.linear(x_fp16, left_gate_w).sigmoid()   # (fp16)
    right_gate = F.linear(x_fp16, right_gate_w).sigmoid() # (fp16)
    out_gate = F.linear(x_fp16, out_gate_w).sigmoid()     # (fp16)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³) contraction via batched GEMM
    #    out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Rearrange to (B, H, N, N) → (B*H, N, N) for each operand
    # left : rows = i, cols = k
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right : rows = k, cols = j  (so we need rightᵀ)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) with axes (k, j)

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N) # (BH, K=N, N)

    # Batched half‑precision matmul (left @ right)
    out_flat = _batched_matmul(left_flat, right_flat)  # (BH, N, N) fp16

    # Restore [B, N, N, H] layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H]  float32

    # ------------------------------------------------------------------
    # 8) Apply output gate (float32)
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.to(torch.float32)   # broadcastable
    out_gated = out_norm * out_gate_fp32

    # ------------------------------------------------------------------
    # 9) Final projection back to original dim (float32)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    result = F.linear(out_gated, to_out_w)      # [B, N, N, dim] float32

    return result
