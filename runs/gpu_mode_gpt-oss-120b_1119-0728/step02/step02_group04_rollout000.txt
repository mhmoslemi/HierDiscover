"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3).

Algorithm
---------
1. LayerNorm over the feature dimension (float32) → x.
2. Cast x to fp16 for the costly path.
3. Linear projections (no bias) → left, right  (B,N,N,hidden_dim) in fp16.
4. Optional binary mask (broadcast on hidden_dim) applied to left/right.
5. Gating: three linear layers + sigmoid (fp16) → left_gate, right_gate,
   out_gate.  Multiply left/right by their gates.
6. Core O(N³) contraction:
        out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   Implemented as a batched half‑precision GEMM (tensor cores) with fp32
   accumulation.  The batch dimension is B·hidden_dim, thus a single
   Triton kernel processes all hidden slices.
7. Convert to float32, apply a second LayerNorm over the hidden dimension,
   multiply by out_gate and a final linear projection back to the original
   feature dimension (float32).

All heavy arithmetic is done in fp16 (except the two LayerNorms for
numerical stability).  The kernel is tuned for H100 (BLOCK_M=N=BLOCK_N=128,
BLOCK_K=64).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (fp16 × fp16 → fp32 accumulation)
#   A : (batch, M, K)   – fp16
#   B : (batch, K, N)   – fp16
#   C : (batch, M, N)   – fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,           # pointers
    M, N, K,                       # matrix dims
    stride_ab, stride_am, stride_ak,  # A strides  (batch, M, K)
    stride_bb, stride_bk, stride_bn,  # B strides  (batch, K, N)
    stride_cb, stride_cm, stride_cn,  # C strides  (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM with fp32 accumulation."""
    pid_b = tl.program_id(2)           # batch (B * hidden_dim)
    pid_m = tl.program_id(0)           # tile row
    pid_n = tl.program_id(1)           # tile col

    # --------------------------------------------------------------
    # Offsets for the output tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers for the current tile of A and B
        a_ptrs = (
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Out‑of‑bounds masks
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # Tensor‑core dot (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back result in fp16
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision matrix multiplication using the Triton kernel.
    a: (B*H, M, K) fp16
    b: (B*H, K, N) fp16
    Returns c: (B*H, M, N) fp16
    """
    assert a.is_cuda and b.is_cuda
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    BH, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((BH, M, N), device=a.device, dtype=torch.float16)

    # Tuned tile sizes for H100 (fits in shared memory + registers)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N] or None (float32/bool)
            weights_dict : dict of torch.Tensor model parameters
            config_dict  : {"dim": int, "hidden_dim": int,
                            "nomask": bool (optional)}

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # --------------------------------------------------------------
    # 2) Cast to fp16 for the heavy part
    # --------------------------------------------------------------
    x_fp16 = x.half()                     # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 3) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())   # (B,N,N,hidden)
    right = F.linear(x_fp16, weights["right_proj.weight"].half()) # (B,N,N,hidden)

    # --------------------------------------------------------------
    # 4) Optional binary mask (broadcast on hidden_dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask_tensor is not None):
        mask = mask_tensor.to(x_fp16.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5) Gating (sigmoid) – keep in fp16
    # --------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].half()).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Core contraction via batched GEMM
    #    out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    # --------------------------------------------------------------
    B, N, _, _ = left.shape

    # Rearrange to (B, hidden, N, N) so that the contraction becomes a GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    # right needs (B, H, k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    # Collapse batch+hidden into a single batch dimension for the kernel
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N) # (BH, K=N, N)

    out_flat = _batched_matmul(left_flat, right_flat)  # (BH, N, N) fp16

    # Restore (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension (float32)
    # --------------------------------------------------------------
    out_fp32 = out.to(torch.float32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, hidden] float32

    # --------------------------------------------------------------
    # 8) Apply output gate
    # --------------------------------------------------------------
    out_gated = out_norm * out_gate.to(torch.float32)

    # --------------------------------------------------------------
    # 9) Final projection back to original dim (float32)
    # --------------------------------------------------------------
    result = F.linear(out_gated, weights["to_out.weight"])  # (B,N,N,dim)

    return result
