    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False); eps = 1e-5

    # Layer norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Cast to fp16
    x_h = x.half()

    # Fuse all linear weights
    # leftproj, leftgate, rightproj, rightgate, outgate
    fused_weight = torch.cat([
        weights["left_proj.weight"],    # (hidden, dim)
        weights["left_gate.weight"],    # (hidden, dim)
        weights["right_proj.weight"],   # (hidden, dim)
        weights["right_gate.weight"],   # (hidden, dim)
        weights["out_gate.weight"],     # (hidden, dim)
    ], dim=0).half()   # shape (5*hidden, dim)

    fused = torch.nn.functional.linear(x_h, fused_weight)  # shape [B,N,N,5*hidden]

    h = hidden_dim
    left_proj = fused[..., 0:h]
    left_gate = torch.sigmoid(fused[..., h:2*h])
    right_proj = fused[..., 2*h:3*h]

@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(1)  # row tile
    pid_n = tl.program_id(0)  # col tile
    pid_b = tl.program_id(2)  # batch task

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # batch offset base pointer
    a_ptrs_base = a_ptr + pid_b * stride_ab
    b_ptrs_base = b_ptr + pid_b * stride_bb
    c_ptrs_base = c_ptr + pid_b * stride_cb

    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # compute a indices
        a_ptrs = a_ptrs_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        # compute b indices
        b_ptrs = b_ptrs_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # load with masks
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptrs_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the input tensor (float32) → `x`.
2. Cast to fp16 and compute a *single* linear projection that yields
   all required tensors:
        left_proj, left_gate, right_proj, right_gate, out_gate
   (concatenated weights of size 5·hidden_dim × dim).
3. Apply the optional 2‑D mask to `left_proj` and `right_proj`.
4. Fuse gates with a sigmoid and compute
        left  = left_proj  * left_gate
        right = right_proj * right_gate
5. Reshape `left` / `right` to (B·hidden, N, N) and transpose `right`
   so each task performs a matrix‑multiply  left @ rightᵀ.
   The batched multiplications are carried out by a custom Triton GEMM
   kernel (fp16 → fp32 accumulation).
6. Reshape back to [B, N, N, hidden] and layer‑norm the result.
7. Fuse the final element‑wise multiplication with the output‑gate
   via a tiny Triton kernel.
8. Apply the output linear projection (hidden → dim) to obtain the
   final tensor of shape [B, N, N, dim] (float32).

The heavy N³ operation (step 5) and the final element‑wise
multiplication (step 7) are the only parts executed on Triton;
all other ops use highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (fp16 inputs → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix dimensions (all equal to N)
    stride_am, stride_ak, stride_ab,  # strides for A (row, col, batch)
    stride_bk, stride_bn, stride_bb,  # strides for B (row, col, batch)
    stride_cm, stride_cn, stride_cb,  # strides for C (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ B   for a batch of independent matrices.
    A : [batch, M, K]   (fp16)
    B : [batch, K, N]   (fp16)   – already transposed so that the
    multiplication implements ∑_k left[..., i,k] * right[..., j,k].
    C : [batch, M, N]   (fp16, stored as fp16, accumulation in fp32)
    """
    pid_m = tl.program_id(1)      # tile index in M dimension
    pid_n = tl.program_id(0)      # tile index in N dimension
    pid_b = tl.program_id(2)      # batch/task index

    # ---- compute tile offsets ------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_base = a_ptr + pid_b * stride_ab
    b_base = b_ptr + pid_b * stride_bb
    c_base = c_ptr + pid_b * stride_cb

    # accumulator in fp32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K ---------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k_idx in range(num_k):
        k_offset = k_idx * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for out‑of‑bounds k
        k_mask = offs_k < K

        # pointers to the current tiles of A and B
        a_ptrs = a_base + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        b_ptrs = b_base + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)

        # load tiles with bounds checking
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(a, b)   # fp16 × fp16 → fp32

    # ---- write back -----------------------------------------------------------
    c_ptrs = c_base + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    c = acc.to(tl.float16)    # store as fp16
    tl.store(c_ptrs,
             c,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply (used for out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, c_ptr, size, BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point: custom Triton‑accelerated TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (bool/float) or None
        - weights      : dict of torch.Tensors (float32)
        - config       : dict containing "dim", "hidden_dim", and optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------
    # unpack
    # -------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # -------------------------------------------------
    # 1) LayerNorm over the channel dimension (float32)
    # -------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                         # [B, N, N, dim]  (fp32)

    # -------------------------------------------------
    # 2) Cast to fp16 for the rest of the heavy work
    # -------------------------------------------------
    x_h = x.half()            # [B, N, N, dim] (fp16)

    # -------------------------------------------------
    # 3) One fused linear that produces all projections & gates
    #    Order: left_proj | left_gate | right_proj | right_gate | out_gate
    # -------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                  # shape (5*hidden, dim)

    fused = torch.nn.functional.linear(x_h, fused_weight)   # [B, N, N, 5*hidden] (fp16)

    # -------------------------------------------------
    # 4) Split and apply sigmoid to gate vectors
    # -------------------------------------------------
    h = hidden
    left_proj = fused[..., 0:h]
    left_gate = torch.sigmoid(fused[..., h:2 * h])
    right_proj = fused[..., 2 * h:3 * h]
    right_gate = torch.sigmoid(fused[..., 3 * h:4 * h])
    out_gate = torch.sigmoid(fused[..., 4 * h:5 * h])     # [B, N, N, hidden] (fp16)

    # -------------------------------------------------
    # 5) Optional mask (applied BEFORE gating, as in the reference)
    # -------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1] fp16
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # -------------------------------------------------
    # 6) Apply element‑wise gates
    # -------------------------------------------------
    left = left_proj * left_gate                     # [B, N, N, hidden] fp16
    right = right_proj * right_gate

    # -------------------------------------------------
    # 7) Batched matrix multiplication (N³) using Triton
    #    out[b, i, j, :] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    # -------------------------------------------------
    total_tasks = B * hidden

    # reshape such that each (batch, hidden) pair is a separate GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(total_tasks, N, N)      # [T, N, N] fp16
    # right needs to be transposed on the last two axes (k ↔ j)
    right_T = right.permute(0, 3, 2, 1).contiguous().view(total_tasks, N, N)      # [T, N, N] fp16

    # allocate output (fp16); will be cast to fp32 later
    out_flat_fp16 = torch.empty_like(left_flat)

    # ------- kernel launch parameters -----------------------------------------
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_T.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat_fp16.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_N),   # N‑tiles (column)
        triton.cdiv(N, BLOCK_M),   # N‑tiles (row)
        total_tasks,               # independent GEMMs
    )

    batch_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat_fp16,
        N,
        N,
        N,
        stride_a_row,
        stride_a_col,
        stride_a_batch,
        stride_b_row,
        stride_b_col,
        stride_b_batch,
        stride_c_row,
        stride_c_col,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # -------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden] and cast to fp32
    # -------------------------------------------------
    out = out_flat_fp16.view(B, hidden, N, N).permute(0, 2, 3, 1).float()   # [B,N,N,hidden]

    # -------------------------------------------------
    # 9) LayerNorm over the hidden dimension (float32)
    # -------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B,N,N,hidden] (fp32)

    # -------------------------------------------------
    # 10) Fuse out_norm * out_gate (both now float32) via Triton
    # -------------------------------------------------
    out_gate_f32 = out_gate.float()
    gated = torch.empty_like(out_norm)            # fp32 buffer
    total_elems = out_norm.numel()
    BLOCK = 8192
    grid_mul = (triton.cdiv(total_elems, BLOCK),)
    elementwise_mul_kernel[grid_mul](
        out_norm,
        out_gate_f32,
        gated,
        total_elems,
        BLOCK_SIZE=BLOCK,
    )

    # -------------------------------------------------
    # 11) Final linear projection back to `dim`
    # -------------------------------------------------
    result = torch.nn.functional.linear(gated, weights["to_out.weight"])

    return result
