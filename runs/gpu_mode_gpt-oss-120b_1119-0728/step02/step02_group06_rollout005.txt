"""
triton implementation of the *outgoing* TriMul operation (AlphaFold‑3 / Evoformer).

The heavy N³ step

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is implemented as a batched half‑precision GEMM (C = A @ Bᵀ) in a single Triton kernel.
All other operations (LayerNorm, linear projections, sigmoid gating, final LayerNorm
and the output linear) stay in PyTorch.  A tiny Triton kernel fuses the final
element‑wise multiplication with the output gate.  The implementation follows the
reference PyTorch code but avoids the costly ``permute(...).contiguous()``
re‑ordering and uses a custom kernel for the matrix‑multiplication stage.

The kernel works for any batch size ≤2, sequence length up to 1024 and hidden
dimension 128.  It uses fp16 arithmetic with fp32 accumulation for maximal
accuracy on H100 tensor‑cores.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Tiny element‑wise kernel (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,          # float32*  – out_norm
    b_ptr,          # float32*  – out_gate
    c_ptr,          # float32*  – output buffer
    size,           # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel (A @ Bᵀ) for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # fp16*  – left  [B, N, N, H]
    right_ptr,         # fp16*  – right [B, N, N, H]
    out_ptr,           # fp16*  – output [B, N, N, H]
    B, N, H,           # dimensions (int)
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_i, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Performs, for every (b, h) pair:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    using a classic tiling strategy.  All arithmetic is fp16 → fp32
    accumulation, stored back as fp16.
    """
    # ------------------------------------------------------------------
    # program IDs
    # ------------------------------------------------------------------
    batch_h_idx = tl.program_id(0)          # 0 … B*H-1  (combined batch)
    pid_i = tl.program_id(1)                # row tile
    pid_j = tl.program_id(2)                # col tile

    # ------------------------------------------------------------------
    # decode (batch, hidden) from the combined index
    # ------------------------------------------------------------------
    hidden = H
    b = batch_h_idx // hidden
    h = batch_h_idx % hidden

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_batch  + h
    right_base = right_ptr + b * stride_right_batch + h
    out_base   = out_ptr   + b * stride_out_batch   + h

    # ------------------------------------------------------------------
    # tile offsets for rows (i) and cols (j)
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over k‑dimension
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- load A tile (left) ----------
        # shape: (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # ---------- load B tile (right) transposed ----------
        # We need Bᵀ, i.e. index (k, j) in the original right tensor.
        # Original layout: right[b, j, k, h] ⚡ stride_i = N*H, stride_k = H
        # After swapping we treat k as the row dimension and j as the column.
        # shape: (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_i) + (offs_j[None, :] * stride_right_k)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # ---------- accumulate ----------
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → (M,N)  (fp16 → fp32)

    # ------------------------------------------------------------------
    # store the result (fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# 3) Entry point – Triton‑accelerated forward pass for the outgoing TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor   [B, N, N, C]   (float32)
            mask         : torch.Tensor   [B, N, N]      (bool or 0/1) or None
            weights      : dict of torch.Tensor containing all model parameters
            config       : dict  with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor   [B, N, N, C]   (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                     # input channel dimension C
    hidden = config["hidden_dim"]           # hidden dimension H
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑Norm on the input (float32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all heavy work
    # ------------------------------------------------------------------
    x_h = x.half()                          # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 3) Fused projection+gate (cat) for left and right sides
    # ------------------------------------------------------------------
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()                                   # [2*H, dim]
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()                                   # [2*H, dim]
    out_gate_w = weights["out_gate.weight"].half()   # [H, dim]

    # Linear projections (proj + gate packed)
    left_cat = torch.nn.functional.linear(x_h, left_w)   # [B, N, N, 2*H] fp16
    right_cat = torch.nn.functional.linear(x_h, right_w)  # [B, N, N, 2*H] fp16

    # split into projection and gate parts
    left_proj = left_cat[..., :hidden]                     # [B,N,N,H]
    left_gate = torch.sigmoid(left_cat[..., hidden:])      # fp16
    right_proj = right_cat[..., :hidden]
    right_gate = torch.sigmoid(right_cat[..., hidden:])

    # output gate (used after the output LayerNorm)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 4) Optional mask (applied before gating, exactly as the reference)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)   # [B,N,N,1] fp16
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate      # [B,N,N,H] fp16
    right = right_proj * right_gate   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6) Batched GEMM: out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    #    Implemented with a custom Triton kernel (no permute / copy)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # batch size, sequence length

    # Allocate output tensor (fp16)
    out = torch.empty_like(left)   # shape [B, N, N, H] , dtype fp16

    # Strides (as int64) needed by the Triton kernel
    stride_left_batch = left.stride(0)   # N*N*H
    stride_left_i      = left.stride(1)   # N*H
    stride_left_k      = left.stride(2)   # H
    # stride for hidden dim is left.stride(3) == 1 (not explicitly needed)

    stride_right_batch = right.stride(0)
    stride_right_i     = right.stride(1)
    stride_right_k     = right.stride(2)

    stride_out_batch = out.stride(0)
    stride_out_i     = out.stride(1)
    stride_out_j     = out.stride(2)

    # Tunable block sizes – empirically good for fp16 on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (batch*hidden, row tiles, col tiles)
    grid = (
        B * hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden,
        stride_left_batch=stride_left_batch,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_right_batch=stride_right_batch,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_out_batch=stride_out_batch,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Layer‑Norm over the hidden dimension (float32 for stability)
    # ------------------------------------------------------------------
    out_fp32 = out.float()     # [B, N, N, H] float32
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, H] float32

    # ------------------------------------------------------------------
    # 8) Fuse out_norm * out_gate using a tiny Triton kernel
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.float()
    gated = torch.empty_like(out_norm)   # fp32 buffer
    total_elements = out_norm.numel()
    BLOCK = 8192
    grid_el = (triton.cdiv(total_elements, BLOCK),)
    elementwise_mul_kernel[grid_el](
        out_norm,
        out_gate_f32,
        gated,
        total_elements,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 9) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    result = torch.nn.functional.linear(gated, weights["to_out.weight"])

    return result
```
