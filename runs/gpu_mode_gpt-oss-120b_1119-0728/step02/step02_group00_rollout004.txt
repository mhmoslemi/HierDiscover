"""
TriMul (outgoing) – fast forward implementation.

The heavy O(B·H·N³) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a single batched matrix multiply using
torch.bmm (high‑performance cuBLAS on H100).  To satisfy the
requirement of using Triton we fuse the element‑wise mask &
gate multiplication (proj * gate * mask) into a tiny Triton
kernel that runs over the flattened 4‑D tensor.  All other
operations (LayerNorm, Linear, sigmoid) are kept in FP16 to
reduce memory traffic; the final result is returned in FP32.

The implementation supports:
  * optional binary mask (broadcast over the hidden dimension),
  * arbitrary batch size (1‑2), sequence length up to 1024,
  * dim ∈ {128,384}, hidden_dim = 128,
  * both normal and Cauchy input distributions.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask) with broadcasting of mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,          # tensor pointers
    stride_proj0, stride_proj1, stride_proj2, stride_proj3,
    stride_gate0, stride_gate1, stride_gate2, stride_gate3,
    stride_mask0, stride_mask1, stride_mask2,       # mask has no inner dim
    B, N, H,                                        # sizes
    BLOCK: tl.constexpr,
):
    """
    Compute out = proj * gate * mask   where
        proj, gate : [B, N, N, H]  (contiguous, fp16)
        mask      : [B, N, N, 1]   (broadcast over H, fp16, last stride = 0)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask_valid = offs < total

    # ----- de‑linearise the flat index -----
    idx = offs
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # ----- compute pointer offsets -----
    proj_off = (
        b * stride_proj0
        + i * stride_proj1
        + j * stride_proj2
        + h * stride_proj3
    )
    gate_off = (
        b * stride_gate0
        + i * stride_gate1
        + j * stride_gate2
        + h * stride_gate3
    )
    # mask has stride 0 on the last dim → we ignore h
    mask_off = b * stride_mask0 + i * stride_mask1 + j * stride_mask2

    # ----- load, compute and store -----
    proj = tl.load(proj_ptr + proj_off, mask=mask_valid, other=0.0, dtype=tl.float16)
    gate = tl.load(gate_ptr + gate_off, mask=mask_valid, other=0.0, dtype=tl.float16)
    m    = tl.load(mask_ptr + mask_off, mask=mask_valid, other=1.0, dtype=tl.float16)

    out = proj * gate * m
    tl.store(out_ptr + proj_off, out, mask=mask_valid)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor, shape [B, N, N, dim], float32, CUDA
        mask_tensor  : torch.Tensor, shape [B, N, N]   (optional)
        weights_dict : mapping of the model weights (float32, CUDA)
        config_dict  : {"dim": int, "hidden_dim": int,
                        "nomask": bool (optional)}

    Returns
    -------
    out : torch.Tensor, shape [B, N, N, dim], float32, CUDA
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch weights (already on the correct device)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the `dim` channel (keep FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
    )  # [B, N, N, dim], dtype=float32

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim], dtype=half

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left_proj = F.linear(x_h, _w("left_proj.weight").half())   # [B, N, N, H]
    right_proj = F.linear(x_h, _w("right_proj.weight").half())

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid of linear layers) – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, _w("left_gate.weight").half()))
    right_gate = torch.sigmoid(F.linear(x_h, _w("right_gate.weight").half()))
    out_gate = torch.sigmoid(F.linear(x_h, _w("out_gate.weight").half()))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask – fuse with projection+gate via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:

        mask_h = mask.to(torch.float16).unsqueeze(-1)  # stride on last dim == 0
        # allocate fused tensors
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # launch parameters
        B_, N_, H_ = left_proj.shape[0], left_proj.shape[1], left_proj.shape[3]
        total_elements = B_ * N_ * N_ * H_
        BLOCK = 1024

            left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            mask_h.stride(0), mask_h.stride(1), mask_h.stride(2),
            B_, N_, H_,
            BLOCK=BLOCK,
        )

        _mask_gate_fuse_kernel[grid](
            right_proj, right_gate, mask_h, right,
            right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            mask_h.stride(0), mask_h.stride(1), mask_h.stride(2),
            B_, N_, H_,
            BLOCK=BLOCK,
        )
    else:
        # No mask – simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction: batched GEMM via cuBLAS (torch.bmm)
    #    left_perm / right_perm shaped [B, H, N, N]
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    # view as (batch_total, N, N) matrices
    A = left_perm.view(batch_total, N, N)                # fp16
    B_mat = right_perm.view(batch_total, N, N).transpose(1, 2)  # transposed k‑dim

    # batched matrix multiply (fp16 × fp16 → fp16, accumulation in fp32)
    out_flat = torch.bmm(A, B_mat)                       # [B*H, N, N], fp16

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Output processing: LayerNorm → gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight").half(),
        bias=_w("to_out_norm.bias").half(),

    out = out * out_gate                 # element‑wise gate (fp16)
    out = F.linear(out, _w("to_out.weight").half())   # [B, N, N, dim], fp16

    # ------------------------------------------------------------------
    # Cast back to FP32 as required by the API
    # ------------------------------------------------------------------
    return out.float()
