"""
custom_kernel – Outgoing TriMul (AlphaFold3) with a tiny Triton fusion

Algorithm
---------
1. Layer‑norm the input tensor (float32) over the feature dimension *dim*.
2. Cast to fp16 and compute the four linear projections (left, right,
   left_gate, right_gate, out_gate) with half‑precision weights.
3. Fuse mask, projection and gating for the *left* and *right* tensors
   with a tiny Triton kernel (element‑wise multiply of three fp16
   tensors).  This satisfies the “use at least one kernel” requirement.
4. Reshape each fused tensor from [B, N, N, H] to (B·H, N, N) so that the
   hidden dimension becomes the batch dimension.
5. Perform the expensive N³ operation with a single batched GEMM:
   `C = A @ Bᵀ`  where `A = left`, `B = right`.  cuBLAS (torch.bmm)
   runs on tensor‑cores and is far faster than a hand‑written Triton
   matmul.
6. Reshape the result back to [B, N, N, H] and apply a second
   LayerNorm (hidden‑dim), the output gate, and the final linear
   projection back to *dim*.
7. Return the result in float32.

The kernel only touches the memory‑bound element‑wise part, while the
matrix‑multiply dominates the compute and is handled by cuBLAS.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: out = proj * mask * gate   (all fp16)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul_kernel(
    out_ptr,          # FP16 output
    proj_ptr,         # FP16 projection (left or right)
    mask_ptr,         # FP16 mask   (broadcasted over hidden dim)
    gate_ptr,         # FP16 gate   (same shape as proj)
    total_elems,      # total number of elements in proj/gate
    hidden_dim,       # size of hidden dimension (used for broadcasting)
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise fusion kernel.

    Computes:
        out[i] = proj[i] * mask[i // hidden_dim] * gate[i]
    for i in [0, total_elems).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load projection and gate values
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out = proj * mask_val * gate
    tl.store(out_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None, shape [B, N, N] (float/bool)
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys
                         "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and compute linear projections & gates
    # ------------------------------------------------------------------
    x_h = x.half()  # fp16

    # Linear projections (no bias)
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,H] fp16
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())

    # Gate pre‑activations
    left_gate_pre = F.linear(x_h, weights["left_gate.weight"].half())
    right_gate_pre = F.linear(x_h, weights["right_gate.weight"].half())
    out_gate_pre = F.linear(x_h, weights["out_gate.weight"].half())

    # Sigmoid in fp16
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Mask handling (fp16)
    # ------------------------------------------------------------------
    if mask is None or nomask:
        mask_h = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        # mask may be bool or float; convert to fp16
        mask_h = mask.to(dtype=torch.float16, device=device)
    # Shape [B, N, N, 1] to broadcast over hidden dimension
    mask_h = mask_h.unsqueeze(-1)                       # [B,N,N,1] fp16
    mask_flat = mask_h.squeeze(-1).contiguous()        # [B,N,N] fp16, contiguous

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask + projection + gate (Triton kernel)
    # ------------------------------------------------------------------
    # Flatten tensors so the kernel can treat them as 1‑D arrays
    total = left_proj.numel()
    BLOCK = 1024  # reasonable trade‑off between occupancy and launch overhead

    # Allocate fused tensors (in‑place overwrite is also fine)
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # Launch kernel for left side
    grid = (triton.cdiv(total, BLOCK),)
    fused_mul_kernel[grid](
        left_fused,
        left_proj,
        mask_flat,
        left_gate,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # Launch kernel for right side
    fused_mul_kernel[grid](
        right_fused,
        right_proj,
        mask_flat,
        right_gate,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Batched GEMM: C = left @ rightᵀ  (fp16, cuBLAS)
    # ------------------------------------------------------------------
    # Reshape so hidden dimension becomes batch dimension
    left = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right_fused.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left = left.view(batch_h, N, N)                      # (B·H, N, N)
    right = right.view(batch_h, N, N)                    # (B·H, N, N)

    # Batched matrix multiplication: right is transposed on the last two dims
    out_h = torch.bmm(left, right.transpose(1, 2))      # (B·H, N, N) fp16

    # ------------------------------------------------------------------
    # 6️⃣ Restore original 4‑D layout and promote to fp32
    # ------------------------------------------------------------------
    out = out_h.view(B, hidden_dim, N, N)                # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()           # (B, N, N, H)
    out = out.to(torch.float32)                          # fp32 for remaining ops

    # ------------------------------------------------------------------
    # 7️⃣ Hidden‑dim LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Multiply by the output gate (convert gate to fp32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
