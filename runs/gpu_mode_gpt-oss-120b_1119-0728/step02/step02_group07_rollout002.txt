"""
custom_kernel – outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Layer‑norm the input tensor (float32) → x.
2️⃣ Cast x to fp16 and fuse the two linear projections per side:
   * left:  [proj + gate]   (2·hidden → dim) → split into proj and pre‑sigmoid‑gate.
   * right: same.
   The weights are kept in fp16, the matmul is performed by cuBLAS.
3️⃣ Compute the out‑gate (float32 sigmoid of a linear on x).
4️⃣ Apply optional mask to the *proj* streams only.
5️⃣ Apply sigmoid to the gate streams and multiply → left, right (fp16).
6️⃣ Batched GEMM   outₕ = left @ rightᵀ   (one half‑precision matmul per
   (batch·hidden) slice, with fp32 accumulation).  Implemented in Triton.
   The GEMM is the O(N³) kernel: for each hidden dimension we compute
   outₕ[b,i,j] = Σₖ left[b,i,k]·right[b,j,k].
7️⃣ Reshape back to [B,N,N,hidden] and promote to fp32.
8️⃣ Hidden‑dim LayerNorm → multiply by out‑gate → final Linear to dim.
All heavy work (step 6) runs in a fused Triton kernel; the rest uses
plain torch / cuBLAS.  The kernel is autotuned for several block sizes
and uses fp16 inputs with fp32 accumulation for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : (batch, M, K);  C : (batch, M, N))
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32, "num_warps": 2},
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32, "num_warps": 4},
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 256, "BLOCK_N": 256, "BLOCK_K": 64, "num_warps": 8},
            num_stages=2,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    batch, M, N, K,                 # dims (batch = B*H)
    stride_ab, stride_am, stride_ak,  # A strides
    stride_bb, stride_bn, stride_bk,  # B strides
    stride_cb, stride_cm, stride_cn,  # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM with FP32 accumulation.

    Each program instance computes a (BLOCK_M × BLOCK_N) tile of C for a
    particular batch index (pid_b).  The K‑dimension is tiled with BLOCK_K.
    """
    pid_b = tl.program_id(2)          # batch * hidden index
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # Offsets for the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (BLOCK_M × BLOCK_K)
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(
            a_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
        )

        # Load Bᵀ tile (BLOCK_K × BLOCK_N) – note the swapped indices
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(
            b_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_b,
            other=0.0,
        )

        # Multiply‑accumulate (BLAS‑style GEMM)
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N) in FP32

    # Write back the result in fp16
    c = acc.to(tl.float16)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        c_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_c,
    )


# ----------------------------------------------------------------------
# Forward pass entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor[B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor[B, N, N] (bool/float); may be ignored
            - weights      : dict of all Linear / LayerNorm parameters (float32)
            - config       : {"dim": ..., "hidden_dim": ..., "nomask": bool (optional)}
    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32, stable)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (fp32)

    # --------------------------------------------------------------
    # 2) Fuse proj + gate for the left and right streams (half‑precision)
    # --------------------------------------------------------------
    # Cast to fp16 – after the stable LN we can lose precision safely
    x_h = x.to(torch.float16)

    # Concatenate proj and gate weights to avoid two separate matmuls
    left_w_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    right_w_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    # Linear (B*N*N, 2*H) with fp16 weights
    left_concat = F.linear(x_h, left_w_concat)   # [B,N,N,2*H]
    right_concat = F.linear(x_h, right_w_concat) # [B,N,N,2*H]

    # Split into projection and pre‑sigmoid gate tensors
    left_proj = left_concat[..., :hidden_dim]          # [B,N,N,H]
    left_gate_pre = left_concat[..., hidden_dim:]      # [B,N,N,H]
    right_proj = right_concat[..., :hidden_dim]        # [B,N,N,H]
    right_gate_pre = right_concat[..., hidden_dim:]    # [B,N,N,H]

    # --------------------------------------------------------------
    # 3) Out‑gate (computed in fp32)
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H]

    # --------------------------------------------------------------
    # 4) Optional mask (applied only to the projections)
    # --------------------------------------------------------------
    if not nomask:
        # mask is broadcastable to [B,N,N,1]
        m = mask_tensor.to(dtype=left_proj.dtype).unsqueeze(-1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # --------------------------------------------------------------
    # 5) Apply sigmoid gates to the streams (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)   # fp16
    right_gate = torch.sigmoid(right_gate_pre) # fp16

    left = left_proj * left_gate   # fp16
    right = right_proj * right_gate # fp16

    # --------------------------------------------------------------
    # 6) Batched GEMM: out_h = left @ rightᵀ   (fp32 accumulation)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Reshape to (B*H, N, N) – each hidden slice becomes an independent GEMM
    left_tile = (
        left.permute(0, 3, 1, 2)   # (B, H, N, N)
        .contiguous()
        .view(B * H, N, N)
    )
    right_tile = (
        right.permute(0, 3, 1, 2)
        .contiguous()
        .view(B * H, N, N)
    )
    out_tile = torch.empty_like(left_tile, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_ab = left_tile.stride(0)
    stride_am = left_tile.stride(1)
    stride_ak = left_tile.stride(2)

    stride_bb = right_tile.stride(0)
    stride_bn = right_tile.stride(1)
    stride_bk = right_tile.stride(2)

    stride_cb = out_tile.stride(0)
    stride_cm = out_tile.stride(1)
    stride_cn = out_tile.stride(2)

    # Kernel launch parameters
    BLOCK_M = 128   # will be selected by autotune
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # grid dim 0 → rows
        (N + BLOCK_N - 1) // BLOCK_N,   # grid dim 1 → cols
        B * H,                          # grid dim 2 → batch × hidden
    )

    # Launch the Triton kernel
    tri_mul_gemm_kernel[grid](
        left_tile,
        right_tile,
        out_tile,
        B * H,
        N,
        N,
        N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bn,
        stride_bk,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore shape [B, N, N, H] (still fp16)
    out = out_tile.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 7) Hidden‑dim LayerNorm (fp32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    # --------------------------------------------------------------
    # 8) Apply out‑gate (fp32) and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
