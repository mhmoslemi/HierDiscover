@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    MASK_PTR, HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_z = tl.program_id(2)  # batch*channel

    # compute batch and hidden
    b = pid_z // H
    h = pid_z % H

    # compute start offsets
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute loop over k
    for k0 in range(0, N, BLOCK_K):
        # Compute actual block size for edges
        cur_k = tl.minimum(N - k0, BLOCK_K)
        cur_i = tl.minimum(N - i0, BLOCK_M)
        cur_j = tl.minimum(N - j0, BLOCK_N)

        # offsets for left
        offs_i = i0 + tl.arange(0, BLOCK_M)
        offs_k = k0 + tl.arange(0, BLOCK_K)
        # mask for out-of-bounds
        mask_i = offs_i < N
        mask_k = offs_k < N
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # offsets for right (transposed on the fly)
        offs_j = j0 + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # matrix multiplication
        acc += tl.dot(left, right)

    # store result
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    x, mask, weights, config = data
    bs, seq_len, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_w = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # left, right
    left = F.linear(x_norm, left_proj_w)   # shape (bs, N, N, hidden_dim)
    right = F.linear(x_norm, right_proj_w)

    # Apply mask if present (non None)
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(left)  # broadcast
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Output tensor
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32).contiguous()

    # Get strides
    stride_b, stride_i, stride_j, stride_h = left_fp16.stride()
    # For left and right: stride_i = seq_len*hidden_dim, stride_j = hidden_dim (k dimension)
    # But in our naming: i is first N axis, k is second N axis (j dimension). So:
    stride_left_b = stride_b
    stride_left_i = stride_i
    stride_left_k = stride_j
    stride_left_h = stride_h

    # For right: same strides
    stride_right_b = stride_b
    stride_right_i = stride_i
    stride_right_k = stride_j
    stride_right_h = stride_h

    # Output strides (output shape same as left)
    stride_out_b = stride_b
    stride_out_i = stride_i
    stride_out_j = stride_j
    stride_out_h = stride_h

    # Run kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = ( (seq_len + BLOCK_M - 1)//BLOCK_M,
             (seq_len + BLOCK_N - 1)//BLOCK_N,
             bs * hidden_dim )
    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        bs, seq_len, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

"""
TriMul (outgoing) implementation for AlphaFold‑3 style pairwise updates.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor over the feature dimension `dim`.
2. Linear projections → `left` and `right` (shape [B,N,N,H]) and three gates
   (`left_gate`, `right_gate`, `out_gate`).  Gates are passed through a sigmoid.
3. Apply the optional pairwise mask and the gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
4. Compute the expensive tensor contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM per hidden channel using a custom Triton kernel.
5. Layer‑norm the result over the hidden dimension, apply the `out_gate`,
   and finally project back to the original `dim`.

Only the N³ contraction is executed on the GPU via Triton; all other
operations use regular PyTorch (they are cheap compared to the triple
product).  The kernel loads `float16` data, uses Tensor‑Core `dot`
operations, and accumulates in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                      # dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute for a single (batch, hidden) pair:
        out[i, j] = Σₖ left[i, k] * right[j, k]   (k‑reduction)
    left  : (M, K)   from input axis (i, k)
    right : (K, N)   from input axis (k, j)
    Both tensors are stored as contiguous float16.
    """
    # 3‑D grid: (tile_i, tile_j, batch*hidden)
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_z = tl.program_id(2)          # batch*hidden index

    # decode batch and hidden channel
    b = pid_z // H
    h = pid_z % H

    # starting coordinates of this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis `k`
    for k_start in range(0, N, BLOCK_K):
        # ---- load left (i, k) ----
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k_start + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_k = offs_k < N

        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], dtype=tl.float16)

        # ---- load right (k, j) – note the transpose on‑the‑fly ----
        offs_j = j_start + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N

        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k      # k dimension (rows)
            + offs_j[None, :] * stride_right_i      # j dimension (cols)
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], dtype=tl.float16)

        # ---- matrix multiply and accumulate (FP16 dot → FP32 accumulator) ----
        prod = tl.dot(left, right)                 # FP16 result (M,N)
        acc += prod.to(tl.float32)

    # ---- store the final block ----
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack arguments
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]          # H
    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gating values
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # optional mask (mask may be all‑ones; still safe to apply)
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)        # [B,N,N,1]
        left  = left * mask
        right = right * mask

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Triton GEMM‑style contraction over K:   out = Σₖ left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    # Cast to half‑precision for the kernel (Tensor‑cores)
    left_h   = left.contiguous().to(torch.float16)
    right_h  = right.contiguous().to(torch.float16)
    out_fp32 = torch.empty_like(left_h, dtype=torch.float32).contiguous()

    # strides of the (B,N,N,H) tensors (contiguous layout)
    stride_b, stride_i, stride_k, stride_h = left_h.stride()   # (B,N,N,H) layout

    # block sizes – tune for H100 (fp16 Tensor‑core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
        B * hidden_dim                       # one program per (batch, hidden)
    )

    _tri_mul_kernel[grid](
        left_h, right_h, out_fp32,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_h,   # left strides
        stride_b, stride_i, stride_k, stride_h,   # right strides (identical)
        stride_b, stride_i, stride_k, stride_h,   # out strides  (i‑stride, j‑stride==k‑stride)
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ------------------------------------------------------------------
    # 4) Output post‑processing (LayerNorm, gating, final Linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # shape -> [B,N,N,dim]

    return out
