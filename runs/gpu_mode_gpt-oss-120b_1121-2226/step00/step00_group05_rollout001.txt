    batch = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
custom_kernel
==============

Implements the *outgoing* TriMul operation (AlphaFold‑3) with a
custom Triton kernel for the heavy N³ einsum.
The pipeline is:

1. Layer‑norm the input (dim).
2. Linear projections (`left_proj`, `right_proj`) and three gates
   (`left_gate`, `right_gate`, `out_gate`).
3. Apply the optional mask and the gates.
4. Core operation:     out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]
   → a batched GEMM for every hidden dimension `d`.  This is
   executed by a Triton kernel that processes a (batch*hidden)×N×N
   matrix multiplication (`left @ rightᵀ`) tile‑wise.
5. Layer‑norm the result, multiply by `out_gate`, and a final linear
   projection back to the original `dim`.

All tensors are kept in `float32` (the kernel accumulates in FP32);
the kernel is deliberately written for clarity and correctness,
while still exploiting the massive parallelism of the H100.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) where A,B have shape
#   (batch, M, K)   and   (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Ar, stride_Ac,     # batch, row, col strides for A
    stride_Bb, stride_Br, stride_Bc,     # batch, row(K), col(N) strides for Bᵀ
    stride_Cb, stride_Cr, stride_Cc,     # batch, row, col strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Program IDs: (block row, block col, batch*hidden)
    # --------------------------------------------------------------
    pid_batch = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # --------------------------------------------------------------
    # Compute tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Ar
    b_ptrs = B_ptr + pid_batch * stride_Bb + offs_n[None, :] * stride_Bc

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptrs + offs_k[None, :] * stride_Ac,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )
        # Load B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptrs + offs_k[:, None] * stride_Br,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )
        # GEMM update
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_Ac
        b_ptrs += BLOCK_K * stride_Br

    # --------------------------------------------------------------
    # Write back the result tile
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Cr + offs_n[None, :] * stride_Cc
    tl.store(
        c_ptrs,
        acc,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Executes the outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Configuration values
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul: out[i,j] = Σₖ left[i,k] * right[j,k]
    # Implemented as a batched GEMM (left @ rightᵀ) using Triton.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Rearrange to (B, H, N, N) for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Transpose the second matrix so we can do a standard GEMM
    right_T = right_perm.transpose(-2, -1).contiguous()  # [B, H, N, N]

    # Collapse the (B, H) dimensions into a single batch for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_T.view(B * H, N, N)

    # Allocate the output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements) for the flattened tensors
    stride_a_batch, stride_a_row, stride_a_col = left_flat.stride()
    stride_b_batch, stride_b_row, stride_b_col = right_flat.stride()
    stride_c_batch, stride_c_row, stride_c_col = out_flat.stride()

    # Tile sizes – chosen to fit well on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (num_row_tiles, num_col_tiles, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    # Launch the Triton kernel
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,          # M
        N,          # N
        N,          # K
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Post‑processing: norm, gating, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate  # broadcast over hidden dimension

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
