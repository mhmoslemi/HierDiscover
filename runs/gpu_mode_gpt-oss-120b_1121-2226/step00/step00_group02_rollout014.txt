def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # True if mask is not applied

    # layernorm on input
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # mask if provided
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # out via Triton
    B, N, _, _ = left.shape

    # allocate output
    out = torch.empty_like(left)

    # determine block sizes (tuned)
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32   # hidden block size (must divide hidden)
    # Ensure hidden dims divisible: pad maybe

    # Strides
    stride_b = left.stride(0)
    stride_i = left.stride(1)
    stride_k = left.stride(2)
    stride_h = left.stride(3)

    # out strides same
    out_stride_b = out.stride(0)
    out_stride_i = out.stride(1)
    out_stride_j = out.stride(2)
    out_stride_h = out.stride(3)

    # compile kernel
    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
    ):

        pid_i = tl.program_id(0)   # i tile
        pid_j = tl.program_id(1)   # j tile
        pid_h = tl.program_id(2)   # hidden tile

        # compute start indices
        i0 = pid_i * BLOCK_N
        j0 = pid_j * BLOCK_N
        h0 = pid_h * BLOCK_H

        # offsets within tile
        i = i0 + tl.arange(0, BLOCK_N)
        j = j0 + tl.arange(0, BLOCK_N)
        h = h0 + tl.arange(0, BLOCK_H)

        # masks
        i_mask = i < N
        j_mask = j < N
        h_mask = h < H

        acc = tl.zeros((BLOCK_N, BLOCK_N, BLOCK_H), dtype=tl.float32)

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # mask
    if not nomask and mask is not None:
        mask = mask.unsqueeze(-1)   # broadcast
        left = left * mask
        right = right * mask

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # output placeholder
    out = torch.empty_like(left)

    B, N, _, _ = left.shape

    # ensure contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # stride values (in elements)
    stride_l_b = left.stride(0)
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    stride_l_h = left.stride(3)

    stride_r_b = right.stride(0)
    stride_r_i = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_h = out.stride(3)

    # Choose block sizes
    BLOCK_N = 64   # tile for i and j
    BLOCK_K = 32   # reduction tile
    BLOCK_H = 32   # hidden tile (must divide hidden_dim nicely)
    # adjust BLOCK_H if hidden_dim smaller
    if hidden_dim < BLOCK_H:
        BLOCK_H = hidden_dim

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        N, H,
        stride_left_i, stride_left_k, stride_left_h,
        stride_right_i, stride_right_k, stride_right_h,
        stride_out_i, stride_out_j, stride_out_h,
        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
    ):
        pid_i = tl.program_id(0)  # tile i
        pid_j = tl.program_id(1)  # tile j
        pid_h = tl.program_id(2)  # hidden tile

        i0 = pid_i * BLOCK_N
        j0 = pid_j * BLOCK_N
        h0 = pid_h * BLOCK_H

        i = i0 + tl.arange(0, BLOCK_N)
        j = j0 + tl.arange(0, BLOCK_N)
        h = h0 + tl.arange(0, BLOCK_H)

        i_mask = i < N
        j_mask = j < N
        h_mask = h < H

        # accumulator
        acc = tl.zeros((BLOCK_N, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # loop over k dimension
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            k0 = k_tile * BLOCK_K
            k = k0 + tl.arange(0, BLOCK_K)
            k_mask = k < N

            # Load left block [i,k,h]
            left_ptrs = left_ptr + \
                i[:, None, None] * stride_left_i + \
                k[None, :, None] * stride_left_k + \
                h[None, None, :] * stride_left_h
            left_block = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :], other=0.0)

            # Load right block transposed [k,j,h] (so that we can dot directly)
            right_ptrs = right_ptr + \
                k[:, None, None] * stride_right_k + \
                j[None, :, None] * stride_right_i + \
                h[None, None, :] * stride_right_h
            right_block = tl.load(right_ptrs, mask=k_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :], other=0.0)

            # for each hidden channel in tile
            for dh in range(0, BLOCK_H):
                left_slice = left_block[:, :, dh]   # (i,k)
                right_slice = right_block[:, :, dh] # (k,j)
                # matmul: (i,k) @ (k,j) -> (i,j)
                acc[:, :, dh] += tl.dot(left_slice, right_slice)

        # store result
        out_ptrs = out_ptr + \
            i[:, None, None] * stride_out_i + \
            j[None, :, None] * stride_out_j + \
            h[None, None, :] * stride_out_h
        tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :])

    # launch kernel for each batch
    element_size = left.element_size()  # bytes per element
    for b in range(B):
        left_ptr = left.data_ptr() + b * stride_l_b * element_size
        right_ptr = right.data_ptr() + b * stride_r_b * element_size
        out_ptr = out.data_ptr() + b * stride_o_b * element_size

        grid = (
            (N + BLOCK_N - 1) // BLOCK_N,
            (N + BLOCK_N - 1) // BLOCK_N,
            (hidden_dim + BLOCK_H - 1) // BLOCK_H,
        )
        tri_mul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            stride_l_i,
            stride_l_k,
            stride_l_h,
            stride_r_i,
            stride_r_k,
            stride_r_h,
            stride_o_i,
            stride_o_j,
            stride_o_h,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel
==============

Implements the forward pass of the “outgoing” TriMul operator (AlphaFold3) with a
high‑performance Triton kernel for the cubic‑cost einsum:

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where

* `left`  = Linear(dim → hidden) (input)  × gate(left)   × mask
* `right` = Linear(dim → hidden) (input)  × gate(right)  × mask

The kernel evaluates a batched GEMM for each hidden channel `d`.  The
computation is tiled over the sequence dimension (`i`, `j`) with `BLOCK_N`,
over the reduction dimension `k` with `BLOCK_K`, and over the hidden
dimension with `BLOCK_H`.  For each tile we load a block of `left` (shape
[i,k,d]) and a transposed block of `right` (shape [k,j,d]), then perform a
matrix‑multiply per hidden slice using `tl.dot`.  The partial results are
accumulated in FP32 registers and stored back to the output tensor.

All other linear layers, layer‑norms and sigmoid gates are performed with
standard PyTorch functional calls; the heavy O(N³·hidden) part is off‑loaded
to Triton.
"""

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                 # input channel dimension
    hidden_dim = config["hidden_dim"]   # hidden dimension (projected)
    nomask = config.get("nomask", True)  # True → ignore mask

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Gates (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4) Optional mask
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.unsqueeze(-1)                  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5) Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) TriMul (cubic reduction) – Triton kernel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                     # left.shape == right.shape
    out = torch.empty_like(left)                # place‑holder for the result

    # Ensure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides in **elements** (not bytes)
    stride_l_b = left.stride(0)
    stride_l_i = left.stride(1)   # i‑dimension
    stride_l_k = left.stride(2)   # k‑dimension
    stride_l_h = left.stride(3)   # hidden dimension

    stride_r_b = right.stride(0)
    stride_r_i = right.stride(1)   # j‑dimension (treated as i for right)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)     # i‑dimension
    stride_o_j = out.stride(2)     # j‑dimension
    stride_o_h = out.stride(3)     # hidden

    # ------------------------------------------------------------------
    # Triton kernel – computes out = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    BLOCK_N = 64   # tile size for i and j axes
    BLOCK_K = 32   # reduction tile along k
    BLOCK_H = 32 if hidden_dim >= 32 else hidden_dim   # hidden‑tile

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        N, H,
        stride_left_i, stride_left_k, stride_left_h,
        stride_right_i, stride_right_k, stride_right_h,
        stride_out_i, stride_out_j, stride_out_h,
        BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
    ):
        # Program IDs → tile indices
        pid_i = tl.program_id(0)    # tile along i
        pid_j = tl.program_id(1)    # tile along j
        pid_h = tl.program_id(2)    # tile along hidden dimension

        i0 = pid_i * BLOCK_N
        j0 = pid_j * BLOCK_N
        h0 = pid_h * BLOCK_H

        # Indices inside the tile
        i = i0 + tl.arange(0, BLOCK_N)
        j = j0 + tl.arange(0, BLOCK_N)
        h = h0 + tl.arange(0, BLOCK_H)

        i_mask = i < N
        j_mask = j < N
        h_mask = h < H

        # Accumulator: shape (BLOCK_N, BLOCK_N, BLOCK_H), FP32
        acc = tl.zeros((BLOCK_N, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over the reduction dimension k in blocks
        # ------------------------------------------------------------------
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            k0 = k_tile * BLOCK_K
            k = k0 + tl.arange(0, BLOCK_K)
            k_mask = k < N

            # Load left block   : [i, k, h]
            left_ptrs = left_ptr + \
                i[:, None, None] * stride_left_i + \
                k[None, :, None] * stride_left_k + \
                h[None, None, :] * stride_left_h
            left_block = tl.load(
                left_ptrs,
                mask=i_mask[:, None, None] &
                     k_mask[None, :, None] &
                     h_mask[None, None, :],
                other=0.0,
            )

            # Load right block **transposed** : [k, j, h]  (so we can dot directly)
            right_ptrs = right_ptr + \
                k[:, None, None] * stride_right_k + \
                j[None, :, None] * stride_right_i + \
                h[None, None, :] * stride_right_h
            right_block = tl.load(
                right_ptrs,
                mask=k_mask[:, None, None] &
                     j_mask[None, :, None] &
                     h_mask[None, None, :],
                other=0.0,
            )

            # Compute contribution for each hidden slice in the tile
            for dh in range(0, BLOCK_H):
                # left_slice: (i, k), right_slice: (k, j)
                left_slice  = left_block[:, :, dh]
                right_slice = right_block[:, :, dh]
                # Accumulate: (i,k) @ (k,j) → (i,j)
                acc[:, :, dh] += tl.dot(left_slice, right_slice)

        # ------------------------------------------------------------------
        # Store the accumulated result
        # ------------------------------------------------------------------
        out_ptrs = out_ptr + \
            i[:, None, None] * stride_out_i + \
            j[None, :, None] * stride_out_j + \
            h[None, None, :] * stride_out_h
        tl.store(
            out_ptrs,
            acc,
            mask=i_mask[:, None, None] &
                 j_mask[None, :, None] &
                 h_mask[None, None, :],
        )

    # --------------------------------------------------------------
    # Launch the kernel for each batch independently (B ≤ 2)
    # --------------------------------------------------------------
    element_size = left.element_size()               # bytes per element
    for b in range(B):
        # Compute base pointers for the current batch (byte offset)
        left_ptr  = left.data_ptr()  + b * stride_l_b * element_size
        right_ptr = right.data_ptr() + b * stride_r_b * element_size
        out_ptr   = out.data_ptr()   + b * stride_o_b * element_size

        grid = (
            (N + BLOCK_N - 1) // BLOCK_N,          # i‑tiles
            (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
            (hidden_dim + BLOCK_H - 1) // BLOCK_H, # hidden tiles
        )
        tri_mul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            stride_l_i,
            stride_l_k,
            stride_l_h,
            stride_r_i,
            stride_r_k,
            stride_r_h,
            stride_o_i,
            stride_o_j,
            stride_o_h,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # --------------------------------------------------------------
    # 7) Output LayerNorm, gate, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
