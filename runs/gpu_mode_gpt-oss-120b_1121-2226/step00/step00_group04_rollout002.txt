def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # core TriMul via Triton kernel
    # Permute to [B, hidden_dim, N, N] for batch matmul
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # Transpose right over last two dims
    right_T = right_perm.transpose(-2, -1).contiguous()
    B, H, N, _ = left_perm.shape
    # flatten batch dimension
    a = left_perm.view(B*H, N, N)
    b = right_T.view(B*H, N, N)
    # allocate output
    c = torch.empty_like(a)  # same dtype as a (float32)
```

"""
custom_kernel: Triton‑accelerated forward of AlphaFold3's outgoing TriMul.
--------------------------------------------------------------
1. Layer‑norm the input (shape [B,N,N,C]).
2. Linear projections (left/right) and gating (left_gate/right_gate/out_gate)
   are done with torch.nn.functional.linear + sigmoid.
3. Apply optional pairwise mask.
4. Compute the core tensor contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (h = hidden_dim)
   as a batched matrix‑multiply using a Triton kernel.  This is
   equivalent to `out = left @ right.transpose(-2,-1)` for every hidden
   slice, i.e. a batch of N×N matmuls with batch size B·hidden_dim.
5. A second layer‑norm, out‑gate gating and final linear projection
   produce the result of shape [B,N,N,C].

Only the O(N³) reduction is implemented in Triton; the remaining
operations use PyTorch which is already highly optimized.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiply  C = A @ B^T
#   A : (B*H, M, K)   – here M = N, K = N
#   B : (B*H, N, K)   – we pass the transpose of the original right tensor,
#                       so B already has shape (B*H, N, K) with the inner
#                       dimension aligned to K.
#   C : (B*H, M, N)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ab, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, row, col) – B is already transposed
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Implements C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]   (B is transposed).
    Each program_id axis indexes:
        program_id(0) -> batch index (b)
        program_id(1) -> block row (i)
        program_id(2) -> block column (j)
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute the actual coordinates of the block we are working on.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for the start of the batch.
    a_batch_ptr = a_ptr + batch_idx * stride_ab
    b_batch_ptr = b_ptr + batch_idx * stride_bb
    c_batch_ptr = c_ptr + batch_idx * stride_cb

    # Initialize accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension in tiles.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute K offsets for this tile.
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile (M x K) – note that A is FP16.
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)

        # Load B tile (N x K) – B is already transposed, we need row = K, col = N.
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # Accumulate the dot product. The dot returns FP16; the addition
        # promotes it to FP32 (our accumulator dtype).
        acc += tl.dot(a, b)

    # Store the result (still FP32) to C.
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same device as inputs).
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------------- #
    # Helper to fetch weight tensors (they already live on the correct device)
    # --------------------------------------------------------------------- #
    def get(name):
        return weights[name]

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the channel dimension (dim)
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=get("norm.weight"),
        bias=get("norm.bias")
    )   # shape [B,N,N,dim]

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias) to hidden_dim
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, get("left_proj.weight"))   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, get("right_proj.weight"))

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------------- #
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------------- #
    left_gate = torch.nn.functional.linear(x, get("left_gate.weight")).sigmoid()
    right_gate = torch.nn.functional.linear(x, get("right_gate.weight")).sigmoid()
    out_gate = torch.nn.functional.linear(x, get("out_gate.weight")).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Core TriMul contraction using Triton
    # --------------------------------------------------------------- #
    # Permute to bring the hidden dimension next to the batch dimension:
    #   [B, N, N, H] -> [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right (so we have ... N,K where K == N)
    right_T = right_perm.transpose(-2, -1).contiguous()

    B, H, N, _ = left_perm.shape

    # Collapse batch*hidden into a single batch dimension for the kernel.
    a = left_perm.view(B * H, N, N).to(torch.float16)   # (B*H, M=N, K=N)
    b = right_T.view(B * H, N, N).to(torch.float16)    # (B*H, N, K=N)  (already transposed)

    # Output buffer (FP32 to keep precision for the following LayerNorm)
    c = torch.empty_like(a, dtype=torch.float32)

    # Strides in element units (not bytes)
    stride_ab = a.stride(0)      # batch stride
    stride_am = a.stride(1)      # row stride (M)
    stride_ak = a.stride(2)      # col stride (K)

    stride_bb = b.stride(0)
    stride_bk = b.stride(1)      # row stride (K)  – because b is (batch, K, N) after transpose
    stride_bn = b.stride(2)      # col stride (N)

    stride_cb = c.stride(0)
    stride_cm = c.stride(1)
    stride_cn = c.stride(2)

    # Tunable tile sizes – chosen to fit well on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=N,
        N=N,
        K=N,
        stride_ab=stride_ab,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bb=stride_bb,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cb=stride_cb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------------- #
    # 6. Second LayerNorm, out‑gate, and final linear projection
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=get("to_out_norm.weight"),
        bias=get("to_out_norm.bias")
    )
    out = out * out_gate                     # gating
    output = torch.nn.functional.linear(out, get("to_out.weight"))  # -> [B,N,N,dim]

    return output
