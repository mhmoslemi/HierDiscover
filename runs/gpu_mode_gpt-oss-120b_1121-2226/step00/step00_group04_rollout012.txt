```
    # retrieve config dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
```

```
import torch
import triton
import triton.language as tl

@triton.jit
def fused_mul_kernel(out, a, b, c, numel, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < numel
    av = tl.load(a + offset, mask=mask, other=0.0)
    bv = tl.load(b + offset, mask=mask, other=0.0)
    cv = tl.load(c + offset, mask=mask, other=0.0)
    tl.store(out + offset, av * bv * cv, mask=mask)
    
@triton.jit
def elementwise_mul_two_kernel(out, a, b, numel, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < numel
    av = tl.load(a + offset, mask=mask, other=0.0)
    bv = tl.load(b + offset, mask=mask, other=0.0)
    tl.store(out + offset, av * bv, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    Performs layernorm, linear projections, gating, applies mask,
    computes the pairwise contraction via an efficient batched
    matrix multiplication (torch.bmm) and final linear projection.
    Elementwise fusions of mask/gate are off‑loaded to Triton kernels.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm over input channel dim -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Using torch.nn.functional.layer_norm which supports weight/bias
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections (no bias) -----
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- Gating linear + sigmoid -----
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ----- Prepare mask (broadcasted) -----
    if nomask:
        # No mask → all ones
        mask = torch.ones_like(x[..., :1], dtype=dtype, device=device)
    else:
        # mask shape: [B, N, N]; we need it broadcastable to [B,N,N,hidden_dim]
        mask = mask_tensor.unsqueeze(-1).to(dtype)   # [B,N,N,1]

    # Expand mask to hidden_dim for fusion kernels
    mask_exp = mask.expand(-1, -1, -1, hidden_dim).contiguous()

    # ----- Fuse projection * mask * gate using Triton -----
    # Ensure tensors are contiguous for 1‑D view
    left_proj_f = left_proj.contiguous()
    right_proj_f = right_proj.contiguous()
    left_gate_f = left_gate.contiguous()
    right_gate_f = right_gate.contiguous()
    # Allocate output tensors
    left = torch.empty_like(left_proj_f)
    right = torch.empty_like(right_proj_f)

    # Flatten everything to 1‑D
    numel = left_proj_f.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(numel, meta['BLOCK']),)

    fused_mul_kernel[grid](left, left_proj_f, mask_exp, left_gate_f, numel, BLOCK_SIZE=BLOCK)
    fused_mul_kernel[grid](right, right_proj_f, mask_exp, right_gate_f, numel, BLOCK_SIZE=BLOCK)

    # ----- Batched contraction via torch.bmm -----
    # Reshape to (B*hidden_dim, N, N)
    B, N, _, _ = left.shape
    left_reshape = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # matmul: left @ right^T
    out_mat = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # Reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- to_out_norm (LayerNorm on hidden_dim) -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ----- Multiply with out_gate (elementwise) via Triton -----
    # Ensure contiguous
    out_gate_f = out_gate.contiguous()
    out_norm_f = out_norm.contiguous()
    out_gated = torch.empty_like(out_norm_f)
    numel2 = out_norm_f.numel()
    grid2 = lambda meta: (triton.cdiv(numel2, meta['BLOCK']),)
    elementwise_mul_two_kernel[grid2](out_gated, out_norm_f, out_gate_f, numel2, BLOCK_SIZE=BLOCK)

    # ----- Final linear projection back to dim -----
    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_gated.view(-1, hidden_dim), to_out_weight)
    out_final = out_final.view(B, N, N, dim)

    return out_final
```

```
import torch
import triton
import triton.language as tl

@triton.jit
def fused_mul_kernel(out_ptr, a_ptr, b_ptr, c_ptr, numel, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < numel
    a = tl.load(a_ptr + offset, mask=mask, other=0.0)
    b = tl.load(b_ptr + offset, mask=mask, other=0.0)
    c = tl.load(c_ptr + offset, mask=mask, other=0.0)
    tl.store(out_ptr + offset, a * b * c, mask=mask)

@triton.jit
def elementwise_mul_two_kernel(out_ptr, a_ptr, b_ptr, numel, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < numel
    a = tl.load(a_ptr + offset, mask=mask, other=0.0)
    b = tl.load(b_ptr + offset, mask=mask, other=0.0)
    tl.store(out_ptr + offset, a * b, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.
    The implementation mirrors the reference PyTorch module but off‑loads
    the three‑way elementwise multiplication (projection × mask × gate)
    and the final gate‑application to custom Triton kernels. The
    expensive contraction ∑_k left[...,k,:]·right[...,k,:] is performed
    via a batched matrix‑multiply (torch.bmm), which yields the same
    result as the original ``einsum('... i k d, ... j k d -> ... i j d')``.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- 1. Input LayerNorm (over dim) -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )

    # ----- 2. Linear projections (no bias) -----
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----- 3. Gating (linear + sigmoid) -----
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ----- 4. Mask handling -----
    if nomask:
        mask = torch.ones_like(x[..., :1], dtype=dtype, device=device)
    else:
        mask = mask_tensor.unsqueeze(-1).to(dtype)   # [B,N,N,1]

    # Broadcast mask to hidden_dim for fused kernel
    mask_exp = mask.expand(-1, -1, -1, hidden_dim).contiguous()

    # ----- 5. Fuse (proj * mask * gate) with Triton -----
    left_proj_f = left_proj.contiguous()
    right_proj_f = right_proj.contiguous()
    left_gate_f = left_gate.contiguous()
    right_gate_f = right_gate.contiguous()

    left = torch.empty_like(left_proj_f)
    right = torch.empty_like(right_proj_f)

    numel = left_proj_f.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(numel, meta["BLOCK_SIZE"]),)

    fused_mul_kernel[grid](
        left, left_proj_f, mask_exp, left_gate_f, numel, BLOCK_SIZE=BLOCK
    )
    fused_mul_kernel[grid](
        right, right_proj_f, mask_exp, right_gate_f, numel, BLOCK_SIZE=BLOCK
    )

    # ----- 6. Contract over the third axis (k) via batched GEMM -----
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # left @ right^T  (each slice multiplies its own hidden channel)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----- 7. Output LayerNorm (over hidden_dim) -----
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ----- 8. Apply final gating (out_gate) with Triton -----
    out_norm_f = out_norm.contiguous()
    out_gate_f = out_gate.contiguous()
    out_gated = torch.empty_like(out_norm_f)

    numel2 = out_norm_f.numel()
    grid2 = lambda meta: (triton.cdiv(numel2, meta["BLOCK_SIZE"]),)
    elementwise_mul_two_kernel[grid2](
        out_gated, out_norm_f, out_gate_f, numel2, BLOCK_SIZE=BLOCK
    )

    # ----- 9. Final linear projection back to `dim` -----
    to_out_weight = weights["to_out.weight"]
    out_final = torch.nn.functional.linear(out_gated.view(-1, hidden_dim), to_out_weight)
    out_final = out_final.view(B, N, N, dim)

    return out_final
```
