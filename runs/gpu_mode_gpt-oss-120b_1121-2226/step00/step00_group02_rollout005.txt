def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, N2, dim = input_tensor.shape
    assert N == N2
    hidden_dim = config["hidden_dim"]
    # Cast to half
    x = input_tensor.to(torch.float16)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    # Mask multiplication if not nomask
    if not config.get("nomask", True):
        # mask shape bs,N,N maybe bool or float.
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape [bs,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16
    else:
        # No mask: skip
        pass

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # TriMul core: compute out = left @ right^T across k dimension per channel
    # Permute to combine batch and channel
    left_ = left.permute(0, 3, 1, 2).contiguous()  # [bs, H, N, N]
    right_ = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_.shape

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)  # one kernel per batch (combined B*H)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # Compute output tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Prepare masks for boundaries
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator with zero (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute base pointers for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (transposed)
        # B is stored as (batch, N, K) but we are accessing as (batch, K, N) by swapping strides
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    # Convert acc back to float16
    acc_fp16 = acc.to(tl.float16)
    tl.store(c_ptrs, acc_fp16, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward implementation with a Triton kernel.

The algorithm follows the reference PyTorch module:

1. Layer‑norm the input tensor (last dim = dim).
2. Linear projections → left/right (dim → hidden_dim) and three gates
   (dim → hidden_dim) + sigmoid.
3. Optional mask multiplication (broadcast over the hidden dimension).
4. Apply gates to the projected tensors.
5. Core operation:   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   This is a batched matrix multiplication where each hidden channel d
   (and each batch) performs a GEMM of shape (N×N) × (N×N)ᵀ.
   The heavy reduction over the sequence dimension k is executed by a
   custom Triton kernel (half‑precision, accumulate in FP32).
6. Layer‑norm on the hidden dimension, elementwise out‑gate multiplication,
   and final linear projection back to the original dim.

Only the reduction (step 5) is off‑loaded to Triton; the surrounding
operations stay in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _tri_mul_kernel(
    a_ptr,              # left  : [B*H, M, K]   (float16)
    b_ptr,              # right : [B*H, N, K]   (float16) – we read it transposed
    c_ptr,              # out   : [B*H, M, N]   (float16)
    BATCH, M, N, K,     # logical sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,   # note: b is accessed as (K,N)
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ
    A – (M×K), B – (N×K) (we load B as K×N), C – (M×N)
    The batch dimension is merged into the first program‑id axis.
    """
    pid_batch = tl.program_id(0)      # combined batch*head dimension
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # -------- A tile (M×K) --------
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m
                            + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # -------- B tile (K×N) --------
        # b is stored as (N, K); we read it as (K, N) by swapping strides
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k
                            + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate A·Bᵀ
        acc += tl.dot(a, b)   # dot returns FP32 when inputs are FP16

    # -------- Write result --------
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m
                        + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------ #
    # Cast everything to half precision (except layer‑norm stats which stay fp32)
    # ------------------------------------------------------------------ #
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------ #
    # Layer‑norm on the last dimension
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------ #
    # Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------ #
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------ #
    if not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------ #
    # Gates + sigmoid
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # Core TriMul (batched GEMM across the sequence dimension)
    # ------------------------------------------------------------------ #
    # Move the hidden dimension to the batch axis: shape -> [bs, hidden, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, _, _ = left.shape
    batch_combined = B * H

    # Merge batch & head dimensions for a simple 3‑D layout
    left_flat = left.view(batch_combined, N, N)
    right_flat = right.view(batch_combined, N, N)
    out_flat = torch.empty_like(left_flat)  # half precision output

    # Strides (must be Python integers)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)   # row stride (N)
    stride_a_k = left_flat.stride(2)   # col stride (1)

    stride_b_batch = right_flat.stride(0)
    # For the transposed access we need (K,N) strides:
    stride_b_k = right_flat.stride(2)   # original col stride -> row stride of transposed
    stride_b_n = right_flat.stride(1)   # original row stride -> col stride of transposed

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # Block sizes – tuned for H100 (fp16 GEMM)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_combined,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _tri_mul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        BATCH=batch_combined,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to [bs, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------ #
    # Post‑processing: hidden‑dim layer‑norm, out‑gate, final linear proj.
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # elementwise gating (broadcasted)

    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
        bias=None,
    )

    # Return in float32 as expected by downstream code
    return out.to(torch.float32)
