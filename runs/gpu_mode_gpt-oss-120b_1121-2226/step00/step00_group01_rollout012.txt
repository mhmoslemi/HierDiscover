    a_ptr, b_ptr, c_ptr,
    B, N,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # define block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, N, BLOCK_K):
        # k offset
        k_end = tl.minimum(k + BLOCK_K, N)
        cur_k = k_end - k
        offs_k = tl.arange(0, BLOCK_K)
        # compute masking for k beyond N
        mask_k = offs_k < cur_k

        # Load A block: rows M, cols K
        a_ptrs = a_ptr + batch_idx * stride_a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + (k + offs_k)[None, :] * stride_a_col
        # load B block: rows K, cols N
        b_ptrs = b_ptr + batch_idx * stride_b_batch \
                 + (k + offs_k)[:, None] * stride_b_row \
                 + offs_n[None, :] * stride_b_col

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
            + offs_m[:, None] * stride_c_row \
            + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if not nomask
    if not config.get("nomask", False) and mask is not None:
        mask = mask.to(dtype=x.dtype).unsqueeze(-1)  # shape [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute and reshape
    bs, N, _, _ = left_h.shape
    # Permute dimensions to bring hidden dim forward
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # (bs, H, N, N)
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    # Transpose right for matmul
    right_t = right_perm.transpose(-2, -1).contiguous()  # (bs, H, N, N)

    # Flatten batch and hidden
    left_flat = left_perm.view(bs * hidden_dim, N, N)
    right_flat = right_t.view(bs * hidden_dim, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Triton kernel call
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (bs * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        bs * hidden_dim,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,

        out,
        (hidden_dim,),

        bias=weights["to_out_norm.bias"],
        eps=eps
    )
    # multiply gating
    out = out * out_gate
    # final linear to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul “outgoing” kernel (AlphaFold3 style).

The heavy part of the TriMul forward consists of the contraction

    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]

for every batch and hidden channel `d`.  This is exactly a batched matrix
multiplication between `left` (shape B×H×N×N) and the transpose of `right`
along the pairwise sequence axes.  The O(N³) work is off‑loaded to a custom
Triton kernel that computes a large batch of GEMMs (one per hidden channel).
All surrounding LayerNorm, linear projections and gating are performed with
PyTorch for simplicity.  The kernel works with FP16 inputs (half‑precision
loads) and accumulates in FP32, returning a FP32 result that is later fed to
the output linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton matmul kernel used for the trisum over the “k” dimension.
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr,          # *half  left matrix   (batch, N, N)
    b_ptr,          # *half  right matrix  (batch, N, N)  (already transposed)
    c_ptr,          # *float out matrix   (batch, N, N)
    batch_dim,      # int   total batch = B*H
    N,              # int   sequence length
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B for each element in the batch dimension.
    A: (batch_dim, N, N)   (row-major, FP16)
    B: (batch_dim, N, N)   (row‑major, FP16)  – already transposed so that
                                 the shared “k” dimension is the 2nd axis of A and
                                 the 1st axis of B.
    C: (batch_dim, N, N)   (FP32)
    """
    batch_idx = tl.program_id(0)                # which batch / hidden channel
    pid_m = tl.program_id(1)                    # block row
    pid_n = tl.program_id(2)                    # block col

    # ----------------- block offsets -----------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------- loop over K dimension -----------------
    for kk in range(0, N, BLOCK_K):
        cur_k = tl.minimum(kk + BLOCK_K, N) - kk
        offs_k = tl.arange(0, BLOCK_K)
        mask_k = offs_k < cur_k

        # A block (M x K)
        a_ptrs = a_ptr + batch_idx * stride_a_batch \
                 + offs_m[:, None] * stride_a_row \
                 + (kk + offs_k)[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B block (K x N)
        b_ptrs = b_ptr + batch_idx * stride_b_batch \
                 + (kk + offs_k)[:, None] * stride_b_row \
                 + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot for this tile (accumulate in FP32)
        acc += tl.dot(a, b)

    # ----------------- write back -----------------
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # -----------------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------------
    # LayerNorm on the input (dim‑last)
    # -----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # -----------------------------------------------------------------------
    # Linear projections to hidden dimension
    # -----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # -----------------------------------------------------------------------
    # Optional mask (applied before gating)
    # -----------------------------------------------------------------------
    if (not config.get("nomask", False)) and mask is not None:
        # broadcast mask over the channel dimension
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------------
    # Gating scalars (sigmoid on linear transforms of the normalized input)
    # -----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # Cast to FP16 for the heavy matmul, keep a copy of out_gate in FP32
    # -----------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Bring hidden dimension forward and make tensors contiguous.
    # Layout after permute: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Transpose right so that the shared “k” dimension is the first of the pair.
    right_t = right_perm.transpose(-2, -1).contiguous()

    # Flatten batch + hidden into a single GEMM batch dimension.
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)      # (batch_h, N, N)  FP16
    right_flat = right_t.view(batch_h, N, N)       # (batch_h, N, N)  FP16

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # -----------------------------------------------------------------------
    # Triton kernel launch
    # -----------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_h,
        N,
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # good default for H100 tensor‑cores
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------------
    # Output LayerNorm, gate, final linear projection to original dim
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])  # -> [B, N, N, dim]

    return out
