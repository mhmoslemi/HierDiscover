@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Map pid_batch to batch and h
    b = pid_batch // H
    h = pid_batch % H

    # Compute base pointers for this (b,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Starting indices for this tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Calculate loop bound
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)

        # Masks for out-of-bounds
        mask_i = i < N
        mask_j = j < N
        mask_k = k_idx < N

        # Compute pointers
        # left[i,k]
        left_ptrs = left_base + i[:, None] * stride_left_i + k_idx[None, :] * stride_left_j
        # right_T[k,j] i.e., right[j,k]
        right_ptrs = right_base + j[None, :] * stride_right_i + k_idx[:, None] * stride_right_j

        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left_tile, right_tile)  # shape (BLOCK_M, BLOCK_N)

    # Write back
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    mask_store = (i[:, None] < N) & (j[None, :] < N)
    tl.store(out_ptrs, acc, mask=mask_store)

def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Layer norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if provided
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Triton matmul
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # strides
    stride_b, stride_i, stride_j, stride_h = left.stride()
    # grid launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid
    num_batches = x.shape[0] * hidden_dim
    grid = (num_batches, (x.shape[1] + BLOCK_M - 1) // BLOCK_M,
                     (x.shape[2] + BLOCK_N - 1) // BLOCK_N)

    # launch kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B=x.shape[0],
        N=x.shape[1],
        H=hidden_dim,
        stride_left_b=stride_b,
        stride_left_i=stride_i,
        stride_left_j=stride_j,
        stride_left_h=stride_h,
        stride_right_b=stride_b,
        stride_right_i=stride_i,
        stride_right_j=stride_j,
        stride_right_h=stride_h,
        stride_out_b=stride_b,
        stride_out_i=stride_i,
        stride_out_j=stride_j,
        stride_out_h=stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Apply to_out_norm and gating
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl

BLOCK_M = 64
BLOCK_N = 64
BLOCK_K = 32

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] for all batches b and hidden channels h.
    Each `(b,h)` pair is processed by a separate Triton program.
    The kernel is a fused batched GEMM over the sequence dimensions.
    """
    pid_batch = tl.program_id(0)   # index over B*H pairs
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile col

    # Decompose batch+channel index
    b = pid_batch // H
    h = pid_batch % H

    # Base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Tile coordinates (row i, col j)
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)

        # Bounds masks
        mask_i = i < N
        mask_j = j < N
        mask_k = k_idx < N

        # Load left[i,k]  (shape BLOCK_M x BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k_idx[None, :] * stride_left_j
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # Load right_T[k,j] = right[j,k]  (shape BLOCK_K x BLOCK_N)
        right_ptrs = right_base + k_idx[:, None] * stride_right_j + j[None, :] * stride_right_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Accumulate product
        acc += tl.dot(left_tile, right_tile)

    # Write back
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    mask_out = (i[:, None] < N) & (j[None, :] < N)
    tl.store(out_ptrs, acc, mask=mask_out)


def batched_triton_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for tensors of shape [B, N, N, H]:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    Returns tensor of same shape as inputs.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Ensure contiguous and on GPU
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Extract strides (in elements)
    stride_b, stride_i, stride_j, stride_h = left.stride()

    # Launch grid
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b=stride_b,
        stride_left_i=stride_i,
        stride_left_j=stride_j,
        stride_left_h=stride_h,
        stride_right_b=stride_b,
        stride_right_i=stride_i,
        stride_right_j=stride_j,
        stride_right_h=stride_h,
        stride_out_b=stride_b,
        stride_out_i=stride_i,
        stride_out_j=stride_j,
        stride_out_h=stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # enough warps for good occupancy on H100
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    The heavy O(N³) triangular multiplication is performed by a fused
    Triton kernel (batched GEMM) while the surrounding LayerNorm,
    projections, gating and final linear projection use native PyTorch
    ops.
    """
    # Unpack inputs
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of a bias‑less linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triangular multiplication – main O(N³) work
    # ------------------------------------------------------------------
    out = batched_triton_gemm(left, right)   # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Output LayerNorm, gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul “outgoing” forward pass.

The heavy O(N³) triangular multiplication is performed by a fused Triton
kernel that treats each (batch, hidden‑channel) pair as an independent
matrix multiply:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

All surrounding operations (LayerNorm, bias‑less linear projections,
sigmoid gates, optional mask, output LayerNorm and final linear
projection) are executed with native PyTorch kernels.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum '... i k d, ... j k d -> ... i j d'
# ----------------------------------------------------------------------
BLOCK_M = 64   # rows  per program
BLOCK_N = 64   # cols  per program
BLOCK_K = 32   # reduction tile size

@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Each (b,h) pair is processed by a separate Triton program.
    """
    pid_batch = tl.program_id(0)   # index over B*H
    pid_m = tl.program_id(1)       # tile row
    pid_n = tl.program_id(2)       # tile column

    # --------------------------------------------------------------
    # 1) Decode batch index and hidden‑channel index
    # --------------------------------------------------------------
    b = pid_batch // H
    h = pid_batch % H

    # Base pointers for the current (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # --------------------------------------------------------------
    # 2) Tile coordinates
    # --------------------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows  (i‑index)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols  (j‑index)

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_offset = kk * BLOCK_K
        k = k_offset + tl.arange(0, BLOCK_K)   # k‑index inside the tile

        # Bounds masks
        mask_i = i < N
        mask_j = j < N
        mask_k = k < N

        # left[i,k]  (shape BLOCK_M × BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_j
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right_T[k,j] = right[j,k]  (shape BLOCK_K × BLOCK_N)
        right_ptrs = right_base + k[:, None] * stride_right_j + j[None, :] * stride_right_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Accumulate dot product
        acc += tl.dot(left_tile, right_tile)

    # --------------------------------------------------------------
    # 4) Write the result back
    # --------------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    mask_out = (i[:, None] < N) & (j[None, :] < N)
    tl.store(out_ptrs, acc, mask=mask_out)


def batched_triton_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the kernel above.
    Input/Output shape: [B, N, N, H].
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # stride in elements (not bytes)
    stride_b, stride_i, stride_j, stride_h = left.stride()

    # grid dimensions: (batch*channel, tiles along i, tiles along j)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the kernel
    tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b=stride_b,
        stride_left_i=stride_i,
        stride_left_j=stride_j,
        stride_left_h=stride_h,
        stride_right_b=stride_b,
        stride_right_i=stride_i,
        stride_right_j=stride_j,
        stride_right_h=stride_h,
        stride_out_b=stride_b,
        stride_out_i=stride_i,
        stride_out_j=stride_j,
        stride_out_h=stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                         # good occupancy on H100
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Steps:
      1. Layer‑norm the 4‑D input.
      2. Linear projections (left/right) without bias.
      3. Optional mask application.
      4. Compute three sigmoid gates (left, right, output).
      5. Apply left/right gates element‑wise.
      6. Perform the O(N³) triangular multiplication via a fused Triton GEMM.
      7. Output LayerNorm, multiply by the output gate, and final linear projection.
    Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    # unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑less)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid of bias‑less linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # 5) Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Triangular multiplication (the O(N³) core)
    # ------------------------------------------------------------------
    out = batched_triton_gemm(left, right)    # shape [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear projection back to dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
