import triton
import triton.language as tl

@triton.jit
def trmul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    
    pid0 = tl.program_id(0)  # combined batch*head
    pid1 = tl.program_id(1)  # block i (row)
    pid2 = tl.program_id(2)  # block j (col)
    
    # Decode batch and hidden index.
    b = pid0 // H
    h = pid0 % H
    
    # offsets for i and j
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for valid i,j < N
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        
        # Compute pointers
        left_ptrs = left_ptr + b*stride_left_b + h*stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # right as (K,N) using transpose via strides
        right_ptrs = right_ptr + b*stride_right_b + h*stride_right_h + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        
        # load
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        # accumulate
        acc += tl.dot(left, right)
    
    # store result
    out_ptrs = out_ptr + b*stride_out_b + h*stride_out_h + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1️⃣ Layer‑normalize the input tensor `x` over the last dimension (`dim`).
2️⃣ Project the normalized tensor into a hidden space (`hidden_dim`) with two
   linear layers → `left`, `right`.
3️⃣ (Optional) Apply the pairwise mask.
4️⃣ Compute three sigmoid gates from the same normalized tensor:
   `left_gate`, `right_gate`, `out_gate`.
5️⃣ Fuse the gates into the projections:
   `left  = left  * left_gate`  ,  `right = right * right_gate`.
6️⃣ **GPU kernel** – for every `(batch, head)` pair compute  
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`  
   i.e. a batched matrix multiplication `left @ rightᵀ` where the
   multiplication is performed in float16 and accumulated in float32.
   The kernel tiles the computation (M, N, K) = (128, 128, 32) and
   respects the batch and hidden‑dim axes via a combined program‑ID.
7️⃣ Layer‑normalize the result over the hidden dimension, multiply by
   `out_gate`, and apply a final linear projection back to `dim`.

At least the cubic‑complex part (step 6) is executed inside a Triton kernel,
while the surrounding linear / layer‑norm steps use highly‑optimized
PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel:   out = left @ rightᵀ   (batch‑wise & head‑wise)
# ------------------------------------------------------------
@triton.jit
def trmul_out_kernel(
    left_ptr,            # float16 *
    right_ptr,           # float16 *
    out_ptr,             # float32 *
    B, N, H,             # int32: batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid0 = tl.program_id(0)          # combined (batch, head) dimension
    pid1 = tl.program_id(1)          # tile along i (rows)
    pid2 = tl.program_id(2)          # tile along j (cols)

    # decode batch index b and hidden‑dim index h
    b = pid0 // H
    h = pid0 % H

    # tile offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (M, K)  (i, k)
        left_ptrs = (left_ptr
                     + b * stride_left_b
                     + h * stride_left_h
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # rightᵀ: we need (K, N) where rows = k, cols = j.
        # Original layout is (b, j, k, h); swapping k and j via strides.
        right_ptrs = (right_ptr
                      + b * stride_right_b
                      + h * stride_right_h
                      + offs_k[:, None] * stride_right_k
                      + offs_j[None, :] * stride_right_i)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate the outer product over k
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # write back
    out_ptrs = (out_ptr
                + b * stride_out_b
                + h * stride_out_h
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ------------------------------------------------------------
# Entry point expected by the evaluator
# ------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N] (float32 or bool)
            - weights: dict of model parameters (LayerNorm, Linear weights)
            - config:  dict with at least `dim`, `hidden_dim`,
                       optional `nomask` (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (`dim`)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections to hidden space (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_broadcast = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gates into projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ GPU kernel: out = left @ rightᵀ  (per batch & head)
    # ------------------------------------------------------------------
    # Convert to fp16 for higher throughput, keep accumulator in fp32.
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_fp16.device)

    # Strides (in elements)
    stride_l_b = left_fp16.stride(0)
    stride_l_i = left_fp16.stride(1)
    stride_l_k = left_fp16.stride(2)
    stride_l_h = left_fp16.stride(3)

    stride_r_b = right_fp16.stride(0)
    stride_r_i = right_fp16.stride(1)
    stride_r_k = right_fp16.stride(2)
    stride_r_h = right_fp16.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_h = out.stride(3)

    # Tiling parameters – chosen to fit H100 shared memory / register budget
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                          # program_id(0) → (batch, head)
        (N + BLOCK_M - 1) // BLOCK_M,           # program_id(1) → i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # program_id(2) → j‑tiles
    )

    trmul_out_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel has finished before the next PyTorch ops
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate          # element‑wise gate
    output = F.linear(out_norm, weights["to_out.weight"])   # → [B, N, N, dim]

    return output
