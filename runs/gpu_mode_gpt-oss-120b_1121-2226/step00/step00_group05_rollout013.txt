@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # batch * hidden
    # Determine batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # Compute offsets for i and j
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for boundary
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load left tile (shape BLOCK_M, BLOCK_K)
        left_offset = b * stride_left_b + i[:, None] * stride_left_i + k[None, :] * stride_left_k + h * stride_left_h
        left = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed (shape BLOCK_K, BLOCK_N)
        # We need right[j, k, h], but we want shape (BLOCK_K, BLOCK_N) to dot.
        right_offset = b * stride_right_b + k[:, None] * stride_right_k + j[None, :] * stride_right_j + h * stride_right_h
        right = tl.load(right_ptr + right_offset, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot over k dimension: left (M,K) dot right (K,N) => (M,N)
        acc += tl.dot(left, right)  # result float32

    # Write back output
    out_offset = b * stride_out_b + i[:, None] * stride_out_i + j[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptr + out_offset, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (Outgoing) – Triton‐accelerated forward pass

The heavy part of the TriMul module is the 4‑D contraction

    out[b,i,j,d] = Σ_k  left[b,i,k,d] * right[b,j,k,d]

where ``left`` and ``right`` are the linearly projected and gated tensors of
shape (B, N, N, H).  This is equivalent to H independent matrix multiplications
 `A_d @ B_d^T` (A_d,B_d ∈ ℝ^{N×N}) and costs O(B·H·N³) FLOPs.
A custom Triton kernel tiles the i‑ and j‑dimensions (M×N tiles) and iterates
over ``k`` in blocks, accumulating a (M,N) matrix per batch‑hidden pair.
All other operations (LayerNorm, Linear, gating, masking) are performed with
standard PyTorch for simplicity.

The kernel works on one hidden channel at a time (program‑id z = batch·H) and
stores the result in float32 while reading ``left``/``right`` in float16 to
reduce bandwidth.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,            # const float16 *
    right_ptr,           # const float16 *
    out_ptr,             # float32 *
    B, N, H,             # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int32
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # int32
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # int32
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Tile‐wise matrix‑multiply for one hidden channel.
    The grid is (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H).
    """
    pid_i = tl.program_id(0)               # i‑tile index
    pid_j = tl.program_id(1)               # j‑tile index
    pid_z = tl.program_id(2)               # combined (batch, hidden) index

    # decode batch and hidden channel
    b = pid_z // H
    h = pid_z % H

    # --------------------------------------------------------------------- #
    #   Tile offsets
    # --------------------------------------------------------------------- #
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # [M]
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # [N]
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------------- #
    #   Loop over the k‑dimension (the “summation” axis)
    # --------------------------------------------------------------------- #
    k = 0
    while k < N:
        offs_k = k + tl.arange(0, BLOCK_K)          # [K]
        mask_k = offs_k < N

        # ---- left tile : shape (M, K) ----------------------------------- #
        left_addr = (
            b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(
            left_ptr + left_addr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
            dtype=tl.float16,
        )

        # ---- right tile : shape (K, N)   (note the transpose) ----------- #
        # right[j, k, h] is stored as (batch, j, k, h); we need (k, j)
        right_addr = (
            b * stride_r_b
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i   # stride_r_i corresponds to the “j” axis
            + h * stride_r_h
        )
        right = tl.load(
            right_ptr + right_addr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0),
            dtype=tl.float16,
        )

        # Promote to FP32 for the dot product
        left_f = left.to(tl.float32)
        right_f = right.to(tl.float32)

        # Accumulate the partial matrix product
        acc += tl.dot(left_f, right_f)   # (M,K) @ (K,N) -> (M,N)

        k += BLOCK_K

    # --------------------------------------------------------------------- #
    #   Store the result for this (batch, hidden) pair
    # --------------------------------------------------------------------- #
    out_addr = (
        b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(
        out_ptr + out_addr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block with a Triton kernel for the
    O(N³·H) contraction.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be ignored
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    #   unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    #   Helper to fetch weight tensors (they are already on the correct device)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    #   1) LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )   # [B,N,N,dim]

    # --------------------------------------------------------------------- #
    #   2) Linear projections (no bias) → half precision for memory bandwidth
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, left_proj_weight).to(torch.float16)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight).to(torch.float16)  # [B,N,N,H]

    # --------------------------------------------------------------------- #
    #   3) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    #   4) Gating linear layers (sigmoid)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight)) # [B,N,N,H]
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))    # [B,N,N,H]

    # Apply gates (convert gates to half for the element‑wise product)
    left = left * left_gate.to(left.dtype)
    right = right * right_gate.to(right.dtype)

    # --------------------------------------------------------------------- #
    #   5) Triton kernel – compute the O(N³·H) tensor contraction
    # --------------------------------------------------------------------- #
    out_pre = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides in elements (not bytes) for all tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_pre.stride()

    # tile sizes – selected to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * hidden_dim,                        # batch × hidden channel
    )

    _trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_pre.data_ptr(),
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------------- #
    #   6) Post‑processing: LayerNorm, out‑gate, final Linear
    # --------------------------------------------------------------------- #
    # layer‑norm on the hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out_pre,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # apply the out‑gate (still float32)
    out_norm = out_norm * out_gate

    # final projection back to `dim`
    output = torch.nn.functional.linear(out_norm, to_out_weight)   # [B,N,N,dim]

    return output
