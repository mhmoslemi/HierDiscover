def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)  # combine batch and output hyper-dim block
    batch = pid // tl.cdiv(H, BLOCK_H)
    h_block = (pid % tl.cdiv(H, BLOCK_H)) * BLOCK_H
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_h = h_block + tl.arange(0, BLOCK_H)

    # Masks for OOB
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N
    mask_h = offs_h < H

    # Pointers offsets
    # left shape: (B, N_i, N_k, H)
    left_ptrs = left_ptr + batch * stride_lb \
                + offs_i[:, None, None] * stride_li \
                + offs_k[None, :, None] * stride_lk \
                + offs_h[None, None, :] * stride_lh

    # right transposed shape: (B, N_k, N_j, H)
    right_ptrs = right_ptr + batch * stride_rb \
                + offs_k[:, None, None] * stride_rk \
                + offs_j[None, :, None] * stride_rj \
                + offs_h[None, None, :] * stride_rh

    # Output ptr: shape (B, N_i, N_j, H)
    out_ptrs = out_ptr + batch * stride_ob \
                + offs_i[:, None, None] * stride_oi \
                + offs_j[None, :, None] * stride_oj \
                + offs_h[None, None, :] * stride_oh

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Iterate over K dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + offs_k
        mask_k_cur = cur_k < N
        # Compute k offset
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k_cur[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=mask_k_cur[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D pair representation `x`  →  `x_norm`.
2. Linear projections (no bias) to the hidden dimension `h_dim`:
      left  = x_norm @ left_proj_weight
      right = x_norm @ right_proj_weight
3. Compute three sigmoid gates (also linear, no bias) and apply them:
      left  = left  * sigmoid(left_gate (x_norm))
      right = right * sigmoid(right_gate(x_norm))
4. Optional pairwise mask (shape [B,N,N]) is broadcast and applied to
   `left` and `right`.
5. Core multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This O(N³·h) contraction is performed by a Triton kernel that
   processes a tile of (i, j) together with a small block of the hidden
   dimension (BLOCK_H).  For each hidden channel the kernel executes a
   small matrix‑multiply `left_tile @ right_tile.T` accumulating over K.
6. Layer‑norm across the hidden dimension, an output gate, and a final
   linear projection back to `dim`.

The kernel is deliberately tiled (BLOCK_M × BLOCK_N × BLOCK_H) to fit
in registers on an H100 while keeping the reduction over K in a tight
loop.  All internal accumulations use fp32 for numerical stability,
while the bulk of the data lives in fp16 to reduce memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,            # *float16
    right_ptr,           # *float16
    out_ptr,             # *float32
    B, N, H,             # int32 scalars
    stride_lb, stride_li, stride_lk, stride_lh,   # int64 strides for left
    stride_rb, stride_rj, stride_rk, stride_rh,   # int64 strides for right
    stride_ob, stride_oi, stride_oj, stride_oh,   # int64 strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Grid dimensions:
        (pid0) = batch * hidden‑block
        (pid1) = i‑tile
        (pid2) = j‑tile
    """

    pid_bh = tl.program_id(0)
    pid_i  = tl.program_id(1)
    pid_j  = tl.program_id(2)

    # --------------------------------------------------------------
    # 1) Derive batch index and hidden‑block offset
    # --------------------------------------------------------------
    hidden_blocks = tl.cdiv(H, BLOCK_H)          # number of blocks in hidden dim
    batch_id   = pid_bh // hidden_blocks
    h_block    = (pid_bh % hidden_blocks) * BLOCK_H

    # --------------------------------------------------------------
    # 2) Offsets within the current tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # [M]
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # [N]
    offs_k = tl.arange(0, BLOCK_K)                     # [K]
    offs_h = h_block + tl.arange(0, BLOCK_H)           # [H]

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # --------------------------------------------------------------
    # 3) Base pointers for the current batch and hidden‑block
    # --------------------------------------------------------------
    # left[b,i,k,h]
    left_base = left_ptr + batch_id * stride_lb \
                + offs_i[:, None, None] * stride_li \
                + offs_h[None, None, :] * stride_lh

    # right[b,j,k,h] – we will load it with the K‑axis as the leading dim
    right_base = right_ptr + batch_id * stride_rb \
                 + offs_j[None, :, None] * stride_rj \
                 + offs_h[None, None, :] * stride_rh

    # --------------------------------------------------------------
    # 4) Accumulator for the tile (M×N×H)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # 5) Loop over K dimension in blocks
    # --------------------------------------------------------------
    for start_k in range(0, N, BLOCK_K):
        cur_k  = start_k + offs_k                 # absolute K indices
        mask_k = cur_k < N

        # left_tile:  (M, K, H)
        a_ptrs = left_base + cur_k[None, :, None] * stride_lk
        # right_tile: (K, N, H) – K is now the leading dim
        b_ptrs = right_base + cur_k[:, None, None] * stride_rk

        # Load with masks (oob elements are zero)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                    other=0.0).to(tl.float32)          # fp16 → fp32
        b = tl.load(b_ptrs,
                    mask=mask_j[None, :, None] & mask_k[:, None, None] & mask_h[None, None, :],
                    other=0.0).to(tl.float32)

        # For each hidden slice compute a small matrix‑multiply
        for hh in range(0, BLOCK_H):
            a_h = a[:, :, hh]          # (M, K)
            b_h = b[:, :, hh]          # (K, N)
            acc[:, :, hh] += tl.dot(a_h, b_h)   # (M, N)

    # --------------------------------------------------------------
    # 6) Store the result (float32)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + batch_id * stride_ob \
               + offs_i[:, None, None] * stride_oi \
               + offs_j[None, :, None] * stride_oj \
               + offs_h[None, None, :] * stride_oh

    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]   (float32)
            - mask  : torch.Tensor [B, N, N]       (bool/float, optional)
            - weights: dict with the exact keys used in the PyTorch reference
            - config : dict with "dim" and "hidden_dim"

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]        # called hidden_dim in the reference

    # ------------------------------------------------------------------
    # 1) LayerNorm on the pair representation (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,dim] float32

    # Cast to float16 for the heavy linear work
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – cast weights to fp16 once
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)          # [B,N,N,hidden_dim] fp16
    right = F.linear(x_f16, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(input_tensor.dtype)               # usually float32
        mask = mask.unsqueeze(-1).to(torch.float16)      # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # Ensure contiguity for the kernel
    left  = left.contiguous()
    right = right.contiguous()

    B, N_, N2, H = left.shape
    assert N_ == N2, "non‑square pair tensor not supported"
    assert H == hidden_dim, "weights hidden dimension mismatch"

    # ------------------------------------------------------------------
    # 5) Triton kernel: multiplicative update over the third axis
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32)  # accumulator will be fp32

    # Strides in element units
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tiling parameters (tuned for H100)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 8   # hidden‑dim is small; 8 gives good register usage

    hidden_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (
        B * hidden_blocks,                           # batch × hidden‑block
        (N + BLOCK_M - 1) // BLOCK_M,                # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,                # j‑tiles
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6) Output post‑processing (LayerNorm → gate → final projection)
    # ------------------------------------------------------------------
    # Layer‑norm on hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )  # still fp32

    # Apply output gate (convert gate to fp32 first)
    out_gate = out_gate.to(out.dtype)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]          # shape [dim, hidden_dim], float32
    out = F.linear(out, to_out_w)                # [B,N,N,dim] float32

    return out
