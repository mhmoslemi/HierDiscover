@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_h_idx = tl.program_id(axis=2)  # 0 .. B*H-1
    pid_i = tl.program_id(axis=0)
    pid_j = tl.program_id(axis=1)

    b = batch_h_idx // H
    h = batch_h_idx % H

    # compute offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # base pointers
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate over k dimension
    for _k in range(0, tl.cdiv(N, BLOCK_K)):
        # load A block: shape (BLOCK_I, BLOCK_K)
        a_ptrs = left_base + (offs_i[:, None] * stride_l_i) + ((offs_k[None, :] + _k*BLOCK_K) * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (offs_k + _k*BLOCK_K < N), other=0.0).to(tl.float32)

        # load B block transposed: shape (BLOCK_K, BLOCK_J)
        k_offset = offs_k + _k*BLOCK_K
        b_ptrs = right_base + (k_offset[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptrs, mask=(k_offset[:, None] < N) & mask_j[None, :], other=0.0).to(tl.float32)

        # compute multiplication
        acc += tl.dot(a, b)  # shapes: (BLOCK_I, BLOCK_K) dot (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)

    # write out
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16) if left.dtype==tl.float16 else acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) fused implementation.

The kernel fuses the core N‑by‑N‑by‑hidden_dim einsum:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is equivalent to a batched matrix multiplication per hidden channel.
The surrounding LayerNorms, linear projections, gating and masking are handled
by PyTorch.  The triton kernel computes the O(N³·H) update efficiently using
tiling over the (i, j, k) dimensions and a combined batch‑hidden dimension
( B·H ).  It works for fp32/fp16 data and supports arbitrary sequence lengths
up to the tested 1024‑element case.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM over the (i,j) matrix for each hidden channel
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    B, N, H,                               # batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    using a tiled matrix‑multiply approach.
    """
    # ------------------------------------------------------------------
    # Determine which (batch, hidden) this program instance works on
    # ------------------------------------------------------------------
    batch_h_idx = tl.program_id(axis=2)          # 0 .. B*H-1
    pid_i = tl.program_id(axis=0)                # tile row (i)
    pid_j = tl.program_id(axis=1)                # tile col (j)

    # De‑compose into batch b and hidden channel h
    b = batch_h_idx // H
    h = batch_h_idx %  H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k_iter in range(num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load A block: shape (BLOCK_I, BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Load B block transposed: shape (BLOCK_K, BLOCK_J)
        # (right is indexed as (b, j, k, h); we need (k, j) order)
        # --------------------------------------------------------------
        b_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # --------------------------------------------------------------
        # Update accumulator
        # --------------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be None when config["nomask"] == True)
            - weights: dict of model weights (see keys in the docstring)
            - config: dict containing "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load weight tensors (all are assumed to be on the same device)
    # ------------------------------------------------------------------
    norm_weight         = weights["norm.weight"]
    norm_bias           = weights["norm.bias"]
    left_proj_weight    = weights["left_proj.weight"]
    right_proj_weight   = weights["right_proj.weight"]
    left_gate_weight    = weights["left_gate.weight"]
    right_gate_weight   = weights["right_gate.weight"]
    out_gate_weight     = weights["out_gate.weight"]
    to_out_norm_weight  = weights["to_out_norm.weight"]
    to_out_norm_bias    = weights["to_out_norm.bias"]
    to_out_weight       = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(x,
                          normalized_shape=(dim,),
                          weight=norm_weight,
                          bias=norm_bias,
                          eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left  = F.linear(x_norm, left_proj_weight)   # (out_features=hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask application (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)      # [B,N,N,1]
        left  = left  * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core triangular multiplicative update (fused Triton kernel)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=dtype)

    # Strides for pointer arithmetic (int64 values)
    s_l = left.stride()   # (B_stride, i_stride, k_stride, h_stride)
    s_r = right.stride()
    s_o = out.stride()

    # Choose tile sizes (tuned for H100, fp32)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Grid: (tiles_i, tiles_j, B*H)
    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate
    out = F.linear(out, to_out_weight)   # -> [B,N,N,dim]

    # Ensure final output is float32 as required
    if out.dtype != torch.float32:
        out = out.to(torch.float32)

    return out
