@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    pid_rem = pid % (num_blocks_m * num_blocks_n)
    block_i = pid_rem // num_blocks_n
    block_j = pid_rem % num_blocks_n

    # offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointer offsets for a and b (b is the transposed version)
    a_ptrs = a_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M) & (tl.arange(0, BLOCK_K) + k * BLOCK_K < K), other=0.0, eviction_policy='evict_last')
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) + k * BLOCK_K < K) & (offs_n < N), other=0.0, eviction_policy='evict_last')
        # compute dot
        acc += tl.dot(a, b)
        # advance ptrs
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write output
    c_ptrs = c_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float32), mask=(offs_m < M) & (offs_n < N))

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Compute block coordinate
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    pid_rem = pid % (num_blocks_m * num_blocks_n)
    block_i = pid_rem // num_blocks_n
    block_j = pid_rem % num_blocks_n

    # Offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = a_ptr + batch_idx * stride_a_batch \
                        + offs_m[:, None] * stride_a_m \
                        + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch \
                        + tl.arange(0, BLOCK_K)[:, None] * stride_b_k \
                        + offs_n[None, :] * stride_b_n

    # Mask for valid indices
    mask_m = offs_m < M
    mask_n = offs_n < N
    # K loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute valid mask for K dimension per tile
        k_valid = (tl.arange(0, BLOCK_K) + k * BLOCK_K) < K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)
        acc += tl.dot(a, b)  # dot returns fp32

        # Move pointers to next K tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Store
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
                        + offs_m[:, None] * stride_c_m \
                        + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # Compute block coordinates
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_blocks_m * num_blocks_n)
    pid_rem = pid % (num_blocks_m * num_blocks_n)
    block_i = pid_rem // num_blocks_n
    block_j = pid_rem % num_blocks_n

    # Offsets within the block
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers in global memory
    a_ptrs = a_ptr + batch_idx * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

    # Masks for out-of-bound accesses
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_valid = (tl.arange(0, BLOCK_K) + k_tile * BLOCK_K) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write back results
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n

    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul “outgoing” implementation (AlphaFold‑3 style) with a fused
N³ reduction.

The heavy part of the forward pass is
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]    (d = hidden_dim)

which is exactly a batched matrix‑multiply
        Lᵈ (N×N)  ×  (Rᵈ)ᵀ (N×N)  →  Oᵈ (N×N)
for every hidden channel d.  The kernel below performs a
standard GEMM on a batch dimension that is the product (batch × hidden_dim).
All other operations (LayerNorms, linear projections, sigmoids,
mask handling) are carried out with regular torch primitives.
The kernel uses FP16 loads/computes and accumulates in FP32,
leveraging H100 Tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                     # logical sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM :   A (M×K) × B (K×N) → C (M×N)  for many batches."""
    pid = tl.program_id(0)

    # blocks per‑matrix
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode 1‑D program id → (batch, block_i, block_j)
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    pid_rem   = pid %  (num_blocks_m * num_blocks_n)
    block_i   = pid_rem // num_blocks_n
    block_j   = pid_rem %  num_blocks_n

    # coordinate offsets inside the tile
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointer arithmetic
    a_ptrs = a_ptr + batch_idx * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K‑tiles
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        k_valid = (tl.arange(0, BLOCK_K) + k_tile * BLOCK_K) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # tensor‑core dot, accumulation in FP32
        acc += tl.dot(a, b)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write result (cast back to FP16)
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape (batch, seq_len, seq_len, dim)
    """
    # ------------------------------------------------------------------
    # unpack inputs & configuration
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]               # input feature dim
    hidden_dim = config["hidden_dim"] # inner (projection) dim
    eps = 1e-5                        # LayerNorm epsilon

    # ------------------------------------------------------------------
    # weight look‑ups
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections & gating logits (all still FP32)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)          # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)    # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4) Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy N³ reduction – fused batched GEMM via Triton
    # ------------------------------------------------------------------
    # Cast to half‑precision before the reduction
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape          # B = batch, H = hidden_dim

    # bring hidden dimension into the *batch* axis
    # left:  [B, H, N, N]   right (transposed K↔J): [B, H, N, N]
    left_perm  = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()   # K ↔ J

    # flatten batch×hidden → one GEMM batch per hidden channel
    left_flat  = left_perm.view(-1, N, N)   # shape (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)  # shape (B*H, N, N)

    total_batches = left_flat.shape[0]      # B * hidden_dim
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # strides for the flattened tensors (row‑major layout)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # block sizes – chosen to hit Tensor Cores on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        total_batches
        * ((N + BLOCK_M - 1) // BLOCK_M)
        * ((N + BLOCK_N - 1) // BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        a_ptr=left_flat.data_ptr(),
        b_ptr=right_flat.data_ptr(),
        c_ptr=out_flat.data_ptr(),
        batch=total_batches,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 and reshape to [B, N, N, hidden_dim]
    out = out_flat.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                         # element‑wise gate
    out = F.linear(out, to_out_weight)           # [B, N, N, dim]

    return out
