@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_j, stride_left_h,
    stride_gate_batch, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_batch, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < (B * N * N * H)
    # compute b,i,j,h indices
    tmp = offsets
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N
    # compute pointers
    left_offset = b * stride_left_batch + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_offset = b * stride_left_batch + i * stride_left_i + j * stride_left_j + h * stride_left_h # right has same shape
    # but right is separate pointer though stride same.
    left_ptrs = left_ptr + left_offset
    right_ptrs = right_ptr + left_offset # same shape
    left_gate_offset = b * stride_gate_batch + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    right_gate_offset = b * stride_gate_batch + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h
    left_gate_ptrs = left_gate_ptr + left_gate_offset
    right_gate_ptrs = right_gate_ptr + right_gate_offset
    mask_offset = b * stride_mask_batch + i * stride_mask_i + j * stride_mask_j
    mask_ptrs = mask_ptr + mask_offset
    # load
    left = tl.load(left_ptrs, mask=mask)
    right = tl.load(right_ptrs, mask=mask)
    left_gate = tl.load(left_gate_ptrs, mask=mask)
    right_gate = tl.load(right_gate_ptrs, mask=mask)
    mask_val = tl.load(mask_ptrs, mask=mask).to(tl.float16) # or cast, mask tensor is maybe bool or float.
    # compute

    right = right * right_gate * mask_val
    # store

    tl.store(right_ptrs, right, mask=mask)

def custom_kernel(data):
   input_tensor, mask, weights, config = data
   bs, N, _, dim = input_tensor.shape
   hidden_dim = config["hidden_dim"]

   # Cast to float32
   x = input_tensor

   # Layernorm
   # compute mean over last dim: using torch.mean for speed with fp32.
   # We'll use torch.nn.functional.layer_norm.
   norm_weight = weights["norm.weight"]
   norm_bias = weights["norm.bias"]
   x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

   # Projections
   left_proj_weight = weights["left_proj.weight"]
   right_proj_weight = weights["right_proj.weight"]
   # Linear returns shape same but last dim = hidden_dim
   left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [bs,N,N,hidden_dim]
   right = torch.nn.functional.linear(x_norm, right_proj_weight)

   # Gates
   left_gate_weight = weights["left_gate.weight"]
   right_gate_weight = weights["right_gate.weight"]
   out_gate_weight = weights["out_gate.weight"]

   left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
   right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
   out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

   # Optionally cast to float16 before mask/gate? We'll remain float16 after multiplication for speed.
   left = left.to(torch.float16)
   right = right.to(torch.float16)
   left_gate = left_gate.to(torch.float16)
   right_gate = right_gate.to(torch.float16)

   # Apply mask and gates with triton kernel (inplace)
   # Note: mask shape [bs,N,N] may be bool or float. We'll cast to float16.
   if mask is None:
       # create broadcast mask of ones
       mask_tensor = torch.ones((bs,N,N), dtype=torch.float16, device=left.device)
   else:
       mask_tensor = mask.to(torch.float16)

   # Launch kernel
   total_elements = bs * N * N * hidden_dim
   BLOCK = 1024  # can be 256? Use 1024 as typical

   grid = (triton.cdiv(total_elements, BLOCK),)

   mask_gate_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_tensor,
        bs, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask_tensor.stride(0), mask_tensor.stride(1), mask_tensor.stride(2),
        BLOCK_SIZE=BLOCK
   )
   # At this point left and right have mask applied and gates.

   # Now compute out via batched matmul.
   # reshape to (B*H,N,N)
   left_reshape = left.permute(0,3,1,2).reshape(bs*hidden_dim, N, N)
   right_reshape = right.permute(0,3,1,2).reshape(bs*hidden_dim, N, N)

   # Compute batched matmul. Use torch.bmm expects (B,N,K)*(B,K,M).
   out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1,2))
   # Reshape back
   out = out_reshape.view(bs, hidden_dim, N, N).permute(0,2,3,1).to(torch.float32)

   # to_out_norm
   to_out_norm_weight = weights["to_out_norm.weight"]
   to_out_norm_bias = weights["to_out_norm.bias"]
   out_normed = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

   # Multiply by out_gate (float32)
   out_gate = out_gate.to(torch.float32)  # shape [bs,N,N,H]
   out_normed = out_normed * out_gate

   # final linear to dim
   to_out_weight = weights["to_out.weight"]
   out_final = torch.nn.functional.linear(out_normed, to_out_weight) # shape [bs,N,N,dim]

   return out_final

"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Linear projections (no bias) → `left` and `right` (shape B×N×N×H).
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. Fuse the elementwise operations
        left = left * left_gate * mask
        right = right * right_gate * mask
   with a small Triton kernel (the only required Triton part).
   All tensors are cast to FP16 before the kernel for bandwidth efficiency.
5. Compute the core TriMul bilinear product:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   This is a batched matrix‑multiply per hidden dimension.
   By reshaping we obtain a single `torch.bmm` call on FP16 data:
        (B·H, N, N) @ (B·H, N, N)^T   ->   (B·H, N, N)
6. Reshape back to (B,N,N,H) and cast to FP32.
7. Apply a second LayerNorm over the hidden dimension, multiply by `out_gate`,
   and finally a linear projection back to `dim`.

Only the elementwise fusion is done in Triton; the heavy matrix multiplication
leverages cuBLAS (via `torch.bmm`) which is highly tuned on H100 GPUs.
The implementation works for any batch size (1‑2), sequence length up to 1024,
and both masked / unmasked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and sigmoid‑gates into left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                 # FP16 tensors (flattened)
    left_gate_ptr, right_gate_ptr,       # FP16 tensors (flattened)
    mask_ptr,                           # FP16 tensor (flattened, same size as left/right)
    size,                               # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise: left = left * left_gate * mask ; right = right * right_gate * mask."""
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load values (masked loads return 0.0 for out‑of‑bounds threads)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    # Fuse
    left = left * left_gate * m
    right = right * right_gate * m

    # Store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor[B, N, N, dim] (float32)
              - mask: torch.Tensor[B, N, N] (bool/float, optional)
              - weights: dict of torch.Tensors (see code for keys)
              - config: dict with at least 'dim' and 'hidden_dim' (ints)

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # often 128
    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over the channel dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim], float32
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # fp32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Cast to FP16 and fuse mask + gates with Triton
    # ------------------------------------------------------------------
    # Convert to contiguous FP16 tensors
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    left_gate  = left_gate.to(torch.float16).contiguous()
    right_gate = right_gate.to(torch.float16).contiguous()

    # Prepare mask (broadcast to hidden_dim)
    if mask is None:
        mask_tensor = torch.ones((bs, N, N), dtype=torch.float16, device=left.device)
    else:
        # mask may be bool or float; ensure float16
        mask_tensor = mask.to(torch.float16)
    mask_expand = mask_tensor.unsqueeze(-1).expand(-1, -1, -1, hidden_dim)
    mask_expand = mask_expand.contiguous()

    # Flatten everything for the 1‑D Triton kernel
    size = left.numel()
    BLOCK = 1024  # good trade‑off for H100
    grid = (triton.cdiv(size, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expand,
        size=size,
        BLOCK_SIZE=BLOCK,
    )
    # ------------------------------------------------------------------
    # 5. Core TriMul: batched matmul per hidden dimension
    # ------------------------------------------------------------------
    # left, right: [B,N,N,H] FP16
    # Reshape to (B*H, N, N)
    left_reshape  = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    # Batched matmul: left @ right^T  (FP16)
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))

    # Reshape back to [B, N, N, H] and convert to FP32 for the rest
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B,N,N,H] float32

    out_gate = out_gate.to(torch.float32)               # already FP32 but ensure dtype
    out = out * out_gate                               # gating

    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)                 # final projection, float32

    return out
