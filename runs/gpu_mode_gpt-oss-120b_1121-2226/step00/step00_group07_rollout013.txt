```
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    d_start,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets within block
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d_start + tl.arange(0, BLOCK_D)

    for b in range(B):
        # init accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        for k_iter in range(0, N, BLOCK_K):
            # compute valid masks for i,j,k,d
            mask_i = offs_i < N
            mask_j = offs_j < N
            mask_k = (k_iter + offs_k) < N
            mask_d = offs_d < H

            # compute offsets for loading
            left_ptrs = left_ptr + b*stride_left_b \
                + (offs_i[:, None, None] * stride_left_i) \
                + ((k_iter + offs_k)[None, :, None] * stride_left_k) \
                + (offs_d[None, None, :] * stride_left_h)
            right_ptrs = right_ptr + b*stride_right_b \
                + ((k_iter + offs_k)[:, None, None] * stride_right_k) \
                + (offs_j[None, :, None] * stride_right_j) \
                + (offs_d[None, None, :] * stride_right_h)

            left = tl.load(left_ptrs, mask=(mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]), other=0.0)
            right = tl.load(right_ptrs, mask=(mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]), other=0.0)

            # now left shape: (M, K, D), right shape: (K, N, D)
            # Compute accumulator += sum_k left[:,:,d] * right[:, :, d] over k dimension
            # We do broadcasted multiply across K dimension then sum.
            # Use tl.dot per d slice:
            for d in range(0, BLOCK_D):
                # If out-of-range d, skip
                if not mask_d[d]:
                    continue
                left_d = left[:, :, d]   # (M,K)
                right_d = right[:, :, d] # (K,N)
                acc[:, :, d] += tl.dot(left_d, right_d)   # (M,N)
        # store accumulator to out
        out_ptrs = out_ptr + b*stride_out_b \
            + (offs_i[:, None, None] * stride_out_i) \
            + (offs_j[None, :, None] * stride_out_j) \
            + (offs_d[None, None, :] * stride_out_h)

        tl.store(out_ptrs, acc, mask=(mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]))
```

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    d_start,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # base indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Compute offsets vectors
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)
    d = d_start + tl.arange(0, BLOCK_D)

    # Masks for out-of-bounds
    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, N, BLOCK_K):
        # compute actual k indices
        cur_k = kk + k
        mask_k = cur_k < N

        # left load
        left_ptrs = left_ptr + (i[:, None, None] * stride_left_i) + (cur_k[None, :, None] * stride_left_k) + (d[None, None, :] * stride_left_h)
        mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptrs, mask=mask_left, other=0.0)

        # right load (K x N)
        right_ptrs = right_ptr + (cur_k[:, None, None] * stride_right_k) + (j[None, :, None] * stride_right_j) + (d[None, None, :] * stride_right_h)
        mask_right = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # compute dot products for each D slice
        for db in tl.static_range(0, BLOCK_D):
            # left_slice: (BLOCK_M, BLOCK_K)
            left_slice = left[:, :, db]
            # right_slice: (BLOCK_K, BLOCK_N)
            right_slice = right[:, :, db]
            acc[:, :, db] += tl.dot(left_slice, right_slice)
    # Store to output
    out_ptrs = out_ptr + (i[:, None, None] * stride_out_i) + (j[None, :, None] * stride_out_j) + (d[None, None, :] * stride_out_h)
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)
```

    BLOCK_N = 32
    BLOCK_K = 32

                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                BLOCK_D=BLOCK_D
            )
    # Continue with to_out_norm, gated, linear
    # out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [bs,N,N,dim]

    return out

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input on the channel dimension.
2. Linear projections (left/right) and gating linear layers are computed in PyTorch.
   The mask (if present) and the element‑wise sigmoid gates are applied there.
3. The core operation
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication per hidden channel `d`.  We implement this
   with a custom Triton kernel that processes a tile of (i,j) and a small
   block of hidden channels `d` at a time, looping over the `k` dimension.
   The kernel accumulates in FP32 and writes the result back to a tensor of
   shape [B, N, N, H].
4. The result is layer‑normed, multiplied by the output gate and finally
   projected back to `dim` with a linear layer (all in PyTorch).

The Triton kernel processes a tile of (BLOCK_M, BLOCK_N) positions and
BLOCK_D hidden channels (here 8) while loading a (BLOCK_M, BLOCK_K, BLOCK_D)
and (BLOCK_K, BLOCK_N, BLOCK_D) slice of the left/right tensors.  Memory
usage stays below the shared‑memory budget on an H100.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    out_ptr,                # *float32
    N,                      # int32
    H,                      # int32
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    d_start,                # int32 start offset in hidden dimension
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute out[..., i, j, d] += Σ_k left[..., i, k, d] * right[..., j, k, d]
    for a tile of i/j and a small block of hidden channels d.
    """

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Tile indices
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)
    d = d_start + tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # accumulator for the tile (M, N, D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over the reduction dimension K
    for kk in range(0, N, BLOCK_K):
        cur_k = kk + k
        mask_k = cur_k < N

        # ----- load left tile [M, K, D] -----
        left_ptrs = left_ptr \
            + i[:, None, None] * stride_left_i \
            + cur_k[None, :, None] * stride_left_k \
            + d[None, None, :] * stride_left_h
        mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)

        # ----- load right tile [K, N, D] (notice K first, then j) -----
        right_ptrs = right_ptr \
            + cur_k[:, None, None] * stride_right_k \
            + j[None, :, None] * stride_right_j \
            + d[None, None, :] * stride_right_h
        mask_right = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # ----- compute dot products for each hidden slice -----
        for db in tl.static_range(0, BLOCK_D):
            # (M, K) @ (K, N) → (M, N)
            acc[:, :, db] += tl.dot(left_tile[:, :, db], right_tile[:, :, db])

    # ----- store the accumulated result -----
    out_ptrs = out_ptr \
        + i[:, None, None] * stride_out_i \
        + j[None, :, None] * stride_out_j \
        + d[None, None, :] * stride_out_h
    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Entry point for the TriMul “outgoing” operation.

    Parameters
    ----------
    data : Tuple containing
        - input : torch.Tensor of shape [B, N, N, dim]
        - mask  : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights : dict of model weight tensors
        - config  : dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, dim]
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------- LayerNorm -------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------- Linear projections -------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])          # -> [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------- Optional mask -------------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------- Gates -------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------- Core TriMul (Triton) -------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)  # [B,N,N,hidden_dim]

    # Choose tile sizes (tuned for H100 shared‑memory)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8  # hidden dimension block; hidden_dim is a multiple of 8 in all tests

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N )

    # Launch the kernel per batch and per hidden‑channel block
    for b in range(B):
        left_b = left[b]   # shape [N, N, hidden_dim]
        right_b = right[b]
        out_b = out[b]

        # Strides without the batch dimension
        stride_li, stride_lk, stride_lh = left_b.stride()
        stride_rj, stride_rk, stride_rh = right_b.stride()
        stride_oi, stride_oj, stride_oh = out_b.stride()

        for d_start in range(0, hidden_dim, BLOCK_D):
            _tri_mul_kernel[grid](
                left_b,
                right_b,
                out_b,
                N,
                hidden_dim,
                stride_li, stride_lk, stride_lh,
                stride_rj, stride_rk, stride_rh,
                stride_oi, stride_oj, stride_oh,
                d_start,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                BLOCK_D=BLOCK_D,
            )
    # Ensure kernel writes are visible before subsequent ops
    torch.cuda.synchronize()

    # ------------------- Post‑processing -------------------
    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    # final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
