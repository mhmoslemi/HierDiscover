def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    # Mask handling (if mask is None or all ones)
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # Flatten for batched matmul
    B,N1,N2, H = left.shape
    assert N1 == N2, "seq_len mismatch"
    N = N1
    # Permute to [B, H, N, N] and reshape to [B*H,N,N]
    left_flat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    # Allocate output tensor
    out_flat = torch.empty_like(left_flat)
    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # Compute grid

def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_ab, stride_am, stride_ak,
    stride_bb, stride_bn, stride_bk,
    stride_cb, stride_cm, stride_cn,

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Predicates for M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    A_ptr = A + pid_b * stride_ab
    B_ptr = B + pid_b * stride_bb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        # Compute effective k range for this tile
        cur_k = k + offs_k
        mask_k = cur_k < K

        # Load A tile (BLOCK_M x BLOCK_K)
        a = tl.load(A_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    x = torch.nn.functional.layer_norm(

        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))

"""
TriMul (outgoing) – Triton‑accelerated elementwise pairwise update.

The forward pass consists of:
1. Layer‑norm over the feature dimension.
2. Linear projections (left/right) and three sigmoidal gates.
3. Optional mask multiplication (broadcasted on the feature axis).
4. Gated left/right tensors are multiplied together:

       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   This is a batch of independent matrix multiplications
   (size N×N) for each hidden channel d, i.e. a GEMM with the
   second operand transposed.  The heavy N³ work is performed by a
   custom Triton kernel that tiles the GEMM in the M, N and K
   dimensions (BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) and uses the
   program‑id axes (M‑tile, N‑tile, batch) to avoid any Python
   loops.

5. A second Layer‑norm over the hidden dimension, multiplication
   by the “out_gate”, and a final linear projection back to the
   original feature dimension.

All tensors are assumed to reside on the CUDA device already.
Only the batched GEMM is written in Triton; the surrounding
operations use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # problem size
    stride_ab, stride_am, stride_ak,   # strides for A: (batch, M, K)
    stride_bb, stride_bn, stride_bk,   # strides for B: (batch, N, K)
    stride_cb, stride_cm, stride_cn,   # strides for C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : [batch, M, K]
    B : [batch, N, K]   (the K‑dimension will be read swapped,
                         i.e. Bᵀ[n, k] = B[k, n])
    C : [batch, M, N]
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_b = tl.program_id(2)          # batch (includes hidden channel)

    # ----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptr = A + pid_b * stride_ab
    b_ptr = B + pid_b * stride_bb
    c_ptr = C + pid_b * stride_cb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension ---------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load B tile transposed:
        #   we need Bᵀ[n, k] = B[k, n]
        b = tl.load(
            b_ptr + (cur_k[:, None] * stride_bn + offs_n[None, :] * stride_bk),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate dot product over the K‑tile
        acc += tl.dot(a, b)

    # ----- write result -------------------------------------------------
    tl.store(
        c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn),
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple(input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: [B, N, N, dim]    (float32)
              - mask_tensor : [B, N, N]        (optional, broadcasted)
              - weights_dict: contains all linear / layer‑norm weights
              - config_dict : {'dim': int, 'hidden_dim': int, 'nomask': bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (over the last dimension)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gate values (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted on the feature axis)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Batched GEMM: out = left @ rightᵀ  (per hidden channel)
    #    left/right: [B, N, N, hidden_dim]  ->  [B*hidden_dim, N, N]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    assert H == hidden_dim, "hidden_dim mismatch"

    # bring hidden dim into the batch dimension
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # allocate output for the batched GEMM
    out_flat = torch.empty_like(left_flat)

    # tile sizes – tuned for NVIDIA H100 (shared‑memory friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # grid dimensions: (tiles_M, tiles_N, batch*hidden)
    num_blocks_M = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_N = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_M, num_blocks_N, B * H)

    # launch kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_ab=left_flat.stride(0),
        stride_am=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        stride_bb=right_flat.stride(0),
        stride_bn=right_flat.stride(1),
        stride_bk=right_flat.stride(2),
        stride_cb=out_flat.stride(0),
        stride_cm=out_flat.stride(1),
        stride_cn=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to the original layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Multiply by out_gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
