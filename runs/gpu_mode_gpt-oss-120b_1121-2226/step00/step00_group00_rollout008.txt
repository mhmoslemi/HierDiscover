    j_mask = j < N

"""
TriMul (outgoing) forward implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over its feature dimension (`dim`).
2. Apply two linear projections (`left_proj`, `right_proj`) to obtain
   tensors `L` and `R` of shape (B, N, N, H) where `H = hidden_dim`.
3. Optionally mask `L`/`R` with the provided pairwise mask.
4. Compute three gates (left, right, output) via a bias‑free linear
   layer followed by a sigmoid and apply the left/right gates element‑wise:
       L ← L * left_gate ,   R ← R * right_gate
5. For every batch `b` and hidden index `h` compute
       out[b, i, j, h] = Σ_k L[b, i, k, h] * R[b, j, k, h]
   This is exactly a batched matrix multiply `L_h @ R_hᵀ`.  The
   heavy O(N³) work is performed by the Triton kernel `trimul_outgoing_kernel`,
   which fuses the load‑compute‑store for each `(b, h)` pair.
6. Layer‑norm the intermediate result over the hidden dimension,
   apply the output gate, and finally a bias‑free linear projection back
   to `dim` features.

The kernel works on contiguous tensors of shape (B, N, N, H)
with standard (row‑major) strides.  It tiles the M and N dimensions
with 64‑element blocks and the reduction K dimension with 32‑element
blocks.  Masking of out‑of‑bounds indices is performed inside the kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *float32
    right_ptr,         # *float32
    out_ptr,           # *float32
    B, N, H,           # scalar int64
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled GEMM where the right operand is accessed transposed.
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_z = tl.program_id(2)          # combined batch * hidden index

    # Decode batch and hidden indices
    b = pid_z // H                     # batch id
    h = pid_z % H                      # hidden‑dim id

    # Offsets for the M (i) and N (j) dimensions
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    i_valid = i < N
    j_valid = j < N

    # Accumulator for the output block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k < N

        # -----------------------------------------------------------------
        # Pointers for left[b, i, k, h]   -> shape (BLOCK_M, BLOCK_K)
        # Pointers for right[b, j, k, h]  -> we need (k, j) order -> shape (BLOCK_K, BLOCK_N)
        # -----------------------------------------------------------------
        a_ptrs = left_ptr \
                 + b * stride_left_b \
                 + h * stride_left_h \
                 + i[:, None] * stride_left_i \
                 + k[None, :] * stride_left_k

        b_ptrs = right_ptr \
                 + b * stride_right_b \
                 + h * stride_right_h \
                 + k[:, None] * stride_right_k \
                 + j[None, :] * stride_right_i

        a = tl.load(a_ptrs,
                    mask=i_valid[:, None] & k_valid[None, :],
                    other=0.0)                     # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & j_valid[None, :],
                    other=0.0)                     # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_N)

    # Write the computed block to the output tensor
    out_ptrs = out_ptr \
               + b * stride_out_b \
               + h * stride_out_h \
               + i[:, None] * stride_out_i \
               + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_valid[:, None] & j_valid[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensor model weights
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32) – the module output.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]        # H
    B, N, _, _ = input_tensor.shape         # shape: [B, N, N, dim]

    # -----------------------------------------------------------------
    # Load model parameters
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]      # (H, dim)
    right_proj_weight = weights["right_proj.weight"]    # (H, dim)
    left_gate_weight = weights["left_gate.weight"]      # (H, dim)
    right_gate_weight = weights["right_gate.weight"]    # (H, dim)
    out_gate_weight = weights["out_gate.weight"]        # (H, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (H,)
    to_out_weight = weights["to_out.weight"]            # (dim, H)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -----------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B,N,N,H)
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3. Optional masking
    # -----------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid of bias‑free linear)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # 5. Heavy O(N³) contraction via Triton
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides (in elements, not bytes)
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Tile sizes – tuned for H100 (64×64 tiles, 32‑wide reduction)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),            # program_id(0) – rows i
        triton.cdiv(N, BLOCK_N),            # program_id(1) – cols j
        B * hidden_dim                      # program_id(2) – (batch, hidden)
    )

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_i, s_right_k, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8
    )
    # Synchronise before next PyTorch ops (optional but safe)
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gate, and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate
    out = F.linear(out, to_out_weight)    # (B,N,N,dim)
    return out
