def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"] # c_z

    # Unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    eps = 1e-5
    # Using torch.nn.functional.layer_norm: expects normalized_shape = dim
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight) # shape [B,N,N,H]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if config.get("nomask", False):
        mask_tensor = torch.ones((bs, seq_len, seq_len), dtype=x_norm.dtype, device=x_norm.device)
    else:
        mask_tensor = mask.to(x_norm.dtype)  # ensure same dtype

    mask_tensor = mask_tensor[..., None] # unsqueeze

    left = left_proj * mask_tensor
    right = right_proj * mask_tensor

    # Gating
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Reshape for batched matmul
    left_reshape = left_fp16.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_reshape = right_fp16.permute(0, 3, 1, 2).contiguous()
    # Transpose right to match matmul
    right_T = right_reshape.permute(0, 1, 3, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_reshape.shape

    A = left_reshape.view(-1, N, N)
    B_mat = right_T.view(-1, N, N)

    batch_hidden = A.shape[0]

    out_mat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=A.device)  # result in fp32

    # Grid launches
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch_hidden, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B_mat,
        C_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_ab=A.stride(0),
        stride_am=A.stride(1),
        stride_ak=A.stride(2),
        stride_bb=B_mat.stride(0),
        stride_bk=B_mat.stride(1),
        stride_bn=B_mat.stride(2),
        stride_cb=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    out = out_mat.view(bs, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # to_out_norm (layernorm on hidden_dim)
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # apply out_gate

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor `x` (shape [B,N,N,C]).
2. Linear projections to hidden dimension `H` (no bias) → `left`, `right`.
3. Apply optional pairwise mask (broadcast over the hidden dim).
4. Compute three sigmoid gates from the normalized tensor.
5. Multiply the projected tensors with their respective gates.
6. **Heavy part:** compute the outer‑product reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix‑multiplication
        out = left @ rightᵀ   (per hidden channel).
   This is performed by a custom Triton kernel that processes
   all `B*H` matrices in one launch, using FP16 loads and FP32 accumulation.
7. Layer‑normalize the result over the hidden dimension, multiply by the
   `out_gate`, and apply the final linear projection back to `C`.
8. Return the tensor of shape [B,N,N,C] (float32).

The kernel fuses the O(N³) reduction and is the main performance hotspot.
All other lightweight operations are carried out with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   where
#   A : [B*H, M, K]   (FP16)
#   B : [B*H, K, N]   (FP16)
#   C : [B*H, M, N]   (FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ab, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, row(K), col(N))
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)                # batch*hidden idx
    pid_m = tl.program_id(1)                    # block row
    pid_n = tl.program_id(2)                    # block col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for A and B
    a_ptrs = (
        A_ptr
        + pid_batch * stride_ab
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_batch * stride_bb
        + offs_k[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K

        # Load tiles with proper masking for the last (partial) tile
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & ((offs_k + cur_k) < K),
            other=0.0,
            dtype=tl.float16,
        )
        b = tl.load(
            b_ptrs,
            mask=((offs_k + cur_k) < K) & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Cast to FP32 for accumulation
        a_f32 = a.to(tl.float32)
        b_f32 = b.to(tl.float32)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a_f32, b_f32)

        # Move pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool or float) – can be ignored if config["nomask"] is True.
        - weights      : dict of torch.Tensor containing all linear / layer‑norm parameters.
        - config       : dict with keys:
            * "dim"        – input/output channel dimension C
            * "hidden_dim" – hidden dimension H
            * "nomask"     – bool indicating whether mask should be applied

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projection weights (no bias)
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    # Gate weights
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Output LayerNorm and final projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask is float and broadcastable
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy reduction: out = Σ_k left[...,i,k,:] * right[...,j,k,:]
    #    Implemented as a batched matmul C = A @ Bᵀ with a Triton kernel.
    # ------------------------------------------------------------------
    # Convert to FP16 to reduce memory bandwidth
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to [B, H, N, N]  (batch, hidden, row, col)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    # Transpose the second N‑axis of right for the matmul (K,N)
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # Flatten batch‑hidden dimension
    A = left_perm.view(batch_hidden, N, N)          # [B*H, M, K]
    B_mat = right_T.view(batch_hidden, N, N)        # [B*H, K, N]

    # Allocate output
    out_mat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=x.device)

    # Kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B_mat,
        C_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_ab=A.stride(0),
        stride_am=A.stride(1),
        stride_ak=A.stride(2),
        stride_bb=B_mat.stride(0),
        stride_bk=B_mat.stride(1),
        stride_bn=B_mat.stride(2),
        stride_cb=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                     # reasonable choice for H100
    )

    # Reshape back to [B, N, N, H]
    out_hidden = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating by out_gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    out_gated = out_norm * out_gate
    final_out = F.linear(out_gated, to_out_w)       # [B, N, N, C]

    return final_out
