def custom_kernel(data):
    # Unpack
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # Cast to float32 in case not
    x = x.to(torch.float32)
    if mask is None:
        mask = torch.ones(bs, N, N, device=x.device, dtype=x.dtype)
    # LayerNorm
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    # Apply mask
    mask_exp = mask.unsqueeze(-1)  # shape (B,N,N,1)
    # Combine
    left = left_proj * mask_exp * left_gate
    right = right_proj * mask_exp * right_gate
    # Convert to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # Permute for batch*hidden
    left_perm = left.permute(0,3,1,2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0,3,2,1).contiguous() # (B, H, N, N) (transposed)
    # Flatten batch and hidden dims
    M = bs * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)
    # Output buffer
    out_half = torch.empty_like(left_flat)  # shape (M,N,N), dtype half
    # Get strides
    a_batch, a_i, a_k = left_flat.stride()
    b_batch, b_k, b_j = right_flat.stride()
    c_batch, c_i, c_j = out_half.stride()
    # launch kernel
    import triton
    import triton.language as tl
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), M)
    batched_gemm_kernel[grid](
        left_flat, right_flat, out_half,
        M, N, N,
        a_batch, a_i, a_k,
        b_batch, b_k, b_j,
        c_batch, c_i, c_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_half.view(bs, hidden_dim, N, N).permute(0,2,3,1).contiguous() # (B,N,N,H)
    # Output norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    # Apply out gate
    out = out * out_gate
    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

Now define kernel:

```
@triton.jit
def batched_gemm_kernel(A, B, C,
                        BATCH, N, K,
                        stride_Ab, stride_Ai, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bj,
                        stride_Cb, stride_Ci, stride_Cj,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)  # row block in N dimension
    pid_n = tl.program_id(1)  # col block in N dimension
    pid_b = tl.program_id(2)  # batch index (M dimension)
    # compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute pointers base for this batch
    a_ptr_base = A + pid_b * stride_Ab
    b_ptr_base = B + pid_b * stride_Bb
    c_ptr_base = C + pid_b * stride_Cb
    # initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for valid k
        mask_k = offs_k < K
        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr_base + offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < N) & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)
        # load B: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr_base + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & (offs_j[None, :] < N), other=0.0)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
    # store
    c_ptrs = c_ptr_base + offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj
    mask_i = offs_i < N
    mask_j = offs_j < N
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(A, B, C,
                        BATCH, N, K,
                        stride_Ab, stride_Ai, stride_Ak,
                        stride_Bb, stride_Bk, stride_Bj,
                        stride_Cb, stride_Ci, stride_Cj,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)   # row tile
    pid_j = tl.program_id(1)   # col tile
    pid_b = tl.program_id(2)   # batch (M)

    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    b_ptr = B + pid_b * stride_Bb
    c_ptr = C + pid_b * stride_Cb

"""
custom_kernel
==============

Implements the *outgoing* Triangle‑Multiplicative update (TriMul) described in
AlphaFold‑3.  The heavy O(N³·C_hidden) part (the einsum
`out[i,j,d] = Σ_k left[i,k,d]·right[j,k,d]`) is accelerated with a custom
Triton GEMM‑style kernel.  All other tensor operations (LayerNorm, Linear,
sigmoid, masking) are performed with PyTorch for simplicity and numerical
stability.

Algorithm
---------
1. **Layer‑norm** over the last dimension of the input tensor.
2. **Linear projections** (`left_proj`, `right_proj`) and **gates**
   (`left_gate`, `right_gate`, `out_gate`) from the normalized tensor.
3. Apply the optional mask and the gates:
   `left = left_proj * mask * left_gate`,
   `right = right_proj * mask * right_gate`.
4. Convert ``left`` and ``right`` to **float16**, permute to shape
   ``(B, H, N, N)`` and flatten the batch + hidden dimensions to obtain
   ``A`` and ``B`` of shape ``(M, N, N)`` with ``M = B·H``.
5. Run the Triton kernel `batched_gemm_kernel` which computes for each
   ``m`` the matrix product ``A[m] @ B[m].T`` (the required einsum) and
   writes the result in half‑precision.
6. Reshape the result back to ``(B, N, N, H)`` (float32), apply the
   output ``LayerNorm`` and the ``out_gate``.
7. Final linear projection back to the original feature dimension ``dim``.

The Triton kernel tiles the matrices with blocks (BM×BN) and accumulates
over K‑tiles, loading data in FP16 and accumulating in FP32 to keep accuracy
while benefiting from the reduced memory bandwidth of half‑precision.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) for many matrices at once
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A,                     # ptr to left operand, shape (M, N, K)
    B,                     # ptr to right operand, shape (M, K, N)  (already transposed)
    C,                     # ptr to output, shape (M, N, N)
    M, N, K,               # matrix dimensions (M = batch*hidden, N = seq_len, K = seq_len)
    stride_Ab, stride_Ai, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bj,   # strides for B
    stride_Cb, stride_Ci, stride_Cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids (block indices)
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_b = tl.program_id(2)          # which matrix in the batch (0 … M‑1)

    # offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this matrix
    a_ptr = A + pid_b * stride_Ab
    b_ptr = B + pid_b * stride_Bb
    c_ptr = C + pid_b * stride_Cb

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # ----- load A[i, k] -------------------------------------------------
        a_mask = (offs_i[:, None] < N) & (offs_k[None, :] < K)
        a_ptrs = a_ptr + offs_i[:, None] * stride_Ai + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)           # fp16
        a = a.to(tl.float32)                                 # promote

        # ----- load B[k, j] -------------------------------------------------
        b_mask = (offs_k[:, None] < K) & (offs_j[None, :] < N)
        b_ptrs = b_ptr + offs_k[:, None] * stride_Bk + offs_j[None, :] * stride_Bj
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)           # fp16
        b = b.to(tl.float32)                                 # promote

        # ----- rank‑K update ------------------------------------------------
        acc += tl.dot(a, b)                                   # (BM,BN)

    # ----- write the result -------------------------------------------------
    c_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    c_ptrs = c_ptr + offs_i[:, None] * stride_Ci + offs_j[None, :] * stride_Cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


# --------------------------------------------------------------
# Entry point expected by the grader
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              input_tensor shape [B, N, N, dim]
              mask shape [B, N, N] (or None)
              weights dict with all linear/ln parameters
              config dict containing "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gating values (sigmoid) – also linear without bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # Optional mask (use ones if mask omitted or nomask flag is True)
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask = torch.ones(B, N, N, device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)  # [B,N,N,1]

    # ------------------------------------------------------------------
    # Apply mask and gates
    # ------------------------------------------------------------------
    left = left_proj * mask * left_gate    # [B,N,N,H], float32
    right = right_proj * mask * right_gate

    # ------------------------------------------------------------------
    # Convert to half‑precision and prepare for the batched GEMM
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B, H, N, N) – rows = i, columns = k for left;
    # rows = k, columns = j for right (already transposed)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # [B,H,N,N] (k,j)

    # Collapse batch and hidden dimensions: M = B * H
    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)    # [M,N,N]
    right_flat = right_perm.view(M, N, N)  # [M,N,N]

    # output buffer (half‑precision)
    out_half = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # Strides (in elements, not bytes) for the three tensors
    # ------------------------------------------------------------------
    stride_Ab, stride_Ai, stride_Ak = left_flat.stride()          # (N*N, N, 1)
    stride_Bb, stride_Bk, stride_Bj = right_flat.stride()        # (N*N, N, 1)
    stride_Cb, stride_Ci, stride_Cj = out_half.stride()          # (N*N, N, 1)

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        M,                              # one program per matrix in the batch
    )
    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_half,
        M,
        N,
        N,
        stride_Ab,
        stride_Ai,
        stride_Ak,
        stride_Bb,
        stride_Bk,
        stride_Bj,
        stride_Cb,
        stride_Ci,
        stride_Cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, H) and convert to FP32 for the remaining ops
    # ------------------------------------------------------------------
    out = out_half.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # Output LayerNorm over hidden dimension, apply out_gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
