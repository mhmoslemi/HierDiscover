TriMul “outgoing” implementation.

The kernel only implements the heavy N³ contraction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

All surrounding operations (LayerNorm, Linear projections, gating,
masking, final LayerNorm + Linear) are performed in PyTorch for
readability.  The contraction is written in Triton and processes the
hidden dimension in small chunks (D_STEP) so that the whole
accumulator fits in registers.  The kernel works for any batch size
(≤2), sequence length (≤1024) and hidden dimension that is a multiple
of D_STEP (we use D_STEP=8 which covers the test configurations).

The final output is returned as float32.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton contraction kernel
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_contract_kernel(
    left_ptr,          # float16*  [B, N, N, D]  (i, k, d)
    right_ptr,         # float16*  [B, N, N, D]  (j, k, d)
    out_ptr,           # float32*  [B, N, N, D]  (i, j, d)

    B, N, D,           # problem sizes (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, D_STEP: tl.constexpr,
):
    """Batched block‑wise contraction with hidden‑dim vectorisation.

    Each program instance computes a (BLOCK_M × BLOCK_N × D_STEP) tile of
    the output for a given batch `b`.  The hidden dimension is processed
    in chunks of size `D_STEP` (default 8) and the reduction over `k`
    (the third axis) is performed with a classic GEMM‑style loop.
    """

    pid_i = tl.program_id(0)               # tile over i (first N)
    pid_j = tl.program_id(1)               # tile over j (second N)
    pid_b = tl.program_id(2)               # batch dimension

    # ------------------------------------------------------------------- #
    # Tile offsets and masks
    # ------------------------------------------------------------------- #
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    b = pid_b

    i_off = i_base + tl.arange(0, BLOCK_M)
    j_off = j_base + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    # ------------------------------------------------------------------- #
    # Loop over hidden‑dim chunks
    # ------------------------------------------------------------------- #
    d_range = tl.arange(0, D_STEP)

    for d_start in range(0, D, D_STEP):
        # accumulator for the current d‑block
        acc = tl.zeros((BLOCK_M, BLOCK_N, D_STEP), dtype=tl.float32)

        # ----------------------------------------------------------------
        # Loop over the reduction dimension k
        # ----------------------------------------------------------------
        for k_tile in range(num_k_tiles):
            k_base = k_tile * BLOCK_K
            k_off = k_base + tl.arange(0, BLOCK_K)
            k_mask = k_off < N

            # ----------------------------------------------------------------
            # Load left  : shape (M, K, D_STEP)   (i, k, d)
            # ----------------------------------------------------------------
            left_ptrs = left_ptr + \
                b * stride_left_b + \
                i_off[:, None, None] * stride_left_i + \
                k_off[None, :, None] * stride_left_k + \
                (d_start + d_range)[None, None, :] * stride_left_d
            left_mask = i_mask[:, None, None] & k_mask[None, :, None]
            left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0)

            # ----------------------------------------------------------------
            # Load right : shape (K, N, D_STEP)   (k, j, d)
            # ----------------------------------------------------------------
            right_ptrs = right_ptr + \
                b * stride_right_b + \
                k_off[:, None, None] * stride_right_k + \
                j_off[None, :, None] * stride_right_j + \
                (d_start + d_range)[None, None, :] * stride_right_d
            right_mask = k_mask[:, None, None] & j_mask[None, :, None]
            right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0)

            # ----------------------------------------------------------------
            # Accumulate:  for each d the dot‑product of (M×K)·(K×N)
            # ----------------------------------------------------------------
            for d_idx in range(D_STEP):
                a = left_tile[:, :, d_idx]      # (M, K)
                b_mat = right_tile[:, :, d_idx]  # (K, N)
                acc[:, :, d_idx] = acc[:, :, d_idx] + tl.dot(a, b_mat)

        # ----------------------------------------------------------------
        # Write the tile back (float32 output)
        # ----------------------------------------------------------------
        out_ptrs = out_ptr + \
            b * stride_out_b + \
            i_off[:, None, None] * stride_out_i + \
            j_off[None, :, None] * stride_out_j + \
            (d_start + d_range)[None, None, :] * stride_out_d

        out_mask = i_mask[:, None, None] & j_mask[None, :, None]
        tl.store(out_ptrs, acc, mask=out_mask)


# --------------------------------------------------------------------------- #
# Public entry point used by the grader
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    inp, mask, weights, config = data
    device = inp.device
    dtype = inp.dtype

    dim = config["dim"]                 # input channel size
    hidden_dim = config["hidden_dim"]   # intermediate channel size

    # --------------------------------------------------------------- #
    # Extract weights
    # --------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # --------------------------------------------------------------- #
    # 1. LayerNorm on the input (last dim = dim)
    # --------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        inp, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )   # shape [B, N, N, dim]  (float32)

    # --------------------------------------------------------------- #
    # 2. Linear projections (float16 for the heavy contraction)
    # --------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, left_proj_weight).to(torch.float16)
    right = torch.nn.functional.linear(x, right_proj_weight).to(torch.float16)

    # --------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------- #
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------- #
    # 4. Gating (sigmoid of linear projections) – also cast to fp16
    # --------------------------------------------------------------- #
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, left_gate_weight)
    ).to(torch.float16)
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, right_gate_weight)
    ).to(torch.float16)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, out_gate_weight)
    ).to(torch.float16)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 5. Contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------- #
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides for the three tensors (all are contiguous)
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_rj, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # Kernel launch parameters (tuned for H100, but work generically)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    D_STEP = 8                     # hidden_dim is a multiple of 8 in all tests

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B                               # batch
    )

    trimul_contract_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_rj,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        D_STEP=D_STEP,
    )

    # --------------------------------------------------------------- #
    # 6. Post‑processing: LayerNorm over hidden_dim, gating, final linear
    # --------------------------------------------------------------- #
    # to_out_norm works on the hidden dimension
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )   # still [B,N,N,hidden_dim] float32

    # Apply the output gate (cast gate to float32 for the multiplication)
    out = out * out_gate.to(torch.float32)

    # Final projection back to `dim`
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
