"""
custom_kernel: Triton‑accelerated forward pass of the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (shape [B,N,N,C]).
2. Linear projections → left, right (shape [..., H]), plus three sigmoid gates.
3. Optional binary mask is applied to left/right before gating.
4. Fuse the heavy N³ reduction:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched mat‑mul of the (i,k) slice with the transpose of the
   (j,k) slice for every hidden dimension *h*.  Implemented with a
   Triton kernel that performs a blocked GEMM for each (batch, head) pair.
5. Layer‑norm the intermediate result (hidden dim H), multiply by the out‑gate,
   and finally a linear projection back to the original channel dimension C.
6. All kernels run in FP16 for speed; accumulation is in FP32.  The final
   output is returned in FP32.

The kernel processes one (batch, head) matrix at a time, using a 2‑D tile
grid (BLOCK_M × BLOCK_N) and a 1‑D tile over the reduction dimension K.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core tensor multiplication (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # left  tensor  [B,N,N,H] (fp16)
    right_ptr,                    # right tensor  [B,N,N,H] (fp16)
    out_ptr,                      # output tensor [B,N,N,H] (fp16)
    B, N, H,                      # dimensions
    # strides of left: (b,i,k,h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides of right: (b,j,k,h)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides of out: (b,i,j,h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 2‑D tile identifiers
    pid_i = tl.program_id(0)                 # tile row   (i dimension)
    pid_j = tl.program_id(1)                 # tile col   (j dimension)
    pid_bn = tl.program_id(2)                # flattened (batch, head) index

    # Recover batch index b and hidden‑dim index h
    b = pid_bn // H
    h = pid_bn % H

    # Offsets inside the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Range masks (handle non‑divisible N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the “pairwise” axis)
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_offset = k_tile * BLOCK_K
        k = offs_k + k_offset
        mask_k = k < N

        # ------------------ load A = left[b,i,k,h] ------------------
        a_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # ------------------ load Bᵀ = right[b,j,k,h]ᵀ ------------------
        # We want Bᵀ[k,j] = right[b,j,k,h]; thus k dimension is first.
        b_ptrs = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # MatMul for this K‑tile (FP16 × FP16 → FP32)
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N)

    # -------------- write the result ------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights, config)
    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # C
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to get strides (int) for a contiguous tensor
    # ------------------------------------------------------------------
    def get_strides(t):
        return tuple(t.stride(i) for i in range(t.dim()))

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the last dimension (C)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.layer_norm expects the normalized shape as a tuple
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2. Linear projections and gates (all float32 for stability)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [..., H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3. Optional mask (binary mask is 0/1 float)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape [B,N,N] → [B,N,N,1]
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy reduction and allocate output buffer
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6. Triton kernel launch parameters
    # ------------------------------------------------------------------
    # Tile sizes – tuned for H100 (work‑group size 64² works well)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,               # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,               # j‑tiles
        B * hidden_dim                              # (batch, head) dimension
    )

    # Strides for left / right / out (in elements, not bytes)
    stride_left = get_strides(left_fp16)    # (b,i,k,h)
    stride_right = get_strides(right_fp16)  # (b,j,k,h)
    stride_out = get_strides(out_fp16)      # (b,i,j,h)

    # Launch the kernel
    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Convert back to FP32 for the remaining operations
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm over hidden_dim, apply out‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)

    # Apply the output gate (broadcasted over the last dim)
    out = out * out_gate

    # Final projection back to original channel dimension C
    to_out_weight = weights["to_out.weight"]          # shape (C, H)
    out = F.linear(out, to_out_weight)               # bias is None by design

    # The result should be float32 with shape [B, N, N, C]
    return out
