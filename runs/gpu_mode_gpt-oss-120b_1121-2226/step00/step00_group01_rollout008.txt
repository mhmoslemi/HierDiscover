def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config contains dim, hidden_dim
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm on dim
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Projections
    left_raw = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_raw = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Mask handling
    nomask = config.get("nomask", False)
    if not nomask:
        # ensure mask dtype matches
        mask_f = mask.to(dtype)
        # mask shape (bs,N,N) -> broadcast
        # Use Triton kernel to fuse mask with gates
        left, right = _apply_mask_gate_triton(left_raw, left_gate, right_raw, right_gate, mask_f)
    else:
        left = left_raw * left_gate
        right = right_raw * right_gate

    # Compute out via batched matmul
    # transpose right for correct contraction
    left_perm = left.permute(0, 3, 1, 2)  # B, H, N, N
    right_perm = right.permute(0, 3, 2, 1)  # B, H, N, N
    B, H, N2, _ = left_perm.shape
    left_mat = left_perm.reshape(B * H, N, N)
    right_mat = right_perm.reshape(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # B, N, N, H

    # Output norm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                              weight=weights["to_out_norm.weight"],
                                              bias=weights["to_out_norm.bias"],
                                              eps=1e-5)
    # Apply out_gate
    out_scaled = out_norm * out_gate

    # Final linear
    out_final = torch.nn.functional.linear(out_scaled, weights["to_out.weight"])

    return out_final

"""
custom_kernel – Triangular Multiplicative Update (outgoing) in Triton + PyTorch.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Project the normalized tensor to the hidden dimension H with four linear layers:
   * left  = Linear(x_norm, left_proj.weight)          → (B,N,N,H)
   * right = Linear(x_norm, right_proj.weight)         → (B,N,N,H)
   * left_gate  = sigmoid(Linear(x_norm, left_gate.weight))   → (B,N,N,H)
   * right_gate = sigmoid(Linear(x_norm, right_gate.weight))  → (B,N,N,H)
   * out_gate   = sigmoid(Linear(x_norm, out_gate.weight))    → (B,N,N,H)
3. Fuse mask, left‑gate and right‑gate multiplication in a Triton
   kernel (if a mask is present). The kernel computes
        left′  = left  * left_gate  * mask
        right′ = right * right_gate * mask
   element‑wise for the whole [B,N,N,H] tensor.
   When no mask is supplied the fusion step is omitted and the
   element‑wise product is performed by PyTorch.
4. Contract the two tensors over the shared sequence axis *k*:
        out[b,i,j,h] = Σ_k left′[b,i,k,h] * right′[b,j,k,h]
   This is a batched matrix multiplication with shape
   (B·H, N, N) @ (B·H, N, N).  It is carried out with
   `torch.bmm` (highly tuned cuBLAS on H100).
5. Apply a second LayerNorm over the hidden dimension,
   multiply by the output gate and project back to the original
   channel dimension C with a final linear layer.
6. Return the tensor of shape [B,N,N,C] (float32).

Only the mask‑gate fusion is written in Triton, while the
dominant O(N³·H) contraction uses cuBLAS for maximal throughput.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# --------------------------------------------------------------------------- #
# Triton kernel: fuse mask, left gate and right gate
# --------------------------------------------------------------------------- #
@triton.jit
def _mask_gate_fuse_kernel(
    # input tensors
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr, mask_ptr,
    # output tensors
    out_left_ptr, out_right_ptr,
    # strides (B,N,N,H) layout
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    # tensor extents
    B, N, H,
    # compile‑time block size
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)                      # [BLOCK]

    total = B * N * N * H
    mask_valid = offs < total

    # ---------------------------------------------------------------------
    # Decode linear index into (b,i,j,h)
    # ---------------------------------------------------------------------
    b, r = tl.divmod(offs, N * N * H)       # b ∈ [0,B)
    i, r2 = tl.divmod(r, N * H)            # i ∈ [0,N)
    j, h = tl.divmod(r2, H)                # j ∈ [0,N), h ∈ [0,H)

    # ---------------------------------------------------------------------
    # Compute pointers for all tensors
    # ---------------------------------------------------------------------
    off_left  = b * stride_left_b  + i * stride_left_i  + j * stride_left_j  + h * stride_left_h
    off_lgate = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    off_right = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    off_rgate = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h
    off_mask  = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # ---------------------------------------------------------------------
    # Load, fuse and store
    # ---------------------------------------------------------------------
    left      = tl.load(left_ptr   + off_left,  mask=mask_valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + off_lgate, mask=mask_valid, other=0.0)
    right      = tl.load(right_ptr   + off_right,  mask=mask_valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + off_rgate, mask=mask_valid, other=0.0)
    mask_val   = tl.load(mask_ptr + off_mask, mask=mask_valid, other=1.0)   # default 1 → “no mask”

    out_left  = left      * left_gate  * mask_val
    out_right = right     * right_gate * mask_val

    tl.store(out_left_ptr  + off_left,  out_left,  mask=mask_valid)
    tl.store(out_right_ptr + off_right, out_right, mask=mask_valid)


def _apply_mask_gate_triton(left, left_gate, right, right_gate, mask):
    """
    Fuse element‑wise mask, left‑gate and right‑gate using Triton.
    Returns gated left and right tensors of the same shape as the inputs.
    """
    B, N, _, H = left.shape
    # Allocate output buffers
    out_left  = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # Strides (int64)
    s_lb, s_li, s_lj, s_lh = left.stride()
    s_lgb, s_lgi, s_lgj, s_lgh = left_gate.stride()
    s_rb, s_ri, s_rj, s_rh = right.stride()
    s_rgb, s_rgi, s_rgj, s_rgh = right_gate.stride()
    s_mb, s_mi, s_mj = mask.stride()

    total = B * N * N * H
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK']),)

    _mask_gate_fuse_kernel[grid](
        left, left_gate, right, right_gate, mask,
        out_left, out_right,
        # strides
        s_lb, s_li, s_lj, s_lh,
        s_lgb, s_lgi, s_lgj, s_lgh,
        s_rb, s_ri, s_rj, s_rh,
        s_rgb, s_rgi, s_rgj, s_rgh,
        s_mb, s_mi, s_mj,
        # extents
        B, N, H,
        BLOCK=BLOCK,
    )
    return out_left, out_right


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, C]
        mask         : torch.Tensor [B, N, N]  (may be None)
        weights      : dict of torch.Tensor containing model weights
        config       : dict with keys:
                       - "dim"          : input channel dimension C
                       - "hidden_dim"   : hidden dimension H
                       - "nomask" (bool) – if True the mask is ignored
    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim        = config["dim"]          # C
    hidden_dim = config["hidden_dim"]   # H
    nomask     = config.get("nomask", False)

    # --------------------------------------------------------------- #
    # 1) Layer‑Norm over the channel dimension C
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------- #
    # 2) Linear projections + gate values (no bias)
    # --------------------------------------------------------------- #
    left_raw      = F.linear(x, weights["left_proj.weight"])      # (B,N,N,H)
    right_raw     = F.linear(x, weights["right_proj.weight"])     # (B,N,N,H)

    left_gate_raw = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate_raw= F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate      = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------- #
    # 3) Fuse mask, left‑gate and right‑gate (Triton kernel)
    # --------------------------------------------------------------- #
    if not nomask and mask is not None:
        # make mask float and broadcast‑compatible
        mask_f = mask.to(x.dtype)          # [B,N,N]
        left, right = _apply_mask_gate_triton(
            left_raw, left_gate_raw,
            right_raw, right_gate_raw,
            mask_f,
        )
    else:
        # No mask → simple element‑wise multiplication
        left  = left_raw * left_gate_raw
        right = right_raw * right_gate_raw

    # --------------------------------------------------------------- #
    # 4) Contract over the shared sequence axis (k) → O(N³·H)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched mat‑mul: (B·H, N, N) @ (B·H, N, N)
    # --------------------------------------------------------------- #
    # transpose right to have the k‑dimension as the inner dimension
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.reshape(B * H, N, N)   # (B·H, N, N)
    right_mat = right_perm.reshape(B * H, N, N)  # (B·H, N, N)

    contracted = torch.bmm(left_mat, right_mat)  # (B·H, N, N)

    # reshape back to [B, N, N, H]
    out = contracted.view(B, H, N, N).permute(0, 2, 3, 1)   # (B, N, N, H)

    # --------------------------------------------------------------- #
    # 5) Second Layer‑Norm + output gate
    # --------------------------------------------------------------- #
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_gated = out_norm * out_gate                      # broadcast over H

    # --------------------------------------------------------------- #
    # 6) Final linear back to original channel dimension C
    # --------------------------------------------------------------- #
    final_out = F.linear(out_gated, weights["to_out.weight"])  # (B,N,N,C)

    return final_out
