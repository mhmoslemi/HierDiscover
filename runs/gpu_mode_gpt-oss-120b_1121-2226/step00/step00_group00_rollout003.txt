def trimul_gram_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0) # combined batch*hidden
    pidi = tl.program_id(1)
    pidj = tl.program_id(2)

    b = pid0 // hidden_dim
    d = pid0 % hidden_dim

    offs_i = pidi * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pidj * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bound i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left sub block: [M,K]
        left_ptrs = left_ptr + b*stride_left_batch + d*stride_left_d + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # load right sub block as [K,N] => right[j,k] transposed as required: [K,N] (k rows)
        # Right sub-block for rows = k, columns = j

    # store output

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # check nomask (if mask is None maybe)
    # Possibly config also includes 'nomask' boolean. We'll handle.

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm
    # using weight and bias as 1D tensors of shape [dim]
    # compute mean and var across dim dimension (last)
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # left, right shape [B,N,N,hidden_dim]

    # Mask handling
    # mask shape [B,N,N] may be bool or 0/1 float
    if mask is not None:
        mask = mask.unsqueeze(-1).to(dtype=x_norm.dtype)
        left = left * mask
        right = right * mask
    # else if nominal no mask we skip

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Optionally cast to fp16 for kernel to improve speed
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare output tensor
    B, N, _, _ = left.shape

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

@triton.jit
def trimul_gram_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for each batch b and hidden dimension d, the Gram matrix:
        out[b, i, j, d] = sum_{k} left[b, i, k, d] * right[b, j, k, d]
    This is equivalent to a batched matrix multiplication of left with right^T.
    The kernel tiles the i (rows) and j (cols) dimensions while looping over k.
    """
    pid0 = tl.program_id(0)  # combined batch*channel
    pid_i = tl.program_id(1)  # i-tile
    pid_j = tl.program_id(2)  # j-tile

    b = pid0 // D
    d = pid0 % D

    # Offsets in i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left sub‑matrix: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (left_ptr
                     + b * stride_l_batch
                     + d * stride_l_d
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right sub‑matrix transposed so that it has shape (BLOCK_K, BLOCK_N)
        right_ptrs = (right_ptr
                      + b * stride_r_batch
                      + d * stride_r_d
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_i)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate GEMM tile
        acc += tl.dot(left, right)

    # Write back the tile
    out_ptrs = (out_ptr
                + b * stride_o_batch
                + d * stride_o_d
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- 1. LayerNorm over the channel dimension -----
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ----- 2. Linear projections (no bias) -----
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- 3. Optional mask -----
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- 4. Gating -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 to exploit tensor‑core acceleration
    left_fp16 = left.contiguous().to(torch.float16)
    right_fp16 = right.contiguous().to(torch.float16)

    out_fp32 = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # Strides (in elements, not bytes)
    stride_l = left_fp16.stride()   # (B_stride, i_stride, k_stride, d_stride)
    stride_r = right_fp16.stride()
    stride_o = out_fp32.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # combined batch & channel dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N        # j‑tiles
    )

    trimul_gram_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- 6. Post‑processing: layer‑norm, gating, final linear -----
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate  # broadcast over hidden_dim

    out = F.linear(out, weights["to_out.weight"])
    return out

Triton‑accelerated outgoing TriMul.

The heavy part of the module is the N³ einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is simply a batched Gram matrix (left @ rightᵀ) for each hidden
channel *d*.  The kernel tiles the i‑ and j‑axes (rows/cols) while
looping over the k‑axis and accumulates in FP32.  All other
operations (LayerNorm, linear projections, gating, final projection)
are performed with PyTorch for simplicity.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def trimul_gram_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for each batch `b` and hidden channel `d`:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    This is equivalent to a matrix multiplication
        out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
    The kernel tiles the i‑ (rows) and j‑ (cols) axes and iterates over k.
    """
    pid0 = tl.program_id(0)          # combined batch * channel
    pid_i = tl.program_id(1)          # i‑tile
    pid_j = tl.program_id(2)          # j‑tile

    b = pid0 // D
    d = pid0 % D

    # i‑ and j‑coordinates for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left sub‑matrix: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (left_ptr
                     + b * stride_l_batch
                     + d * stride_l_d
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k)

                       mask=mask_i[:, None] & mask_k[None, :],

        # right sub‑matrix transposed to shape (BLOCK_K, BLOCK_N)
        # (k dimension first, j second)
        right_ptrs = (right_ptr

                      + d * stride_r_d
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_i)

                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

                + d * stride_o_d
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)

    right_fp16 = right.contiguous().to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out_fp32 = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # strides (in element units, not bytes)
    stride_l = left_fp16.stride()          # (batch, i, k, d)
    stride_r = right_fp16.stride()
    stride_o = out_fp32.stride()           # (batch, i, j, d)

    # tile sizes – values chosen to fit well on H100 tensor‑core units
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # one program per (batch, channel)
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑tiles
    )

    trimul_gram_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])

    return out
