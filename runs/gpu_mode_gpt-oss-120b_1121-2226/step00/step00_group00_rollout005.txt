@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    b = pid_batch // H
    h = pid_batch % H

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointer offsets
        left_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        right_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + offs_j[:, None] * stride_r_j + offs_k[None, :] * stride_r_k

        # Load tiles
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # cast
        left_f32 = tl.cast(left_tile, tl.float32)
        right_f32 = tl.cast(right_tile, tl.float32)

        # dot product
        acc += tl.dot(left_f32, tl.trans(right_f32))

    # Store result
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)

    b = pid_batch // H
    h = pid_batch % H

    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate over reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: (i,k)
        left_ptrs = left_ptr \
            + b * stride_l_b \
            + h * stride_l_h \
            + offs_i[:, None] * stride_l_i \
            + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = tl.cast(left, tl.float32)

        # load right tile: (j,k)
        right_ptrs = right_ptr \
            + b * stride_r_b \
            + h * stride_r_h \
            + offs_j[:, None] * stride_r_j \
            + offs_k[None, :] * stride_r_k
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        right = tl.cast(right, tl.float32)

        # dot product over k: (i,k) * (j,k)^T -> (i,j)
        acc += tl.dot(left, tl.trans(right))

    # store result
    out_ptrs = out_ptr \
        + b * stride_o_b \
        + h * stride_o_h \
        + offs_i[:, None] * stride_o_i \
        + offs_j[None, :] * stride_o_j

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operation.
    The heavy pairwise contraction ``out[b,i,j,:] = sum_k left[b,i,k,:] * right[b,j,k,:]``
    is performed by a custom Triton kernel; all other linear, gating and
    normalization steps are done with PyTorch primitives for simplicity.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --- LayerNorm on the input feature (dim) ---
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # --- Linear projections (no bias) ---
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # --- Optional mask ---
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --- Gating ---
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # Triton kernel: compute pairwise contraction across the third axis `k`
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
    # Cast to half for the kernel (memory‑bandwidth friendly)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer (float32 accumulator)
    out = torch.empty_like(left, dtype=torch.float32)

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    # for the right tensor the `j` dimension uses the same stride as `i`
    stride_r_j = stride_r_i
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # block sizes – chosen to fit well on H100
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        B * H,
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # Post‑processing: LayerNorm on hidden dimension, gating and final linear
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)
    return out

"""
TriMul (outgoing) forward pass with a Triton‐accelerated pairwise contraction.

The module performs
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where
    left  = (LayerNorm(x)·mask)  @ left_proj_weight   (followed by left_gate)
    right = (LayerNorm(x)·mask)  @ right_proj_weight  (followed by right_gate)

The heavy reduction over the third axis *k* is computed by a custom Triton
kernel that evaluates a batched matrix multiplication for every hidden
dimension `d`.  All remaining linear, gating and LayerNorm steps are handled
by PyTorch for clarity.  The kernel works in FP16 for bandwidth‑limited
loads while accumulating in FP32 to preserve numeric quality.

The implementation supports optional masking (controlled by `config["nomask"]`)
and works for all test shapes listed in the problem statement.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM   left[b,i,k,d]  ×  right[b,j,k,d]^T
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *float16*  (B, N, N, H)   – indexed as (b,i,k,h)
    right_ptr,              # *float16*  (B, N, N, H)   – indexed as (b,j,k,h)
    out_ptr,                # *float32*  (B, N, N, H)   – indexed as (b,i,j,h)

    B, N, H,                # static problem size
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    BLOCK_I: tl.constexpr,  # tile size in the i‑dimension
    BLOCK_J: tl.constexpr,  # tile size in the j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in the reduction (k) dimension
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]."""
    pid_i = tl.program_id(0)          # tile index along i
    pid_j = tl.program_id(1)          # tile index along j
    pid_batch = tl.program_id(2)      # combined batch × hidden dimension

    # decode batch and hidden indices
    b = pid_batch // H                # batch index
    h = pid_batch % H                 # hidden (gate) index

    # offsets for the current tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left tile (i,k) ---------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        left = tl.cast(left, tl.float32)          # promote to FP32

        # -------- right tile (j,k) --------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + offs_j[:, None] * stride_r_j
            + offs_k[None, :] * stride_r_k
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )
        right = tl.cast(right, tl.float32)

        # -------- dot product over k --------------------------------
        # left : (BLOCK_I, BLOCK_K)   right.T : (BLOCK_K, BLOCK_J)
        acc += tl.dot(left, tl.trans(right))

    # --------------------------------------------------------------
    # Store the computed tile
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entrypoint expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Shortcut to weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) producing the hidden dimension
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)        # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of separate linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy pairwise contraction – Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Cast to FP16 for the kernel (bandwidth‑friendly)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer – we accumulate in FP32
    out = torch.empty_like(left, dtype=torch.float32)

    # Strides in **elements** (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    # In the right tensor the second axis (j) shares the same stride as the
    # original third axis (i) because the layout is identical.
    stride_r_j = stride_r_i

    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 (64 × 64 tiles give good occupancy)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_I),    # i‑tiles
        triton.cdiv(N, BLOCK_J),    # j‑tiles
        B * H,                      # combined batch × hidden dimension
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
