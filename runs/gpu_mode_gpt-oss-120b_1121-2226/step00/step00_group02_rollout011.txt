```
@triton.jit
def apply_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    N, H,
    stride_l0, stride_l1, stride_l2,  # for left (batch, i, j)
    stride_lg0, stride_lg1, stride_lg2,
    stride_r0, stride_r1, stride_r2,
    stride_rg0, stride_rg1, stride_rg2,
    stride_mask0, stride_mask1, stride_mask2,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    pid = tl.program_id(0) # batch index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    batch = pid
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # compute masks for out-of-bound
    mask_i = i < N
    mask_j = j < N
    # compute offsets
    # left offset: batch * stride0 + i * stride1 + j * stride2
    left_offset = batch * stride_l0 + i[:, None] * stride_l1 + j[None, :] * stride_l2
    left = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_j[None, :], other=0.0)
```

```
import triton
import triton.language as tl

@triton.jit
def gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_left0, stride_left1, stride_left2,
    stride_left_gate0, stride_left_gate1, stride_left_gate2,
    stride_right0, stride_right1, stride_right2,
    stride_right_gate0, stride_right_gate1, stride_right_gate2,
    stride_mask0, stride_mask1, stride_mask2,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    pid = tl.program_id(0)  # combined batch*hidden index
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # index into batch*hidden dimension
    b_h = pid

    # compute original batch index for mask
    b = b_h // H

    # compute i, j indices
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # offsets
    off = b_h * stride_left0 + i[:, None] * stride_left1 + j[None, :] * stride_left2
    left = tl.load(left_ptr + off, mask=mask_ij, other=0.0)

    off_gate = b_h * stride_left_gate0 + i[:, None] * stride_left_gate1 + j[None, :] * stride_left_gate2
    left_gate = tl.load(left_gate_ptr + off_gate, mask=mask_ij, other=0.0)

    # mask value
    off_mask = b * stride_mask0 + i[:, None] * stride_mask1 + j[None, :] * stride_mask2
    mask_val = tl.load(mask_ptr + off_mask, mask=mask_ij, other=0.0)

    out_left = left * left_gate * mask_val
    tl.store(out_left_ptr + off, out_left, mask=mask_ij)

    # right side
    off_r = b_h * stride_right0 + i[:, None] * stride_right1 + j[None, :] * stride_right2
    right = tl.load(right_ptr + off_r, mask=mask_ij, other=0.0)

    tl.store(out_right_ptr + off_r, out_right, mask=mask_ij)
```

import torch

import triton.language as tl

@triton.jit
def _gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    N, H,
    stride_left0, stride_left1, stride_left2,
    stride_left_gate0, stride_left_gate1, stride_left_gate2,
    stride_right0, stride_right1, stride_right2,
    stride_right_gate0, stride_right_gate1, stride_right_gate2,
    stride_mask0, stride_mask1, stride_mask2,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr):
    # combined batch dimension = B*H
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # original batch index for mask
    b_idx = pid // H

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i < N
    mask_j = j < N

    off_left_gate = pid * stride_left_gate0 + i[:, None] * stride_left_gate1 + j[None, :] * stride_left_gate2

def custom_kernel(data):
    """TriMul "outgoing" implementation using Triton for the gating+mask step.
    The heavy triangular reduction is performed as a batched matrix multiplication
    (B*H, N, N) @ (B*H, N, N)^T using torch.bmm, which maps to highly optimized
    cuBLAS kernels. A small Triton kernel fuses the element‑wise mask and gate
    multiplications for both left and right projected tensors.
    """
    input_tensor, mask, weights, config = data
    # unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # mask handling
    if mask is None or config.get("nomask", False):
        mask = torch.ones(input_tensor.shape[:3], dtype=dtype, device=device)
    else:
        mask = mask.to(dtype)

    # --------------------------------------------------------------------
    # 1. Input layer‑norm (per channel)
    # --------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B,N,N,dim]

    # --------------------------------------------------------------------
    # 2. Linear projections and gates (no bias)
    # --------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    B, N, _, _ = x.shape  # _ is dim

    # --------------------------------------------------------------------
    # 3. Fuse mask + gates with a Triton kernel.
    #    We reshape tensors to (B*H, N, N) to get a single batch dimension.
    # --------------------------------------------------------------------
    # bring hidden dimension to the batch axis
    left_proj_f = left_proj.permute(0, 3, 1, 2).contiguous()     # [B*H, N, N]
    right_proj_f = right_proj.permute(0, 3, 1, 2).contiguous()
    left_gate_f = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate_f = right_gate.permute(0, 3, 1, 2).contiguous()

    # allocate output buffers
    left_out = torch.empty_like(left_proj_f)
    right_out = torch.empty_like(right_proj_f)

    batch = B * hidden_dim

    # strides for the flattened tensors (they are contiguous)
    stride_l0, stride_l1, stride_l2 = left_proj_f.stride()
    stride_rg0, stride_rg1, stride_rg2 = right_gate_f.stride()
    stride_lg0, stride_lg1, stride_lg2 = left_gate_f.stride()
    stride_r0, stride_r1, stride_r2 = right_proj_f.stride()
    stride_mask0, stride_mask1, stride_mask2 = mask.stride()

    # Triton launch configuration
    BLOCK_I = 64
    BLOCK_J = 64
    grid = (
        batch,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _gate_mask_kernel[grid](
        left_proj_f,
        left_gate_f,
        right_proj_f,
        right_gate_f,
        mask,
        left_out,
        right_out,
        N,
        hidden_dim,
        stride_l0,
        stride_l1,
        stride_l2,
        stride_lg0,
        stride_lg1,
        stride_lg2,
        stride_r0,
        stride_r1,
        stride_r2,
        stride_rg0,
        stride_rg1,
        stride_rg2,
        stride_mask0,
        stride_mask1,
        stride_mask2,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
    )

    # --------------------------------------------------------------------
    # 4. Batched matrix multiply over the sequence dimension (tri‑mul)
    # --------------------------------------------------------------------
    # (B*H, N, N) @ (B*H, N, N)^T
    # Use FP16 for the gemm if the hardware supports it – convert back afterwards.
    left_fp16 = left_out.to(torch.float16)
    right_fp16 = right_out.to(torch.float16)
    out_fp16 = torch.bmm(left_fp16, right_fp16.transpose(1, 2))
    out = out_fp16.to(torch.float32)  # shape [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------
    # 5. Output layer norm + gate
    # --------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # element‑wise gate (no mask)

    # --------------------------------------------------------------------
    # 6. Final linear projection back to dim
    # --------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

TriMul (outgoing) implementation.

The forward pass consists of:
1. Layer‑norm on the input tensor.
2. Linear projections (left, right) and three sigmoid gates.
3. A small Triton kernel that fuses:
      left  = left_proj  * left_gate  * mask
      right = right_proj * right_gate * mask
   The tensors are reshaped to (B*H, N, N) so the kernel works on a single
   batch dimension (B*H) and the two sequence axes.
4. The triangular multiplication is a batched matrix multiply:
      out_h = left_h @ right_h.T   for each hidden slice h.
   This is performed with `torch.bmm`, which maps to highly‑optimized cuBLAS.
5. Layer‑norm on the hidden dimension, multiplication by the out‑gate,
   and a final linear projection back to the original channel size.
The Triton kernel only handles the element‑wise mask/gate fusion; the
heavy O(N³) reduction is carried out by the fast `torch.bmm` GEMM.

    B, N, _, _ = x.shape  # extract batch and sequence length

    # -----------------------------------------------------------------
    # 3) Fuse mask + per‑position gates with Triton
    #    Reshape tensors to (B*H, N, N) to obtain a single batch axis.
    # -----------------------------------------------------------------
    # bring hidden dim in front and make tensors contiguous
    left_proj_f  = left_proj.permute(0, 3, 1, 2).contiguous()   # (B*H, N, N)
    right_proj_f = right_proj.permute(0, 3, 1, 2).contiguous()
    left_gate_f  = left_gate.permute(0, 3, 1, 2).contiguous()
    right_gate_f = right_gate.permute(0, 3, 1, 2).contiguous()

    # output buffers (same shape)
    left_out  = torch.empty_like(left_proj_f)
    right_out = torch.empty_like(right_proj_f)

    batch_h = B * hidden_dim  # combined batch dimension for the kernel

    # strides of the flattened tensors (they are contiguous)
    stride_l0, stride_l1, stride_l2 = left_proj_f.stride()
    stride_lg0, stride_lg1, stride_lg2 = left_gate_f.stride()
    stride_r0, stride_r1, stride_r2 = right_proj_f.stride()
    stride_rg0, stride_rg1, stride_rg2 = right_gate_f.stride()
    stride_mask0, stride_mask1, stride_mask2 = mask.stride()

    # kernel launch configuration
    BLOCK_I = 64
    BLOCK_J = 64
    grid = (
        batch_h,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _gate_mask_kernel[grid](
        left_proj_f,
        left_gate_f,
        right_proj_f,
        right_gate_f,
        mask,
        left_out,
        right_out,
        N,
        hidden_dim,
        stride_l0,
        stride_l1,
        stride_l2,
        stride_lg0,
        stride_lg1,
        stride_lg2,
        stride_r0,
        stride_r1,
        stride_r2,
        stride_rg0,
        stride_rg1,
        stride_rg2,
        stride_mask0,
        stride_mask1,
        stride_mask2,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
    )

    # -----------------------------------------------------------------
    # 4) Tri‑multiplicative reduction via batched GEMM
    # -----------------------------------------------------------------
    # Use FP16 for the GEMM to exploit Tensor Cores, then cast back.
    left_fp16  = left_out.to(torch.float16)
    right_fp16 = right_out.to(torch.float16)

    # (B*H, N, N) @ (B*H, N, N)^T
    out_fp16 = torch.bmm(left_fp16, right_fp16.transpose(1, 2))
    out = out_fp16.to(torch.float32)  # back to FP32

    # reshape to [B, N, N, hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 5) Output LayerNorm + out‑gate
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # element‑wise multiplicative gate (no mask)

    # -----------------------------------------------------------------
    # 6) Final projection back to original channel dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
