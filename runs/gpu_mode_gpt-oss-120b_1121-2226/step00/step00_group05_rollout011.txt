def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Get shapes
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    # Norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).type_as(x)
        left = left * mask
        right = right * mask
    # gates
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # Now compute out via Triton kernel
    left_contig = left.contiguous()
    right_contig = right.contiguous()
    # allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # get strides
    # left shape (B,N,N,H)
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h = left_contig.stride()
    stride_right_bs, stride_right_i, stride_right_k, stride_right_h = right_contig.stride()
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # kernel launch config
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # compute grid dimensions
    grid_M = triton.cdiv(N, BLOCK_M)
    grid_N = triton.cdiv(N, BLOCK_N)
    # tasks = B * hidden_dim
    grid_BH = B * hidden_dim

    # launch kernel
    trimul_gemm_kernel[(grid_M, grid_N, grid_BH)](
        left_contig,
        right_contig,
        out,
        B,
        N,
        hidden_dim,
        stride_left_bs, stride_left_i, stride_left_k, stride_left_h,
        stride_right_bs, stride_right_i, stride_right_k, stride_right_h,
        stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # final out processing: layernorm, gating, linear out.
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

```python
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h,
    stride_right_bs, stride_right_i, stride_right_k, stride_right_h,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(axis=0)  # tile row
    pid_n = tl.program_id(axis=1)  # tile col
    pid_task = tl.program_id(axis=2)  # combined batch*head

    # compute batch and head index
    b = pid_task // H
    h = pid_task % H

    # offsets for row and col
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for boundaries
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers base for left and right (need left[i,k] and right_T[k,j] for each h)
    # left: shape (B,N,N,H) -> we need tile of size BLOCK_M x BLOCK_K at row i dimension (m) and col k dimension (k)
    # right_T: we load transpose of right to have shape (BLOCK_K, BLOCK_N) where row is k and col is j
    # So we will loop over k tiles
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_tiles_k):
        # compute k offsets
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K)
        left_ptrs = left_ptr + b*stride_left_bs + offs_m[:, None]*stride_left_i + offs_k[None, :]*stride_left_k + h*stride_left_h
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right_T tile (K, N), i.e., right transposed
        # right original shape: (B,N,N,H) indexing (b, i, k, h). We need element right[b, j, k, h] where i=j original; after transpose, row = k, col = j.
        right_ptrs = right_ptr + b*stride_right_bs + offs_k[:, None]*stride_right_k + offs_n[None, :]*stride_right_i + h*stride_right_h
        # Note: Use stride_right_i for j dimension (since original i index maps to j after transpose)
        b_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # multiply-accumulate
        acc += tl.dot(a, b_tile)

    # store result
    out_ptrs = out_ptr + b*stride_out_bs + offs_m[:, None]*stride_out_i + offs_n[None, :]*stride_out_j + h*stride_out_h
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present and not nomask
    if not config.get("nomask", False) and mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel (optional)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    # Allocate output
    out = torch.empty((left.shape[0], left.shape[1], left.shape[2], hidden_dim), dtype=torch.float32, device=device)
    # Compute strides
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_bs, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h = out.stride()

    B, N, _, _ = left_fp16.shape
    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # Grid dimensions
    grid_M = triton.cdiv(N, BLOCK_M)
    grid_N = triton.cdiv(N, BLOCK_N)
    grid_tasks = B * hidden_dim
    num_k_tiles = (N + BLOCK_K - 1) // BLOCK_K

    trimul_gemm_kernel[(grid_M, grid_N, grid_tasks)](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_left_bs, stride_left_i, stride_left_k, stride_left_h,
        stride_right_bs, stride_right_i, stride_right_k, stride_right_h,
        stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
        num_k_tiles,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Continue with out norm and projection
    # LayerNorm on hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

```
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h,
    stride_right_bs, stride_right_i, stride_right_k, stride_right_h,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
    num_k_tiles,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_task = tl.program_id(axis=2)

    # decode batch and head
    b = pid_task // H
    h = pid_task % H

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K)
        left_ptrs = left_ptr + b*stride_left_bs + offs_m[:, None]*stride_left_i + offs_k[None, :]*stride_left_k + h*stride_left_h
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right transposed tile (K, N) - using original right layout where i dimension is j after transpose
        right_ptrs = right_ptr + b*stride_right_bs + offs_k[:, None]*stride_right_k + offs_n[None, :]*stride_right_i + h*stride_right_h
        b_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product accumulate
        acc += tl.dot(a, b_tile)

    # Store result
    out_ptrs = out_ptr + b*stride_out_bs + offs_m[:, None]*stride_out_i + offs_n[None, :]*stride_out_j + h*stride_out_h
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The core operation is
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which can be seen as a batch of independent GEMM computations:
    C_d = A_d @ B_dᴛ   where A_d = left[..., d] (N×N) and B_d = right[..., d] (N×N).

We tile the i‑ and j‑dimensions (M and N) and loop over k in blocks,
accumulating in fp32.  The kernel processes all (batch, hidden) pairs
in parallel (grid dimension 2) and uses `tl.dot` for the inner tile
multiply‑accumulate.  The rest of the TriMul forward pass (LayerNorm,
Linear projections, gating, and output projection) is performed with
standard PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # batch, seq_len, hidden_dim
    stride_left_bs, stride_left_i, stride_left_k, stride_left_h,
    stride_right_bs, stride_right_i, stride_right_k, stride_right_h,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_h,
    num_k_tiles,                                # number of K‑tiles (runtime int)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C_{i,j,d} = Σ_k A_{i,k,d} * B_{j,k,d} for all batches b and heads d.
    Arguments are raw pointers and stride information for the tensors:
        left  : shape (B, N, N, H)  (A)
        right : shape (B, N, N, H)  (B)
        out   : shape (B, N, N, H)  (C)

    The kernel is tiled over M (i) and N (j) with BLOCK_M × BLOCK_N
    output tiles, and iterates over K in blocks of size BLOCK_K.
    """
    pid_m = tl.program_id(axis=0)          # tile row index i
    pid_n = tl.program_id(axis=1)          # tile col index j
    pid_task = tl.program_id(axis=2)       # combined (batch, head) index

    # decode batch and head from linear task id
    b = pid_task // H
    h = pid_task % H

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension blocks
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load A tile (M × K) ------------------------------------
        # left[b, i, k, h]
        left_ptrs = (left_ptr
                     + b * stride_left_bs
                     + offs_m[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k
                     + h * stride_left_h)
        a = tl.load(left_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----- Load Bᵀ tile (K × N) -----------------------------------
        # We need right[b, j, k, h] which is the transpose of the
        # original right tensor in the (i,k) axes.
        # right[b, j, k, h] => stride i for j, stride k for row.
        right_ptrs = (right_ptr
                      + b * stride_right_bs
                      + offs_k[:, None] * stride_right_k   # row dimension = k
                      + offs_n[None, :] * stride_right_i   # col dimension = j
                      + h * stride_right_h)
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)

        # ----- Accumulate ------------------------------------------------
        acc += tl.dot(a, b_tile)   # (M,K)·(K,N) → (M,N)

    # ----- Store the result -------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_out_bs
                + offs_m[:, None] * stride_out_i
                + offs_n[None, :] * stride_out_j
                + h * stride_out_h)
    tl.store(out_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Public entry point expected by the testing harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict containing at least "dim", "hidden_dim", and "nomask"

    Returns
    -------
    out : torch.Tensor [B, N, N, dim] (float32)
    """
    # -------------------------- unpack ------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------- LayerNorm over dim --------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ----------------------- Linear projections ---------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------- optional mask ---------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)     # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ----------------------- Gating --------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------- Triton GEMM (core TriMul) --------------------
    # Ensure tensors are contiguous for correct stride extraction
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Extract strides (in elements, not bytes)
    s_l_bs, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_bs, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_bs, s_o_i, s_o_j, s_o_h = out.stride()

    # Tiling parameters – tweaked for H100 / FP32 compute
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions
    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid_tasks = B * hidden_dim                     # one kernel instance per (batch, head)
    num_k_tiles = (N + BLOCK_K - 1) // BLOCK_K      # runtime int

    trimul_gemm_kernel[(grid_m, grid_n, grid_tasks)](
        left,                     # left_ptr
        right,                    # right_ptr
        out,                      # out_ptr
        B, N, hidden_dim,
        s_l_bs, s_l_i, s_l_k, s_l_h,
        s_r_bs, s_r_i, s_r_k, s_r_h,
        s_o_bs, s_o_i, s_o_j, s_o_h,
        num_k_tiles,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------- Output LayerNorm ----------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ----------------------- Output gating & final linear ---------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out
