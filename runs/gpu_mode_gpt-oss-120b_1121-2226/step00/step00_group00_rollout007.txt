    # Linear projections:
    # Using torch.nn.functional.linear
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,H]

    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)
    # Compute batch and hidden index:
    b = pid_z // H
    h = pid_z % H

    left_base = left_ptr + b * stride_left_b + h * stride_left_h

    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

```python
import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)   # block along i
    pid_j = tl.program_id(1)   # block along j
    pid_z = tl.program_id(2)   # combined batch * hidden

    # derive batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # Compute base pointers for the current (b,h) slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Offsets for the i and j dimensions of the output tile
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for valid i/j indices (for the right‑most edge)
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k):
        k_offsets = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # left tile (i,k)
        left_ptrs = left_base + i_offsets[:, None] * stride_left_i + k_offsets[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (k,j) – note the second matrix is already transposed
        right_ptrs = right_base + k_offsets[:, None] * stride_right_k + j_offsets[None, :] * stride_right_j
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # GEMM on this tile
        acc += tl.dot(left_tile, right_tile)

    # Store the result
    out_ptrs = out_base + i_offsets[:, None] * stride_out_i + j_offsets[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

```python
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,k,j,h}``
    using a Triton kernel. ``right`` is expected to be already transposed
    (shape [B,N,N,H] corresponding to original ``right`` permuted to [B,N,N,H]).
    """
    assert left.dim() == 4 and right.dim() == 4
    B, N, _, H = left.shape
    # allocate output
    out = torch.empty_like(left)
    # strides (int64) – store as python ints
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_k, stride_right_j, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    # block sizes – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_k, stride_right_j, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out
```

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operation.
    It follows the PyTorch reference:
      * LayerNorm over the input channel dimension
      * Linear projections (left, right) with optional mask
      * Gated multiplicative update via a Triton kernel
      * LayerNorm over the hidden dimension, out‑gate, and final linear projection.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure same dtype as the projected tensors
        mask = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating linear + sigmoid
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core multiplicative update (Triton kernel)
    # ------------------------------------------------------------------
    # right is transposed on the second pair of axes to match the einsum contraction.
    right_T = right.permute(0, 2, 1, 3).contiguous()   # shape [B,N,N,hidden_dim] = (b,k,j,h)
    out_hidden = tri_mul_triton(left, right_T)        # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden, (hidden_dim,), weight=to_out_norm_weight,
        bias=to_out_norm_bias, eps=eps
    )
    out_hidden = out_hidden * out_gate

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out_hidden, to_out_weight)  # [B,N,N,dim]

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over its last dimension (dim).
2. Linear projections (left/right) from dim → hidden_dim (no bias).
3. (Optional) element‑wise mask applied to the projected tensors.
4. Gating:  left_gate, right_gate, out_gate = σ(Linear(x)).
   Apply left/right gates to the projected tensors.
5. Core multiplicative update:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,k,j,h}
   The right tensor is first transposed on the (i,k) axes so that the
   contraction is a batched matrix multiplication (N×N)·(N×N) per
   hidden channel.  This contraction is performed by a custom Triton
   kernel that tiles the i‑, j‑ and k‑axes (BLOCK_M×BLOCK_N tiles,
   reduced over BLOCK_K).
6. Layer‑norm over the hidden dimension, gated by out_gate,
   and final Linear hidden_dim → dim (no bias).

Only the heavy N³ contraction is off‑loaded to Triton; all other
operations use regular PyTorch (highly optimized) kernels.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the “outgoing” einsum:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right_T[b,k,j,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,        # const float*   left  [B,N,N,H]   (i,k)
    right_ptr,       # const float*   right_T [B,N,N,H] (k,j) – already transposed
    out_ptr,         # float*         out   [B,N,N,H]
    B, N, H,         # runtime integers
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_k, stride_right_j, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for each hidden channel using tiling."""
    pid_i = tl.program_id(0)                     # tile along i
    pid_j = tl.program_id(1)                     # tile along j
    pid_z = tl.program_id(2)                     # combined batch*hidden index

    # ------------------------------------------------------------------
    # Derive batch index b and hidden index h from the third dimension
    # ------------------------------------------------------------------
    b = pid_z // H
    h = pid_z % H

    # Base pointers for the current (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Tile offsets for the i and j dimensions
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # Accumulator for the tile (float32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for kb in range(num_k_tiles):
        k_offsets = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # ------------------------------------------------------------------
        # Load left tile  (i, k)
        # ------------------------------------------------------------------
        left_ptrs = (left_base
                     + i_offsets[:, None] * stride_left_i
                     + k_offsets[None, :] * stride_left_k)
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)   # ensure fp32 arithmetic

        # ------------------------------------------------------------------
        # Load right tile (k, j) – already transposed
        # ------------------------------------------------------------------
        right_ptrs = (right_base
                      + k_offsets[:, None] * stride_right_k
                      + j_offsets[None, :] * stride_right_j)
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # Block matrix multiplication and accumulate
        # ------------------------------------------------------------------
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write out the result tile
    # ------------------------------------------------------------------
    out_ptrs = (out_base
                + i_offsets[:, None] * stride_out_i
                + j_offsets[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel and returns the contracted tensor
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right_T: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing TriMul contraction using Triton.

    Args:
        left:   Tensor of shape [B, N, N, H]   (i,k)
        right_T:Tensor of shape [B, N, N, H]   (k,j) – already transposed

    Returns:
        out: Tensor of shape [B, N, N, H]
    """
    assert left.dim() == 4 and right_T.dim() == 4
    B, N, _, H = left.shape

    # Output allocation
    out = torch.empty_like(left)

    # Strides (convert to Python ints for Triton)
    s_li_b, s_li_i, s_li_k, s_li_h = left.stride()
    s_rk_b, s_rk_k, s_rk_j, s_rk_h = right_T.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Tuned tile sizes for H100 (fits in shared memory)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H                           # batch × hidden
    )

    tri_mul_kernel[grid](
        left, right_T, out,
        B, N, H,
        s_li_b, s_li_i, s_li_k, s_li_h,
        s_rk_b, s_rk_k, s_rk_j, s_rk_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul module.
    The implementation follows the reference PyTorch code, but the
    N³ contraction is executed by a custom Triton kernel for speed.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask dtype matches the projected tensors
        mask = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid after a linear projection)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core multiplicative update (Triton)
    #    Right tensor is transposed on (i,k) → (k,i) to match the einsum.
    # ------------------------------------------------------------------
    right_T = right.permute(0, 2, 1, 3).contiguous()   # shape [B, N, N, hidden_dim]
    out_hidden = tri_mul_triton(left, right_T)        # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final Linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps
    )
    out_hidden = out_hidden * out_gate

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out_hidden, to_out_weight)  # [B, N, N, dim]

    return out
