@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
    pid = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)  # combined batch-head index

    # compute batch and head
    b = pid_hb // H
    h = pid_hb % H

    # compute offsets
    i_offsets = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # compute pointers
        left_ptrs = left_ptr + (b * stride_l_b + h * stride_l_h
                                 + i_offsets[:, None] * stride_l_i
                                 + k_offset[None, :] * stride_l_k)
        right_ptrs = right_ptr + (b * stride_r_b + h * stride_r_h
                                 + k_offset[:, None] * stride_r_k
                                 + j_offsets[None, :] * stride_r_i)  # note: right dims (k,i?) but we want (k,j). i stride is second dim for right.

        # Load with mask
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # multiply accumulate
        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_ptr + (b * stride_o_b + h * stride_o_h
                         + i_offsets[:, None] * stride_o_i
                         + j_offsets[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Linear proj
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight) # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # Apply mask if needed
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq

"""
TriMul “outgoing” implementation (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections → left/right tensors of shape [B, N, N, H] (H = hidden_dim).
3. Apply an optional mask (broadcast on the channel dim).
4. Compute three sigmoidal gates (left, right, out) from the same normalized input.
5. Fuse gates with the projected tensors.
6. Core operation: for each batch‑head pair compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply per hidden channel.
   This O(B·H·N³) contraction is implemented in a custom Triton kernel
   that tiles the i‑ and j‑dimensions (M×N tiles) and iterates over k in
   BLOCK_K steps, using `tl.dot` for efficient tensor‑core accumulation.
7. Apply a second LayerNorm (to_out_norm) and multiply by the out‑gate.
8. Final linear projection back to the original channel dimension C.

The heavy reduction (step 6) is executed on the GPU with Triton, while all
other operations (LayerNorm, Linear, Sigmoid) use PyTorch for simplicity.
The kernel works for any batch size (1–2), sequence length up to 1024,
and arbitrary hidden/channel dimensions supported by the test matrix.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _trmul_kernel(
    left_ptr,            # *float32   [B, N, N, H]
    right_ptr,           # *float32   [B, N, N, H]
    out_ptr,             # *float32   [B, N, N, H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
    using block‑wise GEMM for each (batch, head) pair.
    """
    pid_i = tl.program_id(0)          # tile over output‑row (i)
    pid_j = tl.program_id(1)          # tile over output‑col (j)
    pid_hb = tl.program_id(2)         # combined batch‑head dimension

    # decode batch and head from the combined index
    b = pid_hb // H
    h = pid_hb % H

    # base offsets for i and j within the tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_offsets = i_base + tl.arange(0, BLOCK_M)
    j_offsets = j_base + tl.arange(0, BLOCK_N)

    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # loop over the reduction dimension k
    # ---------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_base = k * BLOCK_K
        k_offsets = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # left tile:   shape (BLOCK_M, BLOCK_K)  → indices (i, k)
        left_ptrs = left_ptr + (
            b * stride_l_b
            + h * stride_l_h
            + i_offsets[:, None] * stride_l_i
            + k_offsets[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        # right tile:  shape (BLOCK_K, BLOCK_N)  → indices (k, j)
        # (note the swapped order w.r.t. the original tensor)
        right_ptrs = right_ptr + (
            b * stride_r_b
            + h * stride_r_h
            + k_offsets[:, None] * stride_r_k
            + j_offsets[None, :] * stride_r_i
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )

        # accumulate the matrix product for this k‑tile
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)
    # ---------------------------------------------------------

    # write the result tile
    out_ptrs = out_ptr + (
        b * stride_o_b
        + h * stride_o_h
        + i_offsets[:, None] * stride_o_i
        + j_offsets[None, :] * stride_o_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None] & j_mask[None, :],
    )

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension
    assert C == config["dim"], "config dim does not match input channels"

    # ------------------------------------------------------------------
    # 1. First LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"] # (H, C)

    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    mask = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Compute gates (sigmoid(linear(x)))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # allocate output tensor [B,N,N,H]

    # Tile sizes – chosen to fit into H100 shared memory / registers.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (in elements, not bytes) for each tensor
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Grid: (tiles over i, tiles over j, batch*head)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    _trmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # tune for H100; 4 works well for these tile sizes
    )

    # ------------------------------------------------------------------
    # 6. Second LayerNorm + out‑gate
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (H,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final projection back to original channel dimension
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]   # (C, H)
    out = F.linear(out, to_out_weight)        # [B, N, N, C]

    return out
```
