import triton.language as tl

@triton.jit
def mask_and_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_pb0, stride_pb1, stride_pb2, stride_pb3,
    stride_gb0, stride_gb1, stride_gb2, stride_gb3,
    stride_mb0, stride_mb1, stride_mb2,
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,
    BLOCK_H: tl.constexpr, BLOCK_N: tl.constexpr,
):
    pid = tl.program_id(0)  # flatten over batch and N
    # Compute b,i,j index for each program
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N
    # Offsets for block across H dimension
    offs_h = tl.arange(0, BLOCK_H)
    # Compute base pointer offsets
    proj_ptrs = proj_ptr + b*stride_pb0 + i*stride_pb1 + j*stride_pb2 + offs_h*stride_pb3
    gate_ptrs = gate_ptr + b*stride_gb0 + i*stride_gb1 + j*stride_gb2 + offs_h*stride_gb3
    mask_ptrs = mask_ptr + b*stride_mb0 + i*stride_mb1 + j*stride_mb2
    # load
    proj = tl.load(proj_ptrs, mask=offs_h < H, other=0.0)
    gate = tl.load(gate_ptrs, mask=offs_h < H, other=0.0)

@triton.jit
def mask_and_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_pb0, stride_pb1, stride_pb2, stride_pb3,
    stride_gb0, stride_gb1, stride_gb2, stride_gb3,
    stride_mb0, stride_mb1, stride_mb2,
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,
    BLOCK_H: tl.constexpr
):

    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N
    # offsets for h dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_off = b*stride_mb0 + i*stride_mb1 + j*stride_mb2
    # load mask scalar (broadcast)
    mask_val = tl.load(mask_ptr + mask_off)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for mask and gate multiplication
@triton.jit
def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                     B, N, H,
                     stride_pb0, stride_pb1, stride_pb2, stride_pb3,
                     stride_gb0, stride_gb1, stride_gb2, stride_gb3,
                     stride_mb0, stride_mb1, stride_mb2,
                     stride_ob0, stride_ob1, stride_ob2, stride_ob3,
                     BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)                # 1D grid over (b,i,j)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # vector of H indices in this block
    offs_h = tl.arange(0, BLOCK_H)
    mask = tl.load(mask_ptr + b*stride_mb0 + i*stride_mb1 + j*stride_mb2)
    mask_val = mask[0]
    # compute offsets for proj, gate, out
    proj_off = b*stride_pb0 + i*stride_pb1 + j*stride_pb2 + offs_h*stride_pb3
    gate_off = b*stride_gb0 + i*stride_gb1 + j*stride_gb2 + offs_h*stride_gb3
    out_off  = b*stride_ob0 + i*stride_ob1 + j*stride_ob2 + offs_h*stride_ob3

    # guard for tail of H dimension
    mask_h = offs_h < H
    proj = tl.load(proj_ptr + proj_off, mask=mask_h, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask_h, other=0.0)

    out = proj * mask_val * gate
    tl.store(out_ptr + out_off, out, mask=mask_h)

def apply_mask_gate(proj, gate, mask, block_h=64):
    """
    Elementwise: out = proj * mask * gate, using a Triton kernel.
    proj, gate: (B,N,N,H) float tensors
    mask: (B,N,N,1) broadcastable to (B,N,N,H)
    Returns out tensor of same shape.
    """
    B, N, N2, H = proj.shape
    assert N == N2
    out = torch.empty_like(proj)
    grid = (B * N * N,)
    mask_gate_kernel[grid](
        proj_ptr=proj,
        gate_ptr=gate,
        mask_ptr=mask,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_pb0=proj.stride(0),
        stride_pb1=proj.stride(1),
        stride_pb2=proj.stride(2),
        stride_pb3=proj.stride(3),
        stride_gb0=gate.stride(0),
        stride_gb1=gate.stride(1),
        stride_gb2=gate.stride(2),
        stride_gb3=gate.stride(3),
        stride_mb0=mask.stride(0),
        stride_mb1=mask.stride(1),
        stride_mb2=mask.stride(2),
        stride_ob0=out.stride(0),
        stride_ob1=out.stride(1),
        stride_ob2=out.stride(2),
        stride_ob3=out.stride(3),
        BLOCK_H=block_h,
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.
    Steps:
        1. Layer‑norm the input.
        2. Linear projections (left/right) and three gating linear layers.
        3. Fuse mask, projection and gate with a small Triton kernel.
        4. Compute the pairwise multiplication across the sequence dimension
           via a batched GEMM where the hidden dimension is folded into the batch.
        5. Apply a second layer‑norm, the output gate and the final linear map.
    Input:
        data = (x, mask, weights, config)
            x       – [B, N, N, C] tensor
            mask    – [B, N, N] (may be None)
            weights – dict of model weight tensors
            config  – {"dim": C, "hidden_dim": H}
    Output:
        Tensor of shape [B, N, N, C] (float32).
    """
    # unpack arguments
    x, mask, weights, config = data
    B, N, _, C = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1. Input layer‑norm
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # 2. Linear projections and gating pre‑activations
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])
    left_gate_pre = F.linear(x_norm, weights["left_gate.weight"])
    right_gate_pre = F.linear(x_norm, weights["right_gate.weight"])
    out_gate_pre = F.linear(x_norm, weights["out_gate.weight"])

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # 3. Mask handling – broadcast to (B,N,N,1) if needed
    if mask is None or mask.numel() == 0:
        mask_tensor = torch.ones((B, N, N, 1), dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask.unsqueeze(-1).to(dtype=x.dtype)
    # fuse mask and gate with Triton
    left = apply_mask_gate(left_proj, left_gate, mask_tensor)
    right = apply_mask_gate(right_proj, right_gate, mask_tensor)

    # 4. Core Triangle‑multiplicative update via batched GEMM
    # reshape to (B*hidden_dim, N, N) for both sides
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – transpose k axis
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # batched matrix multiplication (B*H, N, N) @ (B*H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # 5. Output layer‑norm, gating, final linear
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
TriMul “outgoing” forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (norm over the last C dimension).
2. Linear projections:
   * left_proj, right_proj  → (B,N,N,H)
   * left_gate, right_gate, out_gate (pre‑sigmoid) → (B,N,N,H)
3. Fuse mask, projection and gate with a small Triton kernel:
   out = proj * mask * sigmoid(gate)
4. Core update:
   left:  (B,N,N,H)   → (B,H,N,N) → (B*H,N,N)
   right: (B,N,N,H)   → (B,H,N,N) (transpose the two sequence axes) → (B*H,N,N)
   Compute batched GEMM  (B*H,N,N) @ (B*H,N,N)ᵀ  → (B*H,N,N)
   Reshape back to (B,N,N,H).
5. Apply a second LayerNorm, the output‑gate and the final linear map
   back to the original dimension C.

Only the element‑wise mask‑gate fusion is written in Triton; the heavy
pairwise reduction is performed with a batched GEMM (cuBLAS), which is
highly optimized on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: out = proj * mask * gate   (proj,gate : [B,N,N,H]; mask : [B,N,N,1])
# ---------------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_pb0, stride_pb1, stride_pb2, stride_pb3,
    stride_gb0, stride_gb1, stride_gb2, stride_gb3,
    stride_mb0, stride_mb1, stride_mb2,
    stride_ob0, stride_ob1, stride_ob2, stride_ob3,
    BLOCK_H: tl.constexpr
):
    # each program handles one (b,i,j) coordinate; all H channels are processed in a block
    pid = tl.program_id(0)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # H‑dimension offsets
    offs_h = tl.arange(0, BLOCK_H)

    # ---- load scalar mask (broadcast to H) ---------------------------------
    mask_off = b * stride_mb0 + i * stride_mb1 + j * stride_mb2
    mask_val = tl.load(mask_ptr + mask_off)            # scalar

    # ---- load proj and gate vectors -----------------------------------------
    proj_off = b * stride_pb0 + i * stride_pb1 + j * stride_pb2 + offs_h * stride_pb3
    gate_off = b * stride_gb0 + i * stride_gb1 + j * stride_gb2 + offs_h * stride_gb3
    out_off  = b * stride_ob0 + i * stride_ob1 + j * stride_ob2 + offs_h * stride_ob3

    mask_h = offs_h < H                     # guard for tail blocks
    proj = tl.load(proj_ptr + proj_off, mask=mask_h, other=0.0)
    gate = tl.load(gate_ptr + gate_off, mask=mask_h, other=0.0)

    out = proj * mask_val * gate
    tl.store(out_ptr + out_off, out, mask=mask_h)


def apply_mask_gate(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor,
                    block_h: int = 64) -> torch.Tensor:
    """
    Triton‑fused elementwise:   out = proj * mask * gate
    proj, gate : (B,N,N,H)  float tensors
    mask       : (B,N,N,1) broadcastable to (B,N,N,H)
    Returns tensor of shape (B,N,N,H).
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    # 1‑D grid: one program per (b,i,j) element
    grid = (B * N * N,)

    mask_gate_kernel[grid](
        proj_ptr=proj,
        gate_ptr=gate,
        mask_ptr=mask,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_pb0=proj.stride(0),
        stride_pb1=proj.stride(1),
        stride_pb2=proj.stride(2),
        stride_pb3=proj.stride(3),
        stride_gb0=gate.stride(0),
        stride_gb1=gate.stride(1),
        stride_gb2=gate.stride(2),
        stride_gb3=gate.stride(3),
        stride_mb0=mask.stride(0),
        stride_mb1=mask.stride(1),
        stride_mb2=mask.stride(2),
        stride_ob0=out.stride(0),
        stride_ob1=out.stride(1),
        stride_ob2=out.stride(2),
        stride_ob3=out.stride(3),
        BLOCK_H=block_h,
    )
    return out


# ---------------------------------------------------------------------------
# Entrypoint
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        x       – torch.Tensor, shape [B, N, N, C]
        mask    – torch.Tensor, shape [B, N, N]  (may be None)
        weights – dict of model parameters
        config  – {"dim": C, "hidden_dim": H}

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, C = x.shape
    dim = config["dim"]               # = C
    hidden_dim = config["hidden_dim"] # = H
    eps = 1e-5

    # -----------------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------------
    # 2. Linear projections + gating pre‑activations
    # -----------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])          # (B,N,N,H)
    right_proj = F.linear(x_norm, weights["right_proj.weight"])       # (B,N,N,H)

    left_gate_pre = F.linear(x_norm, weights["left_gate.weight"])     # (B,N,N,H)
    right_gate_pre = F.linear(x_norm, weights["right_gate.weight"])   # (B,N,N,H)
    out_gate_pre = F.linear(x_norm, weights["out_gate.weight"])       # (B,N,N,H)

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # -----------------------------------------------------------------------
    # 3. Mask handling + Triton fusion
    # -----------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask_tensor = torch.ones((B, N, N, 1), dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask.unsqueeze(-1).to(dtype=x.dtype, device=x.device)

    left = apply_mask_gate(left_proj, left_gate, mask_tensor)   # (B,N,N,H)
    right = apply_mask_gate(right_proj, right_gate, mask_tensor)  # (B,N,N,H)

    # -----------------------------------------------------------------------
    # 4. Core triangle‑multiplicative update via batched GEMM
    # -----------------------------------------------------------------------
    # reshape to (B, H, N, N) and make contiguous for the view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N) – transpose k axis

    left_mat = left_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N) # (B*H, N, N)

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat)         # (B*H, N, N)

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------------
    # 5. Output LayerNorm, output‑gate and final linear
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B,N,N,dim)

    return out
