"""
custom_kernel for TriMul (outgoing) – AlphaFold3 style.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]) using
   `torch.nn.functional.layer_norm` with the provided weight/bias.
2. Linear projections (no bias) produce `left` and `right`
   (shape [B,N,N,H]) where `H = hidden_dim`.
3. Compute three gate tensors (`left_gate`, `right_gate`, `out_gate`) by
   applying a bias‑less linear layer on the normalized `x` followed by a
   sigmoid.
4. **Mask & gate fusion** – a tiny Triton kernel fuses
   `left = left * mask * left_gate` and
   `right = right * mask * right_gate`.  The mask is optional (controlled
   by `config["nomask"]`).
5. Pairwise multiplication over the sequence dimension:
   For each hidden channel `h` we compute
   `out[...,h] = left[...,h] @ right[...,h].T`,
   i.e. a batched matrix multiplication of shape
   `[B*H, N, N]`.  This is performed with `torch.bmm` (highly tuned
   cuBLAS on H100).
6. Layer‑norm the intermediate result (`out`) over the hidden dimension,
   again using `torch.nn.functional.layer_norm`.
7. Apply the output‑gate (`out = out * out_gate`).
8. Final linear projection from hidden_dim → dim (`to_out`).

The kernel only handles the element‑wise mask‑gate fusion – the heaviest
computation (the N³ contraction) is delegated to cuBLAS for optimal
performance.  The whole pipeline runs in float32 and returns a tensor
with shape `[B, N, N, dim]`.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gating for the left/right
# tensors.  Operates on flattened tensors of shape (M, H) where
#   M = B * N * N   (pairwise positions)
#   H = hidden_dim
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [M, H] tensors to be updated in‑place
    left_gate_ptr, right_gate_ptr,      # [M, H] gate tensors (already sigmoid‑ed)
    mask_ptr,                          # [M] mask (float, 0/1)
    M, H,                              # runtime sizes
    stride_left_m, stride_left_h,
    stride_right_m, stride_right_h,
    stride_lgate_m, stride_lgate_h,
    stride_rgate_m, stride_rgate_h,
    stride_mask_m,
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    # program id identifies a tile of rows
    pid = tl.program_id(axis=0)
    row_off = pid * BLOCK_M + tl.arange(0, BLOCK_M)          # (BLOCK_M)
    col_off = tl.arange(0, BLOCK_H)                         # (BLOCK_H)

    # bounds
    row_mask = row_off < M
    col_mask = col_off < H
    active = row_mask[:, None] & col_mask[None, :]

    # ------------------------------------------------------------------
    # Load mask (scalar per row) and broadcast to columns
    # ------------------------------------------------------------------
    mask_val = tl.load(mask_ptr + row_off * stride_mask_m,
                       mask=row_mask,
                       other=0.0)                       # (BLOCK_M)
    mask_val = mask_val[:, None]           # (BLOCK_M, 1)

    # ------------------------------------------------------------------
    # Load tensors
    # ------------------------------------------------------------------
    left = tl.load(left_ptr + row_off[:, None] * stride_left_m
                                 + col_off[None, :] * stride_left_h,
                   mask=active,
                   other=0.0)
    right = tl.load(right_ptr + row_off[:, None] * stride_right_m
                                   + col_off[None, :] * stride_right_h,
                    mask=active,
                    other=0.0)

    lgate = tl.load(left_gate_ptr + row_off[:, None] * stride_lgate_m
                                     + col_off[None, :] * stride_lgate_h,
                    mask=active,
                    other=0.0)
    rgate = tl.load(right_gate_ptr + row_off[:, None] * stride_rgate_m
                                      + col_off[None, :] * stride_rgate_h,
                    mask=active,
                    other=0.0)

    # ------------------------------------------------------------------
    # Apply mask and gates (all element‑wise)
    # ------------------------------------------------------------------
    left = left * mask_val * lgate
    right = right * mask_val * rgate

    # ------------------------------------------------------------------
    # Store results back (in‑place)
    # ------------------------------------------------------------------
    tl.store(left_ptr + row_off[:, None] * stride_left_m
                         + col_off[None, :] * stride_left_h,
             left,
             mask=active)
    tl.store(right_ptr + row_off[:, None] * stride_right_m
                          + col_off[None, :] * stride_right_h,
             right,
             mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor [B, N, N, C]
            - mask: torch.Tensor [B, N, N] (float or bool)
            - weights: Dict[str, torch.Tensor] with all model weights
            - config: Dict with keys:
                "dim": int (C)
                "hidden_dim": int (H)
                "nomask": bool (True ⇒ ignore mask)

    Returns:
        torch.Tensor of shape [B, N, N, C]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (use PyTorch – highly tuned)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Compute gates (sigmoid of bias‑less linear layers)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w)) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))     # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Apply mask and gates with a fused Triton kernel (if mask is used)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is expected to be broadcastable float (0/1)
        mask_float = mask.to(dtype=torch.float32)
        # flatten rows (B*N*N) to treat as a 2‑D tensor
        M = B * N * N
        left_flat = left.view(M, H)
        right_flat = right.view(M, H)
        left_gate_flat = left_gate.view(M, H)
        right_gate_flat = right_gate.view(M, H)
        mask_flat = mask_float.view(M)

        # strides for the flattened tensors (they are contiguous)
        stride_left_m, stride_left_h = left_flat.stride()
        stride_right_m, stride_right_h = right_flat.stride()
        stride_lgate_m, stride_lgate_h = left_gate_flat.stride()
        stride_rgate_m, stride_rgate_h = right_gate_flat.stride()
        stride_mask_m = mask_flat.stride(0)

        # Choose reasonable tile sizes (product must be <= 1024)
        BLOCK_M = 128   # rows per program
        BLOCK_H = 8     # hidden cols per program  (128*8 = 1024 threads)

        grid = (triton.cdiv(M, BLOCK_M),)
        mask_gate_fuse_kernel[grid](
            left_flat, right_flat,
            left_gate_flat, right_gate_flat,
            mask_flat,
            M, H,
            stride_left_m, stride_left_h,
            stride_right_m, stride_right_h,
            stride_lgate_m, stride_lgate_h,
            stride_rgate_m, stride_rgate_h,
            stride_mask_m,
            BLOCK_M=BLOCK_M, BLOCK_H=BLOCK_H,
        )
        # reshape back to original layout
        left = left_flat.view(B, N, N, H)
        right = right_flat.view(B, N, N, H)
    else:
        # No external mask: just apply the element‑wise gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Pairwise multiplication over the sequence dimension
    #    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched matrix‑multiply per hidden channel.
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Batch‑matmul: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    B_mul = B * H
    left_batched = left_perm.view(B_mul, N, N)
    right_batched = right_perm.view(B_mul, N, N)

    out_batched = torch.bmm(left_batched, right_batched.transpose(1, 2))
    out = out_batched.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension C
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)   # [B, N, N, C]

    return out
