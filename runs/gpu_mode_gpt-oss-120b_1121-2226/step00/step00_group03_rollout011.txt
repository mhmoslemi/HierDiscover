def layer_norm(x, weight, bias, eps=1e-5):
    # compute mean across last dim
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm
```

import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_blh, stride_ilh, stride_klh, stride_dlh,
    stride_blr, stride_ilr, stride_klr, stride_dlr,
    stride_bout, stride_iout, stride_jout, stride_dout,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program ids
    pid_z = tl.program_id(axis=2)  # combined batch and hidden dim
    pid_x = tl.program_id(axis=0)  # row block i
    pid_y = tl.program_id(axis=1)  # col block j

    # Compute batch and hidden index
    b = pid_z // hidden_dim
    d = pid_z % hidden_dim

    # Compute block start coordinates
    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N

    # Offsets within tensors
    # left: shape (B, N, N, hidden_dim)
    # left_ptr + b*stride_blh + i0*stride_ilh + k*stride_klh + d*stride_dlh
    # We'll generate range for i and k
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for out-of-bounds in i and j dimensions
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers base for left and right
    left_base = left_ptr + b * stride_blh + d * stride_dlh
    right_base = right_ptr + b * stride_blr + d * stride_dlr
    # Output base
    out_base = out_ptr + b * stride_bout + d * stride_dout

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute actual k index offset
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_ilh) + (cur_k[None, :] * stride_klh)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: we need right as (BLOCK_K, BLOCK_N) i.e., rows over k, cols over j
        right_ptrs = right_base + (offs_j[None, :] * stride_ilr) + (cur_k[:, None] * stride_klr)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute row-col dot product
        acc += tl.dot(left, right)

        # advance pointer maybe not required because we recompute each iteration
    # store acc after casting to appropriate dtype
    # Note output dtype must be float32 to match.
    out_ptrs = out_base + (offs_i[:, None] * stride_iout) + (offs_j[None, :] * stride_jout)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config includes dim, hidden_dim, etc.
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    def layer_norm(x, weight, bias, eps):
        # x shape ...C
        mean = x.mean(dim=-1, keepdim=True)
        var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
        x_norm = (x - mean) * torch.rsqrt(var + eps)
        if weight is not None:
            x_norm = x_norm * weight
        if bias is not None:
            x_norm = x_norm + bias
        return x_norm

    x = layer_norm(input_tensor, norm_weight, norm_bias, eps)

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is not None:
        # Expect mask dtype bool or float
        mask = mask.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gate activations
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute contraction using Triton
    # allocate output tensor
    out = torch.empty_like(left)  # shape B,N,N,hidden_dim

    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # get strides in elements
    stride_blh, stride_ilh, stride_klh, stride_dlh = left.stride()
    stride_blr, stride_ilr, stride_klr, stride_dlr = right.stride()
    stride_bout, stride_iout, stride_jout, stride_dout = out.stride()

    B, N, _, _ = left.shape  # left shape: B,N,N,hidden_dim

    import math
    grid = (
        math.ceil(N / BLOCK_M),
        math.ceil(N / BLOCK_N),
        B * hidden_dim,
    )
    # compile kernel
    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_blh, stride_ilh, stride_klh, stride_dlh,
        stride_blr, stride_ilr, stride_klr, stride_dlr,
        stride_bout, stride_iout, stride_jout, stride_dout,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # after kernel, out is computed

    # apply to_out_norm
    out = layer_norm(out, to_out_norm_weight, to_out_norm_bias, eps)

    # apply out_gate
    out = out * out_gate

    # final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) – fast Triton implementation.

This kernel implements the core O(N³) contraction of the Triangular
Multiplicative Update (outgoing) used in AlphaFold‑3.  The surrounding
operations (LayerNorm, linear projections, gates and the final output
projection) are performed with PyTorch.  The contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is equivalent to a batched matrix multiplication
    out_d = left_d @ right_dᵀ
for each hidden channel *d*.
A Triton kernel computes this GEMM for every (batch, hidden) pair in a
single launch, using tiling (BLOCK_M × BLOCK_N × BLOCK_K) and
accumulating in FP32 for numerical stability.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # B=batch, N=seq_len, H=hidden_dim
    stride_blh, stride_ilh, stride_klh, stride_dlh,   # left strides
    stride_blr, stride_ilr, stride_klr, stride_dlr,   # right strides
    stride_bout, stride_iout, stride_jout, stride_dout,  # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM computing out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    The kernel iterates over the k‑dimension in BLOCK_K tiles and accumulates
    the product of a (BLOCK_M × BLOCK_K) tile from ``left`` and a (BLOCK_K × BLOCK_N)
    tile from ``right`` (the latter is loaded transposed w.r.t. the original
    layout so that a regular tl.dot can be used).
    The third program dimension (axis‑2) enumerates the (batch, hidden) pair.
    """
    pid_z = tl.program_id(axis=2)   # combined batch * hidden
    pid_x = tl.program_id(axis=0)   # block row (i)
    pid_y = tl.program_id(axis=1)   # block col (j)

    # -------------------------------------------------------------
    # decode batch index and hidden‑channel index from pid_z
    # -------------------------------------------------------------
    b = pid_z // H          # batch id
    h = pid_z % H           # hidden‑channel id

    # -------------------------------------------------------------
    # block start indices
    # -------------------------------------------------------------
    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N

    # -------------------------------------------------------------
    # offsets for the three dimensions
    # -------------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the current (batch, hidden) slice
    left_base  = left_ptr  + b * stride_blh + h * stride_dlh
    right_base = right_ptr + b * stride_blr + h * stride_dlr
    out_base   = out_ptr   + b * stride_bout+ h * stride_dout

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # loop over the k dimension
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        mask_k = cur_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_ilh) + (cur_k[None, :] * stride_klh)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # original layout is (b, j, k, h); we need (k, j)
        right_ptrs = right_base + (offs_j[None, :] * stride_ilr) + (cur_k[:, None] * stride_klr)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # matrix multiply‑accumulate
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # -------------------------------------------------------------
    # write back the block
    # -------------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_iout) + (offs_j[None, :] * stride_jout)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] float32 CUDA tensor
            - mask:        [B, N, N] (optional) mask tensor (0/1) or None
            - weights:     dict with all linear / layer‑norm parameters
            - config:     dict containing at least 'dim' and 'hidden_dim'

    Returns:
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # model parameters (all tensors are assumed to be on the same device)
    # ------------------------------------------------------------------
    norm_w      = weights["norm.weight"]
    norm_b      = weights["norm.bias"]
    left_w      = weights["left_proj.weight"]
    right_w     = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w= weights["right_gate.weight"]
    out_gate_w  = weights["out_gate.weight"]
    out_norm_w  = weights["to_out_norm.weight"]
    out_norm_b  = weights["to_out_norm.bias"]
    to_out_w    = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    # Using torch.nn.functional.layer_norm for speed and numerical stability
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_w)          # [B,N,N,hidden_dim]
    right = F.linear(x, right_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool, int or float – cast to the activation dtype
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear + element‑wise multiplication)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction – TriMul – performed with Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=left.dtype,
                      device=left.device)

    # Block sizes – chosen to fit H100 shared memory comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides in elements (not bytes) – required by the kernel
    s_bl, s_il, s_kl, s_dl = left.stride()
    s_br, s_ir, s_kr, s_dr = right.stride()
    s_bo, s_io, s_jo, s_do = out.stride()

    grid = (math.ceil(N / BLOCK_M),
            math.ceil(N / BLOCK_N),
            B * hidden_dim)   # one program per (batch, hidden)

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        s_bl, s_il, s_kl, s_dl,
        s_br, s_ir, s_kr, s_dr,
        s_bo, s_io, s_jo, s_do,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=out_norm_w, bias=out_norm_b, eps=eps)

    out = out * out_gate                     # element‑wise broadcast

    out = F.linear(out, to_out_w)            # [B,N,N,dim]

    return out
