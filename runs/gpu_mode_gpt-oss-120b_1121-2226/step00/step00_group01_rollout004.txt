@triton.jit
def tri_mul_kernel(A_ptr, B_ptr, C_ptr,
                   M, N, K,
                   stride_Am, stride_Ak, stride_Ab, # A stride: batch -> offset? Actually shape A [B*D, M, K] contiguous.
                   stride_Bk, stride_Bn, stride_Bb,
                   stride_Cm, stride_Cn, stride_Cb,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
   pid_m = tl.program_id(0) # block row index
   pid_n = tl.program_id(1) # block col index
   pid_b = tl.program_id(2) # batch index (including hidden)
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # mask for out-of-bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # use fp32 accumulation

   # Loop over K dimension
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < K
       # load A: shape (BLOCK_M, BLOCK_K)
       a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
       b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
       a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
       b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
       # dot product accumulation
       acc += tl.dot(a, b) # dims (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) yields (BLOCK_M, BLOCK_N)
   # Write to C
   c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the last dimension (`dim`).
2. Linear projections (no bias) → `left` and `right` (shape [B,N,N,hidden]).
3. (Optional) apply the pairwise mask.
4. Compute three sigmoid gates from the normalized input and multiply them
   with the projected tensors.
5. For each batch and hidden dimension perform the contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication
      out = left @ rightᵀ .
   This heavy N³ work is carried out by a custom Triton kernel that
   processes many (batch·hidden) matrices in parallel and uses float‑32
   accumulation for accuracy while keeping the tensors in float‑16
   (tensor‑core friendly).
6. Layer‑norm the result over `hidden_dim`, apply the output gate and a
   final linear projection back to `dim`.

Only the N³ contraction (step 5) is performed in Triton; the rest uses
standard PyTorch operators for clarity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched (A @ Bᵀ) where A and B have shape
#   [batch*hidden, M, K]  with M = N, K = N
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_out_kernel(
    A_ptr,                     # *float16
    B_ptr,                     # *float16 (will be accessed transposed)
    C_ptr,                     # *float16
    M, N, K,                   # problem sizes (all equal to seq_len)
    stride_am, stride_ak, stride_ab,   # strides for A (row, col, batch)
    stride_bk, stride_bn, stride_bb,   # strides for B (row, col, batch) – B is read transposed
    stride_cm, stride_cn, stride_cb,   # strides for C (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                 # block row
    pid_n = tl.program_id(1)                 # block column
    pid_b = tl.program_id(2)                 # batched dimension (batch * hidden)

    # --------------------------------------------------------------
    # offsets within the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in float32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A (shape BLOCK_M x BLOCK_K)
        a = tl.load(
            A_ptr
            + pid_b * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load a tile of B transposed (shape BLOCK_K x BLOCK_N)
        # B is accessed as B[k, n] = right[b, n, k]  → stride_bk = 1, stride_bn = N
        b = tl.load(
            B_ptr
            + pid_b * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate the matrix product
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back the result (converted to float16)
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    tl.store(
        C_ptr
        + pid_b * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_transpose(left: torch.Tensor, right: torch.Tensor,
                              BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    Batched GEMM: out = left @ rightᵀ
    left/right shape: [B, N, N, H] (float16)
    Returns out with the same shape and dtype.
    """
    B, N, _, H = left.shape
    # bring the hidden dimension next to the batch for a contiguous view
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # flatten batch*hidden → (B*H) independent matrices
    left_flat = left_perm.view(-1, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(-1, N, N) # [B*H, N, N]

    out_flat = torch.empty_like(left_flat)

    # strides for A (left)
    stride_ab = left_flat.stride(0)  # batch*hidden stride = N*N
    stride_am = left_flat.stride(1)  # row stride = N
    stride_ak = left_flat.stride(2)  # col stride = 1

    # strides for B (right) – we read it transposed, so:
    #   B[k, n] = right[b, n, k]
    stride_bb = right_flat.stride(0) # batch*hidden stride = N*N
    stride_bk = right_flat.stride(2) # 1  (k moves along original inner dim)
    stride_bn = right_flat.stride(1) # N  (n moves along original outer dim)

    # strides for C (output)
    stride_cb = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    num_mats = left_flat.shape[0]                     # B * H
    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks in M dimension
        triton.cdiv(N, BLOCK_N),   # blocks in N dimension
        num_mats,                  # one block per (batch, hidden) pair
    )

    _tri_mul_out_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (bool/float, may be None)
        - weights      : dict of torch.Tensor model parameters
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy contraction: left @ rightᵀ
    #    Cast to float16 before the Triton kernel for tensor‑core speed.
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    out_h = _batched_matmul_transpose(left_h, right_h)   # [B, N, N, hidden_dim] fp16
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]   # [dim, hidden_dim]
    out = F.linear(out, to_out_weight)        # final shape [B, N, N, dim]

    return out
```
