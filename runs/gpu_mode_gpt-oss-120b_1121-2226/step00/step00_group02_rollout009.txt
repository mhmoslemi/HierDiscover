def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Pointers
    left_base = left_ptr + pid_batch * stride_left_batch
    right_base = right_ptr + pid_batch * stride_right_batch
    out_base = out_ptr + pid_batch * stride_out_batch

    # Load row offsets for left: each row i, column k.
    a_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)

    # Load block from right transposed: we want B_T[k, j] = right[j,k]; so row index k, column index j.
    b_ptrs = right_base + (offs_k[:, None] * stride_right_j) + (offs_j[None, :] * stride_right_k)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for start_k in range(0, N, BLOCK_K):
        cur_k = start_k + offs_k
        k_mask = cur_k < N
        a = tl.load(a_ptrs + start_k * stride_left_k, mask=mask_i[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_ptrs + start_k * stride_right_j, mask=k_mask[:, None] & mask_j[None, :], other=0.0)
        # Actually need to adjust pointer increments for each iteration: we can increment a_ptrs and b_ptrs.

        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_left_k

    # Write out

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # devices and dtype
    device = input_tensor.device

    # Get parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if not nomask:
        mask_unsqueeze = mask.unsqueeze(-1).to(x_norm.dtype)  # broadcast to hidden dim
        left = left * mask_unsqueeze
        right = right * mask_unsqueeze

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Prepare for Triton kernel
    B, N, _, _ = left.shape
    # convert to fp16
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Permute and reshape to batch * hidden_dim
    left_reshaped = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshaped = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    out_reshaped = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # Launch Triton
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides
    stride_left_batch = left_reshaped.stride(0)
    stride_left_i = left_reshaped.stride(1)
    stride_left_k = left_reshaped.stride(2)

    stride_right_batch = right_reshaped.stride(0)
    stride_right_j = right_reshaped.stride(1)
    stride_right_k = right_reshaped.stride(2)

    stride_out_batch = out_reshaped.stride(0)
    stride_out_i = out_reshaped.stride(1)
    stride_out_j = out_reshaped.stride(2)

    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    triton_trimul_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        B * hidden_dim,
        N,
        stride_left_batch, stride_left_i, stride_left_k,
        stride_right_batch, stride_right_j, stride_right_k,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,hidden_dim

    # To_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)  # weight shape dim x hidden_dim

    return out

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute global offsets for i and j
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Range of indices
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # Mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers base for this batch
    left_batch_ptr = left_ptr + pid_batch * stride_left_batch
    right_batch_ptr = right_ptr + pid_batch * stride_right_batch
    out_batch_ptr = out_ptr + pid_batch * stride_out_batch

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_batch_ptr + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_T block: shape (BLOCK_K, BLOCK_N) where B_T[k, j] = right[j, k]
        b_ptrs = right_batch_ptr + (offs_k[:, None] * stride_right_j) + (offs_j[None, :] * stride_right_k)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Dot product
        acc += tl.dot(a, b)

    # Write out
    out_ptrs = out_batch_ptr + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel
==============

Implements the forward pass of the TriMul (outgoing) module used in AlphaFold‑3.
The heavy 4‑D pairwise reduction

    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

is fused into a single Triton kernel that performs a batched matrix
multiplication for every hidden dimension (`hidden_dim`).  All other
operations (LayerNorm, Linear projections, sigmoidal gates) are executed
with standard PyTorch primitives.

The kernel works on a flattened batch dimension `B * hidden_dim` and
computes a GEMM of size (N × N) × (N × N) → (N × N) for each slice.
Mixed‑precision (FP16) is used for the two input tensors while the
accumulator is FP32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the core pairwise reduction (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def _triton_trimul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch, M, N, K,                          # dimensions (batch = B*hidden_dim)
    stride_a_batch, stride_a_m, stride_a_k,  # strides for A (left)
    stride_b_batch, stride_b_k, stride_b_n,  # strides for B (rightᵀ)
    stride_c_batch, stride_c_m, stride_c_n,  # strides for C (output)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ B  where:
        A : (batch, M, K)   (fp16)
        B : (batch, K, N)   (fp16)   (right matrix is already transposed)
        C : (batch, M, N)   (fp32)

    grid = (batch, ceil(M/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid_batch = tl.program_id(0)          # index over (B * hidden_dim)
    pid_m = tl.program_id(1)              # block row index
    pid_n = tl.program_id(2)              # block column index

    # ------------------------------------------------------------------
    # Offsets for the output block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current batch slice
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        # Global K offset for this tile
        cur_k = k * BLOCK_K
        # K‑axis offsets (inside tile)
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch_ptr \
                 + (offs_m[:, None] * stride_a_m) \
                 + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)   # promote to fp32 for accumulation

        # Load B tile (K × N) – note B is already transposed (rightᵀ)
        b_ptrs = b_batch_ptr \
                 + (offs_k[:, None] * stride_b_k) \
                 + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # Rank‑K update
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr \
            + (offs_m[:, None] * stride_c_m) \
            + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) block.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask       : torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights    : dict of model parameters (torch.Tensors)
            - config     : dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Weight shortcuts
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask dtype matches the projections
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM for each hidden slice.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to half‑precision for the kernel (saves bandwidth)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Fold batch and hidden_dim together: (B*hidden_dim, N, N)
    left_mat = left_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Transpose the right matrix so that we can use a standard GEMM
    right_mat_T = right_mat.transpose(-2, -1).contiguous()

    # Output buffer (fp32)
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Strides for the three tensors (batch‑first layout)
    # ------------------------------------------------------------------
    stride_a_b, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_b, stride_b_k, stride_b_n = right_mat_T.stride()
    stride_c_b, stride_c_m, stride_c_n = out_mat.stride()

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _triton_trimul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        B * hidden_dim,
        N,          # M
        N,          # N (output cols)
        N,          # K (reduction dim)
        stride_a_b,
        stride_a_m,
        stride_a_k,
        stride_b_b,
        stride_b_k,
        stride_b_n,
        stride_c_b,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm (hidden_dim) + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim], float32

    return out
