@triton.jit
def trimumul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # batch_hidden_idx = tl.program_id(0)
    batch_hidden_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Decode batch (b) and hidden (h)
    b = batch_hidden_idx // H
    h = batch_hidden_idx % H

    # Starting coordinates
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Create offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # K offsets inside loop
    k_base = 0

    # Compute base pointers
    # left base: b*stride_left_b + h*stride_left_h
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    # right base: b*stride_right_b + h*stride_right_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    # out base:
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k_idx in range(num_k):
        k0 = k_idx * BLOCK_K
        # Offsets for k
        offs_k = k0 + tl.arange(0, BLOCK_K)
        # Mask for some offsets may exceed N
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N
        # Load A block left[b,i,k,h]
        # shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        # When i out of bounds: load zeros
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # Load B block right transposed: B_block shape (BLOCK_K, BLOCK_N) = right[b, k, j, h]
        # right indices: (b, k, j, h)
        b_ptrs = right_base + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_k
        # Note: stride_right_i is same as left_stride_i (i dimension)
        # mask for k and j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # Compute dot
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
    # Store result
    # Write out[b,i,j,h]
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Potential epsilon for layer norm
    eps = 1e-6

    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Normalize input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Project
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present
    if mask is not None and not config.get("nomask", False):
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute output using Triton kernel
    out = triton_tri_mul(left, right, hidden_dim)

    # Post processing
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # output dim: dim

    return out
```

```python
def triton_tri_mul(left: torch.Tensor, right: torch.Tensor, hidden_dim: int):
    # left/right shape: [B,N,N, hidden_dim]
    B, N, _, H = left.shape
    assert H == hidden_dim

    # Output allocate
    out = torch.empty_like(left)  # same dtype and device.

    # Get strides
    stride0 = left.stride(0)  # batch
    stride1 = left.stride(1)  # i
    stride2 = left.stride(2)  # k
    stride3 = left.stride(3)  # h

    # For right we have same stride due to same shape
    stride0_r = right.stride(0)
    stride1_r = right.stride(1)
    stride2_r = right.stride(2)
    stride3_r = right.stride(3)

    stride0_o = out.stride(0)
    stride1_o = out.stride(1)
    stride2_o = out.stride(2)
    stride3_o = out.stride(3)

    # Define block sizes. Choose dynamic based on N.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    _trimumul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride0, stride1, stride2, stride3,
        stride0_r, stride1_r, stride2_r, stride3_r,
        stride0_o, stride1_o, stride2_o, stride3_o,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out
```

Define kernel:

```python
import triton
import triton.language as tl

@triton.jit
def _trimumul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    '''
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    where right is transposed on its first two sequence dimensions.
    '''
    # get program ids
    batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden index
    b = batch_hidden // H
    h = batch_hidden % H

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # coordinate ranges
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # pointer to beginning of our batch+head slice
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # masks for boundaries
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # load left block (B, i, k, h)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right block transposed (B, k, j, h)
        b_ptrs = right_base + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate dot product across k
        acc += tl.dot(a, b)
    # store results
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass.

The heavy O(N³·Cₕ) part of AlphaFold3's TriMul is:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `right` is accessed transposed on its first two
sequence dimensions.  This kernel fuses the three‑nested loop
into a tiled matrix‑multiply using Triton's `dot`.  The surrounding
LayerNorms, Linear projections and gating are performed in
regular PyTorch code.  The kernel works for arbitrary batch size,
sequence length and hidden dimension; it uses mixed‑precision
(float16 inputs, float32 accumulation) on H100 to keep the runtime
well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: computes the batched matrix‑multiply described above.
# ----------------------------------------------------------------------
@triton.jit
def _trimumul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  (batch, i, k, h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (batch, i, k, h) – we will read it as (k, j)
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out   (batch, i, j, h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """ compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] """
    pid_batch_h = tl.program_id(0)          # combined batch * head index
    pid_i = tl.program_id(1)                # tile row   (i)
    pid_j = tl.program_id(2)                # tile col   (j)

    # ------------------------------------------------------------------
    # decode batch index and hidden/head index
    # ------------------------------------------------------------------
    b = pid_batch_h // H
    h = pid_batch_h % H

    # starting coordinates of this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in fp32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over the reduction dimension k in tiles
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # bounds checks
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # left[b, i, k, h]  -> shape (M, K)
        a_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # a is fp16 (input dtype)

        # right[b, j, k, h] transposed as (k, j)
        b_ptrs = right_base + offs_k[:, None] * stride_r_i + offs_j[None, :] * stride_r_k
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)          # b is fp16

        # matrix multiply on this tile; accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # write the result (convert back to fp16 before storing)
    # ------------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the kernel.  Works with fp16 tensors.
# ----------------------------------------------------------------------
def _trimumul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B,N,N, H] tensors of dtype torch.float16 (contiguous)
    returns out: same shape and dtype, computed with the Triton kernel.
    """
    B, N, _, H = left.shape
    # output tensor (fp16) – will be cast back to fp32 later
    out = torch.empty_like(left, dtype=torch.float16)

    # strides as plain Python ints (elements, not bytes)
    sl_b, sl_i, sl_k, sl_h = left.stride()
    sr_b, sr_i, sr_k, sr_h = right.stride()
    so_b, so_i, so_j, so_h = out.stride()

    # tile sizes – tuned for H100 (large shared memory, Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                          # batch * head dimension
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
    )

    _trimumul_kernel[grid](
        left, right, out,
        B, N, H,
        sl_b, sl_i, sl_k, sl_h,
        sr_b, sr_i, sr_k, sr_h,
        so_b, so_i, so_j, so_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data[0] : input tensor          – shape [B, N, N, dim] (float32)
        data[1] : mask tensor (or None) – shape [B, N, N] (float32/bool)
        data[2] : dict of model weights (all torch.float32)
        data[3] : config dict containing "dim" and "hidden_dim"
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # same default as torch.nn.LayerNorm

    # ------------------------------------------------------------------
    # retrieve model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)
    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]     # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # ensure matching dtype & broadcast
        mask_f = mask.to(left.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid -> element‑wise)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³·Cₕ) multiplication – Triton kernel (mixed‑precision)
    # ------------------------------------------------------------------
    # Cast to half for faster Tensor‑Core execution
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Triton kernel produces fp16 output
    out_h = _trimumul_fp16(left_h, right_h)
    # Back to fp32 for the remaining ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output normalisation, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                     # broadcast over hidden dim
    out = F.linear(out, to_out_weight)       # back to original dim

    return out
