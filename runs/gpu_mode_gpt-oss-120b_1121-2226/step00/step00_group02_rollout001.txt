import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_am, stride_ak, stride_an,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # block indices
    b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds in M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointer offsets for C
    c_ptrs = C_ptr + b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < K

        # load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + b * stride_am + offs_m[:, None] * stride_ak + cur_k[None, :] * stride_an
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + b * stride_bb + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b_tile = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b_tile)

    # write back
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward implementation with a fused batched‑GEMM Triton kernel.

Algorithm
---------
1. Layer‑Norm the [B,N,N,C] input.
2. Linear projections → left / right  (no bias).
3. Apply optional pair‑wise mask.
4. Compute three gates (left, right, out) by a linear + sigmoid and multiply.
5. Compute the core Einstein‑sum:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of independent matrix‑multiplications for each hidden
   channel (d).  We reshape the tensors to (B*H, N, N) and launch a
   custom Triton kernel that performs a batched GEMM using
   tiling (BLOCK_M×BLOCK_N tiles, reduction over K in BLOCK_K steps).
   The kernel accumulates in FP32 and stores the result in FP16.
6. Reshape the result back to [B,N,N,H].
7. Apply a second Layer‑Norm, multiply by the out‑gate and a final
   linear projection back to the original channel dimension C.

The heavy O(N³·H) part (step 5) is fully executed on the GPU by the
Triton kernel, achieving high occupancy on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) for tensors of shape
#   A : (batch, M, K)
#   B : (batch, K, N)   (here B is already the transposed “right” tensor)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr,                # ptr to A
    B_ptr,                # ptr to B
    C_ptr,                # ptr to output
    batch, M, N, K,       # dimensions
    stride_Ab, stride_Am, stride_Ak,   # strides of A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,   # strides of C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # program IDs
    # ---------------------------------------------------------------
    b = tl.program_id(0)          # batch index (0 … batch‑1)
    pid_m = tl.program_id(1)      # block row index
    pid_n = tl.program_id(2)      # block column index

    # ---------------------------------------------------------------
    # tile offsets within the current block
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ---------------------------------------------------------------
    # bounds masks
    # ---------------------------------------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---------------------------------------------------------------
    # pointers to the beginning of the current batch plane
    # ---------------------------------------------------------------
    a_batch_ptr = A_ptr + b * stride_Ab
    b_batch_ptr = B_ptr + b * stride_Bb
    c_batch_ptr = C_ptr + b * stride_Cb

    # pointer to the output tile (M,N)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # main K‑loop
    # ---------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_Am + cur_k[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + cur_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # ---------------------------------------------------------------
    # write back (store as FP16 for reduced memory traffic)
    # ---------------------------------------------------------------
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(A: torch.Tensor, B: torch.Tensor,
                       BLOCK_M=64, BLOCK_N=64, BLOCK_K=32) -> torch.Tensor:
    """
    Batched matrix multiplication for FP16 tensors using the Triton kernel.
    A : (B, M, K)   FP16, contiguous (row‑major)
    B : (B, K, N)   FP16, contiguous (row‑major)
    Returns C : (B, M, N) FP16.
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, f"Inner dimensions must match, got {K} and {K2}"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in elements (not bytes)
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_gemm_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_Ab=stride_Ab,
        stride_Am=stride_Am,
        stride_Ak=stride_Ak,
        stride_Bb=stride_Bb,
        stride_Bk=stride_Bk,
        stride_Bn=stride_Bn,
        stride_Cb=stride_Cb,
        stride_Cm=stride_Cm,
        stride_Cn=stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Helper: layer‑norm without torch.nn.Module (weight/bias are 1‑D)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-6) -> torch.Tensor:
    """
    x : (..., C)
    weight, bias : (C,)
    """
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    return x_norm * weight + bias


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (AlphaFold‑3 style).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]  (float32)
        - mask         : torch.Tensor [B, N, N]   (bool or float, may be None)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict containing at least:
            * "dim"          – input channel dimension C
            * "hidden_dim"   – hidden projection dimension H
            * "nomask"       – bool, whether mask must be ignored

    Returns
    -------
    torch.Tensor : [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)      # True → ignore mask

    device = input_tensor.device
    dtype = input_tensor.dtype          # usually torch.float32

    # ------------------------------------------------------------------
    # retrieve parameters – they are stored as 2‑D weight matrices and
    # 1‑D layer‑norm scalars
    # ------------------------------------------------------------------
    # LayerNorm on input
    norm_weight = weights["norm.weight"]          # (C,)
    norm_bias   = weights["norm.bias"]            # (C,)

    # Linear projections (no bias)
    left_proj_weight  = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)

    # Gates (no bias)
    left_gate_weight  = weights["left_gate.weight"]   # (H, C)
    right_gate_weight = weights["right_gate.weight"]  # (H, C)
    out_gate_weight   = weights["out_gate.weight"]    # (H, C)

    # Output LayerNorm + projection
    to_out_norm_weight = weights["to_out_norm.weight"]   # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"]     # (H,)
    to_out_weight      = weights["to_out.weight"]        # (C, H)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(input_tensor, norm_weight, norm_bias)   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP32 → FP16 for the heavy path)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects weight (out, in)
    left  = F.linear(x, left_proj_weight)   # (B,N,N, H)
    right = F.linear(x, right_proj_weight)  # (B,N,N, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # ensure mask dtype matches computation dtype
        mask_f = mask.to(dtype).unsqueeze(-1)               # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,H)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,H)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core Einstein‑sum (batched GEMM) – heavy O(N³·H) part
    #    Transform to (B*H, N, N) and use the Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                       # B may be 1 or 2
    # reshape to (B, H, N, N) then fuse B*H into one batch dimension
    left_t   = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t  = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # transposition needed so that we compute Σ_k left[i,k] * right[j,k]
    right_t = right_t.transpose(2, 3).contiguous()    # (B, H, N, N) with dims (k, j)

    # flatten batch*head
    left_flat  = left_t.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right_t.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # cast to FP16 for the kernel (lossless for our range)
    left_half  = left_flat.to(torch.float16)
    right_half = right_flat.to(torch.float16)

    # Batched GEMM: out_half = left_half @ right_half   (via Triton)
    out_half = _batched_gemm_fp16(left_half, right_half,
                                  BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # back to FP32 and reshape to (B, N, N, H)
    out = out_half.to(torch.float32).view(B, hidden_dim, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()       # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gating, final linear
    # ------------------------------------------------------------------
    out = _layer_norm(out, to_out_norm_weight, to_out_norm_bias)   # (B,N,N,H)
    out = out * out_gate                                           # gating

    # final linear projection back to original dim C
    # weight shape (C, H) – torch.nn.functional.linear expects (out, in)
    out = F.linear(out, to_out_weight)                             # (B,N,N,C)

    return out
```
