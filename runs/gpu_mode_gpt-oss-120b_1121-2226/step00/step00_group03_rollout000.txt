```
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bd = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute batch index and channel index
    total_bd = B * D
    b = pid_bd // D
    d = pid_bd % D

    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_off = k_start + tl.arange(0, BLOCK_K)

        # mask out-of-bound
        i_mask = i_off < N
        j_mask = j_off < N
        k_mask = k_off < N

        # Load left tile (i, k)
        left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d + \
                    i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load right tile (k, j)
        right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d + \
                    k_off[:, None] * stride_right_k + j_off[None, :] * stride_right_j
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # Matrix multiplication accumulate
        acc += tl.dot(left, right)  # shape (BLOCK_I, BLOCK_J)

    # Store result
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + \
               i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])
```

"""
TriMul "outgoing" implementation (AlphaFold‑3 style) with a Triton kernel.

Algorithm
---------
1. Layer‐normalize the input tensor `x` (shape [B, N, N, C]).
2. Linear projections to the hidden dimension `H`:
       left  = x @ W_left  (no bias)
       right = x @ W_right
3. (Optional) Apply the 2‑D mask (shape [B,N,N]) – broadcasted on the hidden dim.
4. Compute three sigmoid gates (left, right, out) from `x` and multiply the gates
   element‑wise into `left` and `right`.
5. Core operation – for every batch b and hidden channel h compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication `L_h @ R_hᵀ` for each hidden channel.
   It is executed by a custom Triton kernel that tiles the (i,j) plane and
   reduces over k in blocks, accumulating in float32.
6. Apply a second LayerNorm on the hidden dimension, multiply the result by the
   output gate and finally a linear projection back to the original dimension.

Only the expensive contraction (step 5) is performed in Triton; everything else
uses regular PyTorch ops for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel for the contraction:  out = Σ_k left[...,i,k,d] * right[...,j,k,d]
# ----------------------------------------------------------------------------- #
@triton.jit
def _trimul_kernel(
    left_ptr,                # const *float32
    right_ptr,               # const *float32
    out_ptr,                 # *float32
    B, N, D,                 # i32 scalar dimensions
    # Strides (i32)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # Compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute a tiled matrix‑multiply across the third axis (k).

    Grid:
        program_id(0) -> combined batch & hidden channel index (b, d)
        program_id(1) -> tile in i‑direction
        program_id(2) -> tile in j‑direction
    """
    pid_bd = tl.program_id(0)
    pid_i  = tl.program_id(1)
    pid_j  = tl.program_id(2)

    # ------------------------------------------------------------------ #
    # Decode batch index b and hidden channel d from pid_bd
    # ------------------------------------------------------------------ #
    total_bd = B * D
    b = pid_bd // D                # batch index
    d = pid_bd % D                 # hidden channel index

    # ------------------------------------------------------------------ #
    # Compute tile offsets for i and j
    # ------------------------------------------------------------------ #
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Masks for out‑of‑bounds (required for non‑multiple of BLOCK sizes)
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Accumulator (float32) – initialise to zero
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------ #
    # Loop over k dimension in blocks
    # ------------------------------------------------------------------ #
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ------------------------------------------------------------------
        # Load left tile   shape (BLOCK_I, BLOCK_K)
        #   indexing: (b, i, k, d)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + d * stride_left_d
            + i_offset[:, None] * stride_left_i
            + k_offset[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Load right tile  shape (BLOCK_K, BLOCK_J)  (transposed view)
        #   indexing: (b, k, j, d)  -> we stream k as the leading dimension
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + d * stride_right_d
            + k_offset[:, None] * stride_right_k
            + j_offset[None, :] * stride_right_j
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Accumulate matrix product over the current k‑block
        # ------------------------------------------------------------------
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------ #
    # Store the computed tile into the output tensor
    # ------------------------------------------------------------------ #
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + d * stride_out_d
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None] & j_mask[None, :],
    )

# ----------------------------------------------------------------------------- #
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------------- #
def _trimul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Perform the contraction over the third axis using Triton.
    `left` / `right` shape: [B, N, N, H]  (float32, contiguous)
    Returns: tensor of the same shape.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, N2, H = left.shape
    assert N == N2, "left/right must be square in the second and third dims"

    out = torch.empty_like(left)

    # Extract strides (in elements, not bytes)
    sbl, sli, slk, sld = left.stride()
    sbr, srj, srk, srd = right.stride()
    sbo, soi, soj, sod = out.stride()

    # Tile size – chosen to give good occupancy on H100.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        B * H,                     # pid 0 : batch * hidden_channel
        triton.cdiv(N, BLOCK_I),   # pid 1 : i‑tile
        triton.cdiv(N, BLOCK_J),   # pid 2 : j‑tile
    )

    _trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        sbl,
        sli,
        slk,
        sld,
        sbr,
        srj,
        srk,
        srd,
        sbo,
        soi,
        soj,
        sod,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # 8 warps works well on H100 for these tile sizes
    )
    return out


# ----------------------------------------------------------------------------- #
# Main entry point required by the problem statement
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask_tensor, weights, config = data
    # Expected shapes:
    # input_tensor : [B, N, N, C]   (C == config["dim"])
    # mask_tensor  : [B, N, N]  (may be ignored if config["nomask"] is True)

    dim = config["dim"]               # input feature dimension C
    hidden_dim = config["hidden_dim"] # projection dimension H
    eps = 1e-5                         # LayerNorm epsilon

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------ #
    # Extract weight tensors (all are assumed to be 2‑D, bias‑free unless noted)
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------ #
    # 1) First LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape: [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2) Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------ #
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------ #
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------ #
    if not config.get("nomask", False):
        # mask shape: [B, N, N] -> [B, N, N, 1] to broadcast
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask
    # If nomask == True we skip masking (implicitly all‑ones)

    # ------------------------------------------------------------------ #
    # 4) Compute sigmoid gates and apply them
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5) Core TriMul contraction via Triton
    # ------------------------------------------------------------------ #
    # Ensure tensors are contiguous for the kernel
    left_contig = left.contiguous()
    right_contig = right.contiguous()
    out = _trimul_torch(left_contig, right_contig)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------ #
    # 6) Second LayerNorm on hidden_dim, gating, final projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # Final linear back to `dim` (bias‑free)
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
