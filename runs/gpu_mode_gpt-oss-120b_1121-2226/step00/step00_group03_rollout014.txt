    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers
        a_ptrs = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)

def custom_kernel(data):
    """
    Custom Triton kernel implementation of the outgoing TriMul operator.

    The function follows the reference PyTorch implementation but
    off‑loads the N×N × N reduction (the core einsum) to a Triton
    batched matrix‑multiply kernel to exploit Tensor‑Core performance
    on H100 GPUs.
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # LayerNorm epsilon

    # ----- LayerNorm on input -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape (B, N, N, dim)

    # ----- Linear projections -----
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----- Apply mask if present -----
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gates -----
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ----- Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- Cast to half precision for the heavy matmul -----
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ----- Rearrange tensors so that the reduction axis is contiguous -----
    # left: (B, N, N, H) -> (B, H, N, N) where the last dim is the reduction axis (k)
    left = left.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, H) -> (B, H, N, N) and then swap the two N dims -> (B, H, K, J)
    right = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, K = left.shape
    # flatten batch and channel dimension
    left_flat = left.reshape(-1, N, K)
    right_flat = right.reshape(-1, K, N)  # already in (K,J) layout after permute

    # ----- Triton batched matmul -----
    out_flat = triton_batch_matmul(left_flat, right_flat)  # (B*H, N, N) fp32

    # ----- Reshape back to (B, N, N, H) -----
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----- Output layer norm and final linear -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate  # broadcast over last dim
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)

    return out

"""
TriMul – outgoing version (AlphaFold3)

This implementation follows the reference PyTorch code but accelerates the
core O(N³·C_hidden) operation with a custom Triton kernel.  The heavy
computation

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is a batched matrix‑multiply over the reduction dimension *k* for each
(batch, hidden) pair.  We fuse this by reshaping the tensors to
(batch*hidden, N, N) and launching a Triton kernel that performs a
standard batched GEMM (A @ Bᵀ) using Tensor‑Core friendly 16‑bit arithmetic.
All other steps (LayerNorm, linear projections, sigmoids, final Linear)
are executed with PyTorch for simplicity.

The kernel works for any batch size ≤ 2, sequence length up to 1024 and
hidden dimension up to 384 (as required by the test matrix).  The final
output is returned as a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C[b,i,j] = Σₖ A[b,i,k] * B[b,k,j]
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr,                    # *const half*   A pointer   (batch, M, K)
    b_ptr,                    # *const half*   B pointer   (batch, K, N)
    c_ptr,                    # *float*        C pointer   (batch, M, N)

    BATCH, M, N, K,           # static problem sizes
    stride_az, stride_ai, stride_ak,   # strides for A (batch, row, col)
    stride_bz, stride_bk, stride_bj,   # strides for B (batch, row(K), col(N))
    stride_cz, stride_ci, stride_cj,   # strides for C (batch, row, col)

    BLOCK_M: tl.constexpr,    # tile size in the M dimension
    BLOCK_N: tl.constexpr,    # tile size in the N dimension
    BLOCK_K: tl.constexpr,    # tile size in the reduction dimension
):
    # --------------------------------------------------------------
    # program IDs
    #   pid_z : batch index (0 .. BATCH-1)
    #   pid_m : tile row   (0 .. ceil(M/BLOCK_M)-1)
    #   pid_n : tile col   (0 .. ceil(N/BLOCK_N)-1)
    # --------------------------------------------------------------
    pid_z = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # --------------------------------------------------------------
    # Offsets within the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # Accumulator in FP32 (TensorCore dot returns FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute pointers for the current tile
        a_ptrs = (
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + pid_z * stride_bz
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bj
        )

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # half → fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # half → fp16

        # TensorCore dot product (fp16 x fp16 → fp32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def triton_batch_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A : (B*H, N, K)   fp16
    B : (B*H, K, N)   fp16   (K == N)
    Returns C : (B*H, N, N)   fp32
    """
    assert A.shape == B.shape, "A and B must have identical shape"
    batch, N, K = A.shape
    assert N == K, "The kernel expects square matrices (K == N)"

    # Output in float32 for downstream LayerNorm / Linear
    C = torch.empty_like(A, dtype=torch.float32, device=A.device)

    # Strides in number of elements (not bytes)
    stride_a_batch, stride_a_row, stride_a_col = A.stride()
    stride_b_batch, stride_b_row, stride_b_col = B.stride()
    stride_c_batch, stride_c_row, stride_c_col = C.stride()

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batch_matmul_kernel[grid](
        A,
        B,
        C,
        batch,
        N,
        N,
        K,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a Triton‑accelerated
    N³ reduction.  Input:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # LayerNorm epsilon (matches torch.nn.LayerNorm default)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape = [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating networks
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy multiplication
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Rearrange so that the reduction axis (k) is contiguous:
    #    left  : (B,N,N, H) -> (B, H, N, K) where K == N
    #    right : (B,N,N, H) -> (B, H, K, N) after swapping N dims
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, K)
    right = right.permute(0, 3, 2, 1).contiguous()        # (B, H, K, N)

    B, H, N, K = left.shape   # K == N

    # Flatten batch & channel dimension for the kernel
    left_flat = left.reshape(-1, N, K)    # (B*H, N, K)
    right_flat = right.reshape(-1, K, N)  # (B*H, K, N)

    # ------------------------------------------------------------------
    # 7. Triton batched GEMM – core of the TriMul operation
    # ------------------------------------------------------------------
    out_flat = triton_batch_matmul(left_flat, right_flat)  # (B*H, N, N) FP32

    # Restore original layout (B, N, N, H)
    out = out_flat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate                      # broadcast over last dim
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)

    return out
