"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the feature dimension (dim).
2. Linear projections (left/right) and three gating linear layers.
   * mask (if provided) is applied to left/right before gating.
   * Gating values are obtained with a sigmoid and applied element‑wise.
3. The core “triangle multiplication”:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for each head d:
        out_d = left_d @ right_dᵀ   where left_d has shape (B,N,N)
   The operation is performed by a custom Triton kernel that
   treats each (batch, head) pair as an independent GEMM of size N×N.
4. Layer‑norm on the hidden dimension, an output‑gate (sigmoid) and a final
   linear projection back to the original feature dimension.

Only the heavy N³ multiplication (step 3) is handled by Triton; the remaining
operations use standard PyTorch kernels for simplicity and numerical fidelity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr,                # left   [B, N, N, H]  (float16)
    b_ptr,                # right  [B, N, N, H]  (float16)
    c_ptr,                # out    [B, N, N, H]  (float32)
    B, N, H,
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides for left
    stride_bb, stride_bj, stride_bk, stride_bh,   # strides for right
    stride_cb, stride_ci, stride_cj, stride_ch,   # strides for out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids ---------------------------------------------------------------
    pid_batch_head = tl.program_id(0)   # (batch * H)
    pid_m = tl.program_id(1)           # tile row (i)
    pid_n = tl.program_id(2)           # tile col (j)

    # decode batch and head indices
    batch = pid_batch_head // H
    head  = pid_batch_head %  H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for the current (batch, head)
    a_base = a_ptr + batch * stride_ab + head * stride_ah
    b_base = b_ptr + batch * stride_bb + head * stride_bh
    c_base = c_ptr + batch * stride_cb + head * stride_ch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        cur_k = k_iter * BLOCK_K + offs_k
        mask_k = cur_k < N

        # ----- load a tile ----------------------------------------------------
        # a[b,i,k,head]
        a_ptrs = a_base + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # ----- load b tile (transposed on the two N axes) --------------------
        # we need right[b,j,k,head]  →  treat as (k,j) for the GEMM
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_j[None, :] * stride_bj
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        # ----- accumulate ------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ----- write back ----------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_ci + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched triangle multiplication (outgoing version) using the Triton kernel.
    `left` and `right` must be half‑precision tensors of shape [B, N, N, H].
    Returns a float‑32 tensor of the same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # output in float32
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * H,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # strides (int64) – PyTorch returns Python ints, they are fine for Triton
    a_strides = left.stride()
    b_strides = right.stride()
    c_strides = out.stride()

    _tri_mul_outgoing_kernel[
        grid
    ](
        left,
        right,
        out,
        B,
        N,
        H,
        a_strides[0],
        a_strides[1],
        a_strides[2],
        a_strides[3],
        b_strides[0],
        b_strides[1],
        b_strides[2],
        b_strides[3],
        c_strides[0],
        c_strides[1],
        c_strides[2],
        c_strides[3],
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool / 0‑1) or None
        - weights      : dict of model weights (torch.Tensor)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor  [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1. LayerNorm on the input (over the feature dimension)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 3. Optional mask
    # --------------------------------------------------------------------- #
    use_mask = not config.get("nomask", False)
    if use_mask and mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid) and application
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # used later

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. Triangle multiplication (the N^3 part) – fused Triton kernel
    # --------------------------------------------------------------------- #
    # Cast to half‑precision for the kernel (common in AlphaFold)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Core GEMM: out_h shape [B, N, N, hidden_dim] float32
    out = _tri_mul_outgoing(left_h, right_h)

    # --------------------------------------------------------------------- #
    # 6. Post‑processing: LayerNorm → out‑gate → final linear
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return out
