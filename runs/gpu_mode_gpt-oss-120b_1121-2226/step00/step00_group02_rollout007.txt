@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)  # batch index
    pid_y = tl.program_id(1)  # block row i
    pid_x = tl.program_id(0)  # block col j
    
    # Offsets
    offs_m = pid_y * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_x * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # Masks for OOB
    mask_m = offs_m < M
    mask_n = offs_n < N
    
    # Pointers base for this batch
    left_batch_ptr = left_ptr + pid_z * stride_left_batch
    right_batch_ptr = right_ptr + pid_z * stride_right_batch
    out_batch_ptr = out_ptr + pid_z * stride_out_batch
    
    # Pointers with offsets
    a_ptrs = left_batch_ptr + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
    b_ptrs = right_batch_ptr + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Loop over K dim
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K - k * BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K - k * BLOCK_K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k
    
    # Write back to memory
    out_ptrs = out_batch_ptr + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # maybe also get eps for layernorm
    eps = 1e-6
    # compute norm
    x = input_tensor
    # LayerNorm (dim last)
    # Compute mean and var across dim (last dim)
    # implement manual: 
    #   mean = x.mean(-1, keepdim=True)
    #   var = ((x - mean) ** 2).mean(-1, keepdim=True)
    #   x_norm = (x - mean) / torch.sqrt(var + eps) * norm_weight + norm_bias
    #   (cost small)
    x_norm = (x - x.mean(-1, keepdim=True)) / torch.sqrt(x.var(-1, unbiased=False, keepdim=True) + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
    # compute proj and gates
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # mask
    if mask is not None:
        # Broadcast mask over last dim
        mask_unsqueezed = mask.unsqueeze(-1)
        left = left * mask_unsqueezed
        right = right * mask_unsqueezed
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # reshape for batched matmul
    B, N, _, _ = left.shape
    # flatten batch*hidden into one dimension
    left_flat = left.permute(0,3,1,2).contiguous()  # shape B, hidden, N, N
    right_flat = right.permute(0,3,1,2).contiguous()
    batch_total = B * hidden_dim
    left_flat = left_flat.view(batch_total, N, N)
    right_flat = right_flat.view(batch_total, N, N)
    # allocate out_flat
    out_flat = torch.empty_like(left_flat)
    # launch triton kernel
    import triton
    import triton.language as tl
    # compute grid sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_blocks_n, num_blocks_m, batch_total)  # (pid_x, pid_y, pid_z)
    tri_mul_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape out
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()
    # apply to_out_norm (LayerNorm on hidden_dim)
    out_norm = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    # multiply out_gate (shape B,N,N,hidden_dim)
    out_norm = out_norm * out_gate
    # final linear
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out_final

    stride_right_batch, stride_right_j, stride_right_k,

"""
TriMul (outgoing) forward implementation with a Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor over the channel dimension `dim`.
2. Linear projections (`left_proj`, `right_proj`) and three gating linear layers.
3. Apply the optional mask to the projected tensors.
4. Multiply projected tensors by their respective sigmoid‑gates.
5. Fuse the heavy O(N³) reduction:
   - reshape tensors to (B*hidden, N, N) → a batch of independent matmuls.
   - each batched matmul computes  out[b,d,i,j] = Σₖ left[b,d,i,k] * right[b,d,j,k]
     i.e. a per‑channel matrix multiplication `left @ rightᵀ`.
   - performed by the Triton kernel `tri_mul_matmul_kernel`.
6. Reshape back to (B, N, N, hidden) and apply a second LayerNorm.
7. Multiply by the output gate and a final linear projection back to `dim`.

Only step 5 is done in Triton; the remaining ops use PyTorch (they are
light‑weight compared to the N³ bottleneck).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,          # *float32   left  tensor [B*H, N, N]
    right_ptr,         # *float32   right tensor [B*H, N, N]
    out_ptr,           # *float32   output tensor [B*H, N, N]
    M, N, K,           # matrix dimensions (all equal to seq_len)
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_j, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Perform `out = left @ rightᵀ` for each batch (`pid_z`).
    left, right, out have shape (batch, M, K) and (batch, N, K) respectively.
    """
    pid_z = tl.program_id(2)   # batch index (B*hidden)
    pid_y = tl.program_id(1)   # block row index (i)
    pid_x = tl.program_id(0)   # block col index (j)

    # --------------------------------------------------------------
    # block offsets
    # --------------------------------------------------------------
    offs_m = pid_y * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_x * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    left_batch = left_ptr + pid_z * stride_left_batch
    right_batch = right_ptr + pid_z * stride_right_batch
    out_batch = out_ptr + pid_z * stride_out_batch

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension K
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_offset = kb * BLOCK_K
        # -------- load a block from left (M x K) ----------
        a_ptrs = left_batch \
                 + (offs_m[:, None] * stride_left_m) \
                 + ((k_offset + offs_k)[None, :] * stride_left_k)
        a_mask = mask_m[:, None] & ((k_offset + offs_k) < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # -------- load a block from right (N x K) ----------
        # we need rightᵀ => shape (K x N)
        b_ptrs = right_batch \
                 + ((k_offset + offs_k)[:, None] * stride_right_k) \
                 + (offs_n[None, :] * stride_right_j)
        b_mask = ((k_offset + offs_k) < K)[:, None] & mask_n[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # -------- dot product accumulation ----------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # write the block back to global memory
    # --------------------------------------------------------------
    out_ptrs = out_batch \
               + (offs_m[:, None] * stride_out_m) \
               + (offs_n[None, :] * stride_out_n)
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # convenience aliases for weight tensors
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Step 1: LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm normalises over the last dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Step 2: Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)      # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_w)    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Step 3: Optional mask
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Step 4: Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # Step 5: Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Step 6: Triton‑fused reduction (batched matmul)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B, hidden, N, N) -> (B*hidden, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_flat = torch.empty_like(left_flat)   # float32 output

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        (N + BLOCK_N - 1) // BLOCK_N,          # program_id(0) – column blocks
        (N + BLOCK_M - 1) // BLOCK_M,          # program_id(1) – row blocks
        B * hidden_dim,                        # program_id(2) – batch (B * hidden)
    )

    # strides for the flattened tensors
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Step 7: LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Step 8: Output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate                        # broadcasting over hidden dim
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]

    return out
