def mask_gates_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    BLOCK: tl.constexpr,
):
    # grid: total = B*N*N*H
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < B * N * N * H

    # compute coordinates
    idx = offs

    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    k = idx // H
    h = idx % H

    # compute addresses
    left_offset = b * stride_left_b + i * stride_left_i + k * stride_left_k + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    left_gate_offset = b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + h * stride_lg_h
    right_gate_offset = b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + h * stride_rg_h

    mask_offset = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # Load values
    l = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    r = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + right_gate_offset, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + left_offset, l, mask=mask)
    tl.store(right_ptr + right_offset, r, mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    B, N, H,
    BLOCK: tl.constexpr,
):
    """In‑place fusion of mask, left‑gate and right‑gate.
    left, right, left_gate, right_gate have shape (B,N,N,H).
    mask has shape (B,N,N) and is broadcast over H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    # compute mask for out‑of‑bounds
    active = offs < B * N * N * H

    # flatten index into (b,i,k,h)
    idx = offs

    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    k = idx // H
    h = idx % H

    # pointer offsets for each tensor
    left_off = b * stride_left_b + i * stride_left_i + k * stride_left_k + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    lg_off = b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + h * stride_lg_h
    rg_off = b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + h * stride_rg_h
    mask_off = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k

    # load
    l = tl.load(left_ptr + left_off, mask=active, other=0.0)
    r = tl.load(right_ptr + right_off, mask=active, other=0.0)
    lg = tl.load(left_gate_ptr + lg_off, mask=active, other=0.0)
    rg = tl.load(right_gate_ptr + rg_off, mask=active, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=active, other=0.0)

    # fused multiplication
    l = l * lg * m
    r = r * rg * m

    # store back
    tl.store(left_ptr + left_off, l, mask=active)
    tl.store(right_ptr + right_off, r, mask=active)

def custom_kernel(data):
    """
    Custom forward pass for the outgoing TriMul operator.
    Implements:
        - LayerNorm over input
        - Linear projections (left, right)
        - Gating linear + sigmoid
        - Fusion of mask and gates via a Triton kernel
        - Efficient pairwise multiplication using batched GEMM (via torch.matmul)
        - Output LayerNorm, gating and final linear projection.
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # If mask is omitted (nomask) we treat it as all‑ones
    if mask_tensor is None:
        mask_tensor = torch.ones(input_tensor.shape[0],
                                 input_tensor.shape[1],
                                 input_tensor.shape[2],
                                 device=input_tensor.device,
                                 dtype=input_tensor.dtype)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gates + sigmoid
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    H = hidden_dim

    # ensure contiguous layout for kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask_tensor.contiguous()

    total_elements = B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    apply_mask_gate_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        B, N, H,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Pairwise multiplicative update: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as batched matrix multiplication.
    # ------------------------------------------------------------------
    # left: (B,N,N,H) -> (B,H,N,N) with axes (i,k)
    left_perm = left.permute(0, 3, 1, 2)    # (B,H,i,k)
    # right: (B,N,N,H) -> (B,H,k,j)
    right_perm = right.permute(0, 3, 2, 1)  # (B,H,k,j)

    # batch‑gemm on (i,k)·(k,j) -> (i,j) for each head H
    out = torch.matmul(left_perm, right_perm)   # (B,H,i,j)

    # bring back to original order (B,i,j,H)
    out = out.permute(0, 2, 3, 1)   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # Apply out_gate (broadcasted over hidden dim)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) forward pass.

The module normalises the input, projects it to a hidden dimension, and
applies three sigmoidal gates.  A small Triton kernel fuses the mask
multiplication together with the two gates.  The core “pairwise
multiplicative update”

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is implemented as a batched GEMM (torch.matmul) after permuting the
axes:

    left  -> (B, H, I, K)   where I=i, K=k
    right -> (B, H, K, J)   where J=j

The result is reshaped back to (B, I, J, H), normalised, gated once
more and projected back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    # strides for left (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (B,N,N,H)
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for left_gate (B,N,N,H)
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    # strides for right_gate (B,N,N,H)
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    # strides for mask (B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_k,
    B, N, H,
    BLOCK: tl.constexpr,
):
    """In‑place mask & gate fusion.
    left, right, left_gate, right_gate : (B,N,N,H)
    mask                                   : (B,N,N)  (broadcast over H)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)

    # active mask for out‑of‑bounds elements
    active = offs < B * N * N * H

    # ---- decode linear index into (b,i,k,h) ----
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    k = idx // H
    h = idx % H

    # ---- compute pointer offsets ------------------------------------
    left_off  = b * stride_left_b  + i * stride_left_i  + k * stride_left_k  + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + k * stride_right_k + h * stride_right_h
    lg_off    = b * stride_lg_b    + i * stride_lg_i    + k * stride_lg_k    + h * stride_lg_h
    rg_off    = b * stride_rg_b    + i * stride_rg_i    + k * stride_rg_k    + h * stride_rg_h
    mask_off  = b * stride_mask_b  + i * stride_mask_i  + k * stride_mask_k

    # ---- load, compute, store ---------------------------------------
    l  = tl.load(left_ptr  + left_off,  mask=active, other=0.0)
    r  = tl.load(right_ptr + right_off, mask=active, other=0.0)
    lg = tl.load(left_gate_ptr  + lg_off,    mask=active, other=0.0)
    rg = tl.load(right_gate_ptr + rg_off,   mask=active, other=0.0)
    m  = tl.load(mask_ptr + mask_off,        mask=active, other=0.0)

    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr  + left_off,  l, mask=active)
    tl.store(right_ptr + right_off, r, mask=active)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.
    `data` is a tuple (input, mask, weights, config).

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Fuse mask and gates (Triton) – skip entirely if nomask
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    H = hidden

    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    if not nomask:
        # ensure mask is float and on the correct device
        mask = mask_tensor.to(dtype=input_tensor.dtype, device=input_tensor.device)
        mask = mask.contiguous()

        total = B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)

        _apply_mask_gate_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            B, N, H,
            BLOCK=BLOCK,
        )
    else:
        # simple PyTorch fusion when mask is absent
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Pairwise multiplicative update via batched GEMM
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B, H, I, K) and (B, H, K, J)
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1) # (B, H, K, J)

    # batched matmul on the last two axes
    out = torch.matmul(left_perm, right_perm)  # (B, H, I, J)

    # restore original order (B, I, J, H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + out_gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
