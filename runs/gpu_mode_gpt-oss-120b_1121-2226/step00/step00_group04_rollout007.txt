def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # convert mask to float, shape [bs,N,N]
    if mask is None: # not possible as they pass mask maybe torch.Tensor
        mask = torch.ones_like(input_tensor[...,0])
    mask_expand = mask.unsqueeze(-1).to(input_tensor.dtype)  # [B,N,N,1]
    # Norm
    eps = 1e-6
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Mask multiplication
    left = left * mask_expand
    right = right * mask_expand

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for speed
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Triton kernel for einsum
    out = tri_mul_kernel(
        left, right,
        B=input_tensor.shape[0], N=input_tensor.shape[1], H=hidden_dim,
        BLOCK_M=64, BLOCK_N=64, BLOCK_K=64,
    )
    # out shape [B,N,N,hidden_dim] (float16)
    out = out.to(torch.float32)

    # to_out_norm: layernorm across hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-6,
    )
    out = out * out_gate  # out_gate broadcast.
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

@triton.jit
def tri_mul_kernel(lhs_ptr, rhs_ptr, out_ptr,
                   B, N, H,
                   stride_l0, stride_l1, stride_l2, stride_l3,
                   stride_r0, stride_r1, stride_r2, stride_r3,
                   stride_o0, stride_o1, stride_o2, stride_o3,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(axis=0)
    pid_batch_hidden = tl.program_id(axis=1)

    # Compute i and j block indices from pid
    num_block_n = tl.cdiv(N, BLOCK_N)
    i_block = pid // num_block_n
    j_block = pid % num_block_n

    # Compute batch and hidden index from pid_batch_hidden
    batch = pid_batch_hidden // H
    d = pid_batch_hidden % H

    # Offsets for output
    i_offset = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create masks for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Compute output pointer base for given batch and hidden
    out_ptrs = out_ptr + batch * stride_o0 + d * stride_o3 + i_offset[:, None] * stride_o1 + j_offset[None, :] * stride_o2

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

"""
TriMul “outgoing” forward pass (AlphaFold‑3) implemented with a fused
Triton kernel for the costly N×N×N einsum.

Algorithm
---------

1. Layer‑norm the 4‑D input tensor (shape [B,N,N,C]).
2. Two linear projections (no bias) produce “left” and “right” tensors of
   shape [B,N,N,H] where H = hidden_dim.
3. An optional mask (broadcast on H) is applied to both tensors.
4. Three gate‑linear layers produce sigmoid‑gated scalars of the same
   shape; left and right are multiplied by their respective gates.
5. The core operation
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton kernel that schedules a batched
   matrix‑multiply (M×K)·(K×N) for each (b,h) pair.
   The kernel works in FP16 for the inputs and accumulates in FP32.
6. The result (still in H‑dim) is layer‑normed, gated by the out‑gate,
   and linearly projected back to the original channel dimension C.

Only the heavy N³ contraction is done in Triton; all other ops rely on
highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel:  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    lhs_ptr,                # const half*   left  tensor
    rhs_ptr,                # const half*   right tensor
    out_ptr,                #       float*  output tensor
    B, N, H,                # problem size
    stride_l0, stride_l1, stride_l2, stride_l3,   # left  strides (b,i,k,h)
    stride_r0, stride_r1, stride_r2, stride_r3,   # right strides (b,j,k,h)
    stride_o0, stride_o1, stride_o2, stride_o3,   # out   strides (b,i,j,h)
    BLOCK_M: tl.constexpr,  # tile size in i direction
    BLOCK_N: tl.constexpr,  # tile size in j direction
    BLOCK_K: tl.constexpr,  # tile size in k direction
):
    # --------------------------------------------------------------
    # Grid layout
    #   axis 0 : i‑block × j‑block   (flattened)
    #   axis 1 : (batch, hidden)      – each program works on a single (b,h)
    # --------------------------------------------------------------
    pid = tl.program_id(axis=0)          # i‑block * num_j_blocks + j‑block
    pid_bh = tl.program_id(axis=1)       # b * H + h

    # decode (i_block, j_block)
    num_j_blocks = tl.cdiv(N, BLOCK_N)
    i_block = pid // num_j_blocks
    j_block = pid % num_j_blocks

    # decode (batch, hidden)
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # Offsets for the current 2‑D tile
    # ------------------------------------------------------------------
    i_range = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j_range = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # pointer to output tile (M×N)
    out_ptrs = (
        out_ptr
        + b * stride_o0
        + h * stride_o3
        + i_range[:, None] * stride_o1
        + j_range[None, :] * stride_o2
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the contracted dimension k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_range = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left tile: (M × K)
        lhs_ptrs = (
            lhs_ptr
            + b * stride_l0
            + h * stride_l3
            + i_range[:, None] * stride_l1
            + k_range[None, :] * stride_l2
        )
        lhs = tl.load(
            lhs_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # convert half → float32 for accumulation

        # right tile needs to be transposed to (K × N)
        rhs_ptrs = (
            rhs_ptr
            + b * stride_r0
            + h * stride_r3
            + k_range[:, None] * stride_r2   # K‑stride first
            + j_range[None, :] * stride_r1   # N‑stride second
        )
        rhs = tl.load(
            rhs_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate M×N block
        acc += tl.dot(lhs, rhs)        # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # Store the tile
    # ------------------------------------------------------------------
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_einsum(lhs: torch.Tensor, rhs: torch.Tensor,
                    BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 64) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    lhs / rhs : [B, N, N, H]  (dtype = torch.float16)
    returns   : [B, N, N, H]  (dtype = torch.float32)
    """
    assert lhs.is_contiguous() and rhs.is_contiguous()
    B, N, _, H = lhs.shape

    # allocate output in FP32 (accumulated result)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=lhs.device)

    # strides are expressed in element counts (independent of dtype)
    stride_l0, stride_l1, stride_l2, stride_l3 = lhs.stride()
    stride_r0, stride_r1, stride_r2, stride_r3 = rhs.stride()
    stride_o0, stride_o1, stride_o2, stride_o3 = out.stride()

    # Grid definition
    grid_x = ( (N + BLOCK_M - 1) // BLOCK_M ) * ( (N + BLOCK_N - 1) // BLOCK_N )
    grid_y = B * H
    _tri_mul_kernel[
        (grid_x, grid_y)
    ](
        lhs,
        rhs,
        out,
        B,
        N,
        H,
        stride_l0,
        stride_l1,
        stride_l2,
        stride_l3,
        stride_r0,
        stride_r1,
        stride_r2,
        stride_r3,
        stride_o0,
        stride_o1,
        stride_o2,
        stride_o3,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # modest warp count – works well on H100
    )
    return out


# --------------------------------------------------------------
# Public entry point used by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype = float32)
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Prepare mask (optional)
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        mask = mask_tensor.to(dtype)

    # Expand mask for broadcasting on hidden dim later
    mask_exp = mask.unsqueeze(-1)                           # [B,N,N,1]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Mask application (broadcast on hidden)
    # ------------------------------------------------------------------
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul contraction (FP16 for compute, FP32 accumulator)
    # ------------------------------------------------------------------
    # Cast to half‑precision before the heavy contraction
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Triton kernel does the Σₖ left * right
    out_h = _tri_mul_einsum(left_h, right_h, BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Convert to FP32 for the following LayerNorm
    out = out_h.to(torch.float32)                         # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Output normalisation, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                                 # gate in FP32

    # Final linear back to original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
