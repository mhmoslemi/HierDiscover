"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Project x to a hidden dimension H with two linear layers (left/right).
3. Apply an optional pair mask (broadcast over H) and gating scalars
   (left_gate, right_gate, out_gate) obtained by linear + sigmoid.
4. Core operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]          (H‑dim stays)
   This is a “tri‑multiplicative” update and is performed by a custom
   Triton kernel.  The kernel tiles the (i, j) output plane with
   BLOCK_I×BLOCK_J threads, iterates over k in BLOCK_K‑sized chunks and
   over the hidden dimension in BLOCK_H‑sized chunks, accumulating a
   (BLOCK_I, BLOCK_J, BLOCK_H) accumulator with 32‑bit precision.
5. Layer‑norm the result over the hidden dimension, gate it with out_gate,
   and finally linearly project back to the original channel dimension C.
6. Return a tensor of shape (B, N, N, C).

The implementation uses torch for the cheap linear / norm / sigmoid work
and a Triton kernel for the O(N³·H) multiplication, which is the dominant
cost.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,      # const float*   left  (N×N×H)
    right_ptr,     # const float*   right (N×N×H)
    out_ptr,       # float*         out   (N×N×H)
    N, H,          # i32           sequence length, hidden dim
    # strides (in elements, not bytes)
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 2‑D tiling of the (i, j) output plane
    # --------------------------------------------------------------
    num_pid_j = tl.cdiv(N, BLOCK_J)        # tiles along j
    i_tile = pid // num_pid_j
    j_tile = pid %  num_pid_j

    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # --------------------------------------------------------------
    # Loop over the hidden dimension in chunks
    # --------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        h_offset = h_start + tl.arange(0, BLOCK_H)
        h_mask   = h_offset < H

        # accumulator for this chunk
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # ----------------------------------------------------------
        # Loop over the k dimension (the summation axis)
        # ----------------------------------------------------------
        for k_start in range(0, N, BLOCK_K):
            k_offset = k_start + tl.arange(0, BLOCK_K)
            k_mask   = k_offset < N

            # -----------------------------
            # Load left[i, k, h]  (I×K×H)
            # -----------------------------
            left_ptrs = left_ptr + \
                i_offset[:, None, None] * stride_left_i + \
                k_offset[None, :, None] * stride_left_k + \
                h_offset[None, None, :] * stride_left_h
            left = tl.load(
                left_ptrs,
                mask=(i_mask[:, None, None] &
                      k_mask[None, :, None] &
                      h_mask[None, None, :]),
                other=0.0)

            # -----------------------------
            # Load right[j, k, h] (J×K×H)
            # -----------------------------
            right_ptrs = right_ptr + \
                j_offset[:, None, None] * stride_right_i + \
                k_offset[None, :, None] * stride_right_k + \
                h_offset[None, None, :] * stride_right_h
            right = tl.load(
                right_ptrs,
                mask=(j_mask[:, None, None] &
                      k_mask[None, :, None] &
                      h_mask[None, None, :]),
                other=0.0)

            # ----------------------------------------------
            # Accumulate Σ_k left[i,k,h] * right[j,k,h]
            #   left  : (I, K, H)
            #   right : (J, K, H)
            #   result: (I, J, H)
            # ----------------------------------------------
            for kk in range(0, BLOCK_K):
                # left_vec shape (I, H), right_vec shape (J, H)
                l = left[:, kk, :]          # (I, H)
                r = right[:, kk, :]         # (J, H)
                # broadcast multiply → (I, J, H)
                acc += l[:, None, :] * r[None, :, :]

        # ----------------------------------------------------------
        # Store the computed chunk back to global memory
        # ----------------------------------------------------------
        out_ptrs = out_ptr + \
            i_offset[:, None, None] * stride_out_i + \
            j_offset[None, :, None] * stride_out_j + \
            h_offset[None, None, :] * stride_out_h
        tl.store(
            out_ptrs,
            acc,
            mask=(i_mask[:, None, None] &
                  j_mask[None, :, None] &
                  h_mask[None, None, :])
        )


# ----------------------------------------------------------------------
# Entry point expected by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]           # C (input & output channel dim)
    hidden_dim = config["hidden_dim"]   # H

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (LayerNorm over the last dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after linear: [B, N, N, hidden_dim]
    left  = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_weight)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Apply pair mask (if any) – broadcast over hidden_dim
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()  # (B,N,N,H)
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()    # (B,N,N,H)

    left  = left  * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5) Core tri‑multiplicative update using Triton
    # ------------------------------------------------------------------
    # Allocate output buffer for the hidden‑dim tensor
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=left.dtype,
                             device=left.device)

    # Tile sizes – good trade‑off between register usage and parallelism
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32
    BLOCK_H = 32

    for b in range(B):
        left_b  = left[b]          # (N, N, H)
        right_b = right[b]         # (N, N, H)
        out_b   = out_hidden[b]    # (N, N, H)

        # Strides in *elements* (as returned by torch)
        s_li, s_lk, s_lh = left_b.stride()
        s_ri, s_rk, s_rh = right_b.stride()
        s_oi, s_oj, s_oh = out_b.stride()

        # Launch a 1‑D grid where each program instance computes a (i,j) tile
        grid = ( ( (N + BLOCK_I - 1) // BLOCK_I ) *
                 ( (N + BLOCK_J - 1) // BLOCK_J ), )

        tri_mul_einsum_kernel[grid](
            left_b.data_ptr(),
            right_b.data_ptr(),
            out_b.data_ptr(),
            N, hidden_dim,
            s_li, s_lk, s_lh,
            s_ri, s_rk, s_rh,
            s_oi, s_oj, s_oh,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 6) Output layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 7) Apply the output gate and final projection back to C
    # ------------------------------------------------------------------
    out = out * out_gate                      # gate (B,N,N,H)
    output = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,C)

    return output
