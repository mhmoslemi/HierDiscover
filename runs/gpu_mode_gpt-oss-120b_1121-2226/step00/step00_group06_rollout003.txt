@triton.jit
def tri_mul_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute batch and hidden indices
    pid = tl.program_id(0)  # flattened batch*h index
    batch = pid // H
    hidden = pid % H

    # Offsets for i and j tiles
    pid_i = tl.program_id(1) # block index in M dimension
    pid_j = tl.program_id(2) # block index in N dimension

    # Compute offsets
    # Use range for block's rows (i) and columns (j)
    block_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    block_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-range
    i_mask = block_i < N
    j_mask = block_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # for each k block
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offset for k dimension
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left: shape (BLOCK_M, BLOCK_K)
        left_offset = (
            batch * stride_left_b +
            block_i[:, None] * stride_left_i +
            k_offset[None, :] * stride_left_k +
            hidden * stride_left_h
        )
        left = tl.load(left_ptr + left_offset, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        # Left shape: (M, K)

        # Load right: shape (BLOCK_N, BLOCK_K) we need right[j,k,d] (j row, k col). But our earlier transposition: we need right transposed such that we multiply left[i,k] * right[j,k] and sum over k -> we can load right as (BLOCK_N, BLOCK_K) not transposed. So use same offset for j and k.
        right_offset = (
            batch * stride_right_b +
            block_j[:, None] * stride_right_i +   # careful: right's first spatial dimension is i? Actually right shape as (batch,N,N,H) where indices (i,k,d). Our “j” corresponds to the first index of right? Wait right's dimension ordering is same as left: (i,k,d). The output wants right[j,k,d] where j corresponds to first index (i). So treat j as i index for right. So we can access right via block_j as first index.
            k_offset[None, :] * stride_right_k +
            hidden * stride_right_h
        )
        right = tl.load(right_ptr + right_offset, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        # right shape (N, K)

        # Multiply and accumulate
        acc += tl.dot(left, right)  # shape (M,N) result

    # Write out
    out_offset = (
        batch * stride_out_b +
        block_i[:, None] * stride_out_i +
        block_j[None, :] * stride_out_j +
        hidden * stride_out_h
    )
    # cast to float16 if we used mixed precision? We'll keep float32.

"""
custom_kernel for the “outgoing” triangular multiplicative update (TriMul).

The heavy O(N³·C_hidden) part (the einsum
    out[..., i, j, d] = Σₖ left[..., i, k, d] * right[..., j, k, d])
is implemented as a fused Triton kernel that performs many
independent matrix multiplications of size (N×N)·C_hidden in a single launch.
All lighter operations (LayerNorm, Linear projections, gating, masking)
are done with regular PyTorch kernels.

The Triton kernel processes a (batch, hidden_dim) pair as a single program
instance and tiles the (i, j) output plane with blocks of size
BLOCK_M × BLOCK_N while looping over k in blocks of size BLOCK_K.
Each tile accumulates a float‑32 matrix using tl.dot (which internally uses
Tensor‑Core‑friendly matmul on H100).  The final result is stored as
float‑32, matching the required output datatype.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul with a hidden‑dim loop (the core TriMul)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # const float*  [B, N, N, H]
    right_ptr,                    # const float*  [B, N, N, H]
    out_ptr,                      # float*        [B, N, N, H]
    B, N, H,                      # int32 scalars
    # strides for left
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (same layout as left)
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    # strides for out (same layout as left)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # block size constants
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for a given (batch, hidden) slice:
        out[i, j] = Σₖ left[i, k] * right[j, k]   (all indices 0‑based)
    where left/right have shape (N, N) and the sum runs over the second N axis.
    """
    # --------------------------------------------------------------
    # 1) Identify which batch & hidden channel this program processes
    # --------------------------------------------------------------
    pid = tl.program_id(0)                     # linear index over B*H
    batch_id = pid // H
    hidden_id = pid % H

    # --------------------------------------------------------------
    # 2) Tile indices for the output (i‑tile, j‑tile)
    # --------------------------------------------------------------
    pid_i = tl.program_id(1)                   # tile index along M (i)
    pid_j = tl.program_id(2)                   # tile index along N (j)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)        # [BLOCK_M]
    offs_j = j0 + tl.arange(0, BLOCK_N)        # [BLOCK_N]

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # 3) Accumulate over the k‑dimension in BLOCK_K sized chunks
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left[i, k]   -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + batch_id * stride_l_b
            + hidden_id * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right_T[k, j] (right[j, k] transposed) -> shape (BLOCK_K, BLOCK_N)
        # We load the transposed view directly so that tl.dot works without extra transpose.
        right_ptrs = (
            right_ptr
            + batch_id * stride_r_b
            + hidden_id * stride_r_h
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_i
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate matrix product: (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    # 4) Write the result
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_b
        + hidden_id * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional masking (mask ∈ {0,1})
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; make it broadcastable to [B,N,N,1]
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections of the *pre‑norm* tensor)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # same shape as left
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # shape [B,N,N,hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul (batch * hidden_dim independent GEMMs)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # left.shape == (B,N,N,hidden_dim)

    # Prepare output tensor
    out = torch.empty_like(left)   # [B,N,N,hidden_dim]

    # strides (int64) – required by Triton
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 (supports up to 128×128 tiles comfortably)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                                 # program_id(0)
        (N + BLOCK_M - 1) // BLOCK_M,                  # program_id(1) – i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,                  # program_id(2) – j‑tiles
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm on hidden dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # final projection back to dim

    return output
```
