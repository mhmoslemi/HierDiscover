    BATCH, M, N, K,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # offsets.

   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) – a fused implementation of the core AlphaFold3 update.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor over its last dimension (dim).
2. Linear projections (no bias) produce left/right tensors of shape
   [B, N, N, H] (H = hidden_dim) and three gating tensors (sigmoid).
3. Apply the optional pair mask (B,N,N) and the per‑position gates.
4. For each batch‑and‑head pair compute a dense N×N matrix product:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
   This is exactly a batched mat‑mul across the K‑dimension.
   The heavy N³ work is performed by a custom Triton kernel.
5. Reshape back to [B, N, N, H] and apply a second Layer‑norm
   (to_out_norm) followed by the out‑gate.
6. Final linear layer maps back to the original feature dimension dim.

The kernel works on FP16 tensors (left/right) and accumulates in
FP32 for numerical stability.  All surrounding bookkeeping (norm,
gates, final linear) is done in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # compile‑time constants
    BATCH, M, N, K,
    # strides (in elements)
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    # block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched mat‑mul: out = left @ rightᵀ   (left: [BATCH,M,K], right: [BATCH,K,N])"""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ----- coordinate offsets within the M/N dimensions -----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds (for the last partial block)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- k‑loop (blocked) -----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left block: [BLOCK_M, BLOCK_K]  (FP16 → FP32)
        left_ptrs = (
            left_ptr
            + batch_id * stride_l_batch
            + offs_m[:, None] * stride_l_m
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # right block: [BLOCK_K, BLOCK_N] (FP16 → FP32)
        right_ptrs = (
            right_ptr
            + batch_id * stride_r_batch
            + offs_k[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_n
        )
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        right = right.to(tl.float32)

        # dot‑product accumulation
        acc += tl.dot(left, right)

    # ----- store result (FP32 → FP16) -----
    out_ptrs = (
        out_ptr
        + batch_id * stride_o_batch
        + offs_m[:, None] * stride_o_m
        + offs_n[None, :] * stride_o_n
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor: torch.Tensor of shape [B, N, N] (bool/float)
            - weights_dict: mapping from weight name to torch.Tensor (all on GPU)
            - config_dict: contains `dim`, `hidden_dim`, and optionally `nomask`

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm 1 (over dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # compute mean / var across last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias  # broadcast over dim

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    left = F.linear(x_norm, left_proj_weight)          # [B,N,N,H]
    right = F.linear(x_norm, right_proj_weight)        # [B,N,N,H]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid) – also no bias linear layers
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy N³ mat‑mul
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # permute to bring the hidden dim to the batch axis:
    #   [B, N, N, H] -> [B, H, N, N]
    left_perm = left_f16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    # view as batch of independent matrices: [batch_dim, N, N]
    left_mat = left_perm.view(batch_dim, N, N)
    right_mat = right_perm.view(batch_dim, N, N)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # stride arguments (in elements, not bytes)
    stride_l_batch = left_mat.stride(0)
    stride_l_m = left_mat.stride(1)
    stride_l_k = left_mat.stride(2)

    stride_r_batch = right_mat.stride(0)
    stride_r_k = right_mat.stride(1)
    stride_r_n = right_mat.stride(2)

    stride_o_batch = out_mat.stride(0)
    stride_o_m = out_mat.stride(1)
    stride_o_n = out_mat.stride(2)

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_dim,
        N,
        N,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H] and promote to FP32
    # ------------------------------------------------------------------
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # LayerNorm 2 over the hidden dimension H
    # ------------------------------------------------------------------
    mean2 = out.mean(dim=-1, keepdim=True)
    var2 = out.var(dim=-1, unbiased=False, keepdim=True)
    out = (out - mean2) / torch.sqrt(var2 + eps)
    out = out * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # Apply out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
