@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute batch index (which matrix)
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_id = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_id // tl.cdiv(N, BLOCK_N)
    pid_n = block_id % tl.cdiv(N, BLOCK_N)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds
    m_mask = offs_m < M
    n_mask = offs_n < N

    # pointers to A and B for this batch
    a_ptrs = A_ptr + batch_idx * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = B_ptr + batch_idx * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        a = tl.load(a_ptrs, mask=m_mask[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K - k*BLOCK_K) & n_mask[None, :], other=0.0)
        # dot product
        acc += tl.dot(a, b, out_dtype=tl.float32)
        # increment pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k
    # write back C
    c_ptrs = C_ptr + batch_idx * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=m_mask[:, None] & n_mask[None, :])

def batched_gemm(A, B):
    # A,B shape (batch*H, N, N) float16
    batch, M, K = A.shape
    _, _, N = B.shape
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)
    # get strides

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # LayerNorm - Norm over last dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Compute mean and var across last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias

    # Linear projections / gates
    # left_proj
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present (config may have nomask)
    if not config.get("nomask", False):
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Convert to fp16 for GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape for batched GEMM
    B, N, _, _ = left.shape
    # Transpose to (B, hidden_dim, N, N) then flatten batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # B, H, N, N
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # B, H, N, N (k, j) - we need transpose of last two dims
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # Perform batched GEMM using Triton
    out_flat = batched_gemm_triton(left_flat, right_flat)

    # Convert back to fp32
    out = out_flat.to(torch.float32)

    # Reshape back to B,N,N,H
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # B,N,N,H

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # layernorm over hidden_dim
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) / torch.sqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Linear to output dim
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final

"""
TriMul (outgoing) forward pass with a fused Triton batched‑GEMM kernel.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm over the feature dimension (`dim`).
2. Linear projections (`left_proj`, `right_proj`) and three sigmoid gates.
3. Mask (if provided) is applied to the projections before gating.
4. The core O(N³·hidden_dim) tensor contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is rewritten as a batched matrix multiplication
       out = (left_reshaped) @ (right_reshaped)ᵀ
   and executed with a custom Triton kernel that processes many
   (batch*hidden) GEMMs in parallel.
5. A second layer‑norm over the hidden dimension, multiplication by the
   output gate and a final linear projection back to `dim`.

Only the heavy O(N³) contraction is run on Triton; the remaining
operations use highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
import math


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A: (B, M, K)   – row‑major, contiguous
# B: (B, K, N)   – row‑major, contiguous (already transposed)
# C: (B, M, N)   – row‑major, contiguous, fp16 output (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,               # pointers
    batch_size, M, N, K,   # dimensions
    stride_Ab, stride_Am, stride_Ak,    # A strides (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,    # B strides (batch, row, col)
    stride_Cb, stride_Cm, stride_Cn,    # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)

    # --------------------------------------------------------------
    #  Compute the logical batch index and the 2‑D block indices.
    # --------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_m * num_n)                     # which (b,h) pair
    pid_in_batch = pid % (num_m * num_n)                  # 2‑D block id
    pid_m = pid_in_batch // num_n                         # row block
    pid_n = pid_in_batch % num_n                          # col block

    # --------------------------------------------------------------
    #  Offsets for the current tile.
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over K dimension in tiles.
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current K‑tile of A and B.
        a_ptrs = (A + batch_idx * stride_Ab
                  + offs_m[:, None] * stride_Am
                  + offs_k[None, :] * stride_Ak)
        b_ptrs = (B + batch_idx * stride_Bb
                  + offs_k[:, None] * stride_Bk
                  + offs_n[None, :] * stride_Bn)

        # Load with boundary masks; out‑of‑range values are zero.
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate the dot product.
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # --------------------------------------------------------------
    #  Write the result, casting back to fp16.
    # --------------------------------------------------------------
    c_ptrs = (C + batch_idx * stride_Cb
              + offs_m[:, None] * stride_Cm
              + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    A, B : (B*H, N, N)  FP16, contiguous
    returns C : (B*H, N, N) FP16
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, _, N = B.shape

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides (in elements, not bytes)
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bk, stride_Bn = B.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    # Tile sizes – tuned for H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_blocks_m * num_blocks_n

    grid = (total_blocks,)

    _batched_gemm_kernel[grid](
        A, B, C,
        batch, M, N, K,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be ignored
        - weights      : dict of model parameters (all float32)
        - config       : dict with keys `dim`, `hidden_dim`, `nomask` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # Compute mean / var manually (same numerics as nn.LayerNorm)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_w + norm_b   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and optional mask
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", False):
        # mask is broadcasted over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # Apply gates
    left = left_proj * left_gate    # [B, N, N, hidden_dim]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 before the O(N³) contraction (large memory savings)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare tensors for batched GEMM:
    # left  : (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    # right : (B, N, N, H) -> (B, H, N, N) with (k,j) transposed -> (B, H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()          # B, H, N, N
    right_t = right.permute(0, 3, 2, 1).contiguous()        # B, H, N, N (k, j) already transposed
    left_flat = left_t.view(B * hidden_dim, N, N)
    right_flat = right_t.view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # Batched GEMM via Triton
    # ------------------------------------------------------------------
    out_flat = _batched_gemm_triton(left_flat, right_flat)   # fp16

    # Back to FP32 for the remaining (lighter) ops
    out = out_flat.to(torch.float32)

    # Reshape back to [B, N, N, hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_out) / torch.sqrt(var_out + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # Output gate and final linear projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final
```

**Explanation of key components**

* **Layer normalisation** – performed with explicit mean/variance to avoid an extra
  `torch.nn.LayerNorm` call and to keep the code self‑contained.

* **Projection & gating** – standard `torch.nn.functional.linear` (cuBLAS) for the
  dense multiplies; the mask is applied (if present) before the gates, exactly as
  in the reference implementation.

* **Core contraction** – rewritten as a batched matrix multiplication
  `C = A @ Bᵀ`.  The Triton kernel `_batched_gemm_kernel` processes many independent
  GEMMs (one per `(batch, hidden_dim)` pair) in parallel.  Tiles of size
  64×64×64 give good occupancy on an H100; accumulation is performed in FP32 and
  the result is stored as FP16 to save bandwidth.

* **Post‑processing** – a second LayerNorm, multiplication by the output gate,
  and a final linear projection back to `dim` dimensions are done with the
  highly‑optimised PyTorch primitives.

This combination respects the requirement of using a custom Triton kernel for
the dominant O(N³) work while keeping the rest of the pipeline simple and fast.
