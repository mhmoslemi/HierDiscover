@triton.jit
def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program_id: (pid_i, pid_j, pid_batch)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)
    
    # compute batch and hidden indices
    batch_idx = pid_batch // H
    hidden_idx = pid_batch % H
    
    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # compute start ptr for left and right for this batch and hidden
    left_base = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    out_base = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h

    # iterate over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        
        # Load left tile
        left_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (k_offset[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=(offs_i < N)[:, None] & (k_offset < N)[None, :], other=0.0)
        # left shape (BLOCK_M, BLOCK_K)
        
        # Load right tile (need transposed)
        right_ptrs = right_base + (offs_j[:, None] * stride_right_j) + (k_offset[None, :] * stride_right_k)
        # Without transposition we would have shape (BLOCK_N, BLOCK_K) where j rows; we need right_T[k, j], so using stride_right_k for rows (k) and stride_right_j for columns (j). However we currently have j as rows; need to swap.
        # We'll load as (BLOCK_N, BLOCK_K) then transpose using tl.transpose.
        right = tl.load(right_ptrs, mask=(offs_j < N)[:, None] & (k_offset < N)[None, :], other=0.0)
        # right shape (BLOCK_N, BLOCK_K)
        right_T = tl.transpose(right)  # shape (BLOCK_K, BLOCK_N)
        
        # accumulate
        acc += tl.dot(left, right_T)
    
    # store results
    out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype==tl.float16 else acc, mask=(offs_i < N)[:, None] & (offs_j < N)[None, :])

"""
tri_mul_outgoing: Triton‑accelerated forward pass of the “outgoing” TriMul module.

Algorithm
---------
1. Layer‑norm the input tensor `x`  (shape [B,N,N,dim]).
2. Linear projections → `left` and `right` (shape [B,N,N,hidden_dim]).
3. Optional mask (broadcasted over the hidden dimension) is applied to both
   `left` and `right`.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from the
   same normalized input and apply the first two to the projected tensors.
5. Core operation – a ternary reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication of `left[...,h]` with the
   transpose of `right[...,h]`.  The reduction is performed by a fused
   Triton kernel that treats the (batch, hidden) pair as a single batch axis,
   giving a ∼B·hidden_dim independent matmuls of size N×N.
6. Layer‑norm + output gate are applied to the reduction result.
7. A final linear projection maps the hidden dimension back to `dim`.

The Triton kernel loads tiles of `left` (i,k) and `right` (j,k), transposes the
right tile, performs a dot‑product over the reduction axis `k` and accumulates
into a 2‑D tile of the output.  The kernel is launched with a grid of
(N/BLOCK_M, N/BLOCK_N, B·hidden_dim) blocks.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# --------------------------------------------------------------------------- #
# Triton kernel: sum_k left[b,i,k,h] * right[b,j,k,h] → out[b,i,j,h]
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,          # ptr to left  [B,N,N,H]
    right_ptr,         # ptr to right [B,N,N,H]
    out_ptr,           # ptr to out   [B,N,N,H]
    B, N, H,           # runtime dimensions
    # strides for left  (b,i,k,h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (b,j,k,h)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out   (b,i,j,h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)      # tile row   (i)
    pid_j = tl.program_id(1)      # tile col   (j)
    pid_bh = tl.program_id(2)     # combined batch‑hidden index

    # --------------------------------------------------------------- #
    # De‑compose the combined batch‑hidden index
    # --------------------------------------------------------------- #
    batch_idx = pid_bh // H
    hidden_idx = pid_bh % H

    # --------------------------------------------------------------- #
    # Offsets for the current tile
    # --------------------------------------------------------------- #
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------- #
    # Base pointers for the current (batch, hidden) slice
    # --------------------------------------------------------------- #
    left_base  = left_ptr  + batch_idx * stride_left_b  + hidden_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h
    out_base   = out_ptr   + batch_idx * stride_out_b   + hidden_idx * stride_out_h

    # --------------------------------------------------------------- #
    # Accumulator for the tile
    # --------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # Loop over the reduction axis k
    # --------------------------------------------------------------- #
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # ---------- left[i,k] ----------
        left_mask = (offs_i[:, None] < N) & (k_offset[None, :] < N)
        left_ptrs = left_base \
                    + offs_i[:, None] * stride_left_i \
                    + k_offset[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)          # (BM, BK)

        # ---------- right[j,k] ----------
        right_mask = (offs_j[:, None] < N) & (k_offset[None, :] < N)
        right_ptrs = right_base \
                     + offs_j[:, None] * stride_right_j \
                     + k_offset[None, :] * stride_right_k
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)      # (BN, BK)
        # Transpose to obtain (BK, BN) so that we can perform a matmul
        right_T = tl.transpose(right)                               # (BK, BN)

        # ---------- accumulate ----------
        acc += tl.dot(left, right_T)    # (BM, BN) += (BM, BK) @ (BK, BN)

    # --------------------------------------------------------------- #
    # Write the result back
    # --------------------------------------------------------------- #
    out_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    out_ptrs = out_base \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=out_mask)


# --------------------------------------------------------------------------- #
# Public entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] or None
        - weights      : dict with named weight tensors
        - config       : dict containing `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3) Optional mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------- #
    if mask is not None:
        mask = mask.to(left.dtype).unsqueeze(-1)           # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid of linear projections of the *normalized* input)
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous memory for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------- #
    # 5) Core reduce‑over‑k operation via Triton
    # ------------------------------------------------------------------- #
    out = torch.empty_like(left)   # [B,N,N,hidden_dim], float32

    # Strides (in elements, not bytes)
    stride_left = left.stride()    # (B, N, N, H)
    stride_right = right.stride()
    stride_out = out.stride()

    # Tile sizes – tuned for H100 (pmem ~ 128 KiB per block is safe)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles over j
        B * hidden_dim                         # combined batch‑hidden axis
    )

    tri_mul_einsum_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------- #
    # 6) Output layer‑norm, gate and final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
