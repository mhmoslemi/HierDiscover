```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_rem = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = pid_rem // tl.cdiv(N, BLOCK_N)
    pid_n = pid_rem % tl.cdiv(N, BLOCK_N)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = a_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_m + tl.arange(0, BLOCK_K)[None, :] * stride_a_k
    b_ptrs = b_ptr + batch * stride_b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    c = acc.to(tl.float16)  # or tl.float32? The output should be float32. We can store as float32.
    c_ptrs = c_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

def batched_matmul_triton(left, right):
    # left: (B,N,N,H)
    # right: (B,N,N,H)
    # compute out: (B,N,N,H) = left @ right.transpose(-2,-1) per hidden
    # Implementation steps:
    B, N, _, H = left.shape
    # reshape
    left_perm = left.permute(0,3,1,2).contiguous()  # (B,H,N,N)
    right_t = right.permute(0,3,2,1).contiguous()   # (B,H,N,N)
    # flatten batch and hidden
    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_t.view(B*H, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat)  # float32
    # launch kernel
    grid = lambda meta: ( (B*H * ((N + meta['BLOCK_M'] - 1) // meta['BLOCK_M']) * ((N + meta['BLOCK_N'] - 1) // meta['BLOCK_N']),) )
    batched_matmul_kernel[grid]( left_flat, right_flat, out_flat,
                                 N, N, N,
                                 left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
                                 right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
                                 out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
                                 BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0,2,3,1)  # (B,N,N,H)
    return out

def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # mask broadcast
    if mask is None:
        mask = torch.ones(bs, N, N, device=device, dtype=x.dtype)
    mask = mask.unsqueeze(-1)  # (bs,N,N,1)
    left = left * mask * left_gate
    right = right * mask * right_gate

    # Compute pairwise product (batched matmul)
    # Permute for hidden dimension batching
    left_perm = left.permute(0,3,1,2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0,3,2,1).contiguous()    # (B, H, N, N) (k,j)
    B, H, M, K = left_perm.shape   # M=N, K=N
    # flatten batch and hidden
    left_flat = left_perm.view(B*H, M, K)
    right_flat = right_t.view(B*H, K, M)   # note K,N shape but we need (K,N) so shape (B*H, K, N). But we intend B*H, K, N = (B*H,N,N)
    # Actually right_flat shape (B*H,N,N) but we want (B*H, K, N). Already K=M=N.

    # allocate output
    out_flat = torch.empty_like(left_flat)  # shape (B*H, M, M)

    # launch kernel
    # compute grid size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # compute grid as tuple
    def grid_fn(meta):
        total_blocks_per_batch = ((M + meta['BLOCK_M'] - 1) // meta['BLOCK_M']) * ((K + meta['BLOCK_N'] - 1) // meta['BLOCK_N'])
        return (B*H * total_blocks_per_batch, )
    batched_matmul_kernel[grid_fn]( left_flat, right_flat, out_flat,
                                   M, K, K, # M,N,K where N = K because output square
                                   B*H,
                                   left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
                                   right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
                                   out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
                                   BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K )
    # reshape out
    out = out_flat.view(B, H, M, M).permute(0,2,3,1)  # (B,N,N,H)

    # Output layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel for the
core N³ matrix‑multiplication across the hidden dimension.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left / right (shape [B,N,N,H]).
3. Compute three sigmoid gates (left, right, out) from the same normalized input.
4. Apply mask (if provided) and element‑wise gate weighting.
5. Compute the pairwise product
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that treats the (batch, hidden) pair as a
   single batched dimension and performs a tiled GEMM (A @ Bᵀ) in FP32.
6. Layer‑norm the intermediate output, multiply by out‑gate and apply the final
   linear projection back to the original channel dimension.

The Triton kernel handles the heavy O(N³·H) work, while the remaining
operations (layer‑norm, linear layers, sigmoid) are performed with
torch.nn.functional for simplicity and numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM (A @ Bᵀ) where A,B have shape (B, M, K) and (B, K, N)
# ----------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K, BATCH,               # problem size
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------- #
    # Map pid -> (batch, block_m, block_n)
    # --------------------------------------------------------------- #
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    total_per_batch = num_pid_m * num_pid_n

    batch_idx = pid // total_per_batch
    pid_in_batch = pid % total_per_batch
    pid_m = pid_in_batch // num_pid_n
    pid_n = pid_in_batch % num_pid_n

    # --------------------------------------------------------------- #
    # Offsets for the current tile
    # --------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile for A and B
    a_ptrs = a_ptr + batch_idx * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_b_batch \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------- #
    # Loop over K dimension with reduction tiles
    # --------------------------------------------------------------- #
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = _ * BLOCK_K
        k_mask = (cur_k + offs_k) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_mask[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & mask_n[None, :],
                    other=0.0)
        acc += tl.dot(a, b)

        # advance pointers to the next k‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # --------------------------------------------------------------- #
    # Write back the result
    # --------------------------------------------------------------- #
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------------- #
# Entry point used by the evaluator
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C]
            - mask: torch.Tensor [B, N, N] (may be all‑ones)
            - weights: dict with all linear/LayerNorm parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------- #
    # Unpack arguments
    # --------------------------------------------------------------- #
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    B, N, _, C = x.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------- #
    # Helper to fetch optional bias tensors
    # --------------------------------------------------------------- #
    def _maybe(tensor):
        return tensor if tensor is not None else None

    # --------------------------------------------------------------- #
    # 1. Input LayerNorm
    # --------------------------------------------------------------- #
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=_maybe(weights.get("norm.weight")),
        bias=_maybe(weights.get("norm.bias")),
        eps=1e-5,
    )

    # --------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------- #
    # 3. Gating values (sigmoid)
    # --------------------------------------------------------------- #
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------- #
    # 4. Apply mask (if any) and fuse with gates
    # --------------------------------------------------------------- #
    if mask is None:
        mask = torch.ones(B, N, N, device=device, dtype=dtype)
    mask = mask.unsqueeze(-1)  # (B,N,N,1)

    left = left * mask * left_gate
    right = right * mask * right_gate

    # --------------------------------------------------------------- #
    # 5. Core N³ multiplication: out = left @ rightᵀ  per hidden channel
    # --------------------------------------------------------------- #
    #   left:  (B, N, N, H) → (B, H, N, N)
    #   rightᵀ: (B, N, N, H) → (B, H, N, N) with axes (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t   = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Collapse (batch, hidden) into a single batch dimension for the kernel
    BH = B * H
    left_flat  = left_perm.view(BH, N, N)          # (BH, M, K)
    right_flat = right_t.view(BH, N, N)            # (BH, K, N)  (K == N)
    out_flat   = torch.empty_like(left_flat)       # (BH, M, N)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    def grid_func(meta):
        total_blocks_m = (N + meta["BLOCK_M"] - 1) // meta["BLOCK_M"]
        total_blocks_n = (N + meta["BLOCK_N"] - 1) // meta["BLOCK_N"]
        return (BH * total_blocks_m * total_blocks_n,)

    batched_matmul_kernel[grid_func](
        left_flat,
        right_flat,
        out_flat,
        N,              # M
        N,              # N (output width)
        N,              # K (reduction dimension)
        BH,             # batch count (B * H)
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- #
    # 6. Output LayerNorm, gate and final linear projection
    # --------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=_maybe(weights.get("to_out_norm.weight")),
        bias=_maybe(weights.get("to_out_norm.bias")),
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
