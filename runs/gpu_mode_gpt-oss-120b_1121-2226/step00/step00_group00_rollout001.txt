def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)

    # extract weights
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Apply LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if not nomask and mask is not None:
        mask = mask.to(left.dtype)  # cast
        mask = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute out via Triton kernel
    B, N, _, _ = left.shape
    # allocate result tensor
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Get strides
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    # Launch kernel
    BLOCK_M = 64

"""
TriMul "outgoing" forward pass (AlphaFold3 style) with a custom Triton kernel.

The heavy part of the computation is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batched matrix multiplication for every (batch, hidden) pair.
We implement this contraction with a tiled GEMM kernel in Triton and
use regular PyTorch for the surrounding Linear/LayerNorm/gating logic.
The kernel works on float32 tensors (inputs are cast to float32 before the
kernel launch) and stores the result in float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, N, hidden,
    # strides for left  (B, i, k, d)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_hidden,
    # strides for right (B, j, k, d)
    stride_r_batch, stride_r_j, stride_r_k, stride_r_hidden,
    # strides for out   (B, i, j, d)
    stride_o_batch, stride_o_i, stride_o_j, stride_o_hidden,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = Σ_k left[..., i, k, d] * right[..., j, k, d]"""
    pid_m = tl.program_id(0)          # tile over i (rows)
    pid_n = tl.program_id(1)          # tile over j (cols)
    pid_bh = tl.program_id(2)         # combined batch & hidden dimension

    # decode batch and hidden indices
    b = pid_bh // hidden
    h = pid_bh % hidden

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the current (b, h)
    left_base  = left_ptr  + b * stride_l_batch  + h * stride_l_hidden
    right_base = right_ptr + b * stride_r_batch + h * stride_r_hidden

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  (shape BLOCK_M x BLOCK_K)
        a_ptrs = left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right_T[k, j]  (shape BLOCK_K x BLOCK_N), i.e. we read right as (k,j)
        b_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product over k
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) -> (M,N)

    # write result
    out_base = out_ptr + b * stride_o_batch + h * stride_o_hidden
    c_ptrs = out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul module using a Triton kernel for the
    heavy pairwise contraction.  Arguments:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- unpack weights ----------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

    # ----- LayerNorm ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]

    # ----- Linear projections -------------------------------------------------
    left = F.linear(x, left_proj_weight)   # -> [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ----- Apply mask (if any) ------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] -> [B,N,N,1]
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ----- Gating ------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Cast to float32 for the Triton kernel (ensure contiguity) -------
    left = left.to(torch.float32).contiguous()
    right = right.to(torch.float32).contiguous()

    B, N, _, _ = left.shape
    out = torch.empty_like(left, dtype=torch.float32).contiguous()  # [B,N,N,hidden_dim]

    # ----- Strides (in elements) --------------------------------------------
    stride_l = left.stride()   # (B, i, k, d)
    stride_r = right.stride()
    stride_o = out.stride()

    # ----- Triton kernel launch ---------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * hidden_dim,            # (batch, hidden) dimension
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing ----------------------------------------------------
    # LayerNorm on hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # Apply output gate
    out = out * out_gate
    # Final linear projection back to dim
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
