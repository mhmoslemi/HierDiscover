```
@triton.jit
def triton_gemm_batch_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bn, stride_bk, 
    stride_cm, stride_cn,
    BATCH,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # Pointers for batch offset
    a_batch_ptr = A_ptr + batch_idx * stride_batch_a
    b_batch_ptr = B_ptr + batch_idx * stride_batch_b
    c_batch_ptr = C_ptr + batch_idx * stride_batch_c

    # compute offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    # load A sub-tile: shape [BLOCK_M, BLOCK_K]
    A = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak), mask=mask_m[:, None] & (offs_k < K), other=0.0)
    # load B sub-tile: shape [BLOCK_K, BLOCK_N]
    B = tl.load(b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn), mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load current block of A and B
        a = tl.load(a_batch_ptr + (offs_m[:, None] * stride_am + (k*BLOCK_K + offs_k)[None, :] * stride_ak), mask=mask_m[:, None] & ((k*BLOCK_K+offs_k) < K), other=0.0)
        b = tl.load(b_batch_ptr + ((k*BLOCK_K + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn), mask=((k*BLOCK_K+offs_k)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # (increment pointers automatically not needed because we compute based on k)

    # write back
    c = acc.to(tl.float32) # already float
    tl.store(c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn), c, mask=mask_m[:, None] & mask_n[None, :])
```

    stride_b_batch, stride_b_k, stride_b_n,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    block_m = tl.program_id(1)
    block_n = tl.program_id(2)

    # compute offsets for batch
    a_ptr = A_ptr + batch_idx * stride_a_batch
    b_ptr = B_ptr + batch_idx * stride_b_batch
    c_ptr = C_ptr + batch_idx * stride_c_batch

    # tile indices within M and N dimensions
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator

"""
custom_kernel: highly‑optimized forward pass of the TriMul (outgoing) operator.

Algorithm
---------
1. Layer‑normalize the input tensor `x` (shape [B,N,N,C]).
2. Linear projections to the hidden dimension:
   * left  = x @ W_left   (no bias)
   * right = x @ W_right
   * left_gate  = sigmoid(x @ W_left_gate)
   * right_gate = sigmoid(x @ W_right_gate)
   * out_gate   = sigmoid(x @ W_out_gate)
3. Apply optional mask (broadcast over the hidden dimension) and the
   multiplicative gates to `left` and `right`.
4. Fuse the N‑wise “tri‑multiplication”:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of GEMMs (B·D hidden streams) of size N×N.
   A custom Triton kernel performs the batched GEMM in FP16 with
   FP32 accumulation for speed on H100.
5. Reshape the result back to [B,N,N,D] and cast to FP32.
6. Layer‑normalize over the hidden dimension, multiply by `out_gate`,
   and apply the final linear projection `to_out` (shape D←D_hidden).
7. Return the tensor of shape [B,N,N,C] (float32).

Only the O(N³) tri‑multiplication is implemented in Triton; all other
operations use PyTorch (highly‑optimized and fused in the compiler).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: [batch,M,K], B: [batch,K,N])"""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Pointers to the start of this batch
    a_batch = A_ptr + batch_id * stride_a_batch
    b_batch = B_ptr + batch_id * stride_b_batch
    c_batch = C_ptr + batch_id * stride_c_batch

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Load A tile (M×K) and B tile (K×N)
        a = tl.load(
            a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k),
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n),
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # FP32 accumulation

    # Write back C tile
    tl.store(
        c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n),
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask_tensor: torch.Tensor [B, N, N] (bool/float) or None
            - weights: dict of model weights (torch.Tensors)
            - config: dict with fields "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 1. unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]                # C == dim

    # ------------------------------------------------------------------
    # 2. LayerNorm over the channel dimension (C)
    # ------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,C]  (float32)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) + optional mask
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,HD]
    right = F.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", False) and mask_tensor is not None:
        # mask is broadcast over hidden dimension
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoids) and apply gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Convert to FP16 for the heavy O(N³) GEMM
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # bring hidden dimension into the batch axis for a batched GEMM
    # left_fp16 : [B, N, N, HD] -> [B, HD, N, N]
    left_t = left_fp16.permute(0, 3, 1, 2).contiguous()
    # right must be transposed on the last two axes: (j,k) -> (k,j)
    right_t = right_fp16.permute(0, 3, 2, 1).contiguous()

    # flatten batch*hidden_dim -> [B*HD, N, N]
    batch_hidden = B * hidden_dim
    left_mat  = left_t.view(batch_hidden, N, N)
    right_mat = right_t.view(batch_hidden, N, N)
    out_mat   = torch.empty_like(left_mat, dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6. Launch Triton batched GEMM kernel (C = A @ B)
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # strides in element units (not bytes)
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k     = right_mat.stride(1)   # row stride of B (K dimension)
    stride_b_n     = right_mat.stride(2)   # column stride of B (N dimension)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B,N,N,HD] (float32) and apply output layers
    # ------------------------------------------------------------------
    out = (
        out_mat.view(B, hidden_dim, N, N)          # [B,HD,N,N]
        .permute(0, 2, 3, 1)                       # [B,N,N,HD]
        .contiguous()
        .to(torch.float32)                         # back to FP32 for LayerNorm
    )

    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate and final projection to original dim
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
