def tri_mul_triton(left, right, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    # left shape (B,N,N,D)
    # right shape (B,N,N,D)
    B, N, _, D = left.shape
    # rearrange shape for GEMM
    left_perm = left.permute(0,3,1,2).contiguous()   # (B, D, N_i, N_k)
    right_perm = right.permute(0,3,2,1).contiguous() # (B, D, N_k, N_j)
    left_flat = left_perm.view(-1, N, N)   # (B*D, M=N, K=N)
    right_flat = right_perm.view(-1, N, N) # (B*D, K=N, N=j)
    # allocate output
    out_flat = torch.empty_like(left_flat) # shape (B*D, N, N)
    # compute gemm using triton
    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B * D )
    batched_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_flat.view(B, D, N, N).permute(0,2,3,1).contiguous()
    return out

"""
TriMul “outgoing” kernel – fast forward pass.

The implementation follows the reference PyTorch `TriMul` module:

1. Layer‑norm the input.
2. Linear projections → left/right tensors (shape B×N×N×H) and three
   element‑wise sigmoid gates (left, right, out).  The mask (if any) and
   left/right gates are fused into the projected tensors.
3. Core operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is an Einstein sum equivalent to a batched matrix‑multiply
   over the sequence dimension `k`.  We reshape the tensors so that the
   hidden dimension `H` becomes part of the batch (B*H) and launch a
   Triton GEMM kernel that processes (i,j) tiles while looping over `k`.
4. Layer‑norm over the hidden dimension, multiply by the out‑gate and a
   final linear projection back to the original hidden size.

Only the heavy O(N³) multiplication is performed in Triton; all other
operations use native PyTorch kernels (which are already highly optimized
on H100).  The kernel works for arbitrary batch size, sequence length
and hidden dimension (the tests use B∈{1,2}, N up to 1024, H=128).

The kernel uses 32‑bit accumulation for numerical correctness.  Block
sizes are tuned for H100 (128×128 tiles, K‑tile 32).  The implementation
conforms to Triton 3.3.1.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ over the K dimension)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # strides for A   (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ  (batch, row(K), col(N))
    stride_ch, stride_cm, stride_cn,   # strides for C   (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row index   (i)
    pid_n = tl.program_id(1)   # block column index (j)
    pid_z = tl.program_id(2)   # batch*hidden index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask to avoid out‑of‑bounds reads
    mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
    mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

    # Pointers with the batch offset applied
    a_ptrs = a_ptr + pid_z * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)                           # (BLOCK_M, BLOCK_N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_z * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _triton_tri_mul(left: torch.Tensor, right: torch.Tensor,
                    block_m: int = 128, block_n: int = 128, block_k: int = 32) -> torch.Tensor:
    """
    Perform the core TriMul contraction using Triton.

    Parameters
    ----------
    left, right : torch.Tensor
        Tensors of shape (B, N, N, H) on CUDA.
    block_m / block_n / block_k : int
        Tile sizes for the Triton kernel.

    Returns
    -------
    out : torch.Tensor
        Tensor of shape (B, N, N, H) on CUDA.
    """
    B, N, _, H = left.shape

    # ------------------------------------------------------------------
    # Fuse hidden dimension into the batch dimension
    #   left : (B, H, N, N)  ->  (B*H, N, N)   (row = i, col = k)
    #   right: (B, H, N, N)  ->  (B*H, N, N)   (row = k, col = j)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N_i, N_k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N_k, N_j)

    left_flat = left_perm.view(-1, N, N)   # (B*H, M=N, K=N)
    right_flat = right_perm.view(-1, N, N) # (B*H, K=N, N=j)

    out_flat = torch.empty_like(left_flat)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    grid = (
        (N + block_m - 1) // block_m,   # blocks over rows (i)
        (N + block_n - 1) // block_n,   # blocks over cols (j)
        B * H                           # batch*hidden
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                 # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),# B strides (B is already transposed)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),      # C strides
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

    """

    # ------------------------------------------------------------------

    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------

    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)  # (B,N,N,hidden_dim)

    if mask_tensor is not None:
        # mask shape (B,N,N) -> (B,N,N,1) for broadcasting
        mask_exp = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core TriMul contraction (batched GEMM over the sequence dimension)
    # ------------------------------------------------------------------
    out = _triton_tri_mul(left, right,
                          block_m=128, block_n=128, block_k=32)   # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
```
