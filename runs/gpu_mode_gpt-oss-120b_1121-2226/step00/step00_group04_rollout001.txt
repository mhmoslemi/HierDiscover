@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_bl, stride_bi, stride_bk, stride_bd,
    stride_rl, stride_ri, stride_rk, stride_rd,
    stride_ol, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    epsilon: tl.constexpr
):
    # program ID
    batch_hidden_pid = tl.program_id(0)  # combined batch and hidden dimension index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch index and hidden index
    b = batch_hidden_pid // D
    d = batch_hidden_pid % D

    # compute tile start indices
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bound checks
    mask_i = i < N
    mask_j = j < N

    # pointers offset: base offsets for batch and hidden
    left_base = left_ptr + b * stride_bl + d * stride_bd
    right_base = right_ptr + b * stride_rl + d * stride_rd
    out_base = out_ptr + b * stride_ol + d * stride_od

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # k indices within block
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # load left tile (M x K)
        left_ptrs = left_base + i[:, None] * stride_bi + k[None, :] * stride_bk
        left_vals = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # load right tile (N x K) but we need transpose orientation right[j,k]
        # Use right with i index=j, k index=k
        right_ptrs = right_base + j[:, None] * stride_ri + k[None, :] * stride_rk
        right_vals = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        right_T_ptrs = right_base + k[:, None] * stride_rk + j[None, :] * stride_ri

def custom_kernel(data):
    """
    Compute the outgoing TriMul operator.
    The heavy pairwise interaction (einsum) is executed by a Triton
    kernel that performs a blocked matrix multiplication for each
    batch and hidden channel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # optional config fields
    nomask = config.get("nomask", False)

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F_layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections
    left = F_linear(x, weights["left_proj.weight"])
    right = F_linear(x, weights["right_proj.weight"])

    # Apply mask if provided
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F_linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F_linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F_linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Gather shapes
    B, N, _, _ = left.shape  # left shape [B,N,N,hidden]

    # Ensure contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Strides
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tunable block sizes (adjusted for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_ri, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

"""
TriMul outgoing kernel (AlphaFold3 style)

The heavy pairwise interaction:
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
is a batch of (hidden_dim) independent matrix multiplications
(left @ right^T) of size (seq_len × seq_len).  This kernel implements a
blocked GEMM in Triton: each program processes a TILE_M×TILE_N output patch
for a given (batch, hidden) pair and accumulates over K‑tiles.  The rest of
the TriMul layer (LayerNorm, Linear projections, gating and the final
output norm/linear) are performed with PyTorch which handles the small
tensor ops efficiently.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimensions that fit the test suite.  It uses float32 for accumulation
(and input tensors are kept in float32 for numerical stability).  The
output tensor is returned in float32.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, D,                                    # batch, seq_len, hidden_dim
    stride_lb, stride_li, stride_lk, stride_ld, # left strides (b,i,k,d)
    stride_rb, stride_ri, stride_rk, stride_rd, # right strides (b,i,k,d)
    stride_ob, stride_oi, stride_oj, stride_od, # out strides (b,i,j,d)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    Grid dimensions:
        program_id(0) -> batch * hidden (b,d)
        program_id(1) -> i‑tile index
        program_id(2) -> j‑tile index
    """
    pid0 = tl.program_id(0)  # combined batch*hidden
    pid1 = tl.program_id(1)  # i‑tile
    pid2 = tl.program_id(2)  # j‑tile

    # ------------------------------------------------------------------
    # decode batch and hidden channel
    b = pid0 // D
    d = pid0 % D

    # start indices of the current tile
    offs_m = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (b,d) slice
    left_base  = left_ptr  + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base   = out_ptr   + b * stride_ob + d * stride_od

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = left_base + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        left_tile = tl.load(
            left_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile transposed: we need right_T[k, j] = right[j, k]
        # original right layout: (b, i, k, d) -> stride_ri (i), stride_rk (k)
        # so right_T ptrs = k*stride_rk + j*stride_ri
        right_T_ptrs = right_base + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_ri
        right_T_tile = tl.load(
            right_T_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # GEMM accumulation: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_T_tile)

    # ------------------------------------------------------------------
    # write the result back
    out_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask_tensor : torch.Tensor [B, N, N]  (may be None)
            - weights_dict: dict with the following keys (all torch.Tensors):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict: contains at least "dim", "hidden_dim" and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, same device as inputs)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to same dtype as left/right
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # broadcastable
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid on linear projections of the normalized input)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # shape [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy pairwise interaction: einsum over k (implemented in Triton)
    B, N, _, _ = left.shape  # left shape = [B, N, N, hidden_dim]

    # ensure contiguous layout for efficient pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides for each tensor (as returned by .stride())
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # block sizes – tuned for H100 (TPU‑like Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # launch grid: (batch*hidden, tiles_i, tiles_j)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dimension
    out_final = F.linear(out_norm, weights["to_out.weight"])  # bias is False

    return out_final
