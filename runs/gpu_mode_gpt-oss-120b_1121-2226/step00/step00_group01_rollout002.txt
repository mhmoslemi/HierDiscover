def layernorm_kernel(
    input_ptr, output_ptr,
    weight_ptr, bias_ptr,
    C,
    eps,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    # row offset
    row_offset = pid * C
    # accumulator for sum and sum_sq
    sum_val = 0.0
    # number of columns
    cols = tl.arange(0, BLOCK_SIZE)
    # loop over channels in blocks
    for offset in range(0, C, BLOCK_SIZE):
        col = cols
        valid = offset + col < C
        ptr = input_ptr + row_offset + offset + col
        x = tl.load(ptr, mask=valid, other=0.0)
        sum_val += tl.sum(x, axis=0)
    mean = sum_val / C

    # second pass compute var
    var_val = 0.0
    for offset in range(0, C, BLOCK_SIZE):
        col = cols
        valid = offset + col < C

        x = tl.load(ptr, mask=valid, other=0.0)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)

"""
TriMul forward (outgoing) implementation with a fused LayerNorm Triton kernel.

Algorithm
---------
1. Layer‑norm the input `x` over the last dimension (dim) with learned
   weight/bias (norm.weight / norm.bias).  The kernel works on a flattened
   view of shape (B*N*N, dim) and writes the normalized tensor in float16.
2. Convert the normalized tensor back to float32 and compute the four
   linear projections (left/right) and three gating projections (left_gate,
   right_gate, out_gate) with `torch.nn.functional.linear`.  Gates are passed
   through a sigmoid.
3. Apply the optional pairwise mask to the left/right projections and
   multiply each projection by its gate.
4. Compute the pairwise dot‑product across the sequence dimension:
   for each channel `h` compute `out_h = left_h @ right_h.T`.  This is a
   batched matrix‑multiply (`torch.matmul`) on tensors of shape
   (B, hidden, N, N).
5. Layer‑norm the result over the hidden dimension using the learned
   `to_out_norm` parameters, multiply by `out_gate`, and finally project back
   to `dim` with `to_out`.
6. Return the tensor of shape (B, N, N, dim).

Only the normalization step is executed in Triton; the remaining heavy
operations rely on highly‑optimized cuBLAS / cuDNN kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton fused LayerNorm (float32 input → float16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    inp_ptr,                # *float32  – input [P, C] (flattened)
    out_ptr,                # *float16  – output [P, C]
    weight_ptr,             # *float32  – weight [C]
    bias_ptr,               # *float32  – bias   [C]
    C,                      # int       – channel dimension
    eps,                    # float32   – epsilon
    BLOCK_SIZE: tl.constexpr   # compile‑time constant, e.g. 128
):
    """
    Row‑wise layer‑norm with per‑channel affine transformation.
    One program (block) processes one row of size C.
    """
    pid = tl.program_id(0)                     # row index 0 … P-1
    row_offset = pid * C

    # ------------------------------------------------------------------
    # 1) Compute mean
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    for offs in range(0, C, BLOCK_SIZE):
        cols = tl.arange(0, BLOCK_SIZE)
        mask = offs + cols < C
        ptr = inp_ptr + row_offset + offs + cols
        x = tl.load(ptr, mask=mask, other=0.0, dtype=tl.float32)
        sum_val += tl.sum(x, axis=0)

    mean = sum_val / C

    # ------------------------------------------------------------------
    # 2) Compute variance
    # ------------------------------------------------------------------
    var_val = tl.float32(0.0)
    for offs in range(0, C, BLOCK_SIZE):
        cols = tl.arange(0, BLOCK_SIZE)
        mask = offs + cols < C
        ptr = inp_ptr + row_offset + offs + cols
        x = tl.load(ptr, mask=mask, other=0.0, dtype=tl.float32)
        diff = x - mean
        var_val += tl.sum(diff * diff, axis=0)

    var = var_val / C
    inv_std = 1.0 / tl.sqrt(var + eps)        # 1/sqrt(var+eps)

    # ------------------------------------------------------------------
    # 3) Write normalized + affine transformed output (float16)
    # ------------------------------------------------------------------
    for offs in range(0, C, BLOCK_SIZE):
        cols = tl.arange(0, BLOCK_SIZE)
        mask = offs + cols < C
        ptr = inp_ptr + row_offset + offs + cols
        x = tl.load(ptr, mask=mask, other=0.0, dtype=tl.float32)

        # (x - mean) / sqrt(var+eps)
        y = (x - mean) * inv_std

        # affine: γ * y + β
        w = tl.load(weight_ptr + offs + cols, mask=mask, other=0.0, dtype=tl.float32)
        b = tl.load(bias_ptr   + offs + cols, mask=mask, other=0.0, dtype=tl.float32)
        y = y * w + b

        # store as float16
        tl.store(out_ptr + row_offset + offs + cols, y.to(tl.float16), mask=mask)

def custom_kernel(data):

    Forward pass of the outgoing TriMul operator.
    Parameters

    -------
    torch.Tensor
        Tensor of shape (bs, seq_len, seq_len, dim) with dtype torch.float32

    # Unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, config = data

    hidden = config["hidden_dim"]      # hidden dimension (always 128 in tests)
    eps = 1e-5

    # ------------------------------------------------------------------
    x = inp.contiguous()

    P = B * N * N                             # total number of rows for layernorm

    # LayerNorm parameters
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias   = weights["norm.bias"].contiguous()

    # ------------------------------------------------------------------
    # 1) Triton fused LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    x_norm_fp16 = torch.empty_like(x, dtype=torch.float16, device=x.device)
    # launch kernel: one program per row (P programs)
    grid = (P,)

    # we address memory linearly (row * dim + col).
    layernorm_kernel[grid](
        x,
        x_norm_fp16,
        norm_weight,
        norm_bias,
        dim,
        eps,
        BLOCK_SIZE=128,            # 128 works well for dim up to 384
    )

    x_norm = x_norm_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    x_flat = x_norm.view(-1, dim)          # (B*N*N, dim)

    # Projections
    left  = F.linear(x_flat, weights["left_proj.weight"])
    right = F.linear(x_flat, weights["right_proj.weight"])
    left  = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # Optional mask (if not nomask)
    if not config.get("nomask", True):
        # mask has shape (B,N,N); broadcast to hidden
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # Gating projections (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"])).view(B, N, N, hidden)
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"])).view(B, N, N, hidden)
    out_gate   = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"])).view(B, N, N, hidden)

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Pairwise multiplicative update via batched matmul
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) for efficient bmm per channel
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # out_perm[b, h] = left_perm[b, h] @ right_perm[b, h].T
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, hidden, N, N)

    # Back to (B, N, N, hidden)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Output layer‑norm + final gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 5) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_flat = out.view(-1, hidden)                     # (B*N*N, hidden)
    out_proj = F.linear(out_flat, weights["to_out.weight"])
    out_proj = out_proj.view(B, N, N, dim)              # (B, N, N, dim)

    return out_proj
