@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids:
    pid_i = tl.program_id(0)  # block row (i)
    pid_j = tl.program_id(1)  # block col (j)
    pid_bh = tl.program_id(2) # combined batch and head index

    # decode batch and head:
    bh = pid_bh
    b = bh // H
    h = bh % H

    # compute offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-range
    i_mask = i_offset < N
    j_mask = j_offset < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N
        # load left tile: shape [BLOCK_M, BLOCK_K]
        left_ptrs = left_ptr + b * stride_l_batch + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k + h * stride_l_h
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
        # load right tile: shape [BLOCK_N, BLOCK_K] -> but we need right[b,j,k,h], where we consider j and k swapped: right is accessed as b, j, k, h
        right_ptrs = right_ptr + b * stride_r_batch + j_offset[:, None] * stride_r_j + k_offset[None, :] * stride_r_k + h * stride_r_h
        right = tl.load(right_ptrs, mask=j_mask[:, None] & k_mask[None, :], other=0.0)
        # Compute dot product
        # left: (M,K) , right: (N,K) . For each (i,j), we need sum over k left[i,k] * right[j,k] .
        # We can use tl.dot(left, right, trans_b=True?) Actually tl.dot expects shapes (M,K) and (K,N) produce (M,N). So we need to make right transposed.
        # So we can compute acc += tl.dot(left, right, trans_b=True) or use tl.dot(left, right) after transposing.
        acc += tl.dot(left, right, trans_b=True)
    # store result
    out_ptrs = out_ptr + b * stride_o_batch + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j + h * stride_o_h
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for the core TriMul contraction: 
# out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute the tri‑multiplicative contraction."""
    pid_i = tl.program_id(0)          # block on the i (row) axis
    pid_j = tl.program_id(1)          # block on the j (col) axis
    pid_bh = tl.program_id(2)         # combined batch & head index

    # decode batch and head from the flattened index
    b = pid_bh // H
    h = pid_bh % H

    # base offsets for the i and j axes
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_base < N
    j_mask = j_base < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the k dimension in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_base < N

        # pointers for left[b,i,k,h]
        left_ptrs = left_ptr + \
            b * stride_l_batch + \
            i_base[:, None] * stride_l_i + \
            k_base[None, :] * stride_l_k + \
            h * stride_l_h

        # pointers for right[b,j,k,h]
        right_ptrs = right_ptr + \
            b * stride_r_batch + \
            j_base[:, None] * stride_r_j + \
            k_base[None, :] * stride_r_k + \
            h * stride_r_h

        # load tiles (mask out‑of‑bounds entries)
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)
        right = tl.load(right_ptrs,
                        mask=j_mask[:, None] & k_mask[None, :],
                        other=0.0)

        # accumulate: left (M,K) • right (N,K)^T → (M,N)
        acc += tl.dot(left, right, trans_b=True)

    # store the result
    out_ptrs = out_ptr + \
        b * stride_o_batch + \
        i_base[:, None] * stride_o_i + \
        j_base[None, :] * stride_o_j + \
        h * stride_o_h

    tl.store(out_ptrs, acc,
             mask=i_mask[:, None] & j_mask[None, :])

    """
    Wrapper around ``tri_mul_kernel``.  ``left`` and ``right`` are
    tensors of shape [B, N, N, H] (float32).  Returns a tensor of the
    same shape containing the contraction over the third axis.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    # ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,

        stride_l_i=stride_l[1],
        stride_l_k=stride_l[2],
        stride_l_h=stride_l[3],
        stride_r_batch=stride_r[0],

      * Linear projections & sigmoid gates.
      * Optional masking.
      * Triton‑accelerated contraction over the sequence dimension.
      * Output layer‑norm, gating and final linear projection.

    Args:

    dim = config["dim"]

    # 1) Input layer‑norm
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------

    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

    # ----------------------------------------------------------------------
    # 4) Optional mask (applied before gates)
    # ----------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [B, N, N, 1] for broadcasting
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ----------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 6) Core triangular multiplication using Triton
    # ----------------------------------------------------------------------
    # left/right are now [B, N, N, hidden_dim]
    out = tri_mul_torch(left, right)   # shape [B, N, N, hidden_dim]

    # ----------------------------------------------------------------------
    # 7) Output layer‑norm, gated scaling and final projection
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

"""

import triton

# ----------------------------------------------------------------------
# Triton kernel for the core contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    Grid:
        pid_i : block over i (rows)   → axis 0
        pid_j : block over j (cols)   → axis 1
        pid_bh: combined batch * head → axis 2
    """
    pid_i = tl.program_id(0)          # i‑block
    pid_j = tl.program_id(1)          # j‑block
    pid_bh = tl.program_id(2)         # batch*head index

    # Decode batch and head
    b = pid_bh // H
    h = pid_bh % H

    # Base indices for the i/j tiles
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_base < N
    j_mask = j_base < N

    # Accumulator (FP32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the k‑dimension in tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_base < N

        # Pointers to left[b,i,k,h]   (shape: M×K)
        left_ptrs = left_ptr + \
            b * stride_l_batch + \
            i_base[:, None] * stride_l_i + \
            k_base[None, :] * stride_l_k + \
            h * stride_l_h

        # Pointers to right[b,j,k,h]  (shape: N×K)
        right_ptrs = right_ptr + \
            b * stride_r_batch + \
            j_base[:, None] * stride_r_j + \
            k_base[None, :] * stride_r_k + \
            h * stride_r_h

        # Load tiles, masking out‑of‑bounds entries
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)
        right = tl.load(right_ptrs,
                        mask=j_mask[:, None] & k_mask[None, :],
                        other=0.0)

        # Dot product: left (M,K) • right (N,K)^T → (M,N)
        acc += tl.dot(left, right, trans_b=True)

    # Store the result with proper masking
    out_ptrs = out_ptr + \
        b * stride_o_batch + \
        i_base[:, None] * stride_o_i + \
        j_base[None, :] * stride_o_j + \
        h * stride_o_h

    tl.store(out_ptrs, acc,
             mask=i_mask[:, None] & j_mask[None, :])


def tri_mul_torch(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper calling ``tri_mul_kernel``.
    ``left`` / ``right`` : [B, N, N, H]  (float32, CUDA)
    Returns tensor of the same shape with the contraction performed.
    """
    assert left.is_cuda and right.is_cuda, "Inputs must be on CUDA"
    B, N, _, H = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out.stride()

    # Tiles – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_batch=stride_l[0],
        stride_l_i=stride_l[1],
        stride_l_k=stride_l[2],
        stride_l_h=stride_l[3],
        stride_r_batch=stride_r[0],
        stride_r_j=stride_r[1],
        stride_r_k=stride_r[2],
        stride_r_h=stride_r[3],
        stride_o_batch=stride_o[0],
        stride_o_i=stride_o[1],
        stride_o_j=stride_o[2],
        stride_o_h=stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps give good utilization on H100
    )
    return out


# ----------------------------------------------------------------------
# Public entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32, CUDA)
            - mask  : torch.Tensor [B, N, N] (bool or float) – may be ignored.
            - weights: dict with the following keys (all CUDA tensors):
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict containing at least
                "dim" (input channel dimension) and "hidden_dim" (proj dimension).
                May also contain "nomask": True/False.

    Returns:
        torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating pathways (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Optional mask – applied before the gates
    # ------------------------------------------------------------------
    if not nomask:
        # Broadcast mask over the hidden dimension
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core Tri‑multiplicative update (Triton)
    # ------------------------------------------------------------------
    # left/right shape: [B, N, N, hidden_dim]
    out = tri_mul_torch(left, right)   # -> [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, and final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
