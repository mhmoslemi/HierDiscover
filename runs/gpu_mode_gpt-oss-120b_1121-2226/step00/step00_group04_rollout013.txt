def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over dim
    # compute mean, var
    # Keep dtype as float32.
    x = input_tensor
    # Compute mean & var
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    # Apply weight bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if provided (mask may be None or boolean)
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Allocate output half tensor
    B, N, _, _ = left.shape
    out_half = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left.device)

    # Compute strides
    stride_lb, stride_li, stride_lk, stride_ld = left_half.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_half.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_half.stride()

    # Kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * hidden_dim * grid_m * grid_n
    tri_mul_outgoing_kernel[total_programs](
        left_half,
        right_half,
        out_half,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert out to float32
    out = out_half.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    Triton-accelerated TriMul (outgoing) forward pass.
    This implementation follows the reference PyTorch TriMul module:
      1. LayerNorm over the channel dimension.
      2. Linear projections to a hidden dimension (left/right) and three sigmoid gates.
      3. Optional pair‑wise mask.
      4. Element‑wise gating of left/right.
      5. Batched GEMM `out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]` performed

    keeping the final result in FP32.

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- 1. Input LayerNorm ----------
    # input_tensor shape: [B, N, N, dim]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ---------- 2. Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- 3. Optional masking ----------
    if mask is not None and not config.get("nomask", False):

        right = right * mask_

    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()

    left = left * left_gate

    stride_ob, stride_oi, stride_oj, stride_od = out_h.stride()

    grid_n = (N + BLOCK_N - 1) // BLOCK_N

        stride_od,

        BLOCK_K=BLOCK_K,
    )

    out = out_h.to(torch.float32)

"""
Triton‑accelerated TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module:
    1. LayerNorm over the channel dimension (dim).
    2. Two bias‑free linear projections (left/right) from dim → hidden_dim.
    3. Optional binary mask applied to the projected tensors.
    4. Three sigmoid gates (left, right, out) also from dim → hidden_dim.
    5. Element‑wise gating of the projected tensors.
    6. Core O(N³·hidden_dim) operation:
          out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       performed by a custom Triton kernel (batched GEMM per hidden channel).
    7. LayerNorm over the hidden dimension, followed by the out‑gate.
    8. Final bias‑free linear projection back to dim.

All heavy matrix‑multiply work (step 6) is done in FP16 inside Triton,
with FP32 accumulation to retain numerical quality.  The rest of the
pipeline stays in FP32 for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    L_ptr,                     # pointer to left tensor   [B, N, N, H]
    R_ptr,                     # pointer to right tensor  [B, N, N, H]
    O_ptr,                     # pointer to out tensor    [B, N, N, H]
    B, N, H,                   # batch, sequence length, hidden dim
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the outgoing TriMul:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The kernel processes one (batch, hidden) pair per program ID and
    tiles the i, j dimensions with BLOCK_M × BLOCK_N blocks while
    iterating over k in BLOCK_K‑sized chunks.
    """
    pid = tl.program_id(0)

    # Number of blocks in the M (i) and N (j) dimensions
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Decompose pid -> (b, h, block_m, block_n)
    block_n = pid % num_blocks_n
    pid //= num_blocks_n
    block_m = pid % num_blocks_m
    pid //= num_blocks_m
    h = pid % H
    b = pid // H

    # Coordinates of the current tile
    row = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    col = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    row_mask = row < N
    col_mask = col < N

    # Pointer to the output tile
    out_ptrs = (
        O_ptr
        + b * stride_ob
        + row[:, None] * stride_oi
        + col[None, :] * stride_oj
        + h * stride_od
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ---- Load left tile (M × K) ----
        a_ptrs = (
            L_ptr
            + b * stride_lb
            + row[:, None] * stride_li
            + k_offset[None, :] * stride_lk
            + h * stride_ld
        )
        a = tl.load(
            a_ptrs,
            mask=row_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)

        # ---- Load right tile transposed (K × N) ----
        # Right is indexed as (b, j, k, h); we need the (k, j) view.
        b_ptrs = (
            R_ptr
            + b * stride_rb
            + k_offset[:, None] * stride_rk
            + col[None, :] * stride_rj
            + h * stride_rd
        )
        b = tl.load(
            b_ptrs,
            mask=k_mask[:, None] & col_mask[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)

        # Accumulate the block product
        acc += tl.dot(a, b)

    # Write the result back in FP16
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=row_mask[:, None] & col_mask[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    heavy O(N³·hidden_dim) contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with tensors matching the module's parameters
            - config: dict containing at least "dim", "hidden_dim", and "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply optional mask (only if a mask tensor is supplied and not disabled)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid on linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply element‑wise gates to the projected tensors
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³·hidden_dim) contraction using Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast projection results to FP16 for the kernel (saving bandwidth)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output container in FP16
    out_fp16 = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float16, device=left_fp16.device
    )

    # Strides required by the kernel (batch, i/j, k, hidden)
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    # Tiling parameters – tuned for H100 (feel free to adjust)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * hidden_dim * grid_m * grid_n

    # Launch the Triton kernel
    tri_mul_outgoing_kernel[total_programs](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back‑to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
