    stride_cm, stride_cn,

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Triton need to import triton and tl

    # Step 1: LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Step 2: Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", False):
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask
    # dealing with potential None mask

    # Step 3: Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for matmul
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape for batched matmul
    # Move hidden_dim to batch axis
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    batch = B * hidden_dim
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    # Transpose right for matmul: shape (batch,K,N)
    right_t = right_flat.transpose(1,2).contiguous()

    # Allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Launch Triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute strides
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_bk = right_t.stride(1)
    stride_bn = right_t.stride(2)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    batch_stride_a = left_flat.stride(0)
    batch_stride_b = right_t.stride(0)
    batch_stride_c = out_flat.stride(0)

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             batch )

    batched_matmul_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        N, N, N,  # M,N,K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Synchronize
    # torch.cuda.synchronize()
 
    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).to(torch.float32)

    # Step after matmul: to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Multiply by out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with a Triton‑accelerated N³ contraction.

The heavy part of the TriMul module is the following einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is mathematically a batched matrix multiplication
    out_d = L_d @ R_dᵀ      (L_d,R_d ∈ ℝ^{N×N})
for every hidden channel d and batch b.  We implement this
contraction with a custom Triton kernel that treats each
pair (batch, hidden) as an independent GEMM.  All surrounding
operations (LayerNorm, linear projections, gating, final
normalisation and linear) are performed with PyTorch for
convenience.

The kernel works in fp16 for the loads and dot products while
accumulating in fp32 to keep numerical stability.  After the
matmul the result is cast back to fp32 for the remaining steps.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                 # *Pointer* to A [batch, M, K]
    b_ptr,                 # *Pointer* to B [batch, K, N]  (already transposed)
    c_ptr,                 # *Pointer* to C [batch, M, N]
    M, N, K,               # matrix sizes (runtime scalars)
    stride_am, stride_ak,  # strides of A (row‑major)
    stride_bk, stride_bn,  # strides of B (row‑major, B is K×N)
    stride_cm, stride_cn,  # strides of C
    batch_stride_a,        # stride to move to next batch in A
    batch_stride_b,        # stride to move to next batch in B
    batch_stride_c,        # stride to move to next batch in C
    BLOCK_M: tl.constexpr, # tile size in M dimension
    BLOCK_N: tl.constexpr, # tile size in N dimension
    BLOCK_K: tl.constexpr, # tile size in K dimension
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # tile index in M
    pid_n = tl.program_id(1)          # tile index in N
    pid_b = tl.program_id(2)          # batch (including hidden channel) index

    # ------------------------------------------------------------------
    # Offsets within the output tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the first block of the current tile
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + pid_b * batch_stride_a \
                + offs_m[:, None] * stride_am \
                + tl.arange(0, BLOCK_K)[None, :] * stride_ak

    b_ptrs = b_ptr + pid_b * batch_stride_b \
                + tl.arange(0, BLOCK_K)[:, None] * stride_bk \
                + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main loop over K dimension
    # ------------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for kk in range(num_k):
        # Global offset for the current K‑block (used only for masking)
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A and B tiles with proper masking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # FP16 dot product, accumulation in FP32
        acc += tl.dot(a, b)

        # Move pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_b * batch_stride_c \
                + offs_m[:, None] * stride_cm \
                + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom Triton‑enabled TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor        shape [B, N, N, dim]
            - mask_tensor : torch.Tensor        shape [B, N, N] (optional)
            - weights_dict: dict of torch.Tensors with keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config: dict with at least keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to fp16 for the massive N³ contraction
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # --------------------------------------------------------------
    # 6. Rearrange tensors so that (batch * hidden) is the batch dimension
    #    shape for the Triton kernel: [B*H, N, N]
    # --------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    left_flat = left_t.view(B * hidden_dim, N, N)    # [B*H, M, K]
    right_flat = right_t.view(B * hidden_dim, N, N)  # [B*H, K, N] (will be transposed)

    # We need Bᵀ in the GEMM => transpose the K‑N dimensions
    right_T = right_flat.transpose(1, 2).contiguous()  # [B*H, K, N]

    # Output placeholder
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # --------------------------------------------------------------
    # 7. Launch the Triton batched GEMM kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    stride_am = left_flat.stride(1)   # stride for M (row)
    stride_ak = left_flat.stride(2)   # stride for K (col)
    stride_bk = right_T.stride(1)     # stride for K
    stride_bn = right_T.stride(2)     # stride for N
    stride_cm = out_flat.stride(1)    # stride for M
    stride_cn = out_flat.stride(2)    # stride for N

    batch_stride_a = left_flat.stride(0)
    batch_stride_b = right_T.stride(0)
    batch_stride_c = out_flat.stride(0)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles in M
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles in N
        B * hidden_dim                 # batch × hidden
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_T,
        out_flat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before proceeding
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim] and cast to fp32
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N) \
                 .permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 9. Output LayerNorm (hidden_dim), gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
