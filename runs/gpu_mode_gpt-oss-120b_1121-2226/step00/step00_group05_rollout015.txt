def triangular_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
    h_offset: tl.int32
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_offsets = i_base + tl.arange(0, BLOCK_I)
    j_offsets = j_base + tl.arange(0, BLOCK_J)
    h_offsets = h_offset + tl.arange(0, BLOCK_H)

    mask_i = i_offsets < N
    mask_j = j_offsets < N
    mask_h = h_offsets < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k
    k = tl.zeros([], dtype=tl.int32)  # init 0
    # While loop:
    while k < N:
        # load left (i,k,h)
        left_ptrs = left_ptr + i_offsets[:, None] * stride_left_i + k * stride_left_k + h_offsets[None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_h[None, :], other=0.0, dtype=tl.float16).to(tl.float32)

        # load right (j,k,h)
        right_ptrs = right_ptr + j_offsets[:, None] * stride_right_j + k * stride_right_k + h_offsets[None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_h[None, :], other=0.0, dtype=tl.float16).to(tl.float32)

        # outer product accumulate
        # left shape (BLOCK_I, BLOCK_H)
        # right shape (BLOCK_J, BLOCK_H)
        # compute left[:, None, :] * right[None, :, :] => (BLOCK_I, BLOCK_J, BLOCK_H)
        # This broadcasting works
        acc += left[:, None, :] * right[None, :, :]

        k += 1

    # Store accumulator to out
    out_ptrs = out_ptr + i_offsets[:, None, None] * stride_out_i + j_offsets[None, :, None] * stride_out_j + h_offsets[None, None, :] * stride_out_h

    mask_store = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]

    tl.store(out_ptrs, acc, mask=mask_store)

"""
TriMul (outgoing) – fast Triton implementation of the core cubic update.

The operator works on a 4‑D pair tensor `x` of shape [B, N, N, C].
The forward pass consists of:
  1. Layer‑norm over the last channel (C).
  2. Two linear projections (`left_proj`, `right_proj`) to a hidden
     dimension H (no bias) – result `L` and `R`.
  3. Optional pair mask (broadcast over H) applied to `L` and `R`.
  4. Gated scaling of `L` and `R` via three learned sigmoids.
  5. **Cubic update**:
        out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
     This is implemented as a custom Triton kernel that iterates over the
     shared dimension `k` while accumulating an outer‑product over the
     (i, j) tile for a block of hidden channels `h`.
  6. Layer‑norm over the hidden dimension, multiplication with the
     output‑gate, and a final linear projection back to the original
     channel dimension.

Only the cubic reduction (step 5) is performed in Triton – the rest uses
standard PyTorch ops (which are already highly optimized).  The kernel
operates in fp16 for the inputs and accumulates in fp32, delivering the
final result in fp32.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr,          # fp16 * [B, N, N, H] (batch‑pointer already offset)
    right_ptr,         # fp16 * [B, N, N, H] (batch‑pointer already offset)
    out_ptr,           # fp32 * [B, N, N, H] (batch‑pointer already offset)
    # sizes
    N,                 # int32  – sequence length
    H,                 # int32  – hidden dimension
    # strides (in elements, not bytes)
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_j, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
    # runtime offset of the current hidden‑block
    h_offset: tl.int32,
):
    """Cubic reduction: out[i,j,h] = Σ_k L[i,k,h] * R[j,k,h].

    The kernel processes a tile of size BLOCK_I × BLOCK_J × BLOCK_H.
    `h_offset` selects which slice of the hidden dimension is handled.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # tile indices
    i = i0 + tl.arange(0, BLOCK_I)
    j = j0 + tl.arange(0, BLOCK_J)
    h = h_offset + tl.arange(0, BLOCK_H)

    # out‑of‑bounds masks
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Loop over the shared dimension k
    # ----------------------------------------------------------------------
    k = tl.int32(0)
    while k < N:
        # ----- load L[i, k, h] ------------------------------------------------
        # shape (BLOCK_I, BLOCK_H)
        left_ptrs = left_ptr + i[:, None] * stride_left_i \
                             + k * stride_left_k \
                             + h[None, :] * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_h[None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)

        # ----- load R[j, k, h] ------------------------------------------------
        # shape (BLOCK_J, BLOCK_H)
        right_ptrs = right_ptr + j[:, None] * stride_right_j \
                               + k * stride_right_k \
                               + h[None, :] * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_h[None, :],
                        other=0.0,
                        dtype=tl.float16).to(tl.float32)

        # ----- outer‑product accumulation --------------------------------------
        # left: (BLOCK_I, 1, BLOCK_H)
        # right: (1, BLOCK_J, BLOCK_H)
        # result contribution: (BLOCK_I, BLOCK_J, BLOCK_H)
        acc += left[:, None, :] * right[None, :, :]

        k += 1

    # ----------------------------------------------------------------------
    # Store the tile back to global memory
    # ----------------------------------------------------------------------
    out_ptrs = out_ptr + i[:, None, None] * stride_out_i \
                         + j[None, :, None] * stride_out_j \
                         + h[None, None, :] * stride_out_h
    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict with the following keys (all torch.Tensors on the same device)
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict containing at least:
                * "dim" (C)
                * "hidden_dim" (H)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    (x, mask, weights, cfg) = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    B, N, _, C = x.shape
    assert C == dim, "Input channel dimension does not match config['dim']"

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating scalars (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cubic reduction (custom Triton kernel)
    # ------------------------------------------------------------------
    # Cast to fp16 to reduce memory traffic; the kernel accumulates in fp32
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output tensor (float32)
    out = torch.empty_like(left_fp16, dtype=torch.float32)  # [B,N,N,H]

    # Strides (in elements)
    s_left_b, s_left_i, s_left_k, s_left_h = left_fp16.stride()
    s_right_b, s_right_j, s_right_k, s_right_h = right_fp16.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Tiling parameters – chosen to give good occupancy on an H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32   # hidden‑dim block (runtime offset handles remainder)

    # Grid over i‑tiles and j‑tiles; batch and hidden‑blocks are handled in Python loops
    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J))

    for b in range(B):
        # Batch‑specific base pointers (account for the batch stride)
        left_batch_ptr = left_fp16.data_ptr() + b * s_left_b * left_fp16.element_size()
        right_batch_ptr = right_fp16.data_ptr() + b * s_right_b * right_fp16.element_size()
        out_batch_ptr = out.data_ptr() + b * s_out_b * out.element_size()

        # Loop over hidden‑dimension blocks
        for h_start in range(0, hidden_dim, BLOCK_H):
            tri_mul_kernel[grid](
                left_ptr=left_batch_ptr,
                right_ptr=right_batch_ptr,
                out_ptr=out_batch_ptr,
                N=N,
                H=hidden_dim,
                stride_left_i=s_left_i,
                stride_left_k=s_left_k,
                stride_left_h=s_left_h,
                stride_right_j=s_right_j,
                stride_right_k=s_right_k,
                stride_right_h=s_right_h,
                stride_out_i=s_out_i,
                stride_out_j=s_out_j,
                stride_out_h=s_out_h,
                BLOCK_I=BLOCK_I,
                BLOCK_J=BLOCK_J,
                BLOCK_H=BLOCK_H,
                h_offset=h_start,
                num_warps=8,          # H100 performs best with ≥8 warps per block
            )

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑norm, output‑gate, final linear proj.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])      # back to [B,N,N,dim]

    return out
