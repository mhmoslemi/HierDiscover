def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # get epsilon
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Apply mask if provided
    if mask is not None:
        # mask shape [B,N,N]; may be bool or float
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Convert to FP16 before kernel to gain speed
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape

    # reshape for kernel: (B*H, N, N)
    left_flat = left_fp16.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_flat = right_fp16.permute(0,3,1,2).contiguous().view(B*H, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)  # fp16

    # Launch kernel
    _batched_matmul_kernel[
        (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B*H)
    ](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape back to [B,N,N, H]
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    # convert back to float32 for subsequent ops
    out = out.to(torch.float32)

    # to_out normalization and linear
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

import torch
import triton
import triton.language as tl

@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch index
    pid_m = tl.program_id(0)  # block row index
    pid_n = tl.program_id(1)  # block col index

    # compute starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers for a and b tiles
    a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_b * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k-dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # write back
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing Triangle Multiplicative Update (TriMul).
    The heavy N^3 reduction (einsum over the third axis) is performed as a batched
    matrix multiplication using a Triton kernel. All other linear layers, gating
    and layer‑norms are executed with PyTorch for simplicity.
    """
    import torch
    import triton
    import triton.language as tl

    # unpack
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm on input ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---------- Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ---------- Optional mask ----------
    if not config.get("nomask", False):
        # mask: [B, N, N] broadcast to last dim
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Triton kernel for the big einsum ----------
    # Convert to half precision for speed
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate output FP16 buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    # strides (in number of elements)
    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_flat.stride()
    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ---------- Output normalization, gating and final linear ----------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

"""
custom_kernel – Outgoing Triangle‑Multiplicative Update (TriMul)

The forward pass consists of:
1. Layer‑norm on the input tensor.
2. Two linear projections (left / right) followed by optional masking.
3. Three sigmoid‑gates (left, right, out) that are applied element‑wise.
4. A massive N³ reduction:
   out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]
   → for each hidden channel d this is an N×N matrix multiplication
     (left₍d₎ @ right₍d₎ᵀ).  The set of (B·hidden_dim) matmuls is executed
     on GPU by a Triton kernel that tiles the multiplication in 128×128×32
     blocks and accumulates in FP32.
5. Layer‑norm over the hidden dimension, multiplication with the out‑gate,
   and a final linear projection back to the original channel dimension.

Only the heavy N³ reduction is performed with Triton; the remaining
operations are carried out with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # const float16 *
    b_ptr,                     # const float16 *
    c_ptr,                     # float16 *
    M, N, K,                   # matrix sizes (int32)
    stride_a_batch, stride_a_m, stride_a_k,   # a strides (int64)
    stride_b_batch, stride_b_k, stride_b_n,   # b strides (int64)
    stride_c_batch, stride_c_m, stride_c_n,   # c strides (int64)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched (batch, hidden) N×N matrix multiply: C = A @ Bᵀ.
    A : [batch, M, K]   (float16)
    B : [batch, K, N]   (float16)  – we pass B in its original layout (batch, K, N)
    C : [batch, M, N]   (float16)
    """
    pid_b = tl.program_id(2)               # batch * hidden index
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block column

    # Offsets for the tile we are computing
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds elements
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the start of the tile in A and B
    a_ptrs = a_ptr + pid_b * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_b * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)   # (BLOCK_K, BLOCK_N)
        acc += tl.dot(a, b)                                                    # accumulate in FP32

        # Advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write the result back (store as FP16 to save bandwidth)
    c_ptrs = c_ptr + pid_b * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Execute the outgoing TriMul forward pass.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # Configuration
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    apply_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if apply_mask:
        # mask: [B, N, N]  →  [B, N, N, 1]
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy N³ reduction – batched matmul via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to half to improve throughput
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),      # blocks over rows
        triton.cdiv(N, BLOCK_N),      # blocks over columns
        B * hidden_dim,               # batch * hidden dimension
    )

    # Strides in number of elements
    a_batch, a_m, a_k = left_flat.stride()   # (batch, M, K)
    b_batch, b_k, b_n = right_flat.stride()  # (batch, K, N)
    c_batch, c_m, c_n = out_flat.stride()    # (batch, M, N)

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                   # M, N, K are all equal to seq_len
        a_batch, a_m, a_k,
        b_batch, b_k, b_n,
        c_batch, c_m, c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to the original 4‑D layout and convert to FP32
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
