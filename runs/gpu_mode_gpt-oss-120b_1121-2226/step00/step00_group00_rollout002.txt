def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # layernorm
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # compute mean and var per element across dim dim (last), shape (B,N,N,1)
    # Use torch.mean var
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # apply mask if provided
    if not config.get("nomask", True):
        # mask shape [B,N,N], unsqueeze
        mask_uns = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_uns
        right = right * mask_uns

    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # core einsum using triton
    # left/right shape [B,N,N,hidden_dim]
    out = torch.empty_like(left) # same shape
    triton_einsum_kernel(left, right, out) # defined later

    # to_out_norm layernorm over hidden dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    eps2 = 1e-5
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    out_normed = (out - mean_out) * torch.rsqrt(var_out + eps2)
    out_normed = out_normed * to_out_norm_weight + to_out_norm_bias

    # multiply gating
    out_normed = out_normed * out_gate

    # final linear to original dim
    out_final = torch.nn.functional.linear(out_normed, weights["to_out.weight"])

    return out_final

def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # combined b*h
    pid_i = tl.program_id(axis=1)
    pid_j = tl.program_id(axis=2)

    # compute batch and head indices
    b = pid // H
    h = pid % H

    # compute offsets (in #elements)
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base offsets
    base_left = b * stride_left_b + h * stride_left_h

"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3).

The heavy part of TriMul is the 4‑D Einstein sum

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batched matrix multiplication for every hidden head *h*.
This kernel fuses the N³ computation across all batches and heads, while
the remaining lightweight operations (LayerNorms, linear projections,
gating, final linear) are performed with PyTorch for clarity.

The kernel processes a block (BLOCK_M × BLOCK_N) of the output matrix,
loops over tiles of the reduction dimension K (BLOCK_K), and stores the
result back to the output tensor.  All tensors are assumed to be
contiguous, on the same device, and in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: computes out = left @ rightᵀ for each (batch, head) pair
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *const fp32*  [B,N,N,H]
    right_ptr,              # *const fp32*  [B,N,N,H]
    out_ptr,                # *fp32*        [B,N,N,H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiply for each head (hidden channel).

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    pid = tl.program_id(axis=0)      # batch*head dimension
    pid_i = tl.program_id(axis=1)    # output row tile
    pid_j = tl.program_id(axis=2)    # output col tile

    # derive batch and head indices
    b = pid // H
    h = pid % H

    # offsets inside the output matrix
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # base offsets for the three tensors (batch & head part)
    base_left  = b * stride_left_b  + h * stride_left_h
    base_right = b * stride_right_b + h * stride_right_h
    base_out   = b * stride_out_b   + h * stride_out_h

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # pointers to the current tiles
        a_ptrs = left_ptr + base_left + i_off[:, None] * stride_left_i + k_off[None, :] * stride_left_k
        b_ptrs = right_ptr + base_right + k_off[:, None] * stride_right_k + j_off[None, :] * stride_right_j

        # masks for out‑of‑bounds loads
        mask_a = (i_off[:, None] < N) & (k_off[None, :] < N)
        mask_b = (k_off[:, None] < N) & (j_off[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)      # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)      # (BLOCK_K, BLOCK_N)

        # GEMM on the current tile
        acc += tl.dot(a, b)                              # (BLOCK_M, BLOCK_N)

    # write back the result
    c_ptrs = out_ptr + base_out + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    mask_c = (i_off[:, None] < N) & (j_off[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def tri_mul_einsum(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the tri_mul_kernel.
    left / right: [B, N, N, H] float32 tensors (contiguous).
    Returns out with the same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements (torch gives element strides)
    sl = left.stride()
    sr = right.stride()
    so = out.stride()

    # block sizes – chosen to give good occupancy for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        sl[0], sl[1], sl[2], sl[3],
        sr[0], sr[1], sr[2], sr[3],
        so[0], so[1], so[2], so[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor:  torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict: mapping of weight names to tensors
            - config_dict: contains "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)    # True ⇒ mask is all‑ones

    # --------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool; cast to same dtype as x
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core Einstein sum: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with Triton for maximal throughput.
    # --------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left  = left.contiguous()
    right = right.contiguous()
    out = tri_mul_einsum(left, right)               # shape [B,N,N,hidden]

    # --------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim)
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=1e-5)

    # --------------------------------------------------------------
    # 7. Apply output gate and final linear projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
