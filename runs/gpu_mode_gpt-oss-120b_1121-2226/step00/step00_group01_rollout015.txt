    left_ptr, right_ptr, out_ptr,

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_K: tl.constexpr, HIDDEN: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute indices
    i = pid_i
    j = pid_j

    # bounds
    i_valid = i < N
    j_valid = j < N

    # Initialize accumulator
    acc = tl.zeros([HIDDEN], dtype=tl.float32) # shape (HIDDEN,)

    # Loop over k
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        # k block offsets
        k_offsets = kb * BLOCK_K + tl.arange(0, BLOCK_K)

        # mask for valid k
        k_mask = k_offsets < N

        # Load left slice for all k in block? We'll load vector per k in loop or whole block.
        # Simpler: loop over k within block:
        for kk in range(BLOCK_K):
            k = k_offsets[kk]
            if not (k < N):

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # Check mask: if mask_tensor is None or config['nomask'] True then treat mask as all ones
    if mask_tensor is None:
        mask = None
    else:
        mask = mask_tensor

    # We'll compute full pipeline, maybe using torch.float16 for internal compute
    # Convert to fp16 for speed: left_proj_weight etc. Should be same dtype as input; we can cast.
    # But final output is float32 must be.

    # Use of torch.nn.functional.layer_norm
    # Note: We must pass normalized_shape = (dim,)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # returns shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask exponential
    if mask is not None:
        # mask shape [B,N,N], get broadcast
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()  # shape [B,N,N,hidden]
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # TriMul core using batched GEMM
    # reshape
    B, N, _, H = left.shape  # H = hidden_dim
    # ensure types
    # optional casting to fp16 for speed.
    left_fp = left.to(torch.float16)
    right_fp = right.to(torch.float16)
    # reshape for batched gemm
    # bring hidden dim to batch: shape (B*H, N, N)
    left_reshape = left_fp.permute(0, 3, 1, 2).reshape(B*H, N, N)
    right_reshape = right_fp.permute(0, 3, 1, 2).reshape(B*H, N, N)

    # compute matmul left @ right^T
    # right transposed along the last two dims: shape (B*H, N, N)
    out_mat = torch.bmm(left_reshape, right_reshape.transpose(1, 2))  # (B*H,N,N)

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]
    out = out.to(torch.float32)  # cast back

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias)

    # multiply by out_gate
    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    # compute block start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for this batch
    A_batch = A_ptr + pid_b * stride_ab
    B_batch = B_ptr + pid_b * stride_bb
    C_batch = C_ptr + pid_b * stride_cb

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

    # left and right are [B,N,N,H], dtype float16
    B, N, _, H = left.shape
    # reshape
    left_matrix = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_matrix = right.permute(0,3,2,1).contiguous().view(B*H, N, N)   # transposed

    b_ptr = right_matrix.data_ptr()

def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.
    ...
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias)

    # Cast to fp16 for compute
    x_fp = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x_fp, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_fp, weights["right_proj.weight"].to(torch.float16))

    # Apply mask if present
    if mask is not None and not config.get("nomask", False):
        mask = mask.unsqueeze(-1).to(input_tensor.dtype)  # broadcast
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x_fp, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # TriMul core using Triton batched GEMM
    # Reshape to (batch*hidden, N, N) with right transposed
    B, N, _, _ = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # swap last two dims (k ↔ j)
    # flatten batch*hidden
    left_flat = left_mat.view(B * hidden_dim, N, N)
    right_flat = right_mat.view(B * hidden_dim, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Get strides
    stride_am = left_flat.stride(1)  # row stride (i)
    stride_ak = left_flat.stride(2)  # col stride (k)
    stride_bk = right_flat.stride(1)  # row stride (k) – because right_flat is K×N
    stride_bn = right_flat.stride(2)  # col stride (j)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)
    stride_ab = left_flat.stride(0)
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # launch kernel
    GRID_M = triton.cdiv(N, 128)
    GRID_N = triton.cdiv(N, 128)
    GRID_B = B * hidden_dim
    batched_matmul_kernel[(GRID_M, GRID_N, GRID_B)](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_am, stride_ak, stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=128, BLOCK_N=128, BLOCK_K=32
    )

    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden]
    out = out.to(torch.float32)  # cast back

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

Thus stride_bk = right_flat.stride(1) = N? Wait, after transposition to B_T shape (K,N). Let's re-evaluate:

Right original shape = (B,N,N,H). After permute(0,3,2,1) we get shape (B, H, N, N) where last two dims are (N,N) but swapped (the original indexes j,k become k,j). """
TriMul (outgoing) – Triton‑accelerated forward pass

The module follows the reference implementation in AlphaFold‑3:

1️⃣ Layer‑norm on the input tensor (last dimension = dim).
2️⃣ Linear projections (left_proj / right_proj) → shape [B, N, N, H].
3️⃣ Optional pairwise mask multiplication.
4️⃣ Gating (sigmoid (linear(x))) applied element‑wise to the left/right tensors.
5️⃣ Core “TriMul” operation:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batch of H independent N×N matrix‑multiplications
   (one for each hidden channel).  This is implemented with a
   Triton batched GEMM kernel (no explicit Python loops).
6️⃣ Layer‑norm on the hidden dimension, gating with out_gate, and a final
   linear projection back to dim.

All heavy arithmetic (steps 2‑5) is performed in FP16 for speed;
the final result is returned in FP32.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B^T) where each batch element
# corresponds to one hidden channel of the TriMul operation.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                # pointers to matrices (FP16)
    M, N, K,                # matrix dimensions (all = seq_len)
    stride_am, stride_ak,   # A: row‑stride, col‑stride
    stride_bk, stride_bn,   # B: row‑stride (k), col‑stride (n)
    stride_cm, stride_cn,   # C: row‑stride, col‑stride
    stride_ab, stride_bb, stride_cb,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    """
    Compute C[b] = A[b] @ B[b].T for every batch element b.
    A, B, C are stored as (batch, rows, cols) in row‑major order.
    """
    pid_m = tl.program_id(0)        # tile row index
    pid_n = tl.program_id(1)        # tile col index
    pid_b = tl.program_id(2)        # batch / hidden index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers offset to the selected batch element
    a_batch = A + pid_b * stride_ab
    b_batch = B + pid_b * stride_bb
    c_batch = C + pid_b * stride_cb

    # Accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K‑dimension in blocks
    # --------------------------------------------------------------
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # Load B tile (K × N) – note that B is already transposed
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Matrix multiplication for the tile
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the tile back to C (store in FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor : [B, N, N] or None
            - weights_dict: contains all linear / layer‑norm parameters
            - config_dict : {"dim": ..., "hidden_dim": ..., "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32 → keep precision for norm)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=norm_weight,
                                       bias=norm_bias)  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Cast to FP16 for the bulk of the computation
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (left / right)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the channel dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1).to(x_fp16.dtype)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear(x))
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: batch of H GEMMs (N×N @ N×N^T) via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape        # B = batch size, N = sequence length
    # reshape to (B*H, N, N) for the batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous()               # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()             # (B, H, N, N)  (k ↔ j)

    left_flat = left_mat.view(B * hidden_dim, N, N)   # A: (batch, M, K)
    right_flat = right_mat.view(B * hidden_dim, N, N) # B_T: (batch, K, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)   # C

    # Strides (row‑major layout)
    stride_am = left_flat.stride(1)   # = N
    stride_ak = left_flat.stride(2)   # = 1
    stride_bk = right_flat.stride(1)  # = N (row stride in K‑dim)
    stride_bn = right_flat.stride(2)  # = 1
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    stride_ab = left_flat.stride(0)   # batch stride = N*N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # Tile sizes – chosen to keep per‑program register usage modest
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    _batched_gemm_kernel[(grid_x, grid_y, grid_z)](
        left_flat,                      # A pointer
        right_flat,                     # B_T pointer
        out_flat,                       # C pointer
        N, N, N,                        # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]
    out = out.to(torch.float32)   # back to FP32 for the remaining ops

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=to_out_norm_w,
                                         bias=to_out_norm_b)

    out = out * out_gate  # broadcasted over hidden dim

    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]

    return out
