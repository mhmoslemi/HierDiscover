def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch index and hidden index
    b = pid0 // H
    h = pid0 % H

    # compute start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute range of i,j within block
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # mask for boundaries
    i_mask = i_range < N
    j_mask = j_range < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # left block shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + i_range[:,None]*stride_l_i + k_range[None,:]*stride_l_k
        left = tl.load(left_ptrs, mask = i_mask[:,None] & k_mask[None,:], other = 0.0).to(tl.float32)

    # store result

"""
TriMul (outgoing) forward implementation with a custom Triton kernel.

The heavy part of the TriMul operation is the 4‑D einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We fuse this computation into a Triton kernel that treats each
(batch, hidden) pair as an independent matrix‑multiply:
    left[b,:,:,d] @ right[b,:,:,d].T   →   out[b,:,:,d]

The kernel tiles the (i, j) plane with blocks (BLOCK_M × BLOCK_N) and
accumulates over the k‑dimension in blocks (BLOCK_K).  All
other linear layers, LayerNorms and gating are performed with
standard PyTorch ops (they are comparatively cheap).  The kernel
operates in float32 (loads are cast from float16 if desired) and
stores the result in float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel for the outgoing triangular multiplicative update
# ---------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # *float32   [B, N, N, H]
    right_ptr,         # *float32   [B, N, N, H]
    out_ptr,           # *float32   [B, N, N, H]
    B, N, H,           # int32      batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a (BLOCK_M × BLOCK_N) tile of the
    output matrix for a specific (batch, hidden) pair.
    """
    pid0 = tl.program_id(0)          # combined (batch, hidden) index
    pid_i = tl.program_id(1)         # i‑tile index
    pid_j = tl.program_id(2)         # j‑tile index

    # -----------------------------------------------------------
    # Decode batch and hidden indices from pid0
    # -----------------------------------------------------------
    b = pid0 // H                     # batch index
    h = pid0 %  H                     # hidden channel index

    # -----------------------------------------------------------
    # Compute starting coordinates of the tile
    # -----------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # Index ranges for the tile (including out‑of‑bounds mask)
    # ------------------------------------------------------------------
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    i_mask = i_range < N
    j_mask = j_range < N

    # ------------------------------------------------------------------
    # Accumulator for the tile
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in BLOCK_K steps
    # ------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # -----------------------------------------------------------
        # Load left tile: shape (BLOCK_M, BLOCK_K)
        #   left[b, i, k, h]
        # -----------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + h * stride_l_h
            + i_range[:, None] * stride_l_i
            + k_range[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None] & k_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        # -----------------------------------------------------------
        # Load right tile: shape (BLOCK_K, BLOCK_N)
        #   right[b, j, k, h]   (note the order of j and k)
        # -----------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + h * stride_r_h
            + k_range[:, None] * stride_r_k
            + j_range[None, :] * stride_r_i
        )
        right = tl.load(
            right_ptrs,
            mask=k_mask[:, None] & j_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        # -----------------------------------------------------------
        # Accumulate the matrix product for this k‑block
        # -----------------------------------------------------------
        acc += tl.dot(left, right)

    # -----------------------------------------------------------
    # Write the tile back to global memory
    # -----------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + h * stride_o_h
        + i_range[:, None] * stride_o_i
        + j_range[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


# ---------------------------------------------------------------
# Entry point called by the evaluator
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------
    # Optional mask handling
    # -----------------------------------------------------------
    use_mask = not config.get("nomask", True)
    if use_mask:
        mask = mask_tensor.unsqueeze(-1)                # [B, N, N, 1]
    else:
        mask = torch.ones_like(input_tensor[..., :1])   # all‑ones mask

    # -----------------------------------------------------------
    # LayerNorm on the input (dim)
    # -----------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------
    # Linear projections (no bias) and gating logits
    # -----------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # -----------------------------------------------------------
    # Apply mask (if any) and per‑pair gates
    # -----------------------------------------------------------
    left = left * mask * left_gate
    right = right * mask * right_gate

    # Ensure contiguity for Triton
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------
    # Compute the triangular multiplicative update via Triton
    # -----------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)

    # Strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tiling parameters – chosen to fit H100 tensor‑core size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Launch grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------
    # Final LayerNorm, output gate, and linear projection to dim
    # -----------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])  # bias=False as in reference

    return output
