def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = (input_tensor - input_tensor.mean(dim=-1, keepdim=True)) / torch.sqrt(input_tensor.var(dim=-1, unbiased=False, keepdim=True) + eps)
    x = x * norm_weight + norm_bias

    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        mask_exp = mask.unsqueeze(-1).type_as(x)
        left = left * mask_exp
        right = right * mask_exp
    else:
        # no mask: treat mask as 1
        pass

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # prepare for matmul
    # left: [B,N,N,hidden_dim]
    # right: same

    B, N, _, _ = left.shape

    # Permute to shape (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to [B*hidden_dim, N, N]
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)

    # transpose right for matmul: (batch, K, N)
    right_t = right_flat.transpose(1, 2).contiguous()

    # run triton matmul
    out_flat = batched_matmul_triton(left_flat, right_t)  # shape (B*hidden_dim, N, N)
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape (B,N,N,hidden_dim)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = (out - out.mean(dim=-1, keepdim=True)) / torch.sqrt(out.var(dim=-1, unbiased=False, keepdim=True) + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # apply out_gate
    out_norm = out_norm * out_gate

    # final linear
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out

"""
TriMul “outgoing” kernel (AlphaFold3 style) – forward only.

The computation is:
    x̂   = LayerNorm(x)                                 # [B,N,N,D]
    L   = Linear(x̂, W_left)    (no bias)              # [B,N,N,H]
    R   = Linear(x̂, W_right)   (no bias)              # [B,N,N,H]
    G_L = sigmoid(Linear(x̂, W_left_gate))             # [B,N,N,H]
    G_R = sigmoid(Linear(x̂, W_right_gate))            # [B,N,N,H]
    G_O = sigmoid(Linear(x̂, W_out_gate))              # [B,N,N,H]

    L = L * G_L * mask (optional)
    R = R * G_R * mask (optional)

    # Core “TriMul” – for each hidden channel h:
    #   out[...,i,j,h] = Σ_k L[...,i,k,h] * R[...,j,k,h]
    # This is a batched matrix‑multiply: (B·H, N, N) @ (B·H, N, N)^T
    # Implemented with a custom Triton kernel for high throughput.

    O   = LayerNorm(out, w_out_norm, b_out_norm)       # [B,N,N,H]
    O   = O * G_O
    out = Linear(O, W_out)                              # [B,N,N,D]

All tensors are on CUDA.  The matmul is the only part executed in Triton.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched matrix‑multiply kernel (A @ B^T)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A,                     # ptr, [batch, M, K]
    B,                     # ptr, [batch, K, N]
    C,                     # ptr, [batch, M, N] (output)
    BATCH: tl.constexpr,
    M:     tl.constexpr,
    N:     tl.constexpr,
    K:     tl.constexpr,
    stride_Ab: tl.constexpr, stride_Am: tl.constexpr, stride_Ak: tl.constexpr,
    stride_Bb: tl.constexpr, stride_Bk: tl.constexpr, stride_Bn: tl.constexpr,
    stride_Cb: tl.constexpr, stride_Cm: tl.constexpr, stride_Cn: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # --------------------------------------------------------------
    # Offsets for the current output tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the output tile
    c_ptrs = C + batch_id * stride_Cb \
               + offs_m[:, None] * stride_Cm \
               + offs_n[None, :] * stride_Cn

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # Load a tile of A (M x K)
        a_ptrs = A + batch_id * stride_Ab \
                   + offs_m[:, None] * stride_Am \
                   + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & k_mask[None, :],
                    other=0.0)

        # Load a tile of B (K x N)
        b_ptrs = B + batch_id * stride_Bb \
                   + offs_k[:, None] * stride_Bk \
                   + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # Matrix multiply for this K‑block and accumulate
        acc += tl.dot(a, b)          # a: (M,K), b: (K,N) -> (M,N)
    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    tl.store(c_ptrs,
             acc,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul C = A @ B  (A: [B, M, K], B: [B, K, N]).
    Returns C in float32.
    """
    assert A.is_contiguous() and B.is_contiguous()
    assert A.shape[0] == B.shape[0] and A.shape[2] == B.shape[1]

    batch, M, K = A.shape
    _, _, N = B.shape

    # Casting to float16 for bandwidth – accumulation stays in FP32
    a_fp16 = A.to(torch.float16)
    b_fp16 = B.to(torch.float16)

    # Output in FP32
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides (in elements)
    stride_Ab, stride_Am, stride_Ak = a_fp16.stride()
    stride_Bb, stride_Bk, stride_Bn = b_fp16.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    # Tunable block sizes – chosen for H100/Friendlies
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        a_fp16, b_fp16, C,
        batch, M, N, K,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M, BLOCK_N, BLOCK_K,
        num_warps=4,  # H100 can handle more; 4 is safe
    )
    return C

# ----------------------------------------------------------------------
# Helper: LayerNorm without bias (bias optional)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    # Normalise over the last dimension
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator using a Triton‑accelerated
    batched matrix multiplication.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # input channel dimension
    hidden_dim = config["hidden_dim"]  # projection dimension
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on channel dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] -> [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating vectors (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul (batched matmul across the sequence dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # [B,N,N,hidden_dim]
    # Move hidden dim into batch dimension: (B, hidden, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Flatten batch & hidden => (B*H, N, N)
    left_flat  = left_perm.view(B * hidden_dim, N, N)    # [B*H, M, K]
    right_flat = right_perm.view(B * hidden_dim, N, N)   # [B*H, J, K]

    # We need right transposed on the last two axes for the contraction:
    right_t = right_flat.transpose(1, 2).contiguous()   # [B*H, K, N]

    # Batched matmul using Triton – result: [B*H, N, N] in float32
    out_flat = _batched_matmul_triton(left_flat, right_t)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcasted over last dim

    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
