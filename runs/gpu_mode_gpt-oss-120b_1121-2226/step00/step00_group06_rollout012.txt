def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dtype = input_tensor.dtype
    device = input_tensor.device

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # layernorm
    x_normed = torch.nn.functional.layer_norm(input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"])

    # project
    left = torch.nn.functional.linear(x_normed, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_normed, weights["right_proj.weight"])

    # apply mask if present
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.nn.functional.linear(x_normed, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_normed, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_normed, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Triangular multiplicative update
    # allocate output tensor
    out = torch.empty((bs, N, N, hidden_dim), dtype=input_tensor.dtype, device=device)

"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input pair representation (last dim = dim).
2. Linear projections to a hidden dimension H (no bias).
3. Apply the optional pair mask (broadcast over H).
4. Compute three sigmoid gates (left, right and output) from the
   normalized input and multiply them into the projected tensors.
5. **Heavy part** – compute
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a per‑channel matrix multiplication
   (left[...,d] @ right[...,d].T) for every hidden channel d.
   This O(N³·H) kernel is written in Triton: the (i,j) plane is tiled
   (BLOCK_I×BLOCK_J), the reduction dimension k is tiled (BLOCK_K) and
   the channel dimension is processed in chunks (BLOCK_D).  For each
   tile we load a (I×K×D) block of ``left`` and a (K×J×D) block of
   ``right`` (k is the fast‑moving dimension for the right operand), then
   perform a per‑channel dot‑product with ``tl.dot`` and accumulate.
6. Layer‑norm over the hidden dimension, multiply by the output gate and
   a final linear projection back to ``dim``.

The kernel is deliberately simple, relies on Triton‑level parallelism and
uses FP32 arithmetic (the host code may cast to FP16 before the kernel
if desired).  Only the O(N³·H) part is executed in Triton; the remaining
operations use Torch’s highly‑optimised kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: triangular multiplicative update (outgoing)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # logical sizes
    stride_bl, stride_il, stride_kl, stride_hl, # left  strides
    stride_br, stride_ir, stride_kr, stride_hr, # right strides
    stride_bo, stride_io, stride_jo, stride_ho, # out   strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    # 3‑D grid : (i‑tile, j‑tile, batch)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    b  = pid_b

    # offsets within a tile
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in chunks of BLOCK_D
    # ------------------------------------------------------------------
    for d0 in range(0, H, BLOCK_D):
        offs_d = d0 + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H          # only used for the final store

        # accumulator for this chunk: shape (I, J, D)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # ------------------------------------------------------------------
        # k‑reduction
        # ------------------------------------------------------------------
        for k0 in range(0, N, BLOCK_K):
            offs_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left tile   : (I, K, D)
            left_ptrs = (
                left_ptr
                + b * stride_bl
                + offs_i[:, None, None] * stride_il
                + offs_k[None, :, None] * stride_kl
                + offs_d[None, None, :] * stride_hl
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            ).to(tl.float32)

            # right tile (k‑first) : (K, J, D)
            right_ptrs = (
                right_ptr
                + b * stride_br
                + offs_k[:, None, None] * stride_ir   # k is now the fast dimension
                + offs_j[None, :, None] * stride_kr
                + offs_d[None, None, :] * stride_hr
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                other=0.0,
            ).to(tl.float32)

            # per‑channel dot product: acc[...,d] += left[...,d] @ right[...,d]
            for d in range(BLOCK_D):
                # slice shapes: left (I,K), right (K,J)
                left_mat  = left[:, :, d]   # (I, K)
                right_mat = right[:, :, d]  # (K, J)
                acc[:, :, d] += tl.dot(left_mat, right_mat)

        # ------------------------------------------------------------------
        # Write the accumulated block to global memory
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + b * stride_bo
            + offs_i[:, None, None] * stride_io
            + offs_j[None, :, None] * stride_jo
            + offs_d[None, None, :] * stride_ho
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
        )
# ----------------------------------------------------------------------


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor[B, N, N, dim] (float32/float16)
        * mask_tensor  : torch.Tensor[B, N, N]  (bool or float). May be None.
        * weights_dict : mapping name -> torch.Tensor containing model parameters.
        * config_dict  : must contain "dim" and "hidden_dim".

    Returns
    -------
    torch.Tensor[B, N, N, dim]
        The transformed pair representation.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional pair mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Compute gating tensors (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triangular multiplicative update (heavy O(N³·H) part)
    # ------------------------------------------------------------------
    # Allocate output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Ensure contiguous layout for proper stride handling
    left  = left.contiguous()
    right = right.contiguous()
    out   = out.contiguous()

    # Strides (in elements, not bytes)
    sl = left.stride()
    sr = right.stride()
    so = out.stride()

    # Triton tile sizes – tuned for typical H100 performance.
    # They work for all allowed N and hidden_dim; the kernel masks the
    # remainder tiles.
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,   # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,   # j‑tiles
        B,                               # batch dimension
    )

    trimul_kernel[
        grid
    ](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_bl=sl[0],
        stride_il=sl[1],
        stride_kl=sl[2],
        stride_hl=sl[3],
        stride_br=sr[0],
        stride_ir=sr[1],
        stride_kr=sr[2],
        stride_hr=sr[3],
        stride_bo=so[0],
        stride_io=so[1],
        stride_jo=so[2],
        stride_ho=so[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,               # modest warp count for this kernel
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim layer‑norm, output gating & final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
