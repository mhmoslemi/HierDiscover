@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets within M,N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers offsets
    left_ptrs = left_ptr + pid_batch * stride_left_batch + offs_m[:, None] * stride_left_m + 0 * stride_left_k
    out_ptrs = out_ptr + pid_batch * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a_ptrs = left_ptr + pid_batch * stride_left_batch + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (N x K) but we need transposed => shape K x N
        b_ptrs = right_ptr + pid_batch * stride_right_batch + offs_n[None, :] * stride_right_n + offs_k[:, None] * stride_right_k
        b = tl.load(b_ptrs, mask=mask_n[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)  # a: MxK, b: KxN -> MxN

    # Write out
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Extract weights: they are torch tensors with appropriate shapes.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # LayerNorm x
    # Compute mean and variance across last dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:
        expanded_mask = mask.unsqueeze(-1)    # shape [B,N,N,1]
        left = left * expanded_mask
        right = right * expanded_mask

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # TriMul kernel: compute out = left @ right.T per hidden channel using Triton
    # Permute and reshape to batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape

    # Reshape to (B*H, N, N)
    left_mat = left_perm.reshape(B * H, N, N)
    right_mat = right_perm.reshape(B * H, N, N)

    # we will cast to float16 for kernel compute to improve speed, but accumulate fp32.
    left_mat_fp16 = left_mat.to(torch.float16)
    right_mat_fp16 = right_mat.to(torch.float16)

    out_mat = torch.empty_like(left_mat, dtype=torch.float32)  # keep fp32

    # Launch Triton gemm kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    # Strides
    stride_left_batch = left_mat_fp16.stride(0)
    stride_left_m = left_mat_fp16.stride(1)
    stride_left_k = left_mat_fp16.stride(2)
    stride_right_batch = right_mat_fp16.stride(0)
    stride_right_m = right_mat_fp16.stride(1)   # same as N
    stride_right_k = right_mat_fp16.stride(2)
    stride_out_batch = out_mat.stride(0)
    stride_out_m = out_mat.stride(1)
    stride_out_n = out_mat.stride(2)

    trimul_gemm_kernel[grid](
        left_mat_fp16, right_mat_fp16, out_mat,
        B * H, N, N, N,
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_m, stride_right_k,
        stride_out_batch, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape to original shape
    out = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous() # shape B,N,N,hidden

    # to_out_norm + out_gate + to_out
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # shape B,N,N,dim

    return out

@triton.jit
def trimul_gemm_kernel(
    A, B, C,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_m, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = A + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m
    c_ptrs = C + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M x K)
        a = tl.load(a_ptrs + offs_k[None, :] * stride_a_k,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load B tile (N x K) -> we need transposed (K x N)
        # B layout: row = i (m index) which corresponds to "j" in output, col = k.
        # For transposed we load B as K,N using stride for k and m.
        b = tl.load(B + pid_batch * stride_b_batch + offs_n[None, :] * stride_b_m + offs_k[:, None] * stride_b_k,
                    mask=mask_n[None, :] & mask_k[:, None],
                    other=0.0)

        # convert to float32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # a (M,K) dot b (K,N) -> (M,N)

    # Store result
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul forward (outgoing) implementation with a fused Triton GEMM kernel.

The algorithm follows the reference PyTorch version:

1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Point‑wise linear projections (left/right) and gating linear layers.
3. Apply optional mask, then multiply projections with their gates.
4. Perform the cubic “triangle‑multiplicative” update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel
   (left @ rightᵀ).  The heavy O(N³·H) part is executed by a custom
   Triton kernel that works in FP16 and accumulates in FP32.
5. Apply a second layer‑norm, an output gate and a final linear
   projection back to the original channel dimension.

The kernel processes a batch of (B·H) independent GEMMs of size N×N.
Mixed‑precision (FP16 inputs, FP32 accumulation) fully exploits
Tensor‑Cores on an H100, while the surrounding point‑wise work is done
with regular PyTorch calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    A,               # left  matrix  (batch*H, M, K)  – FP16
    B,               # right matrix  (batch*H, N, K)  – FP16 (will be read transposed)
    C,               # output       (batch*H, M, N)  – FP32
    batch, M, N, K,  # dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for many independent batches.
    A: (batch, M, K)   – row‑major, stride_a_m = K, stride_a_k = 1
    B: (batch, N, K)   – row‑major, we read it as (K, N) by swapping K/N strides.
    C: (batch, M, N)
    """

    pid_batch = tl.program_id(0)          # which (batch * hidden) we are processing
    pid_m = tl.program_id(1)              # block row index
    pid_n = tl.program_id(2)              # block col index

    # ----- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = A + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m
    b_base = B + pid_batch * stride_b_batch + offs_n[None, :] * stride_b_n
    c_ptr  = C + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- main loop over K ---------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (M, K)
        a = tl.load(
            a_base + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )
        # Load B tile transposed: we need Bᵀ -> (K, N)
        b = tl.load(
            b_base + offs_k[:, None] * stride_b_k,
            mask=mask_k[:, None] & mask_n[None, :],
            other=tl.float16(0.0)
        )
        # Up‑cast to FP32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # GEMM on the tile (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b)

    # ----- write back ----------------------------------------------------
    tl.store(
        c_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of model parameters (see skeleton)
        - config       : dict containing "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype torch.float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # layer‑norm parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device).to(dtype)
    norm_bias   = weights["norm.bias"].to(device).to(dtype)

    # ------------------------------------------------------------------
    # linear weights (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"].to(device).to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(device).to(dtype)

    left_gate_weight  = weights["left_gate.weight"].to(device).to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(device).to(dtype)
    out_gate_weight   = weights["out_gate.weight"].to(device).to(dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(device).to(dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device).to(dtype)

    to_out_weight = weights["to_out.weight"].to(device).to(dtype)   # shape (dim, hidden_dim)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣  Layer‑norm over the last dimension (C)
    # ------------------------------------------------------------------
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias                     # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (FP32, point‑wise)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcasted on channel axis)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣  TriMul – batched GEMM per hidden channel (heavy part)
    # ------------------------------------------------------------------
    # Rearrange tensors to (B, hidden, N, N) → (B*hidden, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_total = B * H

    left_mat  = left_perm.reshape(batch_total, N, N)     # [B*H, N, N]
    right_mat = right_perm.reshape(batch_total, N, N)    # [B*H, N, N]

    # Cast to FP16 for the kernel (Tensor‑Core friendly)
    left_fp16  = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)

    # Output will be accumulated in FP32 to retain full precision
    out_fp32 = torch.empty_like(left_mat, dtype=torch.float32, device=device)

    # -------------------------- launch parameters ----------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_total,                     # program_id(0) – each (batch * hidden) pair
        triton.cdiv(N, BLOCK_M),         # program_id(1) – tile rows
        triton.cdiv(N, BLOCK_N),         # program_id(2) – tile cols
    )

    # Strides for the contiguous (batch, M, K) layout
    stride_l_batch = left_fp16.stride(0)
    stride_l_m     = left_fp16.stride(1)   # = N   (row stride)
    stride_l_k     = left_fp16.stride(2)   # = 1   (col stride)

    stride_r_batch = right_fp16.stride(0)
    stride_r_n     = right_fp16.stride(1)  # = N   (row stride, i.e. j‑axis)
    stride_r_k     = right_fp16.stride(2)  # = 1

    stride_o_batch = out_fp32.stride(0)
    stride_o_m     = out_fp32.stride(1)    # = N
    stride_o_n     = out_fp32.stride(2)    # = 1

    _trimul_gemm_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        batch_total, N, N, N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_n, stride_r_k,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,                     # H100 can comfortably run 8 warps per block
    )

    # Reshape back to [B, N, N, hidden]
    out = out_fp32.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6️⃣  Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                # element‑wise output gate

    out = F.linear(out, to_out_weight)  # back to [B, N, N, dim]

    return out
