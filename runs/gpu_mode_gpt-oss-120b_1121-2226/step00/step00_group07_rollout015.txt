@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, hidden_dim,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(axis=2)   # 0 ... batch_hidden-1

    # Compute batch index and hidden dim index
    batch_idx = pid_batch // hidden_dim
    d_idx = pid_batch % hidden_dim

    # Offsets for this hidden channel
    # compute pointers offsets
    pid_i = tl.program_id(axis=0)
    pid_j = tl.program_id(axis=1)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load left tile (M,K) for this hidden channel
        left_ptrs = left_ptr + \
            batch_idx * stride_l_batch + \
            d_idx * stride_l_d + \
            offs_i[:, None] * stride_l_i + \

    a_ptr, b_ptr, c_ptr,
    N, hidden_dim,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bj, stride_bk,
    stride_ch, stride_ci, stride_cj,

    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # compute offsets

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # maybe also config may include "nomask"? To apply mask or not.
    # According to test config, there is "nomask": boolean. We'll handle.

    # Move to device
    device = input_tensor.device
    # Convert to appropriate dtype for speed: use fp16 for linear projections and intermediate
    # compute normalized x in fp32 then cast to fp16; or compute all in fp16 for speed.
    # We'll compute norm using torch.nn.functional.layer_norm (FP32) then cast to fp16.

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # or for speed: F.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # The shape: [B,N,N,dim]

    # Linear projections: left_proj = x_norm @ left_proj_weight.T etc.
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # gating:
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # mask
    if not config.get("nomask", True):
        # mask shape [B,N,N]
        mask = mask_tensor.unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask
    # else: no mask needed, skip

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to FP16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute and reshape for kernel
    B, N, _, hidden_dim_check = left.shape
    assert hidden_dim_check == hidden_dim

    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, hidden_dim, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # Prepare output
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # we want float32 output

    # Launch Triton kernel
    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64  # adjust maybe 32 for hardware

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_hidden,
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),  # note order: stride_bh, stride_bk, stride_bn
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # reshape out_flat back:
    out_perm = out_flat.view(B, hidden_dim, N, N)  # [B, hidden_dim, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # shape [B, N, N, hidden_dim]

    # Now to_out_norm and gating
    # layernorm over hidden_dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate (broadcast)
    out = out * out_gate

    # Linear projection to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul outgoing kernel – a fused implementation of the AlphaFold3 triangular
multiplicative update (the “outgoing” variant).

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Linear projections → left/right tensors of shape [B, N, N, H].
3. Optional mask (broadcasted on the hidden dimension) and sigmoid gates.
   left = left * left_gate, right = right * right_gate.
4. Fuse the heavy cubic computation:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (einsum
      "... i k d, ... j k d -> ... i j d")
   This is a batched matrix‑multiply `left @ rightᵀ` for every hidden
   channel `h`. We reorganise the tensors to shape
   [B·H, N, N] and launch a custom Triton kernel that computes
   the product in FP16 (inputs) with FP32 accumulation.
5. Reshape the result back to [B, N, N, H] and apply a second
   Layer‑Norm (over H), an output gate, and a final linear projection
   back to the original `dim`.
6. The function returns a float‑32 tensor of shape [B, N, N, dim].

The Triton kernel is a classic tiled GEMM with three‑dimensional grid:
   axis‑0 → i‑tile, axis‑1 → j‑tile, axis‑2 → (batch, hidden) index.
It loads tiles of `left` and the transposed view of `right`,
accumulates in FP32 and stores the result in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication A @ Bᵀ
#   A : [BH, M, K]   (float16)
#   B : [BH, N, K]   (float16) – we treat Bᵀ inside the kernel
#   C : [BH, M, N]   (float32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # (batch * hidden) index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K‑dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ------------------- load A (M x K) -------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)       # FP16 → FP32

        # ------------------- load Bᵀ (K x N) -------------------
        # B is originally (N, K); we load its transpose as (K, N)
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)       # FP16 → FP32

        # ------------------- inner product -------------------
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------- write back -------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left_proj * left_gate          # float32
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the cubic kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6. Re‑order to [B, hidden, N, N] and flatten batch*hidden
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]   (fp16)
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]   (fp16)

    # Output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # small enough to fit in shared memory

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # each (batch, hidden) pair
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)                 # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()               # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # ------------------------------------------------------------------
    # 10. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B, N, N, dim]

    return out
```
