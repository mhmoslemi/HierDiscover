    # LayerNorm (faster via torch)
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask handling
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Triton matmul to compute out = left @ right_T over k
    B, N, _, _ = left.shape  # B,N,N,hidden_dim
    # reshape to (B, hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous()
    right_mat = right.permute(0, 3, 1, 2).contiguous()
    # transpose right over last two dims to get right_T
    right_T = right_mat.transpose(-2, -1).contiguous()
    # flatten batch*hidden_dim
    batch_hd = B * hidden_dim
    left_flat = left_mat.view(batch_hd, N, N)
    right_flat = right_T.view(batch_hd, N, N)
    out_flat = torch.empty_like(left_flat)

    # launch kernel
    launch_triton_gemm(left_flat, right_flat, out_flat, N, N, N)

    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul "outgoing" forward pass (AlphaFold3 style) implemented with Triton.

Algorithm
---------
1. Layer‑norm the input (float32) over the feature dimension (dim).
2. Linear projections -> left/right (shape B,N,N,H) and three sigmoidal gates.
3. Apply mask (if present) and element‑wise gate multiplications.
4. Core operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply left @ rightᵀ for every hidden channel h.
   Implemented with a custom Triton GEMM kernel (no PyTorch matmul).
5. Layer‑norm over the hidden dimension, element‑wise multiplication by out_gate.
6. Final linear projection back to the original feature dimension.

The kernel is a straightforward blocked GEMM with float32 accumulation.
It works for B∈{1,2}, N up to 1024 and hidden_dim = 128 (the
configuration used by AlphaFold3).  The rest of the pipeline uses
torch.nn.functional for simplicity and to keep the code short.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# --------------------------------------------------------------------
# Triton batched GEMM kernel (left @ rightᵀ)
# --------------------------------------------------------------------
@triton.jit
def _triton_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # a: (batch, M, K)
    stride_bk, stride_bn,          # b: (batch, K, N)
    stride_cm, stride_cn,          # c: (batch, M, N)
    stride_ab, stride_bb, stride_cb,  # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where A.shape=(B,M,K), B.shape=(B,K,N)."""
    pid = tl.program_id(0)

    # ----------------------------------------------------------------
    # Grid decomposition: (batch, block_i, block_j)
    # ----------------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_pid_m * num_pid_n)
    pid_in_batch = pid % (num_pid_m * num_pid_n)
    pid_m = pid_in_batch // num_pid_n
    pid_n = pid_in_batch % num_pid_n

    # ----------------------------------------------------------------
    # Offsets for the current block
    # ----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ----------------------------------------------------------------
    # Pointers for the current batch
    # ----------------------------------------------------------------
    a_batch = a_ptr + batch * stride_ab
    b_batch = b_ptr + batch * stride_bb
    c_batch = c_ptr + batch * stride_cb

    # ----------------------------------------------------------------
    # Accumulator
    # ----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------------
    # Loop over the K dimension in tiles
    # ----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < K

        # A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & mask_k[None, :],
                    other=0.0)

        # B: (BLOCK_K, BLOCK_N)  (note the transpose is already built‑in)
        b_ptrs = b_batch + (cur_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & (offs_n[None, :] < N),
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # ----------------------------------------------------------------
    # Store the result
    # ----------------------------------------------------------------
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    mask_store = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_store)


def _launch_triton_gemm(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor):
    """
    A: (batch, M, K)   float32, contiguous
    B: (batch, K, N)   float32, contiguous
    C: (batch, M, N)   float32, contiguous (output)
    """
    assert A.is_contiguous()
    assert B.is_contiguous()
    assert C.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    # Strides in element units (not bytes)
    stride_ab, stride_am, stride_ak = A.stride()
    stride_bb, stride_bk, stride_bn = B.stride()
    stride_cb, stride_cm, stride_cn = C.stride()

    # Block sizes – tuned for H100 FP32
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (M + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = batch * num_pid_m * num_pid_n

    _triton_gemm_kernel[total_programs](
        A, B, C,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # No explicit sync needed; the caller can sync if required.


# --------------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul "outgoing" module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------
    # Unpack arguments
    # ----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # layer‑norm epsilon (same as nn.LayerNorm default)

    # ----------------------------------------------------------------
    # Grab weights (all are bias‑free Linear layers except LayerNorms)
    # ----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]    # (hidden_dim, dim)

    left_gate_weight = weights["left_gate.weight"]      # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]    # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]        # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (hidden_dim,)

    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # ----------------------------------------------------------------
    # 1) Input LayerNorm
    # ----------------------------------------------------------------
    # x : [B,N,N,dim] → normalized across last dim
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------
    # Shape after projection: [B,N,N,hidden_dim]
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ----------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ----------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask is [B,N,N]; we need [B,N,N,1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype=left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------
    # 4) Gates (sigmoid after linear)
    # ----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------
    # 5) Core TriMul: out = left @ rightᵀ over the shared sequence dim (k)
    # ----------------------------------------------------------------
    # Prepare tensors for the batched GEMM.
    #   left_mat  : (B, hidden_dim, N, N)
    #   right_mat : (B, hidden_dim, N, N)
    #   right_T   : (B, hidden_dim, N, N)   = right_mat transposed on the last two axes
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    right_T = right_mat.transpose(-2, -1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_mat.shape

    # Merge batch and hidden_dim → single batch dimension for the kernel
    left_flat = left_mat.view(B * H, N, N)
    right_flat = right_T.view(B * H, N, N)
    out_flat = torch.empty_like(left_flat)  # (B*H, N, N)

    # Launch Triton GEMM (left @ right_T)
    _launch_triton_gemm(left_flat, right_flat, out_flat)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------
    # 6) Output LayerNorm (over hidden_dim)
    # ----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ----------------------------------------------------------------
    # 7) Multiply by out_gate (same shape as out) and final projection
    # ----------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # (B,N,N,dim)

    return out
