def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # Compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks to avoid out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # Write back C
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # assign variables
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LN on input
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=weights["norm.weight"],
                                            bias=weights["norm.bias"],
                                            eps=eps)

    # Mask handling
    if mask is None or config.get("nomask", False):
        mask_f = 1.0
    else:
        mask_f = mask.unsqueeze(-1).to(x_norm.dtype)

    # Projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask
    left = left_proj * mask_f
    right = right_proj * mask_f

    # Gating
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Convert to half for high-throughput GEMM
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute to (B, hidden_dim, N, N)
    left_h_perm = left_h.permute(0,3,1,2).contiguous()
    right_h_perm = right_h.permute(0,3,1,2).contiguous()
    # Transpose right for GEMM
    right_h_T = right_h_perm.transpose(-2,-1).contiguous()

    B, H, N, _ = left_h_perm.shape  # each is B, hidden_dim, N, N
    # flatten batch*hidden_dim dimensions
    batch = B * H
    A = left_h_perm.view(batch, N, N)
    Bmat = right_h_T.view(batch, N, N)

    # Call Triton GEMM
    C = batched_gemm(A, Bmat)

    # Reshape back
    out_h_perm = C.view(B, H, N, N)
    out = out_h_perm.permute(0,2,3,1).contiguous()  # shape (B,N,N,hidden_dim)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out,
                                              normalized_shape=(hidden_dim,),
                                              weight=weights["to_out_norm.weight"],
                                              bias=weights["to_out_norm.bias"],
                                              eps=eps)

    # Apply out gate
    out_norm = out_norm * out_gate

    # Final linear
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return output
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # batch base pointers
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # compute the start offsets of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp16 (fast tensor cores)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=tl.float16(0.0))

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def batched_gemm(A, B, BLOCK_M=128, BLOCK_N=128, BLOCK_K=32):
    """
    Batched matrix multiplication C = A @ B   (A: [B, M, K], B: [B, K, N])
    The implementation runs a Triton kernel.
    """
    assert A.shape[0] == B.shape[0]
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        A, B, C,
        batch, M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    Implements the forward pass of AlphaFold3's TriMul module with:
      * LayerNorm on the input.
      * Two gated linear projections (left/right) with optional mask.
      * A batched GEMM (sum over the third pair dimension) performed by a Triton kernel.
      * Output LayerNorm, out‑gate, and final linear projection back to input dimension.
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input layer‑norm
    # -----------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # -----------------------------------------------------------------
    # 2. Optional mask handling (broadcast to the feature dim)
    # -----------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_f = 1.0
    else:
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]

    # -----------------------------------------------------------------
    # 3. Linear projections
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # apply mask
    left = left * mask_f
    right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Prepare tensors for the Triton GEMM (half‑precision)
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # reshape to (B, hidden_dim, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # right needs to be transposed in the last two dimensions for A @ B^T
    right_h_T = right_h.transpose(-2, -1).contiguous()

    B, H, N, _ = left_h.shape
    batch = B * H

    # flatten batch*hidden_dim into a single GEMM batch dimension
    A = left_h.view(batch, N, N)          # (B*H, M=N, K=N)
    Bmat = right_h_T.view(batch, N, N)    # (B*H, K=N, N=N)

    # -----------------------------------------------------------------
    # 6. Triton batched GEMM implementing the summation over the third axis
    # -----------------------------------------------------------------
    C = batched_gemm(A, Bmat)   # (B*H, N, N)  in fp16

    # reshape back to (B, N, N, hidden_dim)
    C = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = C.to(torch.float32)   # back to FP32 for the remaining ops

    # -----------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate

    output = F.linear(out, weights["to_out.weight"])
    return output

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The implementation follows the reference PyTorch module:
    1. LayerNorm over the last dimension.
    2. Linear projections (left/right) with optional pairwise mask.
    3. Gated by sigmoid(linear(x)).
    4. Core operation:  out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d] .
       This is a batched matrix multiplication (A @ Bᵀ) over the
       pair dimension and is executed by a custom Triton kernel.
    5. Output LayerNorm, out‑gate multiplication and a final linear
       projection back to the input dimension.

All heavy arithmetic (the N³ reduction) is performed in FP16 on
Tensor‑cores; the surrounding element‑wise ops stay in FP32 for
numerical stability. The kernel is written for H100 GPUs and targets
sub‑millisecond latency for the supported shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM C = A @ B    (A: [batch,M,K], B: [batch,K,N]) in FP16."""
    batch_id = tl.program_id(0)          # which batch element (B*hidden_dim)
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block column

    # base pointers for this batch
    a_batch = a_ptr + batch_id * stride_a_batch
    b_batch = b_ptr + batch_id * stride_b_batch
    c_batch = c_ptr + batch_id * stride_c_batch

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP16 (Tensor‑core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        acc += tl.dot(a, b)   # FP16 dot, accumulation in FP16

    # write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm(A: torch.Tensor, B: torch.Tensor,
                 BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Wrapper around the Triton GEMM kernel.
    A: [batch, M, K]  (FP16)
    B: [batch, K, N]  (FP16)   – B is already transposed w.r.t. the original right tensor.
    Returns C = A @ B of shape [batch, M, N] (FP16).
    """
    assert A.shape[0] == B.shape[0]
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Mismatched inner dimensions"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    grid = (batch,
            (M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gemm_kernel[grid](
        A, B, C,
        batch, M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

        - config       : dict containing at least "dim" and "hidden_dim"

    Returns

    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"],
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Mask handling (broadcast to last dim)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):

    else:

    # ------------------------------------------------------------------
    # 3. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left = left * mask_f

    # ------------------------------------------------------------------

    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
