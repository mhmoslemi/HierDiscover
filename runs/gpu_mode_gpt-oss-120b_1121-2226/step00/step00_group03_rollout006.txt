    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    batch2, K2, N = b.shape
    assert batch == batch2
    assert K == K2
    # output shape
    out = torch.empty(batch, M, N, device=a.device, dtype=a.dtype)
    # compute strides in number of elements
    stride_a_batch, stride_am, stride_ak = a.stride()
    stride_b_batch, stride_bk, stride_bn = b.stride()

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # extract dims
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # 1. layernorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # 2. linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [b,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # 3. apply mask if any
    if mask is not None:
        # make mask shape broadcastable
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_
    # 4. gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    left = left * left_gate
    right = right * right_gate
    # 5. compute bilinear multiplication via Triton
    # prepare tensors for kernel
    # permute to bring hidden_dim to batch dimension
    b, N, _, _ = left.shape
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # reshape to (batch*hidden_dim, N, N)
    batch_hidden = b * hidden_dim
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)
    # transpose right for multiplication
    right_T = right_flat.transpose(1,2).contiguous()
    # cast to fp16 for kernel
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_T.to(torch.float16)
    # call triton kernel
    out_fp16 = triton_matmul(left_fp16, right_fp16)  # shape (batch_hidden,N,N)

"""
custom_kernel -- Fast “outgoing” Triangle‑Multiplicative Update (TriMul)

The implementation follows the reference PyTorch TriMul module but off‑loads the
O(N³·hidden_dim) tensor contraction to a hand‑written Triton kernel.  The steps
are:

1. Layer‑norm the input tensor.
2. Compute left/right projections (linear layers without bias) and three
   sigmoidal gates.
3. Apply an optional pair‑wise mask and the element‑wise gates.
4. Reshape the tensors so that the hidden dimension becomes part of the batch
   dimension, then perform a batched GEMM:
        out[b·d, i, j] = Σ_k left[b·d, i, k] * right_T[b·d, k, j]
   This is the expensive einsum in the reference code.  The GEMM is implemented
   in Triton with 128×128 tiles and 32‑wide K‑tiles and runs in FP16 while
   accumulating in FP32.
5. Reshape the result back to [B, N, N, hidden_dim].
6. Apply a second layer‑norm, the output gate and the final linear projection
   back to the original feature dimension.

Only the core matrix multiplication (step 4) is written in Triton; the rest
uses fused PyTorch ops for simplicity and numerical stability.
"""

import torch.nn.functional as F

import triton.language as tl

# Triton kernel: batched FP16 GEMM with FP32 accumulation.
# ----------------------------------------------------------------------

def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes
    stride_a_batch, stride_am, stride_ak,    # A strides (batch, row, col)
    stride_b_batch, stride_bk, stride_bn,    # B strides (batch, row, col)
    stride_c_batch, stride_cm, stride_cn,    # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_z = tl.program_id(2)      # batch * hidden_dim
    pid_m = tl.program_id(0)      # block row index
    pid_n = tl.program_id(1)      # block col index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Guard against out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = a_ptr + pid_z * stride_a_batch
    b_batch = b_ptr + pid_z * stride_b_batch
    c_batch = c_ptr + pid_z * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tiles of A and B (FP16)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # FP16
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # FP16

        # Matrix‑multiply the tiles (accumulates in FP32)
        acc += tl.dot(a, b)

    # Write the result back as FP16 (cast from FP32)

    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for tensors A∈[B, M, K] and B∈[B, K, N].
    Both inputs are expected to be contiguous FP16 tensors.
    Returns C∈[B, M, N] as FP16.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    batch2, K2, N = b.shape
    assert batch == batch2 and K == K2, "Shape mismatch for batched GEMM"

    # Output allocation
    c = torch.empty(batch, M, N, dtype=torch.float16, device=a.device)

    # Strides (in element counts)
    stride_a_batch, stride_am, stride_ak = a.stride()
    stride_b_batch, stride_bk, stride_bn = b.stride()
    stride_c_batch, stride_cm, stride_cn = c.stride()

    # Tile sizes – chosen to fit well on H100 Tensor Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(M, BLOCK_M),   # blocks in M direction
        triton.cdiv(N, BLOCK_N),   # blocks in N direction
        batch,                     # one program per batch*hidden_dim
    )

    _batched_gemm_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_b_batch=stride_b_batch,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_c_batch=stride_c_batch,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for H100
    )
    return c


# ----------------------------------------------------------------------
# Forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (no gradients).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [batch, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    # ------------------------------------------------------------------
    # Configuration / shape extraction
    # ------------------------------------------------------------------
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape  # input is [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (already broadcastable)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Batched GEMM in Triton: out = left @ right^T (per hidden channel)
    # ------------------------------------------------------------------
    # Bring hidden_dim into the batch dimension for a plain 3‑D GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse batch & hidden_dim -> B*H
    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)                # [BH, M=N, K=N]
    right_mat = right_perm.view(BH, N, N).transpose(1, 2).contiguous()  # [BH, K=N, N]

    # Cast to FP16 for the kernel (accumulates in FP32)
    left_fp16  = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)

    # Triton GEMM
    out_fp16 = _batched_gemm_fp16(left_fp16, right_fp16)   # [BH, N, N]

    # Back to FP32
    out = out_fp16.to(torch.float32)

    # Restore original layout: [B, N, N, hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 7. Output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
