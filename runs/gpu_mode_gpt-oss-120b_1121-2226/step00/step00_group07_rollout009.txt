def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5

    # layernorm
    # compute mean and var across last dim (dim)
    # Can use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # flatten and linear projections
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # shape (B*N*N, dim)
    left = torch.nn.functional.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.nn.functional.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to batch*hidden
    left_perm = left.permute(0,3,1,2).contiguous()  # shape B, hidden, N, N
    right_perm = right.permute(0,3,1,2).contiguous()

    # Transpose right for matmul: (B, hidden, K, N)
    right_t = right_perm.transpose(-1, -2).contiguous()  # shape B, hidden, N, N (but swapped dim)

    # combine batch and hidden dims
    total_batches = B * hidden_dim
    A = left_perm.view(total_batches, N, N)  # shape [total_batches, M, K] where M=N, K=N
    B_t = right_t.view(total_batches, N, N)  # shape [total_batches, K, N] (but after transpose it's also N,N; we treat as K,N)
    # Actually we want shape (total_batches, M, K) for A and (total_batches, N, K) for B (need to treat B as (total_batches, N, K) then transpose inside kernel). But easier: we'll just compute C = A @ B_t using kernel expecting A[M,K] and B_t[K,N] as we pass B_t.

    # allocate output
    C = torch.empty((total_batches, N, N), dtype=torch.float32, device=left.device)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), total_batches)
    batch_matmul_kernel[grid](
        A_ptr=A, B_ptr=B_t, C_ptr=C,
        M=N, N=N, K=N,
        stride_am=A.stride(1), stride_ak=A.stride(2), stride_ab=A.stride(0),
        stride_bm=B_t.stride(1), stride_bk=B_t.stride(2), stride_bb=B_t.stride(0),
        stride_cm=C.stride(1), stride_cn=C.stride(2), stride_cb=C.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )
    # flatten x for linear
    B, N, _, _ = x.shape
    x_flat = x.view(-1, dim)  # (B*N*N, dim)

    # Cast everything to half for linear layers
    half_dtype = torch.float16

    left_proj_w = weights["left_proj.weight"].to(half_dtype)
    right_proj_w = weights["right_proj.weight"].to(half_dtype)
    left_gate_w = weights["left_gate.weight"].to(half_dtype)
    right_gate_w = weights["right_gate.weight"].to(half_dtype)
    out_gate_w = weights["out_gate.weight"].to(half_dtype)

    # Linear projections
    left = torch.nn.functional.linear(x_flat, left_proj_w).view(B, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(B, N, N, hidden_dim)

    # Apply mask if given
    if mask is not None:
        mask = mask.to(half_dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x_flat, left_gate_w).view(B, N, N, hidden_dim).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_w).view(B, N, N, hidden_dim).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w).view(B, N, N, hidden_dim).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left = left.to(half_dtype)
    right = right.to(half_dtype)

    # Permute for batch matmul
    left_perm = left.permute(0,3,1,2).contiguous()  # (B, hidden, N, N)
    right_perm = right.permute(0,3,1,2).contiguous()
    # Transpose right to (B, hidden, K, N)
    right_t = right_perm.transpose(-1, -2).contiguous()

    total_batches = B * hidden_dim
    A = left_perm.view(total_batches, N, N)  # (batch, M, K)
    B_t = right_t.view(total_batches, N, N)  # (batch, K, N) after transpose

    # allocate output
    out_mat = torch.empty((total_batches, N, N), dtype=torch.float32, device=device)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), total_batches)

    batch_matmul_kernel[grid](
        A, B_t, out_mat,
        M=N, N=N, K=N,
        stride_am=A.stride(1), stride_ak=A.stride(2), stride_ab=A.stride(0),
        stride_bm=B_t.stride(1), stride_bk=B_t.stride(2), stride_bb=B_t.stride(0),
        stride_cm=out_mat.stride(1), stride_cn=out_mat.stride(2), stride_cb=out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8
    )

    # reshape back to original shape
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,hidden)

    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,

    )

    out_flat = out_norm.view(-1, hidden_dim)
    final = torch.nn.functional.linear(out_flat, to_out_w).view(B, N, N, dim)

import triton
import triton.language as tl

@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bm, stride_bk, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ab
    b_batch = B_ptr + pid_b * stride_bb
    c_batch = C_ptr + pid_b * stride_cb

    # pointer to A block (M x K)
    a_ptrs = a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    # pointer to B block (K x N) -- note B is transposed (K,N)
    b_ptrs = b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute mask for the current k block
        k_valid = (tl.arange(0, BLOCK_K) + k * BLOCK_K) < K
        # load tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & k_valid[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=k_valid[:, None] & mask_n[None, :], other=0.0)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core

The module implements the forward of the “outgoing” triangle multiplicative
update used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (dim).
2. Linear projections (no bias) → left, right (shape [B, N, N, H]).
3. Optional mask multiplication (broadcast on the hidden dimension).
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) and apply to
   left / right / final output.
5. Core operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication across the sequence dimension.
   We fuse it in a custom Triton kernel that processes the hidden dimension
   as an extra batch dimension.
6. Layer‑norm the result (over H), multiply by `out_gate`,
   and a final linear projection back to `dim`.

The Triton kernel performs a classic blocked GEMM with fp16 inputs and a
fp32 accumulator, achieving the required O(N³·H) work with tensor‑core
efficiency on an H100.  All other “light‑weight” ops (norms, linear layers,
sigmoid) are delegated to PyTorch which already uses highly tuned kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A: [batch, M, K], Bᵀ: [batch, K, N])
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    # strides (in elements)
    stride_am, stride_ak, stride_ab,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # Bᵀ: (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where each hidden channel is treated as an independent batch."""
    pid_b = tl.program_id(2)                # batch (B*H)
    pid_m = tl.program_id(0)                # block row
    pid_n = tl.program_id(1)                # block column

    # --------------------------------------------------------------
    # Offsets of the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ab
    b_batch = B_ptr + pid_b * stride_bb
    c_batch = C_ptr + pid_b * stride_cb

    # Pointers to the current tile of A (M × K) and Bᵀ (K × N)
    a_ptrs = a_batch + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_batch + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulate over the reduction dimension K
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    num_k = tl.cdiv(K, BLOCK_K)

    for k in range(0, num_k):
        # Valid k‑range for the current tile
        k_valid = (tl.arange(0, BLOCK_K) + k * BLOCK_K) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)                     # fp16 -> fp32 accumulator
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write the output tile
    # --------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]  (may be None)
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    B, N, _, _ = x.shape
    # Flatten for linear layers
    x_flat = x.view(-1, dim)                     # [(B·N·N), dim]

    # ------------------------------------------------------------------
    # 2) Linear projections & gating (cast to fp16 for speed)
    # ------------------------------------------------------------------
    # Cast linear weights once
    w_left  = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    w_right = weights["right_proj.weight"].to(torch.float16, non_blocking=True)
    w_lgate = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    w_rgate = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    w_outgate = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    # Project
    left  = F.linear(x_flat, w_left).view(B, N, N, hidden_dim)   # fp16
    right = F.linear(x_flat, w_right).view(B, N, N, hidden_dim)  # fp16

    # Optional mask (broadcast on hidden dimension)
    if mask is not None:
        mask_f = mask.to(torch.float16, non_blocking=True).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates (sigmoid → fp16)
    left_gate  = F.linear(x_flat, w_lgate).view(B, N, N, hidden_dim).sigmoid().to(torch.float16, non_blocking=True)
    right_gate = F.linear(x_flat, w_rgate).view(B, N, N, hidden_dim).sigmoid().to(torch.float16, non_blocking=True)
    out_gate   = F.linear(x_flat, w_outgate).view(B, N, N, hidden_dim).sigmoid().to(torch.float16, non_blocking=True)

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Core multiplicative update (batched GEMM in Triton)
    # ------------------------------------------------------------------
    # Rearrange tensors to (batch, hidden, N, N) → flatten batch*hidden
    left_perm  = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # Transpose right so we can perform A @ Bᵀ
    right_t = right_perm.transpose(-1, -2).contiguous()   # [B, H, N, N] (now K×N)

    total_batches = B * hidden_dim
    A = left_perm.view(total_batches, N, N)   # (batch, M, K) where M=K=N
    B_T = right_t.view(total_batches, N, N)   # (batch, K, N)

    # Allocate output (float32 accumulator)
    out_mat = torch.empty((total_batches, N, N), dtype=torch.float32, device=device)

    # Tuning parameters – chosen to give high occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), total_batches)
    batch_matmul_kernel[grid](
        A, B_T, out_mat,
        M=N, N=N, K=N,
        stride_am=A.stride(1), stride_ak=A.stride(2), stride_ab=A.stride(0),
        stride_bk=B_T.stride(1), stride_bn=B_T.stride(2), stride_bb=B_T.stride(0),
        stride_cm=out_mat.stride(1), stride_cn=out_mat.stride(2), stride_cb=out_mat.stride(0),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8
    )

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # float32

    # ------------------------------------------------------------------
    # 4) Output layer‑norm, gating, final projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    out_norm = out_norm * out_gate  # broadcast gate (float16 → up‑cast to float32 automatically)

    # Final linear projection back to `dim` (float32)
    w_out = weights["to_out.weight"]                     # shape [dim, hidden_dim]
    out_flat = out_norm.view(-1, hidden_dim)
    final = F.linear(out_flat, w_out).view(B, N, N, dim)  # float32

    return final
