```
@triton.jit
def layer_norm_forward_kernel(
    X_ptr, Y_ptr,
    weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    stride_w, stride_b,
    eps: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # compute b,i,j per off
    b = offs // (N * N)
    residual = offs % (N * N)
    i = residual // N
    j = residual % N

    # Load input
    c_range = tl.arange(0, C)
    X_ptrs = X_ptr + b[:, None] * stride_xb + i[:, None] * stride_xi + j[:, None] * stride_xj + c_range[None, :] * stride_xc
    X = tl.load(X_ptrs, mask=mask[:, None], other=0.0)

    # Compute mean and var
    # sum across C
    sum_X = tl.sum(X, axis=1)  # shape (BLOCK)
    mean = sum_X * (1.0 / C)

    # subtract mean
    X_centered = X - mean[:, None]

    # Compute variance
    var = tl.sum(X_centered * X_centered, axis=1) * (1.0 / C)

    # Normalize
    inv_std = tl.rsqrt(var + eps)
    Y_norm = X_centered * inv_std[:, None]

    # Apply scale and shift (weight, bias)
    w = tl.load(weight_ptr + c_range * stride_w)  # shape (C)
    b_ = tl.load(bias_ptr + c_range * stride_b)
    Y = Y_norm * w[None, :] + b_[None, :]

    # Store output
    Y_ptrs = Y_ptr + b[:, None] * stride_yb + i[:, None] * stride_yi + j[:, None] * stride_yj + c_range[None, :] * stride_yc
    tl.store(Y_ptrs, Y, mask=mask[:, None])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # We assume input Tensor shape B, N, N, dim

    # LayerNorm over dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Triton layernorm
    x_norm = layer_norm_triton(input_tensor, norm_weight, norm_bias, eps=eps)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # Linear without bias
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if mask_tensor is not None:
        # mask shape [B,N,N] might be bool or float. Convert to float.
        mask = mask_tensor[..., None].to(dtype=left.dtype)
        left = left * mask
        right = right * mask

    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute Einsum via batched matmul
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # note transpose of last two dims
    B, H, N, N = left_perm.shape
    # Reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # Use torch.bmm
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)
    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden_dim)

    # to_out_norm layernorm over hidden_dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = layer_norm_triton(out, to_out_norm_weight, to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul “outgoing” implementation (AlphaFold3)

This code follows the reference PyTorch implementation but
* uses a custom Triton kernel for the two LayerNorms (the expensive
  reductions over the channel dimension), and
* computes the core 4‑D reduction  `out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]`
  via a high‑throughput batched GEMM (torch.bmm).  The hidden dimension
  (default 128) is flattened into the batch dimension, so a single
  call to `torch.bmm` performs `B·hidden_dim` independent matrix
  multiplications of size `N×N`.  This avoids the O(N³) Python loops
  and lets the GPU handle the heavy work.

The overall order of operations matches the reference module:
1. LayerNorm over the input channel dimension.
2. Linear projections (`left_proj`, `right_proj`) without bias.
3. Optional pair‑wise mask.
4. Gating (sigmoid of three linear layers) applied element‑wise.
5. Batched GEMM to realize the Einstein summation.
6. LayerNorm over the hidden dimension, followed by an output gate.
7. Final linear projection back to the original channel dimension.

Both LayerNorms are implemented in Triton; the remaining tensor
operations are performed with standard PyTorch kernels (which are
already highly optimized on H100).  The function `custom_kernel`
conforms to the required signature and works for all test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm kernel (reduction over the channel axis)
# ----------------------------------------------------------------------
@triton.jit
def layer_norm_forward_kernel(
    X_ptr,                # input pointer
    Y_ptr,                # output pointer
    weight_ptr,           # LN weight (C,)
    bias_ptr,             # LN bias   (C,)
    B, N, C,              # batch, seq_len, channel
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    stride_w, stride_b,
    eps: tl.constexpr,    # epsilon
    BLOCK: tl.constexpr   # # of (b,i,j) elements processed per program
):
    pid = tl.program_id(0)
    total = B * N * N
    block_start = pid * BLOCK
    offs = block_start + tl.arange(0, BLOCK)                       # (BLOCK,)
    mask = offs < total                                            # (BLOCK,)

    # Decode (b,i,j) from the flattened index
    b = offs // (N * N)
    residual = offs % (N * N)
    i = residual // N
    j = residual % N

    # Channel axis
    c = tl.arange(0, C)

    # ------------------------------------------------------------------
    # Load the input tile X[b,i,j,:]  -> shape (BLOCK, C)
    # ------------------------------------------------------------------
    X_ptrs = X_ptr + \
        b[:, None] * stride_xb + i[:, None] * stride_xi + \
        j[:, None] * stride_xj + c[None, :] * stride_xc
    X = tl.load(X_ptrs, mask=mask[:, None], other=0.0)   # (BLOCK, C)

    # ------------------------------------------------------------------
    # Compute mean and variance over C
    # ------------------------------------------------------------------
    sum_X = tl.sum(X, axis=1)                # (BLOCK,)
    mean = sum_X * (1.0 / C)

    X_centered = X - mean[:, None]           # (BLOCK, C)
    var = tl.sum(X_centered * X_centered, axis=1) * (1.0 / C)

    # ------------------------------------------------------------------
    # Normalize and apply affine transformation
    # ------------------------------------------------------------------
    inv_std = tl.rsqrt(var + eps)
    Y_norm = X_centered * inv_std[:, None]   # (BLOCK, C)

    w = tl.load(weight_ptr + c * stride_w)   # (C,)
    b_ = tl.load(bias_ptr   + c * stride_b)   # (C,)

    Y = Y_norm * w[None, :] + b_[None, :]    # (BLOCK, C)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    Y_ptrs = Y_ptr + \
        b[:, None] * stride_yb + i[:, None] * stride_yi + \
        j[:, None] * stride_yj + c[None, :] * stride_yc
    tl.store(Y_ptrs, Y, mask=mask[:, None])

# ----------------------------------------------------------------------
# Helper: Triton based LayerNorm (float32)
# ----------------------------------------------------------------------
def triton_layer_norm(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block: int = 128) -> torch.Tensor:
    """
    LayerNorm over the last dimension using Triton.
    x, weight, bias must be on the same CUDA device.
    """
    assert x.is_contiguous()
    B, N, N2, C = x.shape
    assert N == N2, "expected square pairwise tensor"

    out = torch.empty_like(x)

    # strides are expressed in element counts (not bytes)
    sx_b, sx_i, sx_j, sx_c = x.stride()
    sy_b, sy_i, sy_j, sy_c = out.stride()
    sw = weight.stride(0)
    sb = bias.stride(0)

    grid = (triton.cdiv(B * N * N, block),)   # 1‑D grid
    layer_norm_forward_kernel[grid](
        x, out,
        weight, bias,
        B, N, C,
        sx_b, sx_i, sx_j, sx_c,
        sy_b, sy_i, sy_j, sy_c,
        sw, sb,
        eps=eps,
        BLOCK=block,
    )
    return out

# ----------------------------------------------------------------------
# Main custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] on the same CUDA device.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the input channel dimension (dim)
    # ------------------------------------------------------------------
    x_norm = triton_layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block=128,
    )  # (B,N,N,dim)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, weights["right_proj.weight"]) # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # Optional mask handling
    # ------------------------------------------------------------------
    # If `nomask` is True we treat the mask as all‑ones.
    if config.get("nomask", True):
        mask_factor = torch.ones_like(mask_tensor, dtype=left.dtype, device=left.device)
    else:
        mask_factor = mask_tensor.to(dtype=left.dtype, device=left.device)
    mask_factor = mask_factor.unsqueeze(-1)   # (B,N,N,1)
    left = left * mask_factor
    right = right * mask_factor

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core Einstein‑summation: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    # Implemented as a batched GEMM (torch.bmm) after reshaping.
    # ------------------------------------------------------------------
    # Convert to half‑precision for the heavy matrix multiply
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring the hidden dimension into the batch axis
    # left:  (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right needs its last two axes swapped to get (k, j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, N, N) with k as dim2

    B, H, N, _ = left_perm.shape
    # Flatten (B,H) into a single batch dimension for torch.bmm
    left_mat = left_perm.view(B * H, N, N)     # (B*H, N, N)   (i,k)
    right_mat = right_perm.view(B * H, N, N)   # (B*H, N, N)   (k,j)

    # Batched matrix multiplication → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)   # FP16 multiplication

    # Reshape back to (B, N, N, H) and cast to FP32 for the next layers
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # LayerNorm over the hidden dimension (hidden_dim)
    # ------------------------------------------------------------------
    out = triton_layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block=128,
    )

    # ------------------------------------------------------------------
    # Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
