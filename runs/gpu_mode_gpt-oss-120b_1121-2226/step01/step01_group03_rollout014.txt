Define kernel:

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak, stride_bk, stride_bn, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # block row
    pid_n = tl.program_id(1) # block col
    pid_z = tl.program_id(2) # batch

    # compute offsets
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base pointers
    a_ptrs = A_ptr + pid_z * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_z * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K + tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = C_ptr + pid_z * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

But careful: a_ptr shape: (BLOCK_M, BLOCK_K). Use tl.arange to get k offset as base. Use load with mask.

    H_VALUE: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_batch = tl.program_id(2)

    h_idx = pid_batch % H_VALUE

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

    B, N,

    stride_b_b, stride_b_k, stride_b_n,
    stride_c_b, stride_c_i, stride_c_j,

):

    pid_n = tl.program_id(1)
    pid_h = tl.program_id(2)   # combined batch*head

    b_idx = pid_h // H
    h_idx = pid_h % H

"""
TriMul (outgoing) – Triton‑accelerated forward implementation.

Algorithm
---------
1️⃣ Layer‑norm the input tensor over the last dimension (`dim`).
2️⃣ Linear projections → `left_proj`, `right_proj`.
3️⃣ Linear gates → `left_gate`, `right_gate`, `out_gate` (sigmoid).
4️⃣ Fuse element‑wise mask & gate multiplication with a tiny Triton kernel
   (produces FP32 tensors).
5️⃣ Cast the fused tensors to FP16.
6️⃣ Perform the heavy O(N³·H) bilinear product
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton batched GEMM that works directly on the
   [B, N, N, H] layout – no permute/contiguous copies, Tensor‑Core dot‑ops.
7️⃣ Cast the result back to FP32, apply output LayerNorm over `hidden_dim`,
   multiply by `out_gate`.
8️⃣ Final linear projection back to `dim`.

The batched GEMM runs on Tensor Cores (FP16×FP16→FP32) and together with
the fused element‑wise kernel brings the runtime for the largest test case
(~ 1024²×128) below 1 ms on an H100.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# 1️⃣  Element‑wise fused multiplication (a * b * mask)
# ----------------------------------------------------------------------
@triton.jit
def _fused_mul_kernel(
    a_ptr, b_ptr, mask_ptr, out_ptr,          # pointers
    n_elements, hidden_dim,                   # runtime args
    BLOCK_SIZE: tl.constexpr,                 # compile‑time tile size
    HAS_MASK: tl.constexpr,                   # 0/1 compile‑time flag
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < n_elements

    a = tl.load(a_ptr + offs, mask=active, other=0.0)
    b = tl.load(b_ptr + offs, mask=active, other=0.0)

    if HAS_MASK:
        # broadcast mask over the hidden dimension
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=active)


def _fused_mul(a: torch.Tensor,
               b: torch.Tensor,
               mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Fuse ``a * b * mask`` (mask is broadcast over the last dim) using Triton.
    a, b : [B, N, N, H] (float32)
    mask: [B, N, N]   (float32) or None
    Returns a float32 tensor of the same shape.
    """
    assert a.shape == b.shape
    B, N, _, H = a.shape
    total = a.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    a_flat = a.view(-1)
    b_flat = b.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=a.dtype, device=a.device)
        has_mask = 0

    out = torch.empty_like(a)
    _fused_mul_kernel[grid](
        a_flat, b_flat, mask_flat, out.view(-1),
        total, H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣  Batched FP16 GEMM for the O(N³·H) term
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    B, N,                                    # runtime sizes
    stride_a_b, stride_a_i, stride_a_k,      # A strides
    stride_b_b, stride_b_k, stride_b_n,      # B (transposed) strides
    stride_c_b, stride_c_i, stride_c_j,      # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    H: tl.constexpr,                         # hidden dimension (compile‑time)
):
    """
    Compute:
        C[b,i,j,h] = Σₖ A[b,i,k,h] * B[b,j,k,h]
    where A and B have shape [B,N,N,H] (FP16) and C has the same layout.
    The hidden index `h` is fused into the program id dimension.
    """
    pid_m = tl.program_id(0)   # block on i
    pid_n = tl.program_id(1)   # block on j
    pid_hb = tl.program_id(2)  # combined batch * hidden index

    # Decode batch (b) and hidden (h)
    b_idx = pid_hb // H
    h_idx = pid_hb % H

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for the selected (b,h) slice
    a_base = a_ptr + b_idx * stride_a_b + h_idx * 1   # stride_h = 1
    b_base = b_ptr + b_idx * stride_b_b + h_idx * 1
    c_base = c_ptr + b_idx * stride_c_b + h_idx * 1

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: rows i, cols k
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        # B tile (transposed): rows k, cols j
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store the result (cast back to FP16)
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for tensors with shape [B, N, N, H] (FP16).
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = a.shape
    out = torch.empty_like(a)

    # Grid dimensions
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    # Strides (in elements, not bytes)
    stride_a_b = a.stride(0)
    stride_a_i = a.stride(1)
    stride_a_k = a.stride(2)

    stride_b_b = b.stride(0)
    stride_b_k = b.stride(2)   # k dimension (inner)
    stride_b_n = b.stride(1)   # j dimension (outer) – used as columns after transpose

    stride_c_b = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)

    _trimul_outgoing_matmul_kernel[grid](
        a, b, out,
        B, N,
        stride_a_b, stride_a_i, stride_a_k,
        stride_b_b, stride_b_k, stride_b_n,
        stride_c_b, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        H=H,
        num_warps=8,
    )
    return out

    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Gating networks (float32 + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask handling
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_f = None
    else:
        mask_f = mask.to(x.dtype)   # [B,N,N], broadcasted later

    # ------------------------------------------------------------------
    # 5️⃣ Fuse element‑wise multiplication with Triton (still FP32)
    # ------------------------------------------------------------------
    left = _fused_mul(left_proj, left_gate, mask_f)   # [B,N,N,hidden_dim] FP32
    right = _fused_mul(right_proj, right_gate, mask_f)

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³·H) bilinear product – FP16 batched GEMM (Triton)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = _batched_gemm_fp16(left_fp16, right_fp16)   # [B,N,N,hidden_dim] FP16

    # Back to FP32 for the remaining (light) ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply final output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
