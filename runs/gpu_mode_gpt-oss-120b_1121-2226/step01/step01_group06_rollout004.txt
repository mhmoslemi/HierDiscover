"""
custom_kernel – Triton‑accelerated outgoing TriMul

Algorithm (forward only)
------------------------
1. Layer‑norm over the last dimension `dim` (float‑32).
2. Cast to FP16 and apply four linear projections (no bias):
      left, right, left_gate, right_gate, out_gate.
   The gates are passed through a sigmoid and multiplied in‑place.
   An optional mask (B,N,N) is broadcasted and applied before gating.
   All these operations run in FP16 to halve memory traffic.
3. Contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication of shape
   (B*H, N, N) @ (B*H, N, N) → (B*H, N, N) performed by a
   custom Triton kernel that uses Tensor‑cores (FP16×FP16 → FP32).
   The kernel tiles the multiplication with BLOCK_M/N/K = 64
   and launches 8 warps per block.
4. Reshape the result back to (B, N, N, H) and convert to FP32.
5. Layer‑norm over the hidden dimension `hidden_dim` (float‑32).
6. Multiply by `out_gate` (broadcasted, FP32).
7. Final linear projection back to `dim` (float‑32).

Only the N³ contraction is executed in Triton; everything else
leverages efficient PyTorch kernels in half‑precision where possible.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A: (batch, M, K)   FP16
#   B: (batch, K, N)   FP16
#   C: (batch, M, N)   FP16  (accumulated in FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,      # pointers
    M, N, K,                  # matrix sizes
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    #  block index decomposition: (batch, block_m, block_n)
    # ------------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # --------------------------------------------------------------
    #  tile offsets and masks
    # --------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  loop over K dimension in tiles of size BLOCK_K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch, offs_m, offs_k]   (M×K tile)
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # B[batch, offs_k, offs_n]   (K×N tile)
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product (uses Tensor‑cores when possible)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    #  write result back
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision matrix multiplication using the Triton kernel above.

    Parameters
    ----------
    a : torch.Tensor
        Shape (B, M, K), dtype=torch.float16, contiguous.
    b : torch.Tensor
        Shape (B, K, N), dtype=torch.float16, contiguous.

    Returns
    -------
    torch.Tensor
        Shape (B, M, N), dtype=torch.float16, contiguous.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Block sizes tuned for H100 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_m * num_n,)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps = 256 threads, optimal for 64×64 tiles
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor            shape [B, N, N, dim]
        - mask         : torch.Tensor or None    shape [B, N, N]
        - weights      : dict of torch.Tensor with keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config       : dict with entries "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # model weights (cast to the precision we will use)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]   # keep FP32 for final projection

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the input `dim` (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim]   FP16

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) in FP16
    # --------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]  FP16
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over the hidden dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1] FP16
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating tensors (sigmoid) – compute in FP16 (torch may upcast internally)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    #    Implemented as a batched GEMM A @ B where:
    #       A = left  (B, H, N, N)   -> (B*H, N, N)
    #       B = rightᵀ (B, H, N, N)   -> (B*H, N, N)
    # --------------------------------------------------------------
    # Permute to bring `hidden_dim` before the two sequence axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N] (i,k)
    right_perm_t = right.permute(0, 3, 2, 1).contiguous()      # [B, H, N, N] (k,j)

    B_, H_, N_, _ = left_perm.shape
    # reshape to (B*H, N, N)
    A = left_perm.view(B_ * H_, N_, N_)          # (batch, M=N, K=N)
    B_mat = right_perm_t.view(B_ * H_, N_, N_)   # (batch, K=N, N)

    # Batched FP16 matmul using Triton
    C = _batched_matmul_fp16(A, B_mat)          # (B*H, N, N)  FP16

    # Restore original 4‑D layout: (B, N, N, H)
    out_h = C.view(B_, H_, N_, N_).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Convert back to FP32 for the second LayerNorm
    # --------------------------------------------------------------
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Apply the output gate (broadcasted)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]   FP32

    return out
