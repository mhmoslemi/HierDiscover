def _fused_proj_gate_mask(left_proj, left_gate, right_proj, right_gate, mask=None):
    B,N,N,H = left_proj.shape
    total = left_proj.numel()
    BLOCK = 1024
    grid = ( (total + BLOCK - 1)//BLOCK, )
    left_out = torch.empty_like(left_proj, dtype=torch.float16) # allocate fp16
    right_out = torch.empty_like(right_proj, dtype=torch.float16)
    left_proj_flat = left_proj.view(-1)
    left_gate_flat = left_gate.view(-1)
    right_proj_flat = right_proj.view(-1)
    right_gate_flat = right_gate.view(-1)
    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=left_proj.dtype, device=left_proj.device)
        has_mask = 0
    _fused_proj_gate_mask_kernel[grid](
        left_proj_flat, left_gate_flat,
        right_proj_flat, right_gate_flat,
        mask_flat,
        left_out.view(-1), right_out.view(-1),
        total, H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return left_out, right_out

@triton.jit
def _proj_gate_mask_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    n_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    # load
    lproj = tl.load(left_proj_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rproj = tl.load(right_proj_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)

    # compute sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-lg))
    right_gate = 1.0 / (1.0 + tl.exp(-rg))

        right = rproj * right_gate * m

        right = rproj * right_gate

    right_h = right.to(tl.float16)

    tl.store(left_out_ptr + offs, left_h, mask=mask)
    tl.store(right_out_ptr + offs, right_h, mask=mask)

def custom_kernel(data):
    x, mask, weights, config = data
    batch_size, seq_len, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 1. input layernorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # combine linear weights
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0)  # shape (5*hidden_dim, dim)

    # flatten and linear
    x_flat = x_norm.view(-1, dim)  # (B*N*N, dim)
    combined_out = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*hidden_dim)
    combined_out = combined_out.view(batch_size, seq_len, seq_len, hidden_dim*5)

    # split
    left_proj, left_gate_logit, right_proj, right_gate_logit, out_gate_logit = torch.split(combined_out, hidden_dim, dim=3)
    # compute out_gate before
    out_gate = torch.sigmoid(out_gate_logit)

    # mask handling
    if config.get("nomask", False) or mask is None:
        mask_f = None
    else:
        mask_f = mask.to(x.dtype)  # float32

    # fused elementwise multiplication + sigmoid gating
    left_fp16, right_fp16 = _proj_gate_mask_fused(left_proj, left_gate_logit, right_proj, right_gate_logit, mask_f)

    # free unneeded tensors
    del left_proj, left_gate_logit, right_proj, right_gate_logit, combined_out, x_norm, x_flat

    # 6. batched GEMM
    B, N, _, H = left_fp16.shape
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # 7. output layernorm (hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 8. apply out_gate
    out = out * out_gate

    # 9. final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    Output tensors are FP16.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < n_elements

    # Load the four source tensors (FP32)
    lp = tl.load(left_proj_ptr + offs, mask=active)
    lg = tl.load(left_gate_ptr + offs, mask=active)
    rp = tl.load(right_proj_ptr + offs, mask=active)
    rg = tl.load(right_gate_ptr + offs, mask=active)

    # Sigmoid = 1 / (1 + exp(-x))  (computed in FP32)
    lgate = 1.0 / (1.0 + tl.exp(-lg))
    rgate = 1.0 / (1.0 + tl.exp(-rg))

    if HAS_MASK:
        # mask index = offset // hidden_dim  (mask is not tiled over H)
        midx = offs // hidden_dim
        m = tl.load(mask_ptr + midx, mask=active)   # FP32
        lval = lp * lgate * m
        rval = rp * rgate * m
    else:
        lval = lp * lgate
        rval = rp * rgate

    # Cast to FP16 before storing
    tl.store(left_out_ptr + offs, lval.to(tl.float16), mask=active)
    tl.store(right_out_ptr + offs, rval.to(tl.float16), mask=active)


def _proj_gate_mask_fused(left_proj, left_gate_logit,
                         right_proj, right_gate_logit,
                         mask):
    """
    Wrapper around the Triton kernel.
    All inputs are FP32 tensors of shape [B, N, N, H].
    `mask` is either None or a float32 tensor of shape [B, N, N].
    Returns (left_fp16, right_fp16) of the same spatial shape, dtype torch.float16.
    """
    B, N, _, H = left_proj.shape
    total = left_proj.numel()                    # B·N·N·H
    BLOCK = 8192                                 # tuned for large element‑wise kernels
    grid = ((total + BLOCK - 1) // BLOCK,)

    left_out  = torch.empty_like(left_proj, dtype=torch.float16)
    right_out = torch.empty_like(right_proj, dtype=torch.float16)

    # flatten tensors for the kernel
    lp_flat = left_proj.view(-1)
    lg_flat = left_gate_logit.view(-1)
    rp_flat = right_proj.view(-1)
    rg_flat = right_gate_logit.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=left_proj.dtype, device=left_proj.device)
        has_mask = 0

    _proj_gate_mask_kernel[grid](
        lp_flat, lg_flat,
        rp_flat, rg_flat,
        mask_flat,
        left_out.view(-1), right_out.view(-1),
        total, H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor   [B, N, N, dim]
        - mask_tensor  : torch.Tensor | None   [B, N, N] (float32 or bool)
        - weights_dict : dict of torch.Tensors with the following keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : dict containing at least
            "dim" (int), "hidden_dim" (int), optional "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    no_mask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Combined linear projection + gate logits (no bias)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape: (5*H, dim))
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )  # (5*H, dim)

    B, N, _, _ = x_norm.shape
    x_flat = x_norm.view(-1, dim)                     # (B*N*N, dim)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (B*N*N, 5*H)
    combined = combined.view(B, N, N, hidden_dim * 5)

    # split into the five tensors (all views, no copy)
    left_proj, left_gate_logit, right_proj, right_gate_logit, out_gate_logit = torch.split(
        combined, hidden_dim, dim=3
    )  # each [B, N, N, H]

    # out_gate (sigmoid) will be used after the output LayerNorm
    out_gate = torch.sigmoid(out_gate_logit)          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if no_mask or mask is None:
        mask_fp = None
    else:
        # ensure float32 mask for the kernel
        mask_fp = mask.to(dtype)

    # ------------------------------------------------------------------
    # 4. Fuse proj * gate * mask → FP16 left/right tensors (Triton)
    # ------------------------------------------------------------------
    left_fp16, right_fp16 = _proj_gate_mask_fused(
        left_proj, left_gate_logit, right_proj, right_gate_logit, mask_fp
    )  # both [B, N, N, H] (float16)

    # Release tensors that are no longer needed (helps with peak memory)
    del left_proj, left_gate_logit, right_proj, right_gate_logit, combined, x_norm, x_flat, combined_weight

    # ------------------------------------------------------------------
    # 5. Batched GEMM over the sequence dimension (FP16, Tensor Cores)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # left @ rightᵀ  →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H] and convert to FP32 for the next steps
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Apply the out‑gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
