"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (last dim = dim).
2️⃣  Linear projections to the hidden dimension:
        left  = x @ W_left   (no bias)
        right = x @ W_right
3️⃣  (Optional) mask the projected tensors.
4️⃣  Compute three element‑wise sigmoid gates from the same normalized input:
        left_gate  = sigmoid(x @ W_left_gate)
        right_gate = sigmoid(x @ W_right_gate)
        out_gate   = sigmoid(x @ W_out_gate)
5️⃣  Apply the left/right gates:   left  = left  * left_gate,
                                  right = right * right_gate
6️⃣  Heavy N³‑like operation:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   This is a batched matrix multiply L @ Rᵀ for every hidden channel.
   We reshape (B,N,N, H) → (B·H, N, N) and run a custom Triton kernel that
   computes C = A @ Bᵀ with fp16 inputs and fp32 accumulation.
7️⃣  Reshape the result back to (B,N,N,H) and apply a second LayerNorm
    over the hidden dimension.
8️⃣  Multiply by out_gate (element‑wise) and a final linear projection
    back to dim.

The Triton kernel is a classic tiled GEMM (BLOCK_M × BLOCK_N tiles,
BLOCK_K‑wise accumulation) with boundary masking.  It avoids an explicit
transpose of the right operand by swapping its stride arguments, thus
saving a memory‑copy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * Bᵀ (N×K) → C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,    # B strides: batch, N, K (NOTE: bn ↔ col‑stride, bk ↔ row‑stride for Bᵀ)
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (here: B·H)

    # -----------------------------------------------------------------
    # Offsets of the current tile (M‑ and N‑dimensions)
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # -----------------------------------------------------------------
    # Pointers to the start of the k‑tile (k = 0)
    # -----------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk   # row‑stride (original N)
        + offs_n[None, :] * stride_bn                 # col‑stride (original K)
    )

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K tiles
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # OOB masks
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16 → fp32 inside dot
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -----------------------------------------------------------------
    # Store the tile
    # -----------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(A: torch.Tensor,
                           B: torch.Tensor,
                           BLOCK_M: int = 128,
                           BLOCK_N: int = 128,
                           BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matmul C = A @ Bᵀ
    A : (batch, M, K) – fp16
    B : (batch, N, K) – fp16 (will be read as transposed)
    Returns C : (batch, M, N) – fp32
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides (in elements)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn_orig, stride_bk_orig = B.stride()
    # To treat B as transposed without materialising it:
    #   - column stride (bn) = original row‑stride (stride_bk_orig)
    #   - row stride    (bk) = original column‑stride (stride_bn_orig)
    stride_bn = stride_bk_orig
    stride_bk = stride_bn_orig

    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor   [B, N, N, dim]  (float32)
        - mask         : torch.Tensor   [B, N, N]       (bool or float)
        - weights      : dict of torch.Tensor – model parameters
        - config       : dict – must contain "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool; convert to same dtype as projection tensors
        m = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()   # [B,N,N,H]
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5️⃣  Apply left/right gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Core N³ operation – batched matmul using Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy kernel (Tensor‑Core friendly)
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape   # B, seq_len, seq_len, hidden_dim

    # reshape to (B·H, N, N) → each hidden channel becomes an independent batch
    # permute moves H next to B without copying (view is cheap)
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Choose tile sizes adapted to seq_len (avoid excessive over‑allocation)
    if N <= 64:
        BLOCK_M = BLOCK_N = 64
    elif N <= 256:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 128   # 128 works well up to 1024
    BLOCK_K = 64

    # Perform C = left_mat @ right_matᵀ  (output dtype = fp32)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )   # shape (B·H, N, N), dtype fp32

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣  Output LayerNorm, out_gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                 # element‑wise gating (broadcasted)
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    # out is already float32
    return out
