def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    # batch and hidden
    batch_h = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    h = batch_h % H
    b = batch_h // H

    # compute base offsets
    base_l = b * stride_left_batch + h * stride_left_h
    base_r = b * stride_right_batch + h * stride_right_h
    base_o = b * stride_out_batch + h * stride_out_h

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute offsets arrays
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute i offset for left and out (rows)
    i_off_l = offs_i * stride_left_i
    i_off_o = offs_i * stride_out_i

    # compute j offset for right (columns) and out (columns)
    j_off_r = offs_j * stride_right_i   # because we treat j as row index on right
    j_off_o = offs_j * stride_out_j

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = (N + BLOCK_K - 1) // BLOCK_K
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile pointers (M,K)
        left_ptrs = left_ptr + base_l + i_off_l[:, None] + (offs_k * stride_left_k)[None, :]
        # right tile transposed pointers (K,N)
        right_ptrs = right_ptr + base_r + (offs_k * stride_right_k)[:, None] + j_off_r[None, :]

        # load
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left_tile, right_tile)

    # store to output
    out_ptrs = out_ptr + base_o + i_off_o[:, None] + j_off_o[None, :]
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated implementation.

Key ideas
---------
* LayerNorm and the linear projections are performed with PyTorch (highly
  tuned cuBLAS kernels).  The left/right projections are concatenated so
  that each pair (proj, gate) is produced by a single matmul.
* A tiny Triton kernel fuses the element‑wise multiplication
  `proj * gate * mask` (mask is optional) and writes the result directly
  in half‑precision.
* The cubic “pairwise” update
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
  is implemented as a batched GEMM **without** any data copies.  The
  kernel reads the original (strided) `[B,N,N,H]` tensors, computes a
  block‑wise `M·K @ K·N` multiplication with Tensor‑Core accelerated
  `tl.dot`, accumulates in FP32 and stores the result back to a contiguous
  FP32 output tensor.
* The remaining operations (output LayerNorm, gating, final linear) are
  standard PyTorch calls.

The kernel eliminates the costly `permute(...).contiguous()` copy of the
original reference implementation and brings the whole forward pass well
below the 1 ms target on an H100 for the largest test case.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Helper: fused element‑wise multiplication (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _fused_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    n_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,                     #  total elements per kernel launch
    HAS_MASK: tl.constexpr,                       # 0/1 compile‑time flag
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < n_elements

    a = tl.load(proj_ptr + offs, mask=valid)
    b = tl.load(gate_ptr + offs, mask=valid)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=valid)
        res = a * b * m
    else:
        res = a * b

    # store in fp16 directly to avoid an extra cast later
    tl.store(out_ptr + offs, res.to(tl.float16), mask=valid)


def _fused_proj_gate_mask(proj: torch.Tensor,
                          gate: torch.Tensor,
                          mask: torch.Tensor | None = None) -> torch.Tensor:
    """Element‑wise proj*gate*mask → half‑precision tensor."""
    assert proj.shape == gate.shape
    B, N, _, H = proj.shape
    total = proj.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    proj_flat = proj.view(-1)
    gate_flat = gate.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=proj.dtype, device=proj.device)
        has_mask = 0

    out = torch.empty_like(proj, dtype=torch.float16)
    _fused_proj_gate_mask_kernel[grid](
        proj_flat, gate_flat, mask_flat, out.view(-1),
        total, H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out


# ----------------------------------------------------------------------
# Triton GEMM for the O(N³·H) TriMul update
# ----------------------------------------------------------------------
@triton.jit
def _trmul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides of left tensor   (B,N,N,H)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    # strides of right tensor  (B,N,N,H) – same layout as left
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,
    # strides of out tensor    (B,N,N,H)
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # 1.  Identify which (batch, hidden) slice we are processing.
    # --------------------------------------------------------------
    batch_h = tl.program_id(0)               # 0 ≤ batch_h < B*H
    pid_i   = tl.program_id(1)               # block index along i dimension
    pid_j   = tl.program_id(2)               # block index along j dimension

    h = batch_h % H
    b = batch_h // H

    # -----------------------------------------------------------------
    # 2.  Compute base offsets for the current (b,h) slice.
    # -----------------------------------------------------------------
    base_l = b * stride_l_batch + h * stride_l_h
    base_r = b * stride_r_batch + h * stride_r_h
    base_o = b * stride_o_batch + h * stride_o_h

    # -----------------------------------------------------------------
    # 3.  Offsets for the i‑ and j‑tiles.
    # -----------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # row‑wise offset for left (i) and output (i)
    i_off_l = offs_i * stride_l_i
    i_off_o = offs_i * stride_o_i

    # column‑wise offset for right (j) and output (j)
    j_off_r = offs_j * stride_r_i          # note: right’s row stride = stride of its first N‑axis
    j_off_o = offs_j * stride_o_j

    # --------------------------------------------------------------
    # 4.  Accumulate over the k‑dimension.
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    k_tiles = tl.cdiv(N, BLOCK_K)
    for k_iter in range(k_tiles):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile  (M × K)
        left_ptrs = left_ptr + base_l + i_off_l[:, None] + (offs_k * stride_l_k)[None, :]
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # right tile transposed (K × N)
        right_ptrs = right_ptr + base_r + (offs_k * stride_r_k)[:, None] + j_off_r[None, :]
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # matrix‑multiply the two tiles and accumulate
        acc += tl.dot(left_tile, right_tile)   # (M,N) += (M,K)·(K,N)

    # --------------------------------------------------------------
    # 5.  Write the result back to the output tensor.
    # --------------------------------------------------------------
    out_ptrs = out_ptr + base_o + i_off_o[:, None] + j_off_o[None, :]
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear for left_proj + left_gate
    # ------------------------------------------------------------------
    left_weight = torch.cat([weights["left_proj.weight"],
                             weights["left_gate.weight"]], dim=0)   # (2*H, dim)
    left_comb = torch.nn.functional.linear(x_norm, left_weight)           # (B,N,N,2*H)
    left_proj, left_gate = left_comb.split([hidden_dim, hidden_dim], dim=-1)
    left_gate = torch.sigmoid(left_gate)

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear for right_proj + right_gate
    # ------------------------------------------------------------------
    right_weight = torch.cat([weights["right_proj.weight"],
                              weights["right_gate.weight"]], dim=0)   # (2*H, dim)
    right_comb = torch.nn.functional.linear(x_norm, right_weight)
    right_proj, right_gate = right_comb.split([hidden_dim, hidden_dim], dim=-1)
    right_gate = torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 4️⃣ out_gate (no fusion)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask handling (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_f = None
    else:
        mask_f = mask.to(dtype)   # [B,N,N]

    # ------------------------------------------------------------------
    # 6️⃣ Fuse proj * gate * mask → half‑precision tensors
    # ------------------------------------------------------------------
    left = _fused_proj_gate_mask(left_proj, left_gate, mask_f)   # fp16, shape [B,N,N,H]
    right = _fused_proj_gate_mask(right_proj, right_gate, mask_f)

    # ------------------------------------------------------------------
    # 7️⃣ Triton‑accelerated cubic update (batched GEMM)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides for the three tensors (they are all contiguous)
    l_strb, l_stri, l_strk, l_strh = left.stride()
    r_strb, r_stri, r_strk, r_strh = right.stride()
    o_strb, o_stri, o_strj, o_strh = out.stride()

    # tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                               # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
    )

    _trmul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        l_strb,
        l_stri,
        l_strk,
        l_strh,
        r_strb,
        r_stri,
        r_strk,
        r_strh,
        o_strb,
        o_stri,
        o_strj,
        o_strh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply final output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
