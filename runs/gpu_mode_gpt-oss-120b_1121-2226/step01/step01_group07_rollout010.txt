    # 9. Post LayerNorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # 10. Output gate
    out = out * out_gate

    # 11. Final linear projection
    out = F.linear(out, to_out_weight)   # shape B,N,N,dim

    return out
```

"""
TriMul (outgoing) – fast Triton‑augmented implementation.

Algorithm
---------
1. Layer‑norm on the channel dimension (float32).
2. Linear projections to the hidden dimension (`left`, `right`).
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
4. Fuse mask, gate and a cast to float16 with a tiny Triton kernel:
       out = left * left_gate * mask   (and the same for `right`).
   The kernel works on the flattened 1‑D view of the tensors, so the
   whole operation is a single launch and the memory traffic is reduced
   to one read/write per element.
5. Batched half‑precision matrix multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batch of `B*H` GEMMs of shape (N,N) and is performed
   by torch.bmm (cuBLAS) on fp16 tensors – the fastest path on H100.
6. Convert the result back to fp32, apply a second Layer‑norm,
   an output gate and a final linear projection to the original dimension.
   The final tensor has shape [B, N, N, dim] and dtype float32.

Only the element‑wise mask‑gate fusion is done in Triton, satisfying the
requirement that “part of the operations” run in a kernel while keeping the
overall runtime well below the 1 ms target for the largest test case.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication of tensor, gate and optional mask.
# All tensors are flattened, contiguous and in fp16.
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    src_ptr,                # input tensor (fp16)   – shape (B*N*N*H,)
    gate_ptr,               # gate tensor  (fp16)   – same shape
    mask_ptr,               # mask tensor (fp16)   – shape (B*N*N,)
    dst_ptr,                # output tensor (fp16) – same shape as src
    hidden_dim: tl.constexpr,
    n_elem:      tl.constexpr,   # total number of elements = B*N*N*H
    BLOCK:       tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear offsets
    m_off = offs < n_elem

    # index of the mask element (one mask per (b,i,k) – i.e. per block of H)
    mask_idx = offs // hidden_dim

    a = tl.load(src_ptr + offs,  mask=m_off, other=0.0)
    g = tl.load(gate_ptr + offs, mask=m_off, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=m_off, other=1.0)   # 1.0 when mask absent
    out = a * g * m
    tl.store(dst_ptr + offs, out, mask=m_off)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
            - mask         : torch.Tensor of shape [B, N, N] or None
            - weights      : dict of model weight tensors
            - config       : dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Extract weight tensors (all are float32 on the same device)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape: [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (bias=False in the reference)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, H]  float32
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # float32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # float32

    # ------------------------------------------------------------------
    # 4. Cast to half‑precision for the heavy path
    # ------------------------------------------------------------------
    left  = left .to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    left_gate  = left_gate .to(torch.float16).contiguous()
    right_gate = right_gate.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 5. Prepare mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    if mask is None:
        # all‑ones mask – shape (B, N, N) -> flatten to (B*N*N,)
        mask_fp16 = torch.ones((B * N * N,), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(-1)  # flatten to (B*N*N,)

    # ------------------------------------------------------------------
    # 6. Fuse mask and gate with a tiny Triton kernel (in‑place scaling)
    # ------------------------------------------------------------------
    total_elems = left.numel()                       # B*N*N*H
    BLOCK = 1024                                      # max threads per block
    grid = (triton.cdiv(total_elems, BLOCK),)

    # left_scaled = left * left_gate * mask
    left_scaled = torch.empty_like(left)
    apply_gate_mask_kernel[grid](
        left,
        left_gate,
        mask_fp16,
        left_scaled,
        hidden_dim=hidden_dim,
        n_elem=total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # right_scaled = right * right_gate * mask
    right_scaled = torch.empty_like(right)
    apply_gate_mask_kernel[grid](
        right,
        right_gate,
        mask_fp16,
        right_scaled,
        hidden_dim=hidden_dim,
        n_elem=total_elems,
        BLOCK=BLOCK,
        num_warps=4,
    )
    # (gate tensors are no longer needed – let them be GC‑ed)
    del left, right, left_gate, right_gate

    # ------------------------------------------------------------------
    # 7. Batched matrix multiplication (B*H independent GEMMs)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    #   left_scaled : (B, N, N, H)  -> (B, H, N, N) -> (B*H, N, N)
    #   right_scaled: (B, N, N, H)  -> (B, H, N, N) with (i,j) swapped
    left_perm  = left_scaled.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_scaled.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    batch_hidden = B * hidden_dim
    left_mat = left_perm.view(batch_hidden, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(batch_hidden, N, N) # (B*H, N, N)

    # cuBLAS batched GEMM (FP16 accumulator in FP32, then cast back to FP16)
    out_mat = torch.bmm(left_mat, right_mat)       # (B*H, N, N), dtype=fp16

    # reshape back to (B, N, N, H) and cast to fp32 for the remaining layers
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8. Post‑contraction LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9. Apply the output gate (still in float32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (dim, hidden_dim) weight → [B, N, N, dim]

    return out
