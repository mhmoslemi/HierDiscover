"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Compute four linear projections (left, right, left_gate, right_gate, out_gate)
   in float32 and cast the results used later to float16.
3. Fuse masking and per‑position gating with a tiny Triton kernel
   (in‑place:  X ← X * G * M  where M is optional mask).
4. Perform the N³ contraction as a batched half‑precision matrix multiplication:
   for each hidden channel d the operation is
        out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
   which is equivalent to a batch of GEMMs (left @ rightᵀ).  The GEMM is executed
   by torch.bmm (Tensor‑Core powered cuBLAS) on tensors reshaped to
   (B·H, N, N).
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   output gate and project back to the original feature dimensionality.

Only the element‑wise masking + gating is done in Triton; the heavy O(N³)
contraction is delegated to the highly‑optimised cuBLAS GEMM which easily
meets the sub‑millisecond target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse (optional) mask with per‑position gate.
# X, G, M and Y are all 1‑D flattened views of tensors with shape
#   (B, N, N, H)  (X and G)  and   (B, N, N, 1) (M).
# The kernel computes    Y = X * G               (no mask)
#                or    Y = X * G * M           (mask present)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    X_ptr,                         # input tensor (fp16)
    G_ptr,                         # gate tensor  (fp16)
    M_ptr,                         # mask tensor  (fp16) – may be unused
    Y_ptr,                         # output tensor (fp16) – can be same as X_ptr
    H: tl.constexpr,               # hidden dimension (compile‑time)
    TOTAL_ELEMS: tl.int64,         # total number of elements in X
    BLOCK_SIZE: tl.constexpr,      # threads per program
    HAS_MASK: tl.constexpr,        # 1 if mask is present, else 0
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)   # [BLOCK_SIZE]
    mask = offs < TOTAL_ELEMS

    x = tl.load(X_ptr + offs, mask=mask, other=0.0)
    g = tl.load(G_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each mask element corresponds to H consecutive X/G elements
        m_off = offs // H
        m = tl.load(M_ptr + m_off, mask=mask, other=1.0)
        y = x * g * m
    else:
        y = x * g

    tl.store(Y_ptr + offs, y, mask=mask)


def _apply_gate_mask(x_fp16: torch.Tensor,
                     gate_fp16: torch.Tensor,
                     mask_fp16: torch.Tensor | None,
                     has_mask: bool) -> torch.Tensor:
    """
    In‑place element‑wise fusion of mask and gate using Triton.
    Returns the same tensor (x_fp16) after modification.
    """
    total = x_fp16.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    # dummy mask tensor required when HAS_MASK == 0 (kernel will ignore it)
    dummy_mask = torch.tensor([1.0],
                              dtype=torch.float16,
                              device=x_fp16.device)

    _apply_gate_mask_kernel[grid](
        x_fp16,
        gate_fp16,
        mask_fp16 if has_mask else dummy_mask,
        x_fp16,                     # store result back into x_fp16
        H=x_fp16.shape[-1],
        TOTAL_ELEMS=total,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=1 if has_mask else 0,
        num_warps=4,                # enough parallelism for the memory‑bound work
    )
    return x_fp16


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of torch.Tensor containing model parameters
            - config: dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Extract weights
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]          # (dim,)
    norm_bias            = weights["norm.bias"]            # (dim,)
    left_proj_weight     = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_weight    = weights["right_proj.weight"]    # (hidden_dim, dim)
    left_gate_weight     = weights["left_gate.weight"]     # (hidden_dim, dim)
    right_gate_weight    = weights["right_gate.weight"]    # (hidden_dim, dim)
    out_gate_weight      = weights["out_gate.weight"]      # (hidden_dim, dim)
    to_out_norm_weight   = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias     = weights["to_out_norm.bias"]     # (hidden_dim,)
    to_out_weight        = weights["to_out.weight"]        # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, D] (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (float32, bias=False)
    # ------------------------------------------------------------------
    left_raw  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right_raw = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gating values (float32, sigmoid)
    # ------------------------------------------------------------------
    left_gate_raw  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate_raw = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]
    out_gate       = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Prepare mask (optional)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).contiguous()  # [B,N,N,1]
        has_mask = True
    else:
        mask_fp16 = None
        has_mask = False

    # ------------------------------------------------------------------
    # 5. Cast to fp16 and fuse mask + per‑position gates via Triton
    # ------------------------------------------------------------------
    left_fp16 = left_raw.to(torch.float16).contiguous()
    right_fp16 = right_raw.to(torch.float16).contiguous()
    left_gate_fp16 = left_gate_raw.to(torch.float16).contiguous()
    right_gate_fp16 = right_gate_raw.to(torch.float16).contiguous()

    # In‑place fusion: X ← X * G * M
    left_fp16 = _apply_gate_mask(left_fp16, left_gate_fp16, mask_fp16, has_mask)
    right_fp16 = _apply_gate_mask(right_fp16, right_gate_fp16, mask_fp16, has_mask)

    # ------------------------------------------------------------------
    # 6. Batched matrix multiplication (the O(N³) contraction)
    #    Out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as left @ rightᵀ for each hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = left_fp16.shape  # B, seq_len, seq_len, H

    # (B, H, N, N) – left: rows = i, cols = k
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    # (B, H, N, N) – rightᵀ: rows = k, cols = j
    right_perm = right_fp16.permute(0, 3, 2, 1).contiguous()

    # Collapse batch and head dimensions → (B·H, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # cuBLAS batched GEMM (Tensor‑Core fp16 → fp32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)               # (B·H, N, N), fp16

    # Reshape back to [B, N, N, H] and convert to float32 for the remaining ops
    out = (
        out_mat.view(B, hidden_dim, N, N)
               .permute(0, 2, 3, 1)                         # (B, N, N, H)
               .contiguous()
               .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 7. Post‑contraction LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8. Apply output gate (still in float32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (dim, hidden_dim) → [B, N, N, D]

    return out
