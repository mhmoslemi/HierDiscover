    right_weights = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)
    right_comb = F.linear(x, right_weights)
    right_proj, right_gate = right_comb[..., :hidden_dim], right_comb[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate)

    # Apply mask if needed.
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # broadcast
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to FP16
    left = left.half()
    right = right.half()

    # Reshape for batched matmul
    B, N, _, H = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B*H, N, N)  # transpose

    # Output container
    out_mat = torch.empty_like(left_mat)

    # Compute strides
    # For left_mat:
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_ab = left_mat.stride(0)

    # Right_mat:
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_bb = right_mat.stride(0)

    # C strides
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # grid dims
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B*H)

    # Launch kernel
    tri_mul_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        M=N, N=N, K=N,
        stride_am=stride_am, stride_ak=stride_ak, stride_ab=stride_ab,
        stride_bk=stride_bk, stride_bn=stride_bn, stride_bb=stride_bb,
        stride_cm=stride_cm, stride_cn=stride_cn, stride_cb=stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )

    # Reshape and cast back
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous().float()
    # Output layernorm + gating + linear
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # shape (B,N,N,hidden_dim)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])
    return out

def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,

    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,

):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_n = offs_n < N
    mask_k = offs_k < K

    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)

"""
TriMul (outgoing) – fast forward implementation.

The kernel works on the heavy N³ reduction:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
where `left`/`right` have shape (B, N, N, H) (H = hidden_dim).
For each head `d` we need a matrix multiplication (N×N)·(N×N)ᵀ.
The implementation fuses the three linear projections (proj + gate) into
two matmuls, applies the optional mask, then launches a Triton kernel that
computes all `B·H` independent matmuls in FP16 with an FP32 accumulator.
The kernel uses a 64×64×64 tile‑size (tuned for H100 Tensor‑Cores) and
stores the result back as FP16 before the final LayerNorm and output
projection in FP32.

All heavy work (the N³ loop) is performed on‑device; the surrounding
operations stay in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A,B,C are FP16)
# A : (batch, M, K)   – fp16
# B : (batch, K, N)   – fp16 (already transposed w.r.t. original tensor)
# C : (batch, M, N)   – fp16
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_am, stride_ak, stride_ab,   # strides of A (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides of B (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # strides of C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch element `b`:
        C[b, :, :] = A[b, :, :] @ B[b, :, :].T
    using a block‑wise fp16 GEMM with an fp32 accumulator.
    """
    pid_m = tl.program_id(0)   # tile index over M
    pid_n = tl.program_id(1)   # tile index over N
    pid_b = tl.program_id(2)   # batch index (B * hidden_dim)

    # ------------------------------------------------------------------
    # Offsets for the tile we are responsible for
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # ------------------------------------------------------------------
    # Pointers to the start of the current batch slice
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # ------------------------------------------------------------------
    # Pointers to the current tiles of A and B
    # ------------------------------------------------------------------
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot: Σₖ a[i,k] * b[k,j]
        acc += tl.dot(a, b)

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result (FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Extract weights (all are FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]      # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2. Fuse projection + gate for LEFT and RIGHT branches
    #    (one matmul per branch, rather than two)
    # --------------------------------------------------------------
    # Concatenate proj and gate weights → (2*hidden_dim, dim)
    left_weights = torch.cat([left_proj_weight, left_gate_weight], dim=0)   # (2H, dim)
    right_weights = torch.cat([right_proj_weight, right_gate_weight], dim=0)  # (2H, dim)

    # Shared linear pass (FP32)
    left_concat = F.linear(x, left_weights)   # (B, N, N, 2H)
    right_concat = F.linear(x, right_weights) # (B, N, N, 2H)

    # Split back into projection & gate tensors
    left_proj, left_gate = left_concat.split(hidden_dim, dim=-1)
    right_proj, right_gate = right_concat.split(hidden_dim, dim=-1)

    # Apply sigmoid to the gate parts (still FP32)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # (B,N,N,H)

    # --------------------------------------------------------------
    # 3. Optional mask (binary 0/1 mask is broadcastable)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → broadcast over hidden dimension
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 4. Combine projection with its gate
    # --------------------------------------------------------------
    left = left_proj * left_gate          # (B,N,N,H)   FP32
    right = right_proj * right_gate       # (B,N,N,H)   FP32

    # --------------------------------------------------------------
    # 5. Cast to FP16 for the N³ reduction
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape   # (batch, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 5a. Reshape to (B*H, N, N) and produce the transposed version for the right
    # ------------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (batch, M, K)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)        # (batch, K, N)

    out_mat = torch.empty_like(left_mat)   # FP16 container for the result

    # ------------------------------------------------------------------
    # 5b. Compute strides needed by the Triton kernel
    # ------------------------------------------------------------------
    stride_am = left_mat.stride(1)   # stride over M dimension
    stride_ak = left_mat.stride(2)   # stride over K dimension
    stride_ab = left_mat.stride(0)   # batch stride

    stride_bk = right_mat.stride(1)  # stride over K dimension
    stride_bn = right_mat.stride(2)  # stride over N dimension
    stride_bb = right_mat.stride(0)  # batch stride

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    # ------------------------------------------------------------------
    # 5c. Launch the Triton batched GEMM
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64   # a relatively large K‑tile works well on H100

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles over rows (M)
        triton.cdiv(N, BLOCK_N),   # tiles over columns (N)
        B * H,                     # one batch slice per hidden head
    )

    tri_mul_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,                # 4‑warps per block is a sweet spot for fp16 on H100
    )

    # ------------------------------------------------------------------
    # 5d. Reshape back to the original 4‑D layout and cast to FP32
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm → gating → final projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # Apply the output gate (still in FP32)
    out = out * out_gate

    # Final linear back to `dim`
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
```

This implementation:

* Fuses the two projection + gate linear layers per side, cutting the number of
  matrix‑multiplies in half.
* Performs the expensive N³ reduction with a custom Triton kernel that uses
  64×64×64 tiles and an FP32 accumulator, fully exploiting H100 Tensor Cores.
* Keeps the rest of the pipeline in PyTorch (LayerNorm, sigmoid, final projection)
  for simplicity and numerical stability.
* Supports all configurations described in the prompt (different batch sizes,
  sequence lengths, hidden dimensions, optional masks, and both normal & Cauchy
  input distributions).
