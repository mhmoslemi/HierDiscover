"""
TriMul (outgoing) – fast half‑precision implementation.

Algorithm
---------
1.  Layer‑norm the 4‑D input over the last channel dimension (half‑precision).
2.  Six linear projections (no bias) are performed in FP16:
        left_proj , right_proj , left_gate , right_gate , out_gate .
   The three gate tensors are passed through a sigmoid.
3.  If a mask is supplied we fuse the three element‑wise operations
        left = left * left_gate * mask
        right = right * right_gate * mask
   with a tiny Triton kernel that works on the flattened tensors.
   When no mask is present we just multiply by the gates in PyTorch.
4.  Core Bilinear update:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   This is a batched matrix‑multiply per hidden dimension.
   We reshape ``left``/``right`` to (B·H, N, N) and call ``torch.bmm``
   (cuBLAS Tensor‑Core GEMM, FP16).
5.  Second LayerNorm over the hidden dimension (half), multiply by the
    ``out_gate`` (half), and a final linear projection back to ``dim``.
6.  Cast the result to FP32 before returning.

All heavy arithmetic stays in FP16 to reduce memory traffic; only the
output is cast back to FP32.  The Triton kernel handles the mask+gate
fusion, which otherwise would require several separate memory passes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                # fp16 *
    right_ptr,               # fp16 *
    left_gate_ptr,           # fp16 *
    right_gate_ptr,          # fp16 *
    mask_ptr,                # fp16 *
    hidden_dim,              # int (constexpr)
    size,                    # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise fusion:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    The mask has shape (B*N*N) and is broadcast over ``hidden_dim``.
    """
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < size

    # Load tensors
    left      = tl.load(left_ptr      + offs, mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=valid, other=0.0)

    # Broadcasted mask index = element // hidden_dim
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)

    # Fuse
    left  = left  * left_gate  * m
    right = right * right_gate * m

    # Store results back in‑place
    tl.store(left_ptr  + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, dim] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of torch.Tensors (see keys in code)
            - config       : dict with at least 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension (half‑precision)
    # --------------------------------------------------------------
    x = input_tensor.half()
    norm_w = weights["norm.weight"].half()
    norm_b = weights["norm.bias"].half()
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=1e-5)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (FP16, no bias)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left  = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 3️⃣ Gate tensors + sigmoid (still FP16)
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 4️⃣ Fuse mask & gates (Triton) – skip if mask is None / all‑ones
    # --------------------------------------------------------------
    if mask is not None:
        # mask → FP16, flatten (B*N*N)
        mask_fp16 = mask.to(torch.float16).contiguous()
        mask_flat = mask_fp16.view(-1)                         # size = B*N*N

        # flatten the tensors to 1‑D for the kernel
        left_flat       = left.view(-1)
        right_flat      = right.view(-1)
        left_gate_flat  = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        total_elems = left_flat.numel()
        BLOCK = 1024                     # reasonable block size for pointwise work
        grid = (triton.cdiv(total_elems, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_ptr=left_flat,
            right_ptr=right_flat,
            left_gate_ptr=left_gate_flat,
            right_gate_ptr=right_gate_flat,
            mask_ptr=mask_flat,
            hidden_dim=hidden_dim,
            size=total_elems,
            BLOCK_SIZE=BLOCK,
        )
        # ``left`` and ``right`` now already contain mask * gate
    else:
        # no mask → just multiply by the gate tensors
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core TriMul – batched matrix multiplication (FP16)
    # --------------------------------------------------------------
    # Reshape to (B*H, N, N) – contiguous layout is required for cuBLAS
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # A @ Bᵀ  (Tensor‑Core GEMM, FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, hidden_dim] (still FP16)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w,
                       bias=to_out_norm_b, eps=1e-5)

    out = out * out_gate               # gate post‑norm (both FP16)

    to_out_w = weights["to_out.weight"].half()   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # final projection, FP16

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the model
    # --------------------------------------------------------------
    return out.to(torch.float32)
