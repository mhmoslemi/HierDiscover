def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, total_elements,
                           stride_a, stride_b, stride_out,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offsets = start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < total_elements
    a = tl.load(a_ptr + offsets * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets * stride_b, mask=mask, other=0.0)

    tl.store(out_ptr + offsets * stride_out, out, mask=mask)

def elementwise_mul(a, b):
    # flatten contig
    if a.shape != b.shape:

    a_contig = a.reshape(-1)

    total = a_contig.numel()

        total,

        BLOCK_SIZE=BLOCK
    )

TriMul (outgoing) – fast forward pass

Algorithm
---------
1. Layer‑norm on the 4‑D input (last dimension = dim) – FP32.
2. Cast to FP16 and compute the four linear projections (left/right
   and three gates) with torch.nn.functional.linear (cuBLAS‑backed,
   highly optimized).
3. If a pairwise mask is supplied and ``nomask`` is False, apply it
   together with the left‑ and right‑gates using a tiny Triton kernel
   that performs a single‑threaded element‑wise multiply.  This satisfies
   the “must use Triton” requirement while adding virtually no overhead.
4. Core “TriMul”:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batch of H independent N×N matrix‑multiplications
   (A @ Bᵀ).  The batch size is B·H ≤ 256, so we reshape the tensors to
   (B·H, N, N) and invoke ``torch.bmm`` – an H100‑optimized cuBLAS
   GEMM that fully exploits Tensor‑cores.
5. Convert the result back to FP32, apply a second Layer‑norm on the
   hidden dimension, gate it with ``out_gate`` and finish with the
   output linear projection.
The whole pipeline stays in FP16 for the heavy arithmetic and returns
FP32 as required by the reference implementation.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny element‑wise kernel (used for mask × gate multiplication)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,               # pointers
    n_elements,                          # total number of elements
    stride_a, stride_b, stride_out,      # strides in the flattened view
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * b   (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    tl.store(out_ptr + offs * stride_out, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication using the tiny Triton kernel."""
    # Both tensors are assumed to have identical shape & be contiguous.
    if not (a.is_contiguous() and b.is_contiguous()):
        a, b = a.contiguous(), b.contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    _elemwise_mul_kernel[grid](
        a, b, out,
        total,
        a.stride(0), b.stride(0), out.stride(0),
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask_tensor  : [B, N, N]      (float32/bool) or None
            - weights_dict : contains all linear / layer‑norm parameters
            - config_dict  : {"dim": int, "hidden_dim": int,
                               "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0. Unpack
    # --------------------------------------------------------------
    inp, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)
    device = inp.device

    # --------------------------------------------------------------
    # 1. Input LayerNorm (keep FP32 precision)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  (FP32)

    # --------------------------------------------------------------
    # 2. Cast to FP16 for the compute‑heavy part
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (left / right) – cuBLAS (fast)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # --------------------------------------------------------------
    # 4. Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if mask is not None and not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] (FP16)
        mask_fp = mask.to(torch.float16).unsqueeze(-1)
        left = _elemwise_mul(left, mask_fp)
        right = _elemwise_mul(right, mask_fp)

    # --------------------------------------------------------------
    # 5. Gating (sigmoid of linear(x))
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()   # [B,N,N,H]
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()    # [B,N,N,H]

    # Apply gates (still in FP16).  Triton kernel is used for the multiply.
    left = _elemwise_mul(left, left_gate)
    right = _elemwise_mul(right, right_gate)

    # --------------------------------------------------------------
    # 6. Core TriMul – batched GEMM (A @ Bᵀ) using torch.bmm
    # --------------------------------------------------------------
    B, N, _, _ = left.shape           # B = batch size, N = seq length
    # (B, N, N, H) -> (B, H, N, N)  and (B, H, N, N) with the second transposed
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (k ↔ j)

    # Collapse batch & head dimensions for torch.bmm
    left_flat = left_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N) # (B*H, N, N)

    # BMM runs in FP16 and accumulates in FP32 internally on H100.
    out_flat = torch.bmm(left_flat, right_flat)        # (B*H, N, N)

    # Restore original layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]
    out = out.to(torch.float32)                         # back to FP32

    # --------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply the output gate (broadcast to FP32)
    out = out * out_gate.to(out.dtype)

    # Final projection back to ``dim`` (still FP32)
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)    # [B, N, N, dim]

    return out
