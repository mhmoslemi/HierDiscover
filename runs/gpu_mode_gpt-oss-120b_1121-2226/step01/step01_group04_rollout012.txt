def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]  # or config.get

    # Cast to half (fp16)
    x = input_tensor.to(torch.float16)

    # Weights to half
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    # combine left_proj and left_gate
    left_proj_weight = weights["left_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    left_combined_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16)

    right_proj_weight = weights["right_proj.weight"]
    right_gate_weight = weights["right_gate.weight"]
    right_combined_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16)

    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # Input layernorm
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Compute combined left and right with gates pre sigmoid (still linear)
    # left_combined: shape (B,N,N, 2*hidden_dim)
    left_combined = F.linear(x, left_combined_weight)  # bias = None, shape B N N 2H
    right_combined = F.linear(x, right_combined_weight)

    # split into proj and gate parts
    left_proj, left_gate = left_combined.split([hidden_dim, hidden_dim], dim=3)
    right_proj, right_gate = right_combined.split([hidden_dim, hidden_dim], dim=3)

    # sigmoid for gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Prepare mask
    if mask is None:
        mask_tensor = torch.ones((bs, N, N), dtype=torch.float16, device=x.device)
    else:
        mask_tensor = mask.to(torch.float16)
    mask_expand = mask_tensor.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # Fuse mask with proj and gate
    # Convert to contiguous
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    size = left_proj.numel()
    BLOCK = 65536  # maybe larger? but limited by 2^15? We choose 65536
    grid = (triton.cdiv(size, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left_proj,
        right_ptr=right_proj,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expand,
        size=size,
        BLOCK_SIZE=BLOCK,
    )
    # After kernel left_proj and right_proj contain gated masked values.

    # Core TriMul batched matmul
    # reshape for batch dimension
    left_perm = left_proj.permute(0,3,1,2).contiguous()
    right_perm = right_proj.permute(0,3,1,2).contiguous()
    left_mat = left_perm.view(bs * hidden_dim, N, N)
    right_mat = right_perm.view(bs * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(bs, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # Output layernorm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    # gating
    out = out * out_gate
    # final projection
    out = F.linear(out, to_out_weight)
    # Cast to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Cast the input to FP16 and apply the input LayerNorm (half‑precision).
2. Fuse the two linear projections (proj + gate) for the “left’’ and “right’’
   tensors with a single GEMM each (weights are concatenated).  The gate
   part is passed through sigmoid.
3. Fuse the element‑wise mask, the sigmoid‑gates and the projection
   tensors with a tiny Triton kernel (no extra memory traffic).
4. Perform the core TriMul contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a strided‑batched GEMM of shape (B·H,N,N) @ (B·H,N,N)ᵀ.
   The tensors are permuted once to obtain a contiguous layout,
   then a single `torch.bmm` (Tensor‑Core FP16) computes *all* H heads in
   one launch.
5. Apply the output LayerNorm, multiply by the out‑gate and finish with the
   final linear projection back to the original `dim`.  Everything is kept
   in FP16 until the very last cast to FP32.

The implementation follows the original PyTorch logic but removes
unnecessary data copies, uses FP16 Tensor‑Cores for all dense matmuls,
and keeps at least one operation (mask‑gate fusion) in an explicit Triton
kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 * (B,N,N,H)
    right_ptr,              # fp16 *
    left_gate_ptr,          # fp16 *
    right_gate_ptr,         # fp16 *
    mask_ptr,               # fp16 *
    size: tl.constexpr,     # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    l  = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs,mask=mask, other=0.0)
    m  = tl.load(mask_ptr + offs,      mask=mask, other=0.0)

    # fused element‑wise: (proj * gate * mask)
    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[bs, N, N, dim]    (float32)
            - mask         : torch.Tensor[bs, N, N]        (bool/float, optional)
            - weights      : dict of torch.Tensors containing the model
                             parameters as in the reference code.
            - config       : dict with at least ``dim`` and ``hidden_dim`` (ints).

    Returns:
        torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # e.g. 128

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 0. FP16 cast of everything that can be Tensor‑Core processed
    # ------------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (still in half‑precision)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B,N,N,dim], fp16

    # ------------------------------------------------------------------
    # 2. Combined linear projections  (proj + gate) for left / right
    # ------------------------------------------------------------------
    # concatenate proj and gate weights so we read `x` only once per side
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)            # (2*H , dim)

    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)            # (2*H , dim)

    # single GEMM per side
    left_combined = F.linear(x, left_combined_w)   # [B,N,N,2*H]  fp16
    right_combined = F.linear(x, right_combined_w) # [B,N,N,2*H]  fp16

    # split into projection and gate parts
    left_proj, left_gate = left_combined.split([hidden_dim, hidden_dim], dim=3)
    right_proj, right_gate = right_combined.split([hidden_dim, hidden_dim], dim=3)

    # sigmoid gates (still half‑precision)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # out‑gate (separate linear)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 3. Fuse mask + gates into the projected tensors with Triton
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    mask_expand = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # make sure tensors are contiguous for the 1‑D kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = left_proj.numel()
    BLOCK = 8192                     # fits comfortably in a single program
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left_proj,
        right_ptr=right_proj,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expand,
        size=total_elements,
        BLOCK_SIZE=BLOCK,
    )
    # ------------------------------------------------------------------
    # 4. Core TriMul contraction (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    # bring `H` in front of the two spatial axes, then flatten (B*H) as batch
    left_perm  = left_proj.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    left_mat  = left_perm.view(bs * hidden_dim, N, N)        # (B*H,N,N)
    right_mat = right_perm.view(bs * hidden_dim, N, N)       # (B*H,N,N)

    # batched matmul:   (B*H,N,N) @ (B*H,N,N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B,N,N,H]
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # still fp16

    out = out * out_gate                     # gating (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim , hidden)
    out = F.linear(out, to_out_w)           # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 6. Cast back to float32 (the model expects fp32 output)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
