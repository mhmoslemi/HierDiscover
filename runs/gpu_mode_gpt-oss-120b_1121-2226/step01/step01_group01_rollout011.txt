"""
TriMul (outgoing) – high‑performance implementation

Algorithm
---------
1.  Layer‑Norm over the feature dimension (FP32 for stability).
2.  Two linear projections (to hidden_dim) → fp16.
3.  Optional binary mask (broadcasted over the hidden dimension).
4.  Three gating networks (linear + sigmoid) → fp16.
5.  Fuse “left * left_gate” and “right * right_gate” with a tiny
    Triton kernel (this satisfies the “use Triton” requirement).
6.  Batched matrix multiplication over the sequence dimension:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Performed with `torch.bmm`, which calls cuBLAS and exploits
    Tensor‑Core FP16 acceleration.
7.  Layer‑Norm on the hidden dimension, gating with `out_gate`,
    and a final linear projection back to `dim` (FP32).

All heavy compute (the O(N³) reduction) runs in FP16 on Tensor‑Cores,
while the surrounding normalisation/gating steps stay in FP32 for
numerical safety. The implementation supports the “nomask” flag and
arbitrary batch sizes (B≤2)."""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny Triton kernel: element‑wise fusion of left/right with their gates
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_kernel(
    left_ptr,            # fp16 *
    right_ptr,           # fp16 *
    left_gate_ptr,       # fp16 *
    right_gate_ptr,      # fp16 *
    total_elems: tl.int64,
    BLOCK_SIZE: tl.constexpr,
):
    """left  = left  * left_gate
       right = right * right_gate
       Operates on the flattened tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load (masked loads return 0.0 for out‑of‑bounds elements)
    l  = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    # Fuse the two multiplications
    l = l * lg
    r = r * rg

    # Store back
    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True ⇒ no mask is applied

    # --------------------------------------------------------------
    # Extract weight tensors (they are stored as float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projection weights (cast once to fp16)
    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    # Gate weights (cast to fp16)
    left_gate_weight  = weights["left_gate.weight"].to(torch.float16)   # (hidden, dim)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(torch.float16)    # (hidden, dim)

    # Output‑side LayerNorm + Linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # (dim, hidden)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)                     # [B, N, N, dim]  (FP32)

    # --------------------------------------------------------------
    # 2. Linear projections to hidden_dim  (FP16)
    # --------------------------------------------------------------
    # `F.linear` works with FP16 when both input and weight are FP16.
    x_fp16 = x.to(torch.float16)

    left  = F.linear(x_fp16, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x_fp16, right_proj_weight)  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] -> fp16
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (linear + sigmoid) – keep FP16 for speed
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, left_gate_weight))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_weight))  # [B, N, N, hidden]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight.to(torch.float32)))  # keep FP32 for the final gate

    # --------------------------------------------------------------
    # 5. Fuse element‑wise gate multiplication with a Triton kernel
    # --------------------------------------------------------------
    # Ensure contiguity (required for the 1‑D flattened kernel)
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = left.numel()                       # B * N * N * hidden
    BLOCK_SIZE = 1024                                    # 1‑K elements per program
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    apply_gate_kernel[grid](
        left, right,
        left_gate, right_gate,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,                 # small kernel, 4 warps are enough
    )

    # --------------------------------------------------------------
    # 6. Batched matrix multiplication (fp16) using cuBLAS
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # N = sequence length
    # reshape to (B*hidden, N, N) where the K‑dimension is the last axis of left/right
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm invokes cuBLAS’s strided‑batched GEMM (FP16 → TensorCores)
    out_mat = torch.bmm(left_mat, right_mat)            # [B*hidden, N, N] (fp16)

    # Reshape back to the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden] (fp16)

    # --------------------------------------------------------------
    # 7. Output LayerNorm (FP32), gate, and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)                         # FP32 inside LayerNorm

    # Apply the output gate (broadcasted over dim)
    out = out * out_gate                                   # FP32 * FP32 → FP32

    # Final linear projection back to the original feature dimension
    out = F.linear(out, to_out_weight)                     # [B, N, N, dim] (FP32)

    return out
