```python
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_k, stride_c_d,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    # Determine batch and hidden index
    batch = pid_hb // hidden_dim
    h = pid_hb % hidden_dim

    # Compute base pointer for this (batch, hidden) slice
    a_base = a_ptr + batch * stride_a_batch + h * stride_a_d
    b_base = b_ptr + batch * stride_b_batch + h * stride_b_d
    c_base = c_ptr + batch * stride_c_batch + h * stride_c_d

    # Compute tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < M
    mask_j = offs_j < N
    mask_k = offs_k < K

    # Pointers for A (left)
    a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
    # Pointers for B (right) - we need shape (K,N)
    b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        acc += tl.dot(a, b)  # a: (M,K), b: (K,N)

        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # Write result
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_k
    tl.store(c_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass

The heavy part of the TriMul layer is the 3‑D reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every batch b and hidden channel d.  This is a batched matrix
multiplication A @ Bᵀ where A and B are (N,N) matrices for each (b,d).
The kernel below works on the original layout
[batch, N, N, hidden] (i.e. no transposes or copies) and fuses the
K‑loop with Tensor‑Core `tl.dot`.  All arithmetic is performed in FP16
with FP32 accumulators for numerical safety.  The surrounding
layer‑norms, linear projections and gates are done in PyTorch (FP32) –
only the N³ reduction is offloaded to Triton.

The implementation follows these steps:

1. Layer‑norm over the input feature dimension.
2. Linear projections (`left_proj`, `right_proj`) and gating projections.
3. Optional mask multiplication.
4. Cast the gated tensors to FP16.
5. Triton kernel computes the batched mat‑mul for every (batch, hidden)
   pair, writing the result in FP16.
6. Cast back to FP32, apply the output‑norm, output‑gate and final
   linear projection.

The kernel is deliberately written with explicit strides so that it can
read the tensors directly from their native [B,N,N,H] layout, avoiding
the costly `permute`/`contiguous` copies that dominate the naïve
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                # base pointers (torch tensors)
    M, N, K,                            # matrix sizes (all equal to seq_len)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,   # strides for left
    stride_b_batch, stride_b_i, stride_b_k, stride_b_d,   # strides for right
    stride_c_batch, stride_c_i, stride_c_k, stride_c_d,   # strides for out
    H,                                    # hidden dimension (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched (B*H) matrix multiplication A @ Bᵀ.
    A = left[b,i,k,h]   (M×K)
    B = right[b,j,k,h]  (N×K)   → we load it as (K×N) to match `tl.dot`.
    C = out[b,i,j,h]    (M×N)
    """
    pid_i = tl.program_id(0)   # tile over M (i dimension)
    pid_j = tl.program_id(1)   # tile over N (j dimension)
    pid_hb = tl.program_id(2)  # flattened (batch, hidden)

    # ------------------------------------------------------------------
    # Decompose flattened index into batch and hidden‑channel indices
    # ------------------------------------------------------------------
    batch_idx = pid_hb // H
    hidden_idx = pid_hb % H

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_d
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_d
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_d

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < M
    mask_j = offs_j < N
    mask_k = offs_k < K

    # ------------------------------------------------------------------
    # Pointers for the current tile
    #   left:  (i, k)   → stride_i for rows, stride_k for cols
    #   right: (k, j)   → stride_k for rows, stride_i for cols
    # ------------------------------------------------------------------
    a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
    b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i

    # Accumulator in FP32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # (BM, BK)  fp16
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # (BK, BN)  fp16
        acc += tl.dot(a, b)            # (BM, BN) ← Σ_k a[i,k]·b[k,j]

        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # ------------------------------------------------------------------
    # Store result (cast back to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_k
    tl.store(
        c_ptrs,
        c,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Load model weights (all FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"] # (hidden_dim, dim)

    left_gate_weight = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"] # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]      # (hidden_dim,)
    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # (B, N, N, dim), FP32

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim), FP32
    right = F.linear(x, right_proj_weight) # (B, N, N, hidden_dim), FP32

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid) – also FP32
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # (B,N,N,hidden_dim)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 6. Apply gates and cast to FP16 (the heavy reduction works in FP16)
    # ------------------------------------------------------------------
    left = (left * left_gate).contiguous().to(torch.float16)
    right = (right * right_gate).contiguous().to(torch.float16)

    B, N, _, _ = left.shape  # N == seq_len

    # ------------------------------------------------------------------
    # 7. Allocate output tensor (FP16) – will hold the result of the
    #    3‑D reduction before the output‑norm.
    # ------------------------------------------------------------------
    out_fp16 = torch.empty_like(left)  # (B,N,N,hidden_dim), FP16

    # ------------------------------------------------------------------
    # 8. Gather strides needed by the Triton kernel
    # ------------------------------------------------------------------
    # left / right / out have identical layout
    stride_b = left.stride(0)   # batch stride
    stride_i = left.stride(1)   # stride for the first N dimension (i / j)
    stride_k = left.stride(2)   # stride for the second N dimension (reduction)
    stride_d = left.stride(3)   # stride for hidden dimension (==1)

    # ------------------------------------------------------------------
    # 9. Launch the Triton kernel
    # ------------------------------------------------------------------
    # Tile sizes – tuned for FP16 Tensor‑Core usage on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles over i
        triton.cdiv(N, BLOCK_N),   # tiles over j
        B * hidden_dim,            # one program per (batch, hidden)
    )

    tri_mul_outgoing_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out_fp16,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_b,
        stride_a_i=stride_i,
        stride_a_k=stride_k,
        stride_a_d=stride_d,
        stride_b_batch=stride_b,
        stride_b_i=stride_i,
        stride_b_k=stride_k,
        stride_b_d=stride_d,
        stride_c_batch=stride_b,
        stride_c_i=stride_i,
        stride_c_k=stride_k,
        stride_c_d=stride_d,
        H=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps give high occupancy on H100
    )

    # ------------------------------------------------------------------
    # 10. Convert back to FP32 for the remaining operations
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)  # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 11. Output LayerNorm, output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                       # gating (still FP32)
    out = F.linear(out, to_out_weight)         # final projection → (B,N,N,dim)

    return out
