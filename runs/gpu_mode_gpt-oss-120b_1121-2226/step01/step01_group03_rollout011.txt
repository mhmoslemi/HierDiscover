def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps=1e-5
    # 1) layer norm
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )
    # 2) linear proj
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # 3) gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # 4) fused multiplication, mask, and pack to half
    if config.get("nomask", False) or mask is None:
        mask_f = None
    else:
        mask_f = mask.to(dtype)  # shape [B,N,N] float32

    left_packed = _fused_pack(left_proj, left_gate, mask_f)  # shape [B*H,N,N] half
    right_packed = _fused_pack(right_proj, right_gate, mask_f)

    # 5) batched GEMM
    # convert to half already done in _fused_pack
    # left_packed shape (B*H,N,N) half
    # right_packed shape (B*H,N,N) half
    # Compute out_mat = left @ right.T
    out_mat = torch.bmm(left_packed, right_packed.transpose(1,2))
    # 6) reshape back to [B,N,N,H]
    B = x.shape[0]; N = x.shape[1]; H = hidden_dim
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # 7) output layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    )
    # 8) multiply by out_gate
    out = out * out_gate
    # 9) final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

def _fused_pack(a: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor | None = None) -> torch.Tensor:
    # a, gate: shape [B,N,N,H] float32 contig
    # mask: shape [B,N,N] float32 or None
    B,N,_,H = a.shape
    total = B * N * N * H
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    a_flat = a.view(-1)
    g_flat = gate.view(-1)
    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=a.dtype, device=a.device)
        has_mask = 0
    # Output shape [B*H,N,N] half
    out = torch.empty((B * H, N, N), dtype=torch.float16, device=a.device)
    out_flat = out.view(-1)
    _fused_pack_kernel[grid](
        a_flat, g_flat, mask_flat, out_flat,
        B, N, H, total,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out

"""
TriMul (outgoing) – fast forward implementation using Triton.

Algorithm
---------
1️⃣ Layer‑norm the input over the last dimension (dim).
2️⃣ Two linear projections → left_proj, right_proj   (float32).
3️⃣ Three linear gates → left_gate, right_gate, out_gate (sigmoid, float32).
4️⃣ Fuse element‑wise multiplication (proj * gate * optional mask) **and**
   pack the result directly into the layout required by the batched GEMM:
   from [B, N, N, H] (float32) to [B*H, N, N] (float16).  This is done by a
   custom Triton kernel that loads the data in the original (coalesced) order
   and writes the packed tensor in the target (batched‑matmul) order.
5️⃣ Perform the heavy O(N³·H) work with a cuBLAS batched GEMM in FP16:
   out = (left ⊗ rightᵀ)  where left,right ∈ [B*H, N, N].
6️⃣ Reshape the result back to [B, N, N, H] and convert to float32.
7️⃣ Layer‑norm over the hidden dimension, multiply by out_gate.
8️⃣ Final linear projection back to `dim`.

The Triton kernel in step 4 ensures we truly use Triton (as required) and
avoids an extra copy+type‑cast+permute.  All other ops stay in PyTorch
so we keep the implementation simple yet very fast on an H100.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused multiplication + packing to [B*H, N, N] FP16
# ----------------------------------------------------------------------
@triton.jit
def _fused_pack_kernel(
    a_ptr,               # float32 left/right projection   (B,N,N,H) flattened
    gate_ptr,            # float32 corresponding gate      (B,N,N,H) flattened
    mask_ptr,            # float32 mask (optional)        (B,N,N)   flattened
    out_ptr,             # float16 output (B*H,N,N)       flattened
    B: tl.constexpr,     # batch size
    N: tl.constexpr,     # sequence length
    H: tl.constexpr,     # hidden dimension
    total_elems: tl.constexpr,  # B*N*N*H
    BLOCK_SIZE: tl.constexpr,   # compile‑time tile size
    HAS_MASK: tl.constexpr,      # 0/1 compile‑time flag
):
    """ 
    For every element idx in the original tensor (layout B,N,N,H):
        out[ (b*H+h), i, j ] = a[idx] * gate[idx] * mask[b,i,j]    (if mask present)
    The kernel loads a,gate (and mask) coalesced and writes the packed
    FP16 tensor required for the batched GEMM.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)   # linear idx in the original tensor
    mask = offs < total_elems

    # ------------------------------------------------------------------
    # Load the two FP32 tensors (they are stored in the original layout)
    # ------------------------------------------------------------------
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # mask index = idx // H  -> shape (B,N,N)
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        val = a * g * m
    else:
        val = a * g

    # ------------------------------------------------------------------
    # Cast result to FP16
    # ------------------------------------------------------------------
    val_f16 = val.to(tl.float16)

    # ------------------------------------------------------------------
    # Decode linear idx -> (b,i,j,h) to compute output index
    # ------------------------------------------------------------------
    h = offs % H
    tmp = offs // H                # = b*N*N + i*N + j
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # Output layout is (B*H, N, N) row‑major:
    # out_idx = ((b*H + h) * N + i) * N + j
    out_idx = ((b * H + h) * N + i) * N + j
    tl.store(out_ptr + out_idx, val_f16, mask=mask)


def _fused_pack(a: torch.Tensor,
                gate: torch.Tensor,
                mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Fuse element‑wise multiplication (proj * gate * mask) and pack the
    result from shape [B, N, N, H] (float32) → [B*H, N, N] (float16).

    Args:
        a   – projection tensor, shape [B,N,N,H], dtype float32.
        gate– gate tensor, same shape as `a`, dtype float32.
        mask– optional mask, shape [B,N,N], dtype float32 (broadcast over H).

    Returns:
        Tensor of shape [B*H, N, N] with dtype torch.float16.
    """
    B, N, _, H = a.shape
    total = B * N * N * H
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    # Ensure contiguous 1‑D views
    a_flat = a.contiguous().view(-1)
    g_flat = gate.contiguous().view(-1)

    if mask is not None:
        mask_flat = mask.contiguous().view(-1)
        has_mask = 1
    else:
        # Dummy tensor – never accessed because HAS_MASK == 0
        mask_flat = torch.empty(0, dtype=a.dtype, device=a.device)
        has_mask = 0

    out = torch.empty((B * H, N, N), dtype=torch.float16, device=a.device)
    out_flat = out.view(-1)

    _fused_pack_kernel[grid](
        a_flat,
        g_flat,
        mask_flat,
        out_flat,
        B,
        N,
        H,
        total,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor with shape [bs, seq_len, seq_len, dim] (float32).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Gating networks (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4️⃣ Optional mask handling
    # --------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_fp = None
    else:
        mask_fp = mask.to(dtype)   # ensure same dtype as other tensors (float32)

    # --------------------------------------------------------------
    # 5️⃣ Fuse element‑wise multiplication + pack to FP16 layout
    # --------------------------------------------------------------
    left_fp16 = _fused_pack(left_proj, left_gate, mask_fp)   # (B*H, N, N) , FP16
    right_fp16 = _fused_pack(right_proj, right_gate, mask_fp)

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM over the sequence dimension (FP16 → Tensor Cores)
    # --------------------------------------------------------------
    # left_fp16/right_fp16 shape: [B*H, N, N]
    out_mat = torch.bmm(left_fp16, right_fp16.transpose(1, 2))

    # --------------------------------------------------------------
    # 7️⃣ Restore original layout [B, N, N, H] and convert to FP32
    # --------------------------------------------------------------
    B = x.shape[0]
    N = x.shape[1]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 9️⃣ Apply the final output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
