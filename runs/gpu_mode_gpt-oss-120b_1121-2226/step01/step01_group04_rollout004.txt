"""
TriMul (outgoing) forward – high‑performance implementation.

Algorithm
---------
1. Layer‑norm the input on the channel dimension (float32).
2. Cast to fp16 and fuse the left/right linear projections with their
   corresponding sigmoid‑gates by concatenating the weight matrices
   (2 × hidden_dim rows). This halves the number of GEMMs.
3. Fuse the element‑wise mask, projection‑gate multiplications with a tiny
   Triton kernel (in‑place on the fp16 tensors).
4. Core triangular update:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   Implemented as a batched GEMM `torch.bmm` on fp16 tensors:
        (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
5. Keep the result in fp16, apply a second LayerNorm (fp16 weights),
   multiply by the output‑gate (fp16) and project back to `dim` with an
   fp16 linear layer.
6. Cast the final result to float32.

All heavy linear work is done in half‑precision using Tensor‑Core
accelerated GEMMs; only a single Triton kernel is required (mask + gate
fusion). The implementation works for any batch size ≤ 2, sequence
length up to 1024 and both masked / unmasked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and sigmoid‑gates into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 tensor, flattened
    right_ptr,              # fp16 tensor, flattened
    left_gate_ptr,          # fp16 tensor, flattened
    right_gate_ptr,         # fp16 tensor, flattened
    mask_ptr,               # fp16 tensor, flattened (broadcasted on hidden dim)
    size: tl.int32,         # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load (out‑of‑bounds loads return 0.0)
    left = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Fuse the three element‑wise operations
    left = left * left_gate * m
    right = right * right_gate * m

    # Store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor[bs, N, N, dim] (float32)
              - mask: torch.Tensor[bs, N, N] (bool/float) or None
              - weights: dict of torch.Tensors (see keys below)
              - config: dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # e.g. 128

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for all subsequent heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fused linear projections (proj + gate) for left & right sides
    # ------------------------------------------------------------------
    #   left_proj + left_gate   → (2*H, dim)
    #   right_proj + right_gate → (2*H, dim)
    left_concat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*H, dim)

    right_concat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*H, dim)

    # flatten the first three dimensions for a single GEMM
    flat_x = x_fp16.view(-1, dim)                       # (B*N*N, dim)

    # one GEMM per side
    left_concat = F.linear(flat_x, left_concat_weight)   # (B*N*N, 2*H)
    right_concat = F.linear(flat_x, right_concat_weight) # (B*N*N, 2*H)

    # reshape back to 4‑D
    left_concat = left_concat.view(bs, N, N, 2 * hidden_dim)
    right_concat = right_concat.view(bs, N, N, 2 * hidden_dim)

    # split proj / gate
    left = left_concat[..., :hidden_dim]
    left_gate_pre = left_concat[..., hidden_dim:]

    right = right_concat[..., :hidden_dim]
    right_gate_pre = right_concat[..., hidden_dim:]

    # sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # ------------------------------------------------------------------
    # 4. Output gate (separate linear)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))
    out_gate = out_gate.view(bs, N, N, hidden_dim)   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5. Fuse mask, projection‑gate multiplication with Triton
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=x.device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # broadcast mask over hidden dimension (no extra memory copy if possible)
    mask_expand = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # all tensors are now contiguous fp16
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elements = left.numel()
    BLOCK = 8192                     # a good compromise for H100
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expand,
        size=total_elements,
        BLOCK_SIZE=BLOCK,
    )
    # after the kernel left/right already contain the masked & gated values

    # ------------------------------------------------------------------
    # 6. Core TriMul – batched matrix multiply (fp16)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H] (still fp16)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7. Second LayerNorm + out‑gate + final linear (all fp16)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # fp16

    out = out * out_gate                     # element‑wise gating (fp16)

    to_out_weight = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)       # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 8. Return float32 tensor as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
