"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The module follows the reference PyTorch code (AlphaFold‑3) but
optimises the dominant O(N³·hidden_dim) reduction:

1. **LayerNorm + fused Linear** – the two linear projections
   (proj & gate) are stacked into a single matmul (`2·hidden_dim × dim`).

2. **Optional mask** – applied after the projection (as in the reference).

3. **Gating** – sigmoid is computed in FP32, the gated values are kept
   in FP16 for the heavy reduction.

4. **Batched matrix multiplication** – we compute
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   as a batched GEMM (`A @ Bᵀ`) with a custom Triton kernel.
   The kernel works on FP16 data and accumulates in FP16 (Tensor‑Core
   friendly).  Tile sizes are chosen adaptively w.r.t. the sequence
   length.

5. **Output LayerNorm, output gate and final projection** – performed
   in FP32 to retain numerical stability.

Only the O(N³) reduction is carried out in Triton; all other
operations use standard PyTorch kernels.  This satisfies the
requirement of “using a Triton kernel for at least part of the
operations” while keeping the overall forward pass fast on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched fp16 GEMM (A @ Bᵀ) – one independent matmul per
# (batch * hidden_dim) slice.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_fp16_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all equal to seq_len)
    total_batch,                       # B * hidden_dim
    stride_am, stride_ak, stride_ab,   # A strides: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B strides: (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # C strides: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b, :, :] = A[b, :, :] @ B[b, :, :].T   for every batch index b.
    The kernel works on fp16 tensors and uses fp16 accumulation (Tensor‑Core
    friendly).  Edge tiles are masked.
    """
    pid_m = tl.program_id(0)   # tile index over M dimension
    pid_n = tl.program_id(1)   # tile index over N dimension
    pid_b = tl.program_id(2)   # flat batch index (B * hidden_dim)

    # --------------------------------------------------------------
    # Offsets for the current output tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # --------------------------------------------------------------
    # Guard against the case where the total number of batches is not a
    # multiple of the launch grid (the last block may be partially empty).
    # --------------------------------------------------------------
    valid_batch = pid_b < total_batch
    if not valid_batch:
        return

    # --------------------------------------------------------------
    # Pointers to the first element of the current batch slice
    # --------------------------------------------------------------
    a_batch_ptr = a_ptr + pid_b * stride_ab
    b_batch_ptr = b_ptr + pid_b * stride_bb
    c_batch_ptr = c_ptr + pid_b * stride_cb

    # --------------------------------------------------------------
    # Load A and B tiles and accumulate the product over K
    # --------------------------------------------------------------
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)   # fp16 accumulator

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                # fp16 load
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                # fp16 load
        acc += tl.dot(a, b)                     # Tensor‑Core dot
        a_ptrs += BLOCK_K * stride_ak           # advance K‑tile
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store the result (masked store for edge tiles)
    # --------------------------------------------------------------
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → no mask is applied

    # --------------------------------------------------------------
    # Extract weight tensors (all are float32 in the checkpoint)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]   # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]    # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]   # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]     # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]      # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2. Fuse (proj + gate) for left and right streams
    # --------------------------------------------------------------
    # Stack projection and gate weights → shape (2*hidden_dim, dim)
    left_combined_w = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16)
    right_combined_w = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16)

    # Linear with fused weights (still bias‑free)
    left_combined = F.linear(x, left_combined_w)   # (B,N,N,2*hidden_dim)
    right_combined = F.linear(x, right_combined_w) # (B,N,N,2*hidden_dim)

    # Split back into projection & gate
    left_proj_raw = left_combined[..., :hidden_dim]          # (B,N,N,hidden_dim)
    left_gate_raw = left_combined[..., hidden_dim:]          # (B,N,N,hidden_dim)

    right_proj_raw = right_combined[..., :hidden_dim]
    right_gate_raw = right_combined[..., hidden_dim:]

    # --------------------------------------------------------------
    # 3. Optional mask (broadcastable to [..., hidden_dim])
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left_proj_raw = left_proj_raw * mask
        right_proj_raw = right_proj_raw * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – keep gating in FP32 for stability
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)    # (B,N,N,hidden_dim), FP32
    right_gate = torch.sigmoid(right_gate_raw)

    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # (B,N,N,hidden_dim), FP32

    # --------------------------------------------------------------
    # 5. Apply gating
    # --------------------------------------------------------------
    left = left_proj_raw * left_gate   # (B,N,N,hidden_dim)
    right = right_proj_raw * right_gate

    # --------------------------------------------------------------
    # 6. Cast to fp16 for the heavy O(N³) reduction
    # --------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape   # batch, seq_len

    # ------------------------------------------------------------------
    # 6a. Prepare tensors for the batched GEMM
    #     left : (B, hidden_dim, N, N) → (B*hidden_dim, N, N)   (A)
    #     right: (B, hidden_dim, N, N) → (B*hidden_dim, N, N)   (Bᵀ)
    # ------------------------------------------------------------------
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output container (fp16)
    out_mat = torch.empty_like(left_mat)

    # ------------------------------------------------------------------
    # 6b. Compute strides required by Triton
    # ------------------------------------------------------------------
    stride_am = left_mat.stride(1)      # stride of M dimension (N)
    stride_ak = left_mat.stride(2)      # stride of K dimension (1)
    stride_ab = left_mat.stride(0)      # batch stride (N*N)

    stride_bk = right_mat.stride(1)     # K stride (N)
    stride_bn = right_mat.stride(2)     # N stride (1)
    stride_bb = right_mat.stride(0)     # batch stride (N*N)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    # ------------------------------------------------------------------
    # 6c. Choose tile sizes based on sequence length (empirical)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8

    # Grid dimensions (M‑tiles, N‑tiles, batch)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )

    # ------------------------------------------------------------------
    # 6d. Launch the Triton kernel
    # ------------------------------------------------------------------
    batched_matmul_fp16_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        total_batch=B * hidden_dim,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,               # double‑buffering for better latency hiding
    )

    # ------------------------------------------------------------------
    # 6e. Reshape back to (B, N, N, hidden_dim) and convert to FP32
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 7. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                     # element‑wise output gate (FP32)
    out = F.linear(out, to_out_weight)       # (B, N, N, dim), bias=False

    return out
```
