"""
TriMul (outgoing) – fast Triton + PyTorch implementation.

* A small Triton kernel fuses the optional mask with the per‑position
  left/right gates (element‑wise multiplication).  This kernel works on
  flattened contiguous fp16 tensors and thus costs virtually no time.
* The heavy N³ contraction is performed as a batched half‑precision
  matrix multiplication using torch.bmm, which on an H100 is far faster
  than a custom Triton GEMM for the sizes used in AlphaFold3.
* All remaining linear projections, LayerNorms and the final linear
  projection stay in PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(
    left_ptr,               # fp16*  (B*N*N*H)
    right_ptr,              # fp16*  (B*N*N*H)
    mask_ptr,               # fp16*  (B*N*N)  – broadcasted over H
    left_gate_ptr,          # fp16*  (B*N*N*H)
    right_gate_ptr,         # fp16*  (B*N*N*H)
    B, N, H,                # runtime integers
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place: left = left * mask * left_gate,
                right = right * mask * right_gate.
       The mask is broadcasted across the hidden dimension H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    active = offs < total

    # Index in the flattened mask (shape B*N*N)
    mask_idx = offs // H

    # Load values (masked loads return 0/1 for out‑of‑bounds entries)
    l_val = tl.load(left_ptr + offs, mask=active, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=active, other=0.0)
    lg_val = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    rg_val = tl.load(right_gate_ptr + offs, mask=active, other=0.0)
    m_val = tl.load(mask_ptr + mask_idx, mask=active, other=1.0)

    # Fuse mask and gates
    tl.store(left_ptr + offs,  l_val * m_val * lg_val, mask=active)
    tl.store(right_ptr + offs, r_val * m_val * rg_val, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            input_tensor – torch.Tensor of shape [B, N, N, D] (float32)
            mask         – torch.Tensor of shape [B, N, N] or None
            weights      – dict of model weight tensors
            config       – dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]            # (dim,)
    norm_bias = weights["norm.bias"]                # (dim,)
    left_proj_weight = weights["left_proj.weight"]  # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]       # (hidden_dim,)
    to_out_weight = weights["to_out.weight"]             # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over channel dim D)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_raw = F.linear(x, left_proj_weight)    # [B, N, N, H]  float32
    right_raw = F.linear(x, right_proj_weight)  # [B, N, N, H]  float32

    # ------------------------------------------------------------------
    # 3. Gating values
    # ------------------------------------------------------------------
    left_gate_raw = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate_raw = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Apply mask (if any) and per‑position gates – fused in Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left_raw.shape
    if mask is not None:
        # Cast everything to fp16 and make sure data are contiguous
        left = left_raw.to(torch.float16).contiguous()
        right = right_raw.to(torch.float16).contiguous()
        left_gate = left_gate_raw.to(torch.float16).contiguous()
        right_gate = right_gate_raw.to(torch.float16).contiguous()
        mask_fp16 = mask.to(torch.float16).reshape(-1).contiguous()  # (B*N*N,)

        total_elems = left.numel()
        BLOCK_SIZE = 32768  # 2^15, a good trade‑off for large tensors
        grid = ((total_elems + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        mask_gate_kernel[grid](
            left,
            right,
            mask_fp16,
            left_gate,
            right_gate,
            B, N, hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        gated_left = left
        gated_right = right
    else:
        # No mask – only gating (still in fp16)
        gated_left = left_raw.to(torch.float16) * left_gate_raw.to(torch.float16)
        gated_right = right_raw.to(torch.float16) * right_gate_raw.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. N³ contraction as batched GEMM (left @ rightᵀ)
    #    Shapes after permute: (B, H, N, N)
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm = gated_left.permute(0, 3, 1, 2).contiguous()
    # Swap the last two axes so that the K dimension aligns for matmul
    right_perm = gated_right.permute(0, 3, 2, 1).contiguous()

    # Flatten batch and hidden dimensions: (B*H, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Batched half‑precision matmul (uses cuBLAS Tensor‑Cores on H100)
    out_mat = torch.bmm(left_mat, right_mat)                     # (B*H, N, N)

    # Reshape back to [B, N, N, H] and convert to fp32 for the remaining ops
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)      # (B, N, N, H)
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 6. Post‑contraction LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (dim, hidden_dim) weight → [B, N, N, D]

    return out
