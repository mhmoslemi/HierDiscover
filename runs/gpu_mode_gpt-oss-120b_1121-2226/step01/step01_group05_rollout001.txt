"""
TriMul “outgoing” implementation (AlphaFold‑3 style) with a fused
batched GEMM kernel written in Triton.

Algorithm
---------
1. **Layer‑norm** the 4‑D input on its last dimension (dim).
2. **Linear projections** → `left` and `right` of shape [B,N,N,H].
3. **Mask & gates** (optional mask, sigmoid gates) are applied element‑wise.
4. **Cast to FP16** – the heavy cubic term is then a batched
   matrix multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly a GEMM `left @ rightᵀ` for every hidden channel *h*.
   The tensors are reshaped to [B·H, N, N] and processed by a
   tiled Triton kernel (`tri_mul_outgoing_kernel`).  The kernel
   uses Tensor‑cores (fp16 → fp32 accumulation) with block sizes
   64×64×64 and 4 warps per block.
5. **Second Layer‑norm**, output gate and final linear projection
   back to `dim` (all in float32).

Only the cubic matrix multiplication is performed in Triton; the
remaining ops use PyTorch which is already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ   (A,B are fp16, C is fp32)
#   A : [BH, M, K]   (M = N, K = N)
#   B : [BH, N, K]   (stored as [BH, K, N] inside the kernel)
#   C : [BH, M, N]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM over the hidden dimension.
    Each program instance computes a BLOCK_M × BLOCK_N tile of C for a
    particular (batch·hidden) index `pid_h`.
    """
    pid_h = tl.program_id(axis=2)      # corresponds to batch * hidden
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- load A (M × K) -------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)   # fp16 → fp32 for the dot

        # ----- load Bᵀ (K × N) -------------------------------------------------
        # B is stored as [BH, N, K]; we interpret it as its transpose.
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)   # fp16 → fp32

        # ----- inner product (Tensor‑core hot path) ---------------------------
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write back C
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (may be None)
        - weights_dict : mapping from weight name → torch.Tensor
        - config_dict  : contains ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` flag.
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, H]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate   # still float32
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the cubic GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6) Rearrange to [B, H, N, N] and flatten batch*hidden → [BH, N, N]
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)   # [BH, N, N] (fp16)
    right_flat = right.view(BH, N, N) # [BH, N, N] (fp16)

    # ------------------------------------------------------------------
    # 7) Allocate output buffer (FP32)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 8) Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per tile
    BLOCK_N = 64   # cols per tile
    BLOCK_K = 64   # reduction tile size (must be multiple of 16 for Tensor‑cores)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one program per (batch, hidden) pair
    )

    # Triton launch – 4 warps (covers 64×64 tile) and 3 stages for prefetch
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N)           # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 10) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # float32

    # ------------------------------------------------------------------
    # 11) Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
