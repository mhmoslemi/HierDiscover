@triton.jit
def mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    size,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # get weight tensors and cast to fp16
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Combine linear weights
    left_proj_weight = weights["left_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_proj_weight = weights["right_proj.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # Cast all to float16 (GPU)
    device = input_tensor.device
    # combine weights
    combined_weight = torch.cat([
        left_proj_weight,
        left_gate_weight,
        right_proj_weight,
        right_gate_weight,
        out_gate_weight], dim=0).to(dtype=torch.float16, device=device)

    # LayerNorm (float32)
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # cast to fp16 for linear
    x_norm_fp16 = x_norm.to(dtype=torch.float16)

    # Combined linear output
    combined_out = F.linear(x_norm_fp16, combined_weight)   # shape [B,N,N,5*hidden_dim]
    # split
    splits = torch.split(combined_out, hidden_dim, dim=-1)
    left = splits[0]                     # [B,N,N,H]
    left_gate = splits[1]                # [B,N,N,H]
    right = splits[2]
    right_gate = splits[3]
    out_gate = splits[4]

    # sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # mask handling
    if mask is None:
        mask_fp16 = torch.ones((input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype=torch.float16, device=device)
    mask_expand = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # flatten pointers for Triton kernel
    size = left.numel()
    BLOCK = 16384  # maybe larger for better occupancy
    grid = (triton.cdiv(size, BLOCK),)

    mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expand,
        size=size,
        BLOCK_SIZE=BLOCK,
    )

    # Core TriMul
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B_H = left_perm.shape[0] * left_perm.shape[1]  # B*H
    left_mat = left_perm.view(-1, N, N)  # (B*H, N, N)
    right_mat = right_perm.view(-1, N, N)  # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # Output LayerNorm + gating + final linear
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, to_out_weight)

    return out
```

"""
Highly‑optimized outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim` (FP32).
2. Cast to FP16 and fuse **all** linear projections (left, left_gate,
   right, right_gate, out_gate) into a single GEMM by concatenating the
   weight matrices.  This reduces the number of cuBLAS launches from five
   to one.
3. Apply sigmoid to the three gate tensors.
4. Fuse element‑wise mask, left‑gate and right‑gate multiplications with
   a custom Triton kernel (one launch, two stores).
5. Core TriMul:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed as a batched half‑precision matrix multiplication
   (`torch.bmm`) after reshaping to (B·H, N, N).
6. Convert the result back to FP32, apply a second LayerNorm over the
   hidden dimension, multiply by `out_gate`, and finish with a linear
   projection back to `dim`.
All heavy arithmetic is done in FP16 (tensor‑core friendly); the final
output is FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and the two gate multiplications.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                # fp16 *, [B,N,N,H]
    right_ptr,               # fp16 *
    left_gate_ptr,           # fp16 *
    right_gate_ptr,          # fp16 *
    mask_ptr,                # fp16 *
    size,                    # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,  # compile‑time constant
):
    """
    left  = left  * left_gate  * mask
    right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim] (float32)
            - mask: torch.Tensor[B, N, N] (bool/float) or None
            - weights: dict of torch.Tensors (see keys below)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # usually 128

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim] FP32

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections into ONE GEMM (FP16)
    # ------------------------------------------------------------------
    # collect weight tensors
    left_proj_w   = weights["left_proj.weight"]   # (H, dim)
    left_gate_w   = weights["left_gate.weight"]   # (H, dim)
    right_proj_w  = weights["right_proj.weight"]  # (H, dim)
    right_gate_w  = weights["right_gate.weight"]  # (H, dim)
    out_gate_w    = weights["out_gate.weight"]    # (H, dim)

    # concatenate along output‑feature axis: shape (5*H, dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(dtype=torch.float16, device=device)

    # cast normalized input to half‑precision for the GEMM
    x_fp16 = x.to(torch.float16)

    # single linear layer → (B,N,N,5*H)
    combined_out = F.linear(x_fp16, combined_weight)  # bias is None

    # split into the five tensors (views, no copy)
    split_tensors = torch.split(combined_out, hidden_dim, dim=-1)
    left        = split_tensors[0]  # [B,N,N,H]
    left_gate   = split_tensors[1]
    right       = split_tensors[2]
    right_gate  = split_tensors[3]
    out_gate    = split_tensors[4]

    # ------------------------------------------------------------------
    # 3. Activate gates (sigmoid) – keep in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate multiplications with Triton
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype=torch.float16, device=device)
    # broadcast mask to hidden dimension
    mask_expand = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # ensure all tensors are contiguous for the kernel
    left       = left.contiguous()
    right      = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_expand = mask_expand.contiguous()

    total_elements = left.numel()
    BLOCK = 1024  # 1K elements per program (tunable)
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expand,
        size=total_elements,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Core TriMul – batched matrix multiplication in FP16
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N)^T  →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H] and cast to FP32 for the rest
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B,N,N,H] FP32

    out = out * out_gate.to(out.dtype)   # gating (out_gate in FP16 → cast)

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)        # final projection, FP32

    return out
