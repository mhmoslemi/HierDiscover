"""
TriMul (outgoing) – Triton‑accelerated forward pass

This implementation follows the reference AlphaFold‑3 module
but replaces the O(H) independent N×N‑GEMMs (where H = hidden_dim)
by a single batched GEMM kernel written in Triton.
The kernel uses larger tiles (128×128×64) that map well to the
Tensor‑Core hardware of an H100, reducing the number of program
launches and memory traffic.  All heavy arithmetic (projections,
gating, the TriMul reduction) is performed in fp16 with fp32
accumulation, and the final result is returned in fp32.

The rest of the pipeline (LayerNorms, linear projections, sigmoid
gates) stays in PyTorch for simplicity; a Triton kernel is still
present, satisfying the “must‑use‑Triton” requirement.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where each batch element
# corresponds to one hidden channel of the TriMul operation.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                     # fp16 pointers
    M, N, K,                     # matrix dimensions (all = seq_len)
    stride_am, stride_ak,        # A row‑stride, col‑stride
    stride_bk, stride_bn,        # B row‑stride (k), col‑stride (n)
    stride_cm, stride_cn,        # C row‑stride, col‑stride
    stride_ab, stride_bb, stride_cb,  # batch strides (size = M*K, K*N, M*N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    """
    Compute C[b] = A[b] @ B[b]   (A: M×K, B: K×N, C: M×N)
    for every batch element b (b = batch * hidden_dim).
    """
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col
    pid_b = tl.program_id(2)      # batch / hidden index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the current batch element
    a_batch = A + pid_b * stride_ab
    b_batch = B + pid_b * stride_bb
    c_batch = C + pid_b * stride_cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks
    # --------------------------------------------------------------
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # B tile (K × N) – B is already stored as (K,N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write‑back
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor : [B, N, N] (float32 or bool) or None
            - weights_dict : model parameters
            - config_dict  : {"dim":..., "hidden_dim":..., "nomask":bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (left/right) – fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_fp16.dtype)    # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of a linear)
    # ------------------------------------------------------------------
    left_gate = (F.linear(x_fp16,
                          weights["left_gate.weight"].to(torch.float16))
                 .sigmoid())
    right_gate = (F.linear(x_fp16,
                           weights["right_gate.weight"].to(torch.float16))
                  .sigmoid())
    out_gate = (F.linear(x_fp16,
                         weights["out_gate.weight"].to(torch.float16))
                .sigmoid())

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul – batched GEMM (one GEMM per hidden channel)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, seq_len, hidden_dim
    # reshape to (B, H, N, N) and then to (B*H, N, N) for the kernel
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (K,N)

    left_flat = left_mat.view(B * hidden_dim, N, N)    # (batch, M, K)
    right_flat = right_mat.view(B * hidden_dim, N, N)  # (batch, K, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # --------------------------------------------------------------
    # Strides (row‑major)
    # --------------------------------------------------------------
    stride_am = left_flat.stride(1)   # = N
    stride_ak = left_flat.stride(2)   # = 1
    stride_bk = right_flat.stride(1)  # = N
    stride_bn = right_flat.stride(2)  # = 1
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    stride_ab = left_flat.stride(0)   # batch stride = N*N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # --------------------------------------------------------------
    # Tile sizes – larger tiles give better occupancy on H100
    # --------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    _batched_gemm_kernel[(grid_x, grid_y, grid_z)](
        left_flat,                      # A
        right_flat,                     # B (already transposed)
        out_flat,                       # C
        N, N, N,                        # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H]  (fp16 → fp32 for the next steps)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate                               # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])      # -> [B, N, N, dim]

    return out
