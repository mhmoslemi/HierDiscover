"""
TriMul (outgoing) – Triton‑accelerated forward pass

The implementation follows the reference in AlphaFold‑3 and fuses the
computationally heavy part (the “TriMul” contraction) into a single
batched GEMM kernel written in Triton.

Algorithm
---------
1️⃣  Layer‑norm on the input tensor (dim‑last).
2️⃣  Linear projections (left/right) and their gates are computed in FP16
    with PyTorch’s highly‑optimized GEMM (torch.nn.functional.linear).
3️⃣  Optional pairwise mask multiplication.
4️⃣  Apply the left/right gates (sigmoid) – element‑wise.
5️⃣  Core contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batch of H independent N×N matrix multiplications (one per
   hidden channel).  The batch dimension is B·H.  A custom Triton kernel
   performs the batched GEMM C = A @ Bᵀ with FP16 inputs and FP32
   accumulation for numerical safety.
6️⃣  Layer‑norm on the hidden dimension, gating with the out‑gate,
    and a final linear projection back to the original dim.

All heavy arithmetic (steps 2‑5) runs in FP16, while the two layer‑norms
use FP32 to keep statistics accurate.  The final result is returned in
FP32.

The kernel is heavily tiled (64‑128) and pipelined (num_stages=4) to
exploit H100’s tensor cores and shared‑memory bandwidth.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) for the TriMul core
# --------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A,                      # ptr to left matrix batch (FP16)
    B,                      # ptr to right matrix batch (FP16, already transposed)
    C,                      # ptr to output batch (FP16)
    M, N, K,                # matrix dimensions (all = seq_len)
    stride_am, stride_ak,   # A: row‑stride, col‑stride
    stride_bk, stride_bn,   # B: row‑stride (k), col‑stride (n)
    stride_cm, stride_cn,   # C: row‑stride, col‑stride
    stride_ab, stride_bb, stride_cb,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr = 4,
    num_warps: tl.constexpr = 8,
):
    """Compute C[b] = A[b] @ B[b].T for each batch element."""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch (B * hidden_dim)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the current batch element
    # ------------------------------------------------------------------
    a_ptr = A + pid_b * stride_ab
    b_ptr = B + pid_b * stride_bb
    c_ptr = C + pid_b * stride_cb

    # Accumulator (FP32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K-loop with double‑buffering (prefetch)
    # ------------------------------------------------------------------
    # Load the first tile outside the loop to hide latency
    k = 0
    offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = offs_k < K

    a_offs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_offs = b_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    a = tl.load(a_offs,
                mask=(mask_m[:, None] & mask_k[None, :]),
                other=0.0).to(tl.float32)
    b = tl.load(b_offs,
                mask=(mask_k[:, None] & mask_n[None, :]),
                other=0.0).to(tl.float32)

    # Prefetch loop
    for k in range(1, tl.cdiv(K, BLOCK_K)):
        # --------------------------------------------------------------
        # Compute with the previously loaded tiles
        # --------------------------------------------------------------
        acc += tl.dot(a, b)

        # --------------------------------------------------------------
        # Load next tiles (prefetch)
        # --------------------------------------------------------------
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_offs = a_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_offs = b_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_offs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)
        b = tl.load(b_offs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

    # multiply‑add for the final tile
    acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result (FP16)
    # ------------------------------------------------------------------
    c_offs = c_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_offs,
             acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# --------------------------------------------------------------
# Public entry point used by the test harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]   (float32)
            - mask_tensor : [B, N, N] or None
            - weights_dict: model parameters
            - config_dict : contains "dim", "hidden_dim", "nomask", ...

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack and move to device
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype_fp16 = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_fp16 = x.to(dtype_fp16)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (left/right) – FP16 GEMM
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_fp16, weights["left_proj.weight"].to(dtype_fp16)
    )   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(
        x_fp16, weights["right_proj.weight"].to(dtype_fp16)
    )   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.unsqueeze(-1).to(dtype_fp16)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear(x))
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_fp16, weights["left_gate.weight"].to(dtype_fp16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_fp16, weights["right_gate.weight"].to(dtype_fp16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_fp16, weights["out_gate.weight"].to(dtype_fp16)
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core TriMul: batch of hidden_dim GEMMs (N×N @ N×Nᵀ)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                      # batch size & seq length
    # Reshape to (B * hidden_dim, N, N) – the batch dimension for the kernel
    left_mat = left.permute(0, 3, 1, 2).contiguous()    # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – transposed on purpose

    left_flat = left_mat.view(B * hidden_dim, N, N)     # A
    right_flat = right_mat.view(B * hidden_dim, N, N)   # Bᵀ
    out_flat = torch.empty_like(left_flat, dtype=dtype_fp16)  # C

    # --------------------------------------------------------------
    # Strides for row‑major storage (FP16 tensors)
    # --------------------------------------------------------------
    stride_am = left_flat.stride(1)   # row‑stride (M dimension)
    stride_ak = left_flat.stride(2)   # col‑stride (K dimension)
    stride_bk = right_flat.stride(1)  # row‑stride in K dim
    stride_bn = right_flat.stride(2)  # col‑stride (N dimension)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    stride_ab = left_flat.stride(0)   # batch stride = N * N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # --------------------------------------------------------------
    # Tiling parameters – tuned for H100 tensor cores
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim                     # total batch count

    _batched_gemm_kernel[(grid_x, grid_y, grid_z)](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # --------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and cast to FP32
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate and final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                      # broadcast over hidden_dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
