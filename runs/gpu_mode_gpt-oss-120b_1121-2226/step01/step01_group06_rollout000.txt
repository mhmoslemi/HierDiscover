"""
Triton‑accelerated forward pass of the *outgoing* TriMul operator.

Algorithm
---------
1. Layer‑norm on the input (last dimension = dim).
2. Two linear projections → left / right (shape [B,N,N,H]).
3. (Optional) mask is applied to left/right.
4. Three gating networks (sigmoid of linear projections) are computed and
   multiplied into left, right and later into the final output.
5. Heavy cubic contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply of shape
        (B·H, N, N) @ (B·H, N, N)ᵀ .
   We implement it with a custom Triton kernel that performs the GEMM in
   FP16 with FP32 accumulation.
6. Second Layer‑norm over the hidden dimension H.
7. Apply the output gate.
8. Final linear projection back to the original dimension dim.

All expensive arithmetic (step 5) runs on the Triton kernel, everything
else is handled with PyTorch primitives.  The kernel is autotuned for
common sizes (N up to 1024) on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({"BLOCK_M": 64,  "BLOCK_N": 64,  "BLOCK_K": 32}, num_warps=4),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32}, num_warps=8),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 64,  "BLOCK_K": 32}, num_warps=8),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ab, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bb, stride_bk, stride_bn,   # B strides: batch, K, N
    stride_cb, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Computes a batched GEMM C = A @ Bᵀ for FP16 inputs with FP32 accumulation."""
    pid = tl.program_id(0)

    # ---- grid layout -------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)               # batch index (B·H)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # ---- tile offsets -------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension ----------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A pointer (batch, M, K)
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # B pointer (batch, K, N) – we need the original B (already transposed)
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # ---- write back ---------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication for FP16 tensors:
        out = a @ bᵀ
    a : (B, M, K)  FP16
    b : (B, K, N)  FP16   (already transposed, i.e. original RHSᵀ)
    Returns out : (B, M, N)  FP16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Block sizes are tuned by @triton.autotune
    _batched_matmul_kernel[
        (B * ((M + 127) // 128) * ((N + 127) // 128),)  # grid size (upper bound)
    ](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        # NOTE: The autotune decorator will pick BLOCK_M/N/K at compile time.
    )
    return out


# ----------------------------------------------------------------------
# Triton‑based TriMul (outgoing) entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator, fully compatible with the
    reference PyTorch implementation.  The dominant N³ contraction is
    performed by a Triton kernel; the rest of the pipeline uses
    standard PyTorch primitives.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict containing all model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load model weights (all are FP32; linear layers are bias‑free)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left / right (shape [..., hidden_dim])
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating tensors (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate   # element‑wise modulation
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy cubic contraction – performed in FP16 via Triton
    # ------------------------------------------------------------------
    # Cast to half‑precision to reduce bandwidth.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, H, N, N) where the trailing two axes correspond to
    # the two matrix dimensions of the GEMM:
    #   left  : (i, k)  -> (..., i, k)
    #   right : (j, k)  -> (..., j, k) → we need (k, j) for Bᵀ.
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N] (i,k)
    right_perm_t = right_h.permute(0, 3, 2, 1).contiguous()      # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions → (B·H, N, N)
    A = left_perm.view(B * H, N, N)          # (M=N, K=N)
    B_mat = right_perm_t.view(B * H, N, N)   # (K=N, N)

    # Batched GEMM: C = A @ B_matᵀ (but B_mat already holds the transposed layout)
    C = _batched_matmul_fp16(A, B_mat)      # (B·H, N, N), FP16

    # Reshape back to [B, N, N, H]
    out_fp16 = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6️⃣ Second Layer‑norm over the hidden dimension (H)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # LayerNorm benefits from FP32 precision
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # [B,N,N,H], float32

    # ------------------------------------------------------------------
    # 7️⃣ Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim], float32

    return out
```
