"""
TriMul (outgoing) – fast forward implementation.

The operator works on a tensor x of shape [B, N, N, C] (C = dim) and
produces a tensor of the same shape.  The heavy O(N³·hidden_dim) part is
the contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

for every hidden channel *d*.  We fuse this contraction with a
batched matrix‑multiply kernel written in Triton.  The kernel computes
C = A @ B for many independent batches (B·hidden_dim), using FP16
operands and FP32 accumulation, which matches the precision used by the
reference PyTorch code.

All cheaper operations (LayerNorm, Linear projections, sigmoid gates,
masking, final LayerNorm and projection) are performed with PyTorch
functions.  This yields a highly‑optimized forward pass that comfortably
meets the sub‑millisecond target on an H100 for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched FP16 matmul C = A @ B  (C = B·hidden_dim independent GEMMs)
# A : (batch, M, K)   – fp16
# B : (batch, K, N)   – fp16
# C : (batch, M, N)   – fp16 (FP32 accumulator internally)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # base pointers
    M, N, K,                        # matrix dimensions (int32)
    stride_az, stride_am, stride_ak,   # strides for A: (batch, M, K)
    stride_bz, stride_bk, stride_bn,   # strides for B: (batch, K, N)
    stride_cz, stride_cm, stride_cn,   # strides for C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b, :, :] = A[b, :, :] @ B[b, :, :]   for each batch b.
    All tensors are in row‑major layout.
    """
    # ------------------------------------------------------------------
    # Program IDs: 0 -> batch, 1 -> tile over M, 2 -> tile over N
    # ------------------------------------------------------------------
    pid_z = tl.program_id(0)  # batch (B * hidden_dim)
    pid_m = tl.program_id(1)  # tile row
    pid_n = tl.program_id(2)  # tile column

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # ------------------------------------------------------------------
    # Pointers to the start of the current batch
    # ------------------------------------------------------------------
    a_batch = a_ptr + pid_z * stride_az
    b_batch = b_ptr + pid_z * stride_bz
    c_batch = c_ptr + pid_z * stride_cz

    # ------------------------------------------------------------------
    # Pointer matrix for the A and B tiles
    # ------------------------------------------------------------------
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in FP32 for better numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                     # Σ_k a[i,k] * b[k,j]
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back the result (cast to FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask

    # --------------------------------------------------------------
    # Load model weights (all stored as float32 tensors)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]     # (hidden_dim,)
    to_out_weight      = weights["to_out.weight"]        # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (over the last dimension = dim)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast to hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left  = left  * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – compute in FP32 then cast to FP16 later
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,hidden_dim)

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to half‑precision for the O(N³) reduction
    # --------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # batch and sequence length

    # --------------------------------------------------------------
    # 5a. Rearrange tensors to (batch*hidden_dim, M, K) layout
    #      left  -> A : (B, hidden_dim, N, N)  → (B*hidden_dim, N, N)
    #      right -> B : (B, hidden_dim, N, N)  → (B*hidden_dim, N, N)   (K,N) layout
    # --------------------------------------------------------------
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output container (FP16)
    out_mat = torch.empty_like(left_mat)

    # --------------------------------------------------------------
    # 5b. Compute strides for the Triton kernel
    # --------------------------------------------------------------
    stride_az = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bz = right_mat.stride(0)
    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)

    stride_cz = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # --------------------------------------------------------------
    # 5c. Launch the batched matmul kernel
    # --------------------------------------------------------------
    # Tuning parameters – they work well on H100 FP16 Tensor Cores.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch dimension
        triton.cdiv(N, BLOCK_M),            # tiles over M
        triton.cdiv(N, BLOCK_N),            # tiles over N
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=stride_az,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bz=stride_bz,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cz=stride_cz,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # 8 warps is a sweet spot for FP16 on H100
        num_stages=2,           # double‑buffering of loads
    )

    # --------------------------------------------------------------
    # 5d. Reshape back to (B, N, N, hidden_dim) and promote to FP32
    # --------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                     # out_gate still FP32
    out = F.linear(out, to_out_weight)       # (B, N, N, dim)

    return out
