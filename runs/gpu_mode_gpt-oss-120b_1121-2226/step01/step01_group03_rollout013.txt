```
@triton.jit
def _fused_gate_mul_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    n_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(proj_ptr + offs, mask=mask) # half
    b = tl.load(gate_ptr + offs, mask=mask) # half
    # compute sigmoid
    # 1/(1+exp(-b))
    # Since tl.exp works on fp16, good
    neg_b = -b
    exp_neg_b = tl.exp(neg_b)
    sigmoid_b = 1.0 / (1.0 + exp_neg_b)
    if HAS_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask)
        out = a * sigmoid_b * m
    else:
        out = a * sigmoid_b
    tl.store(out_ptr + offs, out, mask=mask)

"""
TriMul (outgoing) – highly‑optimised forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input over the feature dimension (dim).
2️⃣  Fuse **all** linear projections (left_proj, left_gate,
    right_proj, right_gate, out_gate) into a *single* GEMM:
       Y = LN(x) @ Wᵀ   where   W ∈ ℝ^{5·hidden_dim × dim}.
   Afterwards we slice Y → left_proj, left_gate, right_proj, right_gate,
   out_gate.
3️⃣  Fuse element‑wise gating, optional mask and the sigmoid in one
   Triton kernel (producing the half‑precision tensors `left` and
   `right`).  This avoids materialising the four intermediate tensors.
4️⃣  Compute the heavy O(N³·hidden_dim) term via a **batched GEMM**
   on FP16 tensors:
        out = (left @ rightᵀ)   ⇢  shape [B, N, N, hidden_dim].
5️⃣  Layer‑norm over the hidden dimension (FP16), multiply by the
   pre‑computed out‑gate (sigmoid), and apply the final linear projection
   back to `dim`.  All these heavy kernels run in FP16 – the final result
   is cast back to FP32 for compatibility.
6️⃣  The implementation reduces the number of GEMMs from 5 to 1 (step 2)
   and replaces multiple point‑wise Tensor operations by a single
   Triton kernel (step 3), delivering a large speed‑up over a naïve
   implementation.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: fused projection * sigmoid(gate) * optional mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mul_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,      # pointers
    n_elements, hidden_dim,                     # runtime args
    BLOCK_SIZE: tl.constexpr,                   # compile‑time tile size
    HAS_MASK: tl.constexpr,                     # 0/1 compile‑time flag
):
    """
    out = proj * sigmoid(gate) * mask   (mask broadcast over hidden_dim)
    proj, gate, out : shape [B, N, N, H]  (flattened)
    mask            : shape [B, N, N]    (flattened) – broadcasted
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < n_elements

    # Load projection and gate (FP16)
    proj = tl.load(proj_ptr + offs, mask=active)
    gate = tl.load(gate_ptr + offs, mask=active)

    # sigmoid(gate) = 1 / (1 + exp(-gate))
    sig = 1.0 / (1.0 + tl.exp(-gate))

    if HAS_MASK:
        # each hidden dimension shares the same mask entry → divide by H
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=active)
        out = proj * sig * m
    else:
        out = proj * sig

    tl.store(out_ptr + offs, out, mask=active)


def _gate_mul(proj: torch.Tensor,
              gate: torch.Tensor,
              mask: torch.Tensor | None,
              hidden_dim: int) -> torch.Tensor:
    """
    Triton‑backed element‑wise fusion:
        out = proj * sigmoid(gate) * mask
    All tensors are expected to be contiguous, half‑precision and have the
    same shape ``[B, N, N, hidden_dim]``.  ``mask`` may be ``None`` (no mask)
    or a half‑tensor of shape ``[B, N, N]``.
    """
    total = proj.numel()                      # B·N·N·H
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    proj_flat = proj.view(-1)
    gate_flat = gate.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=proj.dtype, device=proj.device)
        has_mask = 0

    out = torch.empty_like(proj)
    _gate_mul_kernel[grid](
        proj_flat,
        gate_flat,
        mask_flat,
        out.view(-1),
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor or None, shape [B, N, N]
        - weights_dict : mapping str → torch.Tensor (see code)
        - config_dict  : must contain ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` flag.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear projections into ONE GEMM
    #    W ∈ ℝ^{5·hidden_dim × dim}
    # ------------------------------------------------------------------
    # concatenate weights (no bias)
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    )                                            # (5·H, dim)

    # cast inputs & weights to FP16 for the heavy GEMM
    x_norm_fp16 = x_norm.to(torch.float16)
    combined_w_fp16 = combined_w.to(torch.float16)

    # shape: [B, N, N, 5·hidden_dim]   (still FP16)
    combined = F.linear(x_norm_fp16, combined_w_fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Slice the combined tensor
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = combined[..., 0 * H : 1 * H]
    left_gate   = combined[..., 1 * H : 2 * H]
    right_proj  = combined[..., 2 * H : 3 * H]
    right_gate  = combined[..., 3 * H : 4 * H]
    out_gate    = combined[..., 4 * H : 5 * H]   # pre‑sigmoid

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask handling
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_fp16 = None
    else:
        mask_fp16 = mask.to(torch.float16)   # [B, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gating (sigmoid) + mask multiplication via Triton
    # ------------------------------------------------------------------
    left  = _gate_mul(left_proj, left_gate, mask_fp16, hidden_dim)   # FP16
    right = _gate_mul(right_proj, right_gate, mask_fp16, hidden_dim) # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Heavy O(N³·hidden_dim) operation – batched GEMM (FP16)
    #    left, right : [B, N, N, H]   →   (B·H, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_mat = left_mat @ right_matᵀ   (batched GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back → [B, N, N, H]  (still FP16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply out_gate (sigmoid) – point‑wise multiply
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate)               # FP16, same shape as out_gate pre‑sigmoid
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                          # FP16 → [B, N, N, dim]

    # Cast to FP32 for the final output (matches reference implementation)
    return out.to(torch.float32)
```
