import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul forward (outgoing) implementation – Triton accelerated.

Algorithm
---------
1. Layer‑norm over the last dimension (`dim`) of the 4‑D input tensor.
2. Linear projections (no bias) → left, right (shape [B, N, N, H]).
3. Optional mask: left/right ← left/right * mask[...,None].
4. Compute three gating tensors (sigmoid of linear projections):
   left_gate, right_gate, out_gate (shape [B, N, N, H]).
   Apply gates: left ← left * left_gate; right ← right * right_gate.
5. Compute the cubic contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication over the shared `k` axis.
   We fuse this step with a Triton kernel: for each (b,h) we perform
   a GEMM A (N×N) @ Bᵀ (N×N) → C (N×N) in FP16 accumulation in FP32.
6. Reshape back to [B,N,N,H] and apply a second LayerNorm (over H).
7. Multiply by `out_gate` (shape [B,N,N,H]).
8. Final linear projection back to `dim`.
All heavy arithmetic (step 5) runs on a custom Triton kernel; the rest is
performed with PyTorch primitives (layer‑norm, linear, sigmoid).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----------------------------------------------
    #  block coordinates: (batch, block_m, block_n)
    # ----------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # --------------------------------------------------
    #  offsets for the current tile (including masks)
    # --------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------
    #  Loop over K dimension in tiles of size BLOCK_K
    # --------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for A (shape: batch, M, K)
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # pointers for B (shape: batch, K, N) – we need Bᵀ (K×N)
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product accumulation (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------
    #  Write back the result
    # --------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    a: (B, M, K)   FP16
    b: (B, K, N)   FP16   (already transposed, i.e. original RHSᵀ)
    Returns: (B, M, N) FP16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Tunable block sizes – work well for N up to 1024 on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_m * num_n,)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (or None)
            - weights: dict of torch.Tensor containing all model weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------
    #  unpack arguments
    # --------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    #  Load model weights (all are FP32 tensors without bias unless noted)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    #  1️⃣ Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype=fp32

    # ------------------------------------------------------------------
    #  2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    #  3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1).to(left.dtype)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    #  4️⃣ Gating networks (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  5️⃣ TriMul contraction – heavy N³ work, done in FP16 via Triton
    # ------------------------------------------------------------------
    # Cast to FP16 before the massive GEMM to cut memory bandwidth.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute to (B, H, N, N) where the last two dims are (i,k) and (k,j)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N] (i,k)
    right_perm_t = right_h.permute(0, 3, 2, 1).contiguous()      # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions for the kernel
    A = left_perm.view(B * H, N, N)          # (B*H, M=N, K=N)
    B_mat = right_perm_t.view(B * H, N, N)   # (B*H, K=N, N)

    # Perform the batched matmul C = A @ B_matᵀ  (but B_mat is already transposed)
    C = _batched_matmul_fp16(A, B_mat)      # (B*H, N, N) in FP16

    # Reshape back to 4‑D tensor and restore original layout
    out_perm = C.view(B, H, N, N)            # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] (FP16)

    # ------------------------------------------------------------------
    #  6️⃣ Second Layer‑norm over hidden dimension (H)
    # ------------------------------------------------------------------
    # Convert to FP32 for LayerNorm (more stable)
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    #  7️⃣ Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    #  8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
