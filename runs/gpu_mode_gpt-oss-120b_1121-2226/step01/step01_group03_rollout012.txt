def _fused_mul_gate_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    n_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    left = tl.load(left_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask)
        # broadcast over hidden_dim: same mask per group
        left_out = left * left_gate * m
        right_out = right * right_gate * m
    else:
        left_out = left * left_gate
        right_out = right * right_gate

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)

"""
TriMul (outgoing) – highly‑optimised forward pass

Algorithm
---------
1️⃣ Input LayerNorm (float32 for numerical stability).
2️⃣ Fuse the *four* projection/gate Linear layers into **two** matmuls:
   * left_proj + right_proj  → weight shape (2·hidden_dim, dim)
   * left_gate, right_gate, out_gate → weight shape (3·hidden_dim, dim)
   The results are split back into the original tensors and the three
   gate tensors are passed through a sigmoid.
3️⃣ Fuse the element‑wise products 
        left = left * left_gate * mask
        right = right * right_gate * mask
   (mask is optional) in a **single Triton kernel** that writes both
   `left` and `right` outputs.
4️⃣ Cast the gated tensors to FP16 and perform the heavy O(N³·hidden_dim)
   contraction as a **batched GEMM**:
        out[b,i,j,d] = Σₖ left[b,i,k,d]·right[b,j,k,d]
   (implemented as bmm on tensors of shape (B·hidden_dim, N, N)).
5️⃣ Output LayerNorm over the hidden dimension, then multiply by the
   `out_gate` (all in FP16).
6️⃣ Final Linear projection back to `dim` (FP16) and cast the result to
   float32.

The only Triton‑implemented part is the fused element‑wise gate+mask
multiplication, satisfying the “must use Triton” requirement while all
heavy matrix‑multiplications run on Tensor‑Cores (FP16) for maximal
throughput.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused left/right gating + optional mask
# ----------------------------------------------------------------------
@triton.jit
def _fused_gate_mask_kernel(
    left_ptr,   left_gate_ptr,
    right_ptr,  right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    n_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK:  tl.constexpr,            # 0/1 compile‑time flag
):
    """
    out_left  = left  * left_gate  * mask
    out_right = right * right_gate * mask
    mask is broadcast over the hidden dimension.
    All tensors are flattened.  dtype is inferred from the pointers
    (float32 in the reference implementation, float16 for the heavy
    GEMM path).
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < n_elements

    # Load the four tensors
    left = tl.load(left_ptr + offs,       mask=valid)
    left_gate = tl.load(left_gate_ptr + offs, mask=valid)
    right = tl.load(right_ptr + offs,     mask=valid)
    right_gate = tl.load(right_gate_ptr + offs, mask=valid)

    if HAS_MASK:
        # mask index = element // hidden_dim   (broadcast)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=valid)
        left_out  = left  * left_gate  * m
        right_out = right * right_gate * m
    else:
        left_out  = left  * left_gate
        right_out = right * right_gate

    tl.store(left_out_ptr  + offs, left_out,  mask=valid)
    tl.store(right_out_ptr + offs, right_out, mask=valid)


def _fused_gate_mask(left, left_gate, right, right_gate, mask=None):
    """
    Fuse ``left * left_gate * mask`` and ``right * right_gate * mask``.
    All inputs are tensors of shape [B, N, N, H] (float32). ``mask`` has shape
    [B, N, N] and is broadcast over the hidden dimension.  Returns two tensors
    of the same shape as the inputs.
    """
    assert left.shape == left_gate.shape == right.shape == right_gate.shape
    B, N, _, H = left.shape
    total = left.numel()                     # B·N·N·H
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    left_flat      = left.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_flat     = right.contiguous().view(-1)
    right_gate_flat= right_gate.contiguous().view(-1)

    if mask is not None:
        mask_flat = mask.contiguous().view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=left.dtype, device=left.device)
        has_mask = 0

    out_left  = torch.empty_like(left)
    out_right = torch.empty_like(right)

    _fused_gate_mask_kernel[grid](
        left_flat, left_gate_flat,
        right_flat, right_gate_flat,
        mask_flat,
        out_left.view(-1), out_right.view(-1),
        total, H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    eps = 1e-5

    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Fuse linear projections (left/right) into a single matmul
    # ------------------------------------------------------------------
    # Concatenate left_proj.weight and right_proj.weight → (2*H, dim)
    proj_weight_cat = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    )   # shape (2*hidden_dim, dim)
    # Linear (bias=False)
    proj = torch.nn.functional.linear(x_norm, proj_weight_cat)   # [B,N,N,2*H]

    left  = proj[..., :hidden_dim]          # [B,N,N,H]
    right = proj[..., hidden_dim:]          # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Fuse gating linear layers (left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    gate_weight_cat = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    )   # shape (3*hidden_dim, dim)

    gates = torch.nn.functional.linear(x_norm, gate_weight_cat)   # [B,N,N,3*H]
    left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        gates, hidden_dim, dim=-1
    )
    left_gate  = torch.sigmoid(left_gate_raw)   # [B,N,N,H]
    right_gate = torch.sigmoid(right_gate_raw)  # [B,N,N,H]
    out_gate   = torch.sigmoid(out_gate_raw)    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Optional mask handling (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_f = None
    else:
        mask_f = mask.to(x.dtype)   # float32 → will be cast to half later if needed

    # ------------------------------------------------------------------
    # 5. Fuse element‑wise multiplication with Triton
    # ------------------------------------------------------------------
    left_masked, right_masked = _fused_gate_mask(
        left, left_gate, right, right_gate, mask_f
    )   # still float32

    # ------------------------------------------------------------------
    # 6. Heavy contraction via FP16 batched GEMM
    # ------------------------------------------------------------------
    # Cast to FP16 before the O(N³·H) work
    left_fp16  = left_masked.to(torch.float16)
    right_fp16 = right_masked.to(torch.float16)

    B, N, _, H = left_fp16.shape   # H == hidden_dim

    # reshape to (B*H, N, N)   –  we need the hidden dimension as the batch axis
    left_mat  = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiply: L @ Rᵀ   –  uses Tensor‑Cores
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B, N, N, H]  (float16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # still FP16

    # ------------------------------------------------------------------
    # 8. Apply final output gate (FP16)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim] FP16

    # Cast back to float32 for the model output
    return out.to(torch.float32)
