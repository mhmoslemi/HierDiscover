"""
TriMul (outgoing) – forward only
================================

This implementation follows the reference PyTorch module but accelerates the
quadratic contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

with a custom Triton kernel that performs a batched GEMM
C = A @ Bᵀ where A and B have shapes (B·H, N, N).  The kernel is tuned for
the H100: 128×128 tiles, 64‑wide K‑tiles and FP32 accumulation in FP16
precision.  All other operations (layer‑norms, linear projections,
sigmoids, mask handling) are kept in PyTorch because they are memory‑bound
and already highly optimized.

The heavy N³ work therefore runs entirely on the GPU without any Python
loops, while the rest of the pipeline stays on the fast torch kernels.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A: (B·H, M, K), B: (B·H, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix sizes
    stride_ab, stride_am, stride_ak,     # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,     # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,     # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM for many (B·H) independent (M×K) @ (K×N) multiplications.
    The kernel uses a classic tiling strategy with fp16 inputs and fp32
    accumulation.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------
    #  Decode (batch, tile_m, tile_n) from linear pid
    # --------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)                     # which (B·H) we are processing
    block_idx = pid % (num_m * num_n)
    tile_m = block_idx // num_n
    tile_n = block_idx % num_n

    # --------------------------------------------------
    #  Tile offsets (including boundary mask)
    # --------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (more accurate)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------
    #  Loop over K dimension in tiles of size BLOCK_K
    # --------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for A (batch, M, K)
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # pointers for B (batch, K, N) – we need Bᵀ, so stride order is (K,N)
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # fp16 × fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------
    #  Write back the result as fp16
    # --------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    a: (B·H, M, K)  fp16, contiguous
    b: (B·H, K, N)  fp16, contiguous (already transposed)
    Returns:
        c: (B·H, M, N) fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B_H, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    # Output buffer
    c = torch.empty((B_H, M, N), dtype=torch.float16, device=a.device)

    # Heuristics for block sizes – work well up to N = 1024 on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B_H * num_m * num_n,)

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        c.stride(0),
        c.stride(1),
        c.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data : tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of all model weights (fp32)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------
    #  unpack inputs
    # --------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------
    #  load weight tensors (all fp32, no bias unless noted)
    # --------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------
    #  1️⃣ LayerNorm over input channels (dim)
    # --------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  fp32

    # --------------------------------------------------
    #  2️⃣ Linear projections (no bias) → left/right (hidden_dim)
    # --------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------
    #  3️⃣ Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # --------------------------------------------------
    #  4️⃣ Gating networks (sigmoid of linear projections)
    # --------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------
    #  5️⃣ Contraction (heavy N³ work) – Triton batched GEMM
    # --------------------------------------------------
    # Cast to fp16 to maximise Tensor‑Core utilisation
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Rearrange to (B, H, N, N) where the last two dims are (i,k) and (k,j)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()       # [B, H, N, i=k]
    right_perm_t = right_h.permute(0, 3, 2, 1).contiguous()   # [B, H, N, k=j]

    B, H, N, _ = left_perm.shape

    # Collapse batch and hidden dimensions → (B·H, N, N)
    A = left_perm.view(B * H, N, N)          # (M=N, K=N)
    B_mat = right_perm_t.view(B * H, N, N)   # (K=N, N)

    # Batched matmul using the Triton kernel
    C = _batched_matmul_fp16(A, B_mat)      # (B·H, N, N) fp16

    # Restore original layout: [B, N, N, H]
    out_perm = C.view(B, H, N, N)            # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]  fp16

    # --------------------------------------------------
    #  6️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------
    out = out.to(torch.float32)   # cast for numeric stability in LN
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # --------------------------------------------------
    #  7️⃣ Apply output gate (element‑wise)
    # --------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------
    #  8️⃣ Final linear projection back to `dim`
    # --------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]  fp32

    return out
```
