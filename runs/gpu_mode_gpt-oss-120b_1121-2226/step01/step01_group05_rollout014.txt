```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        a_ptrs = a_ptr + pid_h * stride_ah + offs_m[:, None] * stride_ai + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        b_ptrs = b_ptr + pid_h * stride_bh + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    c_ptrs = c_ptr + pid_h * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    TriMul “outgoing” forward pass (AlphaFold3 style).

    Algorithm
    ---------
    1. Layer‑norm on the input (last dimension = dim).
    2. Linear projections → left/right tensors of shape [B, N, N, H].
    3. Optional mask (broadcast on the hidden dimension).
    4. Compute three sigmoid gates (left, right, output) from the
       normalized input.
    5. Apply the left/right gates: left = left_proj * left_gate,
       right = right_proj * right_gate.
    6. Cast left/right to fp16, permute to [B, H, N, N] and flatten the
       (B, H) axes → tensors of shape [BH, N, N].
    7. Batched matrix‑multiply for every hidden channel:
       out[b*h, i, j] = Σ_k left[b*h, i, k] * right[b*h, j, k]
       (implemented as a tiled GEMM in Triton).
    8. Reshape the result back to [B, N, N, H].
    9. Second Layer‑norm over the hidden dimension, multiply by the
       output gate and project back to the original `dim`.
    The final tensor is float‑32.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------
    # 1. Input LayerNorm
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim], float32

    # ---------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ---------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ---------------------------------------------------------------
    # 4. Gating
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ---------------------------------------------------------------
    # 5. Apply gates – stay in float32 for the multiplication
    # ---------------------------------------------------------------
    left = left_proj * left_gate    # [B, N, N, H], float32

"""
TriMul “outgoing” forward pass (AlphaFold3 style).

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim).
2. Linear projections → left / right tensors of shape [B, N, N, H].
3. (Optional) mask broadcast on the hidden dimension.
4. Compute three sigmoid gates (left, right, output) from the normalized input.
5. Apply the left/right gates:
        left  = left_proj  * left_gate
        right = right_proj * right_gate
6. Cast the gated tensors to fp16, permute to [B, H, N, N] and flatten (B·H)
   → tensors of shape [BH, N, N].
7. Batched matrix‑multiply for every hidden channel:
        out[b·h, i, j] = Σ_k left[b·h, i, k] * right[b·h, j, k]
   (implemented as a tiled GEMM in Triton).
8. Reshape the result back to [B, N, N, H].
9. Second Layer‑norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original `dim`.
The final result is a float‑32 tensor of shape [B, N, N, dim].

The heavy cubic step (7) is a classic tiled GEMM with three axes:
    axis‑0 → i‑tile, axis‑1 → j‑tile, axis‑2 → (batch·hidden) tile.
Tile sizes (128×128×32) and 8 warps per program give high arithmetic
intensity while staying well inside the 64 KB shared‑memory budget of an
H100 SM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                     # left  tensor   (fp16)
    b_ptr,                     # right tensor   (fp16, will be read as transposed)
    c_ptr,                     # output tensor  (fp32)
    M, N, K,                   # problem sizes (all equal to seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A = left
    stride_bh, stride_bk, stride_bn,   # strides for B = right (as transposed)
    stride_ch, stride_ci, stride_cj,   # strides for C = out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    #  Program IDs
    # ------------------------------------------------------------------
    pid_h = tl.program_id(axis=2)               # (batch * hidden) index
    pid_m = tl.program_id(axis=0)               # i‑tile
    pid_n = tl.program_id(axis=1)               # j‑tile

    # ------------------------------------------------------------------
    #  Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    #  Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over K dimension
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----------- load A tile (M × K) --------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)                     # fp16 → fp32

        # ----------- load B tile (K × N)  (B is accessed transposed) ---
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)                     # fp16 → fp32

        # ----------- inner product --------------------------------------
        acc += tl.dot(a, b)                       # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ------------------------------------------------------------------
    #  Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] , float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)  # [B, N, N, 1]

        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Apply gates (still in float32)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to fp16 and reshape for the batched GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Move hidden dim to the batch axis and make the tensor contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)    # [BH, N, N]   fp16
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]   fp16

    # Output buffer (FP32)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7. Launch the Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden)
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2), # B strides (as transposed)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,   # H100 can comfortably run 8 warps per program
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N)               # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # still float32

    # ------------------------------------------------------------------
    # 10. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
