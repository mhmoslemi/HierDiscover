"""
Outgoing TriMul (AlphaFold‑3) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Input LayerNorm (FP32) on the last dimension.
2️⃣  Fuse the projection and gating linear layers for the “left’’ and “right’’
    tensors:
        [B, N, N, dim] → linear → [B, N, N, 2·hidden] → split
        → proj (FP16) and gate (sigmoid, FP16) → proj * gate.
3️⃣  Optional pairwise mask (broadcast on the hidden dimension).
4️⃣  Compute the core TriMul:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, k, j]
   This is a batch of (B·hidden) independent GEMMs (N×N @ N×N) and is
   realized by a custom Triton kernel that tiles the matrices with
   BLOCK_M×BLOCK_N×BLOCK_K and accumulates in FP32.
5️⃣  Layer‑norm on the hidden dimension, element‑wise gating with `out_gate`,
    and a final linear projection back to `dim` (FP32).

All heavy arithmetic (steps 2‑4) runs in FP16 on Tensor‑cores; the final
normalisation and projection run in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (M,K), B: (K,N))
# Batch dimension corresponds to (batch * hidden_dim) in TriMul.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                # pointers to matrices (FP16)
    M, N, K,                # matrix dimensions (all = seq_len)
    stride_am, stride_ak,   # A: row‑stride, col‑stride
    stride_bk, stride_bn,   # B: row‑stride (k‑dim), col‑stride (n‑dim)
    stride_cm, stride_cn,   # C: row‑stride, col‑stride
    stride_ab, stride_bb, stride_cb,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_b = tl.program_id(2)   # batch / hidden index

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the selected batch element
    a_batch = A + pid_b * stride_ab
    b_batch = B + pid_b * stride_bb
    c_batch = C + pid_b * stride_cb

    # Accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # Load B tile (K × N) – B is already in (K,N) layout
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Tile‑wise matrix multiply
        acc += tl.dot(a, b)

    # Write result back (store as FP16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor : [B, N, N] (bool/float) or None
            - weights_dict: {"norm.weight":..., "norm.bias":..., ...}
            - config_dict : {"dim":..., "hidden_dim":..., "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (keep FP32 for statistics)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for the left side
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2*H, dim)
    left_comb = torch.nn.functional.linear(x_fp16, left_comb_w)                 # [B,N,N,2H]

    left_proj = left_comb[..., :hidden_dim]          # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()  # [B,N,N,H]
    left = left_proj * left_gate                     # FP16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for the right side
    # ------------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_comb = torch.nn.functional.linear(x_fp16, right_comb_w)

    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:].sigmoid()
    right = right_proj * right_gate  # FP16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1).to(x_fp16.dtype)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ out‑gate (no mask)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"]
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul – batch of H independent GEMMs (N×N @ N×N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                # B = batch size, N = seq_len
    # Re‑order to (B, H, N, N) where H = hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  – K ↔ J

    # Collapse batch & head dimensions → (B*H, N, N)
    left_flat = left_mat.view(B * hidden_dim, N, N)
    right_flat = right_mat.view(B * hidden_dim, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides for the Triton kernel (row‑major layout)
    stride_am = left_flat.stride(1)   # = N
    stride_ak = left_flat.stride(2)   # = 1
    stride_bk = right_flat.stride(1)  # = N
    stride_bn = right_flat.stride(2)  # = 1
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    stride_ab = left_flat.stride(0)   # batch stride = N*N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # Tile sizes – chosen for H100 Tensor‑cores (multiples of 16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    _batched_gemm_kernel[(grid_x, grid_y, grid_z)](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim] and cast to FP32
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate.to(torch.float32)   # gating (broadcast on hidden dim)

    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)   # final projection (FP32)

    return out
