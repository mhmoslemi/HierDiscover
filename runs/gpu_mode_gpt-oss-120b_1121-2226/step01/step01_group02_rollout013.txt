"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32).
2.  Linear projections → left/right (float32) and three sigmoid gates.
3.  Fuse mask, left‑gate and right‑gate in a single Triton kernel
    (in‑place on the projected tensors).
4.  Cast the gated tensors to fp16 and reshape:
        (B, N, N, H) → (B·H, N, N)   (H = hidden_dim)
5.  Batched matrix multiplication `out = left @ rightᵀ` is performed with
    Torch’s cuBLAS (torch.bmm) which on H100 exploits Tensor‑Cores.
6.  Reshape the result back to (B, N, N, H).
7.  Layer‑norm over the hidden dimension, element‑wise out‑gate,
    and final linear projection back to `dim`.

Only the mask/gate fusion is done in Triton; the heavy N³ matrix
multiplication benefits from the highly‑optimized cuBLAS kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate (in‑place)
# --------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    # tensors
    left_ptr,                 # fp32 [B, N, N, H]
    right_ptr,                # fp32 [B, N, N, H]
    left_gate_ptr,            # fp32 [B, N, N, H]
    right_gate_ptr,           # fp32 [B, N, N, H]
    mask_ptr,                 # fp32 [B, N, N] (broadcasted over H)

    # dimensions
    B, N, H,
    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # index along first sequence axis
    pid_j = tl.program_id(2)   # index along second sequence axis

    # ------------------------------
    #  Offsets for the current tile
    # ------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    # out‑of‑bounds masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # -------------------------------------------------
    #  Load mask (2‑D) and broadcast to the H dimension
    # -------------------------------------------------
    mask_addr = (
        mask_ptr
        + pid_b * stride_mask_b
        + offs_i[:, None] * stride_mask_i
        + offs_j[None, :] * stride_mask_j
    )
    mask_val = tl.load(mask_addr,
                       mask=mask_i[:, None] & mask_j[None, :],
                       other=0.0)               # shape (BLOCK_I, BLOCK_J)
    mask_val = mask_val[:, :, None]                # (BLOCK_I, BLOCK_J, 1)

    # -------------------------------------------------
    #  Process LEFT tensor (in‑place)
    # -------------------------------------------------
    left_addr = (
        left_ptr
        + pid_b * stride_left_b
        + offs_i[:, None, None] * stride_left_i
        + offs_j[None, :, None] * stride_left_j
        + offs_h[None, None, :] * stride_left_h
    )
    left_val = tl.load(
        left_addr,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )
    lgate_addr = (
        left_gate_ptr
        + pid_b * stride_lgate_b
        + offs_i[:, None, None] * stride_lgate_i
        + offs_j[None, :, None] * stride_lgate_j
        + offs_h[None, None, :] * stride_lgate_h
    )
    lgate_val = tl.load(
        lgate_addr,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )
    left_out = left_val * lgate_val * mask_val
    tl.store(
        left_addr,
        left_out,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )

    # -------------------------------------------------
    #  Process RIGHT tensor (in‑place)
    # -------------------------------------------------
    right_addr = (
        right_ptr
        + pid_b * stride_right_b
        + offs_i[:, None, None] * stride_right_i
        + offs_j[None, :, None] * stride_right_j
        + offs_h[None, None, :] * stride_right_h
    )
    right_val = tl.load(
        right_addr,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )
    rgate_addr = (
        right_gate_ptr
        + pid_b * stride_rgate_b
        + offs_i[:, None, None] * stride_rgate_i
        + offs_j[None, :, None] * stride_rgate_j
        + offs_h[None, None, :] * stride_rgate_h
    )
    rgate_val = tl.load(
        rgate_addr,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        other=0.0,
    )
    right_out = right_val * rgate_val * mask_val
    tl.store(
        right_addr,
        right_out,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
            - weights      : dict with all linear / layer‑norm parameters
            - config       : dict with keys ``dim`` and ``hidden_dim``;
                            optional ``nomask`` (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical safety)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – keep in float32 for the gates,
    #    later cast to fp16 for the heavy matmul.
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H] float32
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()  # later used after to_out_norm

    # ------------------------------------------------------------------
    # 4) Fuse mask + left/right gates with a Triton kernel
    # ------------------------------------------------------------------
    if nomask:
        # all‑ones mask – same dtype as the tensors
        mask_f = torch.ones_like(mask, dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)

    # Ensure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_f = mask_f.contiguous()

    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # Strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h = left_gate.stride()
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h = right_gate.stride()
    stride_m_b, stride_m_i, stride_m_j = mask_f.stride()

    # Tile sizes – chosen to keep 1024 threads per program
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _gate_mask_fuse_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_f,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_j, stride_l_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_h,
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
        stride_m_b, stride_m_i, stride_m_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_H=BLOCK_H,
    )
    # ``left`` and ``right`` now contain the gated & masked values.

    # ------------------------------------------------------------------
    # 5) Cast to fp16 and reshape for the batched matmul
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # (B, N, N, H) -> (B*H, N, N)
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # ------------------------------------------------------------------
    # 6) Core TriMul – batched matmul using cuBLAS (very fast on H100)
    #    out = left @ rightᵀ  (sum over the shared K dimension)
    # ------------------------------------------------------------------
    # Note: torch.bmm on fp16 tensors uses Tensor‑Core acceleration.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).float()  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()  # broadcast over the hidden dim
    out = F.linear(out, weights["to_out.weight"])  # back to [B, N, N, dim]

    return out
