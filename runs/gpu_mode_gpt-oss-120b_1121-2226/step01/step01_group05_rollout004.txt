```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)  # combined batch*hidden index
    pid_m = tl.program_id(axis=0)  # i tile
    pid_n = tl.program_id(axis=1)  # j tile

    # compute tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bound rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile (M x K)
        a_ptrs = a_ptr + pid_h * stride_ah + offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B transposed tile (K x N)  (original B is (N x K), we treat it as (K x N))
        b_ptrs = b_ptr + pid_h * stride_bh + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write back the result
    c_ptrs = c_ptr + pid_h * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Combined projection+gate for left
    # Concatenate weights
    left_weights = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)
    left_comb = F.linear(x_norm, left_weights)   # shape [B,N,N,2*hidden_dim]
    left_proj, left_gate_raw = torch.split(left_comb, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_raw)

    # Combined projection+gate for right
    right_weights = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)
    right_comb = F.linear(x_norm, right_weights)
    right_proj, right_gate_raw = torch.split(right_comb, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_raw)

    # Apply mask if needed
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)  # broadcast
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Gated left/right
    left = left_proj * left_gate   # float32
    right = right_proj * right_gate

    # Cast to fp16 for GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    right_perm = right.permute(0, 3, 1, 2).contiguous()

    right_flat = right_perm.view(BH, N, N)

    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    BLOCK_N = 128
    BLOCK_K = 64

"""
TriMul “outgoing” forward pass (AlphaFold‑3).

The heavy part of the computation is the cubic tensor contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix multiply (M=N=seq_len, K=seq_len) for every
hidden channel `h`.  The kernel below implements a high‑throughput tiled
GEMM using Triton (fp16 inputs, fp32 accumulation).  The surrounding
PyTorch code fuses linear+gate projections (two linear calls per side)
and applies the two LayerNorms, masks and the final projection.
All operations are performed on‐device; the result is a float32
tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A : [BH, M, K]   (fp16)
#   B : [BH, N, K]   (fp16)  → we load it as (K, N)
#   C : [BH, M, N]   (fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # combined (batch * hidden) index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for kk in range(num_k_tiles):
        cur_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # --------------------------------------------------------------
        # Load a tile of A (M x K)
        # --------------------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)            # fp16 → fp32

        # --------------------------------------------------------------
        # Load a tile of Bᵀ (K x N)
        # --------------------------------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)            # fp16 → fp32

        # --------------------------------------------------------------
        # Dot‑product accumulation
        # --------------------------------------------------------------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write result back to C
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )

# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                     # input/output channel dim
    hidden_dim = config["hidden_dim"]       # projection dimension (H)
    nomask = config.get("nomask", True)    # True → ignore mask

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over last dimension = dim)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]   float32

    # ------------------------------------------------------------------
    # 2. Fused linear + gate for the left side
    # ------------------------------------------------------------------
    # Stack weights: first hidden_dim rows = projection, next = gate
    left_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0)
    left_comb = F.linear(x_norm, left_w)                     # [B, N, N, 2*H]
    left_proj, left_gate_raw = torch.split(left_comb, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_raw)                # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Fused linear + gate for the right side
    # ------------------------------------------------------------------
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)
    right_comb = F.linear(x_norm, right_w)
    right_proj, right_gate_raw = torch.split(right_comb, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [B, N, N, 1]
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5. Apply gates (still in FP32)
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to fp16 for the cubic GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 7. Rearrange to [BH, N, N] (BH = B * H) for the Triton kernel
    # ------------------------------------------------------------------
    # B, H, N, N layout is more convenient for the batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)    # [BH, N, N]   fp16
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]   fp16

    # Output buffer (FP32 accumulator)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 8. Triton kernel launch (tuned tile sizes for H100)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden) dimension
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # leverage Tensor‑Cores on H100
        num_stages=3,         # triple‑buffering for latency hiding
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 10. Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # ------------------------------------------------------------------
    # 11. Output gate (computed from the same normalized input)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12. Final linear projection back to original `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
