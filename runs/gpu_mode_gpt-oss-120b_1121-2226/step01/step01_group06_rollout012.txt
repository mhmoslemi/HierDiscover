def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # keep fp32

    left = left * left_gate
    right = right * right_gate

    # cast to fp16
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # permute for batched matmul (view as B, H, N, N)
    left_perm = left_h.permute(0, 3, 1, 2)   # B,H,N,N (i,k)
    right_perm_t = right_h.permute(0, 3, 2, 1)  # B,H,N,N (k,j)

    B, H, N, _ = left_perm.shape

    # allocate output
    out_perm = torch.empty((B, H, N, N), dtype=torch.float16, device=device)

    # compute strides
    a_strides = left_perm.stride()
    b_strides = right_perm_t.stride()
    c_strides = out_perm.stride()

    # Choose block sizes
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
    BLOCK_K = 32

    # Grid dimensions
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total = B * H * num_m * num_n
    grid = (total,)

    _batched_matmul_fused_kernel[grid](
        left_perm,
        right_perm_t,
        out_perm,
        N,
        N,
        N,
        # strides A (batch, head, i, k)
        a_strides[0], a_strides[1], a_strides[2], a_strides[3],
        # strides B (batch, head, k, j)
        b_strides[0], b_strides[1], b_strides[2], b_strides[3],
        # strides C (batch, head, i, j)
        c_strides[0], c_strides[1], c_strides[2], c_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # permute back to original layout [B,N,N,H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # second LayerNorm
    out = F.layer_norm(
        out.float(),
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # apply out_gate
    out = out * out_gate

    # final linear
    out = F.linear(out, weights["to_out.weight"])

    return out

    a_ptr, b_ptr, c_ptr,
    M, N, K,
    B, H,
    stride_ab, stride_ah, stride_ai, stride_ak,
    stride_bb, stride_bh, stride_bk, stride_bj,
    stride_cb, stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute block coordinates
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch_head = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # decode batch and head indices
    b = batch_head // H
    h = batch_head % H

    # base pointers for this batch/head
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_bb + h * stride_bh
    c_base = c_ptr + b * stride_cb + h * stride_ch

    # offsets for this tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1.  Layer‑norm over the input last dimension (`dim`).
2.  Linear projections (no bias) → `left`, `right`  (shape [B,N,N,H]).
3.  Optional mask is broadcast‑multiplied.
4.  Gating networks (sigmoid of linear projections) are applied element‑wise.
5.  Heavy contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication `left @ rightᵀ` for each
   hidden slice `h`.  It is performed by a custom Triton kernel that
   fuses the batch and hidden dimensions (total B·H independent
   GEMMs) and works in FP16 with FP32 accumulation.
6.  Second LayerNorm over the hidden dimension `H`.
7.  Multiply by the output gate (`out_gate`).
8.  Final linear projection back to `dim`.

All heavy arithmetic (step 5) runs on the GPU via Triton; the rest
uses PyTorch primitives.  The implementation avoids any unnecessary
contiguity copies – after the proj/gate steps we merely permute the
tensors (views) and pass the resulting strides to the kernel.
"""

import math
from typing import Tuple, Dict

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ
#   A : (B, H, M, K)   – layout (batch, head, i, k)
#   B : (B, H, K, N)   – layout (batch, head, k, j)
#   C : (B, H, M, N)   – layout (batch, head, i, j)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_fused_kernel(
    a_ptr, b_ptr, c_ptr,                # tensor pointers
    M, N, K,                            # matrix dimensions (all equal to seq_len)
    BATCH, HEAD,                        # batch size and hidden dimension
    stride_ab, stride_ah, stride_ai, stride_ak,   # strides for A (b, h, i, k)
    stride_bb, stride_bh, stride_bk, stride_bj,   # strides for B (b, h, k, j)
    stride_cb, stride_ch, stride_ci, stride_cj,   # strides for C (b, h, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------
    #  Grid decomposition
    # --------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # each program processes one (batch, head) tile of size (BLOCK_M, BLOCK_N)
    batch_head = pid // (num_m * num_n)          # index in [0, BATCH*HEAD)
    block_idx = pid % (num_m * num_n)            # linear index of (block_m, block_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # decode batch and head indices
    b = batch_head // HEAD
    h = batch_head % HEAD

    # base pointers for this (b, h) slice
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_bb + h * stride_bh
    c_base = c_ptr + b * stride_cb + h * stride_ch

    # --------------------------------------------------
    #  Offsets for the current tile
    # --------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------
    #  Loop over the K dimension (tiles of size BLOCK_K)
    # --------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: (i, k) tile
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        # Bᵀ: (k, j) tile   – we already have B in (k, j) order
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # inner product accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------
    #  Write the result back (store as FP16)
    # --------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_fused(
    left: torch.Tensor,  # [B, H, N, N]  (i,k)
    right_t: torch.Tensor,  # [B, H, N, N]  (k,j)
) -> torch.Tensor:
    """
    Launch the Triton kernel for the batched matmul C = left @ right_t.
    Both input tensors are expected in FP16.  Returns a FP16 tensor of
    shape [B, H, N, N].
    """
    assert left.is_contiguous() is False  # they can be non‑contiguous views
    assert right_t.is_contiguous() is False
    B, H, N, _ = left.shape
    # output tensor (contiguous, will hold C = A @ Bᵀ)
    out = torch.empty((B, H, N, N), dtype=torch.float16, device=left.device)

    # --------------------------------------------------
    #  Strides (in elements, not bytes)
    # --------------------------------------------------
    a_str = left.stride()
    b_str = right_t.stride()
    c_str = out.stride()

    # --------------------------------------------------
    #  Choose tile sizes (tuned for H100, works for N up to 1024)
    # --------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    # --------------------------------------------------
    #  Grid definition
    # --------------------------------------------------
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * H * num_m * num_n
    grid = (total_tiles,)

    _batched_matmul_fused_kernel[grid](
        left,
        right_t,
        out,
        N,                    # M
        N,                    # N
        N,                    # K
        B, H,                 # batch & head dimensions
        a_str[0], a_str[1], a_str[2], a_str[3],   # A strides
        b_str[0], b_str[1], b_str[2], b_str[3],   # B strides
        c_str[0], c_str[1], c_str[2], c_str[3],   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    #  2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)      # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)    # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    #  3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    #  4️⃣ Gating tensors (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight)).to(left.dtype)   # FP32→FP16
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))  # stay in FP32 for later

    # Apply gates (still FP32 at the moment)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  5️⃣ Cast to FP16 and prepare for the batched matmul
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # View as (B, H, N, N) where the last two dims are (i,k) and (k,j)
    left_perm = left_h.permute(0, 3, 1, 2)          # [B, H, N, i=N, k=N]
    right_perm_t = right_h.permute(0, 3, 2, 1)      # [B, H, k=N, j=N]

    # ------------------------------------------------------------------
    #  6️⃣ Heavy contraction – fused batched GEMM (B·H independent GEMMs)
    # ------------------------------------------------------------------
    out_perm = _batched_matmul_fused(left_perm, right_perm_t)   # [B, H, N, N] FP16

    # Re‑order back to the original layout [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    #  7️⃣ Second Layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = out.to(torch.float32)   # convert for a stable LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    #  8️⃣ Apply the output gate (still FP32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    #  9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim] (FP32)

    return out
```
