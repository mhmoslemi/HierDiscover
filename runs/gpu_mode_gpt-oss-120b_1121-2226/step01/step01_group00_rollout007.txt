"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The implementation follows the AlphaFold‑3 reference:

1️⃣ Layer‑norm on the input (float32 for numeric stability).
2️⃣ Linear projections (left / right) → half‑precision.
3️⃣ Optional pairwise mask (broadcasted on the hidden dimension).
4️⃣ Gating: sigmoid(linear(x)) applied element‑wise to the left/right tensors.
5️⃣ Core “TriMul”:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched GEMM (C = A @ Bᵀ) where each hidden channel is a
   separate batch element.  The heavy matrix‑multiply is performed by a
   custom Triton kernel that tiles the matrices (default 128×128×64) and
   accumulates in FP32 for safety before storing the result in FP16.
6️⃣ Layer‑norm on the hidden dimension, element‑wise out‑gate, and a final
   linear projection back to the original feature dimension.

Only the matrix‑multiply (step 5) is done in Triton; all other
operations use PyTorch which keeps the code simple and the overall
runtime low.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) for the TriMul core.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                    # pointers (FP16)
    M, N, K,                    # matrix dims (all equal to seq_len)
    stride_am, stride_ak,       # A strides: row‑stride, col‑stride
    stride_bk, stride_bn,       # B strides: row‑stride (k), col‑stride (n)
    stride_cm, stride_cn,       # C strides: row‑stride, col‑stride
    stride_ab, stride_bb, stride_cb,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute C[b] = A[b] @ B[b].T for every batch element b. """
    pid_m = tl.program_id(0)   # tile‑row index
    pid_n = tl.program_id(1)   # tile‑col index
    pid_b = tl.program_id(2)   # batch / hidden index

    # ------------------------------------------------------------------
    # Offsets of the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the selected batch element
    a_batch = A + pid_b * stride_ab
    b_batch = B + pid_b * stride_bb
    c_batch = C + pid_b * stride_cb

    # FP32 accumulator (numerically safe)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K‑dimension in blocks
    # ------------------------------------------------------------------
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # Load a tile of B (K × N) – B is already transposed
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Tile‑wise matrix multiplication
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the tile back to C (FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] or None
        - weights_dict : all linear / layer‑norm parameters
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor : [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep in FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy part of the computation
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (left / right)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        mask = mask_tensor.to(x_fp16.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of a linear projection of the *original* x)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul – batched GEMM (C = A @ Bᵀ) via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B = batch size, N = seq_len
    # Rearrange tensors to (B, H, N, N) and make them contiguous
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) – already transposed

    # Collapse batch * hidden_dim into a single GEMM batch dimension
    left_flat = left_mat.view(B * hidden_dim, N, N)    # (B*H, M=N, K=N)
    right_flat = right_mat.view(B * hidden_dim, N, N)  # (B*H, K=N, N)

    # Allocate output buffer (FP16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides for the tiled kernel (row‑major layout)
    stride_am = left_flat.stride(1)   # = N
    stride_ak = left_flat.stride(2)   # = 1
    stride_bk = right_flat.stride(1)  # = N (row stride over K)
    stride_bn = right_flat.stride(2)  # = 1
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    stride_ab = left_flat.stride(0)   # batch stride = N*N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # ------------------------------------------------------------------
    # Tile sizes – chosen to hit the sweet‑spot on H100 tensor cores
    # ------------------------------------------------------------------
    BLOCK_M = 128   # rows per program
    BLOCK_N = 128   # cols per program
    BLOCK_K = 64    # reduction tile size

    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    _batched_gemm_kernel[(grid_x, grid_y, grid_z)](
        left_flat,                      # A pointer
        right_flat,                     # Bᵀ pointer (already transposed)
        out_flat,                       # C pointer
        N, N, N,                        # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # Reshape back to the original 4‑D layout and convert to FP32
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate   # broadcast over hidden dim

    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (float32)

    return out
