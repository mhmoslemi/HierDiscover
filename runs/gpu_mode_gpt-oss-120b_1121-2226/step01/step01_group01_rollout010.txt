"""
custom_kernel
==============

Implementation of the *outgoing* TriMul operation (AlphaFold‑3) with a
hand‑written Triton kernel that computes the O(N³) reduction
`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.

Key ideas
---------
* The heavy matrix‑multiply is performed entirely in FP16 inside Triton,
  accumulating in FP32 for numerical stability.
* No memory‑copy / permute is required: the kernel works directly on the
  original tensor layout (B, N, N, H) and treats (batch, hidden) as a
  fused batch dimension (`pid_hid`).
* All layer‑norms, linear projections and gating are kept in PyTorch;
  the projections stay in FP32 and are only cast to FP16 right before the
  reduction.
* Optional masking is applied in FP16 after the linear projections.
* Final output is cast back to FP32, layer‑normed again and projected to
  the original feature dimension.

The kernel is tuned for H100 Tensor‑Core FP16 MatMuls:
  BLOCK_M = 128, BLOCK_N = 128, BLOCK_K = 64, 8 warps per block.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the TriMul reduction.
# The input tensors have shape (B, N, N, H) in row‑major order.
# We treat (batch, hidden) as a flattened batch index `pid_hid`.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    a_ptr,                     # const uint16*  -> left   (fp16)
    b_ptr,                     # const uint16*  -> right  (fp16) (k‑j transposed)
    c_ptr,                     #       uint16*  -> out    (fp16)
    M, N, K,                   # int32 matrix dimensions (all == seq_len)
    stride_batch,              # int32 stride for the batch dimension (B stride)
    stride_i,                  # int32 stride for the i (row)   dimension (N*H)
    stride_j,                  # int32 stride for the j (col)   dimension (H)
    hidden_dim: tl.constexpr,  # number of hidden channels
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each (batch, hidden) slice:
        C[i, j] = Σ_k A[i, k] * B[k, j]
    where A corresponds to left[b, i, k, h] and B corresponds to
    right[b, j, k, h] (note the transpose on the j‑axis).
    All tensors are stored in FP16; the accumulator is FP32.
    """
    pid_m = tl.program_id(0)          # tile over rows   (i)
    pid_n = tl.program_id(1)          # tile over cols   (j)
    pid_hid = tl.program_id(2)        # flattened (batch, hidden)

    # ------------------------------------------------------------------
    # Decode batch index and hidden index from the flattened pid
    # ------------------------------------------------------------------
    batch_idx = pid_hid // hidden_dim
    hid_idx   = pid_hid % hidden_dim

    # ------------------------------------------------------------------
    # Compute the offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # ------------------------------------------------------------------
    # Base pointer for this (batch, hidden) slice.
    # The hidden dimension has stride 1 (last dimension).
    # ------------------------------------------------------------------
    a_base = a_ptr + batch_idx * stride_batch + hid_idx
    b_base = b_ptr + batch_idx * stride_batch + hid_idx
    c_base = c_ptr + batch_idx * stride_batch + hid_idx

    # ------------------------------------------------------------------
    # Pointers to the tiles of A (i,k) and B (k,j)
    # ------------------------------------------------------------------
    a_ptrs = a_base + (offs_m[:, None] * stride_i + offs_k[None, :] * stride_j)
    b_ptrs = b_base + (offs_k[:, None] * stride_j + offs_n[None, :] * stride_i)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the summed axis)
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # TL's dot performs Σ_k a[i,k] * b[k,j] over the K axis
        acc += tl.dot(a, b)

        # Step to the next K‑tile
        a_ptrs += BLOCK_K * stride_j
        b_ptrs += BLOCK_K * stride_j

    # ------------------------------------------------------------------
    # Store the result back to global memory (FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_m[:, None] * stride_i + offs_n[None, :] * stride_j)
    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Extract weight tensors (all float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]     # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2. Linear projections (FP32) – keep high accuracy
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)  FP32
    right = F.linear(x, right_proj_weight)  # (B, N, N, hidden_dim)  FP32

    # --------------------------------------------------------------
    # 3. Optional mask (broadcastable to hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left  = left  * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid) – keep in FP32, cast to FP16 before mul
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight)).half()
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)).half()
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # keep FP32 for final mul

    # Apply gates (still FP32 for left/right, then cast)
    left  = (left  * left_gate).to(torch.float16)   # now FP16
    right = (right * right_gate).to(torch.float16) # now FP16

    # --------------------------------------------------------------
    # 5. Heavy O(N³) reduction performed by Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # N == seq_len

    # Output container (FP16)
    out = torch.empty_like(left)

    # Strides of the original layout (B, N, N, H) – row‑major
    stride_batch = left.stride(0)   # B stride = N * N * H
    stride_i      = left.stride(1)   # i‑stride = N * H
    stride_j      = left.stride(2)   # j‑stride = H
    # (hidden stride = 1, not needed explicitly)

    # ------------------------------------------------------------------
    # Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64      # 64 gives good Tensor‑Core utilisation on H100

    grid = (
        triton.cdiv(N, BLOCK_M),          # tiles over rows (i)
        triton.cdiv(N, BLOCK_N),          # tiles over cols (j)
        B * hidden_dim                    # one slice per (batch, hidden)
    )

    trimul_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        M=N,
        N=N,
        K=N,
        stride_batch=stride_batch,
        stride_i=stride_i,
        stride_j=stride_j,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps per block works well for FP16 on H100
    )

    # ------------------------------------------------------------------
    # Cast back to FP32 for the remaining post‑processing
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out_fp32 = F.layer_norm(out_fp32,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)

    # Apply the final gate (out_gate is still FP32)
    out_fp32 = out_fp32 * out_gate

    # Final linear back to original dimension
    out_fp32 = F.linear(out_fp32, to_out_weight)   # (B, N, N, dim) FP32

    return out_fp32
