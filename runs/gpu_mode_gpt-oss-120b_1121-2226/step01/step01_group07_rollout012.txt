    A_ptr, B_ptr, C_ptr,
    stride_a_batch, stride_a_h, stride_a_i, stride_a_j,
    stride_b_batch, stride_b_h, stride_b_i, stride_b_j,  # ignore or rename
    stride_c_batch, stride_c_h, stride_c_i, stride_c_j,
    BATCH: tl.constexpr,
    H: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute batch and hidden index.
    batch_hidden = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch_idx = batch_hidden // H
    hidden_idx = batch_hidden % H

    # compute offsets for A and B base
    a_ptr = A_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_ptr = B_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_ptr = C_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_h

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for current tile
    a_tile_ptrs = a_ptr + offs_m[:, None] * stride_a_i + tl.arange(0, BLOCK_K)[None, :] * stride_a_j
    b_tile_ptrs = b_ptr + tl.arange(0, BLOCK_K)[:, None] * stride_b_i + offs_n[None, :] * stride_b_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        b = tl.load(b_tile_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_tile_ptrs += BLOCK_K * stride_a_j
        b_tile_ptrs += BLOCK_K * stride_b_i

    c = acc.to(tl.float16)
    c_tile_ptrs = c_ptr + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_tile_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
Optimized Triton implementation of the outgoing TriMul (TriMul‑outgoing) module.

The expensive N³ contraction
    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
is performed as a batched half‑precision GEMM where the hidden dimension is
treated as an additional batch axis (B·H independent GEMMs).  The kernel works
directly on the non‑contiguous (B, H, N, N) tensors produced by a simple
permute, thus avoiding costly copies.  All other operations (LayerNorm,
linear projections, gating, final projection) are executed with fast
PyTorch primitives.  The kernel uses fp16 inputs and fp32 accumulation,
producing fp16 results that are cast back to fp32 for the remaining
post‑processing.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched half‑precision GEMM kernel.
#   A : (B, H, M, K)  – left tensor after gating (fp16)
#   B : (B, H, K, N)  – right tensor after gating and transposition (fp16)
#   C : (B, H, M, N)  – output (fp16)
# Strides encode the original layout; the hidden dimension (H) is folded
# into the batch axis inside the kernel.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    # strides for A (B, H, M, K)
    stride_a_batch, stride_a_h, stride_a_m, stride_a_k,
    # strides for B (B, H, K, N)
    stride_b_batch, stride_b_h, stride_b_k, stride_b_n,
    # strides for C (B·H, M, N)
    stride_c_batch, stride_c_m, stride_c_n,
    # runtime parameters
    hidden_dim,
    M, N, K,
    # compile‑time tiling
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B for every (batch, hidden) pair.
    A shape = (B, H, M, K)  (fp16)
    B shape = (B, H, K, N)  (fp16)
    C shape = (B·H, M, N)   (fp16)
    """
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    batch_hidden = tl.program_id(0)          # 0‑th axis: combined (B, H)
    pid_m = tl.program_id(1)                 # tile row
    pid_n = tl.program_id(2)                 # tile col

    # ------------------------------------------------------------------
    # Recover batch index and hidden index from the combined ID
    # ------------------------------------------------------------------
    batch_idx = batch_hidden // hidden_dim   # original batch B
    hidden_idx = batch_hidden % hidden_dim   # hidden slice H

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) pair
    # ------------------------------------------------------------------
    a_ptr = A_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_ptr = B_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_ptr = C_ptr + batch_hidden * stride_c_batch   # C already folded B·H

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator in fp32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Pointers for the current K‑tile
        a_tile_ptr = a_ptr + offs_m[:, None] * stride_a_m \
                              + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        b_tile_ptr = b_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_b_k \
                              + offs_n[None, :] * stride_b_n

        # Masks for the edge of the K dimension
        mask_k = (k_offset + tl.arange(0, BLOCK_K)) < K

        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        # Block‑level dot product (fp16 inputs → fp32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result back (cast to fp16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_tile_ptr = c_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_tile_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask:         torch.Tensor of shape [B, N, N] or None
            - weights:      dict of the model's weight tensors
            - config:       dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1️⃣ Extract model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]                # (dim,)
    norm_bias = weights["norm.bias"]                    # (dim,)
    left_proj_weight = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]   # (hidden_dim, dim)
    left_gate_weight = weights["left_gate.weight"]     # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]   # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]       # (hidden_dim, dim)
    to_out_norm_weight = weights["to_out_norm.weight"] # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]     # (hidden_dim,)
    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm over the channel dimension D
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias=False)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Gating values (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 6️⃣ Apply per‑position gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Cast to half‑precision for the heavy N³ contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 8️⃣ Permute to (B, H, N, N) / (B, H, N, N) – **no .contiguous()!**
    # ------------------------------------------------------------------
    # left[b, i, k, h] -> left_perm[b, h, i, k]
    left_perm = left.permute(0, 3, 1, 2)          # shape (B, H, N, N), non‑contiguous
    # right[b, j, k, h] -> right_perm[b, h, k, j] (rightᵀ)
    right_perm = right.permute(0, 3, 2, 1)        # shape (B, H, N, N), non‑contiguous

    B, N, _, H = left.shape                     # B, N, N, H (original shape)
    batch_dim = B * H

    # ------------------------------------------------------------------
    # 9️⃣ Allocate output buffer for the batched GEMM (fp16)
    # ------------------------------------------------------------------
    out_reshaped = torch.empty((batch_dim, N, N),
                               dtype=torch.float16,
                               device=input_tensor.device)

    # ------------------------------------------------------------------
    # 10️⃣ Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # larger K‑tile reduces loop overhead on H100

    grid = (
        batch_dim,                              # combined (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,           # tiles along M
        (N + BLOCK_N - 1) // BLOCK_N,           # tiles along N
    )

    # ------------------------------------------------------------------
    # 11️⃣ Strides for the three tensors (as returned by .stride())
    # ------------------------------------------------------------------
    # A : (B, H, M, K)  = left_perm
    a_batch_stride = left_perm.stride(0)
    a_h_stride = left_perm.stride(1)
    a_m_stride = left_perm.stride(2)
    a_k_stride = left_perm.stride(3)

    # B : (B, H, K, N) = right_perm
    b_batch_stride = right_perm.stride(0)
    b_h_stride = right_perm.stride(1)
    b_k_stride = right_perm.stride(2)
    b_n_stride = right_perm.stride(3)

    # C : (B·H, M, N) = out_reshaped (contiguous)
    c_batch_stride = out_reshaped.stride(0)
    c_m_stride = out_reshaped.stride(1)
    c_n_stride = out_reshaped.stride(2)

    # ------------------------------------------------------------------
    # 12️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    batched_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_reshaped,
        a_batch_stride,
        a_h_stride,
        a_m_stride,
        a_k_stride,
        b_batch_stride,
        b_h_stride,
        b_k_stride,
        b_n_stride,
        c_batch_stride,
        c_m_stride,
        c_n_stride,
        hidden_dim,          # hidden_dim as a runtime scalar
        N, N, N,            # M, N, K (all equal to seq_len)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 13️⃣ Reshape back to [B, N, N, H] (float32) for the remaining ops
    # ------------------------------------------------------------------
    out = (
        out_reshaped
        .view(B, hidden_dim, N, N)        # (B, H, N, N)
        .permute(0, 2, 3, 1)              # (B, N, N, H)
        .contiguous()
        .to(torch.float32)
    )   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 14️⃣ Post‑contraction LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 15️⃣ Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 16️⃣ Final linear projection back to the original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (dim, hidden_dim) → [B, N, N, D]

    return out
```
