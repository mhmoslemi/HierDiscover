```
@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_ab, stride_bb, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    GROUP_M: tl.constexpr):
    pid = tl.program_id(0)
    pid_z = tl.program_id(2)   # batch dimension (hidden)
    # Compute grid mapping similar to Triton matrix multiplication
    # grouping across M tiles
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    # linear id based on pid
    pid_mn = pid
    # Grouping logic
    pid_m = (pid_mn // num_pid_n) % num_pid_m
    pid_n = pid_mn % num_pid_n
    # Offsets for tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N
    # Batch offset
    a_batch = A + pid_z * stride_ab
    b_batch = B + pid_z * stride_bb
    c_batch = C + pid_z * stride_cb

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=(mask_m[:, None] & mask_k[None, :]), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(mask_k[:, None] & mask_n[None, :]), other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store results
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(mask_m[:, None] & mask_n[None, :]))
```

```
    stride_bk, stride_bn,
```

    stride_ab, stride_bb, stride_cb,

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm on the input tensor (last dimension = dim) – performed in FP32.
2. Cast to FP16 and apply two linear projections (left_proj, right_proj) → shape
   [B, N, N, H] where H = hidden_dim.
3. Optional pairwise mask (broadcasted over the hidden dimension) is applied.
4. Gating: sigmoid(linear(x)) for left, right and out gates.  The left/right
   tensors are multiplied element‑wise by their gates.
5. Core “TriMul” operation:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batch of H independent N×N matrix‑multiplications
   (one GEMM per hidden channel).  Implemented with a custom Triton kernel
   that processes the batch dimension (B·H) in parallel.  The kernel performs
   FP16 matrix multiplication with FP32 accumulation (tensor‑core friendly).
6. Layer‑norm on the hidden dimension, element‑wise out‑gate, and a final
   linear projection back to dim.  The result is returned in FP32.

The heavy N³ computation is fully fused inside the Triton kernel; all other
operations are kept in PyTorch (they are memory‑bound and inexpensive).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (B is already transposed w.r.t. the
# original right tensor).  Each program computes a BLOCK_M×BLOCK_N tile for a
# single batch element (batch = B * hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                     # pointers (FP16)
    M, N, K,                     # matrix dims (all = seq_len)
    stride_am, stride_ak,        # A: row‑stride, col‑stride
    stride_bk, stride_bn,        # B: row‑stride (k), col‑stride (n)
    stride_cm, stride_cn,        # C: row‑stride, col‑stride
    stride_ab, stride_bb, stride_cb,   # batch‑strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    """Batched GEMM where each batch element corresponds to one hidden channel."""
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch (B * hidden_dim)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers offset to the selected batch element
    a_batch = A + pid_b * stride_ab
    b_batch = B + pid_b * stride_bb
    c_batch = C + pid_b * stride_cb

    # Accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks
    # --------------------------------------------------------------
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # Load B tile (K × N) – B is already the transposed right matrix
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Matrix multiply for the tile
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write tile back to C (store in FP16)
    # --------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]  (float32)
            - mask_tensor : [B, N, N] or None
            - weights_dict : model parameters
            - config_dict  : {"dim": ..., "hidden_dim": ..., "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the bulk of the computation
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (left / right) – no bias
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,H]
    right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_fp16.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid of linear(x))
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core TriMul (batched GEMM) – Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                       # B = batch size, N = seq_len

    # Reshape to (B, H, N, N) and align the dimensions for GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # Right needs to be transposed on the last two axes: (B, H, K, J)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) where dim2 = K, dim3 = J

    # Collapse batch and hidden dim → (B*H, M, K) and (B*H, K, N)
    left_flat = left_mat.view(B * hidden_dim, N, N)
    right_flat = right_mat.view(B * hidden_dim, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)   # will hold C

    # Strides (row‑major layout)
    stride_am = left_flat.stride(1)   # row‑stride (M)
    stride_ak = left_flat.stride(2)   # col‑stride (K)
    stride_bk = right_flat.stride(1)  # K‑stride (row of B)
    stride_bn = right_flat.stride(2)  # N‑stride (col of B)
    stride_cm = out_flat.stride(1)    # row‑stride of C
    stride_cn = out_flat.stride(2)    # col‑stride of C

    stride_ab = left_flat.stride(0)   # batch stride = N*N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # Tile sizes – chosen to map well to tensor‑core tiles on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions (tiles over M, N, and the collapsed batch)
    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    # Launch the kernel.  8 warps per program give good occupancy on H100.
    _batched_gemm_kernel[(grid_x, grid_y, grid_z), num_warps=8](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,               # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and cast to FP32 for the remaining ops
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                      # broadcast over hidden dim
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
