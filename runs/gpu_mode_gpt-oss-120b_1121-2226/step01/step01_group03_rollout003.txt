"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input over the last dimension (dim) – torch.
2️⃣ Fuse the four “proj / gate” linear layers (left_proj, right_proj,
   left_gate, right_gate) into a single matmul:
       out = x_norm @ W_fused.T    (W_fused ∈ ℝ^(4·hidden_dim × dim))
   and split the result into the four tensors.
   This reduces the number of passes over the huge 4‑D tensor from 5 to 2.
3️⃣ Apply sigmoid to the two gate tensors and (optional) broadcast mask.
4️⃣ Fuse the three‑way element‑wise product (proj * gate * mask) with a tiny
   Triton kernel that works on FP16 data (half the memory bandwidth).
5️⃣ Batched GEMM across the sequence dimension:
       left_mat  = left_scaled.permute(0,3,1,2).reshape(B·H,N,N)   (FP16)
       right_mat = right_scaled.permute(0,3,1,2).reshape(B·H,N,N)
       out_mat   = left_mat @ right_matᵀ   (torch.bmm, half‑precision)
   The GEMM is the O(N³·H) work and runs on Tensor‑Core HW.
6️⃣ Reshape back to [B,N,N,hidden_dim] (FP16 → FP32).
7️⃣ Layer‑norm over the hidden dimension, multiply by the *out_gate* (computed
   separately) and project back to the original dimension.
All heavy element‑wise work (step 4) is performed by Triton, satisfying the
“must use Triton” requirement.  The rest uses highly‑optimised torch kernels.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise multiplication (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _fused_mul_kernel(
    a_ptr,                     # left/right projection       (fp16)
    b_ptr,                     # gate (sigmoid)             (fp16)
    mask_ptr,                  # optional mask broadcast    (fp16)
    out_ptr,                   # output                     (fp16)
    n_elements,                # total number of elements   (int)
    hidden_dim,                # hidden dimension H         (int)
    BLOCK_SIZE: tl.constexpr, # compile‑time tile size
    HAS_MASK: tl.constexpr,    # 0 → no mask, 1 → mask present
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < n_elements

    # load projection and gate (FP16)
    a = tl.load(a_ptr + offs, mask=valid)
    b = tl.load(b_ptr + offs, mask=valid)

    if HAS_MASK:
        # mask is smaller by a factor `hidden_dim`; compute its index
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=valid)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=valid)


def _fused_mul(proj: torch.Tensor,
               gate: torch.Tensor,
               mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Triton‑based element‑wise multiplication:
        out = proj * gate * mask   (mask broadcast over hidden_dim)
    All tensors must be contiguous FP16.
    """
    assert proj.shape == gate.shape
    B, N, _, H = proj.shape
    total = proj.numel()                # B·N·N·H
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    proj_flat = proj.view(-1)
    gate_flat = gate.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)       # shape [B·N·N] (FP16)
        has_mask = 1
    else:
        mask_flat = torch.empty(0,
                                 dtype=proj.dtype,
                                 device=proj.device)   # dummy
        has_mask = 0

    out = torch.empty_like(proj)
    _fused_mul_kernel[grid](
        proj_flat,
        gate_flat,
        mask_flat,
        out.view(-1),
        total,
        H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out

# ----------------------------------------------------------------------

        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Fuse the four linear projections (proj + gate)
    # ------------------------------------------------------------------
    # Cast input & weights to FP16 once
    x_norm_fp16 = x_norm.to(torch.float16)

    # concatenate weights: [left_proj; right_proj; left_gate; right_gate]  (4·H×dim)
    fused_weight_fp16 = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # One big matmul: (B·N·N, dim) @ (dim, 4·H)ᵀ  -> (B·N·N, 4·H)
    fused_out = torch.nn.functional.linear(x_norm_fp16, fused_weight_fp16)
    # split into the four tensors of shape [B,N,N,hidden_dim] (still FP16)
    left_proj, right_proj, left_gate_raw, right_gate_raw = torch.split(
        fused_out, hidden_dim, dim=-1
    )

    # ------------------------------------------------------------------
    # 3. Gate non‑linearities (sigmoid) – stays in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden_dim) – also FP16
    # ------------------------------------------------------------------
    if use_mask:
        mask_fp16 = mask.to(torch.float16)
    else:
        mask_fp16 = None

    # ------------------------------------------------------------------
    # 5. Fuse element‑wise multiplication with Triton
    # ------------------------------------------------------------------
    left_scaled = _fused_mul(left_proj, left_gate, mask_fp16)   # [B,N,N,H] FP16
    right_scaled = _fused_mul(right_proj, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 6. Batched GEMM over the sequence dimension (half‑precision)
    # ------------------------------------------------------------------
    B, N, _, H = left_scaled.shape                      # H == hidden_dim
    # reshape to (B·H, N, N) without extra copies
    left_mat = left_scaled.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_scaled.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_mat = left_mat @ right_matᵀ   (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B,N,N,hidden_dim] (still FP16)
    out_fp16 = out_mat.reshape(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden_dim (float32)
    # ------------------------------------------------------------------
    out_fp32 = out_fp16.to(torch.float32)   # Cast only once
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Compute out_gate (sigmoid) – we can do this in FP16 then cast
    # ------------------------------------------------------------------
    out_gate_raw = torch.nn.functional.linear(
        x_norm_fp16, weights["out_gate.weight"].to(torch.float16)
    )
    out_gate = torch.sigmoid(out_gate_raw).to(torch.float32)

    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out
```
