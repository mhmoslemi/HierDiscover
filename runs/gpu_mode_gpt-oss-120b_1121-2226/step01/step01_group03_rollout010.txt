@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          stride_am, stride_ak, stride_ab,
                          stride_bm, stride_bk, stride_bb,
                          stride_cm, stride_cn, stride_cb,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    # Program ID
    pid = tl.program_id(0)
    # Determine batch id and block indices
    batch = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_idx = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_i = block_idx // tl.cdiv(N, BLOCK_N)
    block_j = block_idx % tl.cdiv(N, BLOCK_N)

    # Compute offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for A and B
    a_ptrs = A_ptr + batch*stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + batch*stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator acc
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k*BLOCK_K + tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :], other=0.0)
        # Convert to fp16? a,b are half input.
        # The kernel compiled with dtype of pointer: we can treat as tl.float16.

        acc += tl.dot(a, b)

        # Increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = C_ptr + batch*stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

import torch
import triton
import triton.language as tl

# fused elementwise multiplication kernel
@triton.jit
def _fused_mul_kernel(
    a_ptr, b_ptr, mask_ptr, out_ptr,
    n_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask).to(tl.float16)
    b = tl.load(b_ptr + offs, mask=mask).to(tl.float16)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask).to(tl.float16)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=mask)


def _fused_mul(a: torch.Tensor,
               b: torch.Tensor,
               mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Fuse elementwise multiplication (a * b * mask) via Triton.
    a, b : [B, N, N, H] (fp16)
    mask : [B, N, N] (fp16) broadcast over H
    Returns fp16 tensor of same shape as a.
    """
    assert a.shape == b.shape, "a and b must share shape"
    total = a.numel()
    BLOCK = 1024  # can be tuned
    grid = ((total + BLOCK - 1) // BLOCK,)

    a_flat = a.view(-1)
    b_flat = b.view(-1)
    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=a.dtype, device=a.device)
        has_mask = 0

    out = torch.empty_like(a)
    _fused_mul_kernel[grid](
        a_flat, b_flat, mask_flat, out.view(-1),
        total, a.shape[-1],
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out


# batched matrix multiplication kernel (fp16 inputs, fp32 accum)
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)

    # compute tile coordinates
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_i = block_idx // num_blocks_n
    block_j = block_idx % num_blocks_n

    # coordinates within the tile
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to the start of the batch
    a_batch_ptr = A_ptr + batch * stride_ab
    b_batch_ptr = B_ptr + batch * stride_bb
    c_batch_ptr = C_ptr + batch * stride_cb

    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(a_batch_ptr + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_batch_ptr + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)  # a,b are fp16, acc fp32

    # write back result in fp16
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication:
    A: [batch, M, K] (fp16)
    B: [batch, K, N] (fp16)
    Returns C: [batch, M, N] (fp16)
    """
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Choose block sizes (tuned for 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Strides
    stride_am, stride_ak, stride_ab = A.stride(1), A.stride(2), A.stride(0)
    stride_bk, stride_bn, stride_bb = B.stride(1), B.stride(2), B.stride(0)
    stride_cm, stride_cn, stride_cb = C.stride(1), C.stride(2), C.stride(0)

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m * num_blocks_n,)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Layer‑norm the input (float32).
2️⃣  Cast to fp16 and apply two linear projections (left / right).
3️⃣  Compute three sigmoid gates (left, right, out) in float32
    and cast the first two to fp16.
4️⃣  Fuse element‑wise multiplication   left = left * left_gate * mask
    (the mask is optional) with a tiny Triton kernel.
5️⃣  Rearrange tensors so that the heavy N³ work becomes a batched
    matrix‑multiply:
        left_mat  : (B·H, N, N)   fp16
        right_mat : (B·H, N, N)   fp16   (already transposed on‑the‑fly)
    Compute   out_mat = left_mat @ right_matᵀ   using a custom
    Triton kernel (fp16 → fp32 accumulation → fp16 store).
6️⃣  Reshape back to [B, N, N, H] and convert to float32.
7️⃣  Layer‑norm over the hidden dimension, multiply by the out‑gate.
8️⃣  Final linear projection back to `dim`.

All heavy work runs in fp16 (Tensor‑cores) while layer‑norms and
sigmoids stay in fp32 for numerical stability.  The Triton kernels
eliminate extra temporaries and guarantee that *some* work is performed
with Triton as required.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Element‑wise fused multiplication (a * b * mask)
# ----------------------------------------------------------------------
@triton.jit
def _fused_mul_kernel(
    a_ptr, b_ptr, mask_ptr, out_ptr,            # pointers
    n_elements, hidden_dim,                     # runtime args
    BLOCK_SIZE: tl.constexpr,                   # compile‑time tile size
    HAS_MASK: tl.constexpr,                     # 0/1 compile‑time flag
):
    """
    out = a * b * mask   (mask is broadcast over the hidden dimension)
    a, b    : [B·N·N·H]  (flattened fp16)
    mask    : [B·N·N]    (flattened fp16) – optional
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    in_range = offs < n_elements

    a = tl.load(a_ptr + offs, mask=in_range).to(tl.float16)
    b = tl.load(b_ptr + offs, mask=in_range).to(tl.float16)

    if HAS_MASK:
        # convert element offset to mask offset (divide by H)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range).to(tl.float16)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=in_range)


def _fused_mul(a: torch.Tensor,
               b: torch.Tensor,
               mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Fuse ``a * b * mask`` with a Triton kernel.
    a, b : [B, N, N, H] (fp16, contiguous)
    mask : [B, N, N] (fp16) or ``None``.
    Returns fp16 tensor of the same shape as ``a``.
    """
    assert a.shape == b.shape, "a and b must have identical shape"
    total = a.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    a_flat = a.view(-1)
    b_flat = b.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=a.dtype, device=a.device)
        has_mask = 0

    out = torch.empty_like(a)
    _fused_mul_kernel[grid](
        a_flat, b_flat, mask_flat, out.view(-1),
        total, a.shape[-1],
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣  Batched matrix multiplication (fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    C[batch, M, N] = A[batch, M, K] @ B[batch, K, N]
    A, B, C are fp16; accumulation is fp32.
    """
    pid = tl.program_id(0)

    # --- tile / batch decomposition ------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    block_i = block_idx // num_blocks_n
    block_j = block_idx % num_blocks_n

    # --- coordinates inside the tile ------------------------------------
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # --- pointer to the beginning of the current batch -----------------
    a_batch = A_ptr + batch * stride_ab
    b_batch = B_ptr + batch * stride_bb
    c_batch = C_ptr + batch * stride_cb

    # --- accumulator ----------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --- loop over K ---------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a = tl.load(a_batch + offs_m[:, None] * stride_am
                              + offs_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_batch + offs_k[:, None] * stride_bk
                              + offs_n[None, :] * stride_bn,
                    mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)          # a,b fp16 → dot result fp32

    # --- write result (fp16) -------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A : [batch, M, K] (fp16)
    B : [batch, K, N] (fp16)
    Returns C : [batch, M, N] (fp16)
    """
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # tile sizes (tuned for H100, N up to 1024)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides (in elements)
    stride_am, stride_ak, stride_ab = A.stride(1), A.stride(2), A.stride(0)
    stride_bk, stride_bn, stride_bb = B.stride(1), B.stride(2), B.stride(0)
    stride_cm, stride_cn, stride_cb = C.stride(1), C.stride(2), C.stride(0)

    # launch grid
    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m * num_blocks_n,)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bn, stride_bb,
        stride_cm, stride_cn, stride_cb,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C
