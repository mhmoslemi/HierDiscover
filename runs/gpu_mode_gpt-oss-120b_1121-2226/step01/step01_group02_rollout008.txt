TriMul (outgoing) forward pass – Triton‑accelerated cubic update.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to float16 and apply the three linear projections (left/right) and the
   three gating linear layers (all in float16, without bias).  The mask (if any)
   and the element‑wise sigmoid gates are fused in half‑precision.
3. The heavy N³ operation
        out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
   is a batched matrix multiplication L @ Rᵀ where each hidden channel is an
   independent batch element.  We reshape (B,N,N,H) → (B·H,N,N) and run a
   custom Triton kernel that computes a batch of (M×K)·(N×K)ᵀ products with
   fp16 inputs and fp32 accumulation (Tensor‑Core friendly).
4. Reshape the result back to (B,N,N,H) and apply a second LayerNorm.
5. Multiply by the output gate and finish with a linear projection back to the
   original feature dimension (float32).

The Triton kernel is a classic tiled matmul with configurable block sizes.
Larger tiles (128×128) are used for big sequences to improve arithmetic
intensity and Tensor‑Core utilization on H100.
"""

import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile index in M
    pid_n = tl.program_id(1)          # tile index in N
    pid_b = tl.program_id(2)          # batch (i.e. hidden‑channel) index

    # ----------------------------------------------------------
    # Offsets for the current M/N tile
    # ----------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the M‑tile and N‑tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator in fp32 (TensorCore mixed‑precision accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over K tiles
    # ----------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Tensor‑core dot, fp16·fp16 → fp32 accumulate
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----------------------------------------------------------
    # Write out the computed tile
    # ----------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matmul for the TriMul update.
    A, B : (batch, M, K) and (batch, N, K) – fp16 tensors.
    Returns C : (batch, M, N) – fp32 tensor.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to half‑precision for all heavy ops
    # ------------------------------------------------------------------
    x_h = x.half()                                   # [B,N,N,dim] (fp16)

    # ------------------------------------------------------------------
    # 2) Linear projections in fp16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left = F.linear(x_h, left_proj_w)                # [B,N,N,hidden_dim] (fp16)
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Masking (optional) – everything stays in fp16
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(left.dtype).unsqueeze(-1)        # [B,N,N,1] (fp16)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – also in fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = F.linear(x_h, left_gate_w).sigmoid()        # [B,N,N,hidden_dim] (fp16)
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()         # (fp16)

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates (still fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core cubic update – batched matmul via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # Reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # Choose tile sizes adaptively (larger tiles for larger N)
    if N <= 384:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B*H, N, N]  (fp32)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    # ------------------------------------------------------------------
    # 8) Output gating and final projection back to dim
    # ------------------------------------------------------------------
    # out_gate was computed in fp16 – cast to fp32 for the final multiplication
    out_gate_f32 = out_gate.float()
    out = out * out_gate_f32

    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim] (fp32)

    return out
