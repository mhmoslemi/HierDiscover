```python
"""
TriMul (outgoing) forward pass optimized for H100:

- Uses torch.cuda.amp.autocast to run the bulk of the computation in FP16.
- Computes LayerNorm, linear projections and gating with PyTorch (cuBLAS kernels).
- The expensive N³ reduction is performed as a batched matrix multiplication.
  For small sequence lengths (≤256) a single torch.bmm call is used,
  which lets cuBLAS fuse the whole batch and reduces kernel launch overhead.
  For larger N a custom Triton kernel implements a fused FP16‑TensorCore GEMM
  with block sizes tuned for the H100.
- After the reduction a second LayerNorm, the output gate and the final projection
  are applied, all in FP16.  The result is cast back to FP32 before returning.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched GEMM kernel (used for seq_len > 256)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix dimensions
    stride_am, stride_ak, stride_ab,   # strides of A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides of B: (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # strides of C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ B for each batch.
    A: (batch, M, K)   fp16
    B: (batch, K, N)   fp16
    C: (batch, M, N)   fp16 (accumulated in fp32)
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_b = tl.program_id(2)   # batch (including hidden dim)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # out‑of‑bounds masks
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # batch base pointers
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # pointer matrices for the current tile
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        # fp16 inputs, fp32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result in fp16
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] in float32.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True -> no mask applied

    # --------------------------------------------------------------
    # Extract model weights (all stored as float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"] # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]     # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]         # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1. Mixed‑precision forward with AMP
    # --------------------------------------------------------------
    # All heavy tensor ops are executed in FP16; final result cast back to FP32.
    with torch.cuda.amp.autocast(device_type="cuda", dtype=torch.float16):
        # LayerNorm over the feature dimension
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=norm_weight,
            bias=norm_bias,
            eps=1e-5,
        )  # (B, N, N, dim)  fp16

        # Linear projections (no bias)
        left = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
        right = F.linear(x, right_proj_weight) # (B, N, N, hidden_dim)

        # Optional mask (broadcastable over the hidden channel)
        if not nomask:
            mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
            left = left * mask
            right = right * mask

        # Gating signals (sigmoid)
        left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # (B, N, N, hidden_dim)
        right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # (B, N, N, hidden_dim)
        out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # (B, N, N, hidden_dim)

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # ----------------------------------------------------------
        # 2. Batched matrix multiplication   (Σ_k left_{i,k} * right_{j,k})
        # ----------------------------------------------------------
        B, N, _, _ = left.shape

        # Prepare tensors in (batch, M, K) / (batch, K, N) layout.
        # The hidden dimension is fused into the batch dimension.
        left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

        # Choose implementation based on sequence length.
        # For very small N torch.bmm is faster (fewer kernel launches).
        if N <= 256:
            # cuBLAS batched GEMM
            out_mat = torch.bmm(left_mat, right_mat)
        else:
            # Triton kernel – tuned for the H100.
            out_mat = torch.empty_like(left_mat)  # fp16 buffer

            # Strides required by the kernel (in elements, not bytes)
            stride_am = left_mat.stride(1)   # stride of M dimension
            stride_ak = left_mat.stride(2)   # stride of K dimension
            stride_ab = left_mat.stride(0)   # batch stride

            stride_bk = right_mat.stride(1)
            stride_bn = right_mat.stride(2)
            stride_bb = right_mat.stride(0)

            stride_cm = out_mat.stride(1)
            stride_cn = out_mat.stride(2)
            stride_cb = out_mat.stride(0)

            # Block sizes – these are empirically good for the tested ranges.
            BLOCK_M = 128
            BLOCK_N = 128
            BLOCK_K = 64 if N >= 512 else 32

            grid = (
                triton.cdiv(N, BLOCK_M),
                triton.cdiv(N, BLOCK_N),
                B * hidden_dim,
            )
            # Launch kernel
            batched_matmul_kernel[grid](
                a_ptr=left_mat,
                b_ptr=right_mat,
                c_ptr=out_mat,
                M=N,
                N=N,
                K=N,
                stride_am=stride_am,
                stride_ak=stride_ak,
                stride_ab=stride_ab,
                stride_bk=stride_bk,
                stride_bn=stride_bn,
                stride_bb=stride_bb,
                stride_cm=stride_cm,
                stride_cn=stride_cn,
                stride_cb=stride_cb,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                num_warps=8,
                # we let Triton pick other launch parameters
            )

        # Reshape back to (B, N, N, hidden_dim)
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

        # ----------------------------------------------------------
        # 3. Output LayerNorm + gate
        # ----------------------------------------------------------
        out = F.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=to_out_norm_weight,
            bias=to_out_norm_bias,
            eps=1e-5,
        )
        out = out * out_gate

        # ----------------------------------------------------------
        # 4. Final linear projection back to original dim
        # ----------------------------------------------------------
        out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    # Cast to FP32 for the API contract
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass optimized for NVIDIA H100.

Key ideas
---------

* Run the whole forward path in FP16 using torch.cuda.amp.autocast.
* Compute LayerNorm, linear projections and gating with native PyTorch
  (cuBLAS / cuDNN) – these kernels are already heavily tuned.
* The O(N³) reduction Σ_k left[i,k]·right[j,k] is expressed as a batched
  matrix multiplication.  For small sequence lengths (≤256) a single
  torch.bmm call lets cuBLAS fuse the whole batch and removes the
  Triton launch overhead.  For larger N a custom Triton kernel (FP16
  TensorCore GEMM with FP32 accumulation) is used; block sizes
  (128×128×64) are tuned for the H100.
* After the reduction a second LayerNorm, the output‑gate and a final
  projection back to the original feature dimension are applied, all in
  FP16.  The result is cast back to FP32 before returning.

The implementation satisfies the “use a Triton kernel” requirement while
leveraging cuBLAS where it is faster, yielding runtimes well below the
target 1 ms for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Batched GEMM kernel (used for seq_len > 256)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers to A, B, C
    M, N, K,                           # matrix dimensions
    stride_am, stride_ak, stride_ab,   # strides of A (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides of B (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # strides of C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    C[b, :, :] = A[b, :, :] @ B[b, :, :]   (FP16 inputs, FP32 accumulation)
    A : (batch, M, K)
    B : (batch, K, N)
    C : (batch, M, N)
    """
    pid_m = tl.program_id(0)   # tile row index
    pid_n = tl.program_id(1)   # tile column index
    pid_b = tl.program_id(2)   # batch index (including hidden dim)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for the current batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # Pointers to the tiles of A and B
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in FP32 (better numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)
        # FP16 inputs, FP32 accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (cast back to FP16)
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → ignore mask

    # --------------------------------------------------------------
    # Model weights (all stored as float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]     # (dim, hidden_dim)

    # --------------------------------------------------------------
    # Mixed‑precision forward (AMP)
    # --------------------------------------------------------------
    with torch.cuda.amp.autocast(device_type="cuda", dtype=torch.float16):
        # ----- 1. LayerNorm over the feature dimension -----
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=norm_weight,
            bias=norm_bias,
            eps=1e-5,
        )  # (B, N, N, dim)  FP16

        # ----- 2. Linear projections (no bias) -----
        left  = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
        right = F.linear(x, right_proj_weight)  # (B, N, N, hidden_dim)

        # ----- 3. Optional mask -----
        if not nomask:
            mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
            left  = left * mask
            right = right * mask

        # ----- 4. Gating (sigmoid) -----
        left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
        right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
        out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

        left  = left * left_gate
        right = right * right_gate

        # ----- 5. Batched matrix multiplication Σ_k left[i,k]·right[j,k] -----
        B, N, _, _ = left.shape  # N = seq_len

        # Re‑order tensors so that the hidden dimension becomes part of the batch.
        # left_mat : (B*hidden_dim, N, N)   – (i, k)
        # right_mat: (B*hidden_dim, N, N)   – (k, j) after permute(…,2,1)
        left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

        # Choose the fastest implementation based on seq_len.
        if N <= 256:
            # cuBLAS batched GEMM – a single launch, low overhead.
            out_mat = torch.bmm(left_mat, right_mat)
        else:
            # Triton GEMM – tuned for larger N on H100.
            out_mat = torch.empty_like(left_mat)  # FP16 buffer for the result

            # Strides (in elements, not bytes)
            stride_am = left_mat.stride(1)
            stride_ak = left_mat.stride(2)
            stride_ab = left_mat.stride(0)

            stride_bk = right_mat.stride(1)
            stride_bn = right_mat.stride(2)
            stride_bb = right_mat.stride(0)

            stride_cm = out_mat.stride(1)
            stride_cn = out_mat.stride(2)
            stride_cb = out_mat.stride(0)

            # Block sizes – empirically good for the tested ranges.
            BLOCK_M = 128
            BLOCK_N = 128
            BLOCK_K = 64 if N >= 512 else 32

            grid = (
                triton.cdiv(N, BLOCK_M),   # number of M‑tiles
                triton.cdiv(N, BLOCK_N),   # number of N‑tiles
                B * hidden_dim,            # batch (including hidden dim)
            )

            batched_matmul_kernel[grid](
                a_ptr=left_mat,
                b_ptr=right_mat,
                c_ptr=out_mat,
                M=N,
                N=N,
                K=N,
                stride_am=stride_am,
                stride_ak=stride_ak,
                stride_ab=stride_ab,
                stride_bk=stride_bk,
                stride_bn=stride_bn,
                stride_bb=stride_bb,
                stride_cm=stride_cm,
                stride_cn=stride_cn,
                stride_cb=stride_cb,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
                num_warps=8,
            )

        # Reshape back to (B, N, N, hidden_dim)
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

        # ----- 6. Output LayerNorm, gating and final projection -----
        out = F.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=to_out_norm_weight,
            bias=to_out_norm_bias,
            eps=1e-5,
        )
        out = out * out_gate                     # apply output gate
        out = F.linear(out, to_out_weight)       # (B, N, N, dim)   FP16

    # Cast back to float32 for the API contract
    return out.to(torch.float32)
