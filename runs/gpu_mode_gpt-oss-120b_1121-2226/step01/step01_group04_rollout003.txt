def mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    size,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    # load
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

"""
TriMul (outgoing) – fast half‑precision implementation.

Algorithm
---------
1. **Half‑precision LayerNorm** on the input tensor (dim‑last).
2. **Linear projections** (no bias) → `left` & `right` (B×N×N×H) in fp16.
3. **Gate linear layers** + sigmoid → `left_gate`, `right_gate`, `out_gate`
   (all fp16).
4. **Fusion kernel** (Triton) that in‑place multiplies:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   The mask is a 3‑D tensor (B,N,N) broadcasted to the hidden dimension
   inside the kernel, thus no extra memory copy.
5. **Batched matrix‑multiply** per hidden dimension:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   Implemented as a single `torch.matmul` on tensors of shape
   (B, H, N, N) → (B, H, N, N), which lets cuBLAS/H100 Tensor‑Cores run
   the whole 3‑D batch without reshaping copies.
6. **Second LayerNorm** (over the hidden dimension), multiply by `out_gate`,
   and a final linear projection back to `dim`.
7. Cast the result to float32 (required output type).

All heavy arithmetic is performed in fp16, only the final output is
converted back to fp32.  The Triton kernel is the only non‑PyTorch piece
and fuses the mask with the two gate multiplications, eliminating three
separate element‑wise passes and saving memory bandwidth.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and two gate multiplications.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 * (B*N*N*H)
    right_ptr,              # fp16 * (B*N*N*H)
    left_gate_ptr,          # fp16 * (B*N*N*H)
    right_gate_ptr,         # fp16 * (B*N*N*H)
    mask_ptr,               # fp16 * (B*N*N)  (broadcasted over H)
    size,                   # total number of elements in left/right (int)
    hidden_dim,             # H (int, not constexpr)
    BLOCK_SIZE: tl.constexpr,
):
    """
    In‑place:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    The mask is expanded on‑the‑fly: each element of left/right belongs
    to a (b,i,j) triple; the corresponding mask entry is at index
    floor(idx / H).
    """
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < size

    # Load fp16 values (out‑of‑bounds loads produce zero)
    l = tl.load(left_ptr + offs,   mask=valid, other=0.0)
    r = tl.load(right_ptr + offs,  mask=valid, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # mask index = (b,i,j)  = offs // H  (integer division)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0)

    # Fuse the three elementwise ops
    l = l * lg * m
    r = r * rg * m

    # Store back the fused tensors
    tl.store(left_ptr + offs,  l, mask=valid)
    tl.store(right_ptr + offs, r, mask=valid)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors (see keys below)
        - config       : dict with ``dim`` and ``hidden_dim`` (ints)

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # e.g. 128

    # ------------------------------------------------------------------
    # Cast everything to half‑precision (fp16) early to cut memory traffic
    # ------------------------------------------------------------------
    # LayerNorm parameters
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    # Linear projection weights (no bias)
    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)   # (H, dim)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    # Gate weights
    left_gate_weight  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(torch.float16)

    # Output norm + final projection
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight      = weights["to_out.weight"].to(torch.float16)     # (dim, H)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (half‑precision)
    # ------------------------------------------------------------------
    # Convert input to fp16 once
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → left / right  (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H] fp16
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – also fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 4. Fuse mask + gate element‑wise multiplies with Triton
    # ------------------------------------------------------------------
    # Prepare mask: if absent use all‑ones
    if mask is None:
        mask_tensor = torch.ones((bs, N, N), dtype=torch.float16,
                                 device=left.device)
    else:
        mask_tensor = mask.to(torch.float16)

    # Flatten tensors for the 1‑D kernel
    total_elems = left.numel()                       # B * N * N * H
    BLOCK = 1024                                      # decent trade‑off
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor.view(-1),   # (B*N*N) flat
        size=total_elems,
        hidden_dim=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Core TriMul: batched matmul per hidden dimension
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) – a (B*H)-batch of NxN matrices.
    # The .contiguous() call forces a layout that cuBLAS can read efficiently.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Batched matmul: (B*H,N,N) @ (B*H,N,N)^T  → (B*H,N,N)
    out_perm = torch.matmul(left_perm,
                            right_perm.transpose(-1, -2))   # fp16

    # Restore original layout: (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()      # fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # fp16

    out = out * out_gate                                 # gating, fp16

    # Final linear back to `dim`
    out = F.linear(out, to_out_weight)                   # [B,N,N,dim] fp16

    # Return in float32 as required by the API
    return out.to(torch.float32)
