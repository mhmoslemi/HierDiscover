def _mask_gate_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, size, H, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    # load
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # compute mask index: element index in left dims divided by H
    # Equivalent to floor(offs / H)
    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0) # broadcast (mask may be 0 or 1)
    # fuse
    l = l * lg * m
    r = r * rg * m
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
                          mask_ptr, size, H: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < size

    left = tl.load(left_ptr + offs, mask=active, other=0.0)
    right = tl.load(right_ptr + offs, mask=active, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    mask_idx = offs // H
    m = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)

    left = left * left_gate * m
    right = right * right_gate * m

    tl.store(left_ptr + offs, left, mask=active)
    tl.store(right_ptr + offs, right, mask=active)

def custom_kernel(data):
    """
    Outgoing TriMul forward pass (no gradient).

    Implements the AlphaFold3 TriMul operation:
        1) Layer‑norm over the channel dimension.
        2) Linear projections to a hidden dimension.
        3) Sigmoid gates.
        4) One Triton kernel fuses mask, left‑gate and right‑gate.
        5) Batched matrix‑multiply over the sequence dimension.
        6) Second Layer‑norm, output gate and final projection.
    All heavy arithmetic is performed in FP16; the final output is FP32.
    """
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2. Cast to half and project
    # ------------------------------------------------------------------
    x_h = x.half()  # FP16 view

    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left = F.linear(x_h, left_proj_w)          # [B,N,N,hidden_dim] FP16
    right = F.linear(x_h, right_proj_w)

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # later used after LN

    # ------------------------------------------------------------------
    # 3. Fuse mask and gates with a Triton kernel
    # ------------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones((bs, N, N), dtype=torch.float16, device=input_tensor.device)
    else:
        mask_tensor = mask.to(torch.float16)
    mask_tensor = mask_tensor.contiguous()

    # Ensure contiguous memory for all tensors used in the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total_elems = left.numel()                     # B * N * N * hidden_dim
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_tensor,
        size=total_elems,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # ------------------------------------------------------------------
    # 4. Core bilinear product: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) for batched gemm
    left_mat = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    # Batched matmul: left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim] and convert to float32 for LN
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).float()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # FP32

    out_gate = out_gate.float()
    out = out * out_gate

    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                # FP32 final projection

    return out

# -*- coding: utf-8 -*-
"""
Optimized outgoing TriMul forward pass (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input over the channel dimension (float32).
2. Linear projections + sigmoid gates (still float32).
3. Cast projections and gates to FP16.
4. One Triton kernel fuses:
        left = left * left_gate * mask
        right = right * right_gate * mask
   The mask is *not* expanded to the hidden dimension – the kernel
   computes the correct mask index on‑the‑fly, which saves a huge
   memory copy.
5. Core bilinear product:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is realised as a batched matrix‑multiply of shape
   (B·H, N, N) @ (B·H, N, N)ᵀ using FP16 tensor‑core kernels
   (torch.bmm).  The result is cast back to FP32.
6. Second Layer‑norm over the hidden dimension, multiplication by the
   output gate and a final linear projection back to the original
   channel dimension (float32).

Only step 4 is implemented in Triton; all other heavy arithmetic
leverages highly‑optimised cuBLAS / torch kernels.  The implementation
conforms to the required `custom_kernel` signature and works for all
test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate into left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # FP16 tensor, flattened
    right_ptr,              # FP16 tensor, flattened
    left_gate_ptr,          # FP16 tensor, flattened
    right_gate_ptr,         # FP16 tensor, flattened
    mask_ptr,               # FP16 mask, shape (B, N, N) flattened
    size,                   # total number of elements in left/right
    H: tl.constexpr,        # hidden dimension (size of the last axis)
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise fused operation:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    The mask is broadcast over the hidden dimension inside the kernel,
    avoiding an explicit expansion copy.
    """
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < size

    # Load the FP16 tensors
    left       = tl.load(left_ptr + offs,       mask=active, other=0.0)
    right      = tl.load(right_ptr + offs,      mask=active, other=0.0)
    left_gate  = tl.load(left_gate_ptr + offs,  mask=active, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=active, other=0.0)

    # Compute 1‑D index for the mask (mask does NOT have the hidden dim)
    mask_idx = offs // H                     # integer division
    m = tl.load(mask_ptr + mask_idx, mask=active, other=0.0)

    # Fuse the three element‑wise multiplications
    left = left * left_gate * m
    right = right * right_gate * m

    # Store the results back in‑place
    tl.store(left_ptr + offs,  left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or ``None``
        - weights      : dict of model parameters (see code for keys)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]  # e.g. 128

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]  dtype=float32

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (still in float32)
    # ------------------------------------------------------------------
    # Linear projections (no bias)
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Cast to FP16 and fuse mask + gates with a Triton kernel
    # ------------------------------------------------------------------
    # Cast the large tensors to half‑precision (still correct because the
    # following matmul uses tensor‑core accumulation in FP32).
    left      = left.to(torch.float16).contiguous()
    right     = right.to(torch.float16).contiguous()
    left_gate = left_gate.to(torch.float16).contiguous()
    right_gate= right_gate.to(torch.float16).contiguous()
    out_gate  = out_gate.to(torch.float16)               # keep for later

    # Prepare the mask – keep it FP16 and *do not* expand it.
    if mask is None:
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16,
                               device=input_tensor.device)
    else:
        mask_fp16 = mask.to(torch.float16)
    mask_fp16 = mask_fp16.contiguous()

    # ------------------------------------------------------------------
    # 4. Triton kernel launch (mask & gate fusion)
    # ------------------------------------------------------------------
    total_elements = left.numel()                     # B * N * N * H
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp16,
        size=total_elements,
        H=hidden_dim,          # compile‑time constant per launch
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5. Core TriMul bilinear product  (batched FP16 matmul)
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) for a single batched GEMM.
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Batched matrix‑multiply:   left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore the original 4‑D shape and cast back to float32 for the
    # remaining (LayerNorm, gating, projection) steps.
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).float()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )   # [B, N, N, hidden_dim]  float32

    # Apply the output gate (cast back to float32 first)
    out_gate = out_gate.float()
    out = out * out_gate

    # Final projection back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])     # float32

    return out
