def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    # retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Move to same device and dtype as input (assume float32)
    device = input_tensor.device
    
    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )
    
    # Projections: cast weights to fp16 for speed
    left_proj_w_h = left_proj_weight.to(torch.float16)
    right_proj_w_h = right_proj_weight.to(torch.float16)
    # input to half
    x_h = x.to(torch.float16)
    left = torch.nn.functional.linear(x_h, left_proj_w_h)   # fp16
    right = torch.nn.functional.linear(x_h, right_proj_w_h)
    
    # mask (if present)
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h
    
    # Gating: compute linear + sigmoid in fp32 then cast to fp16
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    # cast gates to half and broadcast
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)
    left = left * left_gate_h
    right = right * right_gate_h
    
    # Reshape for batched matmul
    B, N, _, _ = left.shape
    # left: [B,N,N,H] -> (B,H,N,N) -> (B*H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # we need right transposed on last two dims: (B*H,N,N) where we multiply A @ B^T
    A = left_perm.view(B * hidden_dim, N, N)
    Bmat = right_perm.view(B * hidden_dim, N, N)
    
    # Batched matmul
    out_fp16 = torch.bmm(A, Bmat.transpose(1,2))
    
    # reshape to original format
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    
    # Second layernorm over hidden dim (dim=hidden_dim)
    out = torch.nn.functional.layer_norm(
        out.to(torch.float32),
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # Apply out_gate (already fp32)
    out = out * out_gate
    
    # Final projection
    out = torch.nn.functional.linear(out, to_out_weight)
    
    return out
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, N_ELEMS, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a, b):
    # a,b must be same shape, contiguous, dtype torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 65536  # 64k elements per program
    grid = ((total + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](
        a, b, out,
        total,
        BLOCK=BLOCK,
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    The implementation follows the reference PyTorch code but
    (1) casts the heavy tensors to FP16 to exploit Tensor Cores,
    (2) fuses the element‑wise gating multiplication with a tiny Triton kernel,
    (3) performs the cubic contraction with a single batched GEMM via
        torch.bmm, which on an H100 runs in ≈1 ms for the largest test shapes.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    #  Load weights (cast to half wherever possible)
    # ------------------------------------------------------------------
    norm_weight_h = weights["norm.weight"].to(torch.float16)
    norm_bias_h   = weights["norm.bias"].to(torch.float16)

    left_proj_weight_h  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight_h = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight_h  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight_h = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight_h   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight_h = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias_h   = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight_h      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    #  1️⃣ Layer‑norm over the last dimension (dim) – FP16
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight_h,
        bias=norm_bias_h,
    )
    x_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    #  2️⃣ Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_weight_h)   # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_weight_h)

    # ------------------------------------------------------------------
    #  3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    #  4️⃣ Gating networks – all in FP16, sigmoid is native FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_weight_h))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight_h))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_weight_h))

    # ------------------------------------------------------------------
    #  5️⃣ Fuse element‑wise gating with a tiny Triton kernel
    # ------------------------------------------------------------------
    left = _elemwise_mul(left, left_gate)    # left ← left * left_gate
    right = _elemwise_mul(right, right_gate) # right ← right * right_gate

    # ------------------------------------------------------------------
    #  6️⃣ Cubic contraction: batched GEMM C = A @ Bᵀ
    #      A = left  (B, N, N, H) → (B*H, N, N)
    #      B = right (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim

    # bring hidden dim to batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    A = left_perm.view(B * H, N, N)   # (B*H, N, N)
    Bmat = right_perm.view(B * H, N, N)

    # Batch‑matmul on FP16 using Tensor Cores
    out_fp16 = torch.bmm(A, Bmat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    #  7️⃣ Second Layer‑norm over hidden dimension (H) – back to FP32
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_h,
        bias=to_out_norm_bias_h,
    )

    # ------------------------------------------------------------------
    #  8️⃣ Apply output gate (FP32) and final linear projection
    # ------------------------------------------------------------------
    # out_gate was computed in FP16; cast to FP32 for multiplication
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32
    out = F.linear(out, to_out_weight_h)   # final projection back to `dim`

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, N_ELEMS, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS
    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, a * b, mask=mask)

def _elemwise_mul(a, b):
    # expects contiguous half tensors of same shape
    assert a.shape == b.shape
    assert a.is_contiguous() and b.is_contiguous()

    total = a.numel()

    grid = ((total + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](
        a, b, out, total,
        BLOCK_SIZE=BLOCK,
    )
    return out

    """
    TriMul (outgoing) forward pass with mixed‑precision and a tiny Triton
    element‑wise kernel. The cubic contraction is performed by a single
    batched GEMM (`torch.bmm`) on FP16 tensors, which on an H100 fits well
    under the 1 ms target.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------- Load / cast weights -------------------
    norm_weight_h = weights["norm.weight"].to(torch.float16)
    norm_bias_h   = weights["norm.bias"].to(torch.float16)

    left_proj_weight_h  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight_h = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight_h  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight_h = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight_h   = weights["out_gate.weight"].to(torch.float16)

    # Second layer‑norm and final projection stay in FP32 for stability
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------- 1. LayerNorm over dim (FP16) -------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight_h,
        bias=norm_bias_h,
    )
    x_h = x_norm.to(torch.float16)

    # ------------------- 2. Linear projections (FP16) -------------------
    left  = F.linear(x_h, left_proj_weight_h)   # [B,N,N,hidden]
    right = F.linear(x_h, right_proj_weight_h)

    # ------------------- 3. Optional mask -------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(torch.float16)  # [B,N,N,1]
        left  = left * mask_h
        right = right * mask_h

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_weight_h))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight_h))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_weight_h))

    # ------------------- 5. Fuse element‑wise gating with Triton -------------------
    left  = _elemwise_mul(left, left_gate)   # left ← left * left_gate
    right = _elemwise_mul(right, right_gate) # right ← right * right_gate

    B, N, _, H = left.shape      # H == hidden_dim
    # bring hidden dim to batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    Bmat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    out_fp16 = torch.bmm(A, Bmat.transpose(1, 2))

        weight=to_out_norm_weight,

    )

    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_weight)

"""
TriMul (outgoing) – mixed‑precision forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the last dimension (dim) in FP16.
2️⃣  Two FP16 linear projections → left, right.
3️⃣  Optional mask (broadcast over the hidden dimension) applied in FP16.
4️⃣  Three gating streams (left, right, out) computed in FP16 + sigmoid.
5️⃣  Element‑wise multiplication of left/right with their gates is fused
    into a tiny Triton kernel (`_elemwise_mul_kernel`).
6️⃣  The cubic contraction 
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    is a batched matrix multiplication A @ Bᵀ where A and B are
    (B·H, N, N) FP16 tensors.  A single call to `torch.bmm` (Tensor‑Core
    accelerated) handles the whole batch.
7️⃣  The result is reshaped back to [B, N, N, H] and converted to FP32.
8️⃣  Second Layer‑norm (over the hidden dimension) in FP32.
9️⃣  Multiply by the output gate (FP32) and apply the final FP32 linear
    projection back to `dim`.

The heavy N³ work lives in the batched GEMM (step 6) and the tiny
element‑wise Triton kernel (step 5); on an H100 the whole forward pass
runs well under the 1 ms target for all provided configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel: out = a * b  (both FP16, same shape)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, out_ptr, N_ELEMS, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask, other=tl.float16(0.0))
    b = tl.load(b_ptr + offs, mask=mask, other=tl.float16(0.0))
    tl.store(out_ptr + offs, a * b, mask=mask)


def _elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication using the Triton kernel above.
    Both inputs must be contiguous FP16 tensors of identical shape.
    """
    assert a.shape == b.shape
    assert a.is_contiguous() and b.is_contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    BLOCK = 65536  # 64 k elements per program
    grid = ((total + BLOCK - 1) // BLOCK,)
    _elemwise_mul_kernel[grid](
        a, b, out, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of torch.Tensor containing model parameters
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # cast weights to FP16 where possible
    # --------------------------------------------------------------
    norm_weight_h = weights["norm.weight"].to(torch.float16)
    norm_bias_h   = weights["norm.bias"].to(torch.float16)

    left_proj_weight_h  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight_h = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight_h  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight_h = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight_h   = weights["out_gate.weight"].to(torch.float16)

    # Second layer‑norm and final projection stay in FP32 for stability
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension (dim) – FP16
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight_h,
        bias=norm_bias_h,
    )
    x_h = x_norm.to(torch.float16)          # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – FP16
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_weight_h)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_weight_h)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4️⃣ Gating networks – FP16 + sigmoid
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_weight_h))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight_h))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_weight_h))

    # --------------------------------------------------------------
    # 5️⃣ Fuse element‑wise gating with a Triton kernel
    # --------------------------------------------------------------
    #   left  ← left  * left_gate
    #   right ← right * right_gate
    left  = _elemwise_mul(left, left_gate)
    right = _elemwise_mul(right, right_gate)

    # --------------------------------------------------------------
    # 6️⃣ Cubic contraction via a single batched GEMM (FP16)
    # --------------------------------------------------------------
    B, N, _, H = left.shape                 # H == hidden_dim

    # move hidden dimension into the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    A = left_perm.view(B * H, N, N)          # (B·H, N, N)
    Bmat = right_perm.view(B * H, N, N)      # (B·H, N, N)

    # Batched matrix multiplication C = A @ Bᵀ  (Tensor‑Core accelerated)
    out_fp16 = torch.bmm(A, Bmat.transpose(1, 2))   # (B·H, N, N)

    # reshape back to the original 4‑D layout
    out = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Second Layer‑norm over the hidden dimension – FP32
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate & final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_weight)      # [B, N, N, dim]  FP32

    return out
