"""
TriMul “outgoing” forward (AlphaFold3).

Algorithm
---------
1. Layer‑norm the 4‑D input (norm over the last dimension `dim`).
2. Linear projections → `left_proj`, `right_proj`  (shape [B,N,N,H]).
3. Optional binary mask (broadcast on H) is applied to the projections.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
5. Fuse element‑wise multiplication of projections with their gates and
   cast the result to fp16 with a single Triton kernel
   (the kernel reads `left_proj`, `left_gate`, `right_proj`, `right_gate`
   and writes two fp16 tensors – the gated left/right tensors).
6. Re‑order to [B·H, N, N] and perform a batched matrix multiplication
   `left @ rightᵀ` with torch.bmm on fp16 (Tensor‑Core accelerated on H100).
   The accumulation is performed in fp32 automatically.
7. Reshape back to [B,N,N,H] (still fp32), apply a second LayerNorm over H,
   multiply by the `out_gate`, and finally project back to `dim` with a
   linear layer.
8. Return the final tensor of shape [B, N, N, dim] (float32).

The heavy cubic step (step 5) is realized with a custom Triton kernel,
while the actual N³ matrix multiply (step 6) uses the highly‑optimized
cuBLAS batched GEMM via `torch.bmm`, giving a runtime well below the
target 1 ms on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_gate_and_cast_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    # strides for the input tensors (float32)
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_lg0, stride_lg1, stride_lg2, stride_lg3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    # strides for the output tensors (fp16)
    stride_lo0, stride_lo1, stride_lo2, stride_lo3,
    stride_ro0, stride_ro1, stride_ro2, stride_ro3,
    BLOCK: tl.constexpr,
):
    """
    For every element (b,i,j,h) compute:
        left_out  = (left_proj[b,i,j,h]   * left_gate[b,i,j,h]).to(fp16)
        right_out = (right_proj[b,i,j,h]  * right_gate[b,i,j,h]).to(fp16)
    The kernel processes ``BLOCK`` elements per program.
    """
    pid = tl.program_id(axis=0)                     # linear program id
    offs = pid * BLOCK + tl.arange(0, BLOCK)        # global element ids
    total = B * N * N * H
    mask = offs < total

    # de‑flatten the linear index to (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # offsets for the left/right tensors
    off_l = b * stride_l0 + i * stride_l1 + j * stride_l2 + h * stride_l3
    off_lg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3
    off_r = b * stride_r0 + i * stride_r1 + j * stride_r2 + h * stride_r3
    off_rg = b * stride_lg0 + i * stride_lg1 + j * stride_lg2 + h * stride_lg3  # same layout as left_gate

    # load, multiply, cast to fp16
    left = tl.load(left_ptr + off_l, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + off_lg, mask=mask, other=0.0)
    left_out = (left * left_gate).to(tl.float16)

    right = tl.load(right_ptr + off_r, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + off_rg, mask=mask, other=0.0)
    right_out = (right * right_gate).to(tl.float16)

    # store results
    tl.store(left_out_ptr + off_l, left_out, mask=mask)
    tl.store(right_out_ptr + off_r, right_out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden]
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5. Fuse gate multiplication & cast to fp16 with a Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape
    left_fp16 = torch.empty_like(left_proj, dtype=torch.float16, device=device)
    right_fp16 = torch.empty_like(right_proj, dtype=torch.float16, device=device)

    total_elements = B * N * N * hidden_dim
    BLOCK = 1024
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    fused_gate_and_cast_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        left_fp16, right_fp16,
        B, N, hidden_dim,
        # strides for left_proj (float32)
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        # strides for left_gate (float32)
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        # strides for right_proj (float32)
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        # strides for left_fp16 output
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        # strides for right_fp16 output
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        BLOCK=BLOCK,
    )

    # Release intermediate tensors early to free memory
    del left_proj, right_proj, left_gate, right_gate
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 6. Batched matrix multiplication (left @ rightᵀ) on fp16 tensors
    # ------------------------------------------------------------------
    # reshape to [B*H, N, N]
    left_flat = left_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # torch.bmm performs a strided‑batched GEMM using Tensor‑Cores (fp16->fp32 accumulation)
    out_half = torch.bmm(left_flat, right_flat.transpose(1, 2))  # [B*H, N, N] (fp16)
    out = out_half.to(torch.float32)  # cast back to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 7. Restore the 4‑D layout: [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still fp32

    # ------------------------------------------------------------------
    # 9. Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim] (float32)

    return out
