def fused_mask_gate_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,

def custom_kernel(data):
    import torch; import torch.nn.functional as F
    import triton; import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # Extract weights (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Convert needed weights to half
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1. LayerNorm (float32) then cast to fp16
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. One fused linear for left/right projections and gates + out_gate
    #   combined weight shape (5*hidden_dim, dim)
    combined_weight = torch.cat([
        left_proj_weight,
        left_gate_weight,
        right_proj_weight,
        right_gate_weight,
        out_gate_weight,
    ], dim=0)   # (5*H, dim)

    combined = F.linear(x, combined_weight)   # shape (B, N, N, 5*H)

    # split
    B, N, _, _ = combined.shape
    slice0 = combined[..., :hidden_dim]                                      # left_proj
    slice1 = torch.sigmoid(combined[..., hidden_dim:2*hidden_dim])          # left_gate
    slice2 = combined[..., 2*hidden_dim:3*hidden_dim]                      # right_proj
    slice3 = torch.sigmoid(combined[..., 3*hidden_dim:4*hidden_dim])       # right_gate
    out_gate = torch.sigmoid(combined[..., 4*hidden_dim:])                 # out_gate, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Mask handling & fuse elementwise multiplication using Triton
    if mask is None:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=input_tensor.device)
    else:
        mask_flat = mask.to(torch.float16).reshape(-1)   # shape B*N*N

    # flatten tensors for kernel
    left_proj_flat = slice0.reshape(-1)
    right_proj_flat = slice2.reshape(-1)
    left_gate_flat = slice1.reshape(-1)
    right_gate_flat = slice3.reshape(-1)

    left_out_flat = torch.empty_like(left_proj_flat)
    right_out_flat = torch.empty_like(right_proj_flat)

    total_elems = left_proj_flat.shape[0]  # B*N*N*H

    BLOCK_SIZE = 1024

    @triton.jit
    def fuse_kernel(
        left_ptr, right_ptr,
        left_gate_ptr, right_gate_ptr,
        mask_ptr,
        out_left_ptr, out_right_ptr,
        total: tl.constexpr,
        hidden: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK + tl.arange(0, BLOCK)
        mask = offs < total

        lp = tl.load(left_ptr + offs, mask=mask, other=0.0)
        rp = tl.load(right_ptr + offs, mask=mask, other=0.0)
        lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
        rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

        # mask index each hidden group shares same mask
        mask_idx = offs // hidden
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

        lo = lp * lg * m
        ro = rp * rg * m

        tl.store(out_left_ptr + offs, lo, mask=mask)
        tl.store(out_right_ptr + offs, ro, mask=mask)

    grid = (triton.cdiv(total_elems, BLOCK_SIZE), )
    fuse_kernel[grid](
        left_proj_flat,
        right_proj_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        left_out_flat,
        right_out_flat,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # reshape back
    left = left_out_flat.view(B, N, N, hidden_dim)
    right = right_out_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. N^3 contraction via batched FP16 GEMM (torch.bmm)
    # reshape for batch matmul: (B*H, N, N) @ (B*H, N, N)^T
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right need to be transposed on last two dims before matmul:
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat)   # fp16

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5. post‑norm over hidden_dim (float32)
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,

    out = out * out_gate   # out_gate was half? Convert to float32

    # ------------------------------------------------------------------
    # 6. Final projection back to original dim
    out = F.linear(out, to_out_weight)  # weight fp16, input fp32 -> result fp32

    return out

"""
Optimized outgoing TriMul forward pass.

Algorithm
---------
1. Layer‑norm the input (FP32) → cast to FP16.
2. Fuse *all* linear projections (left/right projections, their gates and the
   output gate) into a single matmul: weight shape (5·H, D).  Split the result
   into left_proj, left_gate, right_proj, right_gate, out_gate and apply the
   sigmoid to the gate tensors.
3. A tiny Triton kernel fuses mask broadcasting, element‑wise masking and the
   per‑position gates:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
   The kernel works on flattened tensors and uses integer division to obtain
   the mask index (each mask element is shared by `hidden_dim` entries).
4. The costly N³ contraction is performed as a batched half‑precision GEMM
   (left @ rightᵀ) using `torch.bmm`, which maps to cuBLAS Tensor‑Core kernels.
5. Post‑contraction LayerNorm (FP32), multiplication by the output gate and a
   final half‑precision linear projection back to the original dimension.
6. The final result is returned in FP32.

The Triton kernel provides the required custom‑kernel component while the
heavy linear algebra is delegated to highly tuned cuBLAS kernels, achieving
sub‑millisecond runtimes on the tested configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,          # projected values     (fp16)
    left_gate_ptr, right_gate_ptr,# gate values (fp16)
    mask_ptr,                     # mask (fp16) of size B·N·N
    out_left_ptr, out_right_ptr,  # fused results (fp16)
    total: tl.constexpr,          # total #elements = B·N·N·H
    hidden: tl.constexpr,         # hidden_dim = H
    BLOCK: tl.constexpr,          # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # load the four tensors
    l_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask index – every `hidden` consecutive elements share one mask entry
    mask_idx = offs // hidden
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # fuse mask & gates
    lo = l_val * l_gate * m_val
    ro = r_val * r_gate * m_val

    tl.store(out_left_ptr + offs, lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict with model parameters
            - config       : dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # weight tensors (convert to half where beneficial)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]            # (dim,)
    norm_bias   = weights["norm.bias"]              # (dim,)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    to_out_norm_w = weights["to_out_norm.weight"]    # (H,)
    to_out_norm_b = weights["to_out_norm.bias"]      # (H,)

    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (D, H)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                   # [B, N, N, D] float32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. One fused linear for all projections & gates (5·H output features)
    # ------------------------------------------------------------------
    # stack weights: [left_proj, left_gate, right_proj, right_gate, out_gate]
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                   # (5*H, D)

    fused = F.linear(x, fused_weight)   # [B, N, N, 5*H] (FP16)

    B, N, _, _ = fused.shape
    # split the fused tensor
    offset = 0
    left_proj = fused[..., offset : offset + hidden_dim];            offset += hidden_dim
    left_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]); offset += hidden_dim
    right_proj = fused[..., offset : offset + hidden_dim];           offset += hidden_dim
    right_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]); offset += hidden_dim
    out_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]);   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 3. Fuse mask, left/right gates via a tiny Triton kernel
    # ------------------------------------------------------------------
    if mask is None:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=input_tensor.device)
    else:
        mask_flat = mask.to(torch.float16).reshape(-1)  # shape B·N·N

    # flatten tensors for the kernel (contiguous view)
    left_proj_f = left_proj.reshape(-1)
    right_proj_f = right_proj.reshape(-1)
    left_gate_f = left_gate.reshape(-1)
    right_gate_f = right_gate.reshape(-1)

    left_out_f = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elems = left_proj_f.shape[0]                # B·N·N·H
    BLOCK_SIZE = 1024

    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    _mask_gate_fuse_kernel[grid](
        left_proj_f,
        right_proj_f,
        left_gate_f,
        right_gate_f,
        mask_flat,
        left_out_f,
        right_out_f,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # reshape back to `[B, N, N, H]`
    left = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. N³ contraction via batched half‑precision GEMM (torch.bmm)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) matrices
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right needs a transpose on the last two axes before matmul
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # batched matmul (FP16, uses Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat)           # (B*H, N, N), FP16

    # reshape back to `[B, N, N, H]`
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                                   # FP16

    # ------------------------------------------------------------------
    # 5. Post‑contraction LayerNorm (FP32) + output gate
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)               # broadcast over H

    # ------------------------------------------------------------------
    # 6. Final linear projection back to original dimension D
    #    (compute in FP16 then cast to FP32)
    # ------------------------------------------------------------------
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)                       # (D, H) weight, FP16 output
    out = out.to(torch.float32)

    return out
