def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. layer norm
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"])

    # 2. linear projections (no bias)
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. mask
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 4. gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # 5. Apply gates (and cast to half)
    left = (left_proj * left_gate).to(torch.float16)
    right = (right_proj * right_gate).to(torch.float16)

    B, N, _, _ = left.shape

    # 6. Permute and reshape to [BH, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Output buffer
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # 7. Triton kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 16

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            BH)

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
        # optionally num_stages=2
    )

    # 8. Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 9. second layernorm
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"])

    # 10. output gate
    out = out * out_gate.to(out.dtype)

    # 11. final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul – “outgoing” variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm over the last dimension (dim).
2️⃣  Two bias‑free linear projections → tensors L,R of shape
    [B, N, N, H] (H = hidden_dim).
3️⃣  Optional mask (broadcast on H) and three sigmoid gates.
    L ← L * sigmoid(left_gate),   R ← R * sigmoid(right_gate).
4️⃣  Cast L,R to FP16.
5️⃣  Fuse the cubic contraction
        out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
    (the “outgoing” einsum) as a batched GEMM:
        C_h = L_h @ R_hᵀ   with   L_h,R_h ∈ ℝ^{N×N}.
    The tensors are permuted to [B,H,N,N] → [BH,N,N] and a
    custom Triton kernel performs the GEMM in FP16 (tensor‑cores)
    with FP32 accumulation.
6️⃣  Reshape C back to [B,N,N,H] and apply a second LayerNorm over H.
7️⃣  Multiply by the output‑gate and finish with a bias‑free linear
    projection back to the original dim.
8️⃣  Return a float‑32 tensor of shape [B, N, N, dim].

The Triton kernel is a classic tiled matmul (BLOCK_M×BLOCK_N tiles,
BLOCK_K=16 to match tensor‑core shape).  It uses 4 warps per block
(`num_warps=4`) and pipelines the loads with two stages for high
throughput on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ
#   A : [BH, M, K]   (fp16)
#   B : [BH, N, K]   (fp16) – we access its transpose inside the kernel
#   C : [BH, M, N]   (fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes
    stride_ah, stride_ai, stride_ak,   # A strides  (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # index over (batch * hidden)
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension (tile size = BLOCK_K)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----------- load A (M × K) --------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)               # fp16 → fp32 for tensor‑core matmul

        # ----------- load Bᵀ (K × N) -------------------------------
        # B is stored as (batch, N, K).  We treat it as transposed
        # inside the kernel: (K, N).
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)

        # ---------- inner product (tensor‑core dot) ----------------
        acc += tl.dot(a, b)                 # (BLOCK_M×BLOCK_K) @ (BLOCK_K×BLOCK_N)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (over `dim`)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates and cast to FP16 (the data that enters the cubic kernel)
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)   # [B, N, N, H]  fp16
    right = (right_proj * right_gate).to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6️⃣ Re‑order to [B, H, N, N] and flatten batch*hidden → [BH, N, N]
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]   fp16
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]   fp16

    # ------------------------------------------------------------------
    # 7️⃣ Output buffer (FP32)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 8️⃣ Triton kernel launch (batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 64               # rows per tile
    BLOCK_N = 64               # cols per tile
    BLOCK_K = 16               # inner‑tile (tensor‑core friendly)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one program per (batch,hidden) pair
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),    # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # 4 warps → 128 threads per program
        num_stages=2,         # double‑buffered load pipeline
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 🔟 Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
