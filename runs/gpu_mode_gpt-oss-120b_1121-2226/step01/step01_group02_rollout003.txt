"""
TriMul (outgoing) – fast forward pass
------------------------------------
The implementation follows the reference PyTorch module but moves the
N³‑like “outer‑product” into a custom Triton kernel that performs a
batched half‑precision matrix multiplication with fp32 accumulation
(using Tensor‑Core dot).  All linear layers and the two LayerNorms are
run in fp16 (mixed‑precision inference) to cut memory bandwidth.
The heavy matmul is done on the tensor (B·H, N, N) where H is the hidden
dimension; each hidden channel is an independent batch.  The kernel is
tuned for H100 (large BLOCK_M/N/K and 8 warps) and falls back to smaller
tiles for very small N.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matmul  A (M×K) * B (N×K) → C (M×N)
#   A : (batch, M, K)   fp16
#   B : (batch, N, K)   fp16   (K is the inner dimension, N is the row dim)
#   C : (batch, M, N)   fp32   (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,            # pointers
    M, N, K,                        # matrix dimensions
    stride_ah, stride_am, stride_ak,  # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,  # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,  # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)            # block index in M
    pid_n = tl.program_id(1)            # block index in N
    pid_b = tl.program_id(2)            # batch (hidden‑channel) index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this tile (k‑tile starts at 0)
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator (fp32) – Tensor‑Core dot promotes fp16 → fp32 automatically
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # Mask out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        acc += tl.dot(a, b)   # fp16·fp16 → fp32 (Tensor‑Core)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 128,
) -> torch.Tensor:
    """
    Batched matmul used in the outgoing TriMul:
        C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]

    A : (batch, M, K) – fp16
    B : (batch, N, K) – fp16   (note the K‑dimension is the innermost)
    Returns C : (batch, M, N) – fp32
    """
    assert A.is_cuda and B.is_cuda, "inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in elements (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C

    # ------------------------------------------------------------------

    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional masking (broadcasted to hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is float32 or bool → cast to fp16 and add singleton dim
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gates – all in fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = F.linear(x_h, left_gate_w).sigmoid()   # [B, N, N, H] fp16
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul – batched matmul on fp16 with fp32 accumulation
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (B*H, N, N)  – each hidden channel becomes a separate batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Choose tile size adaptively (larger tiles for bigger N)
    if N >= 512:
        BM, BN, BK = 128, 128, 128
    else:
        BM, BN, BK = 64, 64, 64

    # C = Σ_k left[i,k,:] * right[j,k,:]   → shape (B*H, N, N), fp32
    out_mat_fp32 = _batched_matmul_triton(
        left_mat, right_mat, BLOCK_M=BM, BLOCK_N=BN, BLOCK_K=BK
    )  # float32

    # back to (B, N, N, H) and cast to fp16 for the following LayerNorm
    out = out_mat_fp32.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().half()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + gating + final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    out = out * out_gate          # element‑wise gating (fp16)

    to_out_w = weights["to_out.weight"].half()
    out = F.linear(out, to_out_w)  # [B, N, N, dim]  fp16

    # Cast back to float32 as required by the API
    return out.float()
```
