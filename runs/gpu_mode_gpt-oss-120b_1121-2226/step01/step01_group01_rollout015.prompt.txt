```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
custom_kernel
==============

Implements the *outgoing* TriMul operation from AlphaFold‑3 using a
mixed‑precision fusion:

1. **LayerNorm + Linear projections** (FP32) → `x_norm`, `left`, `right`.
2. **Optional mask** multiplication.
3. **Gating** (sigmoid) and element‑wise modulation.
4. **Batched matrix multiplication** over the sequence dimension:
   for every hidden channel `d` we compute
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
   This is performed by a custom Triton kernel that fuses the
   `B·hidden_dim` independent matmuls (`A @ Bᵀ`) using block‑wise
   loads, `tl.dot` and FP16 arithmetic.
5. **LayerNorm + output gate** (FP32) and final linear projection back
   to the original feature dimension.

All heavy compute is done in FP16 inside the Triton kernel, while the
surrounding operations stay in FP32 for numerical stability. The final
output tensor is returned in `float32` as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ)
# A : (batch, M, K)   – fp16
# B : (batch, K, N)   – fp16   (already transposed wrt the original tensor)
# C : (batch, M, N)   – fp16
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_am, stride_ak, stride_ab,   # strides of A (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides of B (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # strides of C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b, :, :] = A[b, :, :] @ B[b, :, :].T  for each batch b.
    """
    pid_m = tl.program_id(0)   # tile index over M
    pid_n = tl.program_id(1)   # tile index over N
    pid_b = tl.program_id(2)   # batch index (B * hidden_dim)

    # ------------------------------------------------------------------
    # Compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds handling
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Pointers to the start of the current batch
    a_batch_ptr = a_ptr + pid_b * stride_ab
    b_batch_ptr = b_ptr + pid_b * stride_bb
    c_batch_ptr = c_ptr + pid_b * stride_cb

    # Compute the pointer matrix for A and B (tile‑wise)
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles (masked loads return 0.0 for OOB elements)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # tl.dot performs Σ_k a[i,k] * b[k,j] over the K axis
        acc += tl.dot(a, b)

        # Move pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Cast accumulator back to FP16 for storage
    c = acc.to(tl.float16)

    # Store the result (masked store for edge tiles)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True => no mask is applied

    # --------------------------------------------------------------
    # Extract weight tensors (all are float32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]     # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    # Normalized over the last dimension (dim)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight) # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Optional mask (mask shape [B, N, N])
    # --------------------------------------------------------------
    if not nomask:
        # mask is broadcastable to [..., hidden_dim]
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to fp16 for the heavy O(N³) reduction
    # --------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape   # N = seq_len

    # ------------------------------------------------------------------
    # 5a. Prepare tensors for batched matmul:
    #     left:  (B, hidden_dim, N, N)   -> A
    #     right: (B, hidden_dim, N, N)   -> Bᵀ  (k dimension already in last axis)
    # ------------------------------------------------------------------
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)  # transpose k axis

    # Output container
    out_mat = torch.empty_like(left_mat)   # fp16

    # ------------------------------------------------------------------
    # 5b. Compute strides required by Triton
    # ------------------------------------------------------------------
    stride_am = left_mat.stride(1)   # stride for M dimension
    stride_ak = left_mat.stride(2)   # stride for K dimension
    stride_ab = left_mat.stride(0)   # stride for batch dimension

    stride_bk = right_mat.stride(1)  # stride for K dimension (second dim after batch)
    stride_bn = right_mat.stride(2)  # stride for N dimension
    stride_bb = right_mat.stride(0)  # batch stride

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    # ------------------------------------------------------------------
    # 5c. Launch Triton kernel
    # ------------------------------------------------------------------
    # Block sizes – tuned for H100 FP16 TensorCores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N),
            B * hidden_dim)   # one grid entry per hidden‑dim batch

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,    # H100 prefers 8‑warps per block for FP16 TensorCore usage
    )

    # ------------------------------------------------------------------
    # 5d. Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate   # out_gate is still FP32
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
```
