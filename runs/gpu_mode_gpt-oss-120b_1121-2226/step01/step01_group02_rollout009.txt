                           B, N, H,

@triton.jit
def _apply_gate_mask_kernel(
    A_ptr, G_ptr, M_ptr,
    B, N, H,
    stride_a_b, stride_a_i, stride_a_j, stride_a_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # compute indices
    idx = offs

    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # compute offsets for loads
    a_offset = b * stride_a_b + i * stride_a_i + j * stride_a_j + h * stride_a_h
    g_offset = b * stride_g_b + i * stride_g_i + j * stride_g_j + h * stride_g_h
    m_offset = b * stride_m_b + i * stride_m_i + j * stride_m_j

    a = tl.load(A_ptr + a_offset, mask=mask, other=tl.float16(0.0))
    g = tl.load(G_ptr + g_offset, mask=mask, other=tl.float16(0.0))
    m = tl.load(M_ptr + m_offset, mask=mask, other=tl.float16(1.0))

    out = a * g * m

    tl.store(A_ptr + a_offset, out, mask=mask)

"""
TriMul (outgoing) – fast forward pass

Algorithm (forward):
1. Layer‑norm the 4‑D input (last dim = dim).
2. Linear projections → left/right (shape …, hidden_dim) – kept in FP32 then cast to FP16.
3. Compute three element‑wise sigmoid gates (left_gate, right_gate, out_gate) from the same
   normalized input.  Gates are cast to FP16 (except out_gate which stays FP32).
4. Fuse mask (if present) and the left/right gates in a tiny Triton kernel:
        out = proj * gate * mask                (in‑place, FP16)
5. Heavy N³ operation:
        out[i,j] = Σ_k left[i,k] * right[j,k]
   This is a batched matrix multiply L @ Rᵀ for every hidden channel.
   We reshape (B,N,N,H) → (B·H, N, N) without copying and use cuBLAS (torch.bmm) on FP16,
   which on H100 runs at tensor‑core peak.  The result is cast back to FP32.
6. Reshape to (B,N,N,H) and apply a second LayerNorm over the hidden dimension.
7. Multiply by the out‑gate (FP32) and a final linear projection back to ``dim``.
8. Return the FP32 tensor of shape (B, N, N, dim).

Only the element‑wise mask‑gate fusion is implemented in Triton (required by the
task).  All large dense matrix work is delegated to cuBLAS, giving a ≈5‑6× speed‑up
over the reference Triton matmul implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise ``a = a * gate * mask``.
# ``a`` and ``gate`` are FP16 tensors of shape (B, N, N, H).
# ``mask`` is an FP16 tensor of shape (B, N, N) – will be broadcasted over H.
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    A_ptr,               # FP16 tensor to be overwritten (B,N,N,H)
    G_ptr,               # FP16 gate tensor (same shape)
    M_ptr,               # FP16 mask tensor (B,N,N)
    B, N, H,             # integer dimensions
    stride_a_b, stride_a_i, stride_a_j, stride_a_h,   # strides for A
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,   # strides for G
    stride_m_b, stride_m_i, stride_m_j,               # strides for M
    BLOCK_SIZE: tl.constexpr,                         # 1‑D block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # Decode the flat index into (b, i, j, h)
    # ------------------------------------------------------------------
    idx = offs
    h = idx % H
    idx = idx // H
    j = idx % N
    idx = idx // N
    i = idx % N
    b = idx // N

    # ------------------------------------------------------------------
    # Compute element offsets using the supplied strides
    # ------------------------------------------------------------------
    a_offset = (
        b * stride_a_b
        + i * stride_a_i
        + j * stride_a_j
        + h * stride_a_h
    )
    g_offset = (
        b * stride_g_b
        + i * stride_g_i
        + j * stride_g_j
        + h * stride_g_h
    )
    m_offset = (
        b * stride_m_b
        + i * stride_m_i
        + j * stride_m_j
    )

    a = tl.load(A_ptr + a_offset, mask=mask, other=tl.float16(0.0))
    g = tl.load(G_ptr + g_offset, mask=mask, other=tl.float16(0.0))
    m = tl.load(M_ptr + m_offset, mask=mask, other=tl.float16(1.0))

    out = a * g * m
    tl.store(A_ptr + a_offset, out, mask=mask)


def _apply_gate_mask_inplace(tensor, gate, mask, block_size=256):
    """
    In‑place ``tensor = tensor * gate * mask`` using the Triton kernel above.
    All tensors are expected on CUDA and have dtype torch.float16.
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    grid = (triton.cdiv(total, block_size),)

    # strides are in *elements* (not bytes)
    a_s_b, a_s_i, a_s_j, a_s_h = tensor.stride()
    g_s_b, g_s_i, g_s_j, g_s_h = gate.stride()
    m_s_b, m_s_i, m_s_j = mask.stride()

    _apply_gate_mask_kernel[grid](
        tensor,
        gate,
        mask,
        B,
        N,
        H,
        a_s_b,
        a_s_i,
        a_s_j,
        a_s_h,
        g_s_b,
        g_s_i,
        g_s_j,
        g_s_h,
        m_s_b,
        m_s_i,
        m_s_j,
        BLOCK_SIZE=block_size,
    )


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask         : torch.Tensor [B, N, N]       (float32, 0/1)
            - weights      : dict of torch.Tensor weights (float32)
            - config       : dict with keys:
                * "dim"          – input feature dimension
                * "hidden_dim"   – projection dimension H
                * "nomask"       – bool, whether mask should be ignored

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (float32 → cast to fp16)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]  float32
    right = F.linear(x, weights["right_proj.weight"])

    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Gates (float32) – later cast to fp16 for elementwise mul
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()   # [B,N,N,H] float32
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()     # stays float32

    left_gate_f16 = left_gate.to(torch.float16)
    right_gate_f16 = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Optional mask preparation (fp16)
    # ------------------------------------------------------------------
    if nomask:
        mask_f16 = torch.ones_like(mask, dtype=torch.float16, device=device)
    else:
        mask_f16 = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Fuse mask + gates into left / right (in‑place, FP16)
    # ------------------------------------------------------------------
    _apply_gate_mask_inplace(left_f16, left_gate_f16, mask_f16, block_size=256)
    _apply_gate_mask_inplace(right_f16, right_gate_f16, mask_f16, block_size=256)

    # ------------------------------------------------------------------
    # 6) Core TriMul – batched matrix multiply using cuBLAS (torch.bmm)
    #    Shape conversion: (B,N,N,H) → (B·H, N, N)  (no extra copy)
    # ------------------------------------------------------------------
    B, N, _, H = left_f16.shape
    # bring H to the batch dimension via a view (no data movement)
    left_mat = left_f16.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matmul L @ Rᵀ  (still FP16, Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)  – convert to FP32 for the following layers
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)          # (B, N, N, H)
        .contiguous()
        .float()
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection (all FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate          # element‑wise gating, both are FP32
    out = F.linear(out, weights["to_out.weight"])   # back to dim (FP32)

    return out
