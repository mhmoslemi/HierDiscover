import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_m_b, stride_m_i, stride_m_k,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < B * N * N * H

    # flatten index to 4d coordinates
    b = offs // (N * N * H)
    idx = offs % (N * N * H)
    i = idx // (N * H)
    idx2 = idx % (N * H)
    k = idx2 // H
    h = idx2 % H

    # pointers
    left_off = b * stride_l_b + i * stride_l_i + k * stride_l_k + h * stride_l_h
    right_off = b * stride_r_b + i * stride_r_i + k * stride_r_k + h * stride_r_h
    lg_off = b * stride_lg_b + i * stride_lg_i + k * stride_lg_k + h * stride_lg_h
    rg_off = b * stride_rg_b + i * stride_rg_i + k * stride_rg_k + h * stride_rg_h
    mask_off = b * stride_m_b + i * stride_m_i + k * stride_m_k

    # loads
    l = tl.load(left_ptr + left_off, mask=mask, other=0.0)
    r = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + lg_off, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=0.0)

    # fused multiplication
    l = l * m * lg
    r = r * m * rg

    tl.store(left_ptr + left_off, l, mask=mask)
    tl.store(right_ptr + right_off, r, mask=mask)

    Triton‑accelerated forward pass of the outgoing TriMul operator.

    The final output is returned in FP32.

    hidden_dim = config["hidden_dim"]

    # Linear weights (no bias) and LayerNorm parameters
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16) if weights["norm.bias"] is not None else None

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16) if weights["to_out_norm.bias"] is not None else None

    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ----- 1. Input layer‑norm (FP16) -----------------------------------------
    x = F.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B,N,N,dim] half

    # ----- 2. Linear projections ------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    # ----- 3. Gating tensors (sigmoid, FP16) -----------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ----- 4. Optional mask & gate fusion (Triton) -----------------------------
    if mask is not None:
        # mask shape [B,N,N] -> broadcast to hidden dim
        mask_h = mask.to(torch.float16)
        # ensure tensors are contiguous for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_h = mask_h.contiguous()
        _apply_mask_and_gate(left, right, left_gate, right_gate, mask_h)
    else:
        # simple elementwise gating
        left = left * left_gate
        right = right * right_gate

    # ----- 5. TriMul contraction: batched matmul (torch.bmm) --------------------
    # reshape for batch GEMM
    # left : (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    # right needs transposition on the N axes (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)

    B, H, N, _ = left_perm.shape
    A = left_perm.view(B * H, N, N)          # (B*H,N,N)
    B_mat = right_perm.view(B * H, N, N)     # (B*H,N,N)

    # torch.bmm uses Tensor Cores for FP16
    out_mat = torch.bmm(A, B_mat)            # (B*H,N,N), half

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ----- 6. Second LayerNorm over hidden dimension -------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ----- 7. Output gate ------------------------------------------------------
    out = out * out_gate

    # ----- 8. Final projection back to `dim` ----------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim), half

    # Cast back to FP32 (as required by the original module)
    return out.float()

Triton‑accelerated forward pass of the outgoing TriMul operator.

Algorithm
---------
1️⃣ Cast everything to fp16 (TensorCore) and apply a layer‑norm on the last
   dimension of the input tensor.
2️⃣ Linear projections → left / right (no bias).
3️⃣ Compute three gating tensors (sigmoid of linear projections).
4️⃣ If a mask is supplied, a tiny Triton kernel fuses the mask with the left
   and right gates:
        left  ← left  * mask * left_gate
        right ← right * mask * right_gate

5️⃣ Contraction: for each hidden channel we need
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h] .

6️⃣ Second layer‑norm over the hidden dimension.
7️⃣ Multiply by the output gate.
8️⃣ Final linear projection back to `dim` and cast to fp32.

    # --------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------
    # 7️⃣ Apply output gate
    # --------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # --------------------------------------------------
    out = F.linear(out, to_out_w)   # (B, N, N, dim), fp16

    # Return in float32 as required by the reference implementation
    return out.float()
```
