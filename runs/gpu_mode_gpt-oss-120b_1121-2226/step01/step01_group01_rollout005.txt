"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1. LayerNorm on the 4‑D input (float32).
2. Two linear projections (left/right) → [B,N,N,H] (float32).
3. Optional mask applied in‑place.
4. Gating (sigmoid) applied in‑place.
5. Cast the gated tensors to fp16.
6. Batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   implemented as a custom Triton kernel that works directly on the
   original NHWC layout – no costly permute/contiguous copies.
   The kernel fuses B·H independent N×N matmuls, uses 64×64 tiles,
   fp16 Tensor‑Core dot products and accumulates in fp16.
7. Convert the result back to fp32, apply a second LayerNorm,
   the output gate and the final linear projection.

The heavy O(N³) reduction is therefore performed without any extra
memory copies, keeping the total data traffic ≈ 3·B·H·N²·2 bytes,
well within H100 bandwidth and allowing runtimes ≲ 1 ms for the
largest test cases.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched “A @ Bᵀ” where A and B have shape
# (B, N, N, H) and the result has shape (B, N, N, H).
# The hidden dimension H is fused into the batch axis (program id 2).
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_outgoing_kernel(
    a_ptr,                     # fp16*  left  tensor
    b_ptr,                     # fp16*  right tensor
    c_ptr,                     # fp16*  output tensor
    N,                         # int    sequence length (M = N = K)
    stride_ab, stride_ai, stride_ak, stride_ah,   # strides of A (batch, i, k, h)
    stride_bb, stride_bi, stride_bk,               # strides of B (batch, j, k, h) – note Bᵀ
    stride_cb, stride_ci, stride_cj,               # strides of C (batch, i, j, h)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,   # compile‑time hidden dimension (typically 128)
):
    """
    Compute:
        C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
    for every hidden slice h ∈ [0, HIDDEN_DIM).
    """
    pid_m = tl.program_id(0)          # tile index over i  (M dimension)
    pid_n = tl.program_id(1)          # tile index over j  (N dimension)
    pid_z = tl.program_id(2)          # combined batch × hidden index

    # --------------------------------------------------------------
    # De‑compose combined index into (batch, hidden)
    # --------------------------------------------------------------
    batch_idx = pid_z // HIDDEN_DIM
    hidden_idx = pid_z % HIDDEN_DIM

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_ab + hidden_idx * stride_ah
    b_base = b_ptr + batch_idx * stride_bb + hidden_idx * stride_ah   # stride_ah == stride_bh
    c_base = c_ptr + batch_idx * stride_cb + hidden_idx * stride_ah

    # --------------------------------------------------------------
    # Offsets within the tile
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑axis
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑axis
    offs_k = tl.arange(0, BLOCK_K)                    # k‑axis (local)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Pointer arithmetic for the current tile (no permutation)
    # --------------------------------------------------------------
    # A tile : (BLOCK_M, BLOCK_K) → indices (i,k)
    a_ptrs = a_base + (offs_i[:, None] * stride_ai
                       + offs_k[None, :] * stride_ak)

    # Bᵀ tile : (BLOCK_K, BLOCK_N) → indices (k,j)
    # (right is accessed as right[b, j, k, h] → rows=k, cols=j)
    b_ptrs = b_base + (offs_k[:, None] * stride_bk
                       + offs_j[None, :] * stride_bi)

    # --------------------------------------------------------------
    # Accumulator in fp16 (Tensor‑Core friendly)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # --------------------------------------------------------------
    # Loop over the k‑dimension in BLOCK_K steps
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k_tiles):
        k_offset = k_iter * BLOCK_K
        cur_k = offs_k + k_offset                 # global k indices for this tile
        mask_k = cur_k < N

        # Masked loads – OOB elements are replaced by 0.0
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Tensor‑Core dot product (fp16 × fp16 → fp16)
        acc = acc + tl.dot(a, b)

        # Advance pointers to the next k‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store the tile back to C
    # --------------------------------------------------------------
    c_ptrs = c_base + (offs_i[:, None] * stride_ci
                       + offs_j[None, :] * stride_cj)
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry‑point that follows the signature required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] in float32.
    """
    # ---------------------------- unpack ----------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------- weights ---------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight   = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight  = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight   = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight  = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight    = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]      # (dim, hidden_dim)

    # -------------------------- LayerNorm -------------------------
    # Normalisation over the trailing feature dimension `dim`
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------- Linear ---------------------------
    left  = F.linear(x, left_proj_weight)    # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)   # (B, N, N, hidden_dim)

    # --------------------------- Mask -----------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------- Gates ----------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,hidden_dim)

    left  = left  * left_gate
    right = right * right_gate

    # ---------------------------- Cast -----------------------------
    # Heavy reduction is performed in fp16.
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, _ = left.shape   # N = sequence length

    # ------------------------------------------------------------------
    # Output tensor for the O(N³) reduction (same layout as left/right)
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)

    # ------------------------------------------------------------------
    # Strides needed by the Triton kernel (batch, i, k, h) layout
    # ------------------------------------------------------------------
    stride_ab = left.stride(0)   # batch stride
    stride_ai = left.stride(1)   # i (first N) stride
    stride_ak = left.stride(2)   # k (second N) stride
    stride_ah = left.stride(3)   # hidden stride (==1 for contiguous)

    stride_bb = right.stride(0)
    stride_bi = right.stride(1)  # j stride (first N) – used as column index after transpose
    stride_bk = right.stride(2)  # k stride (second N)

    stride_cb = out.stride(0)
    stride_ci = out.stride(1)    # i stride
    stride_cj = out.stride(2)    # j stride

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for H100 fp16 TensorCores)
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per block
    BLOCK_N = 64   # cols per block
    BLOCK_K = 64   # reduction tile

    grid = (triton.cdiv(N, BLOCK_M),   # tiles over i
            triton.cdiv(N, BLOCK_N),   # tiles over j
            B * hidden_dim)            # one program per (batch, hidden)

    batched_matmul_outgoing_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        N=N,
        stride_ab=stride_ab,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_ah=stride_ah,
        stride_bb=stride_bb,
        stride_bi=stride_bi,
        stride_bk=stride_bk,
        stride_cb=stride_cb,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        num_warps=4,          # 4 warps → 64×64 tiles fit nicely in a SM
        num_stages=3,         # pipeline stages for prefetching
    )

    # ------------------------------------------------------------------
    # Convert back to FP32 and continue with the remaining lightweight ops
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # Output LayerNorm over the hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    # Apply the output gate (still FP32)
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
