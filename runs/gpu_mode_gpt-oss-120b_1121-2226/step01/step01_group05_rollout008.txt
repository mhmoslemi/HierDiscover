"""
TriMul outgoing (AlphaFold3) – fast Triton implementation
========================================================

We compute

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

with
    left  = (x @ W_left)   * sigmoid(x @ W_left_gate)
    right = (x @ W_right)  * sigmoid(x @ W_right_gate)

All linear layers are performed in FP16, the heavy cubic part is a batched
matrix‑multiply “A @ Bᵀ’’ for every hidden channel h (and batch b).  By
re‑interpreting the tensors as [B·H, N, N] we can reuse a classic tiled GEMM
kernel written in Triton.  The kernel uses FP16 inputs, FP32 accumulation,
and is tuned for the Hopper H100 (128×128 tiles, 8 warps → full Tensor‑Core
utilisation).  The rest of the network (LayerNorms, gating, final projection)
remains in PyTorch.  The final result is returned as FP32.

The implementation avoids an explicit ``contiguous`` copy of the
[ B, N, N, H ] tensors – the permute changes the stride order only, and the
Triton kernel works with arbitrary strides, so the data are streamed directly
from the permuted view.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ
#   A : [BH, M, K]  (FP16)
#   B : [BH, N, K]  (FP16) – we load its transpose inside the kernel
#   C : [BH, M, N]  (FP32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,         # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,         # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,         # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # batch*hidden index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- load A (M×K) ------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)   # FP16 → FP32

        # ---- load Bᵀ (K×N) ------------------------------------------------
        # B is stored as (N, K); we read it as its transpose.
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)   # FP16 → FP32

        # ---- inner product ------------------------------------------------
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---- write result ----------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing’’ operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 early – all subsequent linear ops are done in FP16
    # ------------------------------------------------------------------
    x_h = x_norm.half()                     # FP16 view – safe because LayerNorm is already done

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) in FP16
    # ------------------------------------------------------------------
    left_proj = F.linear(
        x_h, weights["left_proj.weight"].half()
    )  # [B, N, N, hidden_dim] (FP16)
    right_proj = F.linear(
        x_h, weights["right_proj.weight"].half()
    )  # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast on hidden dimension) – applied before gating
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)  # [B, N, N, 1] (FP16)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid) – also in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_h, weights["left_gate.weight"].half())
    )  # [B, N, N, hidden_dim] (FP16)
    right_gate = torch.sigmoid(
        F.linear(x_h, weights["right_gate.weight"].half())
    )
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].half())
    )  # [B, N, N, hidden_dim] (FP16)

    # ------------------------------------------------------------------
    # 6. Apply gates (still FP16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7. Prepare tensors for the batched GEMM
    #    We permute to (B, H, N, N) – this changes strides only,
    #    no memory copy is performed (no .contiguous()).
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2)      # [B, H, N, N]   (FP16, non‑contiguous)
    right_perm = right.permute(0, 3, 1, 2)    # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    BH = B * H

    # View as [BH, N, N] – still a view because the H‑stride is 1.
    left_flat = left_perm.view(BH, N, N)    # [BH, N, N]   (FP16)
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]

    # Output buffer (FP32 accumulator)
    out_flat = torch.empty(
        (BH, N, N), dtype=torch.float32, device=device
    )  # [BH, N, N]

    # ------------------------------------------------------------------
    # 8. Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64                     # fits comfortably in shared memory

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden) dimension
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # best for FP16 Tensor‑Core on H100
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N)      # [B, H, N, N] (FP32)
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 10. Second LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # FP32

    # ------------------------------------------------------------------
    # 11. Apply the output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate          # out_gate is FP16 – will be up‑cast automatically
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim] (float32)

    return out
