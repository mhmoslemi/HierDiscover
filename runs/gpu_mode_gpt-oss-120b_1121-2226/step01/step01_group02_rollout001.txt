def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1 layer_norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        m = mask.to(x.dtype).unsqueeze(-1)  # shape broadcast to [B,N,N,1]
        left = left * m
        right = right * m

    # gates
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # convert to half
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape

    # reshape to (B*H,N,N)
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (batch, M=N, K=N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # choose block sizes
    BLOCK_M = 64 if N <= 256 else 128
    BLOCK_N = BLOCK_M
    BLOCK_K = BLOCK_M

    out_mat = _batched_matmul_triton(left_mat, right_mat,
                                     BLOCK_M=BLOCK_M,
                                     BLOCK_N=BLOCK_N,
                                     BLOCK_K=BLOCK_K)

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul “outgoing” forward pass (AlphaFold3 style) – Triton accelerated.

The heavy N³‑like operation
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is a batched matrix multiplication where every hidden channel d is an
independent batch:
    left   → (B·H, N, N)   (M=N, K=N)
    right  → (B·H, N, N)   (N, K=N)
The Triton kernel computes C = A @ Bᵀ for each batch with fp16 inputs and
fp32 accumulation.  The rest of the module (layer‑norms, linear layers,
gates, masking) is performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block index in M dimension
    pid_n = tl.program_id(1)          # block index in N dimension
    pid_b = tl.program_id(2)          # batch (i.e. hidden‑channel) index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the beginning of the tile in A and B
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # OOB masks
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        acc += tl.dot(a, b)   # fp16·fp16 → fp32

        # Advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matmul implemented in Triton.

    A : (batch, M, K)  – fp16
    B : (batch, N, K)  – fp16
    Returns C : (batch, M, N) – fp32
    """
    assert A.is_cuda and B.is_cuda, "A and B must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "A and B must be fp16"

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in elements (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    data = (input_tensor, mask, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # input feature dimension
    hidden_dim = config["hidden_dim"]  # projection dimension (H)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → left/right (shape [..., hidden_dim])
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to the same dtype as left/right
        m = mask.to(dtype).unsqueeze(-1)        # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gates – sigmoid activations from the same normalized input
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply gating (element‑wise)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul: batched matmul over hidden channels
    #    Cast to fp16 for the heavy kernel, accumulate in fp32.
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape  # B, seq_len, seq_len, hidden_dim

    # (B, H, N, N) → (B·H, N, N)  each hidden channel is an independent batch
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Choose tile sizes – larger tiles for bigger N to reduce loop count
    if N <= 256:
        BLOCK_M = BLOCK_N = BLOCK_K = 64
    else:
        BLOCK_M = BLOCK_N = BLOCK_K = 128

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )   # shape (B·H, N, N), dtype=float32

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # back to dim

    return out
