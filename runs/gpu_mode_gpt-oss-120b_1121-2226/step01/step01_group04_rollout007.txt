    pid_n = tl.program_id(2)

    # Compute offsets...

    # Offsets: row range = pid_m*BLOCK_M + arange

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Input LayerNorm over the channel dimension `dim`.
2️⃣ Two linear projections (`left_proj`/`right_proj`) → tensors L,R of shape
   [B, N, N, H] (H = hidden_dim).  No bias.
3️⃣ Three sigmoidal gates (left‑gate, right‑gate, out‑gate) are computed from the
   normalized input.
4️⃣ **Triton kernel #1** fuses the element‑wise operations
        L ← L * left_gate * mask
        R ← R * right_gate * mask
   This avoids an extra memory pass and works in FP16.
5️⃣ **Triton kernel #2** performs the heavy O(N³) bilinear product
        out₍b,i,j,c₎ = Σₖ L₍b,i,k,c₎ * R₍b,j,k,c₎
   The hidden dimension and batch are collapsed to a single leading dimension,
   so the computation is a batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ.
   The kernel is a classic tiled fp16 GEMM that accumulates in fp32
   (Tensor‑core friendly) and writes the result in fp16.
6️⃣ The product is reshaped back to [B, N, N, H] and converted to FP32.
7️⃣ Second LayerNorm over the hidden dimension, multiplication by `out_gate`
   and a final linear projection back to `dim`.

Only the element‑wise gating and the large matrix‑multiply are written in
Triton; everything else uses the highly‑optimized PyTorch kernels.
Both kernels run on the H100, use FP16 for bandwidth‑critical paths and
accumulate in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask + sigmoid‑gates into the projected tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,          # FP16 tensors (flattened)
    left_gate_ptr, right_gate_ptr,
    mask_ptr,                     # FP16 tensor (flattened, same size)
    size,                         # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
    """
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load (masked loads return 0.0 for out‑of‑bounds threads)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    # Fuse
    l = l * lg * m
    r = r * rg * m

    # Store back
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: batched fp16 GEMM (A @ Bᵀ) for the TriMul core.
# ----------------------------------------------------------------------
@triton.jit
def _triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ   where A,B are (batch, M, K) and (batch, N, K) respectively.
       The kernel processes many batches (batch = B × H) in parallel.
    """
    batch_id = tl.program_id(0)   # leading dimension (B·H)
    pid_m = tl.program_id(1)      # tile row
    pid_n = tl.program_id(2)      # tile col

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Allocate accumulator in FP32 (numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in BLOCK_K sized chunks
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A has shape (M, K); Bᵀ has shape (K, N)
        a_ptrs = a_ptr + batch_id * batch_stride_a + \
                 (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
        b_ptrs = b_ptr + batch_id * batch_stride_b + \
                 (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # BLAS‑like dot product over the K‑dimension
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Write the result back to C (stored in FP16)
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * batch_stride_c + \
             (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expand,
        size=total_elems,
        BLOCK_SIZE=BLOCK_ELEM,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul – batched GEMM in Triton
    # ------------------------------------------------------------------
    # Collapse batch and hidden dimension → (B·H, N, N)
    left_reshape  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Output buffer
    out_reshape = torch.empty_like(left_reshape, dtype=torch.float16, device=left.device)

    # Strides (all tensors are contiguous)
    stride_am = left_reshape.stride(1)   # = N
    stride_ak = left_reshape.stride(2)   # = 1
    batch_stride_a = left_reshape.stride(0)   # = N * N

    stride_bk = right_reshape.stride(1)   # = N   (row stride of original right)
    stride_bn = right_reshape.stride(2)   # = 1   (col stride of original right)
    batch_stride_b = right_reshape.stride(0)   # = N * N

    stride_cm = out_reshape.stride(1)    # = N
    stride_cn = out_reshape.stride(2)    # = 1
    batch_stride_c = out_reshape.stride(0)   # = N * N

    # Tiling parameters – chosen for H100 Tensor‑core sweet‑spot.
    BLOCK_M = 64   # multiple of 2 (split inside kernel)
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        bs * hidden_dim,                        # batch × head
        triton.cdiv(N, BLOCK_M),               # tile rows
        triton.cdiv(N, BLOCK_N),               # tile cols
    )
    _triton_matmul_kernel[grid](
        a_ptr=left_reshape,
        b_ptr=right_reshape,
        c_ptr=out_reshape,
        M=N, N=N, K=N,
        stride_am=stride_am, stride_ak=stride_ak,
        stride_bk=stride_bk, stride_bn=stride_bn,
        stride_cm=stride_cm, stride_cn=stride_cn,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and promote to FP32 for the following ops
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )   # [B, N, N, H]  (float32)

    out = out * out_gate.to(torch.float32)   # gating

    out = F.linear(out, weights["to_out.weight"])  # final projection, float32

    return out
