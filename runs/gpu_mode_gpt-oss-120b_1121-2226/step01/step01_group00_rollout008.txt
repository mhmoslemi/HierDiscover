"""
TriMul (outgoing) – Triton‑accelerated forward pass

Algorithm
---------
1️⃣ Layer‑norm on the input tensor (last dimension = dim) – stays in FP32.
2️⃣ Cast to FP16 and apply the four linear projections (left/right and three
   gates).  These use the highly‑optimized cuBLAS kernels provided by
   `torch.nn.functional.linear`.
3️⃣ Optional pairwise mask multiplication (broadcasted over the hidden channel).
4️⃣ Gating: sigmoid(linear(x)).
5️⃣ Core “TriMul”   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   – This is a batched GEMM of shape (B·hidden, N, N) @ (B·hidden, N, N)ᵀ.
   A custom Triton kernel performs the GEMM with FP16 Tensor‑Core
   arithmetic and FP32 accumulation.  Tile sizes (64×64×64) and 8 warps
   give good occupancy on an H100.
6️⃣ Layer‑norm on the hidden dimension, element‑wise out‑gate, and final
   linear projection back to `dim`.  Result is returned in FP32.

The kernel only handles the heavy matrix‑multiply; all other ops use
standard PyTorch kernels (which are already highly optimized).
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ B) where each batch element
# corresponds to one hidden channel of the TriMul operation.
# A: (batch, M, K)   – left tensor   (fp16)
# B: (batch, K, N)   – right tensor  (fp16)  (already transposed)
# C: (batch, M, N)   – result        (fp16, accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_fp16_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix dimensions (all = seq_len)
    stride_am, stride_ak,         # A strides: row‑stride, col‑stride
    stride_bk, stride_bn,         # B strides: row‑stride (k), col‑stride (n)
    stride_cm, stride_cn,         # C strides: row‑stride, col‑stride
    stride_ab, stride_bb, stride_cb,   # batch‑strides for A,B,C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile column
    pid_b = tl.program_id(2)      # batch / hidden index

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the start of the current batch element
    # ------------------------------------------------------------------
    a_batch = A_ptr + pid_b * stride_ab
    b_batch = B_ptr + pid_b * stride_bb
    c_batch = C_ptr + pid_b * stride_cb

    # accumulator in FP32 (tensor‑core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K‑sized chunks
    # ------------------------------------------------------------------
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load sub‑tiles of A and B (FP16)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=tl.float16(0.0))

        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=tl.float16(0.0))

        # Tensor‑core matmul: FP16 * FP16 → FP32 accumulator
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store tile back to C (cast to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[:, None]))


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim] (float32)
            - mask_tensor : [B, N, N] (float32) or None
            - weights_dict : all linear / layer‑norm parameters
            - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (keep FP32 for numerical stability)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                               # [B, N, N, dim]  (FP32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy arithmetic
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (left / right)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w) # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid of a linear projection of the *original* input)
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core TriMul – batched GEMM (B·H independent N×N matmuls)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                                   # batch size, seq_len
    # Re‑order to (B, H, N, N) – left: (i,k), right: (k,j)
    left_mat  = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()         # (B, H, N, N)  (k,j)

    # Collapse batch & hidden dim → (B·H, N, N)
    left_flat  = left_mat.view(B * hidden_dim, N, N)           # A
    right_flat = right_mat.view(B * hidden_dim, N, N)          # B (already transposed)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # Strides (row‑major)
    stride_am = left_flat.stride(1)   # = N
    stride_ak = left_flat.stride(2)   # = 1
    stride_bk = right_flat.stride(1)  # = N
    stride_bn = right_flat.stride(2)  # = 1
    stride_cm = out_flat.stride(1)    # = N
    stride_cn = out_flat.stride(2)    # = 1

    # Batch strides
    stride_ab = left_flat.stride(0)   # N*N
    stride_bb = right_flat.stride(0)  # N*N
    stride_cb = out_flat.stride(0)    # N*N

    # Tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    _batched_gemm_fp16_kernel[
        (grid_x, grid_y, grid_z),
        num_warps=8,
        num_stages=2
    ](
        left_flat,
        right_flat,
        out_flat,
        N,                # M
        N,                # N
        N,                # K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim] and cast to FP32 for the rest
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm, out‑gate, final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out = out * out_gate.to(torch.float32)   # broadcast over hidden dim

    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim)  – keep FP32
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] (FP32)

    return out
