"""
TriMul (outgoing) – fast forward with a tiny Triton fuse

Algorithm
---------
1. Layer‑norm the 4‑D input (last dim = `dim`).
2. Linear projections → left/right (shape …×hidden_dim) and sigmoid gates.
3. Fuse the three element‑wise products
        left = (proj_left  * gate_left  * mask)
        right = (proj_right * gate_right * mask)
   with a small Triton kernel (avoids extra materialisations).
4. Cast the gated tensors to FP16, reshape to (B·H, N, N) and run a
   *single* batched GEMM via `torch.bmm`.  FP16 inputs let the H100
   Tensor‑Cores work at full speed; the result is converted back to FP32.
5. Layer‑norm over the hidden dimension, multiply by the output gate
   (again with a Triton kernel) and finish with a linear projection back
   to the original `dim`.
6. The implementation uses only the forward pass – no gradients are
   required.

Both Triton kernels are tiny element‑wise fusions; the heavy N³ work is
performed by the highly‑optimised cuBLAS batched GEMM, which comfortably
fits the 1 ms target on an H100 for the test configurations.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask  (all tensors are 1‑D‑flattened)
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_kernel(
    proj_ptr,           # float32*  – projection
    gate_ptr,           # float32*  – sigmoid gate
    mask_ptr,           # float32*  – mask (1.0 where kept, 0.0 otherwise)
    out_ptr,            # float32*  – destination
    size,               # int64    – total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # default 1.0 → no masking

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (out_norm * out_gate)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    a_ptr,              # float32*
    b_ptr,              # float32*
    out_ptr,            # float32*
    size,               # int64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        * input_tensor : torch.Tensor [B, N, N, dim] (float32)
        * mask         : torch.Tensor [B, N, N] (float32 or bool)
        * weights_dict : dict of torch.Tensor containing all model weights
        * config_dict  : dict containing `dim`, `hidden_dim`, `nomask` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim         = config["dim"]          # input feature dimension
    hidden_dim  = config["hidden_dim"]   # projection dimension H
    nomask      = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4) Fuse (proj * gate) and optional mask with a Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape
    # Allocate fused tensors
    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    if not nomask:
        # expand mask to the hidden dimension, keep it contiguous
        mask_exp = mask.to(dtype).unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
    else:
        # a dummy mask of all 1.0 (same size as left_proj) – cheap to allocate
        mask_exp = torch.ones_like(left_proj, dtype=dtype, device=device)

    # --------------------------------------------------------------
    # Kernel launch parameters (tunable)
    # --------------------------------------------------------------
    BLOCK_SIZE = 8192
    size = left_proj.numel()
    grid = ( (size + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    # left = left_proj * left_gate * mask
    _proj_gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask_exp,
        left,
        size,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # right = right_proj * right_gate * mask
    _proj_gate_mask_kernel[grid](
        right_proj,
        right_gate,
        mask_exp,
        right,
        size,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 5) Heavy N³ computation – batched GEMM in FP16
    # ------------------------------------------------------------------
    # Cast to half for maximal Tensor‑Core throughput
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # reshape to (B*H, N, N)  – each hidden channel becomes an independent batch
    left_mat  = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul:   (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
    # torch.bmm is highly‑optimised (Tensor‑Core, FP16→FP32 accumulation)
    out_mat_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Convert back to float32 for the remaining FP32‑only layers
    out_mat = out_mat_fp16.to(torch.float32)

    # Restore the original 4‑D layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # still [B, N, N, H]

    # Fuse out_norm * out_gate via Triton
    out_scaled = torch.empty_like(out)
    size_out = out.numel()
    grid_out = ( (size_out + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elemwise_mul_kernel[grid_out](
        out,
        out_gate,
        out_scaled,
        size_out,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Final linear back to the original dimension
    out = F.linear(out_scaled, weights["to_out.weight"])     # -> [B, N, N, dim]

    return out
