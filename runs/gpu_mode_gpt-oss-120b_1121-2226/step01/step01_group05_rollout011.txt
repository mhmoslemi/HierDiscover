```python
"""
TriMul outgoing kernel – batched GEMM `left @ rightᵀ` fused across the hidden
dimension.  The implementation

1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and apply the two linear projections + sigmoid gates.
3. Reshape to [B·H, N, N] and run a custom Triton GEMM with
   BLOCK_M=N_BLOCK_M, BLOCK_N=N_BLOCK_N, BLOCK_K=N_BLOCK_K.
   The kernel loads tiles of A and Bᵀ, accumulates in fp32 (Tensor‑Core
   friendly) and writes the result in fp32.
4. Reshape back to [B, N, N, H], apply a second LayerNorm, an output gate,
   and a final linear projection to the original dimension.

The kernel uses large tiles (128×128×64) to reduce memory traffic
(~2× fewer loads compared with the reference 64×64×32 tiles) and
exploits Tensor‑Cores (fp16 × fp16 → fp32) on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B are fp16, C fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A: (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ: (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C: (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # (batch * hidden) index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M x K)
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load Bᵀ tile (K x N)
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # Write back
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Triton‑backed forward for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] fp32

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy‑lift linear projections
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – fp16 → fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – keep in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM.
    #    Shape after permute: [B, H, N, N] → view as [BH, N, N]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]  fp16
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]  fp16

    # Output buffer (float32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    # Tile sizes – chosen to maximise Tensor‑Core utilisation while staying
    # within shared‑memory limits on an H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden) dimension
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,  # M
        N,  # N
        N,  # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output gate and the final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate          # broadcasting `out_gate` (fp16 → fp32)
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim] fp32

    return out

"""
TriMul outgoing kernel – batched GEMM `left @ rightᵀ` fused across the hidden
dimension.  The implementation

1️⃣ Layer‑norm the 4‑D input (float32).
2️⃣ Cast to fp16 and apply the two linear projections + sigmoid gates.
3️⃣ Reshape to [B·H, N, N] and run a custom Triton GEMM with
   BLOCK_M=128, BLOCK_N=128, BLOCK_K=64.  The kernel loads tiles of A and
   Bᵀ, accumulates in FP32 (Tensor‑Core friendly) and writes the result in
   FP32.
4️⃣ Reshape back to [B, N, N, H], apply a second LayerNorm, an output gate,
   and a final linear projection to the original dimension.

Large tiles (128 × 128 × 64) cut the number of memory loads by ~2× compared
to a naïve 64 × 64 × 32 tiling and fully exploit H100 Tensor‑Cores (FP16 × FP16→FP32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B are fp16, C fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A  (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # (batch * hidden) index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # --------------------------------------------------------------
    # Tile offsets & bounds
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # ----- Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # ----- Accumulate -----
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write the result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.half()

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – fp16 → fp16
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – keep everything in fp16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Reshape to [B·H, N, N] for the batched GEMM
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, N, N, H) → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]  fp16
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]  fp16

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # --------------------------------------------------------------
    # 7️⃣ Launch the Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden) dimension
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # --------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # --------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection
    # --------------------------------------------------------------
    out = out * out_gate                # broadcast gate (fp16 → fp32)
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
