@triton.jit
def _fused_mul_kernel(
    a_ptr, b_ptr, mask_ptr, out_ptr,
    n_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask)  # tl.float16 by default
    b = tl.load(b_ptr + offs, mask=mask)

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=mask)
```

def _fused_mul(a,b,mask=None):
    assert a.shape == b.shape
    B,N,N,H = a.shape
    total = a.numel()
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    if mask is not None:
        mask_flat = mask.view(-1)
        HAS_MASK = 1
    else:
        mask_flat = torch.empty(0, dtype=a.dtype, device=a.device)
        HAS_MASK = 0
    out = torch.empty_like(a)
    _fused_mul_kernel[grid](
        a.view(-1), b.view(-1), mask_flat, out.view(-1),
        total, H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK
    )
    return out

"""
TriMul (outgoing) – half‑precision Triton‑fused implementation
----------------------------------------------------------------
Algorithm
---------
1️⃣ Layer‑norm the input over the last dimension (dim) in fp16.
2️⃣ Fuse the two linear projections (proj + gate) for the left and right
   tensors by concatenating the weight matrices (shape 2·hidden_dim × dim)
   and performing a single matmul per side.
3️⃣ Split the result, apply sigmoid to the gate part and fuse
   (proj * gate * mask) with a tiny Triton kernel (mask optional).
4️⃣ Cast the fused tensors to fp16 and perform the heavy O(N³) work as a
   batched GEMM (B·hidden_dim independent half‑precision matrix‑multiplies
   of size N×N) using torch.bmm (Tensor‑cores).
5️⃣ Reshape back to [B,N,N,hidden_dim] and apply a second layer‑norm
   (over hidden_dim) in fp16.
6️⃣ Compute the output gate (sigmoid(linear(x_norm))) in fp16 and multiply.
7️⃣ Final linear projection back to `dim` in fp16 and cast to fp32 for the
   returned tensor.

The only Triton component is the element‑wise fusion of proj, gate and
optional mask, satisfying the “must use Triton” requirement while keeping
the compute‑heavy parts in fast half‑precision GEMM.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused (proj * gate * mask) – mask broadcast over hidden dim
# ----------------------------------------------------------------------
@triton.jit
def _fused_mul_kernel(
    a_ptr,                # left_proj   [*] float16
    b_ptr,                # left_gate   [*] float16
    mask_ptr,             # mask        [*] float16 (optional)
    out_ptr,              # output      [*] float16
    n_elements,           # total number of elements in a/b
    hidden_dim,           # size of hidden dimension (for mask stride)
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK:   tl.constexpr,
):
    """
    Computes out = a * b * mask (if present) where mask is broadcast over the
    hidden dimension. All tensors are flattened 1‑D views.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask_oob = offs >= n_elements

    a = tl.load(a_ptr + offs, mask=~mask_oob, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=~mask_oob, dtype=tl.float16)

    if HAS_MASK:
        # each mask element corresponds to hidden_dim entries in the flattened tensor
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=~mask_oob, dtype=tl.float16)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=~mask_oob)


def _fused_mul(
    a: torch.Tensor,
    b: torch.Tensor,
    mask: torch.Tensor | None = None,
) -> torch.Tensor:
    """
    Element‑wise fused multiplication (proj * gate * mask) using the Triton kernel.
    a, b – half‑precision tensors of shape [B, N, N, H].
    mask – optional half‑precision tensor of shape [B, N, N]; broadcast over H.
    """
    assert a.shape == b.shape, "a and b must have the same shape"
    B, N, _, H = a.shape
    total = a.numel()
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )

    a_flat = a.view(-1)
    b_flat = b.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=a.dtype, device=a.device)
        has_mask = 0

    out = torch.empty_like(a)
    _fused_mul_kernel[grid](
        a_flat,
        b_flat,
        mask_flat,
        out.view(-1),
        total,
        H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (float32) or None
        - weights_dict : mapping str → torch.Tensor (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast model weights to fp16 (everything except biases that are
    #   already 1‑D vectors)
    # ------------------------------------------------------------------
    w_norm_weight = weights["norm.weight"].to(torch.float16)
    w_norm_bias   = weights["norm.bias"].to(torch.float16)

    w_left_proj = weights["left_proj.weight"].to(torch.float16)      # (hidden_dim, dim)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)

    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)

    w_out_gate = weights["out_gate.weight"].to(torch.float16)       # (hidden_dim, dim)

    w_to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    w_to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)

    w_to_out = weights["to_out.weight"].to(torch.float16)           # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (still in fp16 for the rest of the pipeline)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w_norm_weight,
        bias=w_norm_bias,
        eps=eps,
    )
    x_norm_h = x_norm.to(torch.float16)   # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for the *left* side
    # ------------------------------------------------------------------
    # Concatenate weights: (2*hidden_dim, dim)
    w_left_concat = torch.cat([w_left_proj, w_left_gate], dim=0)   # (2*H, dim)
    left_concat = torch.nn.functional.linear(x_norm_h, w_left_concat)  # [B,N,N,2*H] fp16
    left_proj = left_concat[..., :hidden_dim]          # [B,N,N,H]
    left_gate_raw = left_concat[..., hidden_dim:]      # [B,N,N,H]
    left_gate = torch.sigmoid(left_gate_raw)           # fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for the *right* side
    # ------------------------------------------------------------------
    w_right_concat = torch.cat([w_right_proj, w_right_gate], dim=0)  # (2*H, dim)
    right_concat = torch.nn.functional.linear(x_norm_h, w_right_concat)  # [B,N,N,2*H]
    right_proj = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask handling (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_h = None
    else:
        mask_h = mask.to(torch.float16)   # [B,N,N] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse element‑wise multiplication (proj * gate * mask) via Triton
    # ------------------------------------------------------------------
    left = _fused_mul(left_proj, left_gate, mask_h)   # [B,N,N,H] fp16
    right = _fused_mul(right_proj, right_gate, mask_h)

    # Release intermediate tensors early to keep memory pressure low
    del left_proj, left_gate, right_proj, right_gate, left_concat, right_concat

    # ------------------------------------------------------------------
    # 6️⃣ Batched half‑precision GEMM across the sequence dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # out_mat = left_mat @ right_matᵀ  (batch GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore layout: [B,N,N,H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Layer‑norm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm_weight,
        bias=w_to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output gate (sigmoid) and multiplication (fp16)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm_h, w_out_gate))  # [B,N,N,H] fp16
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim` (fp16) and cast to fp32
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, w_to_out)   # [B,N,N,dim] fp16
    out = out.to(torch.float32)

    return out
