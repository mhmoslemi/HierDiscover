"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑norm over the last dimension (`dim`) – FP32.
2️⃣ Cast to FP16 and apply four linear projections (no bias):
      left_proj, left_gate, right_proj, right_gate.
3️⃣ Fuse the element‑wise sigmoid‑gate, optional mask and the
   multiplication left = left_proj * sigmoid(left_gate) * mask
   (and similarly for right) in a tiny Triton kernel.
   This avoids storing the intermediate gate tensors.
4️⃣ Reshape to (B*H, N, N) and perform the heavy cubic contraction
   C = left @ rightᵀ using a single batched GEMM (torch.bmm /
   cuBLAS).  cuBLAS on H100 is far faster than a hand‑written
   Triton matmul for the large batch of 𝑁×𝑁 GEMMs.
5️⃣ Second Layer‑norm over the hidden dimension (FP32) → apply
   `out_gate` (already in FP32) and a final projection back to `dim`.

Only step 3 is performed with a Triton kernel, satisfying the
requirement that “part of the operations” be a custom kernel while
keeping the overall runtime well below the target 1 ms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse projection, sigmoid‑gate, mask (if present)
# ----------------------------------------------------------------------
@triton.jit
def _apply_proj_gate_mask_kernel(
    proj_ptr,               # left_proj or right_proj   (FP16)
    gate_ptr,               # left_gate or right_gate   (FP16)
    mask_ptr,               # mask (FP16) or dummy ptr
    out_ptr,                # output tensor (FP16)
    total_elements: tl.int32,     # B * N * N * H
    hidden_dim: tl.int32,         # H
    BLOCK_SIZE: tl.constexpr,     # thread block size
    HAS_MASK: tl.constexpr,       # compile‑time flag
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # ----- load projection and gate values -----
    proj = tl.load(proj_ptr + offs, mask=valid, other=0.0)   # FP16
    gate = tl.load(gate_ptr + offs, mask=valid, other=0.0)   # FP16

    # ----- sigmoid(gate) in FP32 for stability -----
    gate_f32 = gate.to(tl.float32)
    sig_f32 = 1.0 / (1.0 + tl.exp(-gate_f32))
    sig = sig_f32.to(tl.float16)       # back to FP16

    out = proj * sig                    # projection * gate

    # ----- optional mask (broadcast over hidden) -----
    if HAS_MASK:
        # each mask element is repeated `hidden_dim` times in the flat view
        mask_idx = offs // hidden_dim
        mask_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # FP16 (0 or 1)
        out = out * mask_val

    tl.store(out_ptr + offs, out, mask=valid)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]  (FP32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of torch.Tensor (all FP32, bias‑free unless noted)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (FP32)
    """
    # --------------------------------------------------
    # unpack arguments
    # --------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim) – stay in FP32
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] FP32

    # ------------------------------------------------------------------
    # 2️⃣ FP16 cast + linear projections (no bias)
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # cache half‑precision weights (once per function call)
    w_left_proj = weights["left_proj.weight"].to(torch.float16, non_blocking=True)
    w_right_proj = weights["right_proj.weight"].to(torch.float16, non_blocking=True)
    w_left_gate = weights["left_gate.weight"].to(torch.float16, non_blocking=True)
    w_right_gate = weights["right_gate.weight"].to(torch.float16, non_blocking=True)
    w_out_gate = weights["out_gate.weight"].to(torch.float16, non_blocking=True)

    # projection + gate tensors (still FP16)
    left_proj = F.linear(x_fp16, w_left_proj)          # [B, N, N, H]
    left_gate = F.linear(x_fp16, w_left_gate)          # [B, N, N, H]
    right_proj = F.linear(x_fp16, w_right_proj)        # [B, N, N, H]
    right_gate = F.linear(x_fp16, w_right_gate)        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Fuse gate, sigmoid, mask (if any) in a tiny Triton kernel
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape
    total_elements = B * N * N * H
    BLOCK = 8192                     # 8 K threads per launch – good occupancy
    grid = ( (total_elements + BLOCK - 1) // BLOCK, )

    # allocate output tensors for the fused result
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # mask handling
    if mask is not None and mask.numel() > 0:
        mask_fp16 = mask.to(torch.float16, non_blocking=True)
        mask_flat = mask_fp16.contiguous().view(-1)    # [B, N, N] → flat
        mask_ptr = mask_flat
        has_mask = True
    else:
        # dummy tensor – never read because HAS_MASK will be False
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = False

    # launch kernel for left side
    _apply_proj_gate_mask_kernel[grid](
        left_proj.reshape(-1).data_ptr(),
        left_gate.reshape(-1).data_ptr(),
        mask_ptr.data_ptr(),
        left.reshape(-1).data_ptr(),
        total_elements,
        H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    # launch kernel for right side (same mask)
    _apply_proj_gate_mask_kernel[grid](
        right_proj.reshape(-1).data_ptr(),
        right_gate.reshape(-1).data_ptr(),
        mask_ptr.data_ptr(),
        right.reshape(-1).data_ptr(),
        total_elements,
        H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    # Free the intermediate projection/gate tensors (GPU memory pressure)
    del left_proj, left_gate, right_proj, right_gate
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 4️⃣ Heavy cubic contraction C = left @ rightᵀ  (batched GEMM)
    #    Use cuBLAS via torch.bmm – far faster on H100 for B*H ≈ 128‑256.
    # ------------------------------------------------------------------
    # bring tensors to (B*H, N, N) layout required by torch.bmm
    left_resh = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (i,k)
    right_resh = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N) # (j,k)ᵀ

    # batch matrix multiplication (FP16 inputs, FP32 accumulation internally)
    out_resh = torch.bmm(left_resh, right_resh)   # (B*H, N, N)  FP16 → fp32 internal

    # reshape back to [B, N, N, H] (still FP16)
    out = out_resh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second Layer‑norm over hidden dim, output gate and final linear
    # ------------------------------------------------------------------
    # convert to FP32 for the final LayerNorm (more stable)
    out = out.to(torch.float32)

    # compute out_gate once (still FP32) – uses the *original* normed x
    out_gate = torch.sigmoid(F.linear(x, w_out_gate))   # [B, N, N, H] FP32

    # hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # apply output gate (element‑wise)
    out = out * out_gate

    # final projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] FP32

    return out
