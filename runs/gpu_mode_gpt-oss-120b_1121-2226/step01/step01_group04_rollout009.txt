    left_proj_weight = weights["left_proj.weight"].half()

    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

def _gate_fuse_kernel(left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, size, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    l = l * lg
    r = r * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    realised by a single batched GEMM on FP16:
        (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
7.  Reshape back to (B,N,N,H) and run a second LayerNorm (FP16) over the
    hidden dimension.
8.  Apply the output gate and the final linear projection back to ``dim``.
9.  Cast the result to FP32 for the caller.

All heavy arithmetic is done in FP16, exploiting tensor‑cores on an H100.
Only the gate‑fusion step is written in Triton (the rest uses highly‑optimised
cuBLAS / CUTLASS kernels via PyTorch).  The implementation works for any
batch size (1‑2), sequence length up to 1024 and both masked / un‑masked
inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place multiplication of left/right with their gates.
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_kernel(
    left_ptr, right_ptr,          # FP16 tensors, flattened
    left_gate_ptr, right_gate_ptr,
    size,                         # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise: left *= left_gate ; right *= right_gate."""
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    l = l * lg
    r = r * rg

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[bs, N, N, dim] (float32)
            - mask: torch.Tensor[bs, N, N] (bool/float) OR ``None``.
            - weights: dict of torch.Tensors (see keys in the docstring).
            - config: dict containing at least ``dim`` and ``hidden_dim`` (ints).

    Returns:
        torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]               # usually 128

    # ------------------------------------------------------------------
    # 1. FP16 cast + LayerNorm (over the last dimension)
    # ------------------------------------------------------------------
    # Cast input to FP16 once
    x = input_tensor.half()

    # LayerNorm parameters (already on device)
    norm_weight = weights["norm.weight"].half()
    norm_bias   = weights["norm.bias"].half()

    # torch.layer_norm works on FP16 tensors as well
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]  (FP16)

    # ------------------------------------------------------------------
    # 2. Apply mask *before* the projections (broadcasted automatically)
    # ------------------------------------------------------------------
    if mask is None:
        # all‑ones mask → no effect
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=x.device)
    else:
        mask_fp16 = mask.to(dtype=torch.float16, device=x.device)

    # broadcast over the channel dimension
    x_masked = x * mask_fp16.unsqueeze(-1)           # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias) → left / right  (FP16)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"].half()   # (H, dim)
    right_proj_weight = weights["right_proj.weight"].half()  # (H, dim)

    left  = F.linear(x_masked, left_proj_weight)   # [B,N,N,H] FP16
    right = F.linear(x_masked, right_proj_weight)

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) – computed from the *un‑masked* normalized tensor
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"].half()
    right_gate_weight = weights["right_gate.weight"].half()
    out_gate_weight   = weights["out_gate.weight"].half()

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H] FP16
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 5. Fuse the two gate‑multiplications with a Triton kernel
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for the 1‑D kernel
    left  = left.contiguous()
    right = right.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    size = left.numel()
    BLOCK = 1024                         # 1‑D launch block size (empirically good)
    grid = (triton.cdiv(size, BLOCK),)

    _gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        size=size,
        BLOCK_SIZE=BLOCK,
    )
    # ------------------------------------------------------------------
    # 6. Core TriMul – batched matrix‑multiply per hidden dimension
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for a single torch.bmm call
    # (B,N,N,H) -> (B, H, N, N) -> (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_batch  = left_perm.view(bs * hidden_dim, N, N)                # (B·H, N, N)
    right_batch = right_perm.view(bs * hidden_dim, N, N)               # (B·H, N, N)

    # batched GEMM: (B·H, N, N) @ (B·H, N, N)^T  →  (B·H, N, N)
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_batch.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)    # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].half()
    to_out_norm_bias   = weights["to_out_norm.bias"].half()

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 8. Apply output gate and final linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = out * out_gate               # element‑wise gate, FP16

    to_out_weight = weights["to_out.weight"].half()   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)               # (B,N,N,dim) FP16

    # ------------------------------------------------------------------
    # 9. Cast to float32 for the caller and return
    # ------------------------------------------------------------------
    return out.float()
```
