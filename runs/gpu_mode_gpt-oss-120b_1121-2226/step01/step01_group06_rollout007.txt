def _trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H, K,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_i, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # axes
    pid_m = tl.program_id(0)  # block row index
    pid_n = tl.program_id(1)  # block col index
    pid_bh = tl.program_id(2) # combined batch*hidden

    # compute batch and hidden indices
    hidden = H  # passed as argument
    batch_idx = pid_bh // hidden
    hidden_idx = pid_bh % hidden

    # compute offsets for i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers for this batch and hidden
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h

    # loop over K dimension
    for k in range(tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Layer‑norm over the last dimension (`dim`) of the 4‑D pair representation.
2. Linear projections (no bias) → left, right  (shape [B, N, N, H]).
3. Optional mask (broadcast over H) and three gating networks (sigmoid of linear
   projections).  Apply the gates to `left` and `right`.
4. **Heavy cubic contraction**  
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]          (∑ over k)
   This is a batched matrix multiplication for each hidden channel h.
   We implement it with a custom Triton kernel that works on the original
   (B,N,N,H) layout – no costly ``permute``/``contiguous`` copies are performed.
5. Second Layer‑norm over the hidden dimension H, multiply by the output gate
   and a final linear projection back to `dim`.

All arithmetic for the N³ contraction is performed in FP16 (accumulated in
FP32) which gives a large bandwidth reduction on H100 Tensor‑Cores.
The rest of the network stays in FP32 for numerical stability.

The kernel uses a 3‑D launch grid:
    program_id(0) → tile row   (i‑axis)
    program_id(1) → tile col   (j‑axis)
    program_id(2) → combined batch‑hidden index (b*H + h)

The kernel receives the original strides of the input tensors, so the
per‑channel view is obtained *without* materialising a copy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A : (B, N, N, H)  – accessed as A[b,i,k,h]
#   B : (B, N, N, H)  – accessed as B[b,j,k,h]  (we read it transposed: B[k,j])
#   C : (B, N, N, H)  – result out[b,i,j,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    a_ptr, b_ptr, c_ptr,               # base pointers
    B, N, H, K,                        # sizes (K == N)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_k, stride_b_i, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    #  Grid layout
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)                 # block row   (i)
    pid_n = tl.program_id(1)                 # block col   (j)
    pid_bh = tl.program_id(2)                # combined batch*hidden index

    # ------------------------------------------------------------------
    #  Derive batch and hidden indices from the combined id
    # ------------------------------------------------------------------
    batch_idx  = pid_bh // H
    hidden_idx = pid_bh % H

    # ------------------------------------------------------------------
    #  Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    #  Base pointers for this (batch, hidden) slice
    # ------------------------------------------------------------------
    a_base = a_ptr + batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_base = b_ptr + batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_base = c_ptr + batch_idx * stride_c_batch + hidden_idx * stride_c_h

    # ------------------------------------------------------------------
    #  Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over the reduction dimension K in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile  A[i,k] : stride_i (row) , stride_k (col)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile B[k,j] : we read the transposed view,
        # stride_k -> row (k) , stride_i -> col (j)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot‑product accumulation (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    #  Write the result back (store as FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _batched_trimul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched N³ contraction for TriMul.
    left, right : [B, N, N, H] (FP16)
    Returns      : out  [B, N, N, H] (FP16)
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2, "pair dimensions must be square"

    # output tensor
    out = torch.empty_like(left, dtype=torch.float16, device=left.device)

    # strides are expressed in number of elements (not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tunable block sizes – empirically good for N up to 1024 on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_bh = B * H

    _trimul_kernel[
        (grid_m, grid_n, grid_bh)
    ](
        left,
        right,
        out,
        B,
        N,
        H,
        N,                               # K == N
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_k,                      # note: row stride = original k‑dim
        stride_b_i,                      # col stride = original i‑dim (transposed)
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    #  Load model weights
    # ------------------------------------------------------------------
    norm_weight        = weights["norm.weight"]
    norm_bias          = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    #  1️⃣ Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    #  2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]  fp32
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]  fp32

    # ------------------------------------------------------------------
    #  3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]  fp32
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    #  4️⃣ Gating tensors (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]  fp32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]  fp32
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,H]  fp32

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  5️⃣ Heavy N³ contraction – performed in FP16 via Triton
    # ------------------------------------------------------------------
    left_h  = left .to(torch.float16)
    right_h = right.to(torch.float16)

    # result in FP16, shape [B,N,N,H]
    out_fp16 = _batched_trimul_fp16(left_h, right_h)

    # ------------------------------------------------------------------
    #  6️⃣ Second Layer‑norm over hidden dimension (H)
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)   # cast back to fp32 for LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    #  7️⃣ Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    #  8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]  fp32

    return out
```
