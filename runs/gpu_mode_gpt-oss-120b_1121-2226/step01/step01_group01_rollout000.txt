"""
TriMul outgoing kernel (AlphaFold‑3)

The heavy O(N³) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel that fuses the
B·hidden_dim independent matrix multiplications A @ Bᵀ.
All other operations (LayerNorm, linear projections, gating,
final LayerNorm + linear projection) stay in FP32 for
numerical stability.
The kernel works on FP16 data, accumulates in FP32 and
stores the result in FP16, which is then cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   (A: (M,K), B: (K,N))
# One independent matmul per (batch, hidden) pair.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                   # tensor pointers
    M, N, K,                               # matrix dimensions
    stride_am, stride_ak, stride_ab,       # strides of A (batch, M, K)
    stride_bk, stride_bn, stride_bb,       # strides of B (batch, K, N)
    stride_cm, stride_cn, stride_cb,       # strides of C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each batch element:
        C[b, m, n] = Σ_k A[b, m, k] * B[b, k, n]
    where A and B are FP16 tensors, accumulation is done in FP32.
    """
    pid_m = tl.program_id(0)          # tile over M dimension
    pid_n = tl.program_id(1)          # tile over N dimension
    pid_b = tl.program_id(2)          # batch * hidden index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Out‑of‑bounds masks
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # Pointer matrices for the current tile
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # FP32 accumulator (better precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # (BLOCK_K, BLOCK_N)

        # Dot‑product over the K axis
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result (cast back to FP16)
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point called by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack the inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Load model weights (all FP32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]      # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (over last dimension)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)  # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast to hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)      # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy reduction (batched matmul) in FP16
    # --------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # N = sequence length

    # Rearrange tensors so that hidden dimension becomes the batch dimension
    # A: (B, hidden, M=N, K=N)
    # B: (B, hidden, K=N, N=N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Output buffer
    out_mat = torch.empty_like(left_mat)   # FP16

    # --------------------------------------------------------------
    # 5a. Compute strides required by Triton
    # --------------------------------------------------------------
    stride_am = left_mat.stride(1)   # stride over M dimension
    stride_ak = left_mat.stride(2)   # stride over K dimension
    stride_ab = left_mat.stride(0)   # batch stride

    stride_bk = right_mat.stride(1)  # stride over K dimension
    stride_bn = right_mat.stride(2)  # stride over N dimension
    stride_bb = right_mat.stride(0)  # batch stride

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    # --------------------------------------------------------------
    # 5b. Launch the Triton kernel
    # --------------------------------------------------------------
    # Block sizes tuned for FP16 TensorCores on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles over M (seq_len)
        triton.cdiv(N, BLOCK_N),   # tiles over N (seq_len)
        B * hidden_dim,            # one grid entry per (batch, hidden) pair
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,               # optimal for FP16 on H100
        num_stages=3,              # pipeline stages for better L2 reuse
    )

    # --------------------------------------------------------------
    # 5c. Reshape back to (B, N, N, hidden_dim) and cast to FP32
    # --------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate                     # out_gate still FP32
    out = F.linear(out, to_out_weight)       # (B, N, N, dim)

    return out
