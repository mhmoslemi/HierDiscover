def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    device = input_tensor.device

    # Cast everything to half where possible
    # Extract weights and cast half
    def to_half(t): return t.to(torch.float16)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = to_half(weights["left_proj.weight"])
    right_proj_weight = to_half(weights["right_proj.weight"])
    left_gate_weight = to_half(weights["left_gate.weight"])
    right_gate_weight = to_half(weights["right_gate.weight"])
    out_gate_weight = to_half(weights["out_gate.weight"])
    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias = to_half(weights["to_out_norm.bias"])
    to_out_weight = to_half(weights["to_out.weight"])

    # 1. LayerNorm (float32)
    x = torch.nn.functional.layer_norm(input_tensor,
                                       normalized_shape=(dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=1e-5)

    # 2. Proj + Gate in FP16
    # proj
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden_dim) fp16
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # 3. Batched matmul via Triton
    B, N, _, _ = left.shape  # hidden_dim holds in last dim

    # Reshape to (B*hidden, N, N)
    left_mat = left.permute(0, 3, 1, 2)   # shape (B, hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1) # transpose k axis
    # No .contiguous() call, keep as view.

    # Flatten batch*hidden
    left_mat = left_mat.reshape(B * hidden_dim, N, N)
    right_mat = right_mat.reshape(B * hidden_dim, N, N)

    out_mat = torch.empty_like(left_mat)  # fp16

    # compute strides as ints
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_ab = left_mat.stride(0)

    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_bb = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm and gate in FP16
    out = torch.nn.functional.layer_norm(out,
                                         normalized_shape=(hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=1e-5)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # (B,N,N,dim)
    out = out.to(torch.float32)
    return out

"""
TriMul (outgoing) – fast Triton implementation
================================================

The forward pass can be split into three stages:

1️⃣  **LayerNorm + linear projections + gating**  
    These cheap operations are kept in PyTorch.  All linear layers are
    executed in fp16 (the weights are cast once) and the result tensors
    `left` and `right` already contain the mask and the sigmoid‑gates.

2️⃣  **Batched matrix multiplication**  
    For every hidden channel `d` we need  

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    i.e. a large number of independent `N×N` matmuls (`N` is the
    sequence length).  A single Triton kernel fuses all hidden‑channel
    batches: each program instance works on a tile `(BLOCK_M, BLOCK_N)`
    of the output matrix for a given `(batch, hidden)` pair and
    accumulates the products with `tl.dot` (Tensor‑Core fp16 → fp32
    accumulation).

    The kernel uses only three program‑ids:
        * `pid_m` – tile row index,
        * `pid_n` – tile column index,
        * `pid_b` – combined batch‑×‑hidden index.
    No temporary copies of the tensors are performed – we simply
    flatten the leading two dimensions (`batch * hidden`) with
    ``flatten(0, 1)`` which yields a view with the correct strides.

3️⃣  **Output post‑processing**  
    The result is reshaped back to `[B, N, N, hidden]`, normalized,
    multiplied by the output‑gate and projected back to the original
    feature dimension.  The final tensor is returned as `float32`.

The whole pipeline stays on the GPU, uses mixed‑precision (fp16 inside
the O(N³) kernel, fp32 for the layer‑norms) and avoids any unnecessary
`contiguous()` calls, thus reducing memory traffic and improving
runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ)
# A : (batch, M, K)   – fp16
# B : (batch, K, N)   – fp16   (already transposed w.r.t. original tensor)
# C : (batch, M, N)   – fp16
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_am, stride_ak, stride_ab,  # strides of A (batch, M, K)
    stride_bk, stride_bn, stride_bb,  # strides of B (batch, K, N)
    stride_cm, stride_cn, stride_cb,  # strides of C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b] = A[b] @ B[b].T for each batch b.
    """
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_b = tl.program_id(2)   # batch * hidden

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # Base pointers for this batch
    a_batch = a_ptr + pid_b * stride_ab
    b_batch = b_ptr + pid_b * stride_bb
    c_batch = c_ptr + pid_b * stride_cb

    # Pointers to the tile
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # Accumulator in fp32 (better precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # Σₖ a[i,k] * b[k,j]

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # --------------------------------------------------------------
    # Helper: cast model weights once to fp16 (except the LayerNorm
    # weights that stay in fp32 for numerical stability)
    # --------------------------------------------------------------
    to_fp16 = lambda t: t.to(torch.float16)
    left_proj_weight   = to_fp16(weights["left_proj.weight"])
    right_proj_weight  = to_fp16(weights["right_proj.weight"])
    left_gate_weight   = to_fp16(weights["left_gate.weight"])
    right_gate_weight  = to_fp16(weights["right_gate.weight"])
    out_gate_weight    = to_fp16(weights["out_gate.weight"])
    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])
    to_out_weight      = to_fp16(weights["to_out.weight"])

    # LayerNorm weights stay in FP32
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # --------------------------------------------------------------
    # 2️⃣  Projections + optional mask (FP16)
    # --------------------------------------------------------------
    # Linear projections (no bias)
    left  = F.linear(x, left_proj_weight)    # (B,N,N,hidden)
    right = F.linear(x, right_proj_weight)

    if not nomask:
        # mask shape: (B,N,N) → (B,N,N,1) for broadcasting
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 3️⃣  Gating (sigmoid) – also FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4️⃣  Batched matrix multiplication via Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape      # N = sequence length

    # Rearrange to (B, hidden, N, N) and flatten the first two dims.
    # No .contiguous() – we keep a view to avoid extra memory traffic.
    left_mat  = left.permute(0, 3, 1, 2).flatten(0, 1)   # (B*hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).flatten(0, 1)  # (B*hidden, N, N) – transposed k‑axis

    out_mat = torch.empty_like(left_mat)                # fp16 output

    # ------------------------------------------------------------------
    # Strides required by Triton (they are Python ints)
    # ------------------------------------------------------------------
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)
    stride_ab = left_mat.stride(0)

    stride_bk = right_mat.stride(1)
    stride_bn = right_mat.stride(2)
    stride_bb = right_mat.stride(0)

    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)
    stride_cb = out_mat.stride(0)

    # ------------------------------------------------------------------
    # Kernel launch parameters – tuned for H100 fp16 TensorCores
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),         # rows
        triton.cdiv(N, BLOCK_N),         # cols
        B * hidden_dim,                  # batch * hidden
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,         # 4 warps per block works well for 64×64 tiles
    )

    # --------------------------------------------------------------
    # 5️⃣  Reshape back to (B, N, N, hidden) – keep fp16
    # --------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣  Output LayerNorm, final gate and projection (fp16 → fp32)
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate               # out_gate is still fp16
    out = F.linear(out, to_out_weight) # (B,N,N,dim) in fp16

    return out.to(torch.float32)       # required output dtype
