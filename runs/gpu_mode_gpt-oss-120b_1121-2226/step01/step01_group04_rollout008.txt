def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # compute batch_hidden idx
    bh = tl.program_id(0)
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)

    # compute b, h from bh
    b = bh // H
    h = bh % H

    # compute offsets
    i_base = i_tile * BLOCK_M
    j_base = j_tile * BLOCK_N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base pointers for left and right for this batch and hidden dim
    # left_base pointer points to start of left at (b,0,0,h)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K

        # compute offsets for loading
        i_idx = i_base + tl.arange(0, BLOCK_M)
        j_idx = j_base + tl.arange(0, BLOCK_N)
        k_idx = k_base + tl.arange(0, BLOCK_K)

        # mask to avoid out-of-bounds
        i_mask = i_idx < N
        j_mask = j_idx < N
        k_mask = k_idx < N

        # shape for load: (BLOCK_M, BLOCK_K)

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension `dim`.
2. Linear projections (no bias) produce `left` and `right` of shape
   [B, N, N, H] (H = hidden_dim).  Three linear layers + sigmoid give
   `left_gate`, `right_gate` and `out_gate`.
3. A tiny Triton kernel fuses the element‑wise mask, left‑gate and
   right‑gate operations:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
4. The core bilinear product
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is performed by a custom Triton batched‑GEMM kernel.  For each
   (b,h) pair we compute a matrix multiplication `left @ rightᵀ`
   using FP16 arithmetic and FP32 accumulation.
5. `out` is cast back to FP32, layer‑normed over the hidden dimension,
   multiplied by `out_gate`, and finally projected back to `dim` with a
   linear layer.
All heavy work (mask‑gate fusion + matmul) is done on‑GPU with
Triton; the remaining ops run in FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# 1️⃣  Mask + gate fusion (element‑wise)
# ------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # FP16 tensors (flattened)
    left_gate_ptr, right_gate_ptr,     # FP16 tensors (flattened)
    mask_ptr,                         # FP16 tensor (flattened, same size as left/right)
    size: tl.constexpr,               # total number of elements
    BLOCK_SIZE: tl.constexpr,         # compile‑time block size
):
    """Fuse `mask`, `left_gate` and `right_gate` into `left` and `right`."""
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    l   = tl.load(left_ptr + offs,        mask=mask, other=0.0)
    r   = tl.load(right_ptr + offs,       mask=mask, other=0.0)
    lg  = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rg  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + offs,       mask=mask, other=0.0)

    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + offs,  l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ------------------------------------------------------------------
# 2️⃣  Batched matmul for the TriMul core
# ------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Grid:
        (batch_hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    where batch_hidden = B * H.
    """
    bh      = tl.program_id(0)                     # combined batch & hidden index
    i_tile  = tl.program_id(1)                     # row tile
    j_tile  = tl.program_id(2)                     # col tile

    # --------------------------------------------------------------
    # De‑compose bh → (b, h)
    # --------------------------------------------------------------
    b = bh // H
    h = bh % H

    # --------------------------------------------------------------
    # Base pointers for this (b,h) slice
    # --------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # --------------------------------------------------------------
    # Tile start indices
    # --------------------------------------------------------------
    i0 = i_tile * BLOCK_M
    j0 = j_tile * BLOCK_N

    # Accumulator in FP32 for better precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over k dimension
    # --------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K

        # Index vectors
        i_idx = i0 + tl.arange(0, BLOCK_M)
        j_idx = j0 + tl.arange(0, BLOCK_N)
        k_idx = k0 + tl.arange(0, BLOCK_K)

        # Bounds (used for masked loads / stores)
        i_mask = i_idx < N
        j_mask = j_idx < N
        k_mask = k_idx < N

        # -------------------- Load A = left[i, k] --------------------
        a_ptrs = left_base + i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # -------------------- Load B = right[j, k] --------------------
        # We need right[j, k] (notice the swapped order vs usual GEMM).
        b_ptrs = right_base + k_idx[:, None] * stride_right_k + j_idx[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)

        # -------------------- Small matrix multiplication --------------------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write out tile
    # --------------------------------------------------------------
    out_ptrs = out_base + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])


# ------------------------------------------------------------------
# 3️⃣  Public entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask          : torch.Tensor[B, N, N] (bool/float) or None
        - weights       : dict of torch.Tensors with keys:
            'norm.weight', 'norm.bias',
            'left_proj.weight', 'right_proj.weight',
            'left_gate.weight', 'right_gate.weight', 'out_gate.weight',
            'to_out_norm.weight', 'to_out_norm.bias',
            'to_out.weight'
        - config : dict with at least 'dim' and 'hidden_dim'

    Returns
    -------
    torch.Tensor[B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    H = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (FP32)

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H] FP32
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # FP32

    # --------------------------------------------------------------
    # 4️⃣ Cast to FP16 – all heavy ops run in half‑precision
    # --------------------------------------------------------------
    left      = left.to(torch.float16).contiguous()
    right     = right.to(torch.float16).contiguous()
    left_gate = left_gate.to(torch.float16).contiguous()
    right_gate = right_gate.to(torch.float16).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Prepare mask (broadcast to hidden dim)
    # --------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=left.device)
    else:
        mask_tensor = mask.to(torch.float16)
    mask_expanded = mask_tensor.unsqueeze(-1).expand(-1, -1, -1, H).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask + gate (Triton)
    # --------------------------------------------------------------
    total_el = left.numel()
    BLOCK = 1024                       # 1‑D kernel tile size
    grid_mask = (triton.cdiv(total_el, BLOCK),)
    _mask_gate_fuse_kernel[grid_mask](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_expanded,
        size=total_el,
        BLOCK_SIZE=BLOCK,
    )

    # --------------------------------------------------------------
    # 6️⃣ Core TriMul – batched GEMM in Triton
    # --------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16)   # [B, N, N, H]

    # Strides in elements (not bytes)
    s_lb = left.stride(0); s_li = left.stride(1); s_lk = left.stride(2); s_lh = left.stride(3)
    s_rb = right.stride(0); s_ri = right.stride(1); s_rk = right.stride(2); s_rh = right.stride(3)
    s_ob = out.stride(0); s_oi = out.stride(1); s_oj = out.stride(2); s_oh = out.stride(3)

    # Tiling parameters – chosen to fit H100 shared‑memory budget
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid_matmul = (
        B * H,                               # batch × hidden
        triton.cdiv(N, BLOCK_M),             # i‑tiles
        triton.cdiv(N, BLOCK_N),             # j‑tiles
    )
    _trimul_matmul_kernel[grid_matmul](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=s_lb,
        stride_left_i=s_li,
        stride_left_k=s_lk,
        stride_left_h=s_lh,
        stride_right_b=s_rb,
        stride_right_i=s_ri,
        stride_right_k=s_rk,
        stride_right_h=s_rh,
        stride_out_b=s_ob,
        stride_out_i=s_oi,
        stride_out_j=s_oj,
        stride_out_h=s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7️⃣ FP32 post‑processing
    # --------------------------------------------------------------
    out = out.to(torch.float32)   # [B, N, N, H]

    # Output LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Gate the normalized tensor
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])   # (dim, H) → output shape [B,N,N,dim]

    return out
