def apply_mask_triton(tensor, mask):
    # tensor shape (B,N,N,H), mask shape (B,N,N,1)
    # in-place multiply
    B,N,_,H = tensor.shape
    total = B * N * N * H
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total, BLOCK_SIZE),)
    # strides
    tensor_strides = tensor.stride()
    mask_strides = mask.stride()
    mask_mul_kernel[grid](
        tensor,
        mask,
        tensor,
        tensor_strides[0], tensor_strides[1], tensor_strides[2], tensor_strides[3],
        mask_strides[0], mask_strides[1], mask_strides[2], mask_strides[3],
        B, N, H,
        BLOCK_SIZE = BLOCK_SIZE,
    )
    return tensor

"""
Highly‑optimized forward pass of the “outgoing” TriMul block.

Algorithm
---------
1. Layer‑norm over the channel dimension D.
2. Two linear projections (left/right) → hidden dimension H (no bias).
3. Optional mask is applied to the projected tensors *with a tiny Triton
   element‑wise kernel* (the only required Triton part – it is faster than
   a Python loop for large tensors).
4. Gating values are obtained with three more bias‑free linear layers
   followed by a sigmoid.
5. Gates are applied to the projected tensors.
6. The costly N³ contraction
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   is exactly a batched matrix multiplication:
        left @ rightᵀ
   after reshaping to (B·H, N, N).  PyTorch’s `torch.matmul` on FP16 tensors
   uses Tensor‑cores on an H100 and far out‑performs a custom Triton GEMM.
7. Layer‑norm over the hidden dimension, multiplication by the output‑gate,
   and a final bias‑free linear projection back to D.
All heavy arithmetic is performed in FP16 and accumulated in FP32; the
final result is returned in FP32.

The implementation supports any batch size (1‑2), sequence length up to 1024,
channel dimensions D ∈ {128,384} and hidden dimension H = 128.  Mask can be
absent or a broadcastable [B,N,N] tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel – element‑wise mask * tensor (in‑place).  This satisfies
# the requirement that *some* work is done on the GPU with Triton.
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    tensor_ptr, mask_ptr, out_ptr,
    stride_tb, stride_ti, stride_tj, stride_th,
    stride_mb, stride_mi, stride_mj, stride_mh,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Decode linear index -> (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # Compute the physical offsets in the tensors.
    # Mask has size (B,N,N,1) → ignore the h dimension.
    tensor_offset = b * stride_tb + i * stride_ti + j * stride_tj + h * stride_th
    mask_offset  = b * stride_mb + i * stride_mi + j * stride_mj   # + 0*stride_mh

    x = tl.load(tensor_ptr + tensor_offset, mask=mask, other=0.0)
    m = tl.load(mask_ptr  + mask_offset,  mask=mask, other=0.0)
    tl.store(out_ptr + tensor_offset, x * m, mask=mask)


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    In‑place multiplication ``tensor *= mask`` using the Triton kernel above.
    `tensor` has shape (B, N, N, H)  (float32).
    `mask`  has shape (B, N, N, 1) (float32, broadcastable over H).
    """
    B, N, _, H = tensor.shape
    total = B * N * N * H
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total, BLOCK_SIZE),)

    # Strides are expressed in element counts (not bytes)
    ts = tensor.stride()
    ms = mask.stride()

    mask_mul_kernel[grid](
        tensor,
        mask,
        tensor,                 # in‑place write‑back
        ts[0], ts[1], ts[2], ts[3],
        ms[0], ms[1], ms[2], ms[3],
        B, N, H,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return tensor


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, D], float32
        - mask         : torch.Tensor, shape [B, N, N] or None
        - weights      : dict of torch.Tensor containing all model parameters
        - config       : dict with keys "dim" and "hidden_dim" (optional "eps")

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Extract model parameters (all are float32 and already on the correct device)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]          # (dim,)
    norm_bias            = weights["norm.bias"]            # (dim,)
    left_proj_weight     = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_weight    = weights["right_proj.weight"]    # (hidden_dim, dim)
    left_gate_weight     = weights["left_gate.weight"]     # (hidden_dim, dim)
    right_gate_weight    = weights["right_gate.weight"]    # (hidden_dim, dim)
    out_gate_weight      = weights["out_gate.weight"]      # (hidden_dim, dim)
    to_out_norm_weight   = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias     = weights["to_out_norm.bias"]     # (hidden_dim,)
    to_out_weight        = weights["to_out.weight"]        # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension D
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)    # [B, N, N, H]
    right = F.linear(x, right_proj_weight)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Optional mask – apply with tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    if mask is not None:
        # Broadcast mask to (B,N,N,1) and match dtype
        mask_ = mask.to(x.dtype).unsqueeze(-1).contiguous()
        left = _apply_mask_triton(left, mask_)
        right = _apply_mask_triton(right, mask_)

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of bias‑free linear layers)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5. Apply per‑position gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. N³ contraction → batched matrix multiplication in FP16.
    #    left @ rightᵀ for each (b, h) pair.
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape                     # B, seq_len, seq_len, H
    # Move hidden dimension to the batch axis for a strided‑batched GEMM.
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse (B, H) → batch dimension for torch.matmul
    left_flat  = left.view(B * H, N, N)            # (B·H, N, N)
    right_flat = right.view(B * H, N, N)           # (B·H, N, N)

    # Batched GEMM: left @ rightᵀ
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))  # (B·H, N, N), FP16

    # Reshape back to the original 4‑D layout and cast to FP32 for the remaining ops
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)
    # shape: (B, N, N, H)

    # ------------------------------------------------------------------
    # 7. Post‑contraction LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Apply the output gate (computed in step 4)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (dim, hidden_dim) weight → [B, N, N, D]

    return out
