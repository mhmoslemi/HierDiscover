def custom_kernel(data):
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # load weight tensors; cast to half when applicable:
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # Cast weight halves:
    left_proj_weight_h = left_proj_weight.to(torch.float16)
    right_proj_weight_h = right_proj_weight.to(torch.float16)
    left_gate_weight_h = left_gate_weight.to(torch.float16)
    right_gate_weight_h = right_gate_weight.to(torch.float16)
    out_gate_weight_h = out_gate_weight.to(torch.float16)
    # First LayerNorm (fp32)
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)
    # Cast to half for projection
    x_h = x.to(torch.float16)
    # Projections
    left = F.linear(x_h, left_proj_weight_h)   # (B,N,N,H) half
    right = F.linear(x_h, right_proj_weight_h)
    # optional mask
    if mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h
    # Gates: compute in half then sigmoid
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_weight_h))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_weight_h))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_weight_h))
    left = left * left_gate
    right = right * right_gate
    # Convert to half already
    # Triton matmul
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    B, H, N, _ = left_perm.shape
    A = left_perm.view(B * H, N, N)
    B_mat = right_perm.view(B * H, N, N)
    out_mat = _batched_matmul_fp16(A, B_mat)   # (B*H,N,N) half
    # Reshape back
    out_perm = out_mat.view(B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # (B,N,N,H) half
    # Second LayerNorm over hidden_dim (fp32)
    out_fp32 = out.to(torch.float32)
    out_fp32 = F.layer_norm(out_fp32, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm over the last dimension (`dim`) of the 4‑D input.
2. Linear projections → `left`, `right` (shape [B,N,N,H]).
3. Optional mask (broadcast over H).
4. Gating: three linear → sigmoid → element‑wise multiplication
   (`left_gate`, `right_gate`, `out_gate`).
5. Heavy N³ contraction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (A @ B) with
   A = left[b,h,i,k] , B = right[b,h,k,j].
   The multiplication is performed in FP16 with FP32 accumulation
   by a custom Triton kernel (`_batched_matmul_kernel`).
6. Layer‑norm over the hidden dimension H.
7. Apply `out_gate`.
8. Final linear projection back to the original `dim`.

All non‑cubic operations are done with PyTorch (FP32 for numerical
stability).  The cubic contraction is the only part that uses Triton.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes
    stride_ab, stride_am, stride_ak,         # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,         # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,         # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    #   Determine (batch, block_m, block_n) for this program id
    # --------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # --------------------------------------------------------------
    #   Compute offsets for the current tile
    # --------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (higher precision than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #   Loop over K dimension in tiles of size BLOCK_K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for A (batch, M, K)
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # pointers for B (batch, K, N)
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    #   Write back C
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision matrix multiplication using the Triton kernel above.

    Args:
        a: Tensor (B, M, K)  dtype=torch.float16, contiguous.
        b: Tensor (B, K, N)  dtype=torch.float16, contiguous.

    Returns:
        Tensor (B, M, N) dtype=torch.float16.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Tunable block sizes – work well for N up to 1024 on an H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_m * num_n,)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None, shape [B, N, N] (float32/bool)
        - weights      : dict of torch.Tensor containing the model weights
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Load model weights (all are FP32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim) – keep FP32 for stability
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim], dtype=float32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) → left, right  (still FP32)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float32 and has a trailing singleton dim for broadcasting
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating networks (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Heavy N³ contraction – Triton batched matmul in FP16
    # --------------------------------------------------------------
    # Cast to half‑precision before the massive GEMM (memory‑bandwidth saving)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Bring the hidden dimension to the batch axis:
    # left_h : [B, N, N, H] → [B, H, N, N] (i,k)
    # right_h: [B, N, N, H] → [B, H, N, N] (k,j)  (note the transpose on the two N axes)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape
    A = left_perm.view(B * H, N, N)          # (B*H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)     # (B*H, K=N, N)

    # Batched matrix multiplication
    C = _batched_matmul_fp16(A, B_mat)      # (B*H, N, N)  FP16

    # Restore original layout: (B, N, N, H)
    out_perm = C.view(B, H, N, N)           # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] (FP16)

    # --------------------------------------------------------------
    # 6️⃣ Layer‑norm over hidden dimension (H) – back to FP32
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate (converted to FP32)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)      # [B, N, N, dim], dtype=float32

    return out
