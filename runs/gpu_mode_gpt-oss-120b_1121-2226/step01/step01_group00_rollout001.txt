"""
TriMul (outgoing) – Triton‑accelerated forward pass

Algorithm
---------
1️⃣ Input LayerNorm (dim) – kept in FP32 for numerical stability.
2️⃣ Cast to FP16 and apply two linear projections (left/right) → (B,N,N,H).
3️⃣ Optional binary mask multiplication (broadcasted over H).
4️⃣ Gating: sigmoid(linear(x)) per side, then element‑wise scale left/right.
5️⃣ Core “TriMul”:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. a batch of H independent N×N matrix‑multiplications.
   Implemented with a custom Triton kernel that computes a batched
   GEMM C = A @ B (where B is already transposed) for all
   (batch * hidden) matrices in FP16, accumulating in FP32.
6️⃣ Output LayerNorm over the hidden dimension, gating with out_gate,
   and final linear projection back to dim (FP32 output).

All heavy arithmetic (steps 2‑5) runs in FP16; the final result is FP32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B  (A: M×K, B: K×N)
# Batch dimension = batch * hidden_dim.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix sizes (all = seq_len)
    stride_am, stride_ak,        # A: row and col strides
    stride_bk, stride_bn,        # B: row‑stride (k) and col‑stride (n)
    stride_cm, stride_cn,        # C: row and col strides
    stride_ab, stride_bb, stride_cb,   # batch stride for A, B, C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Program IDs
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile column
    pid_b = tl.program_id(2)   # batch (B * hidden)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the selected batch element
    a_batch = A + pid_b * stride_ab
    b_batch = B + pid_b * stride_bb
    c_batch = C + pid_b * stride_cb

    # Accumulator (FP32 for safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (blocked)
    # --------------------------------------------------------------
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # Load B tile (K x N)
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Compute partial product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Write result back (store FP16)
    # --------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Public entry point for the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor or None, shape [B, N, N]
            - weights_dict : dict of torch.Tensors containing all linear /
                             layer‑norm parameters.
            - config_dict  : {"dim": int, "hidden_dim": int,
                              "nomask": bool, ...}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0. Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep in FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (left / right)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcasted over H)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_fp16.dtype)    # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Gating (sigmoid of linear(x))
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core TriMul (batched GEMM via Triton)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B = batch size, N = seq_len

    # Rearrange to (batch, hidden, N, N) where:
    #   left  : rows = i, cols = k
    #   right : rows = k, cols = j   (k↔j transpose)
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) = (B, H, K, N)

    # Collapse batch and hidden dim → effective batch for GEMM
    left_flat  = left_mat.view(B * hidden_dim, N, N)   # (B*H, M, K)
    right_flat = right_mat.view(B * hidden_dim, N, N)  # (B*H, K, N)

    # Output allocation
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # (B*H, M, N)

    # Strides (row‑major)
    stride_am = left_flat.stride(1)   # = N  (row stride, M dimension)
    stride_ak = left_flat.stride(2)   # = 1  (col stride, K dimension)

    stride_bk = right_flat.stride(1)  # = N  (row stride, K dimension)
    stride_bn = right_flat.stride(2)  # = 1  (col stride, N dimension)

    stride_cm = out_flat.stride(1)    # = N
    stride_cn = out_flat.stride(2)    # = 1

    stride_ab = left_flat.stride(0)   # = N * N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # Tile sizes – chosen for H100 (128‑bit registers, high occupancy)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim   # batch * hidden

    _batched_gemm_kernel[(grid_x, grid_y, grid_z)](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                        # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)
    out = out.to(torch.float32)   # back to FP32 for remaining ops

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B, N, N, H]

    out = out * out_gate  # gating (broadcast over hidden dim)

    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]

    return out
