"""
TriMul (outgoing) – highly‑optimized forward pass.

Algorithm
---------
1.   Cast the pair representation to float16 and apply LayerNorm (FP16).
2.   Fuse the linear projection and the gate linear layer:
        [proj | gate] = Linear(x)               # shape (..., 2*H)
        proj = proj[..., :H] ; gate = sigmoid(gate[..., H:])
     This halves the number of GEMM calls.
3.   Apply the pair mask and the two element‑wise gates with a tiny
     Triton kernel (in‑place, no temporary buffers).
4.   Core bilinear update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     This is exactly a batched matrix multiplication of shape
        (B·H, N, N) @ (B·H, N, N)^T .
     We use `torch.bmm` (cuBLAS) on FP16 tensors – the fastest
     implementation on H100.
5.   Layer‑norm over the hidden dimension, gating by `out_gate`,
     and a final linear projection back to `dim`.
6.   Cast the result back to float32 for downstream layers.

Only the mask‑gate fusion is performed with Triton (as required);
all heavy linear algebra stays on cuBLAS which is far superior to a
hand‑written Triton GEMM for the N³ work‑load.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask & gate fusion (left/right tensors).
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                 # FP16 tensors (flattened)
    left_gate_ptr, right_gate_ptr,       # FP16 tensors (flattened)
    mask_ptr,                            # FP16 tensor (flattened)
    size,                                # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load (masked loads return 0.0 for out‑of‑bounds threads)
    l  = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr      + offs, mask=mask, other=0.0)

    # Fuse mask and sigmoid‑gates
    l = l * lg * m
    r = r * rg * m

    # Store back in‑place
    tl.store(left_ptr      + offs, l, mask=mask)
    tl.store(right_ptr     + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor[B, N, N, dim] (float32)
            - mask         : torch.Tensor[B, N, N] (bool/float) or None
            - weights      : dict of torch.Tensor (model parameters)
            - config       : dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # H

    ######################################################################
    # 0. Helper to move a weight to FP16 on‑device only once
    ######################################################################
    def to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    ######################################################################
    # 1. Input LayerNorm (FP16)
    ######################################################################
    # Cast input once
    x = input_tensor.to(dtype=torch.float16)

    norm_weight = to_fp16(weights["norm.weight"])
    norm_bias   = to_fp16(weights["norm.bias"])
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B,N,N,dim]  FP16

    ######################################################################
    # 2. Fused linear + gate for LEFT side
    ######################################################################
    left_proj_weight  = to_fp16(weights["left_proj.weight"])   # (H, dim)
    left_gate_weight  = to_fp16(weights["left_gate.weight"])   # (H, dim)
    left_fused_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0)  # (2H, dim)

    # 2b. Fused linear + gate for RIGHT side
    right_proj_weight = to_fp16(weights["right_proj.weight"])
    right_gate_weight = to_fp16(weights["right_gate.weight"])
    right_fused_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0)  # (2H, dim)

    # Flatten spatial dimensions for the GEMM
    x_flat = x.reshape(-1, dim)                     # (B·N·N, dim)

    # Left:  (B·N·N, 2H) → split into proj & gate
    left_and_gate = F.linear(x_flat, left_fused_weight, bias=None)   # FP16
    left   = left_and_gate[:, :hidden_dim]                         # (B·N·N, H)
    left_g = torch.sigmoid(left_and_gate[:, hidden_dim:])          # (B·N·N, H)

    # Right: (B·N·N, 2H) → split
    right_and_gate = F.linear(x_flat, right_fused_weight, bias=None)
    right   = right_and_gate[:, :hidden_dim]
    right_g = torch.sigmoid(right_and_gate[:, hidden_dim:])

    # Out‑gate (separate linear)
    out_gate_weight = to_fp16(weights["out_gate.weight"])
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight, bias=None))

    # Restore the 4‑D layout (still half‑precision)
    left    = left.view(bs, N, N, hidden_dim)
    left_g  = left_g.view(bs, N, N, hidden_dim)
    right   = right.view(bs, N, N, hidden_dim)
    right_g = right_g.view(bs, N, N, hidden_dim)
    out_gate = out_gate.view(bs, N, N, hidden_dim)

    ######################################################################
    # 3. Fuse mask + gates with a tiny Triton kernel (in‑place)
    ######################################################################
    if mask is None:
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype=torch.float16, device=device)
    # Broadcast to hidden dim
    mask_expand = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # Flatten everything for the 1‑D kernel
    total_elements = left.numel()
    BLOCK = 8192                       # larger block reduces launch overhead
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_g,
        right_gate_ptr=right_g,
        mask_ptr=mask_expand,
        size=total_elements,
        BLOCK_SIZE=BLOCK,
    )
    # At this point `left` and `right` already contain mask‑gated values.

    ######################################################################
    # 4. Core TriMul: batched matrix multiplication (FP16)
    ######################################################################
    # (B, N, N, H) -> (B·H, N, N)  (contiguous for cuBLAS)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)

    # Batched GEMM:   (B·H, N, N) @ (B·H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to (B, N, N, H)
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)   # FP16

    ######################################################################
    # 5. Output LayerNorm, out‑gate, final projection
    ######################################################################
    to_out_norm_weight = to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias   = to_fp16(weights["to_out_norm.bias"])
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )   # FP16

    out = out * out_gate                     # gating, still FP16

    to_out_weight = to_fp16(weights["to_out.weight"])   # (dim, H)
    out = F.linear(out, to_out_weight)        # (B,N,N,dim)  FP16

    # Cast back to float32 as required by downstream code
    return out.to(dtype=torch.float32)
