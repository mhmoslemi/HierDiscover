"""
TriMul (outgoing) – fused, fp16 Triton implementation.

Algorithm
---------
1. Input LayerNorm (float‑32) → cast to fp16.
2. Fuse the five linear projections (left, right, left‑gate, right‑gate,
   out‑gate) into a single `torch.nn.functional.linear` using a concatenated
   weight matrix (5·H × dim).  All operations stay in FP16 to exploit
   Tensor‑Cores.
3. Apply the three sigmoid gates.
4. Fuse the element‑wise mask and gate multiplications for *left* and *right*
   with a small 1‑D Triton kernel.  The mask is broadcast over the hidden
   dimension on‑the‑fly (no materialised expansion).
5. Core TriMul: for each hidden slice `h` compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply (`torch.bmm`) on FP16 tensors.
6. Apply a second LayerNorm over the hidden dimension (still in FP16),
   multiply by the `out_gate`, and finish with a linear projection back to
   `dim`.
7. Return the result in float‑32.

All heavy element‑wise work (mask + gate) is performed in Triton, while
the large matrix multiply leverages cuBLAS (via `torch.bmm`).  The whole
pipeline stays in half‑precision except for the two LayerNorms (which
operate on FP16 tensors) and the final cast back to float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate multiplications for `left` and `right`.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    left_gate_ptr,          # fp16 *
    right_gate_ptr,         # fp16 *
    mask_ptr,               # fp16 *
    size,                   # total #elements (int64)
    hidden_dim,             # hidden dimension H (int)
    BLOCK_SIZE: tl.constexpr,
):
    """
    For each element (b,i,k,h) compute:
        left  = left  * left_gate  * mask[b,i,k]
        right = right * right_gate * mask[b,i,k]
    The mask is a 3‑D tensor [B, N, N]; we locate the corresponding
    mask entry by integer‑dividing the flat index by `hidden_dim`.
    """
    pid = tl.program_id(axis=0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # Load values (oob threads get 0.0)
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g   = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    # mask index = flat_index // H  (broadcast over the hidden dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # Fuse the three element‑wise ops
    left  = left  * left_g  * m
    right = right * right_g * m

    # Store results back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    # 5. Core TriMul – batched GEMM per hidden slice
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) – left & right are now ready for matmul
    left_reshape  = left.permute(0, 3, 1, 2).reshape(bs * H, N, N)   # (B·H,N,N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(bs * H, N, N)

    # Batched matmul: left @ rightᵀ   (fp16, tensor‑cores)
    out_reshape = torch.bmm(left_reshape,
                             right_reshape.transpose(1, 2))      # (B·H,N,N)

    # Back to [B, N, N, H]
    out = out_reshape.view(bs, H, N, N).permute(0, 2, 3, 1)          # fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # fp16

    # Apply the out‑gate (already sigmoid’ed, fp16)
    out = out * out_gate

    # Final linear back to `dim` (fp16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)
    out = F.linear(out, to_out_w)                         # fp16

    # ------------------------------------------------------------------
    # 7. Cast back to float32 (model output type)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
