"""
TriMul (outgoing) – Triton‑accelerated forward pass

The implementation follows the AlphaFold‑3 reference:

1️⃣ Layer‑norm on the input tensor (last dimension = dim).  
2️⃣ Linear projections → left/right tensors of shape [B,N,N,H] (H = hidden_dim).  
3️⃣ Optional pair‑wise mask multiplication (broadcast on H).  
4️⃣ Gating: sigmoid(linear(x)) applied element‑wise to left/right (and a final out‑gate).  
5️⃣ Core “TriMul” kernel: for every hidden channel d,
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
   This is a batch of H independent N×N × N×Nᵀ matrix‑multiplications.
   Implemented with a custom Triton kernel that computes C = A @ Bᵀ for
   the whole (B·H) batch, using FP16 arithmetic and FP32 accumulation.
6️⃣ Layer‑norm on the hidden dimension, multiplication with out_gate,
   and a final linear projection back to dim.

All heavy arithmetic (steps 2‑5) is performed in FP16; the final result is
returned in float32.  The Triton kernel can be tuned automatically for any
sequence length N up to 1024 and hidden dimension up to 768.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (C = A @ Bᵀ) for the TriMul core.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                     # pointers to matrices (FP16)
    M, N, K,                     # matrix dimensions (all = seq_len)
    stride_am, stride_ak,        # A: row‑stride, col‑stride
    stride_bk, stride_bn,        # B: row‑stride (k), col‑stride (n)
    stride_cm, stride_cn,        # C: row‑stride, col‑stride
    stride_ab, stride_bb, stride_cb,   # batch strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr):
    """
    Compute   C[b] = A[b] @ B[b]ᵀ    for every batch element b.
    A : (batch, M, K)   (row‑major)
    B : (batch, K, N)   (row‑major, already transposed w.r.t. the original right tensor)
    C : (batch, M, N)
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_b = tl.program_id(2)          # batch / hidden index

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch element
    a_batch = A + pid_b * stride_ab
    b_batch = B + pid_b * stride_bb
    c_batch = C + pid_b * stride_cb

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks
    # --------------------------------------------------------------
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)

        # Load B tile (K × N) – B is already the transposed matrix
        b_ptrs = b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        # Matrix multiplication for this tile
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write result back (store in FP16)
    # --------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [B, N, N, dim]  (float32)
            - mask_tensor  : [B, N, N]      (float32) or None
            - weights_dict : mapping from layer‑name to torch.Tensor
            - config_dict  : {"dim": int,
                              "hidden_dim": int,
                              "nomask": bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (keep it in FP32 for precision)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                   # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the bulk of the work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (left / right) – no bias
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(device, dtype=torch.float16)
    right_proj_w = weights["right_proj.weight"].to(device, dtype=torch.float16)

    left  = torch.nn.functional.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.to(device, dtype=x_fp16.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of a linear projection of the *original* normed input)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(device, dtype=torch.float16)
    right_gate_w = weights["right_gate.weight"].to(device, dtype=torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(device, dtype=torch.float16)

    left_gate  = torch.nn.functional.linear(x_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_fp16, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul – batched GEMM (C = A @ Bᵀ) via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                # B = batch size, N = seq_len
    # ------------------------------------------------------------------
    # Rearrange tensors so the hidden dimension becomes the batch dimension
    #   left : (B, N, N, H) → (B, H, N, N)  (rows = i, cols = k)
    #   right: (B, N, N, H) → (B, H, N, N)  (rows = k, cols = j) i.e. transposed
    # ------------------------------------------------------------------
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) → (B, H, K, N)

    # Flatten the first two dimensions to obtain a 3‑D tensor (batch, M, K) / (batch, K, N)
    left_flat  = left_mat.view(B * hidden_dim, N, N)   # A : (batch, M=N, K=N)
    right_flat = right_mat.view(B * hidden_dim, N, N)  # B : (batch, K=N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)  # C

    # --------------------------------------------------------------
    # Strides (row‑major layout)
    # --------------------------------------------------------------
    stride_am = left_flat.stride(1)      # = K (= N)
    stride_ak = left_flat.stride(2)      # = 1
    stride_bk = right_flat.stride(1)     # = N (row stride for K dimension)
    stride_bn = right_flat.stride(2)     # = 1
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    stride_ab = left_flat.stride(0)      # batch stride = N * N
    stride_bb = right_flat.stride(0)
    stride_cb = out_flat.stride(0)

    # --------------------------------------------------------------
    # Choose tile sizes.  64×64 is a good compromise for all tested N.
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim                     # one program per (batch, hidden)

    _batched_gemm_kernel[(grid_x, grid_y, grid_z)](
        left_flat,                         # A pointer
        right_flat,                        # B (already transposed)
        out_flat,                          # C pointer
        N, N, N,                           # M, N, K
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        stride_ab, stride_bb, stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Reshape back to the original layout [B, N, N, H]
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]
    out = out.to(torch.float32)                                   # back to FP32 for LN

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + out‑gate + final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                   # [B, N, N, H]  (fp32)

    # out_gate was computed in FP16; cast to FP32 for the multiplication
    out = out * out_gate.to(torch.float32)

    to_out_w = weights["to_out.weight"].to(device)              # (dim, H)
    out = torch.nn.functional.linear(out, to_out_w)            # [B, N, N, dim] (fp32)

    return out
