"""
TriMul (outgoing) – fast Triton implementation

The module receives a 4‑D tensor X ∈ ℝ^{B×N×N×D} and an optional mask M ∈
{0,1}^{B×N×N}.  The forward pass consists of:

1. LayerNorm over the channel dimension D.
2. Linear projections to a hidden dimension H (no bias):  left = X·W_l,
   right = X·W_r.
3. Three gates (sigmoid of linear projections): left_gate, right_gate,
   out_gate.
4. Optional masking of left/right before gating.
5. Element‑wise gating:
        left  = left  * left_gate
        right = right * right_gate
6. Heavy N³ contraction (outgoing TriMul):
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batch of H independent matrix multiplications
   left_d @ right_dᵀ.  The contraction is carried out in half‑precision
   with a custom Triton kernel that implements a double‑buffered
   tiled GEMM (fp16 → fp32 accumulation → fp16 write‑back).
7. LayerNorm over the hidden dimension H, multiplication by out_gate,
   and a final linear projection back to D.

Only the N³ contraction is performed on the GPU via Triton; the
remaining operations use regular PyTorch kernels (which are already
highly optimized).  The kernel is written to be generic w.r.t. the
sequence length N and the (batched) hidden dimension B·H, and automatically
chooses tiling parameters that fit into the H100 shared‑memory budget.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from functools import lru_cache


# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM (A @ Bᵀ)
#   A : (BATCH, M, K)   – left tensor after gating (fp16)
#   B : (BATCH, K, N)   – right tensor after gating, already transposed (fp16)
#   C : (BATCH, M, N)   – result (fp16, accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bk, stride_bj,
    stride_cb, stride_ci, stride_cj,
    BATCH: tl.constexpr,
    M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)               # which (batch, hidden) slice
    pid_m   = tl.program_id(1)                # tile row
    pid_n   = tl.program_id(2)                # tile col

    # --------------------- tile start indices -------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = tl.arange(0, BLOCK_K) < K

    # --------------------- base pointers for this batch ---------------
    a_base = A_ptr + batch_id * stride_ab
    b_base = B_ptr + batch_id * stride_bb
    c_base = C_ptr + batch_id * stride_cb

    # --------------------- pointers to the current tile ---------------
    a_ptrs = a_base + (offs_m[:, None] * stride_ai) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_base + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bj)

    # --------------------- accumulator -------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------- first load (double buffer) -----------------
    a = tl.load(
        a_ptrs,
        mask=mask_m[:, None] & mask_k[None, :],
        other=0.0,
        cache_modifier=tl.constexpr('ca')
    )
    b = tl.load(
        b_ptrs,
        mask=mask_k[:, None] & mask_n[None, :],
        other=0.0,
        cache_modifier=tl.constexpr('ca')
    )

    # number of whole K‑tiles
    num_tiles = tl.cdiv(K, BLOCK_K)

    for tile_idx in range(num_tiles):
        # accumulate product of current tile
        acc += tl.dot(a, b)

        # advance pointers for the next tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

        # pre‑fetch next tile unless we are at the last one
        if tile_idx + 1 < num_tiles:
            a = tl.load(
                a_ptrs,
                mask=mask_m[:, None] & mask_k[None, :],
                other=0.0,
                cache_modifier=tl.constexpr('ca')
            )
            b = tl.load(
                b_ptrs,
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0,
                cache_modifier=tl.constexpr('ca')
            )

    # --------------------- write‑back ---------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_base + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper to pick tiling parameters that fit well for a given N.
# ----------------------------------------------------------------------
def _choose_tiles(N: int):
    """
    Very simple heuristic: use larger tiles for larger N.
    """
    if N <= 256:
        return 64, 64, 64
    elif N <= 512:
        return 128, 128, 64
    else:   # N up to 1024 in the test suite
        return 256, 256, 64


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of weight tensors
        - config       : dict with keys "dim", "hidden_dim", optionally "eps"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments (mask may be omitted)
    # ------------------------------------------------------------------
    if len(data) == 3:
        input_tensor, weights, config = data
        mask = None
    else:
        input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Extract model weights
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]          # (dim,)
    norm_bias            = weights["norm.bias"]            # (dim,)
    left_proj_weight     = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_weight    = weights["right_proj.weight"]    # (hidden_dim, dim)
    left_gate_weight     = weights["left_gate.weight"]     # (hidden_dim, dim)
    right_gate_weight    = weights["right_gate.weight"]    # (hidden_dim, dim)
    out_gate_weight      = weights["out_gate.weight"]      # (hidden_dim, dim)
    to_out_norm_weight   = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias     = weights["to_out_norm.bias"]     # (hidden_dim,)
    to_out_weight        = weights["to_out.weight"]        # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over the channel dimension D)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)    # [B, N, N, H]
    right = F.linear(x, right_proj_weight)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5. Apply per‑position gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Heavy N³ contraction – batched GEMM in half precision
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape                    # B, seq_len, seq_len, hidden_dim

    # Rearrange tensors so that the hidden dimension becomes part of the batch
    # (B, H, N, N) → (B*H, N, N).  Right tensor must be transposed on the
    # two sequence axes to implement the “… i k d, … j k d -> … i j d”
    # contraction as a matrix multiplication left @ rightᵀ.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, i)

    batch_hidden = B * H
    left_reshaped  = left_perm.view(batch_hidden, N, N)   # (B*H, M=N, K=N)
    right_reshaped = right_perm.view(batch_hidden, N, N)  # (B*H, K=N, N)

    # Output buffer (fp16, accumulator will be fp32)
    out_reshaped = torch.empty_like(left_reshaped, dtype=torch.float16)

    # Choose tile sizes based on N to keep shared memory < 64 KB per block
    BLOCK_M, BLOCK_N, BLOCK_K = _choose_tiles(N)

    # Grid: (batch, block_rows, block_cols)
    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    batched_matmul_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        left_reshaped.stride(0),   # stride over batch*hidden
        left_reshaped.stride(1),   # stride over rows (M)
        left_reshaped.stride(2),   # stride over cols (K)
        right_reshaped.stride(0),  # stride over batch*hidden
        right_reshaped.stride(1),  # stride over rows (K)
        right_reshaped.stride(2),  # stride over cols (N)
        out_reshaped.stride(0),    # stride over batch*hidden
        out_reshaped.stride(1),    # stride over rows (M)
        out_reshaped.stride(2),    # stride over cols (N)
        BATCH=batch_hidden,
        M=N, N=N, K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and promote to float32 for the rest
    out = (
        out_reshaped.view(B, H, N, N)
        .permute(0, 2, 3, 1)          # (B, N, N, H)
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 7. Post‑contraction LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Apply the output gate (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)       # (dim, hidden_dim) → [B, N, N, D]

    return out
