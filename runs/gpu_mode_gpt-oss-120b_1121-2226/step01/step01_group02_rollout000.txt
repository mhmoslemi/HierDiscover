"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the AlphaFold‑3 “outgoing” TriMul pipeline:
1. Layer‑norm over the input channel dimension (dim).
2. Linear projections → left / right (hidden_dim) and three sigmoid gates.
3. Optional mask application and element‑wise gating.
4. Core O(N³) operation:
        out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
   which is a batched matrix‑multiply L @ Rᵀ for every hidden channel.
   We flatten the hidden dimension into the batch axis and run a custom
   Triton kernel that computes many independent matmuls at once.
5. Layer‑norm over the hidden dimension, multiplication by the output gate,
   and a final linear projection back to *dim*.

The heavy part (step 4) is implemented in the `_batched_matmul_kernel`
with a 128×128×32 tiling (BLOCK_M/N/K).  FP16 tensors are used for the
matrix multiply while the accumulator is FP32, which gives Tensor‑Core
throughput on an H100.  All other operations remain in PyTorch for
readability, but the kernel satisfies the requirement that “part of the
operations” be written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
# (no transpose on B – we multiply rows of B with rows of A)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                 # tensor pointers
    M, N, K,                             # matrix sizes (int64)
    stride_ah, stride_am, stride_ak,     # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,     # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,     # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul kernel.
    C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]  (FP16 × FP16 → FP32)
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_b = tl.program_id(2)          # batch (hidden‑channel) index

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Starting pointers for A and B (k‑tile starts at 0)
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop (tiling)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K) FP16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N) FP16

        # Tensor‑core dot (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

        # Advance pointers for the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(A: torch.Tensor,
                           B: torch.Tensor,
                           BLOCK_M: int = 128,
                           BLOCK_N: int = 128,
                           BLOCK_K: int = 32) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    A : (batch, M, K)   – FP16
    B : (batch, N, K)   – FP16
    Returns C : (batch, M, N) – FP32
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using a Triton‑based batched matmul.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:          torch.Tensor [B, N, N] (float32 or bool)
            - weights:       dict containing all required weight tensors
            - config:        dict with keys:
                * "dim"          – input feature dimension
                * "hidden_dim"   – hidden projection dimension
                * "nomask"       – bool, True if mask should be ignored

    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – stay in float32 for now
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])       # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool; convert to the same dtype as the projections
        m = mask.to(dtype).unsqueeze(-1)                 # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) – also float32
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates (element‑wise)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core TriMul – batched matmul on FP16 tensors
    # ------------------------------------------------------------------
    # Cast to half‑precision for the compute‑heavy part
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    B, N, _, H = left_f16.shape           # B, seq_len, seq_len, hidden_dim

    # Reshape to (B*H, N, N) to treat each hidden channel as an independent batch
    # (permute → (B, H, N, N) → view)
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul: out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
    out_mat = _batched_matmul_triton(left_mat, right_mat,
                                     BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # Restore the original 4‑D shape [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, H, N, N)               # (B, hidden, N, N)
        .permute(0, 2, 3, 1)                  # (B, N, N, hidden)
        .contiguous()
    )   # FP32 (accumulator was FP32)

    # ------------------------------------------------------------------
    # 7️⃣ Output layer‑norm, output‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                       # element‑wise gating

    # Final projection back to dim
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
