"""
TriMul (outgoing) – fast forward pass.

The N³ contraction  
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  
is equivalent to a batched matrix multiply for each hidden channel:
    left   (B,N,N,H) → (B·H, N, N)
    right  (B,N,N,H) → (B·H, N, N) after swapping the last‑two axes.
We perform this massive GEMM with PyTorch’s fused `torch.bmm` (cuBLAS) which
is heavily tuned on H100 and runs in mixed‑precision (FP16 inputs, FP32
accumulation).  

A tiny Triton kernel is used to fuse the optional mask and the per‑position
gates (`left_gate`, `right_gate`).  The kernel works on the flattened
FP16 tensors, applying  

    left  = left  * mask * left_gate  
    right = right * mask * right_gate  

element‑wise in‑place.  All other linear layers, LayerNorms and the final
projection are done with PyTorch primitives.

The implementation supports:
* batch size B ∈ {1,2}
* sequence length N up to 1024
* hidden dimension H (called `hidden_dim`) up to 384
* optional mask (shape B×N×N)
* mixed‑precision arithmetic (FP16 for the heavy contraction, FP32 for
  normalisation and final projection)

The result is returned as a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr,               # fp16*   left tensor (flattened)
    right_ptr,              # fp16*   right tensor (flattened)
    mask_ptr,               # fp16*   mask tensor (flattened, size = total / hidden_dim)
    left_gate_ptr,          # fp16*   left gate (flattened)
    right_gate_ptr,         # fp16*   right gate (flattened)
    total_elems,            # i32     total number of elements in left/right (B*N*N*H)
    hidden_dim,             # i32     hidden dimension H
    BLOCK_SIZE: tl.constexpr
):
    """
    Apply mask and per‑position gates in‑place.
    The tensors are contiguous 1‑D views; each group of `hidden_dim` elements
    shares the same mask element (broadcast over the hidden channel).
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    # Guard against OOB threads
    mask = offs < total_elems

    # Load data
    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # mask index = offs // hidden_dim   (integer division)
    mask_index = offs // hidden_dim
    # Number of mask elements = total_elems // hidden_dim
    mask_valid = mask_index < (total_elems // hidden_dim)
    m = tl.load(mask_ptr + mask_index, mask=mask_valid, other=tl.float16(0.0))

    # Fuse mask and gates
    l = l * m * lg
    r = r * m * rg

    # Store results back in‑place
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask:         torch.Tensor of shape [B, N, N] or None
            - weights:      dict of weight tensors (see docstring)
            - config:       dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Linear weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over channel dimension D
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias=False)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gating values (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Apply mask and per‑position gates.
    #    If a mask is present we fuse mask+gate with a tiny Triton kernel.
    #    Otherwise we just multiply with the gates.
    # ------------------------------------------------------------------
    if mask is not None:
        # Cast everything to fp16 (half precision)
        left = left.to(torch.float16)
        right = right.to(torch.float16)
        left_gate = left_gate.to(torch.float16)
        right_gate = right_gate.to(torch.float16)
        mask_fp16 = mask.to(torch.float16).contiguous()

        total = left.numel()                     # B * N * N * H
        BLOCK_SIZE = 16384                       # enough threads per block
        grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE,)

        # Launch kernel – in‑place modification of left/right
        _mask_gate_kernel[grid](
            left,
            right,
            mask_fp16.view(-1),      # flatten for 1‑D indexing
            left_gate,
            right_gate,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # After the kernel left/right already contain mask * gate * proj.
    else:
        # No mask → simple element‑wise multiplication
        left = (left * left_gate).to(torch.float16)
        right = (right * right_gate).to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Heavy N³ contraction via batched half‑precision GEMM (cuBLAS)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and make the hidden dimension the batch axis
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)   (k,j) swapped

    B, H, N, _ = left.shape
    batch_dim = B * H

    # Flatten hidden & batch together so that torch.bmm sees (batch_dim, N, N)
    left_mat = left.view(batch_dim, N, N)   # (B*H, N, N)
    right_mat = right.view(batch_dim, N, N)  # (B*H, N, N)

    # Batched matrix multiply – cuBLAS does FP16 × FP16 → FP32 accumulation
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)  fp16 → fp32 internally

    # Reshape back to [B, N, N, H] (float32)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)           # (B, N, N, H)
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 6. Post‑contraction LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate (still in float32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim D
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (dim, hidden_dim) weight → [B, N, N, D]

    return out
```
