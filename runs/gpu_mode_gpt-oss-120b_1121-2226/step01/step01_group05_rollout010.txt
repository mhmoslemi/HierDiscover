@triton.jit
def tri_mul_outgoing_kernel(a_ptr, b_ptr, c_ptr,
                            M, N, K,
                            stride_ah, stride_ai, stride_ak,
                            stride_bh, stride_bk, stride_bn,
                            stride_ch, stride_ci, stride_cj,
                            BLOCK_M: tl.constexpr,
                            BLOCK_N: tl.constexpr,
                            BLOCK_K: tl.constexpr):
    # similar to baseline, but final store as half
    pid_h = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K
        a_ptrs = a_ptr + pid_h * stride_ah + offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)
        b_ptrs = b_ptr + pid_h * stride_bh + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
    # store as fp16
    c_ptrs = c_ptr + pid_h * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),

                     bias=weights["norm.bias"])

    x_h = x.half()
    
    # 2. Combine linear projections and gates
    # Concatenate weight matrices
    weight_list = [
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ]

    combined = F.linear(x_h, combined_weight)  # using half matmul
    B, N, _, _ = combined.shape  # shape (B,N,N,5*hidden_dim)

    combined = combined.view(B, N, N, 5, hidden_dim)  # (B,N,N,5, H)
    left_proj = combined[..., 0, :]  # (B,N,N,H)
    right_proj = combined[..., 1, :]
    left_gate = torch.sigmoid(combined[..., 2, :])
    right_gate = torch.sigmoid(combined[..., 3, :])
    out_gate = torch.sigmoid(combined[..., 4, :])

        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)  # [B,N,N,1]

        right_proj = right_proj * mask

    left = left_proj * left_gate
    right = right_proj * right_gate

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The forward pass consists of three stages:

1️⃣ Input Layer‑Norm (float32) → half.
2️⃣ **Fused linear stage** – the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate)
   are concatenated into a single matrix (5·hidden_dim × dim) and
   multiplied with the input in one half‑precision GEMM.
   The result is reshaped and split into the three projection tensors
   and the three gating tensors (sigmoid applied to the gates).
   An optional mask is broadcast and applied before gating.

3️⃣ **Cubic update** – a batched matrix‑multiply  
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.  
   We treat the hidden dimension as an extra batch dimension
   (BH = B·hidden_dim) and launch a tiled Triton kernel that computes
   `A @ Bᵀ` for each hidden slice.  The kernel works in FP16 with FP32
   accumulation and writes the result in FP16 (halving memory traffic).

4️⃣ Second Layer‑Norm + output‑gate (float32) → final linear projection
   (half → float32).  The output tensor has shape `[B, N, N, dim]`
   and dtype ``float32`` as required.

The major speed‑ups compared to the reference implementation are:
* **Fusion of the five linear layers** into a single GEMM.
* **Half‑precision arithmetic** for the heavy linear and cubic parts.
* **Larger tile size (128×128×64)** and double‑buffered Triton launch
  (`num_warps=8, num_stages=3`), fully exploiting the H100 tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B half → C half, acc in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)      # (batch * hidden) index
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # -------- A tile (M × K) ------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)               # FP16 → FP32

        # -------- Bᵀ tile (K × N) ----------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)               # FP16 → FP32

        # -------- inner product ------------------------------------
        acc += tl.dot(a, b)                # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # write out as half (store half‑precision to save bandwidth)
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, dim]  float32
    x_h = x.half()                               # half for the heavy mats

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 × hidden_dim outputs)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices once (shape (5*H, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # half

    # single half‑precision GEMM → shape [B,N,N,5*H]
    fused = F.linear(x_h, fused_weight)          # half
    B, N, _, _ = fused.shape

    # split the fused result
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H] (half)
    left_proj  = fused[..., 0, :]                # [B,N,N,H]
    right_proj = fused[..., 1, :]
    left_gate  = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # ------------------------------------------------------------------
    left  = left_proj  * left_gate   # [B,N,N,H] half
    right = right_proj * right_gate  # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 5️⃣ Rearrange for batched GEMM (BH = B·H)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N]  half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]  half

    # ------------------------------------------------------------------
    # 6️⃣ Triton GEMM kernel (A @ Bᵀ) → half output
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # each (batch, hidden) slice
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)   # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) and output gate
    # ------------------------------------------------------------------
    out = out_half.float()   # cast to fp32 for layer‑norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                       # [B,N,N,H] float32

    out = out * out_gate.float()                 # apply output gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out.half(), to_out_weight_h)   # half output
    out_final = out_proj.float()                       # cast back to fp32

    return out_final
```
