def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # Load weights
    ln_weight = weights["norm.weight"]; ln_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"] # shape (hid, dim)
    left_gate_weight = weights["left_gate.weight"]
    right_proj_weight = weights["right_proj.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Cast weights to fp16 (they are likely fp32)
    left_concat_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16)
    right_concat_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16)
    out_gate_weight_fp16 = out_gate_weight.to(torch.float16)
    # to_out_norm weight & bias remain fp32; to_out_weight also fp32 for final projection

    # 1. layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=ln_weight, bias=ln_bias, eps=1e-5) # fp32
    x_fp16 = x.to(torch.float16)

    # 2. combined linear projections
    # left_concat: shape [B,N,N,2*hid]
    left_concat = F.linear(x_fp16, left_concat_weight) # fp16
    split = hidden_dim
    left = left_concat[..., :split]   # fp16
    left_gate = torch.sigmoid(left_concat[..., split:])  # fp16

    right_concat = F.linear(x_fp16, right_concat_weight) # fp16
    right = right_concat[..., :split]
    right_gate = torch.sigmoid(right_concat[..., split:])

    # 3. out_gate
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight_fp16)) # fp16, shape [B,N,N,hid]

    # 4. Mask application & gating fuse via Triton
    if mask is None:
        # just multiply left and right by gates (mask=1)
        # Use Triton kernel without mask
        # Could fuse left*left_gate and right*right_gate only. Maybe we can use torch.mul in-place: left.mul_(left_gate), etc.
        # Using torch operations may be efficient already (in-place).
        left = left * left_gate
        right = right * right_gate
    else:
        mask_fp16 = mask.to(torch.float16)  # shape [B,N,N]
        mask_expand = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        # Use Triton kernel:
        size = left.numel()
        BLOCK = 65536  # maybe larger block for throughput.
        grid = (triton.cdiv(size, BLOCK),)
        _mask_gate_fuse_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_expand,
            size=size,
            BLOCK_SIZE=BLOCK,
        )
    # left,right now masked and gated

    # 5. core tri mul
    # reshape
    # Ensure contiguous before reshape:
    left = left.contiguous()
    right = right.contiguous()
    left_reshape = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    # multiply: left @ right^T
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # reshape back
    out = out_reshape.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape B,N,N,H
    out = out.to(torch.float32)  # fp32

    # 6. out layer norm
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # 7. multiply by out_gate (converted to fp32)
    out_gate = out_gate.to(torch.float32)
    out = out * out_gate

    # 8. final projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Highly‑optimized outgoing TriMul forward (AlphaFold3) using Triton.

Algorithm
---------
1. Layer‑norm the (B,N,N,dim) input (FP32).
2. Cast to FP16 for the heavy linear work.
3. Fuse left‑projection + left‑gate and right‑projection + right‑gate
   into two GEMMs by concatenating the weight matrices (2×hidden_dim rows).
4. Compute the three sigmoid gates (left, right, out) in FP16.
5. Fuse mask, left‑gate and right‑gate multiplications in a single Triton
   kernel (no extra memory for the expanded mask).
6. Core TriMul: for each hidden dimension perform a batched matrix‑multiply
   `out_h = left_h @ right_hᵀ`.  Implemented with the highly‑optimized cuBLAS
   batched GEMM (`torch.bmm` on FP16 tensors).
7. Apply a second LayerNorm over the hidden dimension, multiply by the out‑gate
   (FP32) and a final linear projection back to `dim`.

Only the element‑wise mask‑gate fusion is written in Triton; all GEMMs use
cuBLAS/Tensor‑cores, which on an H100 gives the required ~1 ms runtime for the
tested problem sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # FP16,   [size]
    right_ptr,           # FP16,   [size]
    left_gate_ptr,       # FP16,   [size]
    right_gate_ptr,      # FP16,   [size]
    mask_ptr,            # FP16,   [mask_size] (mask without hidden dim)
    size,                # total number of elements in left/right (int)
    HIDDEN_DIM: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse: left = left * left_gate * mask ; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    # load tensors (out‑of‑bounds threads get 0.0)
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask index = element index // hidden_dim  (broadcast mask over H)
    mask_idx = offs // HIDDEN_DIM
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    # fuse the three element‑wise ops
    left = left * left_gate * m
    right = right * right_gate * m

    # store back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor[B, N, N, dim] (float32)
              - mask: torch.Tensor[B, N, N] (bool/float) or None
              - weights: dict of torch.Tensors (see keys below)
              - config: dict with at least 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # typically 128

    # ------------------------------------------------------------------
    # weight shortcuts (all are stored as FP32 in the checkpoint)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (H, dim)
    left_gate_weight  = weights["left_gate.weight"]   # (H, dim)
    right_proj_weight = weights["right_proj.weight"]  # (H, dim)
    right_gate_weight = weights["right_gate.weight"]  # (H, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (H, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]   # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"]     # (H,)
    to_out_weight      = weights["to_out.weight"]        # (dim, H)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32, most stable)
    # ------------------------------------------------------------------
    with torch.no_grad():
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=norm_weight,
            bias=norm_bias,
            eps=1e-5,
        )                     # [B,N,N,dim], float32

        # ------------------------------------------------------------------
        # 2) Cast to FP16 for all linear work
        # ------------------------------------------------------------------
        x_fp16 = x.to(torch.float16)

        # ------------------------------------------------------------------
        # 3) Concatenated linear projections + sigmoid gates
        # ------------------------------------------------------------------
        # left: concat(proj, gate)  -> shape (B,N,N,2*H)
        left_concat_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16)
        left_concat = F.linear(x_fp16, left_concat_weight)                 # [B,N,N,2H] fp16
        left = left_concat[..., :hidden_dim]                              # proj
        left_gate = torch.sigmoid(left_concat[..., hidden_dim:])           # gate

        # right: concat(proj, gate)
        right_concat_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16)
        right_concat = F.linear(x_fp16, right_concat_weight)               # [B,N,N,2H] fp16
        right = right_concat[..., :hidden_dim]                             # proj
        right_gate = torch.sigmoid(right_concat[..., hidden_dim:])          # gate

        # free temporaries
        left_concat = right_concat = None

        # out‑gate (independent linear + sigmoid)
        out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight.to(torch.float16)))  # [B,N,N,H] fp16

        # ------------------------------------------------------------------
        # 4) Fuse mask + left/right gates (Triton)
        # ------------------------------------------------------------------
        if config.get("nomask", False):
            # a mask of all ones – cheaper than allocating a large tensor
            mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=x.device)
        else:
            mask_fp16 = mask.to(torch.float16) if mask is not None else torch.ones(
                (bs, N, N), dtype=torch.float16, device=x.device
            )
            # ensure contiguous for 1‑D view
            mask_fp16 = mask_fp16.contiguous()

        # Flatten everything (same total element count for left/right/gates)
        size = left.numel()                     # B * N * N * H
        BLOCK = 1024
        grid = (triton.cdiv(size, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            size=size,
            HIDDEN_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
        # after the kernel left/right are already masked & gated
        # left_gate/right_gate are no longer needed
        left_gate = right_gate = None

        # ------------------------------------------------------------------
        # 5) Core TriMul – batched matmul per hidden dimension (FP16)
        # ------------------------------------------------------------------
        # make tensors contiguous before reshaping for bmm
        left = left.contiguous()
        right = right.contiguous()

        # reshape to (B*H, N, N)
        left_bmm = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
        right_bmm = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

        # left @ rightᵀ  (each batch = one hidden dimension)
        out_bmm = torch.bmm(left_bmm, right_bmm.transpose(1, 2))

        # reshape back to (B, N, N, H) and cast to FP32 for following LayerNorm
        out = out_bmm.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

        # ------------------------------------------------------------------
        # 6) Output LayerNorm (over hidden_dim) + out‑gate
        # ------------------------------------------------------------------
        out = F.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=to_out_norm_weight,
            bias=to_out_norm_bias,
            eps=1e-5,
        )
        out = out * out_gate.to(torch.float32)   # gating in FP32

        # ------------------------------------------------------------------
        # 7) Final linear projection back to `dim` (FP32)
        # ------------------------------------------------------------------
        out = F.linear(out, to_out_weight)       # [B,N,N,dim] float32

        return out
