#!/usr/bin/env python
"""
TriMul (outgoing) – fast fused implementation.

The forward pass consists of:
1️⃣ Layer‑norm the input pair representation (last dim = dim).
2️⃣ Two linear projections (left / right) → hidden_dim, optional mask
   and three sigmoid gates (left, right, out).
3️⃣ Fuse the cubic “∑_k left[i,k,h] * right[j,k,h]” using a
   batched GEMM kernel written in Triton.  The kernel computes
   A @ Bᵀ for each (batch, head) pair, where A and B have shape
   (N, N) and are stored in FP16; accumulation is performed in FP32.
   Tiling parameters (128×128×64) are chosen to fully exploit the
   H100 tensor‑core units while keeping shared‑memory usage low.
4️⃣ Layer‑norm over the hidden dimension, an output‑gate, and a final
   linear projection back to the original dim.

Only the heavy cubic contraction is executed on Triton – everything
else uses highly‑optimised PyTorch operators.  The implementation
covers all configurations required by the test suite and works for
both masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) with FP16 inputs / FP32 acc.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # tensor pointers
    M, N, K,                           # problem size (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ.
    A : [BH, M, K]   (fp16)
    B : [BH, N, K]   (fp16) – we treat Bᵀ inside the kernel
    C : [BH, M, N]   (fp32)
    """
    # --------------------------------------------------------------
    # Program IDs – one kernel instance per (M‑tile, N‑tile, BH) triple
    # --------------------------------------------------------------
    pid_h = tl.program_id(axis=2)          # batch * hidden index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑loop
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----------- Load A tile (M × K) -------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)        # fp16 → fp32

        # ----------- Load Bᵀ tile (K × N) -----------
        # B is stored as (BH, N, K); we load its transpose.
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)        # fp16 → fp32

        # ----------- Inner product (matrix multiply) ----------
        acc += tl.dot(a, b)         # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Store result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul outgoing update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float)
        - weights_dict : mapping of all learnable parameters
        - config_dict  : must contain ``dim`` and ``hidden_dim``.
          Optional keys: ``nomask`` (bool, default True).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias) – keep them in FP32 for the moment
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of separate linear maps) – keep in FP32
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates (still FP32)
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 – the cubic kernel works in FP16 (inputs) / FP32 (acc)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape   # B, seq_len, seq_len, hidden_dim

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to [BH, N, N] where BH = B * hidden_dim
    # ------------------------------------------------------------------
    #   left_perm : [B, H, N, N]  (contiguous)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)        # [BH, N, N]   (fp16)
    right_flat = right_perm.view(BH, N, N)      # [BH, N, N]   (fp16)

    # Output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Choose tiling parameters (tuned for H100)
    # ------------------------------------------------------------------
    # For N ≥ 512 we use the larger tile; for very small N we fall back.
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
        num_warps = 4
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
        num_warps = 8

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # batch * hidden
    )

    # ------------------------------------------------------------------
    # 8️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    _tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),    # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()        # [B, N, N, H]

    # ------------------------------------------------------------------
    # 🔟 Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # still FP32

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
