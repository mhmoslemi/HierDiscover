"""
TriMul (outgoing) – fast forward implementation using mixed‑precision + Triton.

Algorithm
---------
1️⃣  Layer‑norm the input over the last dimension (dim) – FP32.
2️⃣  Cast the normalized tensor to FP16 and also cast all linear weights to FP16.
3️⃣  Linear projections → left, right   (shape [B,N,N,hidden_dim], FP16).
4️⃣  Compute three sigmoid gates in FP32, cast them to FP16.
5️⃣  Fuse element‑wise multiplication (proj * gate * optional mask) with a tiny
    Triton kernel that works entirely in FP16 → produces left_fp16 & right_fp16.
6️⃣  Reshape to (B*hidden_dim, N, N) and run a batched GEMM (torch.bmm) on the
    FP16 tensors:  out = left @ rightᵀ .   (use Tensor Cores on H100)
7️⃣  Convert the result back to FP32, apply a second LayerNorm over the hidden
    dimension, multiply by the output gate (FP32).
8️⃣  Final linear projection back to `dim` (FP16 weight, FP16 mat‑vec, then cast).

Only step 5 is performed with a custom Triton kernel – all other
operations use the highly‑optimised PyTorch/CUDA kernels.  The heavy
O(N³·hidden_dim) work stays in FP16, which gives a ≈2‑3× speed‑up versus
the previous FP32‑only implementation while preserving full numerical
accuracy after the final FP32 conversion.
"""

import triton
import triton.language as tl

# Triton kernel: element‑wise fusion (proj * gate * mask) – FP16
# ----------------------------------------------------------------------
@triton.jit
def _fused_mul_fp16_kernel(
    a_ptr,               # left/right projection (FP16)
    b_ptr,               # left/right gate      (FP16)
    mask_ptr,            # optional mask        (FP16)
    out_ptr,             # output               (FP16)
    n_elements,          # total number of elements (=B*N*N*H)
    hidden_dim,          # H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):

    out = a * b * mask   (mask broadcast over hidden_dim)
    a, b : [B, N, N, H] (flattened)
    mask : [B, N, N]    (flattened)
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    # Load FP16 values
    a = tl.load(a_ptr + offs, mask=mask, dtype=tl.float16)
    b = tl.load(b_ptr + offs, mask=mask, dtype=tl.float16)

    if HAS_MASK:
        # Compute mask index (mask is smaller by factor hidden_dim)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, dtype=tl.float16)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=mask)


def _fused_mul_fp16(a: torch.Tensor,
                    b: torch.Tensor,
                    mask: torch.Tensor | None = None) -> torch.Tensor:
    """
    Fuse ``a * b * mask`` with a Triton kernel (all tensors FP16).

    a, b : [B, N, N, H]   (contiguous, dtype=torch.float16)
    mask : [B, N, N]      (dtype=torch.float16) or None
    Returns: tensor of same shape & dtype as ``a``.
    """
    assert a.shape == b.shape, "Projection and gate shapes must match"
    B, N, _, H = a.shape
    total = a.numel()                       # B * N * N * H
    BLOCK = 131072                           # 128 KB tiles → good occupancy
    grid = ( (total + BLOCK - 1) // BLOCK, )

    a_flat = a.view(-1)
    b_flat = b.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        # Dummy tensor – never accessed because HAS_MASK == 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=a.device)
        has_mask = 0

    out = torch.empty_like(a)
    _fused_mul_fp16_kernel[grid](
        a_flat,
        b_flat,
        mask_flat,
        out.view(-1),
        total,
        H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor – [bs, seq_len, seq_len, dim]  (float32)
        * mask_tensor  – [bs, seq_len, seq_len]      (float32) or ``None``
        * weights_dict – model weights (float32 tensors)
        * config_dict  – contains ``dim`` and ``hidden_dim`` and
                        a boolean ``nomask`` flag.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣  Cast to FP16 and prepare FP16 weights
    # ------------------------------------------------------------------
    x_norm_f16 = x_norm.to(torch.float16)

    # Linear weights → FP16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # final projection

    # ------------------------------------------------------------------
    # 3️⃣  Linear projections (FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm_f16, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm_f16, right_proj_w) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣  Compute gating tensors (sigmoid, keep FP32 for stability)
    # ------------------------------------------------------------------
    left_gate_fp32 = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate_fp32 = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Cast gates to FP16 (they will only be used in element‑wise mul)
    left_gate = left_gate_fp32.to(torch.float16)
    right_gate = right_gate_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 5️⃣  Optional mask (broadcast over hidden_dim) – FP16
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_f16 = None
    else:
        mask_f16 = mask.to(torch.float16)   # [B, N, N]

    # ------------------------------------------------------------------
    # 6️⃣  Fuse element‑wise multiplication with Triton (FP16)
    # ------------------------------------------------------------------
    left = _fused_mul_fp16(left, left_gate, mask_f16)   # still FP16
    right = _fused_mul_fp16(right, right_gate, mask_f16)

    # ------------------------------------------------------------------
    # 7️⃣  Batched GEMM across the sequence dimension (FP16)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                               # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_mat = left_mat @ right_matᵀ   (FP16, Tensor‑cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim] (still FP16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 8️⃣  Output LayerNorm (over hidden_dim) – back to FP32
    # ------------------------------------------------------------------
    out_fp32 = out.to(torch.float32)

    out_fp32 = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9️⃣  Apply final output gate (FP32)
    # ------------------------------------------------------------------
    out_fp32 = out_fp32 * out_gate

    # ------------------------------------------------------------------
    # 🔟  Final linear projection back to `dim` (FP16 weight, then cast)
    # ------------------------------------------------------------------
    out_fp16 = out_fp32.to(torch.float16)
    out_fp16 = torch.nn.functional.linear(out_fp16, to_out_w)   # [B,N,N,dim] FP16
    out = out_fp16.to(torch.float32)

    return out
