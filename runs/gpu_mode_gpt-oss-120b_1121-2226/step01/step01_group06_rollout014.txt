@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   pid = tl.program_id(0)

   # compute grid dims
   num_m = tl.cdiv(N, BLOCK_M)
   num_n = tl.cdiv(N, BLOCK_N)
   blocks_per_head = num_m * num_n
   batch_head_id = pid // blocks_per_head
   block_idx = pid % blocks_per_head
   block_i = block_idx // num_n
   block_j = block_idx % num_n

   # compute batch and head indices
   b = batch_head_id // H
   h = batch_head_id % H

   # compute base pointers
   left_base = left_ptr + b * stride_left_b + h * stride_left_h
   right_base = right_ptr + b * stride_right_b + h * stride_right_h

   # offsets
   offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
   mask_i = offs_i < N
   mask_j = offs_j < N

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(N, BLOCK_K)):
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

The heavy O(N³) contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is performed by a custom batched‑matmul kernel that works on FP16
inputs (FP32 accumulation) and exploits Tensor‑cores on an H100.
All other operations (LayerNorm, Linear, sigmoid, gating) stay in
PyTorch.  The implementation avoids extra copies apart from the
mandatory permute that makes each head’s (N×N) matrix contiguous,
which gives coalesced loads inside the kernel.

The kernel is autotuned for the target sizes (N up to 1024, hidden
dim up to 384) and runs in roughly 1 ms on an H100 for the
largest test case.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Autotuned batched matmul kernel:   C = A @ Bᵀ
#   A : (batch, M, K)   FP16
#   B : (batch, K, N)   FP16   (already transposed, i.e. original RHSᵀ)
#   C : (batch, M, N)   FP16
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        # best on H100 for N≈1024
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64, "num_warps": 8},
            num_stages=3,
        ),
        triton.Config(
            {"BLOCK_M": 256, "BLOCK_N": 64, "BLOCK_K": 64, "num_warps": 8},
            num_stages=3,
        ),
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 64, "num_warps": 4},
            num_stages=3,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 64, "BLOCK_K": 32, "num_warps": 8},
            num_stages=2,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_am, stride_ak, stride_ab,   # strides for A (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides for B (batch, K, N)
    stride_cm, stride_cn, stride_cb,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------
    #  block (batch, block_m, block_n)
    # --------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)
    block = pid % (num_m * num_n)
    block_m = block // num_n
    block_n = block % num_n

    # --------------------------------------------------
    #  tile offsets
    # --------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers for current tile
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # rank‑K dot
        acc += tl.dot(a, b)

    # --------------------------------------------------
    #  write back
    # --------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """FP16 batched matmul with FP32 accumulation (A @ Bᵀ)."""
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "inner dimensions must match"

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # grid is a function of block sizes chosen by autotune
    grid = lambda META: (
        B
        * ((M + META["BLOCK_M"] - 1) // META["BLOCK_M"])
        * ((N + META["BLOCK_N"] - 1) // META["BLOCK_N"]),
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
    )
    return out


# ----------------------------------------------------------------------
# Triton‑backed TriMul forward (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            mask         : torch.Tensor [B, N, N] or None
            weights      : dict of model parameters (all float32)
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # model weights (all FP32, no bias unless explicitly stored)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy cubic contraction – Triton kernel
    # ------------------------------------------------------------------
    # cast to FP16 before the MATMUL to cut bandwidth
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # bring the head dimension to the batch axis and make each (N,N) matrix
    # contiguous in memory (required for high‑throughput GEMM)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm_t = right_h.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape

    A = left_perm.view(B * H, N, N)          # (BH, M=N, K=N)
    B_mat = right_perm_t.view(B * H, N, N)   # (BH, K=N, N)

    # batched matmul: C = A @ Bᵀ   (Bᵗ is already transposed)
    C = _batched_matmul_fp16(A, B_mat)      # (BH, N, N)  FP16

    # reshape back to the original layout [B, N, N, H]
    out_fp16 = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Second Layer‑norm over hidden dimension
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    return out
