def custom_kernel(data):
   input_tensor, mask, weights, config = data
   dim = config['dim']
   hidden_dim = config['hidden_dim']
   nomask = config.get('nomask', True)

   # 1. LayerNorm
   x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

   # 2. Linear projections and gates
   left = F.linear(x, weights['left_proj.weight'])
   right = F.linear(x, weights['right_proj.weight'])

   left_gate = F.linear(x, weights['left_gate.weight']).sigmoid()
   right_gate = F.linear(x, weights['right_gate.weight']).sigmoid()
   out_gate = F.linear(x, weights['out_gate.weight']).sigmoid()

   # Cast gates to fp16
   left_gate_f16 = left_gate.to(torch.float16)
   right_gate_f16 = right_gate.to(torch.float16)

   # 3. Gating
   left = (left * left_gate_f16).to(torch.float16)
   right = (right * right_gate_f16).to(torch.float16)

   # 4. Optional mask (elementwise)
   if not nomask:
       mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
       # Use Triton kernel to multiply mask in-place
       left, right = _apply_mask(left, right, mask_f16)   # We'll implement function returning same left,right

   # 5. Reshape for batched matmul
   B, N, _, H = left.shape
   left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # (B*H, N, N)
   right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

   # 6. Batched matmul using torch.bmm
   out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)
   # Convert to fp32
   out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

   out = F.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'])

   # 8. Apply out_gate and final linear
   out = out * out_gate
   out = F.linear(out, weights['to_out.weight'])

   return out

    right = right * right_gate

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights['to_out_norm.weight'],
        bias=weights['to_out_norm.bias']
    )

    # 11. Apply final gate and projection
    out = out * out_gate
    out = F.linear(out, weights['to_out.weight'])

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass
==================================================

Algorithm (batch‑first, 4‑D tensors ``[B, N, N, C]``)
----------------------------------------------------
1. **Input LayerNorm** (float32).
2. Cast the normalized tensor to ``float16`` – all heavy linear work will use
   half‑precision (the H100 Tensor‑Core kernels are far faster in fp16).
3. **Linear projections** – ``left`` and ``right`` (fp16) and three sigmoid
   gates (fp16 for ``left_gate``/``right_gate``; fp32 for ``out_gate``).
4. Apply the element‑wise gates to the projected tensors.
5. **Optional mask** – a tiny Triton kernel multiplies both ``left`` and
   ``right`` by the mask (broadcast over the hidden dimension).
6. Reshape ``[B, N, N, H]`` → ``[B*H, N, N]`` (contiguous) and perform a
   **batched matrix multiplication** using ``torch.bmm``:
   ``out_h = left_h @ right_hᵀ`` for every hidden channel *h*.
   Inputs are fp16, the result is cast to fp32.
7. Reshape back to ``[B, N, N, H]``.
8. **Output LayerNorm** over the hidden dimension (fp32), multiply by the
   ``out_gate`` and apply the final linear projection back to ``dim``.
9. Return the tensor ``[B, N, N, dim]`` (float32).

Only the mask‑application step is performed with a custom Triton kernel,
while the dominant ``N³`` workload (batched matmul) leverages the highly
optimised cuBLAS implementation (`torch.bmm`) on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask multiplication (broadcast over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_mul_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    mask_ptr,          # fp16 *
    total_elems,       # int64 – number of elements in left/right (B·N·N·H)
    total_mask,        # int64 – number of elements in mask (B·N·N)
    H,                 # int64 – hidden dimension size
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)               # linear offsets

    # mask index = element offset divided by H (broadcast)
    mask_offs = offs // H

    # bounds
    valid = offs < total_elems
    mask_valid = mask_offs < total_mask

    # load
    left = tl.load(left_ptr + offs, mask=valid, other=0.0)
    right = tl.load(right_ptr + offs, mask=valid, other=0.0)
    mask = tl.load(mask_ptr + mask_offs, mask=mask_valid, other=1.0)

    # apply
    left = left * mask
    right = right * mask

    # store result back in‑place
    tl.store(left_ptr + offs, left, mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


def _apply_mask_triton(left: torch.Tensor,
                       right: torch.Tensor,
                       mask: torch.Tensor) -> None:
    """
    In‑place elementwise multiplication of ``left`` and ``right`` by ``mask``.
    All tensors must be contiguous, on the same CUDA device and of dtype
    ``torch.float16`` (the mask is already broadcasted to a trailing singleton
    dimension and flattened).

    Parameters
    ----------
    left, right : torch.Tensor
        Shape ``[B, N, N, H]`` (fp16), will be updated in‑place.
    mask : torch.Tensor
        Shape ``[B, N, N]`` (fp16) – flatten before calling.
    """
    assert left.is_cuda and right.is_cuda and mask.is_cuda
    total = left.numel()
    total_mask = mask.numel()
    H = left.shape[-1]

    # A fairly large block size gives good occupancy on an H100.
    BLOCK = 65536
    grid = ((total + BLOCK - 1) // BLOCK, )
    _mask_mul_kernel[grid](
        left,
        right,
        mask,
        total,
        total_mask,
        H,
        BLOCK_SIZE=BLOCK,
    )


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        ``(input_tensor, mask, weights_dict, config_dict)`` where
        * ``input_tensor`` – torch.Tensor of shape ``[B, N, N, dim]`` (float32)
        * ``mask`` – torch.Tensor of shape ``[B, N, N]`` (bool/int/float) or
          ``None`` when ``config['nomask']`` is ``True``
        * ``weights_dict`` – dictionary containing all model weights as
          torch.Tensors (float32)
        * ``config_dict`` – contains ``dim``, ``hidden_dim`` and optional
          ``nomask`` flag.

    Returns
    -------
    torch.Tensor
        Tensor of shape ``[B, N, N, dim]`` (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, cfg = data
    device = input_tensor.device

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Cast to fp16 – all subsequent heavy work uses half precision
    # ------------------------------------------------------------------
    x_fp16 = x.half()               # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias) in fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x_fp16, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 4) Gates – sigmoid, keep in fp16 for left/right, fp32 for out_gate
    # ------------------------------------------------------------------
    left_gate = F.linear(x_fp16, weights["left_gate.weight"]).sigmoid()   # fp16
    right_gate = F.linear(x_fp16, weights["right_gate.weight"]).sigmoid() # fp16
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()          # fp32

    # ------------------------------------------------------------------
    # 5) Apply left/right gates (still fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Optional mask – fused Triton kernel (in‑place)
    # ------------------------------------------------------------------
    if not nomask:
        # mask → fp16 and broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).contiguous()
        mask_fp16 = mask_fp16.unsqueeze(-1).contiguous()   # [B, N, N, 1]
        # flatten mask for the kernel
        mask_flat = mask_fp16.view(-1)
        # ensure left/right are contiguous before launching the kernel
        left = left.contiguous()
        right = right.contiguous()
        _apply_mask_triton(left, right, mask_flat)

    # ------------------------------------------------------------------
    # 7) Reshape for the batched matrix multiplication
    #      (B, N, N, H) → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # ------------------------------------------------------------------
    # 8) Core TriMul: batched matmul L @ Rᵀ (fp16 inputs, fp32 accumulation)
    # ------------------------------------------------------------------
    # torch.bmm internally uses cuBLAS Tensor‑Core kernels on H100.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).float()   # fp32

    # ------------------------------------------------------------------
    # 9) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 10) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    # ------------------------------------------------------------------
    # 11) Gating with out_gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                         # element‑wise (fp32)
    out = F.linear(out, weights["to_out.weight"])  # fp32, final shape [B, N, N, dim]

    return out
