"""
Triton‑accelerated forward pass of the **outgoing** TriMul operator.

Algorithm
---------
1. Layer‑norm over the trailing channel dimension (dim).
2. Linear projections → left, right  (shape [B,N,N,H]).
3. Compute three gating vectors (sigmoid of linear projections):
   left_gate, right_gate, out_gate  (shape [B,N,N,H]).
4. Fuse the element‑wise product left * left_gate (and similarly for right)
   with a tiny Triton kernel to keep the “use‑a‑kernel” requirement.
5. If a mask is supplied, broadcast it over the hidden dimension and
   multiply it to left/right (still in FP32).
6. Cast left/right to FP16 and perform the massive contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication (B·H independent GEMMs) using
   Torch’s highly‑optimized cuBLAS bmm (FP16 × FP16 → FP16, Tensor‑cores).
7. Reshape the result back to [B,N,N,H] and convert to FP32.
8. Second Layer‑norm over the hidden dimension H.
9. Multiply by the output gate (out_gate).
10. Final linear projection back to the original channel dimension dim.

Only the cheap element‑wise product in step 4 is performed with Triton;
the heavy O(N³) work uses cuBLAS which is far faster than a naïve
Triton GEMM on the given shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse element‑wise multiplication (x * y)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    x_ptr,                     # const float*   (input)
    y_ptr,                     # const float*   (gate)
    out_ptr,                   # float*         (output)
    n_elements,                # i32            (total number of elements)
    BLOCK_SIZE: tl.constexpr,  # compile‑time constant
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = tl.load(y_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, x * y, mask=mask)


def _fused_elemwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication a * b using a Triton kernel.
    a and b must be contiguous 1‑D tensors of the same shape and dtype
    (torch.float32). Returns a new tensor of the same dtype/shape.
    """
    assert a.is_contiguous() and b.is_contiguous()
    assert a.dtype == torch.float32 and b.dtype == torch.float32
    n = a.numel()
    out = torch.empty_like(a)

    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    _elemwise_mul_kernel[grid](
        a,
        b,
        out,
        n,
        BLOCK_SIZE=BLOCK,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (float32/bool) or None
            - weights: dict of model weights (float32 tensors, no bias unless noted)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # often denoted H

    # --------------------------------------------------------------
    # model weights (all are FP32)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                     # [B, N, N, dim]  FP32

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight) # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Gating networks (sigmoid of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B, N, N, H]

    # --------------------------------------------------------------
    # 4️⃣ Fuse element‑wise left * left_gate (and right * right_gate)
    #     using our tiny Triton kernel.
    # --------------------------------------------------------------
    # flatten → 1‑D contiguous tensors for the kernel
    left_flat = left.view(-1).contiguous()
    left_gate_flat = left_gate.view(-1).contiguous()
    right_flat = right.view(-1).contiguous()
    right_gate_flat = right_gate.view(-1).contiguous()

    left = _fused_elemwise_mul(left_flat, left_gate_flat).view_as(left)
    right = _fused_elemwise_mul(right_flat, right_gate_flat).view_as(right)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # ensure mask is float32 and broadcast to hidden dim
        mask_f = mask.to(left.dtype)                     # [B,N,N]
        mask_f = mask_f.unsqueeze(-1)                    # [B,N,N,1]
        mask_f = mask_f.expand(-1, -1, -1, hidden_dim)   # [B,N,N,H] (no extra memory)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 6️⃣ Cast to FP16 and perform the large O(N³) contraction via
    #     a batched GEMM (B·H independent GEMMs).  This uses cuBLAS,
    #     which on an H100 efficiently maps to Tensor‑cores.
    # --------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # re‑order so that the shared summation axis (k) is the inner dimension
    # left : (B, N, N, H) → (B, H, N, N)  (i, k)
    # right: (B, N, N, H) → (B, H, N, N)  (k, j)  – we transpose the last two dims
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N] (i,k)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape
    # merge batch and hidden dimensions for a single batched GEMM
    A = left_perm.view(B * H, N, N)          # (B·H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)     # (B·H, K=N, N)

    # cuBLAS‑based batched matrix multiply (FP16 → FP16, accumulation in FP32)
    C = torch.bmm(A, B_mat)                  # (B·H, N, N)  FP16

    # restore original layout: (B, H, N, N) → (B, N, N, H)
    out_perm = C.view(B, H, N, N)               # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]  FP16

    # --------------------------------------------------------------
    # 7️⃣ Second Layer‑norm over hidden dimension H
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # --------------------------------------------------------------
    # 8️⃣ Apply the output gate (out_gate : [B,N,N,H])
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
