```python
@triton.jit
def _mask_gate_fuse_kernel(left_ptr, right_ptr,
                           left_gate_ptr, right_gate_ptr,
                           mask_ptr,
                           size,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=0.0)
    l = l * lg * m
    r = r * rg * m
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)
```

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm on the channel axis.
2. Fuse *all* linear maps (left/right projections + three gates) into a **single**
   half‑precision GEMM using `torch.nn.functional.linear`.  This eliminates 4
   separate matmuls and reduces memory traffic.
3. Apply sigmoid to the three gate tensors.
4. Fuse mask, left‑gate and right‑gate element‑wise with a tiny Triton kernel
   (no extra temporaries).
5. Core TriMul reduces to a batched half‑precision GEMM
   `left @ rightᵀ` for every hidden channel.  We reshape to
   (B·H, N, N) and use `torch.bmm`, which invokes cuBLAS Tensor‑Core kernels.
6. Second Layer‑norm, out‑gate multiplication and final projection back to
   `dim`.
All heavy arithmetic is done in FP16; the final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate multiplications into left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                 # FP16 tensors (flattened)
    left_gate_ptr, right_gate_ptr,       # FP16 tensors (flattened)
    mask_ptr,                           # FP16 tensor (flattened, broadcasted)
    size,                               # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    """left = left * left_gate * mask ; right = right * right_gate * mask."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size

    l  = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + offs, mask=mask, other=0.0)

    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim] (float32)
            - mask: torch.Tensor[B, N, N] (bool/float) or None
            - weights: dict with the nine weight tensors required by the module
            - config: dict containing at least ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # e.g. 128

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel axis ``dim``
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )                                 # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2. Single fused linear that produces left/right projections and all gates
    # ------------------------------------------------------------------
    # gather all five weight matrices (no bias) – shape (hidden_dim, dim)
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # concatenate -> (5*hidden_dim, dim) and cast to FP16 for Tensor‑Core GEMM
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    ).to(torch.float16)

    # flatten input and cast to FP16
    x_fp16 = x.to(torch.float16).reshape(-1, dim)                     # (B·N·N, dim)

    # fused projection (all five outputs together)
    fused = F.linear(x_fp16, fused_weight)                           # (B·N·N, 5*H)  FP16

    # reshape back to 4‑D
    fused = fused.view(bs, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 3. Slice and apply sigmoid to the three gate tensors
    # ------------------------------------------------------------------
    h0 = hidden_dim
    left_proj   = fused[..., 0:h0]
    right_proj  = fused[..., h0:2*h0]
    left_gate   = fused[..., 2*h0:3*h0]
    right_gate  = fused[..., 3*h0:4*h0]
    out_gate    = fused[..., 4*h0:5*h0]          # FP16 now

    # Sigmoid on gates – compute in FP32 for numerical safety, then cast back
    left_gate   = torch.sigmoid(left_gate.to(torch.float32)).to(torch.float16)
    right_gate  = torch.sigmoid(right_gate.to(torch.float32)).to(torch.float16)
    out_gate    = torch.sigmoid(out_gate.to(torch.float32)).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Fuse mask, left‑gate and right‑gate with Triton
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp = torch.ones((bs, N, N), dtype=torch.float16,
                             device=left_proj.device)
    else:
        mask_fp = mask.to(torch.float16)

    # broadcast mask over the hidden dimension
    mask_exp = mask_fp.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

    # ensure all tensors are contiguous before the kernel
    left_proj  = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask_exp   = mask_exp.contiguous()

    # element‑wise fusion kernel
    total_elements = left_proj.numel()
    BLOCK = 8192                     # 8 K elements per program (multiple of 32)
    grid = (triton.cdiv(total_elements, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left_ptr=left_proj,
        right_ptr=right_proj,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_exp,
        size=total_elements,
        BLOCK_SIZE=BLOCK,
    )
    # after the kernel ``left_proj`` holds the gated‑and‑masked left tensor,
    # ``right_proj`` holds the gated‑and‑masked right tensor.

    # ------------------------------------------------------------------
    # 5. Core TriMul: batched FP16 GEMM  left @ rightᵀ
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left = left_proj.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)
    right = right_proj.permute(0, 3, 1, 2).reshape(bs * hidden_dim, N, N)

    # batched matrix multiplication (Tensor‑Core GEMM)
    out = torch.bmm(left, right.transpose(1, 2))                    # (B·H, N, N) FP16

    # reshape back to 4‑D and convert to FP32 for the next LayerNorm
    out = (
        out.view(bs, hidden_dim, N, N)
          .permute(0, 2, 3, 1)
          .to(torch.float32)
    )                                                               # [B,N,N,H] FP32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                                               # [B,N,N,H] FP32

    out = out * out_gate                                            # gating

    to_out_w = weights["to_out.weight"]                             # (dim, hidden_dim)
    out = F.linear(out, to_out_w)                                   # final proj → [B,N,N,dim]

    return out
