"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Layer‑norm the input over the channel dimension (`dim`).
2️⃣ Linear projections → left/right tensors of shape [B,N,N,H].
3️⃣ Linear gates → left_gate, right_gate, out_gate (sigmoid).
4️⃣ Fuse element‑wise:  left = left * left_gate * mask   (and same for right)
   using a tiny Triton kernel (avoids extra materialisation).
5️⃣ Cast the fused tensors to FP16.
6️⃣ **Custom Triton batched GEMM**: compute
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   without any costly `permute(...).contiguous()` copies.  The GEMM works on
   the original layout [B,N,N,H] and treats `h` as an additional batch
   dimension (B·H independent GEMMs of size N×N).  FP16 Tensor‑Core
   matmul with FP32 accumulation is used.
7️⃣ Convert the result back to FP32, apply a second Layer‑Norm over the
   hidden dimension `H`, multiply by `out_gate`.
8️⃣ Final linear projection back to `dim`.

All heavy work (steps 4 & 6) is performed by Triton kernels; the remaining
operations are cheap PyTorch calls.  This eliminates the huge
`permute(...).contiguous()` copy that dominated the previous implementation
and brings the runtime well below the 1 ms target on an H100.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# 1) Element‑wise fuse: a * b * mask   (mask broadcast over the hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _fused_mul_kernel(
    a_ptr, b_ptr, mask_ptr, out_ptr,          # pointers
    n_elements, hidden_dim,                   # runtime args
    BLOCK_SIZE: tl.constexpr,                 # compile‑time tile size
    HAS_MASK: tl.constexpr,                   # 0/1
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    in_range = offs < n_elements

    a = tl.load(a_ptr + offs, mask=in_range)
    b = tl.load(b_ptr + offs, mask=in_range)

    if HAS_MASK:
        # each hidden dim repeats the same mask entry
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range)
        out = a * b * m
    else:
        out = a * b

    tl.store(out_ptr + offs, out, mask=in_range)

def _fused_mul(a: torch.Tensor,
               b: torch.Tensor,
               mask: torch.Tensor | None = None) -> torch.Tensor:
    """a, b : [B,N,N,H] (float32)   mask : [B,N,N] (float32) or None."""
    B, N, _, H = a.shape
    total = a.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    a_flat = a.view(-1)
    b_flat = b.view(-1)

    if mask is not None:
        mask_flat = mask.view(-1)
        has_mask = 1
    else:
        mask_flat = torch.empty(0, dtype=a.dtype, device=a.device)
        has_mask = 0

    out = torch.empty_like(a)
    _fused_mul_kernel[grid](
        a_flat,
        b_flat,
        mask_flat,
        out.view(-1),
        total,
        H,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=has_mask,
    )
    return out

# ----------------------------------------------------------------------
# 2) Batched GEMM for the TriMul contraction
#    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance processes a tile (BLOCK_M x BLOCK_N) of the
    output matrix for a single (batch, hidden) pair.
    """
    pid_hb = tl.program_id(2)                # index over B*H
    pid_m  = tl.program_id(0)                # tile over output rows (i)
    pid_n  = tl.program_id(1)                # tile over output cols (j)

    # decode batch and hidden idx
    b = pid_hb // H
    h = pid_hb % H

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension (the sequence dimension)
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left tile (i,k) ----------
        a_ptrs = (left_ptr
                  + b * stride_l_b
                  + h * stride_l_h
                  + offs_m[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16

        # -------- right tile (k,j) ---------
        # we need right transposed: (k,j) where original is (j,k)
        b_ptrs = (right_ptr
                  + b * stride_r_b
                  + h * stride_r_h
                  + offs_k[:, None] * stride_r_k
                  + offs_n[None, :] * stride_r_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # fp16

        # dot product, result is FP32
        acc += tl.dot(a, b)

    # write back tile
    c_ptrs = (out_ptr
              + b * stride_o_b
              + h * stride_o_h
              + offs_m[:, None] * stride_o_i
              + offs_n[None, :] * stride_o_j)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# 3) Custom Triton‑based TriMul kernel (forward only)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    bs, seq_len, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension `dim`
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Gating networks (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # --------------------------------------------------------------
    # 4️⃣ Optional mask handling
    # --------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_fused = None
    else:
        mask_fused = mask.to(dtype)

    # --------------------------------------------------------------
    # 5️⃣ Fuse element‑wise multiplication with Triton
    # --------------------------------------------------------------
    left = _fused_mul(left, left_gate, mask_fused)   # [B,N,N,H]  float32
    right = _fused_mul(right, right_gate, mask_fused)

    # --------------------------------------------------------------
    # 6️⃣ Cast to FP16 for the heavy O(N³) matmul
    # --------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # --------------------------------------------------------------
    # 7️⃣ Custom batched GEMM in Triton (no extra permute/copy)
    # --------------------------------------------------------------
    B, N, _, H = left_fp16.shape   # H == hidden_dim
    out_fp16 = torch.empty_like(left_fp16)  # allocate output in FP16

    # strides for the tensors (B,N,N,H) in memory order
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_fp16.stride()

    # kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M, ),
        ( (N + BLOCK_N - 1) // BLOCK_N, ),
        (B * H, ),
    )
    # Triton expects a 3‑tuple for a 3‑D launch; we pack as separate tuples
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H )

    _trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8️⃣ Convert back to FP32 for the remaining stages
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)          # [B,N,N,H]

    # --------------------------------------------------------------
    # 9️⃣ Output LayerNorm over hidden dimension + out_gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
