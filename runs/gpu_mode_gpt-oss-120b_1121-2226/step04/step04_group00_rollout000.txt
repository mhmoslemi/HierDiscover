---------
1️⃣ Input LayerNorm (FP32, high‑precision).
2️⃣ Cast to FP16 for the heavy work.
3️⃣ **Fused linear**: concatenate the five bias‑free projection
   matrices (left_proj, left_gate, right_proj, right_gate, out_gate)
   into a single weight of shape `(5*hidden_dim, dim)`.  One `F.linear`
   call therefore computes all five tensors at once, eliminating four
   extra passes over the huge `[B,N,N,dim]` input.
4️⃣ Split the result, apply `sigmoid` to the three gate tensors.
5️⃣ **Fused gate‑multiplication kernel** (Triton) multiplies
   `left *= left_gate` and `right *= right_gate` in a single kernel,
   keeping the data in FP16 and avoiding extra memory traffic.
6️⃣ Optional binary mask (broadcasted over the hidden dimension) is
   applied after gating – a cheap element‑wise multiplication.
7️⃣ Core cubic update: for every hidden channel we need
   `out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]`.
   This is exactly a batched GEMM `L @ Rᵀ`.  We reshape the tensors to
   `(B*hidden_dim, N, N)` and use `torch.bmm`, which on H100 runs on
   Tensor‑Cores (FP16×FP16 → FP32 accumulation).
8️⃣ Output LayerNorm (FP16), scaling by `out_gate`, final bias‑free linear
   projection back to `dim`.
9️⃣ Cast back to FP32 for the API.

Only the gate‑multiplication step is implemented in Triton; all
matrix‑multiplications are delegated to cuBLAS, which is the fastest
solution for the large `N²×N` GEMMs required.  The fused linear reduces
memory‑bandwidth pressure dramatically, bringing the whole forward pass
well below the 1 ms target on the provided test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gate multiplications (no mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_kernel(
    left_ptr,                # fp16, [total]
    left_gate_ptr,           # fp16, [total]
    right_ptr,               # fp16, [total]
    right_gate_ptr,          # fp16, [total]
    N_TOTAL: tl.constexpr,   # total number of elements
    BLOCK_SIZE: tl.constexpr # work‑group size
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_TOTAL

    # Load values
    l = tl.load(left_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0).to(tl.float16)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0).to(tl.float16)

    # Apply gates
    l = l * lg
    r = r * rg

    # Store back in‑place
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def _apply_gates_triton(left: torch.Tensor,
                        right: torch.Tensor,
                        left_gate: torch.Tensor,
                        right_gate: torch.Tensor,
                        block_size: int = 1024):
    """
    Fuse `left *= left_gate` and `right *= right_gate` with a tiny Triton kernel.
    All tensors must be contiguous fp16 and have identical shape.
    The operation is performed in‑place for `left` and `right`.
    """
    assert left.shape == right.shape == left_gate.shape == right_gate.shape
    total = left.numel()
    grid = ( (total + block_size - 1) // block_size, )

    _gate_fuse_kernel[grid](
        left,
        left_gate,
        right,
        right_gate,
        total,
        BLOCK_SIZE=block_size,
        num_warps=4,        # 4 warps gives good occupancy for fp16
        num_stages=2,
    )
    # kernel writes directly into `left` and `right`
    return left, right


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension (D)
    hidden_dim = config["hidden_dim"]  # projection dimension (H)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 – high‑precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.half()  # [B, N, N, D] (fp16)

    # ------------------------------------------------------------------
    # 4) Fused linear projection + gate computation (single matmul)
    # ------------------------------------------------------------------
    # Concatenate all five weight matrices (out_features, in_features)
    # Order: left_proj, left_gate, right_proj, right_gate, out_gate
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # All are shape (hidden_dim, dim)
    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).contiguous().half()  # shape (5*H, D)

    # Single linear pass
    y = F.linear(x_f16, combined_weight)  # [B, N, N, 5*H] (fp16)

    # Split the concatenated output
    H = hidden_dim
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(y, H, dim=-1)

    # Apply sigmoid to gate tensors (in‑place to save memory)
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()      # shape [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 5) Fuse gate multiplication via Triton (in‑place)
    # ------------------------------------------------------------------
    # Ensure contiguous layout for the kernels
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    _apply_gates_triton(left_proj, right_proj, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcasted after gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] (bool/int). Convert to fp16 and expand.
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)       # [B, N, N, 1]
        mask_exp = mask_f16.expand(-1, -1, -1, H).contiguous()
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # ------------------------------------------------------------------
    # 7) Core N³ batched matrix multiply (outgoing update)
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape  # B, seq_len, seq_len, hidden_dim
    # Reshape to (B*H, N, N) where each hidden channel is an independent batch
    left_mat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: left @ rightᵀ  (FP16×FP16 → FP16, FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] (fp16)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (FP16), out‑gate scaling, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Scale by out_gate (broadcasted over hidden dim)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].half()
    out = F.linear(out, to_out_w)      # [B, N, N, D] (fp16)

    # ------------------------------------------------------------------
    # 9) Cast back to FP32 (API requirement)
    # ------------------------------------------------------------------
    return out.float()
