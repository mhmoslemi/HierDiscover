    start += L
    out_gate_pre = combined[..., start:start+L]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # apply mask and gates
    if not nomask and mask is not None:
        # mask to fp16
        mask_f16 = mask.to(torch.float16).contiguous()
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        total = left.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # core N^3 update
    B, N, _, _ = left.shape
    left_mat = left.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # [B,N,N,H]

    # output layernorm and gating
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate

    # final linear
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # [B,N,N,dim] fp16
    out = out.to(torch.float32)
    return out

"""
Triton‑accelerated outgoing TriMul (AlphaFold3) forward pass.

Algorithm
---------
1. Input LayerNorm (float32 – stable).
2. Cast to fp16 for all subsequent heavy work.
3. **Single fused linear**: concatenate the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) into one
   matrix of shape (dim, hidden_dim * 5) and perform a single GEMM.
   This reads the input only once and eliminates five separate kernel
   launches.
4. Split the result → left, right, left_gate_pre, right_gate_pre,
   out_gate_pre. Apply sigmoid to the three gate tensors.
5. **Mask‑gate fusion** (Triton):
   `left  = left  * left_gate  * mask`
   `right = right * right_gate * mask`
   All three element‑wise multiplications are fused into one memory‑
   bound kernel that streams through the tensors once.
   When `nomask=True` the kernel is skipped and gates are applied
   with an in‑place mul.
6. **Core N³ update** – batched matrix multiplication:
   for each head h ∈ [0, hidden_dim) compute
   `out[..., :, :, h] = left[..., :, :, h] @ right[..., :, :, h].T`.
   This is performed with a single `torch.bmm` on the view
   (B·hidden_dim, N, N), letting cuBLAS/Tensor‑cores do the heavy
   lifting.
7. Output LayerNorm (fp16) + output gate (element‑wise mul).
8. Final linear projection back to model dimension `dim`.
9. Cast back to float32 for the caller.

Only the mask‑gate step is implemented in Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# `mask` has shape [B, N, N] and is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 * left   (B,N,N,H)
    left_gate_ptr,       # fp16 * left_gate
    right_ptr,           # fp16 * right
    right_gate_ptr,      # fp16 * right_gate
    mask_ptr,            # fp16 * mask (B,N,N)
    total_elems: tl.int32,        # B*N*N*H
    HEAD_DIM: tl.constexpr,       # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Index of the mask element (broadcast over H)
    mask_idx = offs // HEAD_DIM

    # Load values (mask uses `other=1.0` when out‑of‑bounds, but mask
    # guarantees we never read beyond its size)
    l_val   = tl.load(left_ptr       + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr       + mask_idx, mask=mask, other=tl.float16(1.0))

    # Fuse three element‑wise multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights      : dict containing all required model parameters
        - config       : dict with keys ``dim``, ``hidden_dim``, optional
                        ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, D] (fp32)

    # --------------------------------------------------------------
    # 3) Cast to fp16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Single fused linear projection (+ gates) → one GEMM
    # --------------------------------------------------------------
    # Concatenate the five weight matrices along the output‑feature axis.
    # Each weight has shape (out_features, in_features) = (hidden_dim, dim)
    # After cat we obtain (5*hidden_dim, dim) which can be passed to
    # F.linear directly.
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(torch.float16)                         # (5*H, D)

    # One GEMM: (B*N*N, D) @ (D, 5*H) → (B*N*N, 5*H)
    combined = F.linear(x_f16, combined_weight)  # [B, N, N, 5*H]

    # --------------------------------------------------------------
    # 5) Slice the combined tensor + sigmoids for the three gates
    # --------------------------------------------------------------
    H = hidden_dim
    left          = combined[..., 0:H]
    right         = combined[..., H:2 * H]
    left_gate_pre = combined[..., 2 * H:3 * H]
    right_gate_pre = combined[..., 3 * H:4 * H]
    out_gate_pre  = combined[..., 4 * H:5 * H]

    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # --------------------------------------------------------------
    # 6) Optional mask & gate fusion (Triton)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → fp16, contiguous (no extra broadcast dim)
        mask_f16 = mask.to(torch.float16).contiguous()

        # Ensure contiguous tensors for pointer arithmetic
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                     # B * N * N * H  (int32)
        BLOCK = 1024                             # threads per program
        grid = ((total + BLOCK - 1) // BLOCK, )  # 1‑D grid

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,        # memory‑bound kernel – enough warps for latency hiding
        )
    else:
        # No mask – just apply the gates in‑place
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core N³ update – batched GEMM: L @ Rᵀ for every hidden channel
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # _ == hidden_dim
    # reshape to (B·H, N, N) – each head becomes a separate batch entry
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()

    # Batched matrix multiplication (tensor‑core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (fp16)

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # broadcasted element‑wise product

    # --------------------------------------------------------------
    # 9) Final linear projection back to model dimension D (fp16 → fp32)
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))  # [B,N,N,D] (fp16)
    out = out.to(torch.float32)                                      # required dtype

    return out
