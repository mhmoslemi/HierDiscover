"""
TriMul “outgoing” update – fast Triton‐accelerated forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32 → FP16).
2. Two bias‑free linear projections (left/right) give tensors L,R ∈
   [B, N, N, H] (H = hidden_dim) in FP16.
3. (Optional) binary mask is applied element‑wise to L and R.
4. Three sigmoid gates are obtained from the normalized input and
   multiplied into L, R and the final output.
5. Core cubic operation:
        out_{b,i,j,h} = Σ_k L_{b,i,k,h} * R_{b,j,k,h}
   which is a batched matrix‑multiply L @ Rᵀ for each hidden channel.
   We reshape (B,N,N,H) → (B·H, N, N) and run a custom Triton kernel that
   tiles the work with (BLOCK_M×BLOCK_N×BLOCK_K) = (128×128×64) (or
   64×64×64 for small N).  The kernel uses FP16 inputs, FP32 accumulation
   (tensor‑core friendly) and stores the result back in FP16.
6. Reshape back to [B, N, N, H] (still FP16).
7. Output LayerNorm (FP16), out‑gate multiplication and final linear
   projection back to the original dimension D.
8. Cast the result to FP32 for the API contract.

All heavy computation (the N³ mat‑mul) is performed in Triton; the rest
uses mixed‑precision PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
#   A, B : fp16 (shape: batch, M, K) and (batch, N, K)
#   C    : fp16 (shape: batch, M, N)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                 # pointers
    M, N, K,                             # matrix dimensions
    stride_ah, stride_am, stride_ak,     # A: (batch, M, K) strides
    stride_bh, stride_bn, stride_bk,     # B: (batch, N, K) strides
    stride_ch, stride_cm, stride_cn,     # C: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (flattened B·H)

    # Compute offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load A tile (M × K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # Load B tile transposed (K × N)
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Dot product (FP32 accumulation)
        acc += tl.dot(a_tile, b_tile)

    # Store result (cast back to fp16)
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) as fp16.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, f"Inner dimensions must match (got {K} vs {Kb})"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps provides good occupancy on H100
        num_stages=4,         # double‑buffering for latency hiding
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys:
                               * dim (int) – input feature dimension D
                               * hidden_dim (int) – projection dimension H
                               * nomask (bool, optional)

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # D
    hidden_dim = config["hidden_dim"]       # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for the bulk of the work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – fp16 weights
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)    # [B, N, N, H]  (fp16)
    right = F.linear(x_f16, right_proj_w) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional binary mask (broadcasted)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 6) Gates (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8) Core N³ batched mat‑mul (outgoing update)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # Reshape to (B·H, N, N) – each hidden channel is an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Choose block sizes based on sequence length (heuristic)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B·H, N, N] (fp16)

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16, shape [B, N, N, D]

    # ------------------------------------------------------------------
    # 10) Cast back to FP32 for the external API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
