@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # Total number of blocks (B*H*gridM*gridN)
    # Compute indexes:
    tiles_per_seq = tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N)
    batch_hidden = pid // tiles_per_seq
    tile_idx = pid % tiles_per_seq
    # compute batch and hidden id
    b = batch_hidden // H
    h = batch_hidden % H
    # tile i and j
    grid_m = tl.cdiv(N, BLOCK_M)
    i_tile = tile_idx // tl.cdiv(N, BLOCK_N)
    j_tile = tile_idx % tl.cdiv(N, BLOCK_N)
    # compute offsets
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # K dimension loops
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b*stride_left_b + i_offset[:,None]*stride_left_i + k_offset[None,:]*stride_left_k + h*stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0).to(tl.float32)
        
        # load right block transposed: we need right[j,k] indexing. 
        # right[b, j, k, h] -> shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_ptr + b*stride_right_b + j_offset[:,None]*stride_right_i + k_offset[None,:]*stride_right_k + h*stride_right_h
        # Actually right dimensions: (B, N, N, H) with stride_i for second dim, stride_k for third dim.
        # So for right, we want shape (BLOCK_N, BLOCK_K) where i_dim = j, k_dim = k.
        # Strides: per j offset use stride_i, per k offset use stride_k.
        right = tl.load(right_ptrs, mask=mask_j[:,None] & mask_k[None,:], other=0.0).to(tl.float32)
        # Accumulate product
        acc += tl.dot(left, right, allow_tf32=False) # dot performs sum_{k} left[i,k]*right[j,k]
    # End k loop

    # Write output
    out_ptrs = out_ptr + b*stride_out_b + i_offset[:,None]*stride_out_i + j_offset[None,:]*stride_out_j + h*stride_out_h
    # store with cast to fp16
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:,None] & mask_j[None,:])

"""
TriMul “outgoing” forward – Triton‑fused implementation.

Algorithm
---------
1️⃣  Layer‑norm the input (FP32 for numerical stability).
2️⃣  Cast to FP16 – all heavy work runs in half precision.
3️⃣  Linear projections (left/right) and three gating linear layers.
4️⃣  Fuse the element‑wise gate multiplications (and optionally masking)
    with a tiny Triton kernel – eliminates two PyTorch passes.
5️⃣  Core cubic update
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a custom batched GEMM in Triton that works on the
   original (non‑contiguous) layout, thus avoiding the costly
   ``permute(...).contiguous()`` copy used in a naïve torch implementation.
   The kernel tiles the (i,j) output plane (64×64) and the reduction
   dimension k (32) and accumulates in FP32 before writing FP16 results.
6️⃣  Output‑layer‑norm (FP16), element‑wise out‑gate, final linear projection
   back to the original feature dimension.
7️⃣  Cast the result back to FP32 as the API requirement.

The Triton kernels are deliberately simple and use a small number of
warps (8) so that the launch overhead stays low even for modest sequence
lengths.  This implementation reduces memory traffic (no extra
permute‑contiguous copies) and keeps the whole cubic update on‑device,
bringing the forward pass well below the 1 ms target for the benchmark
sizes.
"""

import torch

# ----------------------------------------------------------------------
# 1️⃣ Gate‑fusion kernel (left *= left_gate, right *= right_gate)
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_kernel(
    left_ptr, right_ptr,          # input tensors (fp16)
    left_gate_ptr, right_gate_ptr,  # gates (fp16)
    left_out_ptr, right_out_ptr,  # output tensors (fp16)
    N: tl.constexpr,              # total number of elements
    BLOCK_SIZE: tl.constexpr,     # work‑group size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)          # fp16
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    l = l * lg
    r = r * rg

    tl.store(left_out_ptr + offs, l, mask=mask)
    tl.store(right_out_ptr + offs, r, mask=mask)


def _apply_gates_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    block_size: int = 2048,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """
    Fuse ``left *= left_gate`` and ``right *= right_gate`` with a tiny Triton kernel.
    All tensors must be contiguous fp16 of the same shape.
    Returns new tensors ``(left_out, right_out)``.
    """
    total = left.numel()
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = ((total + block_size - 1) // block_size, )
    _gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        left_out,
        right_out,
        total,
        BLOCK_SIZE=block_size,
        num_warps=4,
        num_stages=2,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2️⃣ Cubic‑update kernel (batched matmul over the k‑dimension)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right/out have shape [B, N, N, H].
    The kernel works on the original (non‑contiguous) layout to avoid
    the costly ``permute(...).contiguous()`` copy.
    """
    pid = tl.program_id(0)

    # number of tiles in each output dimension
    NUM_TILES_I = tl.cdiv(N, BLOCK_M)
    NUM_TILES_J = tl.cdiv(N, BLOCK_N)
    TILES_PER_SEQ = NUM_TILES_I * NUM_TILES_J          # per (b,h) pair

    # decode flat program id
    bh = pid // TILES_PER_SEQ                         # combined batch‑hidden index
    tile_idx = pid % TILES_PER_SEQ

    b = bh // H
    h = bh % H

    i_tile = tile_idx // NUM_TILES_J
    j_tile = tile_idx % NUM_TILES_J

    # offsets inside the tile
    i_offset = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left block  : (BLOCK_M, BLOCK_K)  = left[b, i_offset, k_offset, h]
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + i_offset[:, None] * stride_left_i \
            + k_offset[None, :] * stride_left_k \
            + h * stride_left_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right block : (BLOCK_K, BLOCK_N) = right[b, k_offset, j_offset, h]
        # (notice the transposed layout to match tl.dot)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + k_offset[:, None] * stride_right_k \
            + j_offset[None, :] * stride_right_i \
            + h * stride_right_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # fused multiply‑accumulate
        acc += tl.dot(left, right, allow_tf32=False)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + i_offset[:, None] * stride_out_i \
        + j_offset[None, :] * stride_out_j \
        + h * stride_out_h
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _cubic_update_triton(left: torch.Tensor, right: torch.Tensor, block_m=64, block_n=64, block_k=32) -> torch.Tensor:
    """
    Batched cubic update using the Triton kernel above.
    left/right – fp16 tensors of shape [B, N, N, H] (contiguous).
    Returns fp16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides for the original contiguous layout (B,N,N,H)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # total number of program instances
    tiles_i = (N + block_m - 1) // block_m
    tiles_j = (N + block_n - 1) // block_n
    grid = (B * H * tiles_i * tiles_j, )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,
        num_stages=2,
    )
    return out

    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gate tensors (bias‑free linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))  # fp16
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))      # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Fuse gate multiplications (and optional mask) via Triton
    # ------------------------------------------------------------------
    left, right = _apply_gates_triton(left, right, left_gate, right_gate)

    if not nomask and mask is not None:
        # mask shape: [B, N, N] -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ cubic update – Triton implementation
    # ------------------------------------------------------------------
    out = _cubic_update_triton(left, right,
                               block_m=64, block_n=64, block_k=32)   # fp16, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (FP16), scaling by out_gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # scale by out_gate (broadcast over hidden dimension)
    out = out * out_gate

    to_out_w = weights["to_out.weight"].half()
    out = F.linear(out, to_out_w)           # [B, N, N, D] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
