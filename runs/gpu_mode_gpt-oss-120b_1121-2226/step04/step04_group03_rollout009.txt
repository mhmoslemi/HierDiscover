"""
TriMul “outgoing” forward pass (αFold‑3 style) – highly‑optimized implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) → *x*.
2️⃣ Cast *x* to FP16 for all dense work.
3️⃣ Compute the two bias‑free linear projections (left/right) and the three
   sigmoid gates (left_gate, right_gate, out_gate) in FP16.
4️⃣ If a pair‑wise mask is provided, broadcast‑multiply it into *left* and *right*.
5️⃣ Apply element‑wise gating (left←left·left_gate, right←right·right_gate).
6️⃣ Rearrange tensors to shape (B·H, N, N) (H = hidden_dim) and launch a custom
   Triton kernel that evaluates

        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]

   i.e. a batched GEMM C = A @ Bᵀ with FP16 inputs, FP32 accumulation
   and FP16 write‑back.
   • Tile sizes:  (BLOCK_M, BLOCK_N, BLOCK_K) = (128, 128, 128)
   • 8 warps, 3 pipeline stages → excellent occupancy on an H100.
7️⃣ Reshape the result back to [B, N, N, H] (still FP16).
8️⃣ Second Layer‑Norm over the hidden dimension (FP16).
9️⃣ Multiply by *out_gate* and apply the final bias‑free linear projection
   (FP16 → original dim).
🔟 Cast the final tensor to float32 (API contract).

Only the cubic O(N³) multiplication is performed in Triton; everything else
uses highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (A,B : fp16,  C : fp16, acc fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N]   (fp16)
    b_ptr,               # right : [BH, N, N]   (fp16) – interpreted as (K,N)
    c_ptr,               # out   : [BH, N, N]   (fp16)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ where A,B∈fp16 and the accumulator is fp32."""
    pid_h = tl.program_id(axis=2)      # batch × hidden dimension
    pid_m = tl.program_id(axis=0)      # tile row
    pid_n = tl.program_id(axis=1)      # tile column

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator (kept entirely in registers)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (block‑wise)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # Prefetch the first K‑tile
    cur_k = tl.arange(0, BLOCK_K)
    a_ptrs = (a_ptr
              + pid_h * stride_ah
              + offs_m[:, None] * stride_ai
              + cur_k[None, :] * stride_ak)
    b_ptrs = (b_ptr
              + pid_h * stride_bh
              + cur_k[:, None] * stride_bk
              + offs_n[None, :] * stride_bn)

    a = tl.load(a_ptrs, mask=mask_m[:, None] & (cur_k[None, :] < K), other=0.0)
    b = tl.load(b_ptrs, mask=(cur_k[:, None] < K) & mask_n[None, :], other=0.0)

    # Iterate over remaining K‑tiles
    for _ in range(1, num_k_tiles):
        # ---- accumulate current tile -------------------------------------------------
        acc += tl.dot(a, b).to(tl.float32)

        # ---- pre‑fetch next tile ----------------------------------------------------
        cur_k = (tl.program_id(axis=0) * 0) + _ * BLOCK_K + tl.arange(0, BLOCK_K)  # recompute
        a_ptrs = (a_ptr
                  + pid_h * stride_ah
                  + offs_m[:, None] * stride_ai
                  + cur_k[None, :] * stride_ak)
        b_ptrs = (b_ptr
                  + pid_h * stride_bh
                  + cur_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bn)

        a = tl.load(a_ptrs, mask=mask_m[:, None] & (cur_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(cur_k[:, None] < K) & mask_n[None, :], other=0.0)

    # accumulate the final tile
    acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back in fp16 (halves write bandwidth)
    # ------------------------------------------------------------------
    c_ptrs = (c_ptr
              + pid_h * stride_ch
              + offs_m[:, None] * stride_ci
              + offs_n[None, :] * stride_cj)

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – stable accumulation
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) – fp16 → fp16
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"].half()
    right_proj_weight = weights["right_proj.weight"].half()

    left = F.linear(x_fp16, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [B, N, N, 1] and broadcast over H
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].half())
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].half())
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].half())
    )

    # Apply element‑wise gates in‑place
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to [B·H, N, N] for the batched GEMM
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → flatten batch×head
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim,
                                                          seq_len,
                                                          seq_len)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim,
                                                            seq_len,
                                                            seq_len)

    # Output buffer (fp16) for the GEMM
    out_flat = torch.empty((bs * hidden_dim, seq_len, seq_len),
                           dtype=torch.float16,
                           device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Launch the Triton kernel (batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128          # larger K tile → half the loop iterations

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # rows
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # cols
        bs * hidden_dim,                      # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,  # M
        seq_len,  # N
        seq_len,  # K
        # A strides (batch, i, k)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # Bᵀ strides (batch, k, j) – note the transpose of the last two dims
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # C strides (batch, i, j)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H] (still fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)      # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()                # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )  # fp16

    # ------------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate                       # element‑wise gate
    out = F.linear(out, weights["to_out.weight"].half())  # fp16 → dim

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Cast to float32 (API contract) and return
    # ------------------------------------------------------------------
    return out.float()
