@triton.jit
def mask_mul_kernel(left_ptr, right_ptr, mask_ptr, total_elems, H):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offset = start + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elems

    # compute mask index: since mask broadcast over H
    mask_idx = offset // H
    # load
    l = tl.load(left_ptr + offset, mask=mask)
    r = tl.load(right_ptr + offset, mask=mask)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
    # compute
    l = l * m
    r = r * m
    # store
    tl.store(left_ptr + offset, l, mask=mask)
    tl.store(right_ptr + offset, r, mask=mask)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. **Input LayerNorm** – float32 for stability.
2. **Cast** to FP16 for the heavy work.
3. **Fused linear projection** – concatenate the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) into a single
   `(5*hidden, dim)` matrix and perform one large GEMM.  The result is
   sliced back into the five tensors.
4. **Sigmoid gates** for the three gate tensors.
5. **Mask + gate fusion** – a tiny Triton kernel multiplies
   `left *= left_gate * mask` and `right *= right_gate * mask` in one pass
   (or just the gates when no mask is present).
6. **Core N³ update** – batched matrix multiply `L @ Rᵀ` (shape
   `(B*hidden, N, N)`).  All work stays in FP16 and uses Tensor‑Core‑
   accelerated cuBLAS.
7. **Output LayerNorm + output‑gate fusion** – a Triton kernel computes, for
   each `(b,i,j)` position, the mean/variance over the hidden dimension,
   applies the LayerNorm (`to_out_norm.weight/bias`), multiplies by the
   `out_gate`, and writes the result back in‑place.
8. **Final linear projection** back to the model dimension (`to_out.weight`).
9. Cast the result to float32 and return.

Only two Triton kernels are used (mask‑gate fusion and output‑norm‑gate
fusion); the rest relies on highly‑optimized cuBLAS kernels via
`torch.nn.functional.linear` and `torch.bmm`.  This reduces the number of
expensive GEMMs from five to one, cuts memory traffic, and keeps the
overall runtime well below the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask multiplication with left/right gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 * left
    left_gate_ptr,       # fp16 * left_gate
    right_ptr,           # fp16 * right
    right_gate_ptr,      # fp16 * right_gate
    mask_ptr,            # fp16 * mask  (shape: B*N*N, broadcast over H)
    total_elems: tl.int32,      # B*N*N*H
    HEAD_DIM: tl.constexpr,     # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # index of the mask element (broadcast over hidden dim)
    m_idx = offs // HEAD_DIM

    l_val   = tl.load(left_ptr      + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr     + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr+ offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr      + m_idx, mask=mask, other=tl.float16(1.0))

    # fused element‑wise multiplications
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: fuse output LayerNorm, to_out_norm, and out‑gate multiplication.
# Input & out_gate have shape (total_positions, H).
# to_out_norm.weight/bias have shape (H,).
# ----------------------------------------------------------------------
@triton.jit
def _out_norm_gate_fuse_kernel(
    out_ptr,                # fp16 * out  (to be overwritten with normed*gate)
    out_gate_ptr,           # fp16 * out_gate
    norm_weight_ptr,        # fp16 * to_out_norm.weight  (H,)
    norm_bias_ptr,          # fp16 * to_out_norm.bias    (H,)
    total_positions: tl.int32,
    H: tl.constexpr,
    EPS: tl.constexpr,      # small epsilon for numerical stability
):
    pid = tl.program_id(0)                 # one block per position
    if pid >= total_positions:
        return

    # thread index over the hidden dimension
    hid = tl.arange(0, H)                  # [0, H)
    base = pid * H
    off = base + hid

    # load the raw values
    val = tl.load(out_ptr + off, mask=hid < H, other=tl.float16(0.0))
    val_f32 = val.to(tl.float32)

    # ----- compute mean & variance (float32) -----
    sum_val = tl.sum(val_f32, axis=0)                  # scalar
    inv_H   = 1.0 / H
    mean    = sum_val * inv_H

    diff = val_f32 - mean
    var  = tl.sum(diff * diff, axis=0) * inv_H
    inv_std = 1.0 / tl.sqrt(var + EPS)                # scalar

    # ----- load LayerNorm parameters (fp16 → fp32) -----
    w = tl.load(norm_weight_ptr + hid, mask=hid < H, other=tl.float16(0.0)).to(tl.float32)
    b = tl.load(norm_bias_ptr   + hid, mask=hid < H, other=tl.float16(0.0)).to(tl.float32)

    # ----- normalize -----
    normed = diff * inv_std * w + b                     # fp32

    # ----- apply out‑gate -----
    gate = tl.load(out_gate_ptr + off, mask=hid < H, other=tl.float16(0.0)).to(tl.float32)
    out_f32 = normed * gate

    # write back as fp16
    tl.store(out_ptr + off, out_f32.to(tl.float16), mask=hid < H)


# ----------------------------------------------------------------------
# Entry‑point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32 – stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, D], dtype float32

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Concatenate all five projection weights (shape: (5*H, D))
    # ------------------------------------------------------------------
    w_left   = weights["left_proj.weight"]   # (H, D)
    w_right  = weights["right_proj.weight"]  # (H, D)
    w_lgate  = weights["left_gate.weight"]   # (H, D)
    w_rgate  = weights["right_gate.weight"]  # (H, D)
    w_outgate= weights["out_gate.weight"]    # (H, D)

    # concatenate along the out‑feature dimension
    weight_concat = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate],
        dim=0,
    ).to(torch.float16)                       # (5*H, D)

    # ------------------------------------------------------------------
    # 5) Single GEMM for all projections
    # ------------------------------------------------------------------
    proj = F.linear(x_f16, weight_concat)    # [B, N, N, 5*H]  (fp16)

    # ------------------------------------------------------------------
    # 6) Slice the concatenated result
    # ------------------------------------------------------------------
    H = hidden_dim
    left       = proj[..., 0:H]
    right      = proj[..., H:2*H]
    left_gate_raw  = proj[..., 2*H:3*H]
    right_gate_raw = proj[..., 3*H:4*H]
    out_gate_raw   = proj[..., 4*H:5*H]

    # ------------------------------------------------------------------
    # 7) Sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 8) Apply mask + gate (fused Triton kernel when mask is present)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask -> fp16, shape [B, N, N]
        mask_f16 = mask.to(torch.float16).contiguous()

        # ensure contiguous tensors for pointer arithmetic
        left       = left.contiguous()
        right      = right.contiguous()
        left_gate  = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                      # B*N*N*H
        BLOCK = 1024                               # threads per block
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – just apply the gates in‑place
        left  = left.mul_(left_gate)
        right = right.mul_(right_gate)

    # ------------------------------------------------------------------
    # 9) Core N³ update – batched GEMM  L @ Rᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # B, N, N, H
    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # batched matrix multiplication (fp16, Tensor‑core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 10) Fuse Output LayerNorm, to_out_norm and out_gate (Triton)
    # ------------------------------------------------------------------
    # needed parameters (all fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # make contiguous for pointer math
    out = out.contiguous()
    out_gate = out_gate.contiguous()

    total_pos = B * N * N                      # number of (b,i,j) positions
    grid = (total_pos, )
    _out_norm_gate_fuse_kernel[grid](
        out.data_ptr(),
        out_gate.data_ptr(),
        to_out_norm_w.data_ptr(),
        to_out_norm_b.data_ptr(),
        total_pos,
        H=hidden_dim,
        EPS=1e-5,
        num_warps=8,
    )
    # after the kernel, `out` already holds (LayerNorm(out) * out_gate)

    # ------------------------------------------------------------------
    # 11) Final linear projection back to model dimension D
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)           # [B, N, N, D], fp16

    # ------------------------------------------------------------------
    # 12) Cast back to float32 for the caller
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
