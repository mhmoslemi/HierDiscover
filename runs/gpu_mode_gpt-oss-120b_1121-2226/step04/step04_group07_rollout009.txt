@triton.jit
def _triamul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, N, H,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    H: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row i
    pid_n = tl.program_id(1)  # block col j
    pid_head = tl.program_id(2)  # combined batch*head idx

    # Compute global head_idx and batch_idx
    head_idx = pid_head % H
    batch_idx = pid_head // H

    # Offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Base pointers for this batch and head
    a_base = A_ptr + batch_idx * stride_a_batch + head_idx * stride_a_h
    b_base = B_ptr + batch_idx * stride_b_batch + head_idx * stride_b_h
    c_base = C_ptr + batch_idx * stride_c_batch + head_idx * stride_c_h

    # accumulators
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension (seq_len)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K
        # Load A tile
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        # masks for out-of-bounds
        mask_a = (offs_i[:, None] < N) & (k_off + tl.arange(0, BLOCK_K)[None, :] < N)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # Load B tile (transposed)
        b_ptrs = b_base + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_b_k + offs_j[None, :] * stride_b_j
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < N) & (offs_j[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    mask_c = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

"""
Triton‑accelerated “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1. Layer‑Norm the input tensor (FP32 → FP16 for the heavy part).
2. Fuse the four linear projections (left_proj, left_gate, right_proj,
   right_gate) *and* the out‑gate into a single mat‑mul:
        Y = X @ W_fused                # Y shape: [B, N, N, 5·H]
   where W_fused = concat([W_left, W_left_gate,
                           W_right, W_right_gate,
                           W_out_gate], dim=0).
3. Split Y, apply sigmoid to gates, mask (if present) and compute the
   gated left/right tensors.
4. Core cubic update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched mat‑mul C = A @ Bᵀ with A,B ∈ [B·H, N, N].
   A custom Triton kernel loads tiles of A and transposed B, uses
   FP16 inputs, FP32 accumulation and Tensor‑Core MMA (128×128×64
   tiles, 8 warps, 3 pipeline stages).  The kernel works directly on
   the original [B,N,N,H] layout, therefore no costly permute+contiguous
   copies are required.
5. Apply LayerNorm on the hidden dimension, multiply by the out‑gate,
   and a final bias‑free linear projection back to the original dimension.
6. Cast the result to FP32 (API contract).

The implementation fuses linear layers, removes memory‑intensive copies,
and speeds up the N³ update with a highly‑tuned Triton kernel.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ  (A/B shape: [B·H, N, N])
#   Input tensors are stored as [B, N, N, H] (row‑major).
#   The combined batch index `pid_z` is split into (batch, head).
# ------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,               # [B,N,N,H] pointers (fp16)
    B, N, H,                           # runtime sizes
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Block indices
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)          # block row   → i‑dimension
    pid_n = tl.program_id(1)          # block col   → j‑dimension
    pid_z = tl.program_id(2)          # combined (batch, head)

    # --------------------------------------------------------------
    # Derive batch index and head index
    # --------------------------------------------------------------
    head_idx = pid_z % H
    batch_idx = pid_z // H

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # Base pointers for this (batch, head)
    # --------------------------------------------------------------
    a_base = A_ptr + batch_idx * stride_a_batch + head_idx * stride_a_h
    b_base = B_ptr + batch_idx * stride_b_batch + head_idx * stride_b_h
    c_base = C_ptr + batch_idx * stride_c_batch + head_idx * stride_c_h

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K‑dimension (the third axis of the input tensors)
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = kk * BLOCK_K

        # ---- load A tile (M × K) -------------------------------------------------
        a_ptrs = a_base + offs_i[:, None] * stride_a_i \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        mask_a = (offs_i[:, None] < N) & (k_off + tl.arange(0, BLOCK_K)[None, :] < N)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)          # fp16

        # ---- load B tile (K × N)  – transposed view --------------------------------
        b_ptrs = b_base + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_j
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < N) & (offs_j[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)          # fp16

        # ---- dot product (FP32 accumulation) ---------------------------------------
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_N) in FP32

    # --------------------------------------------------------------
    # Write back result (cast to FP16)
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    mask_c = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched mat‑mul C = left @ rightᵀ  computed with the Triton kernel above.
    Input tensors:
        left, right : [B, N, N, H]   (fp16)
    Returns:
        out : [B, N, N, H]            (fp16)
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    # output tensor
    out = torch.empty_like(left)

    # strides in elements (not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # grid dimensions
    BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_j,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps = 256 threads, good for H100
        num_stages=3,         # double‑buffering
    )
    return out


# ------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys
                                "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused linear projections + gates + out‑gate
    # --------------------------------------------------------------
    # concatenate the five weight matrices (out‑gate is included)
    w_left = weights["left_proj.weight"]          # (H, D)
    w_left_gate = weights["left_gate.weight"]    # (H, D)
    w_right = weights["right_proj.weight"]       # (H, D)
    w_right_gate = weights["right_gate.weight"]  # (H, D)
    w_out_gate = weights["out_gate.weight"]      # (H, D)

    # shape → (5·H, D)
    fused_weight = torch.cat(
        [w_left, w_left_gate, w_right, w_right_gate, w_out_gate], dim=0
    ).to(torch.float16)

    # single linear pass
    fused = F.linear(x_f16, fused_weight)               # [B, N, N, 5·H]

    # split
    H = hidden_dim
    left_proj = fused[..., 0:H]
    left_gate = fused[..., H:2 * H]
    right_proj = fused[..., 2 * H:3 * H]
    right_gate = fused[..., 3 * H:4 * H]
    out_gate = fused[..., 4 * H:5 * H]

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # --------------------------------------------------------------
    # 6) Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # --------------------------------------------------------------
    # 7) Gated left/right tensors
    # --------------------------------------------------------------
    left = left_proj * left_gate          # [B, N, N, H]  fp16
    right = right_proj * right_gate       # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 8) Core N³ batched mat‑mul (outgoing update) via Triton
    # --------------------------------------------------------------
    # left/right are already in the layout expected by the kernel
    out = _tri_mul_outgoing_triton(left, right)   # fp16, shape [B,N,N,H]

    # --------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # fp16

    out = out * out_gate            # apply the out‑gate (fp16)

    # final linear projection back to original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)                         # fp16, shape [B,N,N,D]

    # --------------------------------------------------------------
    # 10) Cast back to FP32 (API contract) and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
