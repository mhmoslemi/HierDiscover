"""
TriMul “outgoing” forward (AlphaFold3) – Triton‑accelerated cubic update.

Algorithm
---------
1. Layer‑norm the input (FP32) → x.
2. Cast x to FP16 and compute bias‑free linear projections:
      left_raw  = x @ W_left      (shape [B,N,N,H])
      right_raw = x @ W_right
3. Apply optional binary mask (broadcasted) to the raw projections.
4. Compute three sigmoid gates from x (FP16):
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
5. Core cubic operation (batched GEMM):
      L = left_raw  * left_gate
      R = right_raw * right_gate
      out_raw = Σ_k L[:,i,k] * R[:,j,k]  for each hidden channel.
   This is a batched mat‑mul L @ Rᵀ.  It is performed by a custom Triton
   kernel that multiplies the raw and gate tensors on‑the‑fly, uses FP16
   arithmetic with FP32 accumulation and tensor‑core friendly tiling.
6. Reshape back to [B,N,N,H] (FP16).
7. Output LayerNorm (FP16) → out_norm, multiply by out_gate,
   and a final bias‑free linear projection back to the original dimension.
8. Cast the result to FP32 (API contract).

Only the heavy N³ mat‑mul is executed in Triton; everything else is done
with PyTorch (mixed precision) to keep the implementation simple yet fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched matmul with per‑channel gates
#   out = (A_raw * A_gate) @ (B_raw * B_gate)ᵀ
#   A_raw, A_gate, B_raw, B_gate : fp16   (shape: batch, M, K) and (batch, N, K)
#   out                           : fp16   (shape: batch, M, N)
# ------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    A_raw_ptr, A_gate_ptr,               # left side raw and gate
    B_raw_ptr, B_gate_ptr,               # right side raw and gate
    C_ptr,                               # output
    M, N, K,                             # matrix dimensions
    stride_ah, stride_am, stride_ak,     # A_raw strides (batch, M, K)
    stride_bh, stride_bn, stride_bk,     # B_raw strides (batch, N, K)
    stride_ch, stride_cm, stride_cn,     # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (flattened B·H)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch_ptr = A_raw_ptr + pid_b * stride_ah
    a_gate_batch_ptr = A_gate_ptr + pid_b * stride_ah
    b_batch_ptr = B_raw_ptr + pid_b * stride_bh
    b_gate_batch_ptr = B_gate_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (blocked)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ---- Load A tile (raw * gate) ----
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        a_gate_ptrs = a_gate_batch_ptr + offs_m[:, None] * stride_am \
                      + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_raw = tl.load(a_ptrs,   mask=mask_a, other=0.0)   # fp16
        a_gate = tl.load(a_gate_ptrs, mask=mask_a, other=0.0)
        a_tile = a_raw * a_gate                         # fp16

        # ---- Load B tile (raw * gate) transposed ----
        # B is stored as (batch, N, K); we need the K‑dim first for the dot.
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b_gate_ptrs = b_gate_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                      + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_raw = tl.load(b_ptrs,   mask=mask_b, other=0.0)   # fp16, shape (K, N)
        b_gate = tl.load(b_gate_ptrs, mask=mask_b, other=0.0)
        b_tile = b_raw * b_gate                             # fp16, shape (K, N)

        # ---- Dot product (FP32 accumulation) ----
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store result (cast to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_gated_triton(
    A_raw: torch.Tensor,
    A_gate: torch.Tensor,
    B_raw: torch.Tensor,
    B_gate: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication with per‑channel gates using the Triton
    kernel above.  All tensors must be fp16 and on the same CUDA device.
    Returns C = (A_raw * A_gate) @ (B_raw * B_gate)ᵀ  with shape
    (batch, M, N) and dtype fp16.
    """
    assert A_raw.is_cuda and B_raw.is_cuda
    assert A_raw.dtype == torch.float16 and B_raw.dtype == torch.float16
    assert A_gate.dtype == torch.float16 and B_gate.dtype == torch.float16

    batch, M, K = A_raw.shape
    _, N, Kb = B_raw.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A_raw.device)

    # strides (in elements)
    stride_ah, stride_am, stride_ak = A_raw.stride()
    stride_bh, stride_bn, stride_bk = B_raw.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _tri_mul_matmul_kernel[grid](
        A_raw,
        A_gate,
        B_raw,
        B_gate,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,    # H100 prefers many warps
        num_stages=4,   # pipeline depth
    )
    return C


# ------------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing) entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys:
                               * dim (int)          – input feature dimension D
                               * hidden_dim (int)   – projection dimension H
                               * nomask (bool, optional)

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].t().to(torch.float16)   # (D, H)
    right_proj_w = weights["right_proj.weight"].t().to(torch.float16)

    left_raw  = F.linear(x_f16, left_proj_w)   # [B, N, N, H] fp16
    right_raw = F.linear(x_f16, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Optional binary mask (broadcasted)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left_raw  = left_raw * mask_f16
        right_raw = right_raw * mask_f16

    # ------------------------------------------------------------------
    # 6) Gates (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].t().to(torch.float16)
    right_gate_w = weights["right_gate.weight"].t().to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].t().to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7) Core N³ batched mat‑mul (gated)
    # ------------------------------------------------------------------
    B, N, _, H = left_raw.shape   # B, seq_len, seq_len, hidden_dim

    # Bring hidden dim to the batch dimension for the kernel
    #   -> (B*H, N, N)
    left_raw_t   = left_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    left_gate_t  = left_gate.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_raw_t  = right_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_gate_t = right_gate.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Heuristic block sizes – larger blocks for bigger sequences
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_t = _batched_matmul_gated_triton(
        left_raw_t,
        left_gate_t,
        right_raw_t,
        right_gate_t,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )   # [B*H, N, N] (fp16)

    # Restore the original layout: [B, N, N, H]
    out = out_t.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (fp16), out‑gate multiplication,
    #    and final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    out = out * out_gate   # fp16

    to_out_w = weights["to_out.weight"].t().to(torch.float16)   # (H, D)
    out = F.linear(out, to_out_w)   # [B, N, N, D] fp16

    # ------------------------------------------------------------------
    # 9) Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
