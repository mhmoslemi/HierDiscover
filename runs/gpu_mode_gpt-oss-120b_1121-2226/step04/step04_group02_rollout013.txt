"""
TriMul “outgoing” update – Triton‑accelerated gating+mask fusion and
torch.einsum for the N³ contraction.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → fp16 for heavy work).
2. Linear projections (bias‑free) → L,R ∈ [B,N,N,H]  (H = hidden_dim).
3. Compute three sigmoid gates from the same X.
4. Fuse element‑wise mask (if present) and the left/right gates in a tiny
   Triton kernel:   L ← L * sigmoid(left_gate) * mask,
                    R ← R * sigmoid(right_gate) * mask.
5. Core cubic operation
        out_{b,i,j,h} = Σ_k L_{b,i,k,h} * R_{b,j,k,h}
   performed with a single torch.einsum – this avoids the expensive
   permute‑/reshape‑copy dance required by a manual batched gemm.
6. Output LayerNorm (FP16), out‑gate multiplication and final linear
   projection back to the original dimension.
7. Cast to FP32 for the API.

Only the gating+mask fusion is custom‑written in Triton (requirement:
at least one custom kernel).  All other compute‑intensive work uses
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,      # [total_elements]   FP16
    right_ptr,     # [total_elements]   FP16
    left_gate_ptr, # [total_elements]   FP16
    right_gate_ptr,# [total_elements]   FP16
    mask_ptr,      # [total_masks]      FP16 (flattened), only used when HAS_MASK=1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place element‑wise fusion:
       left  = left  * left_gate  * (mask   if present)
       right = right * right_gate * (mask   if present)
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Optional mask (shared across hidden dimension)
    if HAS_MASK:
        mask_idx = offs // hidden_dim               # index in flattened mask
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * m_val
        right = right * m_val

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """Launch the Triton kernel that fuses mask and gate multiplication.
       The tensors are modified in‑place."""
    total = left.numel()
    BLOCK = 1024  # good trade‑off between launch overhead and occupancy
    grid = ( (total + BLOCK - 1) // BLOCK, )

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1)        # [B * N * N]
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys "dim", "hidden_dim",
                               optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input feature dimension D
    hidden_dim = config["hidden_dim"]   # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (high‑precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid) – FP16 weights
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))    # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))   # fp16
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))    # fp16

    # ------------------------------------------------------------------
    # 6) Fuse mask (if any) and gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left,
        right,
        left_gate,
        right_gate,
        mask_f16,
        hidden_dim,
    )  # left/right now contain gated & masked values (in‑place)

    # ------------------------------------------------------------------
    # 7) Core N³ contraction (outgoing update) via einsum
    # ------------------------------------------------------------------
    # out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # torch.einsum implements this efficiently and avoids the costly
    # permute/reshape steps used in a manual batched GEMM.
    out = torch.einsum('bikd,bjkd->bijd', left, right)   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (FP16), out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Apply out‑gate (already fp16)
    out = out * out_gate

    # Final projection back to the original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # [B, N, N, D] fp16

    # ------------------------------------------------------------------
    # 9) Cast to FP32 for the API
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    return out
