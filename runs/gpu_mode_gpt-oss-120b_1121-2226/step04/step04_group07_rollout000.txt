"""
custom_kernel – Triton‑accelerated “outgoing” TriMul

Algorithm
---------
1.  Layer‑norm the input tensor (FP32) and cast to FP16.
2.  Fuse each projection + gate into a single mat‑mul:
        left  = (X·W_left)   * sigmoid(X·W_left_gate)
        right = (X·W_right)  * sigmoid(X·W_right_gate)
   The projection and gate weights are concatenated so that each side
   needs only one FP16 GEMM (torch linear – highly optimized).
3.  Apply the optional binary mask (broadcast over the hidden dimension).
4.  Core cubic update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication C = A @ Bᵀ where
       A = left   → (B·H, N, N)
       B = right  → (B·H, N, N)
   A custom Triton kernel performs the multiplication using
   fp16 loads, fp32 accumulation and Tensor‑Core MMA.
   Block sizes are chosen adaptively to the sequence length.
5.  Output LayerNorm (FP16), element‑wise multiplication by the
   out‑gate (sigmoid of a linear projection), and a final FP16 linear
   projection back to the original dimension.
6.  Cast the result to FP32 for the API contract.

Only the heavy N³ mat‑mul is implemented in Triton; everything else
leverages PyTorch’s fast fused kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
#  Triton kernel : batched mat‑mul   C = A @ Bᵀ
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr,               # [batch, M, K]   fp16
    B_ptr,               # [batch, N, K]   fp16
    C_ptr,               # [batch, M, N]   fp16 (output)
    M, N, K,
    stride_ah, stride_am, stride_ak,   # strides for A
    stride_bh, stride_bn, stride_bk,   # strides for B
    stride_ch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)               # batch dimension (B·H)
    pid_m = tl.program_id(1)               # output row tile
    pid_n = tl.program_id(2)               # output col tile

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # pointers to the start of the batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # ------------------- load A tile -------------------
        a_ptrs = a_batch + \
                 offs_m[:, None] * stride_am + \
                 (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)          # (BLOCK_M, BLOCK_K) fp16

        # ------------------- load B tile (transposed) -------------------
        b_ptrs = b_batch + \
                 (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)          # (BLOCK_K, BLOCK_N) fp16

        # ------------------- dot product (Tensor‑Core MMA) -------------------
        acc += tl.dot(a, b)      # acc is fp32

    # ------------------- write back C (cast to fp16) -------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int,
    BLOCK_N: int,
    BLOCK_K: int,
) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ with a Triton kernel.
    A : (batch, M, K)   FP16
    B : (batch, N, K)   FP16
    Returns C : (batch, M, N) FP16
    """
    assert A.is_cuda and B.is_cuda, "inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "FP16 required"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in elements (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    num_warps = 4 if max(M, N) <= 256 else 8
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C


# --------------------------------------------------------------
#  Public entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask         : torch.Tensor [B, N, N]    (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
                         "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, D = x.shape
    dim = cfg["dim"]                # D
    hidden = cfg["hidden_dim"]      # H
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) input LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                        # [B,N,N,D]  FP32
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) fused projection + gate for the left side
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (H, D)
    left_gate_w = weights["left_gate.weight"]          # (H, D)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    left_comb = F.linear(x_f16, left_comb_w)          # [B,N,N,2H] fp16
    left_proj = left_comb[..., :hidden]               # [B,N,N,H]
    left_gate = torch.sigmoid(left_comb[..., hidden:])# [B,N,N,H]

    # ------------------------------------------------------------------
    # 4) fused projection + gate for the right side
    # ------------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_comb = F.linear(x_f16, right_comb_w)        # [B,N,N,2H] fp16
    right_proj = right_comb[..., :hidden]
    right_gate = torch.sigmoid(right_comb[..., hidden:])

    # ------------------------------------------------------------------
    # 5) apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B,N,N,H] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 7) core N³ update: batched mat‑mul left @ rightᵀ
    #    reshape to (B·H, N, N) → each hidden channel is an independent batch
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # pick block sizes adaptively (helps for very small N)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )   # [B*H, N, N]  fp16

    # restore original layout: (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 8) out‑gate (sigmoid of a linear projection of the *original* normed x)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # (H, D)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))      # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 9) output LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )   # fp16

    # ------------------------------------------------------------------
    # 10) apply out‑gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 11) final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))   # fp16

    # ------------------------------------------------------------------
    # 12) cast to FP32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
