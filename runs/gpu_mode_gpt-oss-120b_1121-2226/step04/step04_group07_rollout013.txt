"""
custom_kernel – Triton‑accelerated “outgoing” TriMul

The implementation follows the reference PyTorch module but moves the
N³ update (the cubic mat‑mul) to a custom Triton kernel.  The heavy
copy that a naïve view/permute would introduce is avoided: the kernel
operates directly on the original layout `[B, N, N, H]` (where `H` is the
hidden dimension) and writes the result into the same layout, i.e.
hidden dimension stays innermost – this makes the subsequent LayerNorm
and linear projection cheap.

Algorithm
---------
1. Layer‑norm the input (FP32) → cast to FP16.
2. Fuse the two linear projections with their sigmoid gates:
      left  = (X·W_left ) * sigmoid(X·W_left_gate )
      right = (X·W_right) * sigmoid(X·W_right_gate)
   The projection and gate weights are concatenated so each side uses a
   single `F.linear`.
3. Optional binary mask (broadcast over the hidden dimension) is applied
   after the gates – the order is immaterial because the mask is 0/1.
4. Core update:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}
   This is a batched matrix multiplication `A @ Bᵀ` with `A,B ∈
   [B·H, N, N]`.  A custom Triton kernel performs the multiplication
   using FP16 inputs, FP32 accumulation and Tensor‑Core MMA
   (`BLOCK_M×BLOCK_N×BLOCK_K = 128×128×64`).  The kernel writes directly
   into the output tensor with shape `[B, N, N, H]` (hidden dim contiguous).
5. Output LayerNorm (FP16), multiplication by the out‑gate and final
   bias‑free linear projection back to the original dimension.
6. Cast the result to FP32 (public API contract).

Only the cubic kernel is written in Triton; everything else uses fast
PyTorch FP16 kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel:  C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
#   A,B : fp16 tensors with shape [B, N, N, H]
#   C   : fp16 tensor with shape [B, N, N, H]  (H is innermost)
# ------------------------------------------------------------------
@triton.jit
def _triton_outgoing_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (M = N = K)
    stride_ah, stride_am, stride_ak,         # strides for A  (batch, row, col)
    stride_bh, stride_bn, stride_bk,         # strides for B  (batch, row, col)
    stride_ch, stride_cm, stride_cn,         # strides for C  (batch, row, col)
    stride_a_h, stride_b_h,                  # hidden‑dim stride for A & B (normally 1)
    hidden_dim,                              # H  (runtime integer)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) Identify tiles
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_b = tl.program_id(2)          # combined batch = B * H

    # decode batch and hidden‑channel indices
    b_idx = pid_b // hidden_dim
    h_idx = pid_b % hidden_dim

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # -----------------
    # base pointers for this (batch, hidden) pair
    # -----------------
    a_batch = A_ptr + b_idx * stride_ah + h_idx * stride_a_h
    b_batch = B_ptr + b_idx * stride_bh + h_idx * stride_b_h
    # output base – hidden stride in C is 1, so we just add h_idx
    c_batch = C_ptr + b_idx * stride_ch + h_idx

    # accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # ---- A tile (M × K) ----
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ---- B tile (K × N) ----
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ---- MMA (FP32 accumulation) ----
        acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # 3) Write back (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_outgoing_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Perform the cubic outgoing update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    left/right: [B, N, N, H]  (fp16)
    Returns out:   [B, N, N, H]  (fp16)
    """
    assert left.is_cuda and right.is_cuda, "CUDA tensors required"
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)
    # Output tensor (same layout)
    out = torch.empty((B, N, N, H), dtype=torch.float16, device=left.device)

    # strides for A (left)
    stride_ah, stride_am, stride_ak, stride_a_h = left.stride()
    # strides for B (right)
    stride_bh, stride_bn, stride_bk, stride_b_h = right.stride()
    # strides for C (output)
    stride_ch, stride_cm, stride_cn, stride_c_h = out.stride()   # stride_c_h should be 1

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,                     # combined batch × hidden
    )
    _triton_outgoing_matmul_kernel[grid](
        left,
        right,
        out,
        N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_cm, stride_cn,
        stride_a_h, stride_b_h,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out


# ------------------------------------------------------------------
# Public entry point used by the testing harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys
                                "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Fuse projection + gate for the left side
    # --------------------------------------------------------------
    # concatenate projection and gate weights (bias‑free linear)
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, D)
    left_combined = F.linear(x_f16, left_combined_w)       # [B, N, N, 2H]

    left_proj = left_combined[..., :hidden_dim]                     # [B,N,N,H]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])      # [B,N,N,H]

    # --------------------------------------------------------------
    # 4) Same for the right side
    # --------------------------------------------------------------
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_combined = F.linear(x_f16, right_combined_w)             # [B, N, N, 2H]

    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # --------------------------------------------------------------
    # 5) Apply element‑wise gates
    # --------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 6) Optional binary mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask shape [B,N,N] → broadcast to [B,N,N,1]
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 7) Core N³ batched mat‑mul (outgoing update)
    # --------------------------------------------------------------
    out = _batched_outgoing_matmul_triton(
        left, right,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    # out‑gate (sigmoid of a linear projection of the original x)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B,N,N,H] fp16

    # to_out_norm (LayerNorm over the hidden dimension H)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16, hidden dim remains innermost

    # apply out‑gate
    out = out * out_gate

    # final linear projection back to original dimension D
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)    # [B,N,N,D]  fp16

    # --------------------------------------------------------------
    # 9) Cast to FP32 to satisfy the public API
    # --------------------------------------------------------------
    return out.to(torch.float32)
