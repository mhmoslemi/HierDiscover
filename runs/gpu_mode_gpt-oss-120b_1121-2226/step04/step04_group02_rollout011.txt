    M, N, K,

    BLOCK_K: tl.constexpr,

    num_pid_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_pid_m * num_pid_n)
    rem = pid % (num_pid_m * num_pid_n)
    pid_m = rem // num_pid_n
    pid_n = rem % num_pid_n

"""
TriMul “outgoing” forward pass – Triton‑accelerated

Algorithm
---------
1.  Layer‑norm on the 4‑D input (FP32 → high‑precision statistics).
2.  Cast to FP16 for the heavy work.
3.  Linear projections (bias‑free) → left/right tensors of shape [B,N,N,H].
4.  Three linear gates → sigmoid activations.
5.  Fuse mask (if present) and the two element‑wise gates in a tiny Triton
    kernel so that left/right are multiplied in‑place.
6.  Core cubic operation:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix‑multiply (left @ rightᵀ) for each hidden
   channel.  A custom Triton kernel performs the whole batched GEMM,
   avoiding the overhead of many small torch.bmm calls.
7.  Output LayerNorm (FP16), multiplication with the out‑gate and a final
    linear projection back to the original dimension.
8.  Cast to FP32 for the API output.

Only the point‑wise gate + mask fusion and the batched GEMM are written in
Triton – the rest uses highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Gate + mask fusion (left/right in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr,          # fp16  [total_elems]
    right_ptr,         # fp16  [total_elems]
    left_gate_ptr,     # fp16  [total_elems]
    right_gate_ptr,    # fp16  [total_elems]
    mask_ptr,          # fp16  [total_masks] – flattened, only if HAS_MASK=1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place  f = x * gate * mask (mask optional)."""
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # optional mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        # each hidden element shares the same mask entry → mask_idx = offs // hidden_dim
        mask_idx = offs // hidden_dim
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * m_val
        right = right * m_val

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # write back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """Launch the Triton kernel that fuses mask and gates (in‑place)."""
    total = left.numel()
    BLOCK = 1024  # works well for the present tensor sizes
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1).contiguous()
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# 2️⃣  Batched GEMM for the N³ core (left @ rightᵀ for every hidden slice)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    stride_az, stride_am, stride_ak,          # a: (batch, M, K)
    stride_bz, stride_bk, stride_bn,          # b: (batch, K, N)
    stride_cz, stride_cm, stride_cn,          # c: (batch, M, N)
    M, N, K,                                  # matrix sizes (per batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiply: C[b] = A[b] @ B[b]  (FP16→FP32 accumulation)."""
    pid = tl.program_id(0)                     # linear program id across all blocks

    # number of 2‑D blocks per matrix
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # each batch gets a contiguous sub‑grid of (num_pid_m * num_pid_n) blocks
    batch = pid // (num_pid_m * num_pid_n)
    rem = pid % (num_pid_m * num_pid_n)
    pid_m = rem // num_pid_n
    pid_n = rem % num_pid_n

    # --------------------------------------------------------------
    # offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the K dimension
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch * stride_az \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)

        # B block: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + batch * stride_bz \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # accumulate
        acc += tl.dot(a, b)       # FP16 × FP16 → FP32

    # --------------------------------------------------------------
    # write back
    # --------------------------------------------------------------
    c_ptrs = c_ptr + batch * stride_cz \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul(
    a: torch.Tensor,
    b: torch.Tensor,
) -> torch.Tensor:
    """
    Compute C = A @ B for tensors of shape (B, M, K) and (B, K, N)
    using the Triton kernel above.  All tensors must be FP16 and
    contiguous.
    """
    assert a.is_contiguous() and b.is_contiguous()
    assert a.dtype == torch.float16 and b.dtype == torch.float16

    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    # allocate output
    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # strides (in elements, not bytes)
    stride_az, stride_am, stride_ak = a.stride()
    stride_bz, stride_bk, stride_bn = b.stride()
    stride_cz, stride_cm, stride_cn = c.stride()

    # kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_pid_m = (M + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * num_pid_m * num_pid_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return c


# ----------------------------------------------------------------------
# 3️⃣  Full forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys "dim", "hidden_dim",
                               optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (bias‑free) – FP16 weights
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H] (fp16)
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 4) Gates (sigmoid) – FP16 weights
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H] (fp16)
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))  # [B, N, N, H] (fp16)
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B, N, N, H] (fp16)

    # --------------------------------------------------------------
    # 5) Fuse mask (if any) and the two element‑wise gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left,
        right,
        left_gate,
        right_gate,
        mask_f16,
        hidden_dim,
    )  # left/right are now gated (and masked) in‑place

    # --------------------------------------------------------------
    # 6) Core cubic operation – batched matmul with Triton
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # --------------------------------------------------------------
    B, N, _, H = left.shape                # B, seq_len, _, hidden_dim

    # reshape for batched GEMM: (B*H, N, N)  where the inner dimension K = N
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)      # (batch, M, K)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (batch, K, N)

    # high‑performance batched matmul (fp16 × fp16 → fp32 accumulation, result fp16)
    out_mat = _batched_matmul(left_mat, right_mat_T)  # (B*H, N, N) fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()      # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate multiplication and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # gate (fp16)

    # final linear projection back to original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # [B, N, N, D] (fp16)

    # --------------------------------------------------------------
    # 8) Cast to FP32 for the public API
    # --------------------------------------------------------------
    return out.to(torch.float32)
