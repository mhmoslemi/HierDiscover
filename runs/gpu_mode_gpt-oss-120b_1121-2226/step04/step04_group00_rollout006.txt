"""
Optimized Triton implementation of the “outgoing” TriMul block.

Algorithm
---------
1️⃣ Layer‑norm the input (float32).
2️⃣ Cast to fp16 – all heavy work runs in half precision.
3️⃣ Linear projections and gate logits are computed with a single
   torch.nn.functional.linear call each (highly optimized cuBLAS GEMM).
4️⃣ A tiny Triton kernel fuses:
      * element‑wise multiplication by the learned gates,
      * optional binary mask,
      * and re‑ordering from (B,N,N,H) → (B,H,N,N) (contiguous).
   This removes the costly `permute(...).contiguous()` copy of the original code.
5️⃣ The cubic update is a batched matrix multiplication:
      out[b,i,j,d] = Σₖ left[b,d,i,k] * right[b,d,j,k] .
   Using the reordered tensors we can invoke a single `torch.bmm`
   on a batch of size B·H, which lets cuBLAS exploit Tensor‑Cores.
6️⃣ Output LayerNorm (fp16), per‑position out‑gate scaling and final
   bias‑free linear projection back to the original dimension.
7️⃣ Cast the result back to float32 for the API.

The kernel in step 4 is the only custom Triton code; all large linear‑algebra
operations are delegated to cuBLAS for maximal throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse gating, optional mask and reorder to (B,H,N,N)
# ----------------------------------------------------------------------
@triton.jit
def _gate_reorder_kernel(
    left_proj_ptr,   # fp16  [B,N,N,H] projection
    right_proj_ptr,  # fp16  [B,N,N,H] projection
    left_gate_ptr,   # fp16  [B,N,N,H] sigmoid‑gate
    right_gate_ptr,  # fp16  [B,N,N,H] sigmoid‑gate
    mask_ptr,        # fp16  [B,N,N]   optional mask (all‑ones if unused)
    left_out_ptr,    # fp16  [B,H,N,N] output tensor (contiguous)
    right_out_ptr,   # fp16  [B,H,N,N] output tensor (contiguous)
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    HASMASK: tl.constexpr,          # compile‑time bool, true iff mask is used
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # flat index in (B,N,N,H)
    in_bounds = offs < (B * N * N * H)

    # ------------------------------------------------------------------
    # Decode flat index → (b,i,j,d)
    # ------------------------------------------------------------------
    stride_b = N * N * H
    stride_i = N * H
    stride_j = H

    b  = offs // stride_b
    rem = offs % stride_b
    i  = rem // stride_i
    rem = rem % stride_i
    j  = rem // stride_j
    d  = rem % stride_j          # same as rem % H

    # ------------------------------------------------------------------
    # Load the four operand scalars
    # ------------------------------------------------------------------
    l_val = tl.load(left_proj_ptr + offs,   mask=in_bounds)
    r_val = tl.load(right_proj_ptr + offs,  mask=in_bounds)
    lg   = tl.load(left_gate_ptr + offs,    mask=in_bounds)
    rg   = tl.load(right_gate_ptr + offs,   mask=in_bounds)

    # apply gates
    l_val = l_val * lg
    r_val = r_val * rg

    # ------------------------------------------------------------------
    # Optional binary mask (broadcast over H)
    # ------------------------------------------------------------------
    if HASMASK:
        # mask index = b * (N*N) + i * N + j  (no H dimension)
        mask_idx = b * (N * N) + i * N + j
        m = tl.load(mask_ptr + mask_idx, mask=in_bounds, other=1.0)
        l_val = l_val * m
        r_val = r_val * m

    # ------------------------------------------------------------------
    # Compute output offset for layout (B, H, N, N)
    # ------------------------------------------------------------------
    out_off = b * (H * N * N) + d * (N * N) + i * N + j

    tl.store(left_out_ptr + out_off,  l_val, mask=in_bounds)
    tl.store(right_out_ptr + out_off, r_val, mask=in_bounds)


def _apply_gates_and_reorder(
    left_proj,
    right_proj,
    left_gate,
    right_gate,
    mask,
    block_size: int = 1024,
):
    """
    Fuse `left = left_proj * mask * left_gate`,
          `right = right_proj * mask * right_gate`
    while simultaneously re‑ordering to (B, H, N, N) layout.
    All tensors are fp16 and have identical shape (B,N,N,H).
    Returns:
        left_out, right_out  – shape (B, H, N, N), contiguous.
    """
    B, N, _, H = left_proj.shape
    total = B * N * N * H

    left_out  = torch.empty((B, H, N, N), dtype=left_proj.dtype, device=left_proj.device)
    right_out = torch.empty((B, H, N, N), dtype=right_proj.dtype, device=right_proj.device)

    has_mask = mask is not None
    if not has_mask:
        # dummy mask of ones – never used when HASMASK=False
        mask_ptr = torch.empty((1,), dtype=left_proj.dtype, device=left_proj.device)
    else:
        mask_ptr = mask

    grid = ((total + block_size - 1) // block_size,)

    _gate_reorder_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_ptr,
        left_out,
        right_out,
        B,
        N,
        H,
        HASMASK=has_mask,
        BLOCK_SIZE=block_size,
        num_warps=8,
        num_stages=2,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 – high precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.half()  # [B, N, N, D] (fp16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # ------------------------------------------------------------------
    left_proj = F.linear(x_f16, weights["left_proj.weight"].half())
    right_proj = F.linear(x_f16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 5) Gate tensors (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_f16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_f16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_f16, weights["out_gate.weight"].half()))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcasted after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to fp16 – shape [B, N, N]
        mask_f16 = mask.to(torch.float16)
    else:
        mask_f16 = None  # signal to kernel that mask is unused

    # ------------------------------------------------------------------
    # 7) Fuse gate multiplication, optional mask, and reorder to (B,H,N,N)
    # ------------------------------------------------------------------
    left, right = _apply_gates_and_reorder(
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_f16,
        block_size=1024,
    )
    # Release intermediate tensors asap
    del left_proj, right_proj, left_gate, right_gate

    # ------------------------------------------------------------------
    # 8) Core N³ batched matrix multiply (outgoing update)
    # ------------------------------------------------------------------
    B, H, N, _ = left.shape  # now left.shape == (B, H, N, N)

    # reshape to (B*H, N, N) for a single torch.bmm call
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Batched GEMM: left @ rightᵀ   (FP16×FP16→FP16, FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (FP16), out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )  # fp16

    # Apply per‑position out‑gate
    out = out * out_gate

    # Final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"].half())  # [B,N,N,D] fp16

    # ------------------------------------------------------------------
    # 10) Cast back to FP32 (API requirement)
    # ------------------------------------------------------------------
    return out.float()
