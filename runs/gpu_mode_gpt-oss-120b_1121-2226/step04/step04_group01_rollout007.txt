"""TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1. Input LayerNorm in FP32 (stable) → half‑precision tensor.
2. Fuse the five linear transformations (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single GEMM.  This
   eliminates five separate reads of the huge `[B,N,N,dim]` tensor
   and reduces memory traffic.
3. Slice the fused output, apply a sigmoid to the three gate
   blocks.
4. If a mask is present, a tiny Triton kernel fuses the three
   element‑wise products:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   otherwise the gates are applied in‑place.
5. Core N³ update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched GEMM with `torch.bmm` on the reshaped
   tensors (tensor‑core friendly, FP16).
6. Output LayerNorm (FP16) + element‑wise `out_gate`.
7. Final projection `to_out` (FP16 → FP32 output).

Only the mask‑gate fusion is written in Triton; the remaining
operations are performed with highly‑optimised Torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# `mask` has shape [B, N, N] and is broadcast over the hidden dim.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 * left
    left_gate_ptr,       # fp16 * left_gate
    right_ptr,           # fp16 * right
    right_gate_ptr,      # fp16 * right_gate
    mask_ptr,            # fp16 * mask (shape B*N*N)
    total_elems: tl.int32,      # B*N*N*H
    HEAD_DIM: tl.constexpr,     # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each (b,i,k) location repeats HEAD_DIM times → divide to obtain mask idx
    mask_idx = offs // HEAD_DIM

    # Load values (masked loads use 0.0 / 1.0 defaults)
    l_val   = tl.load(left_ptr       + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr       + mask_idx, mask=mask, other=tl.float16(1.0))

    # Fuse three multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse the five linear layers into a single GEMM
    #    (left_proj, right_proj, left_gate, right_gate, out_gate)
    # ------------------------------------------------------------------
    # All weight tensors are stored as (out_features, in_features) = (H, D)
    # Concatenate along the out_features dimension.
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*H, D)
    ).to(torch.float16)

    # One linear pass -> shape (B, N, N, 5*H)
    fused = F.linear(x_f16, fused_weight)  # fp16

    # Slice the fused tensor
    fused = fused.view(*x.shape[:-1], 5, hidden_dim)  # (B,N,N,5,H)
    left      = fused[..., 0, :]          # (B,N,N,H)
    right     = fused[..., 1, :]          # (B,N,N,H)
    left_gate = torch.sigmoid(fused[..., 2, :])   # (B,N,N,H)
    right_gate = torch.sigmoid(fused[..., 3, :])  # (B,N,N,H)
    out_gate  = torch.sigmoid(fused[..., 4, :])   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()

        # Ensure contiguous memory layouts for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                # B*N*N*H (fits in int32)
        BLOCK = 1024                         # threads per program
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – simply apply the three gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core N³ update: batched GEMM L @ Rᵀ for each hidden channel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape            # H == hidden_dim
    # (B, H, N, N) → (B*H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # BMM with Tensor‑Core acceleration (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore shape (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + output‑gate (still fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to the model dimension D
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)                          # (B,N,N,D) fp16
    out = out.to(torch.float32)                            # required dtype

    return out
