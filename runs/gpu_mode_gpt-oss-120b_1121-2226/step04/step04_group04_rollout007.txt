def cast_fp16_to_fp32_kernel(inp_ptr, out_ptr, N,
                             BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _fp16_to_fp32_kernel(inp_ptr, out_ptr, N, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    val = tl.load(inp_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, val.to(tl.float32), mask=mask)

def _fp16_to_fp32(x: torch.Tensor) -> torch.Tensor:
    assert x.is_cuda and x.dtype == torch.float16
    n = x.numel()
    out = torch.empty_like(x, dtype=torch.float32)
    BLOCK = 1024
    grid = ((n + BLOCK - 1) // BLOCK,)
    _fp16_to_fp32_kernel[grid](
        x.view(-1),
        out.view(-1),
        n,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out

    """Triton-accelerated TriMul “outgoing” update.

    Implements the forward pass of AlphaFold3’s TriangularMultiplicativeUpdate
    (outgoing) using a fused linear projection for left/right gates and a
    Triton kernel for the final fp16 → fp32 cast. The heavy cubic contraction
    is performed with a batched GEMM (torch.bmm) which maps to cuBLAS tensor‑core
    kernels.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    nomask = config.get("nomask", True)

    # 1. Input layer norm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] FP32

    # 2. Cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------

            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],

    right_proj = all_out[..., split1:split2]

    right = right_proj * right_gate
