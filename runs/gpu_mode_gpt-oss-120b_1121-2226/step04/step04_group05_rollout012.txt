"""
TriMul “outgoing” update – highly‑optimized forward pass.

Key ideas
---------
1. Fuse all five bias‑free Linear layers (left/right projection,
   left/right/out gates) into a **single** F.linear call by concatenating
   their weight matrices.  This reduces the number of passes over the
   massive input tensor from five to one, cutting memory traffic
   dramatically.

2. Perform the cubic ∑_k L_{i,k,h}·R_{j,k,h} using a custom Triton kernel
   that carries out a batched matrix‑multiply (B·H) × (N×N).  The kernel
   tiles the work with (BLOCK_M×BLOCK_N×BLOCK_K) = (128×128×64) and
   uses FP16 inputs with FP32 accumulation (Tensor‑Core friendly).

3. All remaining operations (LayerNorms, gating, final projection) are
   kept in mixed‑precision FP16, with the final result cast to FP32 as
   required by the API.

The implementation respects the required signature `custom_kernel(data)`
and works for the full range of test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K)ᵀ → C (M×N)
#   A : (batch, M, K)  – fp16
#   B : (batch, N, K)  – fp16 (will be read as K×N)
#   C : (batch, M, N)  – fp16 (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions
    stride_ah, stride_am, stride_ak,    # A strides (batch, M, K)
    stride_bh, stride_bn, stride_bk,    # B strides (batch, N, K)
    stride_ch, stride_cm, stride_cn,    # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (flattened B·H)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load a tile of A: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)      # fp16

        # Load a tile of B as (BLOCK_K, BLOCK_N) i.e. Bᵀ
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)      # fp16

        # Dot product (fp16×fp16 → fp32 accumulate)
        acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # Write back the result (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper for the Triton batched matmul kernel.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) in fp16.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must reside on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,
        num_stages=4,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * dim (int)          – input feature dimension D
                * hidden_dim (int)   – projection dimension H
                * nomask (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy path
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse the five bias‑free linear layers into one
    # ------------------------------------------------------------------
    # weights have shape (hidden_dim, dim) – concatenate along out_features
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)   # (5*H, D)

    # Single linear pass → shape [B, N, N, 5*H]
    combined = F.linear(x_f16, fused_weight)   # fp16

    # Split into the five tensors (views, no copy)
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        combined, hidden_dim, dim=-1
    )

    # ------------------------------------------------------------------
    # 5) Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 6) Apply gates to the projections
    # ------------------------------------------------------------------
    left = left_proj * left_gate           # [B, N, N, H] (fp16)
    right = right_proj * right_gate        # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 7) Optional binary mask (broadcasted)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 8) Core cubic update: out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Rearrange to (B*H, N, N) so each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Choose block sizes – slight heuristic based on sequence length
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B*H, N, N] (fp16)

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate               # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)       # fp16 → [B, N, N, D]

    # ------------------------------------------------------------------
    # 10) Cast back to FP32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
