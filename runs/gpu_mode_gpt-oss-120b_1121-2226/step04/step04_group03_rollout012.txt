"""
Optimized TriMul “outgoing” forward pass.

The kernel implements a batched GEMM C = A @ Bᵀ where
A = left projection (fp16) and B = right projection (fp16).  Each batch‑hidden
pair (BH) is processed independently (grid axis 2).  The GEMM is tiled with
M‑tile = N‑tile = 64 and K‑tile = 64.  Double‑buffering hides the latency of
global loads: the next K‑tile is prefetched while the previous one is
accumulated.  Accumulation uses FP32 registers and the final result is stored
as FP16, allowing tensor‑core execution on H100.  The rest of the TriMul
pipeline (LayerNorm, linear projections, gating, final projection) is kept in
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM out = left @ rightᵀ
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left:  [BH, N, N]   (fp16)
    b_ptr,               # right: [BH, N, N]   (fp16) – interpreted as (K,N)
    c_ptr,               # out:   [BH, N, N]   (fp16)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM with fp32 accumulation.
    Each program processes one (BH, tile_i, tile_j) block.
    Double‑buffered K‑loop is used to overlap loads with the dot product.
    """
    pid_h = tl.program_id(axis=2)            # batch‑hidden index
    pid_m = tl.program_id(axis=0)            # tile row
    pid_n = tl.program_id(axis=1)            # tile column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator (one register per output element)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop with double‑buffering
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # Load first K‑tile
    k = tl.zero()
    cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = cur_k < K

    a_ptrs = (
        a_ptr
        + pid_h * stride_ah
        + offs_m[:, None] * stride_ai
        + cur_k[None, :] * stride_ak
    )
    a = tl.load(
        a_ptrs,
        mask=mask_m[:, None] & mask_k[None, :],
        other=0.0,
    )

    b_ptrs = (
        b_ptr
        + pid_h * stride_bh
        + cur_k[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )
    b = tl.load(
        b_ptrs,
        mask=mask_k[:, None] & mask_n[None, :],
        other=0.0,
    )

    # advance to the next tile index
    k = k + 1

    # Main loop: pre‑fetch next tile while accumulating the current one
    while k < num_k_tiles:
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Prefetch next A‑tile
        a_ptrs_next = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a_next = tl.load(
            a_ptrs_next,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Prefetch next B‑tile
        b_ptrs_next = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b_next = tl.load(
            b_ptrs_next,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate current tile (fp16×fp16 → fp32)
        acc += tl.dot(a, b).to(tl.float32)

        # Move to next tile
        a = a_next
        b = b_next
        k = k + 1

    # Accumulate the last tile
    acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Store result (fp32 → fp16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------

    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] (fp16)

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # 4️⃣ Optional pair‑wise mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5️⃣ Gating – sigmoid (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # Apply elementwise gates in‑place
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Reshape to (B·H, N, N) for the batched GEMM
    # --------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    BLOCK_N = 64
    BLOCK_K = 64

        (seq_len + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                                   # batch × hidden
    )

        left_flat,
        right_flat,
        out_flat,
        seq_len,  # M
        seq_len,  # N
        seq_len,  # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right – note the transposed layout)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, H] (still fp16)
    # --------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # 🔟 Second LayerNorm over hidden dimension (fp16)
    # --------------------------------------------------------------

        normalized_shape=(hidden_dim,),

    # --------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].half())

    # --------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 (API contract)
    # --------------------------------------------------------------
    return out.float()

"""
Optimized TriMul “outgoing” forward pass.

The heavy part of the operation is a batched GEMM:
    out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]
where H = hidden_dim.  This is implemented as a Triton kernel that
computes C = A @ Bᵀ with FP16 operands, FP32 accumulation and FP16
write‑back, thus fully exploiting Tensor‑cores on an H100.  The kernel
uses a 64×64×64 tiling scheme and a double‑buffered K‑loop to hide
global‑memory latency.  All surrounding ops (LayerNorm, projections,
sigmoid gates, final projection) are performed with highly‑optimized
torch kernels.  The final tensor is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (A, B : fp16;  C : fp16;  acc : fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N]   (fp16)
    b_ptr,               # right : [BH, N, N]   (fp16) – interpreted as (K,N)
    c_ptr,               # out   : [BH, N, N]   (fp16)
    M, N, K,             # problem sizes (all equal to seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM with double‑buffered K‑loop.

    Each program processes a tile (BLOCK_M × BLOCK_N) for one
    (batch × hidden) slice.  The loop over K is tiled with size BLOCK_K.
    FP16 × FP16 → FP32 accumulation uses tensor‑cores.
    """
    pid_h = tl.program_id(axis=2)      # batch * hidden
    pid_m = tl.program_id(axis=0)      # tile row
    pid_n = tl.program_id(axis=1)      # tile column

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator (kept in registers)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Double‑buffered K‑loop
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # ---- Load first K‑tile ------------------------------------------------
    k = tl.zero()
    cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
    mask_k = cur_k < K

    a_ptrs = (
        a_ptr
        + pid_h * stride_ah
        + offs_m[:, None] * stride_ai
        + cur_k[None, :] * stride_ak
    )
    a = tl.load(
        a_ptrs,
        mask=mask_m[:, None] & mask_k[None, :],
        other=0.0,
    )                       # fp16

    b_ptrs = (
        b_ptr
        + pid_h * stride_bh
        + cur_k[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )
    b = tl.load(
        b_ptrs,
        mask=mask_k[:, None] & mask_n[None, :],
        other=0.0,
    )                       # fp16

    k = k + 1

    # ---- Main K‑loop ------------------------------------------------------
    while k < num_k_tiles:
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Prefetch next A‑tile
        a_ptrs_next = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a_next = tl.load(
            a_ptrs_next,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Prefetch next B‑tile
        b_ptrs_next = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b_next = tl.load(
            b_ptrs_next,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate current tile (FP16×FP16 → FP32)
        acc += tl.dot(a, b).to(tl.float32)

        # Advance buffers
        a = a_next
        b = b_next
        k = k + 1

    # ---- Accumulate the last tile -----------------------------------------
    acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write back (FP32 → FP16)
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )
