import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elemwise_mul_inplace(a_ptr, b_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(a_ptr + offs, a * b, mask=mask)

def _elemwise_mul_inplace_torch(a: torch.Tensor, b: torch.Tensor):
    assert a.is_cuda and b.is_cuda
    assert a.shape == b.shape
    a_flat = a.view(-1)
    b_flat = b.view(-1)
    n = a_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    _elemwise_mul_inplace[grid](
        a_flat,
        b_flat,
        n,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return a

def custom_kernel(data):
    """Highly‑optimized forward pass of the TriMul “outgoing” module.

    The implementation fuses the four linear projections (left_proj,
    left_gate, right_proj, right_gate) and the out‑gate projection into a
    single matrix multiplication, applies the optional binary mask, performs
    the cubic update via a batched matrix‐multiply, normalises the result,
    multiplies by the out‑gate using a tiny Triton kernel, and finally maps
    back to the original dimension.

    All heavy computation uses half‑precision (fp16) while the final
    result is returned in fp32 to match the reference implementation.
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 2) Cast to fp16 and fuse the four projections + out‑gate
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # Concatenate the five weight matrices (left_proj, left_gate,
    # right_proj, right_gate, out_gate) along the output dimension.
    # Shapes:
    #   left_proj   : (H, D)
    #   left_gate   : (H, D)
    #   right_proj  : (H, D)
    #   right_gate  : (H, D)
    #   out_gate    : (H, D)
    # Resulting shape -> (5*H, D)
    comb_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # Single linear projection: (B,N,N,D) @ (5*H,D)^T -> (B,N,N,5*H)
    combined = F.linear(x_f16, comb_weight)  # no bias

    # --------------------------------------------------------------
    # 3) Split the combined tensor
    # --------------------------------------------------------------
    H = hidden_dim
    left_proj = combined[..., :H]
    left_gate = combined[..., H : 2 * H]
    right_proj = combined[..., 2 * H : 3 * H]
    right_gate = combined[..., 3 * H : 4 * H]
    out_gate_raw = combined[..., 4 * H : 5 * H]

    # --------------------------------------------------------------
    # 4) Optional binary mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] -> broadcast to [B, N, N, 1]
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # --------------------------------------------------------------
    # 5) Gating (sigmoid) and final left/right tensors
    # --------------------------------------------------------------
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    out_gate = torch.sigmoid(out_gate_raw)  # (B, N, N, H)

    # --------------------------------------------------------------
    # 6) Core cubic update: Σ_k left[..., i, k, h] * right[..., j, k, h]
    #    Implemented as batched matrix‑multiply L @ Rᵀ.
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched GEMM: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm (fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    # --------------------------------------------------------------
    # 8) Multiply by out‑gate (Triton kernel)
    # --------------------------------------------------------------
    out = _elemwise_mul_inplace_torch(out, out_gate)

    # --------------------------------------------------------------
    # 9) Final projection back to original dimension (fp16)
    # --------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),  # (D, H)
        bias=None,
    )  # (B, N, N, D) fp16

    # --------------------------------------------------------------
    # 10) Cast to FP32 for API compatibility
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    return out

    # Concatenate: left_proj, left_gate, right_proj, right_gate, out_gate
    # Shape after cat -> (5·H, D)
    comb_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # One large linear layer: (B,N,N,D) @ (5·H,D)ᵀ → (B,N,N,5·H)
    combined = F.linear(x_f16, comb_weight)   # no bias

    # --------------------------------------------------------------
    # 3) Split the combined tensor
    # --------------------------------------------------------------
    H = hidden_dim
    left_proj = combined[..., :H]                       # (B,N,N,H)
    left_gate = combined[..., H:2 * H]                  # (B,N,N,H)
    right_proj = combined[..., 2 * H:3 * H]             # (B,N,N,H)
    right_gate = combined[..., 3 * H:4 * H]             # (B,N,N,H)
    out_gate_raw = combined[..., 4 * H:5 * H]           # (B,N,N,H)

    # --------------------------------------------------------------
    # 4) Optional binary mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] for broadcasting
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # --------------------------------------------------------------
    # 5) Apply sigmoid gates
    # --------------------------------------------------------------
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate_raw)   # (B,N,N,H)

    # --------------------------------------------------------------
    # 6) Core cubic update: batched mat‑mul L @ Rᵀ
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, H
    # Reshape to (B·H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout → (B, N, N, H) in fp16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output LayerNorm (fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    # --------------------------------------------------------------
    # 8) Multiply by out‑gate (Triton kernel)
    # --------------------------------------------------------------
    out = _elemwise_mul_inplace(out, out_gate)

    # --------------------------------------------------------------
    # 9) Final linear projection back to original dimension (fp16)
    # --------------------------------------------------------------
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),   # (D, H)
        bias=None,
    )  # (B, N, N, D)

    # --------------------------------------------------------------
    # 10) Cast to fp32 for the public API
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    return out
```
