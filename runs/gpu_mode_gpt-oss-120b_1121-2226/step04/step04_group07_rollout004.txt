"""
custom_kernel – highly‑optimized Triton implementation of the “outgoing”
TriMul module (AlphaFold3).

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → float16 for speed).
2. Fuse each linear projection with its sigmoid gate via a single
   `F.linear` (weight matrix = [proj;gate]).
3. Apply the optional binary mask (broadcast over the hidden dimension).
4. Core cubic update:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication `C = A @ Bᵀ` with
   `A,B ∈ [B·H, N, N]`.  A custom Triton kernel performs the
   multiplication using FP16 inputs, FP32 accumulation and Tensor‑Core
   MMA.  An `@triton.autotune` decorator selects the best tiling
   configuration for the current shape.
5. Output LayerNorm (FP16), element‑wise out‑gate, final projection back
   to the original dimension.
6. Cast the result to float32 (the API contract).

The implementation keeps the heavy N³ mat‑mul in Triton while the
remaining operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ
#   A : (batch, M, K)   fp16
#   B : (batch, N, K)   fp16
#   C : (batch, M, N)   fp16 (written from fp32 accumulator)
# ------------------------------------------------------------------
@triton.autotune(
    configs=[
        # (M,N,K) = (128,128,64) – good default
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64},
                      num_stages=2, num_warps=8),
        # faster for very wide matrices
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 64},
                      num_stages=3, num_warps=8),
        # try a tall‑skinny configuration
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64, 'BLOCK_K': 64},
                      num_stages=3, num_warps=8),
        # try a short‑wide configuration
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 128, 'BLOCK_K': 64},
                      num_stages=3, num_warps=8),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                 # pointers
    M, N, K,                             # matrix sizes
    stride_ah, stride_am, stride_ak,     # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,     # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,     # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ."""
    pid_m = tl.program_id(0)   # block row index
    pid_n = tl.program_id(1)   # block column index
    pid_b = tl.program_id(2)   # batch index (B·H)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension with tiling
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # ------------------- Load A tile (M × K) -------------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ------------------- Load B tile (N × K) transposed -------------------
        # B is stored as (batch, N, K); we need the K × N view for Bᵀ
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ------------------- Compute tile contribution -------------------
        #   acc (FP32) += a_tile (FP16) • b_tile (FP16)  -> Tensor‑Core MMA
        acc += tl.dot(a_tile, b_tile)

    # ------------------- Write out C tile (cast to FP16) -------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper for the Triton batched mat‑mul kernel.
    A : (batch, M, K)  fp16
    B : (batch, N, K)  fp16
    Returns C : (batch, M, N) fp16
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Expect fp16 tensors"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides are in elements (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # heuristics: H100 works well with 8 warps & 2‑3 stages
        num_warps=8,
        num_stages=3,
    )
    return C


# ------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys "dim", "hidden_dim", optional
                                "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] fp32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the compute‑intensive parts
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused projection + gate (bias‑free)
    # --------------------------------------------------------------
    # --- left side ---
    left_proj_w = weights["left_proj.weight"]          # (H, D)
    left_gate_w = weights["left_gate.weight"]          # (H, D)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2H, D)
    left_combined = F.linear(x_f16, left_combined_w)   # [B, N, N, 2H] fp16
    left_proj = left_combined[..., :hidden_dim]                     # (B,N,N,H)
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])      # (B,N,N,H)
    left = left_proj * left_gate

    # --- right side ---
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_combined = F.linear(x_f16, right_combined_w)   # [B, N, N, 2H]
    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask is either bool or int; convert to 0/1 FP16
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 6) Core N³ update – batched mat‑mul (A @ Bᵀ)
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # B, N, N, H
    # reshape to (B·H, N, N)  – each hidden channel is an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Triton batched GEMM
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B·H, N, N] fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    # out‑gate (single linear + sigmoid)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B,N,N,H] fp16

    # LayerNorm on hidden dimension (FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # apply out‑gate
    out = out * out_gate

    # final linear projection back to original dimension (bias‑free)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16, shape [B,N,N,D]

    # --------------------------------------------------------------
    # 8) Cast back to FP32 for the public API
    # --------------------------------------------------------------
    return out.to(torch.float32)
