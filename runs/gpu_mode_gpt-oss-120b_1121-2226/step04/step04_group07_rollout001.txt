"""
custom_kernel – Triton‑accelerated “outgoing” TriMul

Algorithm
---------
1. **Layer‑norm the 4‑D input** (FP32 → FP16 for compute).
2. **Fused linear projections + sigmoid gates**  
   – Concatenate projection and gate weights so each side needs a single
     `F.linear` call.  
   – Apply mask (if present) and element‑wise multiplication with the
     corresponding gates.
3. **Core cubic update**  
   – For each hidden channel treat `left` and `right` as independent batches
     and compute `out = left @ rightᵀ` using a custom Triton kernel
     (`_batched_matmul_kernel`).  The kernel uses FP16 inputs, FP32
     accumulation and Tensor‑Core MMA (block size 128×128×64, 8 warps).
4. **Output LayerNorm, out‑gate and final projection**  
   – All remaining ops are cheap FP16 PyTorch kernels.
5. Cast the result back to FP32 (API contract).

Only the O(N³) matrix multiplication is performed in Triton; the rest
leverages highly‑optimized PyTorch primitives.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ
#   A, B : fp16   (batch, M, K) and (batch, N, K)
#   C    : fp16   (batch, M, N) – result is cast from fp32 accumulator.
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides (batch, M, K)
    stride_bh, stride_bn, stride_bk,    # B strides (batch, N, K)
    stride_ch, stride_cm, stride_cn,    # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # col tile
    pid_b = tl.program_id(2)   # batch tile (B·H)

    # Offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)      # fp16

        # B tile (transposed view): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)      # fp16

        # Tensor‑Core dot (fp16×fp16 → fp32)
        acc += tl.dot(a, b)

    # Store C (cast back to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper for the Triton kernel performing C = A @ Bᵀ.
    A, B: (batch, M, K) & (batch, N, K)  FP16
    Returns C: (batch, M, N)  FP16
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads, tuned for H100
        num_stages=2,
    )
    return C


# ------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys
                                "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 1) Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numerical safety)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused projection + gating (bias‑free)
    # --------------------------------------------------------------
    # left side
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2·H, D)
    left_combined = F.linear(x_f16, left_combined_w)          # [B,N,N,2H]
    left_proj = left_combined[..., :hidden_dim]               # [B,N,N,H]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])# [B,N,N,H]

    # right side
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_combined = F.linear(x_f16, right_combined_w)
    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # Apply gates
    left = left_proj * left_gate   # [B,N,N,H]
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5) Optional binary mask
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 6) Core cubic update – batched mat‑mul over hidden dimension
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B·H, N, N) → each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # A @ Bᵀ via Triton
    out_mat = _batched_matmul_triton(
        left_mat, right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B·H, N, N]  (fp16)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # [B,N,N,H] fp16

    # LayerNorm on hidden dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    out = out * out_gate

    # Final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16

    # --------------------------------------------------------------
    # 8) Cast back to FP32 (required by the API)
    # --------------------------------------------------------------
    return out.to(torch.float32)
