"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – highly optimized.

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (FP32) → x.
2️⃣  Cast to FP16 (the heavy part runs in half‑precision).
3️⃣  Linear projections (left/right) and three bias‑free sigmoid gates
    (left_gate, right_gate, out_gate).  The optional pair‑wise mask is applied
    before the element‑wise gates.
4️⃣  Rearrange tensors to a batched 3‑D shape (B·H, N, N) where H = hidden_dim.
5️⃣  Triton kernel computes, for every head *h*:

        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]

    i.e. a batched GEMM C = A @ Bᵀ  (FP16 inputs, FP32 accumulation, FP16 write‑back).
    Tile sizes (M,N,K) = (128,128,64) give high occupancy on H100.
6️⃣  Reshape back to [B, N, N, H] (still FP16).
7️⃣  Second Layer‑Norm over the hidden dimension (FP16).
8️⃣  Apply the output gate and final bias‑free linear projection back to
    the original dimension D (still FP16).
9️⃣  Cast to FP32 for the public API.

Only the N³ matrix multiplication is performed in Triton; all surrounding
operations use highly‑optimized torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B : fp16,  C : fp16, acc fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N]   (fp16)
    b_ptr,               # right : [BH, N, N]   (fp16) – interpreted as (K,N) = (N,N)ᵀ
    c_ptr,               # out   : [BH, N, N]   (fp16)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute batched matrix multiplication C = A @ Bᵀ.
    Each program instance works on a tile (BLOCK_M × BLOCK_N) of one
    batch‑head (BH = batch * hidden_dim).
    """
    pid_h = tl.program_id(axis=2)  # which head (batch * hidden)
    pid_m = tl.program_id(axis=0)  # tile row   (i)
    pid_n = tl.program_id(axis=1)  # tile col   (j)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (kept in registers)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ------- Load A tile (M × K) --------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ------- Load Bᵀ tile (K × N) -------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ------- Accumulate (FP16×FP16 → FP32) -----------------------
        acc += tl.dot(a, b)  # result is automatically promoted to FP32

    # --------------------------------------------------------------
    # Write back – store as FP16 to halve write bandwidth
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (bool / float)
        - weights_dict : mapping str → torch.Tensor with model weights
        - config_dict  : contains keys
            * "dim"          – input feature dimension D
            * "hidden_dim"   – hidden dimension H
            * "nomask" (optional) – if True, the mask can be ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the hot path
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] FP16

    # ------------------------------------------------------------------
    # 3️⃣ Bias‑free linear projections (FP16 → FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; we need [B, N, N, 1] to broadcast over H
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # In‑place scaling by gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to (B·H, N, N) for the batched GEMM
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim

    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    # Output buffer for the GEMM
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Launch the Triton kernel (batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # tiles over i
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # tiles over j
        BH,                                   # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right, interpreted as transposed)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H] (still FP16)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )  # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 🔟 Output gate & final linear projection (still FP16)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].half())  # → [B, N, N, D] fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
