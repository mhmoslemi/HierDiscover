"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (high‑level)
----------------------
1️⃣ Input LayerNorm (float32 – stable).
2️⃣ Cast to fp16 for the heavy work.
3️⃣ Linear projections (no bias) and sigmoid gates (all fp16).
4️⃣ Fuse mask‑multiplication with the three element‑wise products
   (left * left_gate * mask, right * right_gate * mask) in a tiny
   Triton kernel – avoids multiple launch overheads.
5️⃣ Core N³ update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a *batched* matrix multiplication using a custom
   Triton kernel that works directly on the original `[B,N,N,H]`
   layout – no costly transpose / copy.  The kernel treats each
   `(b, h)` pair as an independent matrix and computes a tiled GEMM.
6️⃣ Output LayerNorm + output‑gate (still fp16).
7️⃣ Final linear projection back to the model dimension `dim`
   (fp16 → fp32 output).

The Triton kernels cover the mask‑gate fusion (step 4) and the
batched GEMM (step 5).  All other operations are performed with
high‑performance PyTorch kernels (layer‑norm, linear, sigmoid,
final projection).  The implementation respects the required
signature `custom_kernel(data)` and works for all test configurations
including optional masks and multiple hidden‑dim sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask multiplication with gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 *
    left_gate_ptr,     # fp16 *
    right_ptr,         # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 *  (shape: B * N * N)
    total_elems: tl.int32,
    HEAD_DIM: tl.constexpr,   # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # One mask element per H values (mask is broadcast over the hidden dim)
    mask_idx = offs // HEAD_DIM

    l_val  = tl.load(left_ptr       + offs, mask=mask, other=tl.float16(0.0))
    l_gate = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    r_val  = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val  = tl.load(mask_ptr       + mask_idx, mask=mask, other=tl.float16(1.0))

    # three fused multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: batched N³ update without any transpose / copy.
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
#   Left/right are stored as [B, N, N, H] with H as the innermost
#   dimension (contiguous).  The kernel treats each (b, h) pair as
#   a separate (N×N) matrix and computes a tiled GEMM.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,   # fp16 *
    right_ptr,  # fp16 *
    out_ptr,    # fp16 *
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    BLOCK_M: tl.constexpr,  # tile size for rows   (i)
    BLOCK_N: tl.constexpr,  # tile size for cols   (j)
    BLOCK_K: tl.constexpr,  # tile size for reduction (k)
):
    # program ids (3‑D launch grid)
    pid_bh = tl.program_id(0)   # combined batch‑head dimension
    pid_m  = tl.program_id(1)   # i‑tile
    pid_n  = tl.program_id(2)   # j‑tile

    # decode (b, h) from the linear batch‑head index
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # Base pointers for this (b, h) slice.
    #   offset = b * (N * N * H) + h   (H is innermost)
    # ------------------------------------------------------------------
    slice_offset = b * (N * N * H) + h
    a_base = left_ptr  + slice_offset
    b_base = right_ptr + slice_offset
    c_base = out_ptr   + slice_offset

    # Strides for the *contiguous* layout [B, N, N, H]
    #   row‑stride (i) = N * H
    #   col‑stride (k) = H
    STRIDE_ROW = N * H
    STRIDE_COL = H

    # ------------------------------------------------------------------
    # Tile start indices
    # ------------------------------------------------------------------
    off_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = off_i < N
    mask_j = off_j < N

    # accumulator in FP32 (more accurate, still fast)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        off_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # pointers for the current K‑tile
        # left:   A[i, k]   → row‑stride = STRIDE_ROW, col‑stride = STRIDE_COL
        # right:  Bᵀ[k, j] → row‑stride = STRIDE_COL, col‑stride = STRIDE_ROW
        a_ptrs = a_base + off_i[:, None] * STRIDE_ROW + off_k[None, :] * STRIDE_COL
        b_ptrs = b_base + off_k[:, None] * STRIDE_COL + off_j[None, :] * STRIDE_ROW

        # masked loads (out‑of‑bounds elements are treated as 0)
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=tl.float16(0.0))
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=tl.float16(0.0))

        # dot‑product for the K‑tile
        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write back the result (still FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_base + off_i[:, None] * STRIDE_ROW + off_j[None, :] * STRIDE_COL
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)


# ----------------------------------------------------------------------
# Entry‑point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the *outgoing* TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (float32 – stable)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to fp16 for the expensive linear work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Cast all needed parameters to fp16 once
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 5) Linear projections + sigmoid gates (all fp16)
    # --------------------------------------------------------------
    left      = F.linear(x_f16, left_proj_w)                 # [B, N, N, H]
    right     = F.linear(x_f16, right_proj_w)                # [B, N, N, H]

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w)) # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B, N, N, H]

    # --------------------------------------------------------------
    # 6) Optional mask + gate fusion (Triton)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is bool/int → convert to fp16 (0.0 / 1.0)
        mask_f16 = mask.to(torch.float16).contiguous()
        # ensure contiguous on the tensors we will modify in‑place
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                     # B·N·N·H (fits in int32)
        BLOCK = 1024                               # 1‑D launch (≤ 1024 threads)
        grid = ((total + BLOCK - 1) // BLOCK, )

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask → just fuse gates in‑place
        left  = left.mul_(left_gate)
        right = right.mul_(right_gate)

    # --------------------------------------------------------------
    # 7) Core N³ update – custom Triton batched GEMM.
    #    Operates directly on the layout [B, N, N, H] → no copy.
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    out = torch.empty_like(left)   # [B, N, N, H], fp16

    # Block sizes – tuned for H100 fp16 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden dim
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles along j
    )

    _trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate      # element‑wise product, broadcast over dim

    # --------------------------------------------------------------
    # 9) Final linear projection back to D (fp16 → fp32)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # fp16 result
    out = out.to(torch.float32)     # required output dtype

    return out
