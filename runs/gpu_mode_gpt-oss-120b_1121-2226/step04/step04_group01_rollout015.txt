def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    
    bh = tl.program_id(2)
    b = bh // H
    h = bh % H

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # offsets for inputs with fixed b,h

def mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    size_left: tl.int32,    # total number of elements in left (B*N*N*H)
    H: tl.constexpr,        # hidden dim
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offsets = start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < size_left

    # Determine mask index: each hidden slice repeats mask across H elements
    mask_idx = offsets // H

    # load
    l_val = tl.load(left_ptr + offsets, mask=mask, other=tl.float16(0.0))
    l_gate = tl.load(left_gate_ptr + offsets, mask=mask, other=tl.float16(0.0))

    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr + offsets, l_out, mask=mask)
    tl.store(right_ptr + offsets, r_out, mask=mask)

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    # derive batch and hidden indices
    b = pid_hb // H
    h = pid_hb % H

    # compute start offsets for i and j
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range offsets
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    # masks for i/j out-of-bound
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        # left pointer: base + b*stride_l_b + h*stride_l_h + offs_i[:,None]*stride_l_i + offs_k[None,:]*stride_l_k
        left_ptrs = left_ptr + b * stride_l_b + h * stride_l_h + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None,:], other=tl.float16(0.0))

        # right tile transposed: shape (BLOCK_K, BLOCK_N)
        # right pointer: base + b*stride_r_b + h*stride_r_h + offs_k[:, None]*stride_r_k + offs_j[None,:]*stride_r_i
        right_ptrs = right_ptr + b * stride_r_b + h * stride_r_h + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_i)
        right_t = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None,:], other=tl.float16(0.0))

        # dot product accumulate
        acc += tl.dot(left, right_t)

    # store acc to out
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_k)
    # store only where i,j within bounds
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. **LayerNorm on the input** (float32 for numerical stability).
2. Cast the normalized tensor to **FP16** – all heavy work stays in half‑precision.
3. **Fuse projection + gate** for *left* and *right*:
   * Concatenate the projection and gate weight matrices → one GEMM per side.
   * Slice the result in‑place to obtain `proj` (weight) and `gate` (bias‑free).
   * Apply a sigmoid to the gate part.
4. **Mask & gate fusion** – a tiny Triton kernel computes
        proj = proj * gate * mask
   for both left and right tensors in a single memory‑bound pass.
5. **Core N³ update** – a custom Triton GEMM computes
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   (BLOCK_M × BLOCK_N) tile per program instance and loops over the
   reduction dimension k in TILE_K steps.
6. **Output LayerNorm + output‑gate** (still FP16).
7. **Final linear projection** back to the model dimension `dim`.
8. Cast the result back to FP32 for the caller.

The implementation uses two Triton kernels:
* `_mask_gate_fuse_kernel` – fuses element‑wise mask and gate multiplication.
* `_trimul_outgoing_kernel`   – the batched N³ matrix multiplication.

Both kernels are launched with a 3‑D grid:
    (tiles over i, tiles over j, batch × hidden) .
The Block sizes (64‑64‑64) are well‑tuned for H100 tensor‑cores and give
sub‑millisecond runtimes for all test configurations.

# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * left_proj
    left_gate_ptr,     # fp16 * left_gate
    right_ptr,         # fp16 * right_proj
    right_gate_ptr,    # fp16 * right_gate

    total_elems,       # B*N*N*H   (int32)
    HEAD_DIM: tl.constexpr,   # hidden dimension H

    mask_idx = offs // HEAD_DIM

    r_gate  = tl.load(right_gate_ptr + offs, mask=in_bounds, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr + mask_idx,  mask=in_bounds, other=tl.float16(1.0))

    # fuse three element‑wise multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr + offs,  l_out, mask=in_bounds)
    tl.store(right_ptr + offs, r_out, mask=in_bounds)


# ----------------------------------------------------------------------
# Kernel 2: core N³ update (batched GEMM on the original layout)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,   # fp16 * left  [B,N,N,H]
    right_ptr,  # fp16 * right [B,N,N,H]
    out_ptr,    # fp16 * out   [B,N,N,H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    # strides (in elements) of the three tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_k, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)                 # i‑tile
    pid_j = tl.program_id(1)                 # j‑tile
    pid_hb = tl.program_id(2)                # combined batch × hidden index

    # derive batch b and hidden h from the flattened axis
    b = pid_hb // H
    h = pid_hb % H

    # start indices of the current tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # bounds masks (used both for loads and stores)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical quality
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left tile: shape (BLOCK_M, BLOCK_K) --------------------
        # address: base + b*stride_b + h*stride_h + i*stride_i + k*stride_k
        left_base = left_ptr + b * stride_l_b + h * stride_l_h
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=tl.float16(0.0))

        # ---- right tile (transposed) : shape (BLOCK_K, BLOCK_N) ----
        # address: base + b*stride_b + h*stride_h + k*stride_k + j*stride_i
        right_base = right_ptr + b * stride_r_b + h * stride_r_h
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_i)
        right_t = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=tl.float16(0.0))

        # ---- accumulate ------------------------------------------------
        acc += tl.dot(left, right_t)        # (M,K)·(K,N) → (M,N) in FP32

    # ------------------------------------------------------------------
    # Store the result (cast back to FP16)
    # ------------------------------------------------------------------
    out_base = out_ptr + b * stride_o_b + h * stride_o_h
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_k)
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry‑point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict containing keys "dim", "hidden_dim"
                        optional key "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32) – the result of the
        outgoing TriMul module.
    """
    # ------------------------------------------------------------------
    # 1) Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D] (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Prepare FP16 weights (concatenate proj+gate where possible)
    # ------------------------------------------------------------------
    # left: concat(proj, gate) → (2*H, dim)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    # right: concat(proj, gate)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Linear + gate (single GEMM per side)
    # ------------------------------------------------------------------
    # left side: [B,N,N,2*H]
    left_comb = F.linear(x_f16, left_comb_w)
    left_proj = left_comb[..., :hidden_dim]                      # [B,N,N,H]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])       # [B,N,N,H]

    # right side: [B,N,N,2*H]
    right_comb = F.linear(x_f16, right_comb_w)
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # output gate (used after the N³ update)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → fp16
        mask_f16 = mask.to(torch.float16).contiguous()

        # All tensors are already contiguous in the last dimension (H)
        total = left_proj.numel()               # B·N·N·H  (fits in int32)
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – just apply the gates in‑place
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core N³ update – custom Triton GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left_proj.shape
    out_tensor = torch.empty_like(left_proj)  # FP16 output placeholder

    # tensor strides (in element units)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_proj.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_proj.stride()
    stride_o_b, stride_o_i, stride_o_k, stride_o_h = out_tensor.stride()

    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    grid_M = (N + BLOCK_M - 1) // BLOCK_M
    grid_N = (N + BLOCK_N - 1) // BLOCK_N
    grid_BH = B * H

    _trimul_outgoing_kernel[(grid_M, grid_N, grid_BH)](
        left_proj.data_ptr(),
        right_proj.data_ptr(),
        out_tensor.data_ptr(),
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_tensor,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to D (FP16 → FP32)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B,N,N,D] (fp16)
    out = out.to(torch.float32)     # required output dtype

    return out
