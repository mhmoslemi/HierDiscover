"""
TriMul “outgoing” forward pass (αFold‑3 style) – Triton implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → *x*.
2. Cast *x* to fp16 for the heavy linear work.
3. Compute two bias‑free projections (left/right) and three sigmoid
   gates (left, right, out) on fp16.  Apply the optional pair‑wise mask
   and the element‑wise gates in‑place.
4. Launch a custom Triton kernel that evaluates, for every batch *b*
   and hidden channel *h*,

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   i.e. a batched GEMM  C = A @ Bᵀ  with A,B ∈ fp16, accumulation in fp32
   and write‑back in fp16.  The kernel works directly on the original
   memory layout (no costly transpose/contiguous copy) by using the
   strides of the (B,N,N,H) tensors.
5. Second Layer‑Norm over the hidden dimension (fp16).
6. Apply the output gate and the final bias‑free linear projection
   back to the original dimension *dim* (still fp16).
7. Cast the result to float32 and return.

The kernel tiles are (M,N,K) = (128,128,64) – a good fit for H100
Tensor‑cores while keeping the fp32 accumulator (≈16 KB) inside registers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (A,B : fp16,  C : fp16, acc fp32)
#   A, B, C have shape (B, H, N, N) with strides:
#       batch_stride = N*N*H
#       row_stride   = N*H          (i‑axis for A / j‑axis for B)
#       col_stride   = H            (k‑axis)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [B, H, N, N] (fp16)
    b_ptr,               # right : [B, H, N, N] (fp16)
    c_ptr,               # out   : [B, H, N, N] (fp16)
    M, N, K,             # problem sizes (all == seq_len)
    hidden_dim,          # number of hidden channels (H)
    stride_batch,        # stride over the batch dimension (B)
    stride_row,          # stride over the i / j dimension (N*H)
    stride_col,          # stride over the k dimension (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)   # combined (batch * hidden) index
    pid_m = tl.program_id(axis=0)   # tile row   (i)
    pid_n = tl.program_id(axis=1)   # tile col   (j)

    # --------------------------------------------------------------
    # De‑compose pid_h → batch and hidden indices
    # --------------------------------------------------------------
    batch_idx = pid_h // hidden_dim
    hidden_idx = pid_h - batch_idx * hidden_dim   # pid_h % hidden_dim

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_idx * stride_batch + hidden_idx
    b_base = b_ptr + batch_idx * stride_batch + hidden_idx
    c_base = c_ptr + batch_idx * stride_batch + hidden_idx

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator kept in registers
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- A tile (M × K) ---------------------------------------
        a_ptrs = a_base + offs_m[:, None] * stride_row + cur_k[None, :] * stride_col
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ----- Bᵀ tile (K × N) ---------------------------------------
        b_ptrs = b_base + cur_k[:, None] * stride_col + offs_n[None, :] * stride_row
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # fp16

        # ----- Accumulate in FP32 ------------------------------------
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write back in fp16 (halves the write bandwidth)
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_row + offs_n[None, :] * stride_col
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – stable statistics
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) – fp16 → fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # Apply element‑wise gates in‑place
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Allocate output tensor (same layout as left/right)
    # ------------------------------------------------------------------
    out_hidden = torch.empty_like(left)  # [B, N, N, H] – fp16

    # ------------------------------------------------------------------
    # 7️⃣ Triton kernel launch – batched GEMM C = A @ Bᵀ
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides of the (B, N, N, H) tensors
    stride_batch = left.stride(0)   # B stride = N*N*H
    stride_row   = left.stride(1)   # i / j stride = N*H
    stride_col   = left.stride(2)   # k stride = H

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        bs * hidden_dim,                      # (batch * hidden)
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out_hidden,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        hidden_dim,                  # H
        stride_batch,
        stride_row,
        stride_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )  # fp16

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate and final linear projection (fp16)
    # ------------------------------------------------------------------
    out_hidden = out_hidden * out_gate                # element‑wise fp16
    out = F.linear(out_hidden, weights["to_out.weight"].half())  # fp16 → dim

    # ------------------------------------------------------------------
    # 🔟 Cast back to float32 for the external API
    # ------------------------------------------------------------------
    return out.float()
