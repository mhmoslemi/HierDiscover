"""
TriMul (outgoing) – fast Triton‑augmented forward pass
=====================================================

The implementation follows the reference `TriMul` module but
* fuses the five linear layers (two projections + three gates) into a
  single FP16 GEMM,
* applies the two gates together with the optional mask in a single
  Triton kernel (the sigmoid is evaluated inside the kernel, so no extra
  memory traffic for the gate tensors),
* keeps the heavy N³ contraction as a batched GEMM (`torch.bmm`,
  which on H100 uses tensor‑cores),
* runs the two layer‑norms in FP16 where possible and casts back to
  FP32 only for the final output.

All operations are performed on‑device; the only Python‑level calls are
high‑level PyTorch ops (layer‑norm, batched GEMM and the final linear).
The kernel is deliberately simple and works for all batch‑sizes (1 or 2),
sequence lengths up to 1024 and hidden dimensions 128/384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_sigmoid_kernel(
    left_ptr,          # fp16 * left (output buffer, also input)
    left_gate_ptr,    # fp16 * left_gate (raw, before sigmoid)
    right_ptr,        # fp16 * right (output buffer, also input)
    right_gate_ptr,   # fp16 * right_gate (raw, before sigmoid)
    mask_ptr,         # fp16 * mask (shape B*N*N) – broadcasted over H
    total_elems: tl.int32,       # B * N * N * H
    HEAD_DIM: tl.constexpr,      # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # index of the mask element (broadcasted over the hidden dim)
    mask_idx = offs // HEAD_DIM

    # load tensors
    l_val   = tl.load(left_ptr + offs,            mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr + offs,       mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr + offs,           mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs,      mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr + mask_idx,        mask=mask, other=tl.float16(1.0))

    # sigmoid = 1 / (1 + exp(-x))
    l_sig = 1.0 / (1.0 + tl.exp(-l_gate))
    r_sig = 1.0 / (1.0 + tl.exp(-r_gate))

    # fuse three element‑wise multiplies
    l_out = l_val * l_sig * m_val
    r_out = r_val * r_sig * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32 – stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D] – still float32

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse the five linear layers (proj + 3 gates) into a single GEMM
    # ------------------------------------------------------------------
    # Stack weights: [left_proj; right_proj; left_gate; right_gate; out_gate]
    w_left   = weights["left_proj.weight"].to(torch.float16)
    w_right  = weights["right_proj.weight"].to(torch.float16)
    w_lgate  = weights["left_gate.weight"].to(torch.float16)
    w_rgate  = weights["right_gate.weight"].to(torch.float16)
    w_outgate= weights["out_gate.weight"].to(torch.float16)

    all_weight = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)  # (5*H, D)

    # Single FP16 GEMM
    all_proj = F.linear(x_f16, all_weight)      # [B, N, N, 5*H]

    # ------------------------------------------------------------------
    # 5) Slice the fused result (views – not yet contiguous)
    # ------------------------------------------------------------------
    H = hidden_dim
    left_raw   = all_proj[..., :H]
    right_raw  = all_proj[..., H:2*H]
    left_gate_raw  = all_proj[..., 2*H:3*H]
    right_gate_raw = all_proj[..., 3*H:4*H]
    out_gate_raw   = all_proj[..., 4*H:]               # shape [B, N, N, H]

    # Make contiguous buffers for the tensors that will be used later.
    left  = left_raw.contiguous()
    right = right_raw.contiguous()
    left_gate_raw  = left_gate_raw.contiguous()
    right_gate_raw = right_gate_raw.contiguous()
    # out_gate_raw can stay non‑contiguous – only a sigmoid is applied later.

    # Free the large fused tensor (no longer needed)
    del all_proj
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 6) Mask & gate fusion (Triton) – only if a mask is present.
    #    The kernel also evaluates sigmoid(left_gate_raw) and
    #    sigmoid(right_gate_raw) inside the kernel.
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
        total = left.numel()                     # B * N * N * H  (int32 fits)
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_sigmoid_kernel[grid](
            left.data_ptr(),
            left_gate_raw.data_ptr(),
            right.data_ptr(),
            right_gate_raw.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – just apply gates (sigmoid) in‑place.
        left_gate_sig = torch.sigmoid(left_gate_raw)
        right_gate_sig = torch.sigmoid(right_gate_raw)
        left = left * left_gate_sig
        right = right * right_gate_sig
        # Release the raw gate tensors early.
        del left_gate_sig, right_gate_sig

    # Release raw gate tensors (no longer needed)
    del left_gate_raw, right_gate_raw
    torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 7) Core N³ update – batched GEMM  L @ Rᵀ  (FP16, tensor‑core)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] (fp16)

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still FP16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out_gate = torch.sigmoid(out_gate_raw)          # [B, N, N, H] (fp16)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to model dimension D
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)                         # (fp16)
    out = out.to(torch.float32)                           # required output dtype

    return out
