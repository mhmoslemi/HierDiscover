@triton.jit
def _mask_apply_kernel(left_ptr, right_ptr, mask_ptr, total_elems, HEAD_DIM: tl.constexpr, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # compute mask index (broadcast over H)
    mask_idx = offs // HEAD_DIM

    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    l = l * m
    r = r * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
==================================================

Algorithm
---------
1. **Input LayerNorm** (float32 for numerical stability).
2. Cast to FP16 – all heavy work stays in half‑precision.
3. **Fused projection + gate linear layers**:
   * Concatenate `left_proj`, `right_proj`, `left_gate`, `right_gate`
     weights → a single GEMM `x @ Wᵀ` producing a tensor of shape
     `[B, N, N, 4·H]`.
   * Slice the result into the four tensors, apply sigmoid on the two
     gate halves, and multiply the projection halves by their gates.
4. **Mask & gate fusion** (when a mask is supplied):
   * A tiny Triton kernel loads `left`, `right` and the mask (broadcast
     over the hidden dimension) and performs
       `left  *= mask`; `right *= mask`.
5. **Core N³ update** – for each hidden channel we need
   `Cₕ = Lₕ @ Rₕᵀ`.  By reshaping to `(B·H, N, N)` the work becomes a
   batched GEMM, executed by `torch.bmm` (cuBLAS tensor‑core path).
6. **Output LayerNorm + output‑gate** (still FP16).
7. **Final linear projection** back to the model dimension `D` and cast
   the result to float32 for the caller.

The implementation reduces the original five separate linear layers
to **two** (one 4‑way fused projection/gate and one output‑gate) and
uses a custom Triton kernel only for the mask‑gate step, cutting both
compute and memory traffic while keeping the heavy N³ update on the
highly‑optimized cuBLAS backend.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: apply a binary mask (broadcast over the hidden dim)
#   left  = left  * mask
#   right = right * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_apply_kernel(
    left_ptr,            # fp16 * left  (size = B*N*N*H)
    right_ptr,           # fp16 * right (size = B*N*N*H)
    mask_ptr,            # fp16 * mask  (size = B*N*N)
    total_elems: tl.int32,        # = B*N*N*H
    H: tl.constexpr,              # hidden dimension
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Index of the mask element (broadcast over H)
    mask_idx = offs // H

    l = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    r = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    l = l * m
    r = r * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        `(input_tensor, mask, weights_dict, config_dict)`
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output of shape `[B, N, N, D]` (float32)
    """
    # --------------------------------------------------------------
    # 1) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (float32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to fp16 for the heavy compute
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fuse projection & gate linear layers
    # --------------------------------------------------------------
    # Gather and cast the four projection/gate weights
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)

    # Concatenate -> shape (4*H, D)
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    )  # [4*H, D]

    # Single GEMM: (B·N·N, D) @ (4*H, D).T  →  (B, N, N, 4*H)
    fused = F.linear(x_f16, fused_weight)  # fp16

    # Slice the fused result
    left_proj, right_proj, left_gate_pre, right_gate_pre = torch.split(
        fused, hidden_dim, dim=-1
    )  # each: [B, N, N, H]

    # Apply sigmoid to gate halves
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # Apply gates to the projections
    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5) Output‑gate (independent linear)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # [B,N,N,H]

    # --------------------------------------------------------------
    # 6) Optional mask – fuse with a lightweight Triton kernel
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] → fp16 (0/1)
        mask_f16 = mask.to(torch.float16).contiguous()
        # Ensure contiguous tensors for pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()

        total = left.numel()                     # B·N·N·H  (int32 fits)
        BLOCK = 1024
        grid = (triton.cdiv(total, BLOCK),)
        _mask_apply_kernel[grid](
            left.data_ptr(),
            right.data_ptr(),
            mask_f16.data_ptr(),
            total,
            H=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    # --------------------------------------------------------------
    # 7) Core N³ update: batched GEMM L @ Rᵀ   (tensor‑core path)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B·H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # Batched matrix multiplication.  Right‑hand side transposed.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] (fp16)

    # Restore shape [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # --------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still fp16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # element‑wise gating

    # --------------------------------------------------------------
    # 9) Final linear projection back to model dimension D
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # [D, H]
    out = F.linear(out, to_out_w)  # fp16 → [B, N, N, D]

    # Cast back to float32 as required by the caller
    out = out.to(torch.float32)

    return out
