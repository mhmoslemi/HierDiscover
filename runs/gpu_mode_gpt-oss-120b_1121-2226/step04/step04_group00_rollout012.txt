@triton.jit
def _norm_gate_kernel(
    x_ptr, gate_ptr, weight_ptr, bias_ptr, out_ptr,
    M: tl.constexpr, H: tl.constexpr,
    stride_x_row: tl.constexpr, stride_x_col: tl.constexpr,
    stride_gate_row: tl.constexpr, stride_gate_col: tl.constexpr,
    stride_out_row: tl.constexpr, stride_out_col: tl.constexpr,
    epsilon: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid
    mask_row = row < M
    # compute mean and var
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    for off in range(0, H, BLOCK_H):
        col = off + tl.arange(0, BLOCK_H)
        mask_col = col < H
        mask = mask_row & mask_col

        # load x
        x = tl.load(x_ptr + row * stride_x_row + col * stride_x_col, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # second pass - compute normalized, scale, bias, gate, store
    for off in range(0, H, BLOCK_H):
        col = off + tl.arange(0, BLOCK_H)
        mask_col = col < H
        mask = mask_row & mask_col

        x = tl.load(x_ptr + row * stride_x_row + col * stride_x_col, mask=mask, other=0.0).to(tl.float32)
        g = tl.load(gate_ptr + row * stride_gate_row + col * stride_gate_col, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + col, mask=mask_col, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + col, mask=mask_col, other=0.0).to(tl.float32)

        norm = (x - mean) * inv_std
        out_val = (norm * w + b) * g

        tl.store(out_ptr + row * stride_out_row + col * stride_out_col, out_val.to(tl.float16), mask=mask)

@triton.jit
def _project_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_pre_ptr, right_gate_pre_ptr, out_gate_pre_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    mask_ptr,    # optional broadcast mask of shape (M, )
    M: tl.constexpr, H: tl.constexpr,
    stride_left_row, stride_left_col,
    stride_right_row, stride_right_col,
    stride_gate_row, stride_gate_col,
    stride_out_gate_row, stride_out_gate_col,
    stride_left_out_row, stride_left_out_col,
    stride_right_out_row, stride_right_out_col,
    stride_out_gate_out_row, stride_out_gate_out_col,
    stride_mask_row,
    BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    row = pid
    mask_row = row < M
    for off in range(0, H, BLOCK_H):
        col = off + tl.arange(0, BLOCK_H)
        mask_col = col < H
        mask = mask_row & mask_col
        # load
        l = tl.load(left_ptr + row * stride_left_row + col * stride_left_col, mask=mask).to(tl.float32)
        r = tl.load(right_ptr + row * stride_right_row + col * stride_right_col, mask=mask).to(tl.float32)
        lg = tl.load(left_gate_pre_ptr + row * stride_gate_row + col * stride_gate_col, mask=mask).to(tl.float32)
        rg = tl.load(right_gate_pre_ptr + row * stride_gate_row + col * stride_gate_col, mask=mask).to(tl.float32)
        og = tl.load(out_gate_pre_ptr + row * stride_gate_row + col * stride_gate_col, mask=mask).to(tl.float32)
        # sigmoid
        lg_sig = 1.0 / (1.0 + tl.exp(-lg))
        rg_sig = 1.0 / (1.0 + tl.exp(-rg))
        og_sig = 1.0 / (1.0 + tl.exp(-og))
        # mask
        if HAS_MASK:
            m = tl.load(mask_ptr + row, mask=mask_row, other=1.0) # mask scalar as fp16 with broadcast?
            # mask shape (M,) => broadcast across H
            m_f = m.to(tl.float32)
            lg_sig = lg_sig * m_f
            rg_sig = rg_sig * m_f
        # apply gates
        l_out = l * lg_sig
        r_out = r * rg_sig
        # store

    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr, out_gate_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,

    stride_lg_row, stride_lg_col,
    stride_og_row, stride_og_col,

"""
TriMul “outgoing” forward pass – Triton‑accelerated implementation.

Algorithm (high‑level)
----------------------
1️⃣ Input LayerNorm (FP32) – identical to nn.LayerNorm.
2️⃣ Cast to FP16 for the heavy work.
3️⃣ Fuse all 5 linear projections (left, right, left‑gate, right‑gate,
   out‑gate) into one GEMM (weight shape (5·H, D)).
4️⃣ One Triton kernel fuses:
   • sigmoid on the three gate vectors,
   • multiplication of left/right projections by their gates,
   • optional binary mask (broadcast over the hidden dimension),
   • stores the gated left/right tensors and the sigmoid‑ed out‑gate.
5️⃣ Core cubic update: `out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]`.
   Implemented as a batched GEMM `left @ rightᵀ` with shape
   (B·H, N, N) → (B·H, N, N) (FP16×FP16→FP16, FP32 accumulation).
6️⃣ Fuse the output LayerNorm, multiplication by the out‑gate and the
   final linear projection in a second Triton kernel (mean/var in FP32,
   scaling in FP16).  The final projection back to `dim` dimensions is
   performed by a standard linear (cuBLAS) GEMM.
7️⃣ Cast the result back to float32 (the public API expects FP32).

All heavy linear‑algebraic work is done by cuBLAS tensor‑core kernels,
while the small element‑wise parts are fused into two lightweight
Triton kernels, reducing kernel‑launch overhead and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse linear‑gate‑mask for left/right streams and sigmoid out‑gate
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_mask_fuse_kernel(
    left_ptr, right_ptr,                          # pre‑gate projections (fp16)
    lgate_pre_ptr, rgate_pre_ptr, ogate_pre_ptr, # pre‑sigmoid gate tensors (fp16)
    left_out_ptr, right_out_ptr, out_gate_ptr,   # outputs (fp16)
    mask_ptr,                                     # optional mask (fp16, shape M)
    M: tl.constexpr, H: tl.constexpr,             # rows, hidden dim (compile‑time)
    stride_left_row, stride_left_col,
    stride_right_row, stride_right_col,
    stride_lgate_row, stride_lgate_col,
    stride_rgate_row, stride_rgate_col,
    stride_ogate_row, stride_ogate_col,
    stride_left_out_row, stride_left_out_col,
    stride_right_out_row, stride_right_out_col,
    stride_out_gate_row, stride_out_gate_col,
    stride_mask_row,
    BLOCK_H: tl.constexpr, HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)               # one program per row (M = B·N·N)
    row = pid
    valid = row < M

    # broadcast mask value (if present)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + row * stride_mask_row, mask=valid).to(tl.float32)
    else:
        mask_val = tl.float32(1.0)

    # iterate over the hidden dimension in BLOCK_H sized chunks
    for off in range(0, H, BLOCK_H):
        col = off + tl.arange(0, BLOCK_H)          # column index within row
        col_mask = col < H
        mask = valid & col_mask

        # load data (fp16 → fp32 for arithmetic)
        l  = tl.load(left_ptr          + row * stride_left_row          + col * stride_left_col,
                     mask=mask).to(tl.float32)
        r  = tl.load(right_ptr         + row * stride_right_row         + col * stride_right_col,
                     mask=mask).to(tl.float32)
        lg = tl.load(lgate_pre_ptr     + row * stride_lgate_row         + col * stride_lgate_col,
                     mask=mask).to(tl.float32)
        rg = tl.load(rgate_pre_ptr     + row * stride_rgate_row         + col * stride_rgate_col,
                     mask=mask).to(tl.float32)
        og = tl.load(ogate_pre_ptr     + row * stride_ogate_row         + col * stride_ogate_col,
                     mask=mask).to(tl.float32)

        # sigmoid (stable for fp16 values)
        lg_sig = 1.0 / (1.0 + tl.exp(-lg))
        rg_sig = 1.0 / (1.0 + tl.exp(-rg))
        og_sig = 1.0 / (1.0 + tl.exp(-og))

        # apply gates and optional mask
        l_out = l * lg_sig * mask_val
        r_out = r * rg_sig * mask_val

        # store results (back to fp16)
        tl.store(left_out_ptr        + row * stride_left_out_row        + col * stride_left_out_col,
                 l_out.to(tl.float16), mask=mask)
        tl.store(right_out_ptr       + row * stride_right_out_row       + col * stride_right_out_col,
                 r_out.to(tl.float16), mask=mask)
        tl.store(out_gate_ptr        + row * stride_out_gate_row        + col * stride_out_gate_col,
                 og_sig.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Kernel 2: output LayerNorm + out‑gate scaling
# ----------------------------------------------------------------------
@triton.jit
def _out_norm_gate_kernel(
    x_ptr, gate_ptr,                     # (M, H) tensors (fp16)
    weight_ptr, bias_ptr,                # LayerNorm parameters (fp16, length H)
    out_ptr,                             # output after norm+gate (fp16)
    M: tl.constexpr, H: tl.constexpr,    # rows, hidden dim (compile‑time)
    stride_x_row, stride_x_col,
    stride_gate_row, stride_gate_col,
    stride_out_row, stride_out_col,
    EPS: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)            # one program per row
    row = pid
    valid = row < M

    # ------------------------------------------------------------------
    # First pass – compute mean and variance (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for off in range(0, H, BLOCK_H):
        col = off + tl.arange(0, BLOCK_H)
        col_mask = col < H
        mask = valid & col_mask

        v = tl.load(x_ptr + row * stride_x_row + col * stride_x_col,
                    mask=mask).to(tl.float32)
        sum_val += tl.sum(v, axis=0)
        sum_sq  += tl.sum(v * v, axis=0)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ------------------------------------------------------------------
    # Second pass – apply norm, weight, bias, out‑gate
    # ------------------------------------------------------------------
    for off in range(0, H, BLOCK_H):
        col = off + tl.arange(0, BLOCK_H)
        col_mask = col < H
        mask = valid & col_mask

        v  = tl.load(x_ptr + row * stride_x_row + col * stride_x_col,
                     mask=mask).to(tl.float32)
        g  = tl.load(gate_ptr + row * stride_gate_row + col * stride_gate_col,
                     mask=mask).to(tl.float32)
        w  = tl.load(weight_ptr + col, mask=col_mask).to(tl.float32)
        b  = tl.load(bias_ptr   + col, mask=col_mask).to(tl.float32)

        normed = (v - mean) * inv_std
        scaled = normed * w + b        # LayerNorm affine
        out    = scaled * g            # multiply by out‑gate

        tl.store(out_ptr + row * stride_out_row + col * stride_out_col,
                 out.to(tl.float16), mask=mask)


# ----------------------------------------------------------------------
# Helper that runs kernel 1 (projection + gating + optional mask)
# ----------------------------------------------------------------------
def _proj_gate_mask_fuse(left_raw, right_raw,
                         lgate_raw, rgate_raw, ogate_raw,
                         mask, hidden_dim):
    """
    left_raw / right_raw / … are contiguous fp16 tensors of shape (M, H)
    where M = B·N·N and H = hidden_dim.
    Returns (left, right, out_gate) after sigmoid‑gating and mask.
    """
    device = left_raw.device
    M = left_raw.shape[0]
    H = hidden_dim

    left_out  = torch.empty_like(left_raw)
    right_out = torch.empty_like(right_raw)
    out_gate  = torch.empty_like(ogate_raw)

    # optional mask (broadcasted over H)
    if mask is not None:
        mask_flat = mask.view(-1).to(torch.float16)
        has_mask = True
    else:
        mask_flat = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = False

    # strides (all tensors are (M, H) ⇒ stride0 = H, stride1 = 1)
    s_lr = left_raw.stride()
    s_rr = right_raw.stride()
    s_lg = lgate_raw.stride()
    s_rg = rgate_raw.stride()
    s_og = ogate_raw.stride()
    s_lo = left_out.stride()
    s_ro = right_out.stride()
    s_og_out = out_gate.stride()
    s_mask = mask_flat.stride(0) if has_mask else 0

    grid = (M,)
    _proj_gate_mask_fuse_kernel[grid](
        left_raw, right_raw,
        lgate_raw, rgate_raw, ogate_raw,
        left_out, right_out, out_gate,
        mask_flat,
        M, H,
        s_lr[0], s_lr[1],
        s_rr[0], s_rr[1],
        s_lg[0], s_lg[1],
        s_rg[0], s_rg[1],
        s_og[0], s_og[1],
        s_lo[0], s_lo[1],
        s_ro[0], s_ro[1],
        s_og_out[0], s_og_out[1],
        s_mask,
        BLOCK_H=32,
        HAS_MASK=1 if has_mask else 0,
        num_warps=4,
        num_stages=2,
    )
    return left_out, right_out, out_gate


# ----------------------------------------------------------------------
# Helper that runs kernel 2 (output LayerNorm + out‑gate)
# ----------------------------------------------------------------------
def _out_norm_gate(x, out_gate, norm_w, norm_b):
    """
    x, out_gate – fp16 tensors of shape (B, N, N, H)
    norm_w, norm_b – fp16 vectors of length H
    Returns tensor of same shape after LayerNorm + out‑gate.
    """
    B, N, _, H = x.shape
    M = B * N * N

    x_flat = x.contiguous().view(M, H)
    gate_flat = out_gate.contiguous().view(M, H)
    out_flat = torch.empty_like(x_flat)

    s_x = x_flat.stride()
    s_g = gate_flat.stride()
    s_o = out_flat.stride()

    grid = (M,)
    _out_norm_gate_kernel[grid](
        x_flat,
        gate_flat,
        norm_w,
        norm_b,
        out_flat,
        M, H,
        s_x[0], s_x[1],
        s_g[0], s_g[1],
        s_o[0], s_o[1],
        EPS=1e-5,
        BLOCK_H=32,
        num_warps=8,
        num_stages=2,
    )
    return out_flat.view(B, N, N, H)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # D
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 – the rest of the pipeline works in half precision
    # ------------------------------------------------------------------
    x_f16 = x.half()

    # ------------------------------------------------------------------
    # 3) Fuse the five linear layers into a single GEMM
    # ------------------------------------------------------------------
    #   left_proj, right_proj, left_gate, right_gate, out_gate
    # all have shape (hidden_dim, dim) and no bias.
    weight_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                         # (5·H, D)

    combined = F.linear(x_f16, weight_all)  # [B, N, N, 5·H]  (fp16)

    # ------------------------------------------------------------------
    # 4) Slice and run fused projection‑gate‑mask kernel
    # ------------------------------------------------------------------
    B, N, _, _ = combined.shape
    H = hidden_dim
    # raw tensors for the kernel (contiguous, shape (M, H))
    left_raw   = combined[..., :H].contiguous().view(-1, H)
    right_raw  = combined[..., H:2*H].contiguous().view(-1, H)
    lgate_raw  = combined[..., 2*H:3*H].contiguous().view(-1, H)
    rgate_raw  = combined[..., 3*H:4*H].contiguous().view(-1, H)
    ogate_raw  = combined[..., 4*H:5*H].contiguous().view(-1, H)

    # optional binary mask (broadcasted over hidden dimension)
    if not nomask and mask is not None:
        mask_use = mask
    else:
        mask_use = None

    left, right, out_gate = _proj_gate_mask_fuse(
        left_raw, right_raw,
        lgate_raw, rgate_raw, ogate_raw,
        mask_use,
        hidden_dim,
    )  # all [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 5) Core cubic update – batched matrix multiplication
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) so each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched GEMM: left @ rightᵀ    (FP16 × FP16 → FP16, FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N]

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + out‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()

    out_norm = _out_norm_gate(out, out_gate, to_out_norm_w, to_out_norm_b)  # fp16

    # ------------------------------------------------------------------
    # 7) Final projection back to `dim` (standard linear GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()
    out_final = F.linear(out_norm, to_out_w)  # [B, N, N, D]  fp16

    # ------------------------------------------------------------------
    # 8) Cast back to FP32 for the API
    # ------------------------------------------------------------------
    return out_final.float()
