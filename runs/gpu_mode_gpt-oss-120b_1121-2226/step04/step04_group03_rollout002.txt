"""
TriMul “outgoing” forward pass (high‑performance Triton implementation).

Algorithm
---------
1️⃣ Layer‑Norm the 4‑D input (float32) – stable statistics.
2️⃣ Cast to fp16 for the heavy linear work.
3️⃣ **Fused linear**: concatenate the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) → shape
   (5·hidden_dim, dim) and apply a single torch linear on the whole input.
   This loads the input tensor only once from memory.
4️⃣ Split the fused result → left_proj, right_proj, left_gate‑logits,
   right_gate‑logits, out_gate‑logits.
   Apply sigmoid to the gate logits.
5️⃣ Optional pair‑wise mask (broadcast on the hidden dimension) on
   left_proj/right_proj, then multiply by their respective gates:
   `left = left_proj * left_gate`, `right = right_proj * right_gate`.
6️⃣ Reshape to [B·H, N, N] (contiguous) and launch a custom Triton kernel
   that computes, for every head **h**,
        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]
   i.e. a batched GEMM `C = A @ Bᵀ` with fp16 inputs, fp32 accumulation,
   and fp16 write‑back.
   Tile sizes are (128,128,64) and the kernel uses 8 warps + 3
   pre‑fetch stages for maximal occupancy on H100.
7️⃣ Reshape back to [B, N, N, H] (fp16).
8️⃣ Second Layer‑Norm over the hidden dimension (fp16).
9️⃣ Apply the output‑gate (sigmoid) element‑wise.
🔟 Final bias‑free linear projection back to the original dimension `dim`
   (fp16) and cast to float32 for the API contract.

Only the N³ matrix multiplication is performed in Triton; all surrounding
operations use highly‑optimized torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (A,B: fp16, C: fp16, acc: fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                     # left  : [BH, N, N]   (fp16)
    b_ptr,                     # right : [BH, N, N]   (fp16) – will be read as Bᵀ
    c_ptr,                     # out   : [BH, N, N]   (fp16)
    M, N, K,                   # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)  # batch * hidden
    pid_m = tl.program_id(axis=0)  # tile row
    pid_n = tl.program_id(axis=1)  # tile column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator kept in registers
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M × K) ------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ----- Load Bᵀ tile (K × N) ------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ----- Accumulate ------------------------------------------------
        # tl.dot on fp16 tensors automatically uses fp32 accumulation on
        # NVIDIA GPUs (Tensor Cores).  Convert to fp32 explicitly for safety.
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back result in fp16 (halves write bandwidth)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – stable accumulation
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fused linear: left_proj, right_proj, left_gate, right_gate, out_gate
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (all have shape [hidden_dim, dim])
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device, dtype=torch.float16)  # (5*H, D)

    # Single matmul: shape -> [B, N, N, 5*H]
    fused_out = F.linear(x_fp16, fused_weight)  # fp16

    # Split the fused result
    split_chunks = torch.split(fused_out, hidden_dim, dim=-1)
    left_proj, right_proj, left_gate_logits, right_gate_logits, out_gate_logits = split_chunks

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_logits)
    right_gate = torch.sigmoid(right_gate_logits)
    out_gate = torch.sigmoid(out_gate_logits)

    # ------------------------------------------------------------------
    # 5️⃣ Optional pair‑wise mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_expanded = mask_tensor.unsqueeze(-1).to(left_proj.dtype)  # [B, N, N, 1]
        left_proj = left_proj * mask_expanded
        right_proj = right_proj * mask_expanded

    # ------------------------------------------------------------------
    # 6️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # [B, N, N, H] – fp16
    right = right_proj * right_gate  # [B, N, N, H] – fp16

    # ------------------------------------------------------------------
    # 7️⃣ Reshape to [B·H, N, N] for the batched GEMM
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    # Output buffer for the GEMM
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 8️⃣ Launch the Triton kernel (batched GEMM C = A @ Bᵀ)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                                   # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right interpreted as transposed)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, H] (still fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 🔟 Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Apply output gate and final linear projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate          # element‑wise gate, fp16
    out = F.linear(
        out,
        weights["to_out.weight"].to(torch.float16),
    )  # fp16 → dim

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 for the API contract
    # ------------------------------------------------------------------
    out = out.float()
    return out
