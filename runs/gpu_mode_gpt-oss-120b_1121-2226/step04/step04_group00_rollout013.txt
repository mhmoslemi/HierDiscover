"""
TriMul “outgoing” forward pass (GPU‑accelerated).

Algorithm
---------

1️⃣  **Input LayerNorm** – performed directly in FP16 (no extra FP32 copy).

2️⃣  **Linear projections & gates** – five bias‑free `torch.nn.functional.linear`
    calls (left/right and three gates) all work in FP16.

3️⃣  **Fused gate × mask** – a tiny Triton kernel multiplies the
    projected tensors with their sigmoid‑gates and (optionally) the binary
    mask, producing the final *left* and *right* tensors ready for the cubic
    update.

4️⃣  **Cubic (N³) update** – `out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]`
    = batched matrix‑multiply `L @ Rᵀ` where each hidden channel is an
    independent batch.  This is implemented with a single `torch.bmm`,
    exploiting Tensor‑Cores (FP16 multiplication, FP32 accumulation).

5️⃣  **Fused output norm & projection** – another Triton kernel
    performs (a) LayerNorm over the hidden dimension, (b) element‑wise
    scaling by `out_gate`, and (c) the final linear projection back to
    `dim`.  All three steps share the same memory traversal, eliminating
    extra kernels.

The only Python‑side kernels are the two Triton calls; everything else
uses highly‑optimized PyTorch primitives.  The implementation keeps the
register pressure low (no per‑thread hidden‑dim vectors) and therefore
runs well inside the 1 ms target on an H100.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Fuse left/right gating (and optional mask) into a single kernel
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_kernel(
    left_ptr, right_ptr,               # [total, hidden] (fp16)
    left_gate_ptr, right_gate_ptr,     # [total, hidden] (fp16)
    mask_ptr,                          # [total_positions] (fp16) – optional
    left_out_ptr, right_out_ptr,       # [total, hidden] (fp16)
    total_elements: tl.constexpr,      # total = B·N·N·H
    hidden_dim: tl.constexpr,          # H
    BLOCK_SIZE: tl.constexpr,          # work‑group size
    HAS_MASK: tl.constexpr,            # compile‑time bool
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # ------------------------------------------------------------------
    # Load tensors
    # ------------------------------------------------------------------
    l = tl.load(left_ptr + offs,          mask=mask)
    r = tl.load(right_ptr + offs,         mask=mask)
    lg = tl.load(left_gate_ptr + offs,    mask=mask)
    rg = tl.load(right_gate_ptr + offs,   mask=mask)

    # Apply sigmoid gates
    l = l * lg
    r = r * rg

    # ------------------------------------------------------------------
    # Optional binary mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # one mask element per position (B·N·N)
        pos_idx = offs // hidden_dim          # integer division
        m = tl.load(mask_ptr + pos_idx, mask=mask)
        l = l * m
        r = r * m

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    tl.store(left_out_ptr + offs,  l, mask=mask)
    tl.store(right_out_ptr + offs, r, mask=mask)


def _apply_gates_triton(left, right, left_gate, right_gate, mask_fp16=None, block_size: int = 1024):
    """
    ``left`` and ``right`` have shape [B, N, N, H].
    ``mask_fp16`` – tensor of shape [B, N, N, 1] (fp16) or ``None``.
    Returns gated ``left`` and ``right`` (in‑place‑like tensors).
    """
    total = left.numel()                     # B·N·N·H
    hidden_dim = left.shape[-1]

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    has_mask = mask_fp16 is not None
    if not has_mask:
        # dummy tensor – never read because HAS_MASK = False
        mask_fp16 = torch.empty(0, device=left.device, dtype=left.dtype)

    grid = ((total + block_size - 1) // block_size, )
    _gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_fp16,
        left_out,
        right_out,
        total,
        hidden_dim,
        BLOCK_SIZE=block_size,
        HAS_MASK=has_mask,
        num_warps=4,
        num_stages=2,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# 2️⃣  Fuse output LayerNorm, out‑gate scaling and final projection
# ----------------------------------------------------------------------
@triton.jit
def _out_norm_linear_kernel(
    out_ptr,                # [P, H]   (fp16)   – cubic update result
    out_gate_ptr,           # [P, H]   (fp16)
    ln_weight_ptr,          # [H]      (fp16) – to_out_norm.weight
    ln_bias_ptr,            # [H]      (fp16) – to_out_norm.bias
    proj_weight_ptr,        # [H, D]   (fp16) – to_out.weight
    out_final_ptr,          # [P, D]   (fp16) – final result
    hidden_dim: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
):
    pid = tl.program_id(0)                 # one program = one (b,i,j) position
    # ------------------------------------------------------------------
    # Pointers to the per‑position rows
    # ------------------------------------------------------------------
    out_base      = out_ptr      + pid * hidden_dim
    out_gate_base = out_gate_ptr + pid * hidden_dim

    # ------------------------------------------------------------------
    # First pass: compute mean / variance of ``out`` (LayerNorm over H)
    # ------------------------------------------------------------------
    sum_val  = tl.float32(0.0)
    sum_sq   = tl.float32(0.0)

    BLOCK_H = 32  # fits comfortably into registers

    for k in range(0, hidden_dim, BLOCK_H):
        cur_len = hidden_dim - k
        cur_len = tl.where(cur_len > BLOCK_H, BLOCK_H, cur_len)

        idx = k + tl.arange(0, BLOCK_H)
        mask = idx < hidden_dim

        v = tl.load(out_base + idx, mask=mask).to(tl.float32)
        sum_val += tl.sum(v, mask=mask)
        sum_sq  += tl.sum(v * v, mask=mask)

    mean = sum_val / hidden_dim
    var  = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalize, apply out‑gate, LayerNorm affine, and
    #               project back to ``dim``.
    # ------------------------------------------------------------------
    acc = tl.zeros((dim,), dtype=tl.float32)   # accumulator for final vector

    for k in range(0, hidden_dim, BLOCK_H):
        idx = k + tl.arange(0, BLOCK_H)
        mask = idx < hidden_dim

        # load raw tensors
        out_val      = tl.load(out_base      + idx, mask=mask).to(tl.float32)
        gate_val     = tl.load(out_gate_base + idx, mask=mask).to(tl.float32)
        ln_w         = tl.load(ln_weight_ptr + idx,   mask=mask).to(tl.float32)
        ln_b         = tl.load(ln_bias_ptr   + idx,   mask=mask).to(tl.float32)

        # LayerNorm (affine)
        normed = (out_val - mean) * inv_std
        normed = normed * ln_w + ln_b

        # multiply by out‑gate
        scaled = normed * gate_val          # shape [BLOCK_H]

        # weight block: rows (k … k+BLOCK_H-1) of shape (hidden_dim, dim)
        w = tl.load(
            proj_weight_ptr + (k + tl.arange(0, BLOCK_H))[:, None] * dim
                             + tl.arange(0, dim)[None, :],
            mask=mask[:, None]
        )                                     # shape [BLOCK_H, dim]

        # contribution = Σ_h scaled_h * w[h, :]
        contrib = tl.sum(scaled[:, None] * w, axis=0, mask=mask[:, None])
        acc += contrib
    # ------------------------------------------------------------------
    # Store final FP16 output
    # ------------------------------------------------------------------
    tl.store(out_final_ptr + pid * dim + tl.arange(0, dim), acc.to(tl.float16))


def _apply_out_norm_linear(out, out_gate, ln_weight, ln_bias, proj_weight, dim, block_size: int = 256):
    """
    ``out`` / ``out_gate`` : [B, N, N, H] (fp16)
    Returns ``final`` : [B, N, N, dim] (fp16)
    """
    B, N, _, H = out.shape
    total_positions = B * N * N

    out_flat      = out.view(total_positions, H)
    out_gate_flat = out_gate.view(total_positions, H)

    final = torch.empty((total_positions, dim), dtype=torch.float16, device=out.device)

    grid = (total_positions, )
    _out_norm_linear_kernel[grid](
        out_flat,
        out_gate_flat,
        ln_weight,
        ln_bias,
        proj_weight,
        final,
        hidden_dim=H,
        dim=dim,
        eps=1e-5,
        num_warps=4,
        num_stages=2,
    )
    return final.view(B, N, N, dim)


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model tensors (float32)
            - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                             ``nomask`` (bool)

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣  Unpack & config
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                      # D
    hidden_dim = config["hidden_dim"]        # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣  Input LayerNorm directly in FP16 (saves an extra copy)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].half(),
        bias=weights["norm.bias"].half(),
    )  # [B, N, N, D] (fp16)

    # ------------------------------------------------------------------
    # 3️⃣  Linear projections (bias‑free) – keep everything in FP16
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].half())   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].half()))   # fp16
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].half()))   # fp16

    # ------------------------------------------------------------------
    # 5️⃣  Fuse gate multiplication (and optional mask) via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask -> fp16, broadcast over hidden dimension
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
    else:
        mask_fp16 = None

    left, right = _apply_gates_triton(left, right, left_gate, right_gate, mask_fp16)

    # ------------------------------------------------------------------
    # 6️⃣  Cubic update  (batched GEMM: L @ Rᵀ)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (fp16)

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 7️⃣  Output LayerNorm + out‑gate + final projection (fused Triton kernel)
    # ------------------------------------------------------------------
    final = _apply_out_norm_linear(
        out,
        out_gate,
        weights["to_out_norm.weight"].half(),
        weights["to_out_norm.bias"].half(),
        weights["to_out.weight"].half(),
        dim,
    )  # fp16

    # ------------------------------------------------------------------
    # 8️⃣  Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return final.float()
