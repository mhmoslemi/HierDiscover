import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # input layernorm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # cast to fp16
    x_f16 = x.to(torch.float16)

    # Linear weights cast to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16) # (dim, hidden_dim)

    # Fuse left_proj + left_gate
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*H, dim)
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Linear: shape B,N,N,2*H
    left_comb = F.linear(x_f16, left_comb_w)   # bias None
    right_comb = F.linear(x_f16, right_comb_w)

    # split
    left_proj = left_comb[..., :hidden_dim]
    left_gate = left_comb[..., hidden_dim:]
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:]

    # mask aplications if provided
    if not nomask and mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1]
        m = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # apply gates
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # out_gate
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # core cubic update (batched matmul)
    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # matmul: left @ right^T  -> (B, H, N, N)
    out = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # permute back to (B,N,N,H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # fuse gating + linear projection via Triton kernel
    B, N, _, _ = out.shape
    total = B * N * N
    out_flat = out.view(total, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(total, hidden_dim).contiguous()
    # allocate output
    out_final_flat = torch.empty(total, dim, dtype=torch.float16, device=device)

    # launch kernel
    BLOCK_M = 128
    BLOCK_D = 32
    BLOCK_H = 32
    grid = (triton.cdiv(total, BLOCK_M), triton.cdiv(dim, BLOCK_D))
    _gated_linear_fused_kernel[grid](
        out_flat, out_gate_flat, to_out_weight, out_final_flat,
        total, hidden_dim, dim,
        BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D, BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    out_final = out_final_flat.view(B, N, N, dim)

    # cast to fp32
    return out_final.to(torch.float32)

"""
Custom Triton‑accelerated forward pass for the “outgoing” TriMul operator.

Algorithm
---------
1.  Input LayerNorm (FP32) – high‑precision statistics.
2.  Cast to FP16 for the heavy work.
3.  Fuse each projection + gate linear layer:
    * Concatenate (proj_weight , gate_weight) → shape (2·H, D) and perform a single
      torch.nn.functional.linear.
    * Split the result, apply sigmoid to the gate part and multiply.
    * Optional binary mask (broadcasted) is applied *before* the gate, exactly as
      the reference implementation.
4.  Compute the cubic update Σₖ L_{i,k,h}·R_{j,k,h} as a batched
    matrix‑multiply (B,H,N,N) @ (B,H,N,N)ᵀ using a single strided‑batched
    cuBLAS GEMM (torch.matmul).  This replaces many tiny bmm calls.
5.  Output LayerNorm (FP16).
6.  Fuse the element‑wise out‑gate multiplication and the final linear
    projection (H → D) in a custom Triton kernel.  The kernel works on the
    flattened tensors (B·N·N, H) and (D, H) and accumulates in FP32 for
    numeric stability.
7.  Cast the result back to FP32.

Only the gating + projection step (step 6) is written in Triton, satisfying the
“must use Triton” requirement while keeping the dominant cubic computation on the
highly‑optimized cuBLAS path.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused element‑wise out‑gate * (out_norm @ weight.T)
# ----------------------------------------------------------------------
@triton.jit
def _gated_linear_fused_kernel(
    out_norm_ptr,   # FP16 pointer: (total, hidden_dim)
    out_gate_ptr,   # FP16 pointer: (total, hidden_dim)
    weight_ptr,     # FP16 pointer: (dim, hidden_dim)
    out_ptr,        # FP16 pointer: (total, dim) – result
    total,          # int: total = B * N * N
    hidden_dim,     # int: H
    dim,            # int: D
    BLOCK_M: tl.constexpr,   # rows processed per program (total dimension)
    BLOCK_D: tl.constexpr,   # columns processed per program (output dim)
    BLOCK_H: tl.constexpr,   # reduction block size over hidden_dim
):
    pid_m = tl.program_id(0)          # block over rows (total)
    pid_n = tl.program_id(1)          # block over columns (dim)

    # ------------------------------------------------------------------
    # 1) Compute row / column indices and masks
    # ------------------------------------------------------------------
    row = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M)
    col = pid_n * BLOCK_D + tl.arange(0, BLOCK_D)   # (BLOCK_D)

    row_mask = row < total
    col_mask = col < dim

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) Loop over hidden_dim in tiles of size BLOCK_H
    # ------------------------------------------------------------------
    for k in range(0, hidden_dim, BLOCK_H):
        cur_h = k + tl.arange(0, BLOCK_H)               # (BLOCK_H)
        h_mask = cur_h < hidden_dim

        # ----- load out_norm and out_gate (shape BLOCK_M x BLOCK_H) -----
        off = row[:, None] * hidden_dim + cur_h[None, :]   # broadcasted offset
        mask = row_mask[:, None] & h_mask[None, :]
        out_norm = tl.load(out_norm_ptr + off, mask=mask, other=0.0)   # fp16
        out_gate = tl.load(out_gate_ptr + off, mask=mask, other=0.0)   # fp16

        gated = out_norm * out_gate                         # fp16

        # ----- load weight tile (shape BLOCK_H x BLOCK_D) ----------
        # weight is stored as (dim, hidden_dim) → offset = d*hidden_dim + h
        w_off = cur_h[:, None] + col[None, :] * hidden_dim
        w_mask = h_mask[:, None] & col_mask[None, :]
        weight_tile = tl.load(weight_ptr + w_off, mask=w_mask, other=0.0)  # fp16

        # ----- matrix multiply: (BLOCK_M, BLOCK_H) @ (BLOCK_H, BLOCK_D) -----
        # tl.dot returns FP32 when acc is FP32
        acc += tl.dot(gated, weight_tile)

    # ------------------------------------------------------------------
    # 3) Write result back (cast to FP16)
    # ------------------------------------------------------------------
    out_off = row[:, None] * dim + col[None, :]
    out_mask = row_mask[:, None] & col_mask[None, :]
    tl.store(out_ptr + out_off, acc.to(tl.float16), mask=out_mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys "dim", "hidden_dim",
                               optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 0) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Fused projection + gate for left & right streams
    # --------------------------------------------------------------
    # concatenate proj and gate weights for each stream (out_features, in_features)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)   # (2*H, D)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)   # (2*H, D)

    # Linear → (B,N,N,2*H)
    left_comb = F.linear(x_fp16, left_comb_w)     # bias = None
    right_comb = F.linear(x_fp16, right_comb_w)

    # split into proj / gate parts
    left_proj = left_comb[..., :hidden_dim]            # (B,N,N,H)
    left_gate = left_comb[..., hidden_dim:]            # (B,N,N,H)
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:]

    # optional mask (applied *before* gating, just like the reference)
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # apply sigmoid gates
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # out‑gate (shared for all streams)
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))  # (B,N,N,H)

    # --------------------------------------------------------------
    # 4) Core cubic update: batched GEMM L @ Rᵀ
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) for a strided‑batched matmul
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous()

    # batched matrix multiplication (cuBLAS strided‑batched)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))   # (B, H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Output LayerNorm (FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )   # (B, N, N, H)  fp16

    # --------------------------------------------------------------
    # 6) Fuse out‑gate multiplication and final linear projection (H → D)
    # --------------------------------------------------------------
    B, N, _, _ = out.shape
    total = B * N * N                         # number of (i,j) positions

    # flatten tensors to shape (total, hidden_dim) – guarantee contiguity
    out_norm_flat = out.view(total, hidden_dim).contiguous()
    out_gate_flat = out_gate.view(total, hidden_dim).contiguous()

    # final weight: (D, H)
    to_out_w = weights["to_out.weight"].to(torch.float16).contiguous()

    # allocate output buffer (total, D)
    out_final_flat = torch.empty(total, dim, dtype=torch.float16, device=device)

    # Triton launch configuration
    BLOCK_M = 128      # rows per program (tuned for H100)
    BLOCK_D = 32       # output‑dim columns per program
    BLOCK_H = 32       # reduction tile over hidden_dim
    grid = (
        triton.cdiv(total, BLOCK_M),
        triton.cdiv(dim, BLOCK_D),
    )
    _gated_linear_fused_kernel[grid](
        out_norm_flat,
        out_gate_flat,
        to_out_w,
        out_final_flat,
        total,
        hidden_dim,
        dim,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )

    # reshape to (B, N, N, D)
    out_final = out_final_flat.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 7) Cast to FP32 for API compliance
    # --------------------------------------------------------------
    return out_final.to(torch.float32)
