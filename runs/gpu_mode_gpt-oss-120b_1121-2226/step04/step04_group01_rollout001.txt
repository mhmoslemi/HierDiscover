     `right_gate` and `mask`.
   * Otherwise simple in‑place element‑wise multiplication is used.
5. **Core N³ update** – batched GEMM:
   * Reshape `left`/`right` to `(B·H, N, N)` and compute
     `out = left @ rightᵀ` via `torch.bmm`. This runs on Tensor‑cores.
6. **Output LayerNorm + gating** (FP16).
7. **Final linear projection** back to `D` (FP16 → cast to FP32).

Only the mask‑gate step is written in Triton; all other ops use
highly‑optimized PyTorch kernels. The implementation works for any
batch size (1 or 2), sequence length up to 1024 and hidden dimension
128, with or without a mask.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# Mask is broadcasted over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 * left
    left_gate_ptr,       # fp16 * left_gate
    right_ptr,           # fp16 * right
    right_gate_ptr,      # fp16 * right_gate
    mask_ptr,            # fp16 * mask  (shape B*N*N)
    total_elems: tl.int32,    # B*N*N*H
    HEAD_DIM: tl.constexpr,   # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Index of the mask element (broadcast over H)
    mask_idx = offs // HEAD_DIM

    # Load values (out‑of‑bounds loads return zero or one for mask)
    l_val   = tl.load(left_ptr       + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr       + mask_idx, mask=mask, other=tl.float16(1.0))

    # Fuse the three multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    # Store results back in‑place
    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Entry‑point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32 – stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse the five linear layers into a single matmul
    # ------------------------------------------------------------------
    # Cast weights once to fp16
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension (out_features, in_features)
    combined_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16).contiguous()   # shape (5*H, D)

    # Single linear pass -> shape [B, N, N, 5*H]
    combined = F.linear(x_f16, combined_w)   # fp16

    # Split into the five tensors
    left, left_gate, right, right_gate, out_gate = combined.split(
        [hidden_dim, hidden_dim, hidden_dim, hidden_dim, hidden_dim],
        dim=-1,
    )

    # Gate activations
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5) Optional mask & gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Cast mask to fp16 (0/1) and make it contiguous
        mask_f16 = mask.to(torch.float16).contiguous()
        # Ensure contiguous tensors for pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                     # B·N·N·H  (int32 fits)
        BLOCK = 1024                              # 1‑D block size (tuned for H100)
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – just apply the gates in‑place
        left = left.mul_(left_gate)
        right = right.mul_(right_gate)

    # ------------------------------------------------------------------
    # 6) Core N³ update: batched GEMM L @ Rᵀ (all fp16)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # Batched matrix multiply (Tensor‑core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] (fp16)

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + output‑gate (still fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # element‑wise gating

    # ------------------------------------------------------------------
    # 8) Final linear projection back to the model dimension D
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (D, H)
    out = F.linear(out, to_out_w)   # fp16, shape [B, N, N, D]

    # Cast to float32 for the caller (as in the reference implementation)
    out = out.to(torch.float32)

    return out
