"""
Optimized Triton implementation of the TriMul “outgoing” update.

The heavy cubic part (∑_k left_{i,k} * right_{j,k}) is realized as a batched
matrix multiplication L @ Rᵀ where each hidden channel is treated as an
independent batch.  The custom Triton kernel tiles the computation with a
128×128×128 tile (M×N×K) and uses fp16 inputs with fp32 accumulation,
exploiting H100 Tensor‑Cores.  All other operations (LayerNorm, linear
projections, sigmoids, mask) are performed in PyTorch with mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton batched GEMM kernel
# -------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bn, stride_bk,   # B: (batch, N, K)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch

    # off‑sets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension, tiling by BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile transposed (K x N) -> we load B as (K, N)
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # dot: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a_tile, b_tile)

    # store result (cast back to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor,
                         BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 128):
    """
    Batched GEMM A @ Bᵀ for tensors in fp16.

    Args:
        A: (batch, M, K) fp16 tensor
        B: (batch, N, K) fp16 tensor
    Returns:
        C: (batch, M, N) fp16 tensor
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    # Number of warps: 8 works well for 128×128×128 tiles, 16 for larger
    num_warps = 8 if BLOCK_M <= 128 and BLOCK_N <= 128 else 16

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,
    )
    return C

# Custom kernel entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """Forward pass of the TriMul “outgoing” update (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: [B, N, N, D] torch.float32
            - mask: [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys ``dim`` (D) and ``hidden_dim`` (H);
              optional ``nomask`` (bool).

    Returns:
        torch.float32 tensor of shape [B, N, N, D].
    """
    # unpack
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (FP32 for numerical stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,D] FP32

    # -----------------------------------------------------------------
    # 2. Cast to fp16 for the bulk of the work
    # -----------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3. Linear projections (bias‑free) – fp16 weights
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)   # [B,N,N,H] fp16
    right = F.linear(x_f16, right_proj_w) # [B,N,N,H] fp16

    # -----------------------------------------------------------------
    # 4. Optional mask (broadcasted on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 5. Gates (sigmoid) – fp16
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # -----------------------------------------------------------------
    # 6. Apply gates element‑wise
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 7. Core cubic update: batched matmul L @ Rᵀ for each hidden channel
    # -----------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) where each hidden channel is a batch element
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Choose tile size – larger tiles for larger N to reduce launch overhead
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64   # 256→2 K‑tiles
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 128  # 4 K‑tiles
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 128  # 8 K‑tiles for 1024

    # batched GEMM
    out_mat = _batched_gemm_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B*H, N, N] fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------------
    # 8. Output LayerNorm, out‑gate, and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # fp16, shape [B,N,N,D]

    # -----------------------------------------------------------------
    # 9. Cast back to FP32 (API spec)
    # -----------------------------------------------------------------
    return out.to(torch.float32)

    c_batch = C_ptr + pid_b * stride_ch

    # -----------------------------------------------------------------
    # 3) accumulator (FP32)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # 4) loop over K dimension (tiling)
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ----- load A tile (M × K) -------------------------------------------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ----- load B tile as transposed (K × N) ------------------------------------
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ----- dot product (FP32 accumulation) --------------------------------------
        acc += tl.dot(a_tile, b_tile)                     # (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------
    # 5) store result (cast back to fp16)
    # -----------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_gemm_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 128,
) -> torch.Tensor:
    """
    Batched GEMM C = A @ Bᵀ for FP16 tensors using the Triton kernel above.

    Args:
        A: (batch, M, K) fp16 tensor
        B: (batch, N, K) fp16 tensor
    Returns:
        C: (batch, M, N) fp16 tensor
    """
    assert A.is_cuda and B.is_cuda, "Tensors must reside on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    # Choose a reasonable number of warps (8 works well for 128‑tile, 16 for larger)
    num_warps = 8 if max(BLOCK_M, BLOCK_N) <= 128 else 16

    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,
    )
    return C


# -------------------------------------------------------------------------
# Entry point expected by the evaluation harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
                         * dim (int)          – input feature dimension D
                         * hidden_dim (int)   – projection dimension H
                         * nomask (bool, optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]                # D
    hidden_dim = cfg["hidden_dim"]  # H
    nomask = cfg.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for numerical stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  float32

    # -----------------------------------------------------------------
    # 2) Cast to FP16 – the heavy lifting will be done in half precision
    # -----------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Linear projections (bias‑free) – fp16 weights
    # -----------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H] fp16

    # -----------------------------------------------------------------
    # 4) Optional binary mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # -----------------------------------------------------------------
    # 5) Gates (sigmoid) – fp16
    # -----------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # -----------------------------------------------------------------
    # 6) Apply element‑wise gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 7) Core cubic update: batched GEMM L @ Rᵀ (each hidden channel = batch)
    # -----------------------------------------------------------------
    B, N, _, H = left.shape  # B = batch size, H = hidden_dim

    # reshape to (B·H, N, N) – each hidden channel becomes an independent batch item
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # pick tile sizes – larger K‑tiles for larger N to reduce loop count
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 128

    out_mat = _batched_gemm_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B·H, N, N]  fp16

    # reshape back to the original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate multiplication and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16, shape [B, N, N, D]

    # -----------------------------------------------------------------
    # 9) Cast back to FP32 (API contract)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
