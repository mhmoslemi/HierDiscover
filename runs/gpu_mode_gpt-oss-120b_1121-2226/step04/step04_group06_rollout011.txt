@triton.jit
def _ln_gate_kernel(
    out_ptr, out_gate_ptr, out_processed_ptr,
    ln_weight_ptr, ln_bias_ptr,
    stride_out_row, stride_out_col, # strides for out (pos, h)
    stride_gate_row, stride_gate_col,
    N_POS: tl.constexpr, H: tl.constexpr, BLOCK_POS: tl.constexpr, BLOCK_H: tl.constexpr, EPS: tl.constexpr
):
    pid = tl.program_id(0)
    pos = pid * BLOCK_POS + tl.arange(0, BLOCK_POS)
    mask_pos = pos < N_POS

    # compute mean and var across H
    # initialize sums
    sum_val = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    for i in range(0, H, BLOCK_H):
        off_h = i + tl.arange(0, BLOCK_H)
        mask_h = off_h < H
        # load values for each pos
        off = pos[:, None] * stride_out_row + off_h[None, :] * stride_out_col
        val = tl.load(out_ptr + off, mask=mask_pos[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq += tl.sum(val * val, axis=1)
    mean = sum_val / H
    var = sum_sq / H - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Now compute normalized and apply weight, bias, gating -> write result
    for i in range(0, H, BLOCK_H):
        off_h = i + tl.arange(0, BLOCK_H)
        mask_h = off_h < H
        off_out = pos[:, None] * stride_out_row + off_h[None, :] * stride_out_col
        val = tl.load(out_ptr + off_out, mask=mask_pos[:, None] & mask_h[None, :], other=0.0).to(tl.float32)
        # norm
        norm = (val - mean[:, None]) * inv_std[:, None]
        # scale & bias
        w = tl.load(ln_weight_ptr + off_h, mask=mask_h, other=0.0)
        b = tl.load(ln_bias_ptr + off_h, mask=mask_h, other=0.0)
        norm = norm * w + b
        # out_gate
        off_gate = pos[:, None] * stride_gate_row + off_h[None, :] * stride_gate_col
        gate = tl.load(out_gate_ptr + off_gate, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        out_val = norm * gate
        tl.store(out_processed_ptr + off_out, out_val.to(tl.float16), mask=mask_pos[:, None] & mask_h[None, :])
```

```
@triton.jit
def _ln_gate_kernel(
    out_ptr, gate_ptr, out_processed_ptr,
    weight_ptr, bias_ptr,
    stride_out_row, stride_out_col,
    stride_gate_row, stride_gate_col,
    n_pos: tl.constexpr,
    hidden_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_POS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    start_pos = pid * BLOCK_POS
    offs_pos = start_pos + tl.arange(0, BLOCK_POS)
    mask_pos = offs_pos < n_pos

    # first pass: compute sum and sumsq for each pos
    sum_val = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    # loop over hidden dim
    for off_h in range(0, hidden_dim, BLOCK_H):
        cur_h = off_h + tl.arange(0, BLOCK_H)
        mask_h = cur_h < hidden_dim

        # compute offset for out values
        out_ptrs = out_ptr + offs_pos[:, None] * stride_out_row + cur_h[None, :] * stride_out_col
        val = tl.load(out_ptrs, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        # cast to float32 for reduction
        val_f32 = val.to(tl.float32)
        sum_val += tl.sum(val_f32, axis=1)
        sum_sq += tl.sum(val_f32 * val_f32, axis=1)

    # compute mean and inv_std
    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass: apply norm, scale, bias, multiply gate, store
    for off_h in range(0, hidden_dim, BLOCK_H):
        cur_h = off_h + tl.arange(0, BLOCK_H)
        mask_h = cur_h < hidden_dim

        out_ptrs = out_ptr + offs_pos[:, None] * stride_out_row + cur_h[None, :] * stride_out_col
        val = tl.load(out_ptrs, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        # normalize
        norm = (val_f32 - mean[:, None]) * inv_std[:, None]

        # scale and bias
        w = tl.load(weight_ptr + cur_h, mask=mask_h, other=0.0)
        b = tl.load(bias_ptr + cur_h, mask=mask_h, other=0.0)
        # broadcast to each pos
        norm = norm * w + b

        # multiply by gate
        gate_ptrs = gate_ptr + offs_pos[:, None] * stride_gate_row + cur_h[None, :] * stride_gate_col
        gate_val = tl.load(gate_ptrs, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)

        out_val = (norm * gate_val.to(tl.float32)).to(tl.float16)

        # store
        out_processed_ptrs = out_processed_ptr + offs_pos[:, None] * stride_out_row + cur_h[None, :] * stride_out_col

def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm weight and bias
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    # Compute initial layernorm in fp16
    # Use torch.layer_norm with weight/bias dtype match
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    ).to(torch.float16)   # result already fp16 but ensure dtype
    # Actually x is fp16 after cast.

    # Apply mask before projections if needed
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        x = x * mask_f

    # Prepare combined weights
    left_comb_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_comb_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # Linear projection
    left_comb = F.linear(x, left_comb_w)
    right_comb = F.linear(x, right_comb_w)

    # Gated projection via Triton (proj * sigmoid(gate))
    left = _gated_projection(left_comb, hidden_dim)
    right = _gated_projection(right_comb, hidden_dim)

    # Out gate
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(input_tensor.to(torch.float16), out_gate_w))

    # Core cubic update via batched bmm
    B, N, _, _ = left.shape
    # reshape left/right for bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply output layernorm and out_gate via fused kernel
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Kernel for gated projection: proj * sigmoid(gate)
@triton.jit
def _proj_gate_kernel(
    comb_ptr, out_ptr,
    stride_comb_row, stride_comb_col,
    stride_out_row, stride_out_col,
    n_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_POS: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_POS
    offs = start + tl.arange(0, BLOCK_POS)
    mask_pos = offs < n_rows

    # indices within the combined vector
    proj_off = tl.arange(0, hidden_dim)
    gate_off = proj_off + hidden_dim

    # load projection part
    proj = tl.load(
        comb_ptr + offs[:, None] * stride_comb_row + proj_off[None, :] * stride_comb_col,
        mask=mask_pos[:, None],
        other=0.0,
    )
    # load gate part
    gate = tl.load(
        comb_ptr + offs[:, None] * stride_comb_row + gate_off[None, :] * stride_comb_col,
        mask=mask_pos[:, None],
        other=0.0,
    )
    # sigmoid (fp16)
    gate = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * gate

    tl.store(
        out_ptr + offs[:, None] * stride_out_row + proj_off[None, :] * stride_out_col,
        out,
        mask=mask_pos[:, None],
    )

def _gated_projection(comb_tensor: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Apply elementwise sigmoid-gating to a tensor that contains projection and gate
    concatenated along the last dimension.
    """
    # shape (..., 2*hidden_dim)
    flat = comb_tensor.view(-1, hidden_dim * 2)
    n_rows = flat.shape[0]
    out = torch.empty((n_rows, hidden_dim), dtype=flat.dtype, device=flat.device)

    BLOCK_POS = 64  # tuning parameter
    grid = (triton.cdiv(n_rows, BLOCK_POS),)

    _proj_gate_kernel[grid](
        flat,
        out,
        flat.stride(0),
        flat.stride(1),
        out.stride(0),
        out.stride(1),
        n_rows,
        hidden_dim,
        BLOCK_POS=BLOCK_POS,
        num_warps=4,
    )
    # reshape back to original leading dims
    return out.view(comb_tensor.shape[:-1] + (hidden_dim,))

# Kernel for fused output layernorm + out_gate multiplication
@triton.jit
def _ln_gate_kernel(
    out_ptr, gate_ptr, out_proc_ptr,
    weight_ptr, bias_ptr,
    stride_out_row, stride_out_col,
    stride_gate_row, stride_gate_col,
    n_pos: tl.constexpr,
    hidden_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_POS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_POS
    pos_off = start + tl.arange(0, BLOCK_POS)
    mask_pos = pos_off < n_pos

    # first pass: compute sum and sum of squares
    sum_val = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    for off in range(0, hidden_dim, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < hidden_dim

        # load out values (fp16)
        ptr = out_ptr + pos_off[:, None] * stride_out_row + h[None, :] * stride_out_col
        val = tl.load(ptr, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        sum_val += tl.sum(val_f32, axis=1)
        sum_sq += tl.sum(val_f32 * val_f32, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass: normalize, scale+bias, multiply gate and store
    for off in range(0, hidden_dim, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < hidden_dim

        # load out values again
        ptr_out = out_ptr + pos_off[:, None] * stride_out_row + h[None, :] * stride_out_col
        val = tl.load(ptr_out, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        # normalize
        norm = (val_f32 - mean[:, None]) * inv_std[:, None]

        # weight and bias (fp16 -> cast)
        w = tl.load(weight_ptr + h, mask=mask_h, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + h, mask=mask_h, other=0.0).to(tl.float32)

        norm = norm * w + b

        # gate
        ptr_gate = gate_ptr + pos_off[:, None] * stride_gate_row + h[None, :] * stride_gate_col
        gate = tl.load(ptr_gate, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        gate_f32 = gate.to(tl.float32)

        out_val = (norm * gate_f32).to(tl.float16)

        # store
        ptr_out_proc = out_proc_ptr + pos_off[:, None] * stride_out_row + h[None, :] * stride_out_col
        tl.store(ptr_out_proc, out_val, mask=mask_pos[:, None] & mask_h[None, :])

def _ln_gate_fused(
    out_tensor: torch.Tensor,
    out_gate: torch.Tensor,
    ln_weight: torch.Tensor,
    ln_bias: torch.Tensor,
    eps: float = 1e-5,

    The implementation stays entirely in fp16 after an initial layer‑norm,
    fuses projection+gate via a custom Triton kernel, applies the mask before
    the projections, and fuses the output layer‑norm, out‑gate multiplication,
    and scaling into a single Triton kernel before the final linear projection.
    """
    # ------------------------------------------------------------------
    # 1) unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (fp16)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    ).to(torch.float16)   # shape [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 3) Optional binary mask (applied before projection)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        x = x * m

    # ------------------------------------------------------------------
    # 4) Fused projection+gate for left and right streams
    # ------------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, dim)

    left_comb = F.linear(x, left_comb_w)   # (B,N,N,2*H)
    right_comb = F.linear(x, right_comb_w)  # (B,N,N,2*H)

    # Apply sigmoid‑gate via custom Triton kernel
    left = _gated_projection(left_comb, hidden_dim)   # (B,N,N,H)
    right = _gated_projection(right_comb, hidden_dim) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Out‑gate (sigmoid of a linear projection on the *unmasked* tensor)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)  # (H, dim)
    out_gate = torch.sigmoid(F.linear(input_tensor.to(torch.float16), out_gate_w))  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Core cubic update: batched matrix multiply L @ Rᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # (B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 7) Fused output LayerNorm + out‑gate multiplication
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = _ln_gate_fused(out, out_gate, to_out_norm_w, to_out_norm_b, eps=1e-5)  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    out = F.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 9) Cast back to fp32 for API compliance
    # ------------------------------------------------------------------
    return out.to(torch.float32)

@triton.jit
def _ln_simple_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    stride_in_row, stride_in_col,
    n_pos: tl.constexpr,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_POS: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_POS
    pos = start + tl.arange(0, BLOCK_POS)
    mask_pos = pos < n_pos

    # First pass: compute sum and sumsq
    sum_val = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    for off in range(0, dim, BLOCK_D):
        d = off + tl.arange(0, BLOCK_D)
        mask_d = d < dim

        ptr = inp_ptr + pos[:, None] * stride_in_row + d[None, :] * stride_in_col
        val = tl.load(ptr, mask=mask_pos[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)
        sum_val += tl.sum(val_f32, axis=1)
        sum_sq += tl.sum(val_f32 * val_f32, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: normalize, apply weight and bias, store
    for off in range(0, dim, BLOCK_D):
        d = off + tl.arange(0, BLOCK_D)
        mask_d = d < dim

        ptr = inp_ptr + pos[:, None] * stride_in_row + d[None, :] * stride_in_col
        val = tl.load(ptr, mask=mask_pos[:, None] & mask_d[None, :], other=0.0)
        val_f32 = val.to(tl.float32)

        norm = (val_f32 - mean[:, None]) * inv_std[:, None]

        w = tl.load(weight_ptr + d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + d, mask=mask_d, other=0.0).to(tl.float32)

        out = (norm * w + b).to(tl.float16)

        out_ptrs = out_ptr + pos[:, None] * stride_in_row + d[None, :] * stride_in_col
        tl.store(out_ptrs, out, mask=mask_pos[:, None] & mask_d[None, :])
```

"""
Triton‑accelerated forward pass for the “outgoing” TriMul operator.

Key ideas
---------
1.  Input LayerNorm is performed entirely in fp16 by a custom Triton kernel.
2.  Binary mask (when present) is applied *before* the linear projections.
    Since the mask is a scalar per (i,k) position it can be merged into the
    linear operation without changing the semantics.
3.  The two linear projections (proj + gate) are fused:
    - A single weight matrix of shape (2·H, D) is used.
    - A Triton kernel turns the concatenated tensor
      (proj,gate) → proj * sigmoid(gate)  (no intermediate tensors).
4.  The core cubic update Σₖ L[i,k,h]·R[j,k,h] is still the dominant work.
    It is performed as a batched GEMM (torch.bmm) which on H100 uses
    Tensor‑cores and is faster than any pure‑Triton implementation.
5.  Output LayerNorm, the element‑wise multiplication with the out‑gate
    and the affine scale+bias are fused into a single Triton kernel.
6.  The final projection back to the original dimension is a standard
    torch.nn.functional.linear (still fp16).  The result is cast back to
    fp32 for API compatibility.

The kernels reduce the number of memory passes (the original reference
used three separate passes for the output norm and a separate Triton
kernel for the element‑wise product).  All heavy linear algebra stays in
fp16, which halves bandwidth and keeps the compute on Tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: simple LayerNorm (no gating) – used for the input tensor
# ----------------------------------------------------------------------
@triton.jit
def _ln_simple_kernel(
    inp_ptr, out_ptr,                # fp16 input / output (flattened)
    weight_ptr, bias_ptr,            # fp16 1‑D weight & bias
    stride_in_row, stride_in_col,    # strides of the flattened tensor
    n_pos: tl.constexpr,             # total number of (B·N·N) positions
    dim: tl.constexpr,               # hidden dimension (input dim)
    eps: tl.constexpr,               # epsilon for stability
    BLOCK_POS: tl.constexpr,         # positions per program
    BLOCK_D: tl.constexpr,           # reduction block for dim
):
    pid = tl.program_id(0)
    start = pid * BLOCK_POS
    pos = start + tl.arange(0, BLOCK_POS)
    mask_pos = pos < n_pos

    # ----- first pass: compute mean & variance -----
    sum_val = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    for off in range(0, dim, BLOCK_D):
        d = off + tl.arange(0, BLOCK_D)
        mask_d = d < dim

        ptr = inp_ptr + pos[:, None] * stride_in_row + d[None, :] * stride_in_col
        v = tl.load(ptr, mask=mask_pos[:, None] & mask_d[None, :], other=0.0)
        v_f32 = v.to(tl.float32)

        sum_val += tl.sum(v_f32, axis=1)
        sum_sq += tl.sum(v_f32 * v_f32, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalize, scale, bias -----
    for off in range(0, dim, BLOCK_D):
        d = off + tl.arange(0, BLOCK_D)
        mask_d = d < dim

        ptr = inp_ptr + pos[:, None] * stride_in_row + d[None, :] * stride_in_col
        v = tl.load(ptr, mask=mask_pos[:, None] & mask_d[None, :], other=0.0)
        v_f32 = v.to(tl.float32)

        norm = (v_f32 - mean[:, None]) * inv_std[:, None]

        w = tl.load(weight_ptr + d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + d, mask=mask_d, other=0.0).to(tl.float32)

        out = (norm * w + b).to(tl.float16)

        out_ptrs = out_ptr + pos[:, None] * stride_in_row + d[None, :] * stride_in_col
        tl.store(out_ptrs, out, mask=mask_pos[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Kernel 2: fused projection + gate (proj * sigmoid(gate))
# ----------------------------------------------------------------------
@triton.jit
def _proj_gate_kernel(
    comb_ptr, out_ptr,               # fp16 input (n,2·H) and output (n,H)
    stride_comb_row, stride_comb_col,
    stride_out_row, stride_out_col,
    n_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_POS: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_POS
    offs = start + tl.arange(0, BLOCK_POS)
    mask_row = offs < n_rows

    # offsets inside the concatenated vector
    proj_off = tl.arange(0, hidden_dim)
    gate_off = proj_off + hidden_dim

    # load projection
    proj = tl.load(
        comb_ptr + offs[:, None] * stride_comb_row + proj_off[None, :] * stride_comb_col,
        mask=mask_row[:, None],
        other=0.0,
    )
    # load gate
    gate = tl.load(
        comb_ptr + offs[:, None] * stride_comb_row + gate_off[None, :] * stride_comb_col,
        mask=mask_row[:, None],
        other=0.0,
    )
    # sigmoid (fp16)
    gate = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * gate

    tl.store(
        out_ptr + offs[:, None] * stride_out_row + proj_off[None, :] * stride_out_col,
        out,
        mask=mask_row[:, None],
    )


def _gated_projection(comb_tensor: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Input:  (..., 2*hidden_dim)   – concatenated (proj,gate)
    Output: (..., hidden_dim)     – proj * sigmoid(gate)
    """
    flat = comb_tensor.view(-1, hidden_dim * 2)
    n_rows = flat.shape[0]
    out = torch.empty((n_rows, hidden_dim), dtype=flat.dtype, device=flat.device)

    BLOCK_POS = 64
    grid = (triton.cdiv(n_rows, BLOCK_POS),)

    _proj_gate_kernel[grid](
        flat,
        out,
        flat.stride(0),
        flat.stride(1),
        out.stride(0),
        out.stride(1),
        n_rows,
        hidden_dim,
        BLOCK_POS=BLOCK_POS,
        num_warps=4,
    )
    return out.view(comb_tensor.shape[:-1] + (hidden_dim,))


# ----------------------------------------------------------------------
# Kernel 3: fused output LayerNorm + out‑gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def _ln_gate_kernel(
    out_ptr, gate_ptr, out_proc_ptr,
    weight_ptr, bias_ptr,
    stride_out_row, stride_out_col,
    stride_gate_row, stride_gate_col,
    n_pos: tl.constexpr,
    hidden_dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_POS: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_POS
    pos = start + tl.arange(0, BLOCK_POS)
    mask_pos = pos < n_pos

    # ----- first pass: mean / variance -----
    sum_val = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    for off in range(0, hidden_dim, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < hidden_dim

        ptr = out_ptr + pos[:, None] * stride_out_row + h[None, :] * stride_out_col
        v = tl.load(ptr, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        v_f32 = v.to(tl.float32)

        sum_val += tl.sum(v_f32, axis=1)
        sum_sq += tl.sum(v_f32 * v_f32, axis=1)

    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ----- second pass: normalize, scale, bias, multiply gate, store -----
    for off in range(0, hidden_dim, BLOCK_H):
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < hidden_dim

        # load out
        ptr_out = out_ptr + pos[:, None] * stride_out_row + h[None, :] * stride_out_col
        v = tl.load(ptr_out, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        v_f32 = v.to(tl.float32)

        # normalize
        norm = (v_f32 - mean[:, None]) * inv_std[:, None]

        # affine transform
        w = tl.load(weight_ptr + h, mask=mask_h, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + h, mask=mask_h, other=0.0).to(tl.float32)
        norm = norm * w + b

        # out‑gate
        ptr_gate = gate_ptr + pos[:, None] * stride_gate_row + h[None, :] * stride_gate_col
        g = tl.load(ptr_gate, mask=mask_pos[:, None] & mask_h[None, :], other=0.0)
        g_f32 = g.to(tl.float32)

        out_val = (norm * g_f32).to(tl.float16)

        # store
        ptr_out_proc = out_proc_ptr + pos[:, None] * stride_out_row + h[None, :] * stride_out_col
        tl.store(ptr_out_proc, out_val, mask=mask_pos[:, None] & mask_h[None, :])


def _ln_gate_fused(
    out_tensor: torch.Tensor,
    out_gate: torch.Tensor,
    ln_weight: torch.Tensor,
    ln_bias: torch.Tensor,
    eps: float = 1e-5,
) -> torch.Tensor:
    """
    Fuse LayerNorm (per last dim) + out_gate element‑wise multiplication.
    Operates in fp16, uses float32 for the statistics.
    """
    B, N, N2, H = out_tensor.shape
    assert N == N2
    n_pos = B * N * N

    out_flat = out_tensor.view(-1, H)
    gate_flat = out_gate.view(-1, H)
    out_proc = torch.empty_like(out_flat)

    BLOCK_POS = 128
    BLOCK_H = 32
    grid = (triton.cdiv(n_pos, BLOCK_POS),)

    _ln_gate_kernel[grid](
        out_flat,
        gate_flat,
        out_proc,
        ln_weight,
        ln_bias,
        out_flat.stride(0),
        out_flat.stride(1),
        gate_flat.stride(0),
        gate_flat.stride(1),
        n_pos,
        H,
        eps,
        BLOCK_POS=BLOCK_POS,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    return out_proc.view(B, N, N, H)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.
    """
    # --------------------------------------------------------------
    # 1) unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (fp16) – custom Triton kernel
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    # flatten to (B·N·N, D)
    x_fp16 = input_tensor.to(torch.float16)
    B, N, _, _ = x_fp16.shape
    n_pos = B * N * N
    x_flat = x_fp16.view(-1, dim)
    x_norm_flat = torch.empty_like(x_flat)

    BLOCK_POS = 128
    BLOCK_D = 32
    grid = (triton.cdiv(n_pos, BLOCK_POS),)

    _ln_simple_kernel[grid](
        x_flat,
        x_norm_flat,
        norm_w,
        norm_b,
        x_flat.stride(0),
        x_flat.stride(1),
        n_pos,
        dim,
        eps=1e-5,
        BLOCK_POS=BLOCK_POS,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    x = x_norm_flat.view(B, N, N, dim)   # now normalized, fp16

    # --------------------------------------------------------------
    # 3) out_gate (sigmoid of a linear projection on the *normalized* tensor)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # (H, D)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))          # (B,N,N,H) fp16

    # --------------------------------------------------------------
    # 4) Optional mask – applied *before* the left/right projections
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        x_proj = x * mask_f
    else:
        x_proj = x

    # --------------------------------------------------------------
    # 5) Fused projection + gate for left and right streams
    # --------------------------------------------------------------
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)          # (2·H, D)
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # (2·H, D)

    left_comb = F.linear(x_proj, left_comb_w)   # (B,N,N,2·H)
    right_comb = F.linear(x_proj, right_comb_w)  # (B,N,N,2·H)

    left = _gated_projection(left_comb, hidden_dim)   # (B,N,N,H)
    right = _gated_projection(right_comb, hidden_dim) # (B,N,N,H)

    # --------------------------------------------------------------
    # 6) Core cubic update: batched matrix multiply L @ Rᵀ
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matmul (Tensor‑core GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # back to (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Fuse output LayerNorm, affine scale/bias and out_gate multiplication
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = _ln_gate_fused(out, out_gate, to_out_norm_w, to_out_norm_b, eps=1e-5)  # (B,N,N,H)

    # --------------------------------------------------------------
    # 8) Final linear projection back to the original dimension
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)   # (B,N,N,D) fp16

    # --------------------------------------------------------------
    # 9) Cast to FP32 for API compliance
    # --------------------------------------------------------------
    return out.to(torch.float32)
