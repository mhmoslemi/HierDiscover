"""
TriMul “outgoing” forward pass – Triton‑accelerated.

Algorithm
---------
1. Layer‑norm the input (FP32) → x.
2. Cast to FP16.
3. Fuse the two projection linear layers per side (proj + gate) into a single
   linear by concatenating the weight matrices → one matmul per side.
   Split the result, apply sigmoid to the gate part.
4. Fuse mask (if present) and the element‑wise gate multiplication in a tiny
   Triton kernel.
5. Core N³ update: for every hidden channel d,
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched GEMM (left @ rightᵀ).  A custom Triton kernel performs the
   whole batch in one launch (grid = (M‑blocks, N‑blocks, batch)).
   Multiplication is FP16 with FP32 accumulation.
6. Output LayerNorm (FP16) → to_out_norm.
7. Multiply by the out‑gate (sigmoid) and a final linear projection back to
   the original dimension.
8. Cast the result to FP32 (required API type).

Both the gate‑mask fusion and the batched GEMM are written in Triton,
fulfilling the “at least one custom kernel” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# 1) Gate + (optional) mask fusion kernel                                    #
# --------------------------------------------------------------------------- #
@triton.jit
def _gate_mask_kernel(
    left_ptr,          # fp16 * total_elems
    right_ptr,         # fp16 * total_elems
    left_gate_ptr,     # fp16 * total_elems
    right_gate_ptr,    # fp16 * total_elems
    mask_ptr,          # fp16 * (B * N * N)  – flattened, only if HAS_MASK=1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * left_gate  * (mask if present)
       right = right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load everything
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask is broadcasted over the hidden dimension
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * m
        right = right * m

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # store back (in‑place)
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """Launch the Triton kernel that fuses mask (if any) and gate multiplication."""
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1).contiguous()
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


# --------------------------------------------------------------------------- #
# 2) Batched GEMM kernel (left @ rightᵀ)                                      #
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    A_ptr,               # left  : [B*H, M, K]  fp16
    B_ptr,               # rightᵀ: [B*H, K, N]  fp16
    C_ptr,               # out   : [B*H, M, N]  fp16
    batch, M, N, K,      # integer dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    C[b, i, j] = Σ_k A[b, i, k] * B[b, k, j]   (FP16×FP16 → FP32 accumulation)
    """
    pid_b = tl.program_id(2)                 # batch index
    pid_m = tl.program_id(0)                 # row block
    pid_n = tl.program_id(1)                 # col block

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # pointers for the current block of A and B
        a_ptrs = (
            A_ptr
            + pid_b * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            B_ptr
            + pid_b * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # bounds checking
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # fp16·fp16 → fp32 accumulation
        acc += tl.dot(a, b)

    # write back
    c_ptrs = (
        C_ptr
        + pid_b * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_gemm(left, right):
    """
    Performs (left) @ (rightᵀ) for a batch of matrices using the Triton kernel.
    left, right: tensors of shape [B, H, N, N] (FP16, contiguous)
    Returns: tensor of shape [B, H, N, N] (FP16, contiguous)
    """
    B, H, N, _ = left.shape
    batch = B * H
    M = N
    K = N
    # reshape to (batch, N, N)
    A = left.view(batch, N, N)
    B_ = right.view(batch, N, N)

    # allocate output
    C = torch.empty_like(A)

    # strides (all tensors are contiguous)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_k, stride_b_n = B_.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # tile sizes – tuned for H100 FP16 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        ( (M + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          batch ),
    )

    _batched_gemm_kernel[grid](
        A,
        B_,
        C,
        batch,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C.view(B, H, N, N)   # shape back to [B, H, N, N]


# --------------------------------------------------------------------------- #
# 3) Full custom kernel                                                       #
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.
    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        Tensor of shape [B, N, N, D] (float32).
    """
    # --------------------------------------------------------------
    # 0) Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] float32

    # --------------------------------------------------------------
    # 2) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Fused projection + gate (two linear calls instead of four)
    # --------------------------------------------------------------
    # left side: concat(proj, gate) → (2*H, D)
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    left_combined = F.linear(x_f16, left_combined_w)               # [B, N, N, 2H]
    left = left_combined[..., :hidden_dim]                         # proj
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])     # gate

    # right side
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_combined = F.linear(x_f16, right_combined_w)             # [B, N, N, 2H]
    right = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # out‑gate (used after the core update)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))          # [B, N, N, H]

    # --------------------------------------------------------------
    # 4) Fuse mask and element‑wise gates (tiny Triton kernel)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(left, right, left_gate, right_gate, mask_f16, hidden_dim)

    # --------------------------------------------------------------
    # 5) Core cubic operation (batched GEMM) – outgoing update
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Equivalent to (left @ rightᵀ) for each hidden channel.
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # [B, N, N, H]
    # reshape to (B, H, N, N) and make contiguous for easy strides
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (transposed K dimension)

    # batched GEMM via Triton
    out_perm = _batched_gemm(left_perm, right_perm)    # [B, H, N, N]

    # back to [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # fp16

    # --------------------------------------------------------------
    # 6) Output LayerNorm (FP16), out‑gate multiplication and final proj
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate                # apply out‑gate (already fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)       # [B, N, N, D] fp16

    # --------------------------------------------------------------
    # 7) Cast back to FP32 (API requirement) and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
