    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

"""
TriMul “outgoing” update – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32 for numerical stability).
2. Cast to FP16 and perform *fused* linear projections:
   - left_proj + left_gate  →  weight (2·H, D)
   - right_proj + right_gate → weight (2·H, D)
   The result is split into projection and gate tensors.
3. (Optional) binary mask is applied to the two projections.
   Gates are passed through a sigmoid and multiplied in‑place.
4. Core cubic contraction Σₖ left[b,i,k,h]·right[b,j,k,h] is computed with a
   purpose‑built Triton batched GEMM kernel.  The kernel works on the
   original (B,N,N,H) layout, avoiding any expensive transposes.
5. Output LayerNorm (FP16), multiplication by the out‑gate and final
   linear projection back to the original feature dimension.
6. Cast the result back to FP32 for the public API.

The heavy N³ work is performed entirely inside the Triton kernel, while
the remaining element‑wise operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel – computes out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float16 *
    B, N, H,           # int
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs – map together (batch, i‑tile, j‑tile)
    # ------------------------------------------------------------------
    batch_id = tl.program_id(0)               # 0 … B·H‑1
    pid_i    = tl.program_id(1)               # i‑tile index
    pid_j    = tl.program_id(2)               # j‑tile index

    # ------------------------------------------------------------------
    # Decode batch_id → (b, h)
    # ------------------------------------------------------------------
    b = batch_id // H
    h = batch_id %  H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop – each iteration processes a BLOCK_K slice of the sum
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile   shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_l_b + h * stride_l_h + \
            offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # Load right tile  shape (BLOCK_K, BLOCK_N)
        # right layout: (b, j, k, h) → need (k, j) ordering
        right_ptrs = right_ptr + \
            b * stride_r_b + h * stride_r_h + \
            offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Matrix multiply (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)  # FP16·FP16 → FP32 accumulation

    # ------------------------------------------------------------------
    # Store the result back (convert to FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_o_b + h * stride_o_h + \
        offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _batched_gemm(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton GEMM kernel.
    left, right : torch.Tensor of shape (B, N, N, H), dtype=torch.float16, contiguous.
    Returns out : torch.Tensor of same shape and dtype.
    """
    assert left.is_cuda and right.is_cuda
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Strides in **elements** (not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Tunable block sizes – a good trade‑off for the tested N (256–1024) on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (B * H,                # batch dimension (b, h)
            tl.cdiv(N, BLOCK_M),  # i‑tiles
            tl.cdiv(N, BLOCK_N))  # j‑tiles

    _batched_gemm_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_l_b=s_l_b,
        stride_l_i=s_l_i,
        stride_l_k=s_l_k,
        stride_l_h=s_l_h,
        stride_r_b=s_r_b,
        stride_r_i=s_r_i,
        stride_r_k=s_r_k,
        stride_r_h=s_r_h,
        stride_o_b=s_o_b,
        stride_o_i=s_o_i,
        stride_o_j=s_o_j,
        stride_o_h=s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update (no backward).

    Args
    ----
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, D] (float32)
        mask         : torch.Tensor [B, N, N] (bool/int) or None
        weights      : dict of model parameters (float32)
        config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack and basic shapes
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – FP32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for all heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused left/right projections + gates (bias‑free linear layers)
    # --------------------------------------------------------------
    # Concatenate projection and gate weights for a single matmul
    left_w_proj = weights["left_proj.weight"]          # (H, D)
    left_w_gate = weights["left_gate.weight"]          # (H, D)
    left_w_fused = torch.cat([left_w_proj, left_w_gate], dim=0).to(torch.float16)  # (2H, D)

    right_w_proj = weights["right_proj.weight"]
    right_w_gate = weights["right_gate.weight"]
    right_w_fused = torch.cat([right_w_proj, right_w_gate], dim=0).to(torch.float16)  # (2H, D)

    # One linear per side
    left_comb = F.linear(x_f16, left_w_fused)   # [B, N, N, 2H]
    right_comb = F.linear(x_f16, right_w_fused) # [B, N, N, 2H]

    # Split into projection / gate tensors
    left_proj = left_comb[..., :hidden_dim]                # (B,N,N,H)
    left_gate_pre = left_comb[..., hidden_dim:]            # (B,N,N,H)
    right_proj = right_comb[..., :hidden_dim]
    right_gate_pre = right_comb[..., hidden_dim:]

    # Apply mask *before* gating (replicates reference implementation)
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * mask_f16
        right_proj = right_proj * mask_f16

    # Sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # Apply gates – keep fp16 throughout
    left = (left_proj * left_gate).contiguous()
    right = (right_proj * right_gate).contiguous()

    # --------------------------------------------------------------
    # 5) Out‑gate (computed from the normalized input)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # (H, D)
    out_gate_pre = F.linear(x_f16, out_gate_w)                # (B,N,N,H)
    out_gate = torch.sigmoid(out_gate_pre)                    # (B,N,N,H)

    # --------------------------------------------------------------
    # 6) Core cubic contraction – batched GEMM via Triton
    # --------------------------------------------------------------
    # left/right are already (B,N,N,H) fp16, contiguous.
    # Output shape is also (B,N,N,H) fp16.
    out_hidden = _batched_gemm(left, right)   # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm, multiplication by out‑gate
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out_hidden = out_hidden * out_gate  # fp16 element‑wise

    # --------------------------------------------------------------
    # 8) Final projection back to original dimension
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (D, H)
    out = F.linear(out_hidden, to_out_w)                  # (B,N,N,D) fp16

    # --------------------------------------------------------------
    # 9) Cast back to FP32 for the public API
    # --------------------------------------------------------------
    return out.to(torch.float32)
