    N, H,

    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    pid_batch = tl.program_id(0)  # range: 0 .. B*H-1
    pid_m = tl.program_id(1)      # row tile
    pid_n = tl.program_id(2)      # col tile

    batch = pid_batch // H
    head = pid_batch % H

    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) → x.
2. Cast x to FP16 for the heavy work.
3. Fuse the two linear projections (proj & gate) of the left and right streams:
   * Concatenate the two weight matrices (2·H × D) and call a single F.linear.
   * Split the result back into projection / gate tensors, apply the binary mask
     (if present) **before** the sigmoid‑gate, and finally multiply
     projection * sigmoid(gate).
4. Compute the cubic update
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM (B·H independent N×N matmuls) using a custom Triton
   kernel.  No intermediate transposes or copies are performed – the kernel
   reads directly from the (B,N,N,H) layout.
5. Apply the output LayerNorm (FP16) and the shared output gate
   (element‑wise product performed by a tiny Triton kernel).
6. Final bias‑free linear projection back to the original dimension.
7. Cast the result to FP32 (API requirement).

The kernel in step 4 is the performance‑critical part: each program handles one
(b, h) pair, tiles the output M×N plane with 64×64 blocks and iterates over the
K‑dimension in 64‑element chunks, accumulating in FP32.  This removes the costly
`.contiguous()` reshapes required by the naïve `torch.bmm` implementation and
exposes the full compute to the GPU tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply for the cubic update
# ------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,                 # fp16*  [B,N,N,H]
    right_ptr,                # fp16*  [B,N,N,H]
    out_ptr,                  # fp16*  [B,N,N,H] – written in‑place
    N,                        # sequence length (int)
    H,                        # hidden dimension (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,   # tile size in the output rows
    BLOCK_N: tl.constexpr,   # tile size in the output cols
    BLOCK_K: tl.constexpr,   # tile size in the reduction dimension
):
    # program IDs
    pid_batch = tl.program_id(0)       # 0 … B*H-1  (each (batch,head) pair)
    pid_m = tl.program_id(1)           # row‑tile of the output
    pid_n = tl.program_id(2)           # col‑tile of the output

    # ------------------------------------------------------------------
    # 1) decode (batch, head) from pid_batch
    # ------------------------------------------------------------------
    b = pid_batch // H                 # batch index
    h = pid_batch % H                  # head index within the batch

    # ------------------------------------------------------------------
    # 2) compute the coordinates of the output tile
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # rows   (i)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # cols   (j)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 3) accumulate over the reduction dimension k
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)      # reduction axis

        mask_k = offs_k < N

        # left:  [i, k]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right: [k, j]  (note the order – we load right[j,k] and treat it as (k,j))
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + h * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # dot product over K
        acc += tl.dot(left, right)   # acc: FP32

    # ------------------------------------------------------------------
    # 4) write the result back (cast to FP16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

# ------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication (out * out_gate)
# ------------------------------------------------------------------
@triton.jit
def _elem_mul_kernel(
    a_ptr,                     # fp16* – both input and output
    b_ptr,                     # fp16* – input
    n_elements,                # total elements in the flat view
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(a_ptr + offs, a * b, mask=mask)


def _elemwise_mul_inplace(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """In‑place a = a * b (both FP16, same shape)."""
    assert a.is_cuda and b.is_cuda
    assert a.shape == b.shape
    a_flat = a.view(-1)
    b_flat = b.view(-1)
    n = a_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    _elem_mul_kernel[grid](
        a_flat,
        b_flat,
        n,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return a


def _tri_mul_compute(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM: out = left @ rightᵀ   for tensors of shape [B,N,N,H].
    Returns a FP16 tensor of the same shape.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides (ints)
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_j, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # tiling parameters – can be tuned per GPU
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch × head dimension
        triton.cdiv(N, BLOCK_M),               # row tiles
        triton.cdiv(N, BLOCK_N),               # column tiles
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        H,
        s_left_b,
        s_left_i,
        s_left_k,
        s_left_h,
        s_right_b,
        s_right_j,
        s_right_k,
        s_right_h,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ------------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module (no gradients).
    Arguments:
        data = (input_tensor, mask, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 – high precision)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (fp32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused projection + gate for left and right streams
    # --------------------------------------------------------------
    # concatenate projection & gate weights -> (2·H, D)
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, D)

    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)  # (2*H, D)

    # Linear → (B,N,N,2*H)  – bias is None
    left_comb = F.linear(x_f16, left_comb_w)    # fp16
    right_comb = F.linear(x_f16, right_comb_w)

    # split into projection & gate components
    left_proj = left_comb[..., :hidden_dim]           # (B,N,N,H)
    left_gate = left_comb[..., hidden_dim:]           # (B,N,N,H)
    right_proj = right_comb[..., :hidden_dim]
    right_gate = right_comb[..., hidden_dim:]

    # optional binary mask (applied before gating, exactly as reference)
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)      # (B,N,N,1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # apply sigmoid gates and combine
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # shared output gate (shape B,N,N,H)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # --------------------------------------------------------------
    # 5) Core cubic update: Σₖ L_{i,k,h} · R_{j,k,h}
    #    (batched mat‑mul implemented in Triton)
    # --------------------------------------------------------------
    out = _tri_mul_compute(left, right)   # fp16, shape (B,N,N,H)

    # --------------------------------------------------------------
    # 6) Output LayerNorm (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # --------------------------------------------------------------
    # 7) Apply out‑gate (element‑wise) – Triton kernel
    # --------------------------------------------------------------
    out = _elemwise_mul_inplace(out, out_gate)   # in‑place fp16 product

    # --------------------------------------------------------------
    # 8) Final linear projection back to the original dimension
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)   # (B,N,N,D)  fp16

    # --------------------------------------------------------------
    # 9) Cast to FP32 for API compliance
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    return out
```

This implementation:
* fuses the two linear projections per stream,
* avoids the costly `contiguous()` reshapes by reading directly from the
  `[B, N, N, H]` layout,
* performs the cubic update with a dedicated Triton kernel (batched GEMM
  across the `(B·H)` independent matrix multiplications),
* uses an in‑place Triton kernel for the output‑gate multiplication,
* retains the original LayerNorm and final projection logic.

The kernel dimensions (`BLOCK_M = BLOCK_N = BLOCK_K = 64`) were chosen to
balance shared‑memory use and occupancy on an H100, giving a substantial
runtime reduction (target < 1 ms for the benchmark configurations).
