Implementation using Triton:

```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_Abatch, stride_Arow, stride_Acol,
    stride_Bbatch, stride_Brow, stride_Bcol,
    stride_Cbatch, stride_Crow, stride_Ccol,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer base for each batch
    a_base = a_ptr + pid_batch * stride_Abatch
    b_base = b_ptr + pid_batch * stride_Bbatch
    c_base = c_ptr + pid_batch * stride_Cbatch

    # a block pointers (M x K)
    a_ptrs = a_base + offs_m[:, None] * stride_Arow + offs_k[None, :] * stride_Acol
    # b block pointers (K x N) for transposed view: original b has row stride stride_Brow, col stride stride_Bcol
    # we need element (k,n) = b[batch, n, k] -> offset = n*stride_Brow + k*stride_Bcol
    b_ptrs = b_base + offs_k[:, None] * stride_Bcol + offs_n[None, :] * stride_Brow

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs = a_ptrs + BLOCK_K * stride_Acol
        b_ptrs = b_ptrs + BLOCK_K * stride_Bcol

    # write back
    c_ptrs = c_base + offs_m[:, None] * stride_Crow + offs_n[None, :] * stride_Ccol
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Coordinates within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-range rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch
    c_base = c_ptr + pid_batch * stride_c_batch

    num_k = tl.cdiv(K, BLOCK_K)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(num_k):
        # Global k index for this tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)   # shape (BLOCK_K,)
        mask_k = k_offset < K

        # Load A tile (M x K)
        a = tl.load(
            a_base + offs_m[:, None] * stride_a_row + k_offset[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )   # (BLOCK_M, BLOCK_K)

        # Load B tile transposed (K x N). Original B layout is (batch, N, N)
        # We need B[batch, n, k] i.e. transposed version.
        b = tl.load(
            b_base + k_offset[:, None] * stride_b_col + offs_n[None, :] * stride_b_row,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)   # accumulate in fp32

    # Write back
    c = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c, acc.to(tl.float16), mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” update.
    The kernel fuses the O(N³) contraction Σₖ L_{i,k}·R_{j,k} as a
    batched matrix multiplication L @ Rᵀ using a custom Triton gemm.
    All other ops (layer‑norms, linear projections, sigmoids) are
    performed in PyTorch (FP16 where possible). The final result is
    returned as float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1) Input LayerNorm (FP32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B,N,N,dim) float32

    # 2) Cast to FP16 for compute‑intensive ops
    x_f16 = x.to(torch.float16)

    # 3) Linear projections (bias‑free)
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # 4) Optional binary mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f16
        right = right * mask_f16

    # 5) Compute gates
    left_gate = torch.sigmoid(F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)))

    # 6) Apply gates
    left = left * left_gate
    right = right * right_gate

    # 7) Reshape for batched GEMM: (B, H, N, N) -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_dim = B * hidden_dim
    left_mat = left_perm.view(batch_dim, N, N)
    right_mat = right_perm.view(batch_dim, N, N)

    # 8) Core O(N³) contraction via Triton batched GEMM
    out_mat = batched_gemm_triton(left_mat, right_mat)   # (B*H, N, N), FP16

    # 9) Reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 10) Output LayerNorm (FP16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # 11) Multiply by out_gate (broadcasted)
    out = out * out_gate

    # 12) Final projection back to original dimension
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # 13) Return in FP32 as required by the API
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a custom Triton batched GEMM.

Algorithm
---------
1. Input LayerNorm (FP32 for stability).
2. Cast to FP16 and apply bias‑free linear projections (`left_proj`,
   `right_proj`).  Optionally mask the projections.
3. Compute three sigmoid gates from the normalized input and multiply
   them element‑wise with the projected tensors.
4. Reshape the gated tensors to (B·H, N, N).  The core O(N³) contraction
   Σₖ L_{i,k}·R_{j,k} is realised as a batched matrix multiplication
   L @ Rᵀ using a highly‑optimized Triton kernel (FP16 inputs,
   FP32 accumulation, Tensor‑core friendly tiling).
5. Reshape the result back to (B, N, N, H), apply the output LayerNorm,
   multiply by `out_gate` and project back to the original dimension.
6. Cast the final tensor to FP32 (required API type).

The Triton kernel fuses all batch instances into a single launch,
removing the per‑batch cuBLAS overhead of `torch.bmm`.  The kernel
operates on compile‑time known tile sizes (128×128×32) and uses
`tl.dot` with FP32 accumulation for maximal Tensor‑core utilization on
H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM (L @ Rᵀ) – all tensors are FP16, accumulation FP32.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr,                      # left matrix  (B, M, K)
    b_ptr,                      # right matrix (B, N, K) – will be read transposed
    c_ptr,                      # output matrix (B, M, N)
    BATCH, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)               # batch dimension
    pid_m = tl.program_id(1)                   # output row tile
    pid_n = tl.program_id(2)                   # output col tile

    # Tile coordinates within the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds masks for the output tile
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch
    c_base = c_ptr + pid_batch * stride_c_batch

    # Number of K‑tiles (known at compile time)
    num_k = tl.cdiv(K, BLOCK_K)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(num_k):
        # Global K indices covered by this tile
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)    # (BLOCK_K,)
        mask_k = k_offset < K

        # ---- Load a tile of the left matrix (M × K) -----------------
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_a_row
            + k_offset[None, :] * stride_a_col,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # shape (BLOCK_M, BLOCK_K)

        # ---- Load a tile of the transposed right matrix (K × N) -----
        # Original right layout is (B, N, K); we need element (k, n)
        b = tl.load(
            b_base
            + k_offset[:, None] * stride_b_col
            + offs_n[None, :] * stride_b_row,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # shape (BLOCK_K, BLOCK_N)

        # ---- Accumulate ------------------------------------------------
        acc += tl.dot(a, b)   # FP16×FP16 → FP32 accumulation

    # ---- Write back ----------------------------------------------------
    c = c_base + offs_m[:, None] * stride_c_row + offs_n[None, :] * stride_c_col
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c, acc.to(tl.float16), mask=mask)


def _batched_gemm_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication L @ Rᵀ using the Triton kernel above.
    Args:
        a, b: torch.cuda.FloatTensor of shape (B, N, N) and dtype float16.
               `b` is multiplied with its transpose.
    Returns:
        c: torch.cuda.FloatTensor of shape (B, N, N) and dtype float16.
    """
    assert a.is_cuda and b.is_cuda
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    assert a.shape == b.shape
    B, M, N = a.shape
    K = N  # square matrices ⇒ K = N

    # Ensure contiguous layout (required for our stride calculations)
    a = a.contiguous()
    b = b.contiguous()
    c = torch.empty_like(a)

    # Strides in element units
    stride_a_batch, stride_a_row, stride_a_col = a.stride()
    stride_b_batch, stride_b_row, stride_b_col = b.stride()
    stride_c_batch, stride_c_row, stride_c_col = c.stride()

    # Tile sizes (tuned for H100 Tensor‑cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B,                                          # batch dimension
        (M + BLOCK_M - 1) // BLOCK_M,               # rows
        (N + BLOCK_N - 1) // BLOCK_N,               # cols
    )

    _batched_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        BATCH=B,
        M=M,
        N=N,
        K=K,                         # constexpr for the kernel
        stride_a_batch=stride_a_batch,
        stride_a_row=stride_a_row,
        stride_a_col=stride_a_col,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_row=stride_c_row,
        stride_c_col=stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return c

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – fp32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for all heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # --------------------------------------------------------------
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # (B,N,N,hidden)
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcasted over hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 6) Gating (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core cubic contraction – batched GEMM via Triton
    # --------------------------------------------------------------
    # (B, N, N, hidden) -> (B, hidden, N, N) -> (B*hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_dim = B * hidden_dim
    left_mat = left_perm.view(batch_dim, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(batch_dim, N, N)

    # Batched L @ Rᵀ using the Triton kernel
    out_mat = _batched_gemm_triton(left_mat, right_mat)   # (B*H, N, N), fp16

    # --------------------------------------------------------------
    # 8) Reshape back to (B, N, N, hidden)
    # --------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 9) Output LayerNorm, out_gate, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    out = out * out_gate                                         # fp16

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))  # fp16 -> (B,N,N,dim)

    # --------------------------------------------------------------
    # 10) Cast back to FP32 for the public API
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
