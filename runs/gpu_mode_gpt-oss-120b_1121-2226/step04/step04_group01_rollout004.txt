"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The original PyTorch implementation performs five independent linear
projections of the input tensor `x` (two projections, two gates and the
output‑gate) and then a large N³ update implemented as an
einsum/loop.  The heavy memory traffic comes from reading `x` repeatedly
and from three separate element‑wise passes (mask, gates, final
output‑gate).

This implementation reduces the memory bandwidth by

1. **Fusing all five linear layers** into a single matrix‑multiply.
   `x` (shape B×N×N×D) is multiplied once by a weight matrix of size
   (5·H, D) where H = hidden_dim.  The result is sliced into
   `left`, `right`, `left_gate`, `right_gate` and `out_gate`.

2. **Fusing mask & gate multiplication** for `left` and `right` with a
   Triton kernel.  The kernel performs the three element‑wise products
   ``left * left_gate * mask`` and ``right * right_gate * mask`` in one
   pass, broadcasting the mask over the hidden dimension.

3. **Keeping the costly N³ update as a batched GEMM** (`torch.bmm`) –
   this already uses tensor‑cores on H100 and is the fastest way to
   compute ``∑_k left[...,k,:]·right[...,k,:]``.

All remaining operations (layer‑norms, sigmoid, final projection) are
performed with built‑in PyTorch kernels in FP16 (except the first
LayerNorm which stays in FP32 for numerical stability).  The final
output is returned as float32.

The kernel works for any batch size (1/2), sequence length up to 1024,
model dimensions 128/384 and hidden dimension 128, with or without a
mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 * left
    left_gate_ptr,       # fp16 * left_gate
    right_ptr,           # fp16 * right
    right_gate_ptr,      # fp16 * right_gate
    mask_ptr,            # fp16 * mask  (B*N*N)
    total_elems: tl.int32,    # B*N*N*H
    HEAD_DIM: tl.constexpr,   # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    """Fuse ``left * left_gate * mask`` and ``right * right_gate * mask``.

    The mask is broadcast over the hidden dimension, so the mask index
    for a given flat offset is ``offs // HEAD_DIM``.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask index (broadcast over H)
    mask_idx = offs // HEAD_DIM

    # Load values (zero for out‑of‑range elements)
    l_val   = tl.load(left_ptr + offs,       mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr + offs,      mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr + mask_idx,   mask=mask, other=tl.float16(1.0))

    # Fuse three multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr + offs,  l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/int/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for all heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse all five linear layers (left_proj, left_gate, right_proj,
    #    right_gate, out_gate) into a single matmul.
    # ------------------------------------------------------------------
    # Concatenate weights: shape (5*H, D)
    weight_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, D)

    # Combined projection: [B, N, N, 5*H]
    combined = F.linear(x_f16, weight_all)  # no bias

    # Slice the combined tensor (no copy, just views)
    H = hidden_dim
    left, left_gate_raw, right, right_gate_raw, out_gate_raw = torch.split(
        combined, H, dim=-1
    )  # each: [B, N, N, H]

    # --------------------------------------------------------------
    # 5) Sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)          # used later

    # --------------------------------------------------------------
    # 6) Apply mask & gates (Triton fusion)  – only when a mask is present
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask → fp16, contiguous
        mask_f16 = mask.to(torch.float16).contiguous()
        # Ensure contiguity of tensors that will be accessed in the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                # B * N * N * H  (fits in int32)
        BLOCK = 1024                        # max threads per block on H100
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=H,
            BLOCK_SIZE=BLOCK,
            num_warps=8,   # memory‑bound kernel – enough warps for latency hiding
        )
    else:
        # No mask: simple in‑place element‑wise multiplication
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core N³ update – batched GEMM (L @ Rᵀ) per hidden channel
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # Batched matrix multiplication (tensor‑core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N], fp16

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate  # element‑wise broadcast

    # --------------------------------------------------------------
    # 9) Final linear projection back to model dimension D
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))  # fp16
    out = out.to(torch.float32)  # required output dtype

    return out
