"""TriMul (outgoing) kernel optimized with Triton

- Input LayerNorm (float32)
- Cast to fp16.
- Fuse left/right projections and gate projections (concatenated weight).
- Apply gates (and optional mask) with a single Triton kernel.
- Compute core N³ update with batched GEMM (tensor cores).
- Output LayerNorm + out_gate.
- Final linear projection back to model dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    total_elems,
    HEAD_DIM: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # index of the mask element (broadcast over hidden dimension)
    mask_idx = offs // HEAD_DIM

    l_val   = tl.load(left_ptr      + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr     + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr+ offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr      + mask_idx, mask=mask, other=tl.float16(1.0))

    tl.store(left_ptr  + offs, l_val * l_gate * m_val, mask=mask)
    tl.store(right_ptr + offs, r_val * r_gate * m_val, mask=mask)

    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32 as required by the caller
    out = out.to(torch.float32)
    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Input LayerNorm (float32) – stable statistics.
2. Cast to FP16 for all heavy work.
3. Fuse the two projection linear layers *and* the two gate linear layers
   by concatenating the weight matrices (shape (2*H, D)).  One `F.linear`
   call per side produces a tensor `[B, N, N, 2*H]` which is then split
   into the projection and gate tensors.
4. Compute the output gate (`out_gate`) with a standard linear+sigmoid.
5. Apply the masks (if present) **and** the gates in a single Triton
   kernel.  The kernel reads
   `left, left_gate, right, right_gate, mask` and writes
   `left = left * left_gate * mask` and
   `right = right * right_gate * mask` in‑place, removing two
   extra memory passes.
6. Core N³ update: for each head `h`,
   `out[..., h] = left[..., h] @ right[..., h].T`.
   This is realised as a batched GEMM (`torch.bmm`) on the reshaped
   tensors `(B*H, N, N)`, which runs on tensor‑cores.
7. Output LayerNorm + element‑wise multiplication with `out_gate`.
8. Final linear projection back to the model dimension `D`.
9. Cast the result back to float32 for the caller.

All linear work after the first LayerNorm stays in FP16, which yields
large speed‑ups on an H100 while preserving enough numerical fidelity
for the test tolerances.
"""

import triton
import triton.language as tl

    total_elems,

        bias=weights["to_out_norm.bias"].to(torch.float16)
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear back to model dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Cast back to float32 as required by the caller
    return out.to(torch.float32)
```
