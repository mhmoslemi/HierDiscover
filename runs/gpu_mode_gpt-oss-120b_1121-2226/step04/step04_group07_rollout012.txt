"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1. Input LayerNorm (FP32) → cast to FP16.
2. **Fused linear step**: concatenate all projection‑ and gate‑weights
   (left_proj, left_gate, right_proj, right_gate, out_gate) into one
   weight matrix.  A single FP16 GEMM produces a tensor of shape
   [B, N, N, 5·H] where H = hidden_dim.
3. Slice the fused output, apply sigmoid to the three gate tensors,
   multiply gates with their corresponding projections and (optionally)
   mask the two side‑tensors.
4. Re‑order the side‑tensors to the layout required by the cubic update:
   (B, N, N, H) → (B, H, N, N) → (B·H, N, N).  This copy is the only
   non‑fused memory movement.
5. **Core cubic update**: for each hidden channel we need
   `out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}`.
   This is a batched matrix multiplication C = A @ Bᵀ with
   A, B ∈ [B·H, N, N].  The heavy work is performed by a custom
   Triton kernel (FP16 inputs, FP32 accumulation) using
   BLOCK_M×BLOCK_N×BLOCK_K = 128×128×64 tiles.
6. Reshape the result back to (B, N, N, H), apply the output LayerNorm,
   multiply by the out‑gate, and finally project back to the original
   model dimension with a bias‑free linear layer.
7. Return the result in FP32 as required by the API.

The implementation fuses three separate linear layers into one, removes
two passes over the large 4‑D tensor and therefore reduces memory traffic
while keeping the cubic update fully Triton‑accelerated.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul   C = A @ Bᵀ
#   A, B : fp16   (batch, M, K) and (batch, N, K)
#   C    : fp16   (batch, M, N) – result is cast from fp32 accumulator.
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,            # pointers
    M, N, K,                        # matrix sizes
    stride_ah, stride_am, stride_ak,  # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,  # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,  # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B tile (transposed view): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # (FP32) dot product via Tensor‑Core
        acc += tl.dot(a_tile, b_tile)

    # write back C (cast to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ using a Triton kernel.
    A, B : (batch, N, N)  FP16
    Returns C : (batch, N, N) FP16
    """
    assert A.is_cuda and B.is_cuda, "CUDA tensors required"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "FP16 tensors required"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in elements (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # tuned for H100
        num_stages=2,
    )
    return C

    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, dim] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys
                                "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]          # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numeric stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused linear projections + gates + out‑gate
    # --------------------------------------------------------------
    # Concatenate all projection / gate weights:
    # order = left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*H, dim)

    # Single GEMM: x_f16 @ combined_weightᵀ  →  [B, N, N, 5*H]
    combined = F.linear(x_f16, combined_weight)  # fp16

    # Slice the fused result
    H = hidden_dim
    left_proj = combined[..., :H]                     # (B,N,N,H)
    left_gate = combined[..., H:2 * H]
    right_proj = combined[..., 2 * H:3 * H]
    right_gate = combined[..., 3 * H:4 * H]
    out_gate_pre = combined[..., 4 * H:]            # (B,N,N,H)

    # Apply sigmoid gates
    left_gate_sig = torch.sigmoid(left_gate)
    right_gate_sig = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate_pre)

    # Element‑wise gating
    left = left_proj * left_gate_sig
    right = right_proj * right_gate_sig

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(dtype=torch.float16, device=device).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 6) Core N³ batched mat‑mul (outgoing update)
    # --------------------------------------------------------------
    # Rearrange to (B*H, N, N) – each hidden channel is an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched mat‑mul using the Triton kernel (C = A @ Bᵀ)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # (B*H, N, N)  fp16

    # Restore to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)

        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate

    # Final linear projection back to the original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)
    out = F.linear(out, to_out_w)   # fp16, shape (B,N,N,dim)

    # --------------------------------------------------------------
    # 8) Cast back to FP32 for API compatibility
    # --------------------------------------------------------------
    return out.to(torch.float32)
