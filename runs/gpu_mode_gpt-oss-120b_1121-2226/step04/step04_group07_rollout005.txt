"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32 → FP16)           : X ∈ [B, N, N, D]
2. Fuse each projection + its sigmoid gate:
       left  = (X·W_left)   * sigmoid(X·W_left_gate)
       right = (X·W_right)  * sigmoid(X·W_right_gate)
   The two linear maps are concatenated so that a single `F.linear`
   call produces both the projection and its gate.
3. (Optional) binary mask broadcast over the hidden dimension.
4. Core cubic update – a batched matrix multiplication
       out_{b,i,j,h}= Σ_k left_{b,i,k,h}·right_{b,j,k,h}
   This is a batched GEMM `A @ Bᵀ` with shape
       A,B ∈ [B·H, N, N]   (H = hidden_dim)
   Implemented with a custom Triton kernel that uses FP16 inputs,
   FP32 accumulation and Tensor‑Core MMA (BLOCK_M×BLOCK_N×BLOCK_K
   = 128×128×64, 8 warps, 2 stages).
5. Layer‑norm on the hidden dimension, multiply by the out‑gate,
   and a final bias‑free linear projection back to the original
   dimension.
6. Cast the result to FP32 (API requirement).

Only the heavy N³ mat‑mul is performed in Triton; all other
operations remain in high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul   C = A @ Bᵀ
#   A, B : fp16   (batch, M, K) and (batch, N, K)
#   C    : fp16   (batch, M, N) – result cast from fp32 accumulator.
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # base pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,   # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) Identify the tile we are responsible for
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointer for the current batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) Loop over the K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # ---- load A tile -------------------------------------------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ---- load B tile (transposed view) -------------------------------
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ---- inner‑product ------------------------------------------------
        acc += tl.dot(a_tile, b_tile)   # FP32 accumulation

    # ------------------------------------------------------------------
    # 3) Write back the result (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ using the Triton kernel above.
    A : (batch, M, K)   – fp16
    B : (batch, N, K)   – fp16
    Returns C : (batch, M, N) – fp16
    """
    assert A.is_cuda and B.is_cuda, "input tensors must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "fp16 required"

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps (256 threads) – good fit for H100
        num_stages=2,         # double‑buffering
    )
    return C


# ------------------------------------------------------------------
# Public entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32).
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (keep FP32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused projection + gate for left and right sides
    # --------------------------------------------------------------
    # left side
    left_proj_w = weights["left_proj.weight"]          # (H, D)
    left_gate_w = weights["left_gate.weight"]          # (H, D)
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2H, D)
    left_comb = F.linear(x_f16, left_comb_w)          # [B, N, N, 2H]
    left_proj = left_comb[..., :hidden_dim]           # [B, N, N, H]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # [B, N, N, H]

    # right side
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_comb = F.linear(x_f16, right_comb_w)        # [B, N, N, 2H]
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])

    # element‑wise gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcast over H)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 6) Core N³ update: batched mat‑mul (outgoing)
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # B, N, H
    # reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched mat‑mul using the Triton kernel defined above
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B·H, N, N]  (fp16)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output layer‑norm, out‑gate and final projection
    # --------------------------------------------------------------
    # out‑gate (linear + sigmoid)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B, N, N, H]

    # to_out_norm
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # apply out‑gate
    out = out * out_gate

    # final linear projection back to original dimension D
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)                         # fp16

    # --------------------------------------------------------------
    # 8) Cast to FP32 (API contract) and return
    # --------------------------------------------------------------
    return out.to(torch.float32)
