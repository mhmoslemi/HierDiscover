"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‐accelerated gating+mask.
Algorithm
---------
1. Layer‑norm the input (FP32) → x.
2. Cast to FP16 for the heavy work.
3. **Fused linear projections**:
   * Concatenate proj‑weight and gate‑weight for the left side
     → one matmul produces both left_proj and left_gate_raw.
   * Same for the right side.
   * Apply sigmoid to the gate halves.
4. Compute the out‑gate (sigmoid) with a separate linear.
5. Fuse mask (if present) and gate multiplication in a tiny Triton kernel:
        left  = left  * left_gate  * mask?
        right = right * right_gate * mask?
6. Core cubic update:  out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   implemented as a batched GEMM (B·H, N, N) @ (B·H, N, N)ᵀ via torch.bmm.
7. Output LayerNorm, out‑gate multiplication and final linear projection.
8. Cast back to FP32 (API requirement).

Only the mask+gate fusion is written in Triton; all dense linear algebra
uses cuBLAS (torch.nn.functional.linear / torch.bmm) which on H100
leverages Tensor‑cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_fused_kernel(
    left_ptr,          # fp16 * (total_elems)
    right_ptr,         # fp16 * (total_elems)
    left_gate_ptr,     # fp16 * (total_elems)
    right_gate_ptr,    # fp16 * (total_elems)
    mask_ptr,          # fp16 * (num_masks) – only used when HAS_MASK=1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * left_gate  * (mask   if present)
       right = right * right_gate * (mask   if present)
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the four tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Optional mask (broadcast over the hidden dimension)
    if HAS_MASK:
        # each hidden element shares the same mask value:
        # mask index = flat_index // hidden_dim
        mask_idx = offs // hidden_dim
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * m_val
        right = right * m_val

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Store back (in‑place)
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """
    Launch the Triton kernel that fuses mask (if any) and gate multiplication.
    All tensors are modified in‑place.
    """
    total = left.numel()
    BLOCK = 1024                       # max threads per block on modern GPUs
    grid = ( (total + BLOCK - 1) // BLOCK, )

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1)                     # flatten [B,N,N]
        HAS_MASK = 1

    _gate_mask_fused_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # ----------------------------------------------------------------------
    # 1) Unpack & basic shapes
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 2) Input LayerNorm (high‑precision FP32)
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ----------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ----------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ----------------------------------------------------------------------
    # 4) Fused linear projections (proj + gate) for LEFT side
    # ----------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]          # (D, H)
    left_gate_w  = weights["left_gate.weight"]          # (D, H)
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=1).to(torch.float16)
    left_concat = F.linear(x_f16, left_concat_w)       # [B, N, N, 2*H]

    left = left_concat[..., :hidden_dim]               # projected part, shape [B,N,N,H]
    left_gate_raw = left_concat[..., hidden_dim:]      # gate pre‑sigmoid, same shape
    left_gate = torch.sigmoid(left_gate_raw)           # sigmoid gate

    # ----------------------------------------------------------------------
    # 5) Fused linear projections (proj + gate) for RIGHT side
    # ----------------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=1).to(torch.float16)
    right_concat = F.linear(x_f16, right_concat_w)     # [B, N, N, 2*H]

    right = right_concat[..., :hidden_dim]
    right_gate_raw = right_concat[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_raw)

    # ----------------------------------------------------------------------
    # 6) Out‑gate (applied later after output norm)
    # ----------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B, N, N, H]

    # ----------------------------------------------------------------------
    # 7) Fuse mask (if any) and gate multiplication via Triton
    # ----------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left, right, left_gate, right_gate, mask_f16, hidden_dim
    )  # left / right are now multiplied by their gates (+ mask)

    # ----------------------------------------------------------------------
    # 8) Core N³ batched matrix multiplication (outgoing update)
    #    left[b,i,k,h] @ right[b,j,k,h]ᵀ  →  out[b,i,j,h]
    # ----------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) for batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch, M, K)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (batch, K, N)

    # batched matmul (FP16×FP16 → FP16, accumulated in FP32 internally)
    out_mat = torch.bmm(left_mat, right_mat_T)   # (B·H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ----------------------------------------------------------------------
    # 9) Output LayerNorm (FP16), out‑gate scaling and final projection
    # ----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    out = out * out_gate                     # apply out‑gate (FP16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)            # [B, N, N, D] FP16

    # ----------------------------------------------------------------------
    # 10) Cast back to FP32 for the API
    # ----------------------------------------------------------------------
    return out.to(torch.float32)
```
