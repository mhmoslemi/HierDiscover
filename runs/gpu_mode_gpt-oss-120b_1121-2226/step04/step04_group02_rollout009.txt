def _gate_mask_fuse_kernel(
    left_concat_ptr,       # fp16
    right_concat_ptr,      # fp16
    left_out_ptr,          # fp16
    right_out_ptr,         # fp16
    mask_ptr,              # fp16
    total_hidden: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_hidden

    # compute position index and hidden idx
    pos = offs // hidden_dim   # integer division by constexpr hidden_dim
    h = offs - pos * hidden_dim  # remainder
    # offset into concat tensors
    two_h = hidden_dim * 2
    left_base = pos * two_h
    right_base = pos * two_h

    left_proj_ptr = left_concat_ptr + left_base + h
    left_gate_ptr = left_concat_ptr + left_base + h + hidden_dim

    right_proj_ptr = right_concat_ptr + right_base + h
    right_gate_ptr = right_concat_ptr + right_base + h + hidden_dim

    left_proj = tl.load(left_proj_ptr, mask=mask, other=0.0)
    left_gate_raw = tl.load(left_gate_ptr, mask=mask, other=0.0)

@triton.jit
def _gate_mask_fuse_kernel(
    left_concat_ptr,   # *fp16
    right_concat_ptr,  # *fp16
    left_out_ptr,
    right_out_ptr,
    mask_ptr,          # *fp16 and maybe length total_positions or empty
    total_hidden: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_hidden

    # each thread works on a hidden element
    # compute position index (pos) and hidden channel index (h)
    # note: hidden_dim known at compile-time? it's a runtime argument but constant for each launch.
    pos = offs // hidden_dim    # integer division in 64-bit?
    h = offs - pos * hidden_dim

    # compute offsets for concatenated tensors
    two_h = hidden_dim * 2
    left_offset_proj = pos * two_h + h
    left_offset_gate = left_offset_proj + hidden_dim
    right_offset_proj = pos * two_h + h
    right_offset_gate = right_offset_proj + hidden_dim

    # load values
    left_proj = tl.load(left_concat_ptr + left_offset_proj, mask=mask, other=tl.float16(0.0))
    left_gate_raw = tl.load(left_concat_ptr + left_offset_gate, mask=mask, other=tl.float16(0.0))
    right_proj = tl.load(right_concat_ptr + right_offset_proj, mask=mask, other=tl.float16(0.0))
    right_gate_raw = tl.load(right_concat_ptr + right_offset_gate, mask=mask, other=tl.float16(0.0))

    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # mask multiplication if needed

        m = tl.load(mask_ptr + pos, mask=mask, other=tl.float16(1.0))
        left_proj = left_proj * m
        right_proj = right_proj * m

    right_out_val = right_proj * right_gate

    # store results
    tl.store(left_out_ptr + offs, left_out_val, mask=mask)
    tl.store(right_out_ptr + offs, right_out_val, mask=mask)

"""
TriMul “outgoing” update – Triton‑accelerated gating + cuBLAS batched GEMM.

Algorithm
---------
1. Layer‑norm the 4‑D input X ∈ [B, N, N, D] in FP32.
2. Cast X to FP16 for the heavy compute.
3. Fuse the two linear projections (proj & gate) for the left and right
   sides by concatenating the weight matrices → shapes (2·H, D).
   This yields tensors Lc, Rc ∈ [B, N, N, 2·H].
4. Launch a Triton kernel that, for every hidden element h at every
   (b,i,k) position, performs:
        left  = Lc[..., h]                * sigmoid(Lc[..., H+h]) * mask
        right = Rc[..., h]                * sigmoid(Rc[..., H+h]) * mask
   The kernel therefore fuses:
        * linear‑gate split,
        * sigmoid,
        * optional mask multiplication,
        * writing the final left/right tensors.
5. Compute the out‑gate via a separate bias‑free linear + sigmoid.
6. Core cubic operation (outgoing update):
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}
   which is a batched matrix multiply left @ rightᵀ.
   We reshape (B,N,N,H) → (B·H, N, N) and call torch.bmm that
   dispatches to cuBLAS Tensor‑cores on H100.
7. Apply the output LayerNorm, multiply by the out‑gate and a final
   linear projection back to the original dimension.
8. Cast the result to FP32 for the public API.

Only the gating‑+‑mask fusion is written in Triton (requirement).
All other compute‑intensive work uses highly‑optimized PyTorch/cuBLAS.
"""

def _gate_mask_fuse_kernel(
    left_concat_ptr,   # const fp16*  [B,N,N,2*H]   (flattened)
    right_concat_ptr,  # const fp16*  [B,N,N,2*H]
    left_out_ptr,      #       fp16*  [B,N,N,H]    (flattened)
    right_out_ptr,     #       fp16*  [B,N,N,H]
    mask_ptr,          # const fp16*  [B,N,N]      (flattened) – only used when HAS_MASK=1
    total_hidden: tl.int32,          # B·N·N·H  (total number of hidden elements)
    hidden_dim: tl.int32,            # H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place fusion:
       left  = left_proj * sigmoid(left_gate) * mask
       right = right_proj * sigmoid(right_gate) * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)          # element index in the flattened hidden tensor
    active = offs < total_hidden

    # ------------------------------------------------------------------
    # Compute position (pos) and hidden index (h) for each thread
    # pos ∈ [0, B·N·N)   – identifies the (b,i,k) slot
    # h   ∈ [0, H)      – hidden‑channel index
    # ------------------------------------------------------------------
    pos = offs // hidden_dim                     # integer division
    h = offs - pos * hidden_dim                  # remainder ( = offs % hidden_dim )

    # offsets inside the concatenated tensors (2*H per position)
    two_h = hidden_dim * 2
    base = pos * two_h

    left_proj_off  = base + h
    left_gate_off  = left_proj_off + hidden_dim
    right_proj_off = base + h
    right_gate_off = right_proj_off + hidden_dim

    # ------------------------------------------------------------------
    # Load values
    # ------------------------------------------------------------------
    left_proj = tl.load(left_concat_ptr + left_proj_off,  mask=active, other=tl.float16(0.0))
    left_gate_raw = tl.load(left_concat_ptr + left_gate_off, mask=active, other=tl.float16(0.0))
    right_proj = tl.load(right_concat_ptr + right_proj_off, mask=active, other=tl.float16(0.0))
    right_gate_raw = tl.load(right_concat_ptr + right_gate_off, mask=active, other=tl.float16(0.0))

    # ------------------------------------------------------------------
    # Sigmoid (stable formulation not required for fp16 on H100)
    # ------------------------------------------------------------------
    left_gate  = 1.0 / (1.0 + tl.exp(-left_gate_raw))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))

    # ------------------------------------------------------------------
    # Optional mask (broadcasted across hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # mask is stored per position (no hidden dim)
        m = tl.load(mask_ptr + pos, mask=active, other=tl.float16(1.0))
        left_proj  = left_proj * m
        right_proj = right_proj * m

    # ------------------------------------------------------------------
    # Apply gates
    # ------------------------------------------------------------------
    left_out  = left_proj  * left_gate
    right_out = right_proj * right_gate

    # ------------------------------------------------------------------
    # Write results back
    # ------------------------------------------------------------------
    tl.store(left_out_ptr  + offs, left_out,  mask=active)
    tl.store(right_out_ptr + offs, right_out, mask=active)


def _fuse_gates_and_mask(
    left_concat: torch.Tensor,
    right_concat: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
):
    """
    Launch the Triton kernel that fuses:
        * split of concatenated projection/gate tensors,
        * sigmoid activation,
        * optional mask multiplication,
    producing the final left and right tensors of shape [B,N,N,H] (fp16).
    """
    B, N, _, two_H = left_concat.shape
    assert two_H == 2 * hidden_dim, "Unexpected concatenated hidden dimension"

    total_positions = B * N * N
    total_hidden = total_positions * hidden_dim          # number of output elements

    BLOCK = 1024
    grid = ((total_hidden + BLOCK - 1) // BLOCK, )

    # allocate output tensors
    left_out  = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left_concat.device)
    right_out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=right_concat.device)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left_concat.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1).contiguous().to(torch.float16)
        HAS_MASK = 1

    # launch kernel
    _gate_mask_fuse_kernel[grid](
        left_concat,
        right_concat,
        left_out,
        right_out,
        dummy_mask,
        total_hidden,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
        num_stages=2,
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys "dim", "hidden_dim",
                               optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the module output.
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature size D
    hidden_dim = config["hidden_dim"]  # projection size H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 – high‑precision)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Concatenated linear projections (proj + gate) for left & right
    # --------------------------------------------------------------
    # left side: [left_proj; left_gate] → (2·H, D)
    left_concat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    left_concat = F.linear(x_f16, left_concat_weight)  # [B, N, N, 2·H]

    # right side: [right_proj; right_gate] → (2·H, D)
    right_concat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_concat = F.linear(x_f16, right_concat_weight)  # [B, N, N, 2·H]

    # --------------------------------------------------------------
    # 5) Fuse mask (if any) and sigmoid gates via Triton
    # --------------------------------------------------------------
    mask_f16 = None if nomask or mask is None else mask.to(torch.float16).contiguous()
    left, right = _fuse_gates_and_mask(left_concat, right_concat, mask_f16, hidden_dim)

    # --------------------------------------------------------------
    # 6) Out‑gate (separate linear + sigmoid)
    # --------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_weight))  # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 7) Core cubic operation – batched GEMM (outgoing update)
    # --------------------------------------------------------------
    B, N, _, H = left.shape                       # B, seq_len, _, hidden_dim
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H, N, N)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat_T)    # (B·H, N, N)   fp16, fp32 accumulation

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 8) Output LayerNorm (FP16), out‑gate multiplication, final projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # fp16

    out = out * out_gate                           # apply out‑gate (fp16)

    to_out_weight = weights["to_out.weight"].to(torch.float16)  # (D, H)
    out = F.linear(out, to_out_weight)           # [B, N, N, D] fp16

    # --------------------------------------------------------------
    # 9) Cast back to FP32 for the public API
    # --------------------------------------------------------------
    return out.to(torch.float32)
