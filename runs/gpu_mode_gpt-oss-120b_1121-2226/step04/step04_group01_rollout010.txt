Implementation details for kernel:

```
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    total_elems:int,
    H:int,  # hidden_dim
    NO_MASK: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask index across B,N,N
    mask_idx = offs // H  # int division
    # load
    l_val = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    l_gate_raw = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_gate_raw = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    mask_val = tl.float16(1.0)
    if not NO_MASK:
        mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
    # sigmoid gates
    l_gate = 1.0 / (1.0 + tl.exp(-l_gate_raw))
    r_gate = 1.0 / (1.0 + tl.exp(-r_gate_raw))

    l_out = l_val * l_gate * mask_val
    r_out = r_val * r_gate * mask_val

    tl.store(left_ptr + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)

"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
=======================================================

Algorithm
---------
1. Input LayerNorm (FP32, stable).
2. Cast to FP16 for all heavy work.
3. Five bias‑free linear projections in FP16:
   * left  = x @ W_left
   * right = x @ W_right
   * l_gate_raw  = x @ W_left_gate   (pre‑sigmoid)
   * r_gate_raw  = x @ W_right_gate  (pre‑sigmoid)
   * out_gate_raw = x @ W_out_gate   (pre‑sigmoid)
4. Fuse **sigmoid**, **mask** (if present) and **gate** multiplication
   for ``left`` and ``right`` in a single Triton kernel.  This removes
   the three separate ``torch.sigmoid`` kernels and the extra
   element‑wise kernels used in a naïve implementation.
5. Core N³ update – batched matrix multiplication
   ``out = (B·H, N, N)`` GEMM: ``left @ rightᵀ`` (FP16 tensor‑core).
6. Output LayerNorm (FP16) → sigmoid(out_gate_raw) → element‑wise gate.
7. Final linear projection back to model dimension D and cast to FP32.

The only Triton kernel needed is the gate‑mask fuse kernel; all
matmul / bmm work stays in cuBLAS/Tensor‑Core and all LayerNorm /
sigmoid steps stay in highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  left  = left  * sigmoid(l_gate_raw)  * mask
#                right = right * sigmoid(r_gate_raw)  * mask
# mask is stored once as [B, N, N] (fp16) and is broadcast over the
# hidden dimension H.  The kernel works for both masked and
# mask‑free cases (MASKED=0 → mask_val = 1.0).
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16 * left   (output in‑place)
    l_gate_ptr,        # fp16 * left_gate_raw (pre‑sigmoid)
    right_ptr,         # fp16 * right  (output in‑place)
    r_gate_ptr,        # fp16 * right_gate_raw (pre‑sigmoid)
    mask_ptr,          # fp16 * mask   (shape: B*N*N) – unused when MASKED=0
    total_elems: tl.int32,   # B*N*N*H
    H: tl.constexpr,         # hidden dimension
    MASKED: tl.constexpr,    # compile‑time flag
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Index into the mask (broadcast across H)
    mask_idx = offs // H

    # Load values
    l_val   = tl.load(left_ptr   + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(l_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr  + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(r_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Sigmoid = 1/(1+exp(-x))
    l_gate = 1.0 / (1.0 + tl.exp(-l_gate))
    r_gate = 1.0 / (1.0 + tl.exp(-r_gate))

    # Mask handling
    if MASKED:
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
    else:
        m_val = tl.float16(1.0)

    # Fuse three element‑wise operations
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Triton entry‑point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool / int) or None
        - weights_dict : mapping of weight names -> torch.Tensor (float32)
        - config_dict  : {"dim": D,
                          "hidden_dim": H,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32 – more stable)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Load / cast model weights to fp16
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 5) Five bias‑free linear projections (all fp16)
    #    – we keep them separate to avoid an extra massive copy.
    # ------------------------------------------------------------------
    left      = F.linear(x_f16, left_proj_w)       # [B, N, N, H]
    right     = F.linear(x_f16, right_proj_w)      # [B, N, N, H]

    l_gate_raw = F.linear(x_f16, left_gate_w)     # [B, N, N, H] (pre‑sigmoid)
    r_gate_raw = F.linear(x_f16, right_gate_w)    # [B, N, N, H] (pre‑sigmoid)

    out_gate_raw = F.linear(x_f16, out_gate_w)    # [B, N, N, H] (pre‑sigmoid)

    # ------------------------------------------------------------------
    # 6) Fuse sigmoid, mask and gate multiplication (single kernel)
    # ------------------------------------------------------------------
    # All tensors are already contiguous because each linear output is
    # produced in a contiguous layout.
    B, N, _, _ = left.shape

    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()   # [B, N, N]
        MASKED = 1
        mask_ptr = mask_f16.data_ptr()
    else:
        MASKED = 0
        # dummy pointer – never dereferenced when MASKED==0
        mask_ptr = left.data_ptr()

    total = left.numel()                     # B·N·N·H
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    _gate_mask_fuse_kernel[grid](
        left.data_ptr(),
        l_gate_raw.data_ptr(),
        right.data_ptr(),
        r_gate_raw.data_ptr(),
        mask_ptr,
        total,
        H=hidden_dim,
        MASKED=MASKED,
        BLOCK_SIZE=BLOCK,
        num_warps=8,
    )

    # l_gate_raw / r_gate_raw are no longer needed – free them early
    del l_gate_raw, r_gate_raw

    # ------------------------------------------------------------------
    # 7) Core N³ update – batched GEMM (FP16 tensor‑core)
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_mat : [B·H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm + out‑gate (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out_gate = torch.sigmoid(out_gate_raw)   # sigmoid on pre‑gate values
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to D and cast to FP32
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # FP16 result
    out = out.to(torch.float32)     # required output dtype

    return out
