def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # Load weights
    # Norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Combined weight for projections and gates
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # final linear
    to_out_weight = weights["to_out.weight"]

    # Input layernorm (float32)
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # Cast to fp16
    x_f16 = x.to(torch.float16)

    # Combine projection and gate weights into a 5*hidden_dim x dim matrix.
    # Ensure they are contiguous for efficient BLAS
    combined_weight = torch.cat([
        left_proj_w,
        left_gate_w,
        right_proj_w,
        right_gate_w,
        out_gate_w,
    ], dim=0).to(torch.float16)

    # Flatten input to (B*N*N, dim)
    B, N, _, _ = x_f16.shape
    x_flat = x_f16.reshape(-1, dim)  # (B*N*N, dim)

    # Linear (single GEMM)
    proj = torch.nn.functional.linear(x_flat, combined_weight)  # shape (B*N*N, 5*hidden_dim)
    proj = proj.view(B, N, N, hidden_dim * 5)

    # Split into parts
    left, left_gate, right, right_gate, out_gate = torch.split(proj, hidden_dim, dim=-1)

    # Apply sigmoid for gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply mask and gating for left/right (fusion via Triton)
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
        # Ensure tensors contiguous
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()  # B * N * N * hidden_dim
        BLOCK = 4096  # choose 4k threads per block to hide latency
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Input LayerNorm (float32) on the tensor `[B, N, N, D]`.
2. Cast to fp16 for the heavy work.
3. **Fuse all five linear layers** (left_proj, left_gate, right_proj,
   right_gate, out_gate) into a *single* GEMM:
   - weight matrix shape `(5*H, D)` where `H = hidden_dim`.
   - One matrix‑multiply produces the concatenated tensors
     `[left, left_gate, right, right_gate, out_gate]`.
4. Apply sigmoid to the three gate tensors.
5. **Mask‑gate fusion** – a tiny Triton kernel multiplies
   `left *= left_gate * mask` and `right *= right_gate * mask`
   (mask is absent → only gate multiplication).
6. Core N³ update: batched GEMM
   `out = left @ rightᵀ` for each hidden channel.
   Implemented with `torch.bmm` on the reshaped tensors
   (tensor‑core friendly fp16).
7. Output LayerNorm (fp16) → element‑wise product with `out_gate`.
8. Final linear projection back to the model dimension `D`.
9. Cast the result back to float32 for the caller.

The most expensive memory traffic (the five separate linear layers)
is reduced to a single GEMM, and the mask‑gate step is kept in a
custom Triton kernel.  All other operations rely on highly‑optimized
cuBLAS / PyTorch kernels.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# The mask is stored only once with shape [B, N, N] and is broadcasted
# across the hidden dimension inside the kernel.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 *
    left_gate_ptr,       # fp16 *
    right_ptr,           # fp16 *
    right_gate_ptr,      # fp16 *
    mask_ptr,            # fp16 *  (shape B*N*N)
    total_elems: tl.int32,        # B*N*N*H
    HEAD_DIM: tl.constexpr,       # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # each element belongs to a mask entry repeated HEAD_DIM times
    mask_idx = offs // HEAD_DIM

    l_val   = tl.load(left_ptr       + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr       + mask_idx, mask=mask, other=tl.float16(1.0))

    # three-way multiplication
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Entry point expected by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (float32 – more stable)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to fp16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fuse the five linear projections (including gates) into one GEMM
    # --------------------------------------------------------------
    # shape of each weight: (hidden_dim, dim)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, D)

    B, N, _, _ = x_f16.shape
    # flatten leading dimensions for a single GEMM
    x_flat = x_f16.view(-1, dim)            # (B*N*N, D)

    # Linear: (B*N*N, 5*H) = (B*N*N, D) @ (5*H, D)^T
    proj_flat = torch.nn.functional.linear(x_flat, combined_weight)  # fp16
    proj = proj_flat.view(B, N, N, hidden_dim * 5)                    # (B, N, N, 5*H)

    # --------------------------------------------------------------
    # 5) Slice into the individual tensors
    # --------------------------------------------------------------
    left, left_gate, right, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=-1
    )  # each: (B, N, N, H)

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 6) Optional mask & gate fusion (Triton)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B, N, N] -> fp16
        mask_f16 = mask.to(torch.float16).contiguous()

        # ensure contiguous storage for the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                     # B*N*N*H
        BLOCK = 4096                              # threads per block
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – just apply the gates in‑place
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core N³ update – batched GEMM  L @ Rᵀ  (fp16, tensor cores)
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 9) Final linear projection back to model dimension D
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(
        out,
        weights["to_out.weight"].to(torch.float16)   # (D, H)
    )   # fp16

    # --------------------------------------------------------------
    # 10) Cast back to float32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
