"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. **Input LayerNorm** – high‑precision (FP32) to keep numerical stability.
2. **Cast to FP16** – the heavy linear work is carried out in half precision.
3. **Linear projections** `left` and `right` into the hidden dimension `H`.
4. **Three sigmoid gates** (`left_gate`, `right_gate`, `out_gate`) are computed
   in FP32 (same as the reference) and cast to FP16.
5. **Mask + gate fusion** – a tiny 1‑D Triton kernel multiplies
   `left` ← `left * left_gate * mask` and `right` ← `right * right_gate * mask`.
   The kernel works in‑place on the FP16 tensors, avoiding extra kernels.
6. **Core cubic update** – for every batch `b` and hidden channel `h`

        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}

   which is a batched matrix‑multiply `left @ rightᵀ`.  We reshape the tensors
   to `(B·H, N, N)` and call `torch.bmm` (cuBLAS/Tensor‑core) – far faster than
   a hand‑written Triton GEMM.
7. **Output LayerNorm** (FP16), multiplication by `out_gate`, and a final
   linear projection back to the original dimension `D`.
8. **Cast back to FP32** for the API output.

Only step 5 is written in Triton (requirement).  All other compute‑intensive
operations reuse highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if present) and the two element‑wise gates.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr,           # FP16  [total_elems]
    right_ptr,          # FP16  [total_elems]
    left_gate_ptr,      # FP16  [total_elems]
    right_gate_ptr,     # FP16  [total_elems]
    mask_ptr,           # FP16  [num_masks] – flattened mask (B,N,N)
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load tensors
    left      = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden element shares the same mask entry:
        #   mask_idx = floor(idx / hidden_dim)
        mask_idx = offs // hidden_dim
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * m_val
        right = right * m_val

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # Store back in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """
    Launch the Triton kernel that fuses mask (if any) and the two gate
    multiplications.  Operates in‑place on ``left`` and ``right``.
    """
    total = left.numel()
    BLOCK = 4096                         # larger block reduces launch overhead
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        # flatten mask [B,N,N] → 1‑D; broadcast inside the kernel
        dummy_mask = mask.view(-1).contiguous()
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=8,          # H100 prefers 8 warps for FP16 workloads
    )


# ----------------------------------------------------------------------
# Public entry point – the Triton‑accelerated TriMul forward.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D  (input feature dimension)
    hidden_dim = config["hidden_dim"]  # H  (projection dimension)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy lifting
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  (fp16)
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H]  (fp16)

    # ------------------------------------------------------------------
    # 5) Gates (sigmoid) – compute in FP32 for numerical fidelity
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]   # still FP32
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate_fp32  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,H]  (fp32)
    right_gate_fp32 = torch.sigmoid(F.linear(x, right_gate_w))  # (fp32)
    out_gate_fp32   = torch.sigmoid(F.linear(x, out_gate_w))   # (fp32)

    # Cast gates to half for the Triton kernel / subsequent ops
    left_gate  = left_gate_fp32.to(torch.float16)
    right_gate = right_gate_fp32.to(torch.float16)
    out_gate   = out_gate_fp32.to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Fuse mask (if any) and the gates using a Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left,
        right,
        left_gate,
        right_gate,
        mask_f16,
        hidden_dim,
    )  # left/right are now gated and masked (in‑place)

    # ------------------------------------------------------------------
    # 7) Core N³ update – batched matrix multiplication (outgoing update)
    #    Each hidden channel forms its own (N×N) matrix; we compute
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B, N, N, H
    # reshape to (B·H, N, N) with left as (i,k) and rightᵀ as (k,j)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)          # (B·H, N, N)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)    # (B·H, N, N)

    # Batched GEMM – fp16×fp16 → fp16 (Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat_T)   # (B·H, N, N)  fp16

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H) fp16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (FP16), out‑gate multiplication, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    out = out * out_gate                # apply out‑gate (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)       # [B, N, N, D]  fp16

    # ------------------------------------------------------------------
    # 9) Cast back to FP32 for the API
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    return out
