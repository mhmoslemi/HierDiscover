def batched_matmul_kernel(
    L_ptr, R_ptr, Out_ptr,
    B, N, H,
    stride_L_batch, stride_L_i, stride_L_k, stride_L_h,
    stride_R_batch, stride_R_i, stride_R_k, stride_R_h,
    stride_O_batch, stride_O_i, stride_O_j, stride_O_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
  pid = tl.program_id(0)   # batch*hidden dimension
  pid_i = tl.program_id(1)
  pid_j = tl.program_id(2)

  # decode batch and hidden index
  hidden = H
  b = pid // hidden
  h = pid - b * hidden

  # calculate offsets for start of tile
  i = pid_i * BLOCK_M
  j = pid_j * BLOCK_N

  # compute base ptr offsets
  off_batch = b * stride_L_batch + h * stride_L_h
  # L ptr base
  L_ptrs = L_ptr + off_batch
  # R ptr base: base for batch is same: b * stride_R_batch + h * stride_R_h (same as left)
  off_batch_R = b * stride_R_batch + h * stride_R_h
  R_ptrs = R_ptr + off_batch_R
  # Output base offset
  off_out = b * stride_O_batch + h * stride_O_h
  Out_ptrs = Out_ptr + off_out

  # Create offset arrays for tiles
  offs_i = i + tl.arange(0, BLOCK_M)
  offs_j = j + tl.arange(0, BLOCK_N)
  mask_i = offs_i < N
  mask_j = offs_j < N

  acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

  for k in range(0, tl.cdiv(N, BLOCK_K)):
      k_offset = k * BLOCK_K
      offs_k = k_offset + tl.arange(0, BLOCK_K)
      mask_k = offs_k < N

      # Load A: shape (BLOCK_M, BLOCK_K)
      a_ptrs = L_ptrs + offs_i[:, None] * stride_L_i + offs_k[None, :] * stride_L_k
      a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

      # Load B transposed: we need shape (BLOCK_K, BLOCK_N)
      # B is accessed as B[b,j,k,h], but we want B_T[k,j] at row=k, col=j.

"""
Triton‑accelerated “outgoing” TriMul.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32 for numeric stability).
2. Cast to FP16 and apply two bias‑free linear projections
   (left_proj, right_proj).  The projections are followed by
   element‑wise sigmoid gates (left_gate, right_gate) and an optional
   binary mask.  The result of each projection is multiplied by its
   gate, producing tensors L and R of shape [B,N,N,H] (H = hidden_dim).
3. Compute the cubic contraction Σₖ L_{b,i,k,h}·R_{b,j,k,h} with a
   custom Triton kernel that performs a batched GEMM:
      out[b,i,j,h] = Σₖ L[b,i,k,h] * R[b,j,k,h]      (k‑reduction)
   The kernel works directly on the original memory layout
   (strided loads with stride_k = hidden_dim) and writes the result
   into the final tensor layout [B,N,N,H] – no extra transposes or
   copies are needed.
4. Apply the output LayerNorm, the out‑gate (sigmoid), and a final
   linear projection back to the original dimension.
5. Cast the result back to FP32 (the public API expects FP32).

The heavy O(B·H·N³) work is done in the fused Triton kernel, which
eliminates the expensive *permute‑reshape‑contiguous* steps used in
the reference implementation and therefore reduces overall runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the cubic contraction.
#   out = Σₖ L[...,i,k,h] * R[...,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    L_ptr,               # fp16 input tensor L  [B,N,N,H]
    R_ptr,               # fp16 input tensor R  [B,N,N,H]
    Out_ptr,             # fp16 output tensor   [B,N,N,H]
    B, N, H,
    # strides for L
    stride_L_batch, stride_L_i, stride_L_k, stride_L_h,
    # strides for R (same layout as L)
    stride_R_batch, stride_R_i, stride_R_k, stride_R_h,
    # strides for output (same layout as L)
    stride_O_batch, stride_O_i, stride_O_j, stride_O_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1)  Decode the combined batch*hidden index into (batch, hidden)
    # ------------------------------------------------------------------
    pid_bn = tl.program_id(0)                     # combined B*H index
    pid_i  = tl.program_id(1)                     # block row (i)
    pid_j  = tl.program_id(2)                     # block col (j)

    hidden = H
    b = pid_bn // hidden                           # batch index
    h = pid_bn - b * hidden                        # hidden index within the batch

    # ------------------------------------------------------------------
    # 2)  Compute the absolute offsets for the start of the tile
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M
    j = pid_j * BLOCK_N

    # base pointers that include batch and hidden offsets
    L_base = L_ptr + b * stride_L_batch + h * stride_L_h
    R_base = R_ptr + b * stride_R_batch + h * stride_R_h
    O_base = Out_ptr + b * stride_O_batch + h * stride_O_h

    # ------------------------------------------------------------------
    # 3)  Masks for the i and j dimensions (boundary handling)
    # ------------------------------------------------------------------
    offs_i = i + tl.arange(0, BLOCK_M)
    offs_j = j + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4)  Loop over the reduction dimension k in tiles
    # ------------------------------------------------------------------
    k = 0
    while k < N:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load L[i, k, h] ---------------------------------
        # shape (BLOCK_M, BLOCK_K)
        L_ptrs = (
            L_base
            + offs_i[:, None] * stride_L_i
            + offs_k[None, :] * stride_L_k
        )
        L = tl.load(
            L_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ----------- load R[j, k, h] (transposed on‑the‑fly) -------
        # we need R[k, j, h] => rows = k, cols = j
        # shape (BLOCK_K, BLOCK_N)
        R_ptrs = (
            R_base
            + offs_k[:, None] * stride_R_k   # row stride = stride over k
            + offs_j[None, :] * stride_R_i   # col stride = stride over i (original “j”)
        )
        R = tl.load(
            R_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # ----------- dot product over the K dimension ---------------
        acc += tl.dot(L, R)     # (BLOCK_M, BLOCK_N), fp32 accumulation

        k += BLOCK_K

    # ------------------------------------------------------------------
    # 5)  Write the result (cast back to fp16)
    # ------------------------------------------------------------------
    O_ptrs = (
        O_base
        + offs_i[:, None] * stride_O_i
        + offs_j[None, :] * stride_O_j
    )
    tl.store(
        O_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper to launch the Triton kernel
# ----------------------------------------------------------------------
def _batched_gemm_triton(L: torch.Tensor, R: torch.Tensor) -> torch.Tensor:
    """
    Compute the cubic contraction Σₖ L[...,i,k,h]·R[...,j,k,h] for tensors
    L,R ∈ ℝ^{B×N×N×H}  (fp16).  Returns a tensor of the same shape.
    """
    assert L.is_cuda and R.is_cuda
    B, N, N2, H = L.shape
    assert N == N2 and H == R.shape[3]

    # output tensor (same layout as inputs)
    out = torch.empty_like(L)

    # strides (in elements, not bytes)
    sL_b, sL_i, sL_k, sL_h = L.stride()
    sR_b, sR_i, sR_k, sR_h = R.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # tile sizes – tuned for H100 tensor‑core (FP16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                              # program id 0 : combined batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # program id 1 : i tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # program id 2 : j tiles
    )

    _tri_mul_outgoing_kernel[grid](
        L,
        R,
        out,
        B,
        N,
        H,
        sL_b,
        sL_i,
        sL_k,
        sL_h,
        sR_b,
        sR_i,
        sR_k,
        sR_h,
        sO_b,
        sO_i,
        sO_j,
        sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul “outgoing” update with a Triton‑accelerated
    cubic contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – fp32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy parts
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # --------------------------------------------------------------
    w_left = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    w_right = weights["right_proj.weight"].to(torch.float16)  # (H, D)

    left = F.linear(x_fp16, w_left)    # [B,N,N,H]
    right = F.linear(x_fp16, w_right)  # [B,N,N,H]

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)    # (B,N,N,1)
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 6) Gating (sigmoid) – FP16
    # --------------------------------------------------------------
    w_lgate = weights["left_gate.weight"].to(torch.float16)
    w_rgate = weights["right_gate.weight"].to(torch.float16)
    w_outgate = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_fp16, w_lgate))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_fp16, w_rgate))  # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x_fp16, w_outgate)) # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core cubic contraction – custom Triton kernel
    # --------------------------------------------------------------
    # The kernel directly writes the result into the final layout
    # (B, N, N, H) and therefore avoids the costly permute‑reshape‑copy
    # steps used in the reference implementation.
    out = _batched_gemm_triton(left, right)   # fp16, shape [B,N,N,H]

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )  # fp16

    out = out * out_gate                     # fp16

    w_to_out = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, w_to_out)            # fp16 → [B,N,N,D]

    # --------------------------------------------------------------
    # 9) Cast back to FP32 (public API expects float32)
    # --------------------------------------------------------------
    return out.to(torch.float32)
