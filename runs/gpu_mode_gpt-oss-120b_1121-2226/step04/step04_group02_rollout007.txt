    left_ptr, right_ptr, out_ptr,
    B, N, H,

    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,

    right_base = right_ptr + b*stride_right_batch + h*stride_right_h
    out_base = out_ptr + b*stride_out_batch + h*stride_out_h

    i0 = pid1 * BLOCK_M

    j_offset = j0 + tl.arange(0, BLOCK_N)

"""
TriMul “outgoing” forward (AlphaFold‑3 style) – Triton‑accelerated

Algorithm
---------
1.  Full‑precision LayerNorm on the 4‑D input    (B,N,N,D) → (B,N,N,D)  (FP32)
2.  Cast to FP16 for the heavy work.
3.  Two bias‑free linear projections producing L,R ∈ (B,N,N,H)  (FP16)
4.  Three sigmoid gates (left, right, out) from the same normalized input.
5.  Fuse mask (if present) and the two gates in a tiny Triton kernel:
       L ← L * sigmoid(left_gate) * mask
       R ← R * sigmoid(right_gate) * mask
6.  Core cubic update (outgoing):
       out_{b,i,j,h} = Σ_k L_{b,i,k,h} * R_{b,j,k,h}
   Implemented as a batched GEMM (L @ Rᵀ) in pure Triton – each (b,h) slice is a
   dense mat‑mul of size (N×N).  No intermediate permute/copy is required.
7.  Output LayerNorm (FP16), element‑wise out‑gate multiplication,
   and final linear projection back to D.
8.  Cast the result to FP32 for the API.

Only the gate+mask fusion **and** the cubic multiplication are written in Triton,
all other ops rely on highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Gate‑+‑mask fusion (tiny kernel)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr,       # fp16   [total_elems]
    right_ptr,      # fp16   [total_elems]
    left_gate_ptr,  # fp16   [total_elems]
    right_gate_ptr, # fp16   [total_elems]
    mask_ptr,       # fp16   [B*N*N]  (flattened) – only used when HAS_MASK=1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask_offs = offs < total_elems

    # load tensors
    l = tl.load(left_ptr + offs,   mask=mask_offs, other=0.0)
    r = tl.load(right_ptr + offs,  mask=mask_offs, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=mask_offs, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask_offs, other=0.0)

    # optional mask (broadcasted over hidden_dim)
    if HAS_MASK:
        # every hidden element shares the same mask entry
        mask_idx = offs // hidden_dim                # integer division
        m = tl.load(mask_ptr + mask_idx, mask=mask_offs, other=1.0)
        l = l * m
        r = r * m

    # apply gates
    l = l * lg
    r = r * rg

    # write back
    tl.store(left_ptr + offs,  l, mask=mask_offs)
    tl.store(right_ptr + offs, r, mask=mask_offs)


def _apply_gates_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    In‑place fusion of mask and left/right gates.
    left / right / left_gate / right_gate are fp16 tensors of shape (B,N,N,H).
    mask is either None or a fp16 tensor of shape (B,N,N) (already broadcast‑ready).
    """

    BLOCK = 1024                     # 1 K elements per launch – low overhead
    grid = ((total + BLOCK - 1) // BLOCK,)

# 2. Outgoing cubic update – batched GEMM in Triton
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr,           # fp16 *
    right_ptr,          # fp16 *
    out_ptr,            # fp16 *
    B: tl.int32,
    N: tl.int32,

    # left strides
    stride_left_batch: tl.int32,
    stride_left_i: tl.int32,
    stride_left_k: tl.int32,
    stride_left_h: tl.int32,
    # right strides
    stride_right_batch: tl.int32,
    stride_right_j: tl.int32,
    stride_right_k: tl.int32,
    stride_right_h: tl.int32,
    # out strides
    stride_out_batch: tl.int32,
    stride_out_i: tl.int32,
    stride_out_j: tl.int32,
    stride_out_h: tl.int32,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] """
    pid0 = tl.program_id(0)   # indexes (b * H + h)
    pid1 = tl.program_id(1)   # block row (i)
    pid2 = tl.program_id(2)   # block col (j)

    # decode batch / head
    b = pid0 // H
    h = pid0 % H

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # tile offsets
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    i_off = i0 + tl.arange(0, BLOCK_M)
    j_off = j0 + tl.arange(0, BLOCK_N)
    k_off = tl.arange(0, BLOCK_K)

    i_mask = i_off < N
    j_mask = j_off < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # main reduction over k
    for kk in range(0, N, BLOCK_K):
        cur_k = kk + k_off
        k_mask = cur_k < N

        # left tile  [M, K]
        left_ptrs = left_base + i_off[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # right tile transposed  [K, N]  (we load (k, j) directly)
        right_ptrs = right_base + cur_k[:, None] * stride_right_k + j_off[None, :] * stride_right_j
        right_tile = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # mma – fp16 × fp16 → fp32 accumulation
        acc += tl.dot(left_tile, right_tile)   # (M,N)
    # end for kk

    # write result (fp16)
    out_tile = acc.to(tl.float16)
    out_ptrs = out_base + i_off[:, None] * stride_out_i + j_off[None, :] * stride_out_j
    tl.store(out_ptrs, out_tile, mask=i_mask[:, None] & j_mask[None, :])


# ----------------------------------------------------------------------
# 3. Public entry point – fully differentiable forward (no grads needed)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args
    ----
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, D]          (float32)
        mask         : torch.Tensor [B, N, N] or None   (bool/int/float)
        weights      : dict of model parameters (float32)
        config       : dict with keys
                       - "dim"          : D
                       - "hidden_dim"  : H
                       - optional "nomask" (bool, default True)

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                         # [B, N, N, D] (float32)

    # --------------------------------------------------------------
    # 2) Cast to FP16
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (bias‑free)
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 4) Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # fp16 (used later)

    # --------------------------------------------------------------
    # 5) Fuse mask (if any) + left/right gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left,
        right,
        left_gate,
        right_gate,
        mask_f16,
        hidden_dim,
    )   # left/right are modified in‑place

    # --------------------------------------------------------------
    # 6) Core cubic multiplication (outgoing) – Triton batched GEMM
    # --------------------------------------------------------------
    B, N, _, H = left.shape                # B, N, N, H  (H == hidden_dim)
    # allocate output tensor
    out = torch.empty_like(left)           # [B,N,N,H] fp16, uninitialized

    # strides (all tensors are contiguous, shape [B,N,N,H])
    s_b = left.stride(0)    # B‑dim stride  = N*N*H
    s_i = left.stride(1)    # i‑dim stride  = N*H
    s_k = left.stride(2)    # k‑dim stride  = H
    s_h = left.stride(3)    # hidden stride = 1

    # grid configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        B * H,                                 # batch*head dimension
        (N + BLOCK_M - 1) // BLOCK_M,          # rows
        (N + BLOCK_N - 1) // BLOCK_N,          # cols
    )

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        # left strides
        s_b, s_i, s_k, s_h,
        # right strides (identical layout)
        s_b, s_i, s_k, s_h,
        # out strides
        s_b, s_i, s_k, s_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 likes 8‑warp kernels
    )

    # --------------------------------------------------------------
    # 7) Output LayerNorm (FP16), out‑gate & final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                 # fp16

    out = out * out_gate                # apply out‑gate (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)       # [B,N,N,D] fp16

    # --------------------------------------------------------------
    # 8) Cast back to FP32 (API requirement)
    # --------------------------------------------------------------
    return out.to(torch.float32)
