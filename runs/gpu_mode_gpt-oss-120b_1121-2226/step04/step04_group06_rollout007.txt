```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elem_mul_kernel(a_ptr, b_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(a_ptr + offs, a * b, mask=mask)

def _elemwise_mul_inplace(a: torch.Tensor, b: torch.Tensor):
    assert a.is_cuda and b.is_cuda
    assert a.shape == b.shape
    a_flat = a.view(-1)
    b_flat = b.view(-1)
    n = a_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    _elem_mul_kernel[grid](
        a_flat,
        b_flat,
        n,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return a

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul module.

    Algorithm
    ---------
    1) Layer‑norm the 4‑D input tensor in FP32.
    2) Cast to FP16 for the heavy compute.
    3) Fuse the three linear projections (left_proj, left_gate,
       right_proj, right_gate, out_gate) into a single GEMM: a
       single `torch.nn.functional.linear` with a weight matrix of shape
       (5·H, D).  Split the result, apply the binary mask (if any) and
       apply the sigmoid gates in place.
    4) Core cubic update Σₖ L_{i,k,h}·R_{j,k,h} is performed with a
       single `torch.einsum`, avoiding the explicit permutation and
       copy required for a batched `torch.bmm`.
    5) Apply the second LayerNorm (FP16) and multiply by the out‑gate
       using a tiny Triton kernel (in‑place element‑wise product).
    6) Final bias‑free linear projection back to the original dimension.
    7) Cast result back to FP32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] fp32

    # ------------------------------------------------------------------
    # 2) Cast to FP16
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fused projection + gate (left/right + out_gate)
    # ------------------------------------------------------------------
    # Concatenate all five weight matrices (proj/gate for left & right, out_gate)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, D)

    # Linear: result shape [B, N, N, 5*H]
    fused = F.linear(x_f16, fused_weight)  # no bias

    # Slice the fused tensor
    #   0:H                     -> left_proj
    #   H:2H                    -> left_gate
    #   2H:3H                   -> right_proj
    #   3H:4H                   -> right_gate
    #   4H:5H                   -> out_gate
    hp = hidden_dim
    left_proj = fused[..., :hp]
    left_gate = fused[..., hp:2*hp].sigmoid_()
    right_proj = fused[..., 2*hp:3*hp]
    right_gate = fused[..., 3*hp:4*hp].sigmoid_()
    out_gate = fused[..., 4*hp:5*hp].sigmoid_()

    # Apply mask before gating (mask broadcasted on the hidden dimension)
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left_proj.mul_(m)
        right_proj.mul_(m)

    # Apply sigmoid gates (in‑place)
    left_proj.mul_(left_gate)
    right_proj.mul_(right_gate)

    # ------------------------------------------------------------------
    # 4) Core cubic update via einsum (no extra copies)
    # ------------------------------------------------------------------
    # left_proj and right_proj now contain the gated tensors L and R
    out = torch.einsum('b i k h, b j k h -> b i j h', left_proj, right_proj)

    # ------------------------------------------------------------------
    # 5) Output LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 6) Multiply by out_gate (Triton kernel, in‑place)
    # ------------------------------------------------------------------
    out = _elemwise_mul_inplace(out, out_gate)

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 8) Cast back to FP32
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiply (used for the out‑gate)
# ----------------------------------------------------------------------
@triton.jit
def _elem_mul_kernel(
    a_ptr,                     # fp16 tensor, also the output
    b_ptr,                     # fp16 tensor
    n_elements,                # total number of elements
    BLOCK_SIZE: tl.constexpr,  # compile‑time block size
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    offs = offset + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask)      # fp16
    b = tl.load(b_ptr + offs, mask=mask)      # fp16
    tl.store(a_ptr + offs, a * b, mask=mask)  # in‑place write


def _elemwise_mul_inplace(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """In‑place a = a * b (both fp16, same shape)."""
    assert a.is_cuda and b.is_cuda
    assert a.shape == b.shape
    a_flat = a.view(-1)
    b_flat = b.view(-1)
    n = a_flat.numel()
    BLOCK = 1024
    grid = (triton.cdiv(n, BLOCK),)
    _elem_mul_kernel[grid](
        a_flat,
        b_flat,
        n,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return a

# Triton‑accelerated forward pass of the “outgoing” TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Algorithm
    ---------
    1) Layer‑norm the 4‑D input (FP32).
    2) Cast to FP16 for the heavy work.
    3) Fuse the three linear projections (left_proj, left_gate,
       right_proj, right_gate, out_gate) into a single GEMM: a single
       torch.nn.functional.linear with a weight matrix of shape (5·H, D).
       Split the result, apply the optional binary mask (if any) and the
       sigmoid gates in‑place.
    4) Core cubic update Σₖ L_{i,k,h}·R_{j,k,h} is performed with a single
       ``torch.einsum``; this avoids the explicit permutation/copy needed
       for a batched ``torch.bmm``.
    5) Apply the second LayerNorm (FP16) and multiply by the out‑gate
       using the Triton kernel above (in‑place).
    6) Final bias‑free linear projection back to the original dimension.
    7) Cast result back to FP32.
    """
    # ------------------------------------------------------------------
    # 0) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 – high precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] (float32)

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Fused projection + gate (left/right + out_gate)
    # ------------------------------------------------------------------
    # Concatenate all five weight matrices (proj & gate for left/right,
    # plus out‑gate).  Shape -> (5*H, D)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, D)

    # One huge linear: output shape [B, N, N, 5*H]
    fused = F.linear(x_f16, fused_weight)  # bias‑free

    hp = hidden_dim
    # Slices (views) of the fused tensor
    left_proj = fused[..., :hp]                               # (B,N,N,H)
    left_gate = fused[..., hp:2 * hp].sigmoid_()              # (B,N,N,H) in‑place sigmoid
    right_proj = fused[..., 2 * hp:3 * hp]                    # (B,N,N,H)
    right_gate = fused[..., 3 * hp:4 * hp].sigmoid_()         # (B,N,N,H) in‑place sigmoid
    out_gate = fused[..., 4 * hp:5 * hp].sigmoid_()          # (B,N,N,H) in‑place sigmoid

    # Apply sigmoid gates (in‑place multiplication)
    left_proj.mul_(left_gate)

    out = torch.einsum('b i k h, b j k h -> b i j h', left_proj, right_proj)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
