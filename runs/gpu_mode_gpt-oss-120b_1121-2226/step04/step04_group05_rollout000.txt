"""
TriMul “outgoing” forward – Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) → x.
2. Cast x → FP16 and apply two bias‑free linear projections
   (left_proj, right_proj) → L,R ∈ [B,N,N,H] (H = hidden_dim).
3. If a mask is present, broadcast‑multiply it onto L,R.
4. Compute three sigmoid gates from x and scale L,R and later the
   mat‑mul result.
5. Core cubic update:
        out_{b,i,j,h} = Σ_k L_{b,i,k,h} * R_{b,j,k,h}
   This is a batch of N×N matrix‑multiplications (L @ Rᵀ) for each
   hidden channel h.  We flatten the (B,H) dimensions → (B·H,N,N) and
   run a custom Triton kernel that tiles the work with
   (BLOCK_M×BLOCK_N×BLOCK_K) = (128×128×64) (or 64×64×64 for small N),
   uses FP16 inputs, FP32 accumulation (tensor‑core friendly) and
   writes the result back in FP16.
6. Reshape back to [B,N,N,H] (still FP16).
7. Output LayerNorm (FP16), out‑gate multiplication and a final
   bias‑free linear projection back to the original feature dimension.
   Cast to FP32 before returning.

All heavy work (step 5) runs on the GPU via Triton; the rest is
performed with PyTorch in mixed precision.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K)ᵀ → C (M×N)
#   A, B : fp16, shape (batch, M, K) and (batch, N, K)
#   C    : fp16, shape (batch, M, N)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K) strides
    stride_bh, stride_bn, stride_bk,   # B: (batch, N, K) strides
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Program IDs (grid: (M/blk_M, N/blk_N, batch))
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)   # row block index
    pid_n = tl.program_id(1)   # col block index
    pid_b = tl.program_id(2)   # flatten batch index (B*H)

    # --------------------------------------------------------------
    # Offsets for this tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ---- Load A tile: (BLOCK_M, BLOCK_K) ---------------------------------
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ---- Load B tile transposed: (BLOCK_K, BLOCK_N) ------------------------
        # B is accessed as Bᵀ, so we load (k, n) where k varies fastest.
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ---- FP32 accumulation -------------------------------------------------
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Write out C (cast back to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) as fp16.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    # Output buffer
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # Grid dimensions
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    # Choose a reasonable number of warps; >=8 for H100
    num_warps = 16 if BLOCK_M >= 128 else 8

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,          # pipelining stages
    )
    return C

# Triton‑accelerated TriMul forward (outgoing update)
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.
    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * dim (int)          – input feature dimension D
                * hidden_dim (int)   – projection dimension H
                * nomask (bool, optional)
    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy path
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Bias‑free linear projections (FP16 weights)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcasted)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 6) Gates (sigmoid) – also FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8) Core cubic update via batched matmul (outgoing)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, N, N, H

    # Reshape to (B·H, N, N) where each hidden channel is an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Choose block sizes heuristically; smaller N -> smaller tiles for better occupancy
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )   # [B·H, N, N]  fp16

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (FP16), out‑gate & final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)   # shape (dim, hidden_dim)
    out = F.linear(out, to_out_w)   # fp16, shape [B, N, N, D]

    # ------------------------------------------------------------------
    # 10) Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
```
