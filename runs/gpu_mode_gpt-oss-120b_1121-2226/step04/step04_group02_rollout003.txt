"""
TriMul “outgoing” forward pass (no gradients) – Triton‑accelerated
================================================================
Algorithm
---------
1. Layer‑norm the 4‑D input X∈[B,N,N,D] in FP32 (high‑precision statistics).
2. Cast to FP16 for the heavy part of the computation.
3. Fuse the two linear projections (proj + gate) on each side by concatenating
   weight matrices → one matmul per side.
4. Compute the three sigmoid gates (left, right, out) in FP16.
5. Fuse mask (if present) and the left/right gates in a tiny Triton kernel:
       left  = left  * sigmoid(left_gate)  * mask?
       right = right * sigmoid(right_gate) * mask?
6. Core cubic update:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched GEMM L @ Rᵀ for every hidden channel.
   We reshape to (B·H, N, N) and call `torch.bmm`, which on H100 uses
   Tensor‑cores (FP16×FP16→FP16 with reduced‑precision reduction).
7. Output LayerNorm (FP16), multiplication by the out‑gate, and final linear
   projection back to the original dimension.
8. Cast the result back to FP32 for the API.

Only step 5 is a custom Triton kernel – the rest uses highly‑optimized
PyTorch/cuBLAS primitives.  The implementation also:
* concatenates projection & gate weights → 2× fewer matmuls,
* enables reduced‑precision FP16 accumulation,
* uses a 1024‑element block with 8 warps for the fusion kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel that fuses mask (if any) and the two gates
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr,          # fp16 * (total_elems)
    right_ptr,         # fp16 * (total_elems)
    left_gate_ptr,     # fp16 * (total_elems)
    right_gate_ptr,    # fp16 * (total_elems)
    mask_ptr,          # fp16 * (total_masks)  (flattened) – only used when HAS_MASK==1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place:
       left  = left  * left_gate  * (mask if present)
       right = right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the four tensors
    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_g   = tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    # Optional mask (broadcasted over hidden dimension)
    if HAS_MASK:
        # Collapse hidden dimension: each H elements share the same mask value
        mask_idx = offs // hidden_dim
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * m_val
        right = right * m_val

    # Apply the gates
    left  = left  * left_g
    right = right * right_g

    # Store back
    tl.store(left_ptr + offs,  left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """Launch the Triton kernel that fuses mask and gate multiplication.
       Operates in‑place on `left` and `right`."""
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1)          # flatten [B,N,N] → 1‑D
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=8,               # more warps give higher occupancy on H100
    )


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys:
            * "dim"         – input feature dimension D
            * "hidden_dim"  – projection dimension H
            * optional "nomask" (True → ignore mask)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32).
    """
    # --------------------------------------------------------------
    # 0) Global settings for maximum speed on H100
    # --------------------------------------------------------------
    torch.backends.cuda.matmul.allow_fp16_reduced_precision_reduction = True
    torch.backends.cuda.matmul.allow_tf32 = False

    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (float32 → high‑precision statistics)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  float32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for all heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused linear projections (proj + gate) for left and right side
    #    Weight matrices are stored as (out_features, in_features)
    # --------------------------------------------------------------
    # ----- left side -----
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    left_cat_w   = torch.cat([left_proj_w, left_gate_w], dim=0)    # (2H, D)
    left_cat = F.linear(x_f16, left_cat_w)                         # (B,N,N,2H)
    left_proj_raw, left_gate_raw = left_cat.split(hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_raw)                       # (B,N,N,H)

    # ----- right side -----
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_cat_w  = torch.cat([right_proj_w, right_gate_w], dim=0)  # (2H, D)
    right_cat = F.linear(x_f16, right_cat_w)                       # (B,N,N,2H)
    right_proj_raw, right_gate_raw = right_cat.split(hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_raw)                     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Compute the out‑gate (used after the output LayerNorm)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 6) Fuse mask (if any) and the left/right gates with Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    # left_proj_raw/right_proj_raw are the "pre‑gate" tensors L,R
    _apply_gates_and_mask(
        left_proj_raw,
        right_proj_raw,
        left_gate,
        right_gate,
        mask_f16,
        hidden_dim,
    )   # left_proj_raw/right_proj_raw are modified in‑place → become L,R

    # ------------------------------------------------------------------
    # 7) Core N³ batched GEMM:  out = L @ Rᵀ   (B·H, N, N) batch
    # ------------------------------------------------------------------
    B, N, _, H = left_proj_raw.shape   # B, seq_len, _, hidden_dim

    # reshape to (B·H, N, N)   – left on the left side, right transposed
    left_mat = left_proj_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat_T = right_proj_raw.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (fp16×fp16 → fp16 with reduced‑precision reduction)
    out_mat = torch.bmm(left_mat, right_mat_T)   # (B·H, N, N), fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm (fp16), out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # Apply the out‑gate (already fp16)
    out = out * out_gate

    # Final linear back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)                         # [B,N,N,D] fp16

    # ------------------------------------------------------------------
    # 9) Cast to FP32 for the public API
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    return out
