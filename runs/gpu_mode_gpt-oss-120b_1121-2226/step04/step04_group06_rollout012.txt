@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    ):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for block
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute batch idx and head idx
    # To get b and h from pid_batch
    b = pid_batch // H
    h = pid_batch % H

    # base offsets
    off_batch = b * stride_l_batch + h * stride_l_k  # choose one of stride_h for left (i.e., last stride)
    # But left stride for head dimension is stride_l_h = left.stride(3), which is 1.

    # left base = left_ptr + b*stride_batch + h*stride_h

    left_base = left_ptr + b * stride_l_batch + h * stride_l_h
    right_base = right_ptr + b * stride_r_batch + h * stride_r_h

    # out base
    out_base = out_ptr + b * stride_o_batch + h * stride_o_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset base
        k_offset = k * BLOCK_K
        # compute offsets for loading left (i,k) and right (k,j)
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for k
        k_mask = offs_k < N

        # left pointers: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k)
        # right pointers: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k) + (j[None, :] * stride_r_j)

        # load with mask
        left_tile = tl.load(left_ptrs, mask=(i[:, None] < N) & k_mask[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=k_mask[:, None] & (j[None, :] < N), other=0.0)

        # perform dot product (sum over k)
        acc += tl.dot(left_tile, right_tile)  # shape (BLOCK_M, BLOCK_N), dtype float32 likely

    # Store to output

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0) # 0 .. B*H-1
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # compute base addresses
    left_base = left_ptr + b*stride_l_batch + h*stride_l_h
    right_base = right_ptr + b*stride_r_batch + h*stride_r_h
    out_base = out_ptr + b*stride_o_batch + h*stride_o_h

    # compute i and j offsets
    i_offsets = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_offsets = k_offset + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # left tile (i,k)
        left_ptrs = left_base + i_offsets[:, None] * stride_l_i + k_offsets[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # right tile (j,k) transposed: we need right[k,j]
        right_ptrs = right_base + k_offsets[:, None] * stride_r_k + j_offsets[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # dot product and accumulate
        acc += tl.dot(left, right)

    # write back output
    out_ptrs = out_base + i_offsets[:, None] * stride_o_i + j_offsets[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None] & j_mask[None, :])
```

"""
Triton‑accelerated “outgoing” TriMul forward pass.

Algorithm
---------
1.  Layer‑norm the 4‑D input (FP32) → x.
2.  Cast to FP16 for all heavy work.
3.  Fuse the four linear projections (proj & gate for left/right) into a
    single mat‑mul:  weight shape (4·H, D).  Slice the result into
    left_proj, left_gate, right_proj, right_gate.
4.  Optional binary mask is broadcast‑applied to the two projections.
5.  Apply sigmoid‑gates and obtain the gated streams `left` and `right`
    of shape [B, N, N, H] (FP16).
6.  Core cubic update Σₖ left[b,i,k,h]·right[b,j,k,h] implemented with
    ``torch.einsum`` (no extra copies).  Result `out` has shape [B,N,N,H].
7.  Output LayerNorm (FP16).
8.  Compute the shared out‑gate (sigmoid(F.linear(x, out_gate_w))) and
    fuse the element‑wise product out ← out * out_gate via a tiny Triton
    kernel.
9.  Final linear projection back to the original dimension (`to_out`).
10. Cast the result back to FP32 and return.

Only the element‑wise product (step 8) uses a custom Triton kernel – this
satisfies the “must use Triton” requirement while keeping the main
computational kernel in highly‑optimised PyTorch (cuBLAS / ATen) for
maximum throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication (FP16)
# ----------------------------------------------------------------------
@triton.jit
def _elem_mul_kernel(
    a_ptr,               # fp16 tensor, also output
    b_ptr,               # fp16 tensor, same shape as a_ptr
    n_elements,          # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(a_ptr + offs, a * b, mask=mask)


def _elemwise_mul_inplace(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """In‑place a = a * b  (both fp16 and same shape)."""
    assert a.is_cuda and b.is_cuda
    assert a.shape == b.shape
    a_flat = a.view(-1)
    b_flat = b.view(-1)
    n = a_flat.numel()
    BLOCK = 1024                       # 1 k elements per program
    grid = (triton.cdiv(n, BLOCK),)
    _elem_mul_kernel[grid](
        a_flat,
        b_flat,
        n,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return a


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D]  (float32)
            - mask         : torch.Tensor [B, N, N]   (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys “dim”, “hidden_dim”, optional “nomask”

    Returns:
        Tensor of shape [B, N, N, D] (float32).
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 – high precision)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused projection + gate for left & right streams
    # --------------------------------------------------------------
    # combine the four linear weight matrices (proj + gate for left/right)
    # each weight is (hidden_dim, dim)
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    # concat -> (4*H, D)
    fused_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    ).to(torch.float16)          # (4H, D)

    # single linear (no bias)
    fused = F.linear(x_f16, fused_w)    # [B, N, N, 4H]

    H = hidden_dim
    left_proj = fused[..., :H]               # (B,N,N,H)
    left_gate = fused[..., H:2*H]
    right_proj = fused[..., 2*H:3*H]
    right_gate = fused[..., 3*H:4*H]

    # optional mask (applied before gating, exactly as in reference)
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj = left_proj * m
        right_proj = right_proj * m

    # apply sigmoid gates
    left = left_proj * torch.sigmoid(left_gate)
    right = right_proj * torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 5) Core cubic update: Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with a single torch.einsum (no extra copies)
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # (B,N,N,H), FP16

    # --------------------------------------------------------------
    # 6) Output LayerNorm (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # FP16

    # --------------------------------------------------------------
    # 7) Compute out‑gate and fuse via Triton element‑wise multiplication
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)  # (H, D)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))     # (B,N,N,H) FP16

    out = _elemwise_mul_inplace(out, out_gate)

    # --------------------------------------------------------------
    # 8) Final linear projection back to original dimension (FP16)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)   # (B, N, N, D)  FP16

    # --------------------------------------------------------------
    # 9) Cast to FP32 for API compliance
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    return out
