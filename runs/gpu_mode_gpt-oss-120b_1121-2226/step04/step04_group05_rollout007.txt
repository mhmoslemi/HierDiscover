"""
TriMul “outgoing” forward pass with Triton‑accelerated batched matrix multiplication.

Algorithm
---------
1. Layer‑norm the input in FP32.
2. Cast to FP16 and apply two bias‑free linear projections (left / right).
3. Optional binary mask (broadcasted) is applied to the projections.
4. Three sigmoid gates are computed from the normalized input and fused
   element‑wise with the projected tensors.
5. Core O(N³) operation:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   -> interpreted as a batched matrix multiply left @ rightᵀ for each hidden
      channel.  The tensors are reshaped to (B·H, N, N) and a custom Triton
      kernel performs the multiply with FP16 loads, FP32 accumulation and
      stores the result in FP32.
6. Reshape back to [B,N,N,H], apply an output LayerNorm (FP32),
   multiply by the output gate, and a final bias‑free linear projection
   back to the original dimension.
7. Return the result in FP32.

All heavy computation (step 5) is performed in the Triton kernel, while
the remaining operations use mixed‑precision PyTorch for minimal overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
#   A : fp16, shape (batch, M, K)
#   B : fp16, shape (batch, N, K)  (the K‑dimension is the inner one)
#   C : fp32, shape (batch, M, N)  (accumulator is stored in FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K) strides
    stride_bh, stride_bn, stride_bk,   # B: (batch, N, K) strides
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Program identifiers
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (flattened B·H)

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tile‑wise)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K) – fp16
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B tile: (BLOCK_K, BLOCK_N) – fp16, loaded as transposed
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Dot product (FP32 accumulation)
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # Store result (FP32)
    # --------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B must be FP16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) as FP32.
    """
    assert A.is_cuda and B.is_cuda, "Inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Inputs must be FP16"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimension mismatch"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 SGEMM‑friendly
        num_stages=4,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward pass (“outgoing” update)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul “outgoing” update.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                              * dim (int)          – input feature dimension D
                              * hidden_dim (int)   – projection dimension H
                              * nomask (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack & config
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # --------------------------------------------------------------
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # --------------------------------------------------------------
    # 5) Optional binary mask
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)        # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 6) Gates (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)))

    # --------------------------------------------------------------
    # 7) Apply gates to projected tensors
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8) Core cubic operation (batched mat‑mul)
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) – each hidden channel is an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # Heuristic block sizes (tuned for H100)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B·H, N, N] (FP32)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP32

    # --------------------------------------------------------------
    # 9) Output LayerNorm (FP32), out‑gate multiplication,
    #    final projection back to original dimension (FP32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # FP32

    out = out * out_gate.to(torch.float32)   # FP32

    out = F.linear(out, weights["to_out.weight"])  # FP32, shape [B, N, N, D]

    # --------------------------------------------------------------
    # 10) Return FP32 result
    # --------------------------------------------------------------
    return out
