"""
TriMul “outgoing” kernel (AlphaFold‑3 style)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) → `x`.
2️⃣ Cast to FP16 for the heavy linear work.
3️⃣ Linear projections (left/right) and three bias‑free sigmoid gates
   (left_gate, right_gate, out_gate) are computed in FP16.
4️⃣ (Optional) pair‑wise mask is applied to `left` / `right` before gating.
5️⃣ `left` and `right` are permuted to (B, H, N, N) and flattened to
   (BH, N, N) where BH = B·hidden_dim.
6️⃣ A custom Triton kernel computes, for each head *h*:

        out[h, i, j] = Σₖ left[h, i, k] * right[h, j, k]

   i.e. a batched GEMM C = A @ Bᵀ with FP16 inputs, FP32 accumulation,
   and FP16 write‑back.
7️⃣ Reshape back to (B, N, N, H) (FP16) and apply a second LayerNorm over
   the hidden dimension.
8️⃣ Multiply by the output gate and project back to the original dimension
   with a bias‑free linear layer.
9️⃣ Cast the final result to float32 (API contract).

Only the N³ matrix multiplication is performed in Triton; the surrounding
operations use highly‑optimized torch kernels.  Block sizes (128×128×64)
are chosen to maximise H100 Tensor‑Core occupancy while keeping the FP32
accumulator (~16 KB) within register limits.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N] (fp16)
    b_ptr,               # right : [BH, N, N] (fp16) – interpreted as (K,N)
    c_ptr,               # out   : [BH, N, N] (fp16)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)            # batch * hidden
    pid_m = tl.program_id(axis=0)            # tile row
    pid_n = tl.program_id(axis=1)            # tile column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator (kept in registers)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- Load A tile (M × K) ------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ---- Load Bᵀ tile (K × N) -----------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ---- Accumulate ------------------------------------------------
        acc += tl.dot(a, b)   # dot is fp16, promotion to fp32 for acc

    # ------------------------------------------------------------------
    # Write back (fp16) – half the bandwidth of fp32 writes
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – stable accumulation
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Load weights once in fp16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (bias‑free) – fp16 → fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Optional pair‑wise mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; unsqueeze for broadcasting
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # fp16
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # Apply element‑wise gates in‑place
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Reshape for batched GEMM: (B, N, N, H) → (BH, N, N)
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    # Output buffer for the GEMM (fp16 result, fp32 accumulation inside)
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 8️⃣ Choose block sizes (constexpr for Triton) – adapt to small N
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    # Reduce block size for very small sequences to avoid over‑tiling
    if seq_len <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    if seq_len <= 128:
        BLOCK_M = BLOCK_N = 32
        BLOCK_K = 32

    # ------------------------------------------------------------------
    # 9️⃣ Launch Triton kernel (batched GEMM C = A @ Bᵀ)
    # ------------------------------------------------------------------
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                                   # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right – transposed on the fly)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10️⃣ Reshape back to [B, N, N, H] (still fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 11️⃣ Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # ------------------------------------------------------------------
    # 12️⃣ Apply output gate and final linear projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate                     # fp16 element‑wise
    out = F.linear(out, to_out_w)            # fp16 → dim

    # ------------------------------------------------------------------
    # 13️⃣ Cast back to float32 (API contract) and return
    # ------------------------------------------------------------------
    return out.float()
