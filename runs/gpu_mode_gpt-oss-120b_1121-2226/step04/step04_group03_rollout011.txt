"""
TriMul “outgoing” forward pass – optimized implementation.

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32) → x.
2.  Cast x to fp16 for the heavy linear work.
3.  Fuse the two projection layers and their sigmoid gates:
        left_proj + left_gate   → one matmul (2·hidden_dim outputs)
        right_proj + right_gate → one matmul (2·hidden_dim outputs)
   The gates are applied with a sigmoid, the masked elements are zeroed
   afterwards (the mask can be applied after the gating because multiplication
   is commutative).
4.  Compute the third gate (out_gate) with a separate matmul.
5.  Reshape tensors to (B·H, N, N) (contiguous) and launch a custom Triton
   kernel that for each head h computes

        out[h, i, j] = Σₖ left[h, i, k] * right[h, j, k]

   i.e. a batched GEMM C = A @ Bᵀ with fp16 inputs, fp32 accumulation and
   fp16 write‑back.
6.  Reshape the result back to [B, N, N, H] (fp16) and apply a second
   LayerNorm over the hidden dimension.
7.  Apply the output gate and the final bias‑free linear projection back to
   the original dimension D.
8.  Cast the result to float32 (API contract).

The kernel uses a 128×128×64 tiling (M, N, K) which fits cleanly into
registers on an H100.  We launch one program (warp) per (head, tile‑row,
tile‑col) and use `tl.dot` to let Tensor‑Cores perform the inner product.
`num_warps=8` and `num_stages=2` give good occupancy without exceeding
register budgets.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (A, B : fp16,  C : fp16, acc fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N]   (fp16)
    b_ptr,               # right : [BH, N, N]   (fp16) – interpreted as (K,N)
    c_ptr,               # out   : [BH, N, N]   (fp16)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)      # batch × hidden
    pid_m = tl.program_id(axis=0)      # tile‑row
    pid_n = tl.program_id(axis=1)      # tile‑col

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator (kept in registers)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- A tile (M × K) ------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ----- Bᵀ tile (K × N) ------------------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ----- Accumulate in FP32 -------------------------------------------
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write back (fp16)
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [bs, seq_len, seq_len, dim]
        - mask_tensor  : torch.Tensor, shape [bs, seq_len, seq_len]
        - weights_dict : dict of weight tensors (see code below)
        - config_dict  : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – stable accumulation
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()           # [B, N, N, D] – fp16
    # Flatten to (B*N*N, D) for the linear layers
    x_flat = x_fp16.view(-1, dim)  # [(B*N*N), D]

    # ------------------------------------------------------------------
    # 3️⃣ Fused projection + gate for left & right sides
    # ------------------------------------------------------------------
    # Concatenate the projection and gate weight matrices (out_dim = 2*hidden_dim)
    left_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # shape (2*H, D)
    right_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # shape (2*H, D)

    # One matmul each (no bias)
    left_cat = F.linear(x_flat, left_weight)   # [(B*N*N), 2*H]
    right_cat = F.linear(x_flat, right_weight) # [(B*N*N), 2*H]

    # Split projection and gate
    left_proj = left_cat[:, :hidden_dim]
    left_gate_raw = left_cat[:, hidden_dim:]
    right_proj = right_cat[:, :hidden_dim]
    right_gate_raw = right_cat[:, hidden_dim:]

    # Apply sigmoid to gates (still fp16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)

    # ------------------------------------------------------------------
    # 4️⃣ out_gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].half()
    out_gate_raw = F.linear(x_flat, out_gate_weight)  # [(B*N*N), H]
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5️⃣ Reshape back to 4‑D tensors
    # ------------------------------------------------------------------
    # shape: [B, N, N, H] – fp16
    left_proj = left_proj.view(bs, seq_len, seq_len, hidden_dim)
    left_gate = left_gate.view(bs, seq_len, seq_len, hidden_dim)
    right_proj = right_proj.view(bs, seq_len, seq_len, hidden_dim)
    right_gate = right_gate.view(bs, seq_len, seq_len, hidden_dim)
    out_gate = out_gate.view(bs, seq_len, seq_len, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Apply element‑wise mask (if present) and gating
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)  # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare data for the Triton GEMM (batch * hidden, N, N)
    # ------------------------------------------------------------------
    # Move hidden dimension next to batch and make contiguous storage
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    # Output buffer for the GEMM
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 8️⃣ Launch the Triton kernel (batched GEMM C = A @ Bᵀ)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # rows
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # cols
        BH,                                   # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right – note the transposed layout)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, H] (still fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 🔟 Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )  # fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Apply output gate and final linear projection back to dim D
    # ------------------------------------------------------------------
    out = out * out_gate                                 # gate – fp16
    out = F.linear(out, weights["to_out.weight"].half())  # fp16 → dim

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
