custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1.  Input LayerNorm (FP32 → FP16)                                   
2.  Fuse each projection with its gate (single linear + sigmoid)
    * left  = (X·W_left)   * sigmoid(X·W_left_gate)
    * right = (X·W_right)  * sigmoid(X·W_right_gate)
3.  (optional) binary mask broadcast over the hidden dimension
4.  Core cubic update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]                 
    This is a batched GEMM with A = left, B = rightᵀ.
    A custom Triton kernel computes C = A @ Bᵀ using FP16 inputs,
    FP32 accumulation and Tensor‑Core MMA.
    Block sizes are chosen adaptively (64 or 128) to hit the
    sweet‑spot of the H100.
5.  Output LayerNorm on the hidden dimension, multiplication by the
    out‑gate, and final bias‑free linear projection back to dim.
6.  Cast the result to FP32 for the public API.

Only the heavy N³ mat‑mul is performed in Triton; the surrounding
operations use fast PyTorch kernels.  All tensors stay on the GPU
throughout the run.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ
#   A : fp16   (batch, M, K)
#   B : fp16   (batch, N, K)
#   C : fp16   (batch, M, N)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,           # pointers
    M, N, K,                       # matrix sizes
    stride_ah, stride_am, stride_ak,  # strides for A  (batch, M, K)
    stride_bh, stride_bn, stride_bk,  # strides for B  (batch, N, K)
    stride_ch, stride_cm, stride_cn,  # strides for C  (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM with A @ Bᵀ.

    Each program instance computes a TILE of C of shape (BLOCK_M, BLOCK_N).
    """
    pid_b = tl.program_id(0)  # batch dimension (B·H)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block column

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # ---- Load A tile -------------------------------------------------
        k_off = k * BLOCK_K
        a_ptrs = a_batch + \
                 offs_m[:, None] * stride_am + \
                 (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)          # fp16

        # ---- Load B tile (transposed view) --------------------------------
        b_ptrs = b_batch + \
                 (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)          # fp16

        # ---- Compute block product ----------------------------------------
        acc += tl.dot(a, b)      # FP32 accumulation (Tensor‑Core)

    # ---- Write back -------------------------------------------------------
    c_ptrs = c_batch + \
             offs_m[:, None] * stride_cm + \
             offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    Arguments:
        A, B : (batch, M, K) and (batch, N, K), dtype=fp16, on CUDA.
    Returns:
        C : (batch, M, N), dtype=fp16.
    """
    assert A.is_cuda and B.is_cuda, "CUDA tensors required"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "fp16 expected"

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps works well on H100 for 128×128 tiles
        num_stages=3,       # double‑buffering of loads
    )
    return C


# ------------------------------------------------------------------
# Public entry point used by the evaluator
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys
                                "dim", "hidden_dim", optional "nomask"
    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    # keep the layer‑norm in FP32 for numerical stability
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fused projection + gating (bias‑free)
    # ------------------------------------------------------------------
    # left side
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                           # (2·H, D)
    left_combined = F.linear(x_f16, left_combined_w)   # [B, N, N, 2H]
    left_proj = left_combined[..., :hidden_dim]        # [B, N, N, H]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])  # [B, N, N, H]

    # right side
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)
    right_combined = F.linear(x_f16, right_combined_w)  # [B, N, N, 2H]
    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # Apply element‑wise gates
    left = left_proj * left_gate          # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Optional binary mask (broadcast over H)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 6) Core N³ update – batched mat‑mul A @ Bᵀ
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B·H, N, N) without extra copy – we view via `permute`
    # (B, N, N, H) -> (B, H, N, N) and then reinterpret the memory.
    # The permute does NOT make the tensor contiguous; we therefore
    # use `as_strided` to obtain a (B·H, N, N) view that respects the
    # original strides.
    # ------------------------------------------------------------------
    # Compute strides for the view (batch*H, N, N)
    # Original strides of left (contiguous) = (N*N*H, N*H, H, 1)
    s_b, s_i, s_j, s_h = left.stride()
    # New batch dimension combines B and H:
    #   offset = b * s_b + h * s_h
    # So we set stride for new batch = s_b (step to next b) and will
    # manually advance h inside the kernel using pid_b = b*H + h.
    # To keep it simple we flatten (b, h) into a single batch id and
    # provide the original batch & hidden strides to the kernel.
    # ------------------------------------------------------------------
    # Prepare flattened tensors (still a view; no copy)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # ------------------------------------------------------------------
    # Choose block sizes based on sequence length (empirical)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 128
        num_warps = 8

    out_flat = _batched_matmul_triton(
        left_flat,
        right_flat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B·H, N, N] (fp16)

    # Restore shape [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    # out‑gate (single linear + sigmoid) – based on the normalized input
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))          # [B, N, N, H]

    # LayerNorm over hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Apply out‑gate
    out = out * out_gate

    # Final linear projection back to original dim
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16

    # ------------------------------------------------------------------
    # 8) Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
