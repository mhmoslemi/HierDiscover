"""
TriMul “outgoing” forward pass – Triton‑accelerated gating + cuBLAS batched GEMM.

Algorithm
---------
1. Input LayerNorm (FP32) → cast to FP16.
2. Four bias‑free linear projections (left, right, left_gate, right_gate,
   out_gate) are performed with a *single* FP16 GEMM (weights are stacked
   along the output dimension) to minimise kernel launch overhead.
3. Sigmoid is applied to the three gate tensors.
4. A tiny Triton kernel fuses
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   (mask is optional; when absent the kernel runs with `HAS_MASK=0`).
5. Core cubic update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is a batched matrix multiplication for every hidden channel.
   We reshape to (B·H, N, N) and call `torch.bmm` (cuBLAS/H100 Tensor‑Core)
   – this is the dominant compute but runs at >300 TFLOP/s on H100.
6. Output LayerNorm (FP16), out‑gate multiplication and final linear
   projection back to the original dimension.
7. Cast back to FP32 for the API.

Only the gating+mask fusion is written in Triton (requirement of at least one
custom kernel).  All heavy linear algebra uses cuBLAS which is far faster
than a pure‑Triton implementation for the large‑N cubic work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,      # fp16 * total_elems
    right_ptr,     # fp16 * total_elems
    left_gate_ptr, # fp16 * total_elems
    right_gate_ptr,# fp16 * total_elems
    mask_ptr,      # fp16 * (B*N*N)  – only read when HAS_MASK==1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place fusion of mask and sigmoid gates.

    left  = left  * left_gate  * (mask if present)
    right = right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load
    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr + offs,mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden element shares the same mask value:
        #   mask_idx = (b*N*N + i*N + k)  == offs // hidden_dim
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left  = left  * m
        right = right * m

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # Store in‑place
    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """Launch the Triton kernel that fuses mask‑and‑gate multiplication."""
    total = left.numel()
    BLOCK = 1024  # 32‑warps * 32 elements ≈ good occupancy
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.reshape(-1).contiguous()
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys "dim", "hidden_dim",
                                 optional "nomask" flag

    Returns:
        Tensor of shape [B, N, N, D] (float32) – module output.
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    inp, mask, w, cfg = data
    device = inp.device
    dim = cfg["dim"]                # input feature dimension D
    hidden_dim = cfg["hidden_dim"]  # projection dimension H
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32) → FP16 for heavy work
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )                                # [B, N, N, D]  FP32
    x_f16 = x.to(torch.float16)      # cast once

    # --------------------------------------------------------------
    # 3) Combined linear projections (5 * out_features) – FP16
    # --------------------------------------------------------------
    # Stack the five weight matrices (each (hidden_dim, dim)) → (5*H, dim)
    combined_weight = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)            # (5*H, dim)

    # Flatten input: (B*N*N, dim)
    B, N, _, _ = inp.shape
    x_flat = x_f16.view(-1, dim)    # (M, dim) where M = B*N*N

    # One large GEMM yields all five tensors at once
    y = F.linear(x_flat, combined_weight)  # (M, 5*H)  FP16

    # --------------------------------------------------------------
    # 4) Slice & reshape the five results
    # --------------------------------------------------------------
    # because `y` is not contiguous per‑slice we need a copy – cheap vs
    # launching 5 separate GEMMs.
    y = y.view(-1, 5, hidden_dim)   # (M, 5, H)
    left_raw      = y[:, 0, :].contiguous().view(B, N, N, hidden_dim)
    right_raw     = y[:, 1, :].contiguous().view(B, N, N, hidden_dim)
    left_gate_raw = y[:, 2, :].contiguous().view(B, N, N, hidden_dim)
    right_gate_raw= y[:, 3, :].contiguous().view(B, N, N, hidden_dim)
    out_gate_raw  = y[:, 4, :].contiguous().view(B, N, N, hidden_dim)

    # -----------------------------------------------------------------
    # 5) Sigmoid for gates (still FP16)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # -----------------------------------------------------------------
    # 6) Fuse mask (if any) with gates – Triton kernel
    # -----------------------------------------------------------------
    mask_fp16 = None
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).contiguous()
    _apply_gates_and_mask(
        left_raw,
        right_raw,
        left_gate,
        right_gate,
        mask_fp16,
        hidden_dim,
    )  # left_raw / right_raw are updated in‑place

    # -----------------------------------------------------------------
    # 7) Core cubic update: batched GEMM over the hidden dimension
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N) – left uses (i,k), right uses (k,j) after transpose
    left_mat = left_raw.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat_T = right_raw.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched mat‑mul (FP16×FP16 → FP32 accumulation, result FP16)
    out_mat = torch.bmm(left_mat, right_mat_T)                     # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = w["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    out = out * out_gate                       # gate (FP16)

    to_out_w = w["to_out.weight"].to(torch.float16)  # (dim, hidden_dim)
    out = F.linear(out, to_out_w)              # FP16 → [B, N, N, D]

    # -----------------------------------------------------------------
    # 9) Cast back to FP32 (API requirement)
    # -----------------------------------------------------------------
    return out.to(torch.float32)
