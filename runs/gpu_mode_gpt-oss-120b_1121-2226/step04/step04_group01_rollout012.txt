    out_gate = proj[..., 4*hidden_dim:5*hidden_dim]
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)
    # Reorder and apply mask/gate using kernel
    total = B * N * N * hidden_dim
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )
    left_out = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=device)
    right_out = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=device)
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
        HAS_MASK = 1
        mask_ptr = mask_f16.data_ptr()
    else:
        HAS_MASK = 0
        mask_ptr = torch.empty((1,), dtype=torch.float16, device=device).data_ptr()
    # Launch kernel
    _tri_mul_prepare_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_ptr,
        left_out.data_ptr(),
        right_out.data_ptr(),
        B,
        N,
        hidden_dim,
        HAS_MASK=HAS_MASK,
        BLOCK_SIZE=BLOCK,
        num_warps=8,
    )
    # At this point left_out and right_out are contiguous (B*H,N,N)
    # Batched GEMM
    out_mat = torch.bmm(left_out, right_out.transpose(1, 2))
    # Reshape back to [B,N,N,H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Output layernorm + out_gate multiplication
    out = F.layer_norm(out, (hidden_dim,), weights["to_out_norm.weight"], weights["to_out_norm.bias"])
    out = out * out_gate
    # Final projection
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))
    # Cast back to float32
    return out.to(torch.float32)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Input LayerNorm (float32 → stable).
2. Cast to FP16 for the heavy work.
3. Fuse the five linear projections (left, right, left_gate,
   right_gate, out_gate) into a single matrix‑multiply:
        [B,N,N,D] → flatten → (B·N·N)×D
        weight_concat = concat([left_proj, right_proj,
                                 left_gate, right_gate, out_gate])
        → (B·N·N)×(5·H) result.
   Split the result and apply `sigmoid` to the three gate tensors.
4. Fuse mask multiplication, gate multiplication and **re‑ordering**
   of `left` and `right` into contiguous buffers of shape
   (B·H, N, N).  This is done by one Triton kernel
   (`_tri_mul_prepare_kernel`).  When no mask is supplied the kernel
   simply computes `left = left * left_gate` and `right = right *
   right_gate` while reshaping.
5. Core N³ update – batched GEMM:
        out_mat = torch.bmm(left_contig, right_contig.transpose(1,2))
   → shape (B·H, N, N).  Reshape back to (B, N, N, H).
6. Output LayerNorm (FP16) + output‑gate (`out_gate`).
7. Final linear projection back to the model dimension `D`.
8. Cast the result back to float32 for the caller.

Only kernel‑level work is the heavy element‑wise pass (step 4);
all matrix‑multiplications stay in FP16 and are handled by
cuBLAS/Tensor‑Cores.  The implementation works for
B∈{1,2}, N up to 1024, hidden_dim∈{128,384} and supports both
masked and un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_prepare_kernel(
    # input tensors (shape B,N,N,H, stored as B·N·N·H linear)
    left_in_ptr,        # fp16 *
    left_gate_ptr,      # fp16 *
    right_in_ptr,       # fp16 *
    right_gate_ptr,     # fp16 *
    # mask (shape B,N,N) – flat pointer, only accessed when HAS_MASK
    mask_ptr,           # fp16 *
    # output buffers (shape B*H,N,N, contiguous)
    left_out_ptr,       # fp16 *
    right_out_ptr,      # fp16 *
    # dimensions
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    # compile‑time constants
    HAS_MASK: tl.constexpr,     # 1 → load mask, 0 → ignore
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # decode linear index into (b,i,j,h)
    # original layout: ((b*N + i)*N + j) * H + h
    # ------------------------------------------------------------------
    tmp = offs
    h = tmp % H
    tmp = tmp // H
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N      # after removing i we are left with b

    # --------------------------------------------------------------
    # load values
    # --------------------------------------------------------------
    l_val   = tl.load(left_in_ptr   + offs,   mask=mask, other=tl.float16(0.0))
    lg_val  = tl.load(left_gate_ptr + offs,   mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_in_ptr  + offs,   mask=mask, other=tl.float16(0.0))
    rg_val  = tl.load(right_gate_ptr+ offs,   mask=mask, other=tl.float16(0.0))

    if HAS_MASK:
        # mask index does **not** contain the H dimension
        mask_idx = ((b * N + i) * N + j)
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))
    else:
        m_val = tl.float16(1.0)

    # --------------------------------------------------------------
    # apply gates and mask
    # --------------------------------------------------------------
    l_out = l_val * lg_val * m_val
    r_out = r_val * rg_val * m_val

    # --------------------------------------------------------------
    # write to contiguous (B*H, N, N) layout
    # dest_offset = ((b*H + h) * N + i) * N + j
    # --------------------------------------------------------------
    dest_offset = ((b * H + h) * N + i) * N + j
    tl.store(left_out_ptr  + dest_offset, l_out,  mask=mask)
    tl.store(right_out_ptr + dest_offset, r_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the *outgoing* TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 – stable)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 2) Cast to fp16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    B, N, _, _ = x_f16.shape

    # --------------------------------------------------------------
    # 3) Fuse the five linear projections (left, right, three gates)
    # --------------------------------------------------------------
    # concatenate weight matrices along the output‑feature dimension
    weight_concat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)                 # shape (5*H, D)

    # flatten [B,N,N,D] → [(B·N·N), D]
    x_flat = x_f16.view(-1, dim)               # (B·N·N, D)
    proj = F.linear(x_flat, weight_concat)     # (B·N·N, 5*H)

    # reshape back to 4‑D and split
    proj = proj.view(B, N, N, 5 * hidden_dim)

    left_proj   = proj[..., 0:hidden_dim]
    right_proj  = proj[..., hidden_dim:2 * hidden_dim]
    left_gate   = proj[..., 2 * hidden_dim:3 * hidden_dim]
    right_gate  = proj[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate    = proj[..., 4 * hidden_dim:5 * hidden_dim]

    # sigmoid for the three gate tensors
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 4) Mask‑gate‑reorder kernel (produces contiguous (B·H,N,N) tensors)
    # --------------------------------------------------------------
    total_elems = B * N * N * hidden_dim
    BLOCK = 1024                                   # max threads per block on H100
    grid = ((total_elems + BLOCK - 1) // BLOCK, )

    # destination buffers (contiguous (B*H, N, N))
    left_contig  = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=device)
    right_contig = torch.empty((B * hidden_dim, N, N), dtype=torch.float16, device=device)

    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
        has_mask = 1
        mask_ptr = mask_f16.data_ptr()
    else:
        # dummy mask – never accessed when HAS_MASK == 0
        dummy = torch.empty((1,), dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy.data_ptr()

    _tri_mul_prepare_kernel[grid](
        left_proj.data_ptr(),
        left_gate.data_ptr(),
        right_proj.data_ptr(),
        right_gate.data_ptr(),
        mask_ptr,
        left_contig.data_ptr(),
        right_contig.data_ptr(),
        B,
        N,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_SIZE=BLOCK,
        num_warps=8,                               # enough warps for latency hiding
    )

    # --------------------------------------------------------------
    # 5) Core N³ update – batched matrix multiplication
    # --------------------------------------------------------------
    # left_contig / right_contig : (B*H, N, N)
    out_mat = torch.bmm(left_contig, right_contig.transpose(1, 2))   # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 6) Output LayerNorm + output‑gate (still fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )
    out = out * out_gate                # element‑wise broadcast (B,N,N,H)

    # --------------------------------------------------------------
    # 7) Final linear projection back to model dimension D
    # --------------------------------------------------------------
    out = F.linear(out,
                   weights["to_out.weight"].to(torch.float16))    # (B,N,N,D) fp16

    # --------------------------------------------------------------
    # 8) Cast to float32 for the caller
    # --------------------------------------------------------------
    return out.to(torch.float32)
