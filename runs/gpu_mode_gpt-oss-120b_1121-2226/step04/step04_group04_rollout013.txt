    x = F.layer_norm(input_tensor, normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"]
    )   # float32

    # Cast to fp16 for main ops
    x_fp16 = x.to(torch.float16)

    # Fuse left projection + left gate
    left_weight_proj = weights["left_proj.weight"]
    left_weight_gate = weights["left_gate.weight"]
    left_weight_fused = torch.cat([left_weight_proj, left_weight_gate], dim=0)  # shape (2*H, D)
    left_comb = F.linear(x_fp16, left_weight_fused)  # shape (B,N,N,2H)

    left_proj = left_comb[..., :hidden_dim]
    left_gate_logit = left_comb[..., hidden_dim:]
    left_gate = torch.sigmoid(left_gate_logit)
    left = left_proj * left_gate

    # Fuse right projection + right gate
    right_weight_proj = weights["right_proj.weight"]
    right_weight_gate = weights["right_gate.weight"]
    right_weight_fused = torch.cat([right_weight_proj, right_weight_gate], dim=0)
    right_comb = F.linear(x_fp16, right_weight_fused)

    right_proj = right_comb[..., :hidden_dim]
    right_gate_logit = right_comb[..., hidden_dim:]
    right_gate = torch.sigmoid(right_gate_logit)
    right = right_proj * right_gate

    # Apply mask
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # broadcast
        left = left * mask_f16
        right = right * mask_f16

    # Compute out_gate
    out_gate_weight = weights["out_gate.weight"]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))

    # Core contraction: batched GEMM
    B = x.shape[0]
    N = x.shape[1]

    # reshape for batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Perform batched matmul: left @ right^T
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # still fp16

    # Output layernorm (with weight/bias)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final projection
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # fp16

    # Convert to float32 using Triton kernel
    out = _fp16_to_fp32_triton(out)

    return out

"""
TriMul “outgoing” update – fast forward pass.

The implementation follows the reference PyTorch module but reduces the
memory‑traffic overhead by fusing each projection (weight matrix) with its
corresponding gate (sigmoid).  The expensive cubic contraction
∑ₖ left[b,i,k,h]·right[b,j,k,h] is performed as a batched GEMM with
Tensor‑Core‑enabled FP16 arithmetic (`torch.matmul`).  All heavy work
remains in FP16, only the final result is converted back to FP32 with a
tiny Triton kernel (demonstrates the required Triton usage).

Key optimisations
-----------------
* **Fused projection+gate** – a single `F.linear` produces both the
  projection and the gating logits (`cat([proj,gate], dim=-1)`).  This
  halves the number of large matrix‑multiplications and reads the input
  tensor only once per branch.
* **In‑place element‑wise ops** – avoids temporary allocations.
* **No unnecessary copies** – the reshape‑permute sequence after the
  contraction is a view; we skip the costly `.contiguous()` call used in
  the baseline.
* **Triton conversion kernel** – replaces the trivial copy kernel from
  the reference and performs the final FP16→FP32 cast in a single GPU
  launch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: FP16 → FP32 copy (used for the final cast)
# ----------------------------------------------------------------------
@triton.jit
def _fp16_to_fp32_kernel(inp_ptr, out_ptr, N,
                         BLOCK_SIZE: tl.constexpr):
    """Copy `N` elements from fp16 `inp_ptr` to fp32 `out_ptr`."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N
    # Load as fp16, cast to fp32, store
    val = tl.load(inp_ptr + offs, mask=mask).to(tl.float32)
    tl.store(out_ptr + offs, val, mask=mask)


def _fp16_to_fp32_triton(x: torch.Tensor) -> torch.Tensor:
    """Convert a contiguous fp16 tensor to fp32 using Triton."""
    assert x.is_cuda and x.dtype == torch.float16
    out = torch.empty_like(x, dtype=torch.float32)
    N = x.numel()
    BLOCK = 1024
    grid = ((N + BLOCK - 1) // BLOCK,)
    _fp16_to_fp32_kernel[grid](
        x.view(-1),
        out.view(-1),
        N,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update (no gradient required).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numeric stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – FP32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy path
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused linear projection + gating for the left branch
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (H, D)
    left_gate_w = weights["left_gate.weight"]          # (H, D)
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2H, D)

    left_comb = F.linear(x_fp16, left_fused_w)          # (B, N, N, 2H)
    left_proj = left_comb[..., :hidden_dim]            # (B, N, N, H)
    left_gate_logit = left_comb[..., hidden_dim:]      # (B, N, N, H)
    left_gate = torch.sigmoid(left_gate_logit)         # (B, N, N, H)
    left = left_proj * left_gate                       # (B, N, N, H)

    # --------------------------------------------------------------
    # 5) Fused linear projection + gating for the right branch
    # --------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"]        # (H, D)
    right_gate_w = weights["right_gate.weight"]        # (H, D)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)  # (2H, D)

    right_comb = F.linear(x_fp16, right_fused_w)       # (B, N, N, 2H)
    right_proj = right_comb[..., :hidden_dim]          # (B, N, N, H)
    right_gate_logit = right_comb[..., hidden_dim:]    # (B, N, N, H)
    right_gate = torch.sigmoid(right_gate_logit)       # (B, N, N, H)
    right = right_proj * right_gate                    # (B, N, N, H)

    # --------------------------------------------------------------
    # 6) Optional binary mask (broadcasted over the hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 7) Gating for the outgoing path
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"]
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w.to(torch.float16)))  # (B, N, N, H)

    # --------------------------------------------------------------
    # 8) Core cubic contraction – batched GEMM (L @ Rᵀ)
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) for a single batched matmul
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # (B*H, N, N)

    # Batched matrix multiplication with Tensor‑Core FP16 (accumulate in FP32)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))      # (B*H, N, N)

    # Reshape back to (B, N, N, H) – view+permute are cheap, no copy
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)        # (B, N, N, H)

    # --------------------------------------------------------------
    # 9) Output LayerNorm (with learned affine parameters)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # (B, N, N, H) – FP16

    # --------------------------------------------------------------
    # 10) Apply the outgoing gate
    # --------------------------------------------------------------
    out = out * out_gate  # in‑place mul is fine (still FP16)

    # --------------------------------------------------------------
    # 11) Final projection back to the original dimension
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))   # (B, N, N, D) – FP16

    # --------------------------------------------------------------
    # 12) Convert result to FP32 using a tiny Triton kernel
    # --------------------------------------------------------------
    out = _fp16_to_fp32_triton(out)

    return out
