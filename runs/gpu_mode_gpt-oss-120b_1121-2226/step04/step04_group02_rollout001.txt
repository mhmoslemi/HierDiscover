"""
TriMul “outgoing” forward pass – highly‑optimized implementation.

Algorithm (FP32 → FP16):
1. Layer‑norm the 4‑D input (per‑channel) in FP32.
2. Cast to FP16 for all heavy work.
3. **Fused linear projections**:
   * Concatenate left/right projection weights → one matmul.
   * Concatenate left/right/out gate weights → one matmul.
   * Split the results → left_raw, right_raw, left_gate, right_gate, out_gate.
4. Apply sigmoid to gate tensors.
5. **Triton kernel** fuses elementwise mask (if any) and gate multiplication:
       left  = left_raw  * left_gate  * mask
       right = right_raw * right_gate * mask
6. Core cubic update:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   implemented as a batched matrix‑multiply L @ Rᵀ using torch.bmm.
7. Output LayerNorm (FP16), multiply by out_gate, final linear projection back
   to the original dimension.
8. Cast result to FP32 (API contract).

Only the mask+gate fusion is written in Triton (mandatory).  All other
compute‑intensive steps use fused torch operations and cuBLAS tensor‑cores,
yielding sub‑millisecond runtimes for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and gate multiplication.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fusion_kernel(
    left_ptr,          # fp16, [total_elems]
    right_ptr,         # fp16, [total_elems]
    left_gate_ptr,     # fp16, [total_elems] (already sigmoid)
    right_gate_ptr,    # fp16, [total_elems] (already sigmoid)
    mask_ptr,          # fp16, [mask_elems] (flattened), only used when HAS_MASK=1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place: left  *= left_gate  * (mask ?)
                right *= right_gate * (mask ?)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the four tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Optional mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        # each hidden element shares the same mask value:
        #   mask_idx = element_index // hidden_dim
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * m
        right = right * m

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Store back (in‑place)
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """Launch Triton kernel that fuses mask (if any) and gating."""
    total = left.numel()
    BLOCK = 1024  # max threads per block on H100
    grid = ( (total + BLOCK - 1) // BLOCK, )

    if mask is None:
        # dummy tensor – never read because HAS_MASK=0
        mask_flat = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        mask_flat = mask.view(-1).contiguous()
        HAS_MASK = 1

    _gate_mask_fusion_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_flat,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numeric stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] float32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy path
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused linear projections
    #    - Concatenate left/right projection weights → (2*H, D)
    #    - Concatenate left/right/out_gate weights → (3*H, D)
    # --------------------------------------------------------------
    # Linear projections (bias‑free)
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, D)
    proj = F.linear(x_f16, proj_weight)      # [B, N, N, 2*H]
    left  = proj[..., :hidden_dim]           # [B, N, N, H]
    right = proj[..., hidden_dim:]           # [B, N, N, H]

    # Gate linear (also bias‑free)
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (3*H, D)
    gates = F.linear(x_f16, gate_weight)    # [B, N, N, 3*H]

    left_gate  = torch.sigmoid(gates[..., :hidden_dim])                     # [B, N, N, H]
    right_gate = torch.sigmoid(gates[..., hidden_dim:2 * hidden_dim])      # [B, N, N, H]
    out_gate   = torch.sigmoid(gates[..., 2 * hidden_dim:])                # [B, N, N, H]

    # --------------------------------------------------------------
    # 5) Fuse mask (if any) and gates with a tiny Triton kernel
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(left, right, left_gate, right_gate, mask_f16, hidden_dim)

    # --------------------------------------------------------------
    # 6) Core cubic operation – batched matmul L @ Rᵀ
    # --------------------------------------------------------------
    B, N, _, H = left.shape   # B, seq_len, _, H   (H == hidden_dim)

    # reshape to (B*H, N, N) where the inner dimensions are (i,k) and (k,j)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)        # (B*H, N, N)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)   # (B*H, N, N)

    # Batched GEMM: fp16×fp16 → fp32 accumulation, result fp16
    out_mat = torch.bmm(left_mat, right_mat_T)  # (B*H, N, N), fp16

    # Back‑to original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate and final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Apply out‑gate (already sigmoid‑ed, fp16)
    out = out * out_gate

    # Final linear back to original dimension D
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)                         # [B, N, N, D] fp16

    # --------------------------------------------------------------
    # 8) Cast to FP32 for API compatibility
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    return out
