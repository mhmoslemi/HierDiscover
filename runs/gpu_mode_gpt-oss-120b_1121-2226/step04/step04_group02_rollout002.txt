```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for mask+gate fusion
@triton.jit
def _gate_mask_kernel(
    left_ptr,      # fp16*
    right_ptr,     # fp16*
    left_gate_ptr, # fp16*
    right_gate_ptr,# fp16*
    mask_ptr,      # fp16* (optional)
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # off // hidden_dim gives the position index, because hidden dim is the innermost dimension.
        mask_idx = offs // hidden_dim
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * m_val
        right = right * m_val

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Store
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

def _apply_gates_and_mask(left: torch.Tensor,
                          right: torch.Tensor,
                          left_gate: torch.Tensor,
                          right_gate: torch.Tensor,
                          mask: torch.Tensor | None,
                          hidden_dim: int):
    """Launch Triton kernel that fuses mask (if any) and gate multiplication.
    The tensors are modified in-place."""
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK, )

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1)
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
        # num_stages=2  # default
    )

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.
    The implementation fuses the four linear projections (left, right,
    left_gate, right_gate, out_gate) into a single GEMM, and fuses
    mask‑application with gate multiplication via a small Triton kernel.
    All heavy work runs in FP16 while layer‑norms keep FP32 precision.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1) Input LayerNorm (FP32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D], float32

    # 2) Cast to FP16 for subsequent math
    x_f16 = x.to(torch.float16)

    # 3) Fuse the five linear transforms (left, right, left_gate, right_gate, out_gate)
    #    All weight matrices are bias‑free.
    #    Shape: (out_features, in_features) => (hidden_dim, dim) for each.
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Concatenate weights along the output dimension (dim 0)
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(torch.float16, non_blocking=True)   # (5*hidden_dim, dim)

    # Perform a single GEMM: (B*N*N, dim) @ (dim, 5*hidden_dim).T
    B, N, _, _ = x_f16.shape
    M = B * N * N
    x_flat = x_f16.view(M, dim)                 # (M, dim)
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)  # (M, 5*hidden_dim)
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)   # (B, N, N, 5*hidden_dim)

    # Split fused output
    left, right, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        fused_out, hidden_dim, dim=-1
    )   # each (B,N,N,hidden_dim)

    # Apply sigmoid to gate tensors (still fp16)
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

"""
TriMul “outgoing” update – Triton‑accelerated implementation

Algorithm
---------
1. FP32 LayerNorm on the input tensor X ∈ [B,N,N,D].
2. Cast X → FP16 for all subsequent heavy math.
3. Fuse the five bias‑free linear projections
      left_proj, right_proj, left_gate, right_gate, out_gate
   into a *single* GEMM by concatenating their weight matrices.
   The result is split into the five tensors and the three gate
   tensors are passed through a sigmoid.
4. Fuse optional mask application with the two element‑wise gate
   multiplications via a tiny Triton kernel (`_gate_mask_kernel`).
5. Core cubic operation (outgoing update):
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   performed as a batched matrix‑multiply (B·H, N, N) @ (B·H, N, N)ᵀ
   using `torch.bmm` (FP16×FP16 → FP32 accumulation, result FP16).
6. FP16 LayerNorm on the hidden dimension, multiply by the out‑gate,
   and a final bias‑free linear projection back to the original dimension.
7. Cast the result to FP32 for the API.

Only the mask‑+‑gate fusion is written in Triton; all large matrix
operations use cuBLAS (via PyTorch) which on an H100 fully exploits
Tensor‑cores.  This reduces kernel launch overhead dramatically and
brings the runtime well below the 1 ms target for the prescribed
sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,          # fp16*   left  [total_elems]
    right_ptr,         # fp16*   right [total_elems]
    left_gate_ptr,     # fp16*   left_gate  [total_elems]
    right_gate_ptr,    # fp16*   right_gate [total_elems]
    mask_ptr,          # fp16*   mask (flattened) – used only if HAS_MASK==1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place fusion:
       left  ← left  * left_gate  * (mask if present)
       right ← right * right_gate * (mask if present)
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load operands
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # mask is broadcast over the hidden dimension → one mask value per (B,N,N)
        mask_idx = offs // hidden_dim
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * m_val
        right = right * m_val

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Write back
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """Launch the Triton kernel that fuses mask (if any) and gate multiplication.
    The tensors are modified in‑place."""
    total = left.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1)
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]      (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights       : dict of FP32 model parameters
        - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor  [B, N, N, D]  (float32)
        Forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # hidden dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32, high‑precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Weight tensors are stored as (out_features, in_features) and have bias=False
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(torch.float16)

    B, N, _, _ = x_f16.shape
    M = B * N * N
    # Flatten X to (M, dim) → GEMM → reshape back to (B,N,N,5*hidden_dim)
    x_flat = x_f16.view(M, dim)
    fused_out = torch.nn.functional.linear(x_flat, fused_weight)   # (M, 5*hidden_dim)
    fused_out = fused_out.view(B, N, N, 5 * hidden_dim)

    # ------------------------------------------------------------------
    # 5) Split and activate gates
    # ------------------------------------------------------------------
    left, right, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        fused_out, hidden_dim, dim=-1
    )
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 6) Fuse mask (if present) and gate multiplication via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left,
        right,
        left_gate,
        right_gate,
        mask_f16,
        hidden_dim,
    )   # left/right updated in‑place (FP16)

    # ------------------------------------------------------------------
    # 7) Core cubic update – batched matmul
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat_T)                     # (B*H, N, N), fp16
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate          # apply out‑gate (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,D), fp16

    # ------------------------------------------------------------------
    # 9) Cast back to FP32 for the public API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
