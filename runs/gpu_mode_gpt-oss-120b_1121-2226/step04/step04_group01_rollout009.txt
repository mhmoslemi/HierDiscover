import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    HEAD_DIM: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Compute mask index (broadcast across hidden dim)
    mask_idx = offs // HEAD_DIM

    # Load values (default 0 for out-of-range)
    l_val   = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)

def custom_kernel(data):
    """Triton‑accelerated outgoing TriMul forward pass.

    The implementation fuses the five linear layers of the PyTorch reference
    into a single GEMM, applies sigmoid gates, optionally merges a mask
    with the gating using a custom Triton kernel, performs the N³ update
    with a batched GEMM (left @ rightᵀ), and finishes with output
    layer‑norm, output‑gate and final projection back to ``dim``.
    All heavy computation uses FP16 while the two layer‑norms stay in FP32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------- 1) Input LayerNorm (float32) ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # [B, N, N, dim]  (float32)

    # ---------- 2) Cast to fp16 ----------
    x_f16 = x.to(torch.float16)

    # ---------- 3) Fuse the 5 linear layers ----------
    # Concatenate weight matrices (out_features, in_features)
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)  # shape (5*H, dim)

    B, N, _, _ = x_f16.shape
    BNN = B * N * N

    # Flatten input to (BNN, dim)
    x_flat = x_f16.view(BNN, dim)  # contiguous view
    # Perform fused GEMM
    proj = F.linear(x_flat, fused_weight)  # (BNN, 5*H)
    proj = proj.view(B, N, N, -1)  # (B, N, N, 5*H)

    # Split into components
    left, right, left_gate, right_gate, out_gate = torch.split(proj, hidden_dim, dim=-1)

    # Apply sigmoid to gate tensors (in‑place)
    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()

    # ---------- 4) Mask + gate fusion ----------
    if not nomask and mask is not None:
        # fp16 mask, shape [B, N, N]
        mask_f16 = mask.to(torch.float16).contiguous()
        # Ensure contiguous tensors for pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()  # B*N*N*H
        BLOCK = 1024
        grid = (total + BLOCK - 1) // BLOCK

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # Simple in‑place multiplication by gates
        left.mul_(left_gate)
        right.mul_(right_gate)

    # ---------- 5) Core N³ update (batched GEMM) ----------
    B, N, _, H = left.shape  # H == hidden_dim
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N), fp16
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ---------- 6) Output LayerNorm + output‑gate ----------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate  # broadcasted element‑wise

    # ---------- 7) Final linear projection back to dim ----------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # (B, N, N, dim), fp16

    # Return as float32 per spec
    return out.to(torch.float32)

"""
Triton‑accelerated outgoing TriMul forward pass.

Algorithm
---------
1. Input LayerNorm (FP32 for numerical stability).
2. Cast to FP16 for the bulk of the computation.
3. Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM:   X·W_fused  →  (B·N·N, 5·H).
   Split the result, apply sigmoid to the three gate tensors.
4. If a mask is present, a custom Triton kernel fuses:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   otherwise perform the two in‑place gate multiplications.
5. Core N³ update: batched GEMM   left @ rightᵀ   (FP16, Tensor‑core).
6. Output LayerNorm + output‑gate (still FP16).
7. Final projection back to the model dimension (FP16 → FP32 output).

All heavy ops stay in FP16; the two LayerNorms are kept in FP32.
Only the mask‑gate fusion is written in Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: mask & gate fusion.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# mask has shape [B, N, N] and is broadcasted over the hidden dim.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 *
    left_gate_ptr,     # fp16 *
    right_ptr,         # fp16 *
    right_gate_ptr,    # fp16 *
    mask_ptr,          # fp16 *
    total_elems: tl.int32,          # B·N·N·H
    HEAD_DIM: tl.constexpr,         # hidden dimension H
    BLOCK_SIZE: tl.constexpr,       # threads per block
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    # Index of the mask element (broadcast over H)
    mask_idx = offs // HEAD_DIM

    # Load operands; out‑of‑range positions get a neutral value.
    l_val   = tl.load(left_ptr       + offs, mask=valid, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr  + offs, mask=valid, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr      + offs, mask=valid, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=valid, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr       + mask_idx, mask=valid, other=tl.float16(1.0))

    # Fuse three multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=valid)
    tl.store(right_ptr + offs, r_out, mask=valid)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate          # broadcasted element‑wise product

    # ------------------------------------------------------------------
    # 8) Final linear projection back to model dimension D
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)          # (B, N, N, D), fp16

    # ------------------------------------------------------------------
    # 9) Cast to float32 for the caller
    # ------------------------------------------------------------------
    return out.to(torch.float32)
