   custom Triton kernel that tiles the work with (BLOCK_M×BLOCK_N×BLOCK_K)
   = (128×128×64) (or 64×64×64 for small N).  Inputs are FP16, accumulation
   is FP32 (Tensor‑core friendly), and the result is stored back in FP16.
6.  Reshape back to [B, N, N, H].
7.  Apply a second LayerNorm (hidden_dim), multiply by the out‑gate,
    and project back to the original dim with a bias‑free linear layer.
8.  Cast the final result to FP32 (as required by the API).

Only the heavy N³ mat‑mul is executed on Triton; all other ops use
high‑performance PyTorch kernels with mixed precision.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
#   A : (batch, M, K) fp16
#   B : (batch, N, K) fp16  (note: K dimension is the last axis, we
#                            load B as transposed to realize A @ Bᵀ)
#   C : (batch, M, N) fp16  (result cast from FP32 accumulator)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions
    stride_ah, stride_am, stride_ak,         # A: (batch, M, K) strides
    stride_bh, stride_bn, stride_bk,         # B: (batch, N, K) strides
    stride_ch, stride_cm, stride_cn,         # C: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (flattened B·H)

    # ------------------------------------------------------------------
    # Offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # (BLOCK_N,)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tile‑wise)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ----- Load A tile: (BLOCK_M, BLOCK_K) -----
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ----- Load B tile as transposed: (BLOCK_K, BLOCK_N) -----
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ----- Dot product (FP32 accumulation) -----
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store result (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B must be FP16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) as FP16.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, f"Inner dimensions must match (got {K} vs {Kb})"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    # Heuristic warps: more warps for larger tile sizes
    num_warps = 16 if (BLOCK_M * BLOCK_N) >= 8192 else 8
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                             * dim (int)          – input feature dimension D
                             * hidden_dim (int)   – projection dimension H
                             * nomask (bool, optional)

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy part
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)    # [B, N, N, H]  (fp16)
    right = F.linear(x_f16, right_proj_w) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool/int; convert to float16 (1.0 for True, 0.0 for False)
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 6) Gates (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w)) # same shape
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))     # same shape

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8) Core batched mat‑mul (outgoing update)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B = batch size

    # Reshape so that each (batch, hidden) pair becomes an independent batch:
    #   (B, N, N, H) → (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Choose tile sizes based on N (heuristic)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B·H, N, N] (fp16)

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (FP16), out‑gate multiplication,
    #    and final projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16, shape [B, N, N, D]

    # ------------------------------------------------------------------
    # 10) Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
