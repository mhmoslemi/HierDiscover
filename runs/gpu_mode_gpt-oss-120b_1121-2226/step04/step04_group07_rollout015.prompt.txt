```python
import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)
```

```python
"""
custom_kernel – Triton‑accelerated “outgoing” TriMul

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32 → FP16)   :  X ∈ [B, N, N, D]
2. Fuse the two linear projections with their sigmoid gates:
      left = (X @ W_left) * sigmoid(X @ W_left_gate)
      right= (X @ W_right)* sigmoid(X @ W_right_gate)
   The two weight matrices are concatenated so that each side needs only a single
   GEMM (reducing the number of passes over X from 4 to 2).
3. Apply the optional binary mask (broadcasted over the hidden dimension) and
   the element‑wise gates.
4. Core cubic update:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication A @ Bᵀ with
   A,B ∈ [B·H, N, N].  A custom Triton kernel performs the
   multiplication using FP16 inputs, FP32 accumulation and Tensor‑Core
   MMA (BLOCK_M×BLOCK_N×BLOCK_K = 128×128×64, 16 warps).
5. Output LayerNorm (FP16), multiplication by the out‑gate,
   final bias‑free linear projection back to the original dimension.
6. Cast the result to FP32 as required by the API.

Only the heavy N³ mat‑mul is performed in Triton; everything else
uses fast FP16 PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul   C = A @ Bᵀ
#   A, B : fp16   (batch, M, K) and (batch, N, K)
#   C    : fp16   (batch, M, N) – result is cast from fp32 accumulator.
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,    # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,    # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B tile (transposed view): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # (FP32) dot product via Tensor‑Core
        acc += tl.dot(a_tile, b_tile)

    # write back C (cast to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Perform batched matrix multiplication C = A @ Bᵀ with a Triton kernel.
    A : (batch, M, K)  FP16
    B : (batch, N, K)  FP16
    Returns C : (batch, M, N) FP16
    """
    assert A.is_cuda and B.is_cuda, "CUDA tensors required"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "FP16 tensors required"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides (in elements, not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps (256 threads) – good for H100
        num_stages=2,
    )
    return C


# ------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys
                                "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (keep FP32 for numeric stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fused linear projections + gates (bias‑free)
    # --------------------------------------------------------------
    # left side: combine projection and gate weights
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2H, D)
    left_combined = F.linear(x_f16, left_combined_w)  # [B, N, N, 2H]  fp16
    left_proj = left_combined[..., :hidden_dim]                     # [B,N,N,H]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])      # [B,N,N,H]

    # right side: combine projection and gate weights
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_combined = F.linear(x_f16, right_combined_w)  # [B, N, N, 2H] fp16
    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # apply element‑wise gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcast over H)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 6) Core N³ batched mat‑mul (outgoing update)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched mat‑mul using Triton (A @ Bᵀ)
    out_mat = _batched_matmul_triton(
        left_mat, right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B·H, N, N]  (fp16)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    # out‑gate (single linear + sigmoid)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # [B,N,N,H] fp16

    # to_out_norm (LayerNorm on the hidden dimension)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # apply out‑gate
    out = out * out_gate

    # final linear projection back to the original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16

    # Cast to FP32 for the API contract
    return out.to(torch.float32)
```
