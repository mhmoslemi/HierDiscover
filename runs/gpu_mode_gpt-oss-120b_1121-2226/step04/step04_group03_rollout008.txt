"""
TriMul “outgoing” operator – fast Triton implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) → x.
2️⃣ Cast to FP16 for the heavy linear work.
3️⃣ Fuse each projection and its sigmoid gate into a single
   bias‑free linear layer (weights are concatenated) → left, left_gate,
   right, right_gate (all FP16).  Apply the optional pair‑wise mask and
   the element‑wise gates in‑place.
4️⃣ Reshape tensors to (B·H, N, N) (contiguous) where H = hidden_dim.
5️⃣ Batched GEMM C = A @ Bᵀ with a custom Triton kernel.
   – A and B are FP16, accumulation is FP32, write‑back is FP16.
   – Tile size 64×64×64 fits H100 Tensor‑cores with low register pressure.
6️⃣ Reshape the result back to [B, N, N, H] (FP16).
7️⃣ Second Layer‑Norm over the hidden dimension (FP16).
8️⃣ Apply the output gate (sigmoid) and the final bias‑free linear
   projection back to the original dimensionality D (still FP16).
9️⃣ Cast to float32 for the API contract.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM  C = A @ Bᵀ
#   A : [BH, M, K]   (row‑major)
#   B : [BH, N, K]   (row‑major, interpreted as K×N after transpose)
#   C : [BH, M, N]   (row‑major)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left   [BH, M, K]   fp16
    b_ptr,               # right  [BH, N, K]   fp16 (will be read as K×N)
    c_ptr,               # out    [BH, M, N]   fp16
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, col, row)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)                 # batch × hidden
    pid_m = tl.program_id(axis=0)                 # tile row
    pid_n = tl.program_id(axis=1)                 # tile col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator (kept in registers)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M × K) ------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ----- Load Bᵀ tile (K × N) ------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ----- Accumulate ------------------------------------------------
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write result (fp16) – halves write bandwidth
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – stable accumulation
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gating linear layers
    #    left_proj + left_gate  → (hidden_dim, dim) + (hidden_dim, dim)
    #    right_proj + right_gate → same
    #    Build concatenated weight matrices (2·hidden_dim, dim)
    # ------------------------------------------------------------------
    left_cat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(device).half()  # (2*H, D)
    right_cat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(device).half()  # (2*H, D)

    # Linear pass (bias‑free)
    left_cat = F.linear(x_fp16, left_cat_weight)   # [B, N, N, 2*H]  fp16
    right_cat = F.linear(x_fp16, right_cat_weight) # [B, N, N, 2*H]  fp16

    # Split projection & gate
    left = left_cat[..., :hidden_dim]                # [B, N, N, H]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])   # [B, N, N, H]

    right = right_cat[..., :hidden_dim]              # [B, N, N, H]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:]) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Apply element‑wise sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Compute output gate (used after the second LayerNorm)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].to(device).half())
    )  # [B, N, N, H] – fp16

    # ------------------------------------------------------------------
    # 7️⃣ Reshape to [B·H, N, N] for the batched GEMM
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)  (contiguous)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, row‑major
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, row‑major

    # Output buffer for the GEMM (fp16 write‑back)
    out_flat = torch.empty(
        (BH, seq_len, seq_len), dtype=torch.float16, device=device
    )

    # ------------------------------------------------------------------
    # 8️⃣ Launch the Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,  # i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,  # j‑tiles
        BH,                                 # batch × hidden
    )

    tri_mul_outgoing_kernel[
        grid,
        num_warps=4,
    ](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                   # M
        seq_len,                   # N
        seq_len,                   # K
        # strides for A (left)
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for Bᵀ (right – interpreted as K×N)
        right_flat.stride(0),
        right_flat.stride(2),
        right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, H] (still fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)  # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 🔟 Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(device).half(),
        bias=weights["to_out_norm.bias"].to(device).half(),
    )  # fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Apply the output gate and final projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate                       # element‑wise fp16
    out = F.linear(
        out, weights["to_out.weight"].to(device).half()
    )                                          # fp16 → dim

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
