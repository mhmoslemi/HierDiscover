"""
Triton‑accelerated outgoing TriMul (AlphaFold3 “TriMul” block)

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32 → FP16) – keep FP32 for stability,
   then cast to FP16 for the heavy work.
2. Fuse each linear projection with its sigmoid gate:
      left  = (X·W_left ) * sigmoid(X·W_left_gate)
      right = (X·W_right) * sigmoid(X·W_right_gate)
   The two weight matrices are concatenated so that only one GEMM
   per side is required.
3. Apply the optional binary mask (broadcast over the hidden dimension)
   and the element‑wise gates.
4. Core cubic update (the “outgoing” term):
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication A @ Bᵀ with
   A,B ∈ [B·H, N, N].  The Triton kernel works directly on the
   original (B,N,N,H) layout – no permute/contiguous copies are
   performed.  The kernel uses FP16 inputs, FP32 accumulation and
   Tensor‑Core MMA (BLOCK_M×BLOCK_N×BLOCK_K = 128×128×64,
   8 warps, 3 pipeline stages).
5. Output LayerNorm (FP16), multiplication by the out‑gate,
   final bias‑free projection back to the original dimension.
6. Cast the result back to FP32 (API contract).

Only the N³ mat‑mul is done in Triton; all other ops use the
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul  C = A @ Bᵀ
#   A, B : fp16  (B·H, N, N) stored as (batch, i, k, h) / (batch, j, k, h)
#   C    : fp16  (B·H, N, N) stored as (batch, i, j, h)
# ------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # sizes
    stride_bl, stride_il, stride_kl, stride_hl,  # left strides (batch,i,k,h)
    stride_br, stride_ir, stride_kr, stride_hr,  # right strides (batch,j,k,h)
    stride_ob, stride_io, stride_jo, stride_ho,  # out strides (batch,i,j,h)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The first program dimension enumerates (batch * hidden) indices,
    the second/third dimensions enumerate tile rows and columns.
    """
    pid_bh = tl.program_id(0)      # combined batch * hidden index
    pid_m  = tl.program_id(1)      # tile row
    pid_n  = tl.program_id(2)      # tile column

    # unpack batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers that already include batch & hidden offsets
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_ob + h * stride_ho

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K

        # ----- left tile (M × K) -----
        a_ptrs = left_base + offs_i[:, None] * stride_il \
                           + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_kl
        mask_a = (offs_i[:, None] < N) & (k_off + tl.arange(0, BLOCK_K)[None, :] < N)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)          # fp16

        # ----- right tile (K × N) : we need the transposed view (k as rows, j as cols) -----
        b_ptrs = right_base + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_kr \
                            + offs_j[None, :] * stride_ir
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < N) & (offs_j[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)          # fp16

        # fused multiply‑add (Tensor‑Core)
        acc += tl.dot(a_tile, b_tile)

    # write result (cast back to fp16)
    c_ptrs = out_base + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    mask_c = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor,
                             BLOCK_M: int = 128,
                             BLOCK_N: int = 128,
                             BLOCK_K: int = 64) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.
    left, right : [B, N, N, H] (fp16, contiguous)
    Returns out : same shape as inputs (fp16)
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # stride information (in elements, not bytes)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_ob, stride_io, stride_jo, stride_ho = out.stride()

    # grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_hr,
        stride_ob,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )
    return out

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 2) Cast to FP16 for the heavy kernels
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3) Fused projection + gate for the “left” side
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"]      # (H, D)
    left_gate_w   = weights["left_gate.weight"]      # (H, D)
    left_combined = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)  # (2H, D)

    # Linear on the last dimension only (shape (B*N*N, 2H) after view)
    # torch.nn.functional.linear works on any leading dimensions.
    left_comb = F.linear(x_f16, left_combined)       # [B, N, N, 2H]  fp16
    left_proj = left_comb[..., :hidden_dim]          # [B, N, N, H]
    left_gate = torch.sigmoid(left_comb[..., hidden_dim:])  # [B, N, N, H]
    left = left_proj * left_gate                     # element‑wise gating

    # --------------------------------------------------------------
    # 4) Fused projection + gate for the “right” side
    # --------------------------------------------------------------
    right_proj_w   = weights["right_proj.weight"]
    right_gate_w   = weights["right_gate.weight"]
    right_combined = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)

    right_comb = F.linear(x_f16, right_combined)    # [B, N, N, 2H]
    right_proj = right_comb[..., :hidden_dim]
    right_gate = torch.sigmoid(right_comb[..., hidden_dim:])
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 6) Core outgoing cubic update (batched mat‑mul)
    # --------------------------------------------------------------
    # left/right are already in the layout expected by the kernel: [B, N, N, H]
    # No permute/contiguous copies are performed.
    out = _tri_mul_outgoing_triton(
        left,
        right,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )   # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 7) Output layer‑norm, out‑gate and final projection
    # --------------------------------------------------------------
    # out‑gate (linear + sigmoid) – using the same normalized x as the reference
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))      # [B, N, N, H] fp16

    # output layer‑norm on hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # apply out‑gate
    out = out * out_gate

    # final projection back to the original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16

    # --------------------------------------------------------------
    # 8) Cast back to FP32 (API contract)
    # --------------------------------------------------------------
    return out.to(torch.float32)
