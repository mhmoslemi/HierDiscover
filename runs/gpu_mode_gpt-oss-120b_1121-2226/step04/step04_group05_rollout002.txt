```python
"""
TriMul outgoing update - Triton accelerated matmul.

Algorithm:
1. Input layer‑norm in FP32.
2. Cast to FP16, apply bias‑free linear projections and sigmoid gates.
3. Apply optional binary mask.
4. Fuse element‑wise gating.
5. Core cubic operation: for each hidden channel compute
   out_{i,j} = Σ_k left_{i,k} * right_{j,k}
   This is a batched matrix‑multiply left @ rightᵀ.
   Implemented with Triton's highly‑tuned `triton.ops.matmul`
   (FP16 inputs, FP32 accumulation, Tensor‑core friendly).
6. Output layer‑norm, out‑gate multiplication and final linear projection.
7. Cast result back to FP32.

All heavy computation sits in the Triton kernel invoked via `triton.ops.matmul`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.ops as ops

def custom_kernel(data):
    """Forward pass of the TriMul “outgoing” module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys:
                * dim (int) – input feature dimension D
                * hidden_dim (int) – projection dimension H
                * nomask (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] FP32

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the bulk of the work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Bias‑free linear projections (FP16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)   # [B, N, N, H] FP16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 5) Optional binary mask (broadcasted)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 6) Sigmoid gates (FP16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8) Core cubic operation – batched matmul via Triton
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B·H, N, N) – each hidden channel becomes a separate batch
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, K)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, K)

    # Triton matmul: left @ rightᵀ
    # The op automatically uses FP16 inputs, FP32 accumulation and Tensor‑cores.
    out_mat = ops.matmul(left_mat, right_mat.transpose(-2, -1))   # (B*H, N, N) FP16

    # Restore the original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (FP16), out‑gate multiplication,
    #    and final projection back to the original dimensionality.
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    out = out * out_gate  # FP16

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # FP16, shape [B, N, N, D]

    # ------------------------------------------------------------------
    # 10) Cast back to FP32 for the public API
    # ------------------------------------------------------------------
    return out.to(torch.float32)

"""
TriMul “outgoing” update – Triton‑accelerated forward pass.

Algorithm
---------
1. Input LayerNorm (FP32) → x.
2. Cast x to FP16 and apply two bias‑free linear projections (left/right).
3. Optional binary mask (broadcasted) is applied.
4. Three sigmoid gates (left, right, out) are computed from x and multiplied
   element‑wise with the projected tensors.
5. Core cubic operation:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication left @ rightᵀ for each hidden
   channel.  We reshape the tensors to (B·H, N, N) and invoke Triton’s
   highly‑tuned `triton.ops.matmul` (FP16 inputs, FP32 accumulation,
   Tensor‑core friendly) to compute the result.
6. The result is reshaped back to [B, N, N, H], LayerNorm‑ed, multiplied by
   the out‑gate, and finally projected back to the original dimension.
7. The final tensor is cast to float32 (API contract).

All heavy computation (step 5) runs inside a Triton kernel; the rest is
performed with mixed‑precision PyTorch for minimal memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.ops as ops

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys
                * dim (int)          – input feature dimension D
                * hidden_dim (int)   – projection dimension H
                * nomask (bool, optional) – True if mask should be ignored

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (high‑precision FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the bulk of the work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)   # FP16 activations

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # ------------------------------------------------------------------
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, H]
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcasted)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 6) Gates (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_f16, weights["right_gate.weight"].to(torch.float16))) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)))     # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8) Core N³ batched mat‑mul (outgoing update) via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B·H, N, K)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # (B·H, N, K)

    # Batched matmul: left @ rightᵀ
    out_mat = ops.matmul(left_mat, right_mat.transpose(-2, -1))   # (B·H, N, N) FP16

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (FP16), out‑gate multiplication,
    #    and final projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # FP16

    out = out * out_gate   # FP16

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))   # FP16 → [B, N, N, D]

    # ------------------------------------------------------------------
    # 10) Cast back to FP32 for the API contract
    # ------------------------------------------------------------------
    return out.to(torch.float32)
