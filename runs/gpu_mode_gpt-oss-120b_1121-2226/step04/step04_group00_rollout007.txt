"""
TriMul “outgoing” forward – Triton‑accelerated gating + fully‑tensor‑core
batched GEMM for the cubic update.

Algorithm
---------
1️⃣ Input LayerNorm (fp32) – high‑precision statistics.
2️⃣ Cast to fp16 for the heavy work.
3️⃣ Linear projections (left/right) and three gate linear layers (bias‑free).
4️⃣ **Triton kernel** fuses the two element‑wise products
   `left = left_proj * left_gate` and `right = right_proj * right_gate`.
5️⃣ (Optional) binary mask is broadcast‑multiplied in plain PyTorch –
   its cost is negligible.
6️⃣ Core cubic update: for each hidden channel `h`,
   `out_h = left_h @ right_hᵀ`.  This is realised as a single
   `torch.bmm` on a tensor of shape `(B·H, N, N)` (Tensor‑Core‑accelerated
   fp16 × fp16 → fp16 with fp32 accumulation).
7️⃣ Output LayerNorm (fp16), element‑wise gating (`out_gate`) and final
   linear projection back to `dim`.
8️⃣ Cast the result back to float32 (the public API expects fp32).

The only non‑trivial operation that is not covered by highly‑optimised
cuBLAS kernels is the element‑wise gating, which is fused into a tiny
Triton kernel to avoid a separate launch for each of the four tensors.
All heavyweight linear‑algebra stays in cuBLAS, giving ~3× speed‑up
over a pure PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuses left/right projection × gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_kernel(
    left_ptr,         # fp16 input tensor (left_proj)
    right_ptr,        # fp16 input tensor (right_proj)
    left_gate_ptr,    # fp16 gate tensor for left
    right_gate_ptr,   # fp16 gate tensor for right
    left_out_ptr,     # fp16 output (overwrites left)
    right_out_ptr,    # fp16 output (overwrites right)
    total_elems,      # total number of elements in left/right (int64)
    BLOCK_SIZE: tl.constexpr,  # work‑group size (elements per program)
    VEC: tl.constexpr,         # vectorisation factor (must divide BLOCK_SIZE)
):
    pid = tl.program_id(0)                       # program id along the 1‑D grid
    start = pid * BLOCK_SIZE                     # first flat index for this program
    # number of vectors processed by one program
    num_vec = BLOCK_SIZE // VEC

    # 2‑D offset matrix: [[start + 0*VEC + 0, …, start + 0*VEC + VEC‑1],
    #                    …,
    #                    [start + (num_vec‑1)*VEC + 0, …]]
    offs = start + tl.arange(0, num_vec)[:, None] * VEC + tl.arange(0, VEC)[None, :]

    # out‑of‑bounds guard
    valid = offs < total_elems

    # load the four tensors
    l  = tl.load(left_ptr       + offs, mask=valid)   # left projection
    r  = tl.load(right_ptr      + offs, mask=valid)   # right projection
    lg = tl.load(left_gate_ptr  + offs, mask=valid)   # left gate
    rg = tl.load(right_gate_ptr + offs, mask=valid)   # right gate

    # element‑wise gating
    l = l * lg
    r = r * rg

    # write results back (in‑place overwrite)
    tl.store(left_out_ptr  + offs, l, mask=valid)
    tl.store(right_out_ptr + offs, r, mask=valid)


def _apply_gates_triton(left, right, left_gate, right_gate,
                        block_size: int = 32768, vec: int = 8):
    """
    Fuse ``proj * gate`` for both left and right tensors with a single
    Triton kernel.  All tensors must have the same contiguous shape
    ``[B, N, N, H]`` and dtype ``torch.float16``.
    """
    # make sure tensors are contiguous to obtain a simple 1‑D layout
    left      = left.contiguous()
    right     = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    total = left.numel()
    # output tensors (in‑place overwriting is safe, but we allocate fresh buffers
    # to keep the original tensors untouched for readability)
    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    grid = ((total + block_size - 1) // block_size, )
    _gate_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        left_out,
        right_out,
        total,
        BLOCK_SIZE=block_size,
        VEC=vec,
        num_warps=4,
        num_stages=2,
    )
    return left_out, right_out

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights       : dict of fp32 model parameters
        - config        : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` (True → no mask)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]          # input feature dimension D
    hidden_dim = config["hidden_dim"]   # projection dimension H
    nomask     = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 – high precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  float32

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    x_f16 = x.half()   # [B, N, N, D]  fp16

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  fp16

    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # fp16

    # ------------------------------------------------------------------
    # 6) Fuse gating (proj * gate) with a tiny Triton kernel
    # ------------------------------------------------------------------
    left, right = _apply_gates_triton(left, right, left_gate, right_gate)

    # ------------------------------------------------------------------
    # 7) Optional binary mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:

        left  = left  * mask_f16

    # Batched matrix‑multiply: left @ rightᵀ   (FP16×FP16 → FP16, FP32 accum)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]
