@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    HEAD_DIM: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Because left/right have layout (B,N,N,H) in contiguous memory with last dim H (fastest)
    # So stride for H dimension is 1.
    # Compute mask index for broadcast (mask shape B,N,N)
    mask_idx = offs // HEAD_DIM

    # load
    l_val = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0))
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0))
    r_val = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0))
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0))
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1))

    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)
```

5️⃣ **Mask‑&‑gate fusion** – a tiny Triton kernel multiplies  
   `left   = left_proj   * left_gate   * mask` and  
   `right  = right_proj  * right_gate  * mask` in‑place.  
   If no mask is present we simply multiply by the gates.

6️⃣ Core N³ update – batched GEMM: for each hidden channel `h` we
   compute `out_h = left_h @ right_h.T`.  By reshaping to
   `(B·H, N, N)` we reuse cuBLAS’s tensor‑core GEMM (`torch.bmm`).

7️⃣ Output LayerNorm + output‑gate (still fp16).

8️⃣ Final linear projection back to the model dimension `dim`
   (fp16 → fp32 output).

All heavy arithmetic stays in FP16 (tensor‑core friendly).  The only
explicit Triton kernel fuses the three element‑wise multiplies required
by the mask‑gate step, dramatically reducing memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# The mask has shape [B, N, N] and is broadcasted over the hidden
# dimension `HEAD_DIM`.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * left  (B,N,N,H)
    left_gate_ptr,     # fp16 * left_gate
    right_ptr,         # fp16 * right
    right_gate_ptr,    # fp16 * right_gate
    mask_ptr,          # fp16 * mask (B,N,N)
    total_elems: tl.int32,      # B·N·N·H
    HEAD_DIM: tl.constexpr,     # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elems

    # index of the mask element (each mask entry covers HEAD_DIM consecutive
    # entries of the left/right tensors)
    mask_idx = offs // HEAD_DIM

    # Load values (use zero/one as a neutral element when out‑of‑bounds)
    l_val   = tl.load(left_ptr      + offs, mask=active, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr + offs, mask=active, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr     + offs, mask=active, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr+ offs, mask=active, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr      + mask_idx, mask=active, other=tl.float16(1.0))

    # Three element‑wise multiplies fused in one pass
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr      + offs, l_out, mask=active)
    tl.store(right_ptr     + offs, r_out, mask=active)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights      : dict with all linear / layer‑norm parameters
        - config       : dict with keys
                         * "dim"         – model dimension D
                         * "hidden_dim" – hidden dimension H
                         * "nomask"      – True if mask is absent

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # D
    hidden_dim = config["hidden_dim"] # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2️⃣ Input LayerNorm (float32 → fp16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # [B, N, N, D]  (float32)
    x = x.to(torch.float16) # cast once for the rest of the pipeline

    # --------------------------------------------------------------
    # 3️⃣ Prepare weights (cast to fp16 once)
    # --------------------------------------------------------------
    # combine 4 linear layers (left, right, left_gate, right_gate)
    left_proj_w   = weights["left_proj.weight"]      # (H, D)
    right_proj_w  = weights["right_proj.weight"]     # (H, D)
    left_gate_w   = weights["left_gate.weight"]      # (H, D)
    right_gate_w  = weights["right_gate.weight"]     # (H, D)

    # concat along output dimension → (4·H, D)
    proj_weight_big = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    ).to(torch.float16)

    # out‑gate weight (H, D)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # final projection weight (D, H) – note nn.Linear(D, H) has weight (H, D)
    # `F.linear` expects (out_features, in_features), so we keep (H, D)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # layer‑norm parameters for the two LN layers
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # --------------------------------------------------------------
    # 4️⃣ One‑shot projection & gate computation (4× faster)
    # --------------------------------------------------------------
    B, N, _, _ = x.shape
    flat_x = x.view(-1, dim)                     # (B·N·N, D)

    # (B·N·N, 4·H)
    proj_big = F.linear(flat_x, proj_weight_big)
    proj_big = proj_big.view(B, N, N, 4 * hidden_dim)

    # split
    left_proj   = proj_big[..., 0               : hidden_dim]          # (B,N,N,H)
    right_proj  = proj_big[..., hidden_dim     : 2 * hidden_dim]      # (B,N,N,H)
    left_gate   = proj_big[..., 2 * hidden_dim : 3 * hidden_dim]      # (B,N,N,H)
    right_gate  = proj_big[..., 3 * hidden_dim : 4 * hidden_dim]      # (B,N,N,H)

    # sigmoid gates
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # out‑gate (separate linear)
    out_gate = torch.sigmoid(F.linear(flat_x, out_gate_w)).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5️⃣ Mask & gate fusion (Triton) – in‑place modification of left/right
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask → fp16 (broadcasted later)
        mask_f16 = mask.to(torch.float16).contiguous()  # (B,N,N)

        # make tensors contiguous for pointer arithmetic
        left_proj   = left_proj.contiguous()
        right_proj  = right_proj.contiguous()
        left_gate   = left_gate.contiguous()
        right_gate  = right_gate.contiguous()

        total_elems = left_proj.numel()                 # B·N·N·H
        BLOCK = 1024
        grid = ( (total_elems + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total_elems,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
        # after the kernel `left_proj` and `right_proj` already contain the
        # masked‑and‑gated values.
    else:
        # No mask → just multiply by the gates (in‑place)
        left_proj  = left_proj.mul_(left_gate)
        right_proj = right_proj.mul_(right_gate)

    # --------------------------------------------------------------
    # 6️⃣ Core N³ update: batched GEMM  (L @ Rᵀ) per hidden channel
    # --------------------------------------------------------------
    # shape to (B·H, N, N)
    left_mat  = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiplication – tensor cores work in fp16
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm + output‑gate (still fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate   # element‑wise broadcasted product

    # --------------------------------------------------------------
    # 8️⃣ Final projection back to model dimension D
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # fp16 result (B,N,N,D)

    # Convert to float32 as required by the caller
    return out.to(torch.float32)
```
