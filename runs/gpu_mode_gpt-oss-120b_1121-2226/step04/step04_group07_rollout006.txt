"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold3)

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32) → x   [B, N, N, D]
2. Cast x to FP16 for the heavy work.
3. Fuse the four linear projections (left_proj, left_gate, right_proj,
   right_gate) into a single GEMM by concatenating their weight matrices.
   Split the result and apply sigmoid gates.
4. Apply the optional binary mask (broadcast over the hidden dimension).
5. Core cubic update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication C = A @ Bᵀ where
   A,B ∈ [B·H, N, N] (FP16).  The multiplication is performed by a
   custom Triton kernel that uses Tensor‑Core MMA (BLOCK_M×BLOCK_N×BLOCK_K
   = 128×128×64) with FP32 accumulation.
6. Output LayerNorm (FP16), multiplication by the out‑gate,
   final linear projection back to the original dimension.
7. Cast the result to FP32 (API contract).

Only the cubic N³ update is done in Triton; all other ops use
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ
#    A : (batch, M, K)  FP16
#    B : (batch, N, K)  FP16   (will be accessed as Bᵀ)
#    C : (batch, M, N)  FP16   (accumulator is FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A : (batch, M, K)
    stride_bh, stride_bn, stride_bk,   # B : (batch, N, K)
    stride_ch, stride_cm, stride_cn,   # C : (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B tile (as Bᵀ): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # MMA (FP32 accumulation)
        acc += tl.dot(a_tile, b_tile)

    # write results (cast back to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ using the Triton kernel above.
    All tensors must be CUDA FP16 and contiguous.
    """
    assert A.is_cuda and B.is_cuda, "inputs must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "FP16 required"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in elements
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=12,          # 12 warps (384 threads) – good for H100
        num_stages=3,          # double‑buffering
    )
    return C


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) input layer‑norm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] float32

    # ------------------------------------------------------------------
    # 3) cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) fused linear projections + gates (four linear layers → one)
    # ------------------------------------------------------------------
    # concatenate all four weight matrices: [4*H, D]
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    proj_gate_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    ).to(device).to(torch.float16)                # (4*H, D)

    # single GEMM
    proj_gate = F.linear(x_f16, proj_gate_w)      # [B, N, N, 4*H] fp16
    left_proj, left_gate, right_proj, right_gate = torch.split(
        proj_gate, hidden_dim, dim=-1
    )
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # element‑wise gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 6) core cubic update (batched mat‑mul A @ Bᵀ)
    # ------------------------------------------------------------------
    Bsz, N, _, H = left.shape
    # reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(Bsz * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(Bsz * H, N, N)

    # batched mat‑mul using the Triton kernel defined above
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B*H, N, N] fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7) output layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    # out‑gate (Linear(dim → H) + sigmoid) – re‑use the same x (normed)
    out_gate_w = weights["out_gate.weight"].to(device).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # [B,N,N,H] fp16

    # output LayerNorm on the hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"].to(device).to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(device).to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # apply out‑gate
    out = out * out_gate

    # final linear projection back to original dimension
    to_out_w = weights["to_out.weight"].to(device).to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16

    # ------------------------------------------------------------------
    # 8) cast back to FP32 as required by the API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
