```python
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,    # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,    # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B tile (transposed view): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # (FP32) dot product via Tensor‑Core
        acc += tl.dot(a_tile, b_tile)

    # write back C (cast to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)
```

"""
TriMul “outgoing” forward – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → float16) .
2. Fuse the two linear projections (proj + gate) of each side by concatenating the
   weight matrices, then split the result and apply the sigmoid gate.
3. (Optional) apply the binary mask – fused into a tiny Triton kernel to satisfy
   the “must use a kernel” rule.
4. Core cubic update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication A @ Bᵀ with
   A,B ∈ [B·H, N, N].  The heavy work is done by a
   custom Triton kernel (BLOCK_M×BLOCK_N×BLOCK_K = 128×128×128,
   8 warps, 4 stages) which uses fp16 inputs, fp32 accumulation and
   Tensor‑Core MMA.
5. Output LayerNorm (fp16), multiply by the out‑gate, final linear
   projection back to the original dimension.
6. Cast to float32 for the API contract.

The kernel implements the N³ matrix multiplication; all other
operations are performed with fast torch kernels.  This layout
avoids costly transposes: we only reshape the tensors once before
and after the mat‑mul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A, B : fp16  (batch, M, K) and (batch, N, K)
#   C    : fp16  (batch, M, N) – result is cast from fp32 accumulator.
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions
    stride_ah, stride_am, stride_ak,    # A strides (batch, M, K)
    stride_bh, stride_bn, stride_bk,    # B strides (batch, N, K)
    stride_ch, stride_cm, stride_cn,    # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # col tile
    pid_b = tl.program_id(2)   # batch tile (B·H)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # batch‑specific pointers
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # ----- load A tile (BLOCK_M × BLOCK_K) -----
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ----- load B tile (BLOCK_K × BLOCK_N) -----
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ----- MMA (Tensor‑Core) -----
        acc += tl.dot(a_tile, b_tile)   # FP32 accumulation

    # write result (cast back to fp16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 128,
) -> torch.Tensor:
    """
    Batched mat‑mul C = A @ Bᵀ using the Triton kernel above.
    A, B – fp16 tensors of shape (batch, M, K) and (batch, N, K)
    Returns C – fp16 tensor of shape (batch, M, N)
    """
    assert A.is_cuda and B.is_cuda, "inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "fp16 required"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in elements (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps (256 threads) – matches H100 warp size
        num_stages=4,         # double‑buffering to hide load latency
    )
    return C


# ------------------------------------------------------------------
# Tiny Triton kernel to fuse mask multiplication (satisfies “use a kernel”)
# ------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    out_ptr, mask_ptr,
    size: tl.constexpr,          # total number of elements (B*N*N*H)
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < size
    val = tl.load(out_ptr + offs, mask=mask, other=0.0)
    m   = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # mask already 0/1
    tl.store(out_ptr + offs, val * m, mask=mask)


def _apply_mask(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication `tensor *= mask` using a Triton kernel.
    Both tensors must be fp16, contiguous and have identical shape.
    """
    assert tensor.is_cuda and mask.is_cuda
    numel = tensor.numel()
    BLOCK = 1024
    grid = ( (numel + BLOCK - 1) // BLOCK, )
    _apply_mask_kernel[grid](
        tensor,
        mask,
        numel,
        BLOCK=BLOCK,
        num_warps=4,
    )
    return tensor

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                                       # fp16

    # apply out‑gate
    out = out * out_gate

    # final linear projection back to original dimension D
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)                         # fp16

    # --------------------------------------------------------------
    # 8) Cast back to FP32 for the external API
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
