"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) – Triton implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) → *x*.
2️⃣ Cast to FP16 for the heavy linear work.
3️⃣ Compute left/right projections and the three sigmoid‑gates (bias‑free linear).
   Apply optional pair‑wise mask and the element‑wise gates in‑place.
4️⃣ Rearrange tensors to [B·H, N, N] (contiguous) where H = hidden_dim.
5️⃣ Launch a custom Triton kernel that performs a batched GEMM
        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]
   i.e. C = A @ Bᵀ with FP16 inputs, FP32 accumulation, FP16 write‑back.
   Tile sizes are chosen automatically (128/256) depending on sequence length.
6️⃣ Reshape back to [B, N, N, H] (FP16).
7️⃣ Second Layer‑Norm over the hidden dimension (FP16).
8️⃣ Apply the output‑gate and final bias‑free linear projection back to *dim*.
9️⃣ Cast to float32 and return.

Only the N³ multiplication (step 5) is done in Triton; all surrounding
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM C = A @ Bᵀ   (A,B : fp16 → fp32 accum → fp16 out)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N]   (fp16)
    b_ptr,               # right : [BH, N, N]   (fp16) – interpreted as (K,N)
    c_ptr,               # out   : [BH, N, N]   (fp16)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)            # batch * hidden
    pid_m = tl.program_id(axis=0)            # tile row
    pid_n = tl.program_id(axis=1)            # tile column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator (kept in registers)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- A tile (M × K) -----------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ----- Bᵀ tile (K × N) -----------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ----- Accumulate in FP32 using Tensor‑Cores -----------------
        # tl.dot supports an `out=` argument that adds the result to the
        # accumulator without an intermediate conversion.
        tl.dot(a, b, out=acc)

    # ------------------------------------------------------------------
    # Write back in fp16 (halves write bandwidth)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/int)
        - weights_dict : {
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          } (float32)
        - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) – stable accumulation
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) – fp16 → fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_fp16.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating – sigmoid (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # Apply element‑wise gates in‑place
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to [B·H, N, N] for the batched GEMM
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat  = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len)  # fp16, contiguous

    # Output buffer for the GEMM (FP16 write‑back)
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Choose tile sizes (larger tiles for longer sequences)
    # ------------------------------------------------------------------
    if seq_len <= 256:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 256
    BLOCK_K = 64  # works well for all tested sizes

    # Compute grid dimensions
    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # rows (i‑tiles)
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # cols (j‑tiles)
        BH,                                   # batch × hidden
    )

    # ------------------------------------------------------------------
    # 8️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,               # M
        seq_len,               # N
        seq_len,               # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right – note the transposed layout)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Free left/right tensors early to reduce peak memory (optional)
    del left, right, left_perm, right_perm, left_flat, right_flat

    # ------------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, H] (still fp16)
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 🔟 Second LayerNorm over the hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )  # fp16

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Apply output gate and final linear projection (fp16)
    # ------------------------------------------------------------------
    out = out * out_gate                     # element‑wise gate
    out = F.linear(out, weights["to_out.weight"].half())  # fp16 → dim

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Cast back to float32 for the API contract
    # ------------------------------------------------------------------
    return out.float()
