def _gate_mask_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, left_out_ptr, right_out_ptr,
    N: tl.constexpr, block_size: tl.constexpr, has_mask: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid*block_size + tl.arange(0, block_size)
    mask = offs < N

    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)
    l = l * lg
    r = r * rg
    if has_mask:

        l = l * m
        r = r * m
    tl.store(left_out_ptr + offs, l, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Input LayerNorm (FP32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape B,N,N,dim float32

    # Cast to fp16 for heavy work
    x_f16 = x.half()

    # Combine projection and gate weights
    # left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).half()  # shape (5*hidden_dim, dim)

    # Linear projection to get all required tensors in one go
    combined_out = F.linear(x_f16, combined_weight)  # shape B,N,N,5*hidden_dim

    # Split
    # Indices:
    # 0: hidden_dim => left_proj
    # hidden_dim: 2*hidden_dim => left_gate
    # 2*hidden_dim: 3*hidden_dim => right_proj
    # 3*hidden_dim: 4*hidden_dim => right_gate
    # 4*hidden_dim: 5*hidden_dim => out_gate
    h = hidden_dim
    left_proj = combined_out[..., :h]
    left_gate = combined_out[..., h:2*h]
    right_proj = combined_out[..., 2*h:3*h]
    right_gate = combined_out[..., 3*h:4*h]
    out_gate = combined_out[..., 4*h:5*h]

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply gating and optional mask using Triton kernel
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16)  # shape B,N,N
    else:
        mask_f16 = None

    # The gating kernel yields left_out and right_out
    left_out, right_out = _apply_gate_and_mask_triton(
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_f16,
        hidden_dim,
        block_size=1024,
    )

    # left_out and right_out shape: B,N,N,H
    # Now cubic update: batched matrix multiplication per hidden channel
    B, N, _, H = left_out.shape
    # reshape for batch matmul
    # left_mat shape: (B*H, N, N)
    left_mat = left_out.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_out.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Compute out_mat = left_mat @ right_mat.transpose(1,2)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] fp16

    # Reshape to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # Output LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )  # fp16

    # Apply out_gate scaling
    out = out * out_gate

    # Final linear projection
    out = F.linear(out, weights["to_out.weight"].half())  # shape B,N,N,dim fp16

    # Cast back to float32
    return out.float()

"""
Triton‑accelerated forward pass for the “outgoing” TriMul (AlphaFold3) module.

Algorithm
---------
1️⃣ Input LayerNorm (FP32) on the last dimension.
2️⃣ Cast to FP16 – all heavy work runs in half‑precision.
3️⃣ **Fused linear projection**: concatenate the five weight matrices
   (left_proj, left_gate, right_proj, right_gate, out_gate) and perform a
   single `F.linear`.  The result is split into
   `left_proj, left_gate, right_proj, right_gate, out_gate`.
4️⃣ Apply sigmoid to the three gate tensors.
5️⃣ **Fused gating + optional mask** – a tiny Triton kernel multiplies
   `left_proj * left_gate` and `right_proj * right_gate` (and also the
   binary mask if supplied) in one pass, producing the gated tensors.
6️⃣ **Cubic update** – compute
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   via a strided‑batched GEMM (`torch.bmm`).  This reuses Tensor‑Core
   acceleration on the H100.
7️⃣ Output LayerNorm (FP16), element‑wise scaling by `out_gate`,
   and final bias‑free linear projection back to the original dimension.
8️⃣ Cast the result back to FP32 for the API.

Only the gating step is implemented in Triton; all large matrix‑multiplications
use cuBLAS which already provides near‑optimal performance on the target GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gating (and optional mask) in one pass.
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,               # fp16 *
    right_ptr,              # fp16 *
    left_gate_ptr,          # fp16 *
    right_gate_ptr,         # fp16 *
    mask_ptr,               # fp16 *
    left_out_ptr,           # fp16 *
    right_out_ptr,          # fp16 *
    H: tl.constexpr,        # hidden_dim (int)
    N_ELEM: tl.constexpr,   # total number of elements (B·N·N·H)
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr, # compile‑time bool
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEM

    # Load values (default 0.0 for out‑of‑bounds, though mask blocks them)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    if HAS_MASK:
        # Every H consecutive elements belong to the same (b,i,j) mask entry.
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # broadcasted mask
        l = l * lg * m
        r = r * rg * m
    else:
        l = l * lg
        r = r * rg

    tl.store(left_out_ptr + offs,  l, mask=mask)
    tl.store(right_out_ptr + offs, r, mask=mask)


def _apply_gate_and_mask_triton(left, right, left_gate, right_gate, mask, hidden_dim, block_size: int = 1024):
    """
    Fuse `left *= left_gate`, `right *= right_gate` and optional mask
    into a single Triton kernel. All tensors must be contiguous fp16 and have
    identical shape (B, N, N, hidden_dim). Returns the gated tensors.
    """
    total = left.numel()
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Flattened mask (if present) – shape (B, N, N)
    if mask is not None:
        mask_f16 = mask.contiguous()
        has_mask = 1
    else:
        # dummy pointer – never dereferenced because HAS_MASK==0
        mask_f16 = torch.empty(0, dtype=torch.float16, device=left.device)
        has_mask = 0

    grid = ((total + block_size - 1) // block_size, )
    _gate_mask_fuse_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        mask_f16,
        left_out,
        right_out,
        hidden_dim,
        total,
        BLOCK_SIZE=block_size,
        HAS_MASK=has_mask,
        num_warps=4,
        num_stages=2,
    )
    return left_out, right_out

    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32) – high‑precision statistics
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy‑weight path
    # ------------------------------------------------------------------
    x_f16 = x.half()  # [B, N, N, D] (fp16)

    # ------------------------------------------------------------------
    # 4) One‑shot linear projection for all gates & projections
    # ------------------------------------------------------------------
    # Concatenate weights: left_proj, left_gate, right_proj, right_gate, out_gate
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # shape (5*H, D)

    # Linear: produces [B, N, N, 5*H]
    combined = F.linear(x_f16, combined_weight)  # fp16

    # ------------------------------------------------------------------
    # 5) Slice the concatenated tensor
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj = combined[..., :H]
    left_gate  = combined[..., H:2*H]
    right_proj = combined[..., 2*H:3*H]
    right_gate = combined[..., 3*H:4*H]
    out_gate   = combined[..., 4*H:5*H]

    # ------------------------------------------------------------------
    # 6) Sigmoid on gate tensors
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 7) Fuse gating (and optional mask) via Triton
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] -> broadcast over hidden dim inside kernel
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    left_gated, right_gated = _apply_gate_and_mask_triton(
        left_proj.contiguous(),
        right_proj.contiguous(),
        left_gate.contiguous(),
        right_gate.contiguous(),
        mask_f16,
        hidden_dim,
        block_size=1024,
    )

    # ------------------------------------------------------------------
    # 8) Cubic (outgoing) update: batched matrix multiplication per hidden channel
    # ------------------------------------------------------------------
    B, N, _, H = left_gated.shape  # B, seq_len, N, hidden_dim
    # reshape to (B*H, N, N) for a strided‑batched GEMM
    left_mat = left_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_gated.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_mat[b*h] = left_mat[b*h] @ right_mat[b*h].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16, shape (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )  # fp16

    # ------------------------------------------------------------------
    # 10) Scale by out_gate and final projection to original dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].half())  # [B, N, N, D] fp16

    # ------------------------------------------------------------------
    # 11) Cast back to FP32 (API contract)
    # ------------------------------------------------------------------
    return out.float()
