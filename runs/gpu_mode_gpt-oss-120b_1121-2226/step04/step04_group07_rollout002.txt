"""custom_kernel – Triton‑accelerated “outgoing” TriMul

Implements the forward pass of the TriMul module (AlphaFold3 “outgoing”
update).  The heavy cubic update
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is a batched matrix multiplication A @ Bᵀ where each hidden channel is
treated as an independent batch.  A custom Triton kernel performs this
multiplication in FP16 (Tensor‑Core MMA) with FP32 accumulation.
All other operations (LayerNorm, linear projections, sigmoid gates,
masking) are performed with fast Torch kernels and the data is kept in
FP16 for the compute‑intensive part.  The final output is cast back to
FP32 to satisfy the API contract.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ  (FP16 inputs, FP32 acc)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # strides for A (batch, M, K)
    stride_bh, stride_bn, stride_bk,    # strides for B (batch, N, K)
    stride_ch, stride_cm, stride_cn,    # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Each program instance computes a tile C[pid_b, pid_m*BM:..., pid_n*BN:...]."""
    pid_b = tl.program_id(0)   # batch index (B·H)
    pid_m = tl.program_id(1)   # block row
    pid_n = tl.program_id(2)   # block col

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B tile (transposed view): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # MMA accumulation (fp32)
        acc += tl.dot(a_tile, b_tile)

    # --------------------------------------------------------------
    # Write back C (cast to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for 3‑D tensors (batch, M, K) and (batch, N, K)
    using the Triton kernel above.
    All tensors must be CUDA FP16.
    """
    assert A.is_cuda and B.is_cuda, "inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "inputs must be fp16"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in elements
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps (256 threads) work well on H100
        num_stages=2,
    )
    return C


# ------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys
                                "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fused linear projection + gate for left and right sides
    # ------------------------------------------------------------------
    # Left side: concatenate projection and gate weights -> single matmul
    left_w_proj = weights["left_proj.weight"]          # (H, D)
    left_w_gate = weights["left_gate.weight"]          # (H, D)
    left_w_cat = torch.cat([left_w_proj, left_w_gate], dim=0).to(torch.float16)  # (2H, D)

    left_all = F.linear(x_fp16, left_w_cat)            # [B, N, N, 2H]
    left_proj = left_all[..., :hidden_dim]            # [B, N, N, H]
    left_gate = torch.sigmoid(left_all[..., hidden_dim:])  # [B, N, N, H]

    # Right side
    right_w_proj = weights["right_proj.weight"]
    right_w_gate = weights["right_gate.weight"]
    right_w_cat = torch.cat([right_w_proj, right_w_gate], dim=0).to(torch.float16)  # (2H, D)

    right_all = F.linear(x_fp16, right_w_cat)          # [B, N, N, 2H]
    right_proj = right_all[..., :hidden_dim]
    right_gate = torch.sigmoid(right_all[..., hidden_dim:])

    # Element‑wise gating
    left = left_proj * left_gate          # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Optional binary mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Core cubic update: batched mat‑mul A @ Bᵀ
    #    Treat each hidden channel as an independent batch.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # Triton batched mat‑mul (FP16 inputs, FP32 accumulation)
    out_mat = _batched_matmul_triton(
        left_mat, right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B*H, N, N]  (fp16)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection back to D
    # ------------------------------------------------------------------
    # out‑gate (sigmoid of a linear on the original normalized input)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)          # (H, D)
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))            # [B, N, N, H]

    # Output LayerNorm over hidden dimension H
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    # Apply out‑gate
    out = out * out_gate

    # Final linear projection back to dim D
    to_out_w = weights["to_out.weight"].to(torch.float16)             # (D, H)
    out = F.linear(out, to_out_w)                                    # [B, N, N, D] fp16

    # Cast back to FP32 as required by the API
    return out.to(torch.float32)
