"""
Triton‑accelerated outgoing TriMul (forward only)

Algorithm
---------
1.  Layer‑norm on the input (float32 → stable).
2.  Cast to fp16 for the heavy part.
3.  **Fused linear + gate**:
    * Compute `left_proj` and `left_gate` together by concatenating their
      weight matrices (shape D×2H).  The first H output channels are the
      projection, the next H are the gate (followed by a sigmoid).
    * Do the same for the right branch.
    * `out_gate` stays a separate linear + sigmoid.
4.  **Mask & gate fusion** – a tiny Triton kernel multiplies
    `left *= left_gate * mask` and `right *= right_gate * mask`
    (broadcast mask over the hidden dimension).  When there is no mask we
    simply do in‑place element‑wise multiplication.
5.  **Core N³ update** – batched GEMM `L @ Rᵀ` for each hidden channel.
    By reshaping to `(B·H, N, N)` we can reuse cuBLAS (`torch.bmm`), which
    runs on tensor cores in fp16.
6.  Layer‑norm on the hidden representation, then apply `out_gate`.
7.  Final linear projection back to the model dimension `dim`.
8.  Cast the result to float32 (required output type).

The implementation keeps the two heaviest memory‑bound steps (the two
linear + gate operations and the mask‑gate fusion) in a single kernel each,
drastically reducing traffic compared to the naïve PyTorch version.
All large matrix multiplications stay in fp16 and benefit from H100
tensor‑core throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
#   mask is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 * left
    left_gate_ptr,       # fp16 * left_gate
    right_ptr,           # fp16 * right
    right_gate_ptr,      # fp16 * right_gate
    mask_ptr,            # fp16 * mask   (shape B*N*N, no hidden dim)
    total_elems: tl.int32,        # B*N*N*H
    HEAD_DIM: tl.constexpr,       # hidden dim H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # mask index (ignore hidden dim)
    mask_idx = offs // HEAD_DIM

    # Load tensors (masked load returns 0/1 for out‑of‑range)
    l_val    = tl.load(left_ptr       + offs, mask=mask, other=tl.float16(0.0))
    l_gate   = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    r_val    = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate   = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val    = tl.load(mask_ptr       + mask_idx, mask=mask, other=tl.float16(1.0))

    # Fuse three multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for the heavy work
    # ------------------------------------------------------------------

    # 5) Fused linear + gate for left and right sides
    # ------------------------------------------------------------------
    # left_combined: [B, N, N, 2H] → split into left & left_gate
    left_combined = F.linear(x_f16, left_comb_w)               # (B,N,N,2H)
    left = left_combined[..., :hidden_dim]                     # (B,N,N,H)
    left_gate_raw = left_combined[..., hidden_dim:]            # (B,N,N,H)
    left_gate = torch.sigmoid(left_gate_raw)

    # right_combined: similarly
    right_combined = F.linear(x_f16, right_comb_w)             # (B,N,N,2H)
    right = right_combined[..., :hidden_dim]                   # (B,N,N,H)
    right_gate_raw = right_combined[..., hidden_dim:]          # (B,N,N,H)
    right_gate = torch.sigmoid(right_gate_raw)

    # out_gate (separate linear)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))      # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6) Optional mask & gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → fp16, contiguous
        mask_f16 = mask.to(torch.float16).contiguous()
        # Ensure contiguous storage for element‑wise kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        total = left.numel()                         # B*N*N*H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask → just apply gates in‑place
        left = left.mul_(left_gate)
        right = right.mul_(right_gate)

    # ------------------------------------------------------------------
    # 7) Core N³ update: batched GEMM L @ Rᵀ for each hidden channel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape      # H == hidden_dim
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    # batched matrix multiplication (fp16 → tensor‑core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))     # (B·H, N, N)
    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate          # element‑wise product (broadcasted)

    # ------------------------------------------------------------------
    # 9) Final linear projection back to model dim D (fp16 → fp32)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, D] (fp16)
    out = out.to(torch.float32)     # required output dtype

    return out
```
