"""
TriMul “outgoing” kernel – Triton‐accelerated implementation
============================================================

Algorithm (forward only)
------------------------
1. Layer‑norm the 4‑D input X ∈ [B, N, N, D] in FP32.
2. Cast to FP16 for the compute‑heavy part.
3. Linear projections (bias‑free) → L, R ∈ [B, N, N, H] (H = hidden_dim).
4. Compute three sigmoid gates from the same X.
5. Fuse the element‑wise mask (if any) and the left/right gates in a tiny
   Triton kernel (one kernel, zero extra launch overhead).
6. Core cubic update:
        out_{b,i,j,h} = Σ_k L_{b,i,k,h} * R_{b,j,k,h}
   i.e. for each head h we need a matrix multiplication  L_h @ R_hᵀ
   with L_h, R_h ∈ ℝ^{N×N}.  We implement this as a *single* Triton
   batched GEMM that works on the original layout (no permute /
   contiguous copy) – each (b, h) pair is a separate batch in the
   kernel.  This avoids the memory copy required by the PyTorch
   `view/permute` trick and lets the kernel reuse the same data layout
   that the linear layers produced.
7. Layer‑norm on the hidden dimension, multiply by the out‑gate and apply
   the final linear projection back to D.
8. Cast the result back to FP32 (API requirement).

Only the mask‑gate fusion and the cubic GEMM are written in Triton;
all other operations use highly‑optimised PyTorch kernels.
The fused GEMM reduces launch overhead and removes an extra
`contiguous` copy, giving a speed‑up that brings the end‑to‑end
runtime well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Tiny kernel that fuses:  left = left * left_gate * mask,
#                              right = right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,               # [total_elems]   fp16
    left_gate_ptr, right_gate_ptr,     # [total_elems]   fp16
    mask_ptr,                          # [total_masks]   fp16 (flattened), only used when HAS_MASK=1
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # optional mask (broadcasted over hidden dimension)
    if HAS_MASK:
        # bring the mask to the hidden‑dim axis: each hidden element shares
        # the same mask entry -> mask_idx = offs // hidden_dim
        mask_idx = offs // hidden_dim
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * m_val
        right = right * m_val

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # store in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
) -> None:
    """Launch the tiny Triton kernel that fuses mask and gates."""
    total = left.numel()
    BLOCK = 1024                      # 1 KB work per program – good trade‑off
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1)    # flatten → [B·N·N]
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


# ----------------------------------------------------------------------
# 2. Batched GEMM for the cubic update (outgoing TriMul)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    A_ptr,                # left  [B,N,N,H]  fp16
    B_ptr,                # right [B,N,N,H]  fp16
    C_ptr,                # out   [B,N,N,H]  fp16
    M, N, K,              # all equal to seq_len (N)
    stride_am, stride_ak,                 # A: i‑stride, k‑stride
    stride_bk, stride_bn,                 # B (transposed): k‑stride, j‑stride
    stride_cm, stride_cn,                 # C: i‑stride, j‑stride
    stride_batch, stride_head,            # stride to next batch & next head (inner dim)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_HEADS: tl.constexpr,
    # NOTE: we flatten (batch, head) into a single 3rd program‑id dimension.
):
    ###########################################################
    # program IDs
    pid_m = tl.program_id(0)          # tile row   (i)
    pid_n = tl.program_id(1)          # tile col   (j)
    pid_h = tl.program_id(2)          # flattened (batch, head)
    ###########################################################

    # compute start indices of this tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # recover original batch and head indices
    batch_idx = pid_h // NUM_HEADS
    head_idx  = pid_h % NUM_HEADS

    # base pointers for this (batch,head)
    A_base = A_ptr + batch_idx * stride_batch + head_idx * stride_head
    B_base = B_ptr + batch_idx * stride_batch + head_idx * stride_head
    C_base = C_ptr + batch_idx * stride_batch + head_idx * stride_head

    # range vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # pre‑compute offsets that do not depend on the k‑loop
    a_ptrs = A_base + offs_i[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_base + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_j[None, :] * stride_bn

    # accumulator (FP32 for Tensor‑core precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # loop over K dimension in tiles
    # ---------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k_offset = kk * BLOCK_K
        cur_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A and B (B is accessed as Bᵀ, hence the stride order above)
        a = tl.load(a_ptrs + k_offset * stride_ak,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs + k_offset * stride_bk,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0)

        # standard GEMM tile
        acc += tl.dot(a, b)         # (BLOCK_M, BLOCK_N)  FP32
    # ---------------------------------------------------------

    # write back only the valid region
    c = acc.to(tl.float16)
    C_ptrs = C_base + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    tl.store(C_ptrs, c, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor) -> None:
    """
    Compute   out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    using a custom Triton GEMM that works directly on the
    [B,N,N,H] layout (no extra reshapes).
    """
    B, N, _, H = left.shape                     # left/right are [B,N,N,H] fp16
    total_heads = H                               # we treat each head independently
    # strides of a contiguous tensor [B,N,N,H]
    stride_b, stride_i, stride_k, stride_h = left.stride()
    # launch grid
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        (tl.cdiv(N, BLOCK_M),            # rows (i)
         tl.cdiv(N, BLOCK_N),            # cols (j)
         B * total_heads),               # batch × head
    )
    _tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        N, N, N,                         # M, N, K
        stride_i, stride_k,              # stride_am, stride_ak
        stride_k, stride_i,              # stride_bk, stride_bn (Bᵀ)
        stride_i, stride_k,              # stride_cm, stride_cn  (output same as left)
        stride_b, stride_h,              # stride across batch & across head
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        NUM_HEADS=H,
        num_warps=8,
    )


# ----------------------------------------------------------------------
# 3. Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.
    Arguments:
        data = (input_tensor, mask, weights_dict, config_dict)
            input_tensor : torch.Tensor [B, N, N, D] (float32)
            mask         : torch.Tensor [B, N, N] (bool/int) or None
            weights       : dict of model parameters (float32)
            config        : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  FP32

    # ------------------------------------------------------------------
    # 2) FP16 cast for the heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16))   # [B, N, N, H]
    right = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – also FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_f16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_f16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(F.linear(x_f16, weights["out_gate.weight"].to(torch.float16)))  # later

    # ------------------------------------------------------------------
    # 5) Fuse mask (if any) and gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left, right, left_gate, right_gate, mask_f16, hidden_dim,
    )  # left/right are updated in‑place

    # ------------------------------------------------------------------
    # 6) Core cubic update: out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    #    Implemented as a single Triton GEMM that works on the original
    #    layout – no extra permute/contiguous copies.
    # ------------------------------------------------------------------
    out_hidden = torch.empty_like(left)   # allocate [B, N, N, H] (fp16)
    _tri_mul_gemm(left, right, out_hidden)   # writes into out_hidden

    # ------------------------------------------------------------------
    # 7) Output LayerNorm + out‑gate + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    out = out * out_gate                         # apply out‑gate (fp16)

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))   # back to D

    # ------------------------------------------------------------------
    # 8) Cast back to FP32 for the public API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
