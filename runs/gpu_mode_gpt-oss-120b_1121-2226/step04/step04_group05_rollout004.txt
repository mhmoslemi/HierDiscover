"""
TriMul “outgoing” update (forward‑only) – Triton‑accelerated core.
-----------------------------------------------------------------
Algorithm
---------
1. Layer‑norm the 4‑D input (FP32 → FP16).
2. Two bias‑free linear projections (left/right) in FP16.
3. Optional binary mask (broadcasted) applied to the projected tensors.
4. Three sigmoid gates (left_gate, right_gate, out_gate) computed from the
   normalized input and applied element‑wise.
5. Core cubic operation:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is a batched mat‑mul L @ Rᵀ for each hidden channel.
   We reshape (B,N,N,H) → (B·H, N, N) and run a custom Triton kernel.
   The kernel tiles the work with (BLOCK_M×BLOCK_N×BLOCK_K) and uses
   FP16 inputs with FP32 accumulation (Tensor‑core friendly).
6. Reshape back to (B,N,N,H) (still FP16).
7. Output LayerNorm (FP16) → out_gate multiplication → final bias‑free
   linear projection back to the original dimension.
8. Cast result to FP32 for the public API.

All heavy computation (the N³ mat‑mul) is performed in Triton; the rest
uses high‑performance PyTorch FP16 kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel (batched mat‑mul A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 64},
            num_warps=4,
            num_stages=2,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64},
            num_warps=8,
            num_stages=3,
        ),
        triton.Config(
            {"BLOCK_M": 256, "BLOCK_N": 128, "BLOCK_K": 64},
            num_warps=16,
            num_stages=4,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 256, "BLOCK_K": 64},
            num_warps=8,
            num_stages=4,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (int32)
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K) strides
    stride_bh, stride_bn, stride_bk,   # B: (batch, N, K) strides
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched mat‑mul C = A @ Bᵀ  (FP16 input → FP32 accumulation → FP16 out)."""
    pid_m = tl.program_id(0)               # block row
    pid_n = tl.program_id(1)               # block col
    pid_b = tl.program_id(2)               # batch (flattened B·H)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Tile loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # ---- Load A tile (M×K) ----
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) \
                 + ((k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        mask_a = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ---- Load B tile transposed (K×N) ----
        b_ptrs = b_batch_ptr + ((k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk) \
                 + (offs_n[None, :] * stride_bn)
        mask_b = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Dot product (FP32 accumulation)
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N)

    # ---- Store result (cast back to fp16) ----
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Calls the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) as fp16.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # a good default; autotune may override
        num_stages=3,
    )
    return C


# ----------------------------------------------------------------------
# Public entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” update (no gradients).

    Args:
        data: (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor  [B, N, N, D]  (float32)
            - mask         : torch.Tensor  [B, N, N]    (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys:
                * dim (int)           – input feature dimension D
                * hidden_dim (int)    – projection dimension H
                * nomask (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1. Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (FP32 -> preserve numeric stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3. Cast to FP16 for the bulk of the computation
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Linear projections (bias‑free) – FP16 weights
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 6. Gates (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w)) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8. Core cubic operation: batched mat‑mul over the sequence dimension
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B=batch, H=hidden_dim

    # reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Pick block sizes (the Triton kernel is autotuned; these are just defaults)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B*H, N, N] (fp16)

    # Restore (B, N, N, H) layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 9. Output LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # apply out‑gate (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)  # shape (dim, hidden_dim)
    out = F.linear(out, to_out_w)  # [B, N, N, D] (fp16)

    # ------------------------------------------------------------------
    # 10. Cast back to FP32 for the API contract
    # ------------------------------------------------------------------
    return out.to(torch.float32)
