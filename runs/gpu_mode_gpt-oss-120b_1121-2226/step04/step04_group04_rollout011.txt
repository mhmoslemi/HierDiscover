```
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_WARPS: tl.constexpr,
    FP16_ACC: tl.constexpr
):
    pid_batch = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets ...
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # K loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) if FP16_ACC else tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + pid_batch * batch_stride_a + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak
        b_ptrs = B_ptr + pid_batch * batch_stride_b + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn
        # load
        a = tl.load(a_ptrs, mask=offs_m < M & k_offset < K, other=0.0) # shape
        b = tl.load(b_ptrs, mask=k_offset < K & offs_n < N, other=0.0)
        # dot
        if FP16_ACC:
            acc = tl.dot(a, b, acc)
        else:
            acc += tl.dot(a, b)
    # store
    c_ptrs = C_ptr + pid_batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=offs_m < M & offs_n < N)
```

"""
Highly‑optimised Triton implementation of the “outgoing” TriMul update.

Algorithm
---------
1. Input LayerNorm (FP32 for numerical stability).
2. Cast to FP16 and fuse *all* linear projections/gates into a single
   batched GEMM by concatenating the 5 weight matrices (left_proj,
   right_proj, left_gate, right_gate, out_gate).  This avoids reading the
   huge 4‑D tensor multiple times.
3. Split the fused result, reshape to (B,N,N,H) and apply the binary
   mask (if present) together with the three sigmoid gates.
4. Core cubic contraction:
   out_{b,i,j,h} = Σₖ left_{b,i,k,h} · right_{b,j,k,h}
   → a batched matrix multiplication L @ Rᵀ for each head h.
   The heavy O(N³·H) work is performed by a custom Triton kernel that
   processes all (B·H) GEMM instances in a single launch, using
   Tensor‑core friendly FP16⋅FP16→FP32 accumulation.
5. Output LayerNorm, multiply by the out‑gate, final linear projection
   back to the original feature dimension.
6. Return FP32 tensor (API‑compatible).

The Triton kernel (batched_gemm_kernel) is the only non‑trivial
operation; everything else uses torch’s highly‑optimised BLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel for the cubic contraction
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_batch_a, stride_m_a, stride_k_a,   # left (L) strides
    stride_batch_b, stride_k_b, stride_n_b,   # right (R) strides (Rᵀ layout)
    stride_batch_c, stride_m_c, stride_n_c,   # output (O) strides
    M, N, K,
    HEADS: tl.constexpr,                      # hidden dimension (=H)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for every (batch=head) a matrix multiplication:
        C[b] = A[b] @ B[b].T      (A,B shape (M,K) and (N,K) )
    where A and B are stored with the stride pattern of the original
    (B,N,N,H) tensors (row‑major, H is the innermost dimension).
    """
    # program ids
    pid_batch = tl.program_id(0)   # 0 … B*HEADS‑1
    pid_m = tl.program_id(1)      # tile row index
    pid_n = tl.program_id(2)      # tile col index

    # recover batch (b) and head (h) from flattened index
    b = pid_batch // HEADS
    h = pid_batch % HEADS

    # global offsets for this (b,h) pair
    batch_offset_a = b * stride_batch_a + h
    batch_offset_b = b * stride_batch_b + h
    batch_offset_c = b * stride_batch_c + h

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + offs_k
        mask_k = k_off < K

        # pointers for the current tile
        a_ptrs = A_ptr + batch_offset_a \
                  + offs_m[:, None] * stride_m_a \
                  + k_off[None, :] * stride_k_a
        b_ptrs = B_ptr + batch_offset_b \
                  + k_off[:, None] * stride_k_b \
                  + offs_n[None, :] * stride_n_b

        # guarded loads
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate in FP32
        acc = tl.dot(a, b, out=acc)

    # write back (cast to FP16)
    c_ptrs = C_ptr + batch_offset_c \
             + offs_m[:, None] * stride_m_c \
             + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches `batched_gemm_kernel`.  Both tensors must have
    shape [B, N, N, H] and dtype torch.float16, be contiguous and on the
    same CUDA device.
    Returns a tensor of the same shape and dtype containing left @ rightᵀ.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    # output placeholder
    out = torch.empty_like(left)

    # strides in *elements* (not bytes)
    stride_batch = left.stride(0)      # B‑stride = N*N*H
    stride_m_a   = left.stride(1)      # row stride = N*H
    stride_k_a   = left.stride(2)      # col stride = H

    # right is accessed as transposed (k, j)
    stride_k_b   = right.stride(2)     # row stride of transposed = H
    stride_n_b   = right.stride(1)     # col stride of transposed = N*H

    stride_m_c   = out.stride(1)       # row stride of output = N*H
    stride_n_c   = out.stride(2)       # col stride of output = H

    # tiling parameters – experimentally good for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # flattened (batch, head)
        (N + BLOCK_M - 1) // BLOCK_M,          # number of row tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # number of col tiles
    )

    batched_gemm_kernel[grid](
        left,
        right,
        out,
        stride_batch, stride_m_a, stride_k_a,
        stride_batch, stride_k_b, stride_n_b,
        stride_batch, stride_m_c, stride_n_c,
        N, N, N,
        HEADS=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul “outgoing” update (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – fp32

    # --------------------------------------------------------------
    # 3) Cast to FP16 and fuse all five linear layers
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)
    # [B, N, N, D] → flatten leading three dims
    x_flat = x_f16.view(-1, D)               # (B·N·N, D)

    # concatenate weight matrices (out_features stacked)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                      # (5·H, D)

    # single large GEMM → (B·N·N, 5·H)
    proj = F.linear(x_flat, w_concat)

    # split back to individual heads
    splits = torch.split(proj, hidden_dim, dim=1)
    left_proj_raw, right_proj_raw, left_gate_raw, right_gate_raw, out_gate_raw = splits

    # reshape to original 4‑D layout
    left_proj_raw = left_proj_raw.view(B, N, N, hidden_dim)
    right_proj_raw = right_proj_raw.view(B, N, N, hidden_dim)
    left_gate_raw = left_gate_raw.view(B, N, N, hidden_dim)
    right_gate_raw = right_gate_raw.view(B, N, N, hidden_dim)
    out_gate_raw = out_gate_raw.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4) Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_proj_raw = left_proj_raw * mask_f16
        right_proj_raw = right_proj_raw * mask_f16

    # --------------------------------------------------------------
    # 5) Gating (sigmoid) and apply gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    left = left_proj_raw * left_gate      # (B,N,N,H) FP16
    right = right_proj_raw * right_gate   # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 6) Core cubic contraction – batched GEMM via Triton
    # --------------------------------------------------------------
    out = batched_gemm_triton(left, right)   # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # FP16

    out = out * out_gate               # FP16

    to_out_w = weights["to_out.weight"].to(torch.float16)  # (D, H)
    out = F.linear(out, to_out_w)      # (B, N, N, D) FP16

    # --------------------------------------------------------------
    # 8) Cast back to FP32 for the public API
    # --------------------------------------------------------------
    return out.to(torch.float32)
