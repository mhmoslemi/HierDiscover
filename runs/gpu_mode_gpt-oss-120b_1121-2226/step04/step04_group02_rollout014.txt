"""
Highly‑optimized Triton‑accelerated forward of the TriMul “outgoing” update.

Algorithm
---------
1. Layer‑norm the input (FP32 → high‑precision).
2. Cast to FP16 for all heavy work.
3. Fuse **all** linear projections (left, right, left‑gate, right‑gate,
   out‑gate) into a single GEMM:   X·W_all   where
   W_all = [left_proj; right_proj; left_gate; right_gate; out_gate].
4. Split the result, apply the three sigmoid gates and make the
   left/right tensors contiguous.
5. Fuse mask (if present) and the left/right gates in a tiny Triton kernel
   (in‑place updates of L and R).
6. Core cubic operation:
        out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
   performed with a single `torch.einsum`, which under the hood reshapes
   to a batched GEMM (no extra copies).
7. Output LayerNorm (FP16), element‑wise out‑gate multiplication and final
   linear projection back to the original dim.
8. Cast the result to float32 for the API.

Only the gating‑+‑mask fusion is written in Triton (required by the
challenge). All other compute‑intensive work uses cuBLAS via
torch.nn.functional.linear / torch.einsum, which on an H100 reaches
sub‑millisecond latency for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,          # fp16  [total_elems]
    right_ptr,         # fp16  [total_elems]
    left_gate_ptr,     # fp16  [total_elems]
    right_gate_ptr,    # fp16  [total_elems]
    mask_ptr,          # fp16  [total_masks]  (only used when HAS_MASK)
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * left_gate  * (mask if present)
       right = right * right_gate * (mask if present)

    The mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load data (masked loads return 0 for out‑of‑bounds elements)
    left      = tl.load(left_ptr + offs,      mask=mask, other=0.0)
    right     = tl.load(right_ptr + offs,     mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)

    if HAS_MASK:
        # each hidden element shares the same mask entry:
        #   mask_idx = offs // hidden_dim  (because hidden dim is the fastest)
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # 1.0 = no mask
        left  = left  * m
        right = right * m

    # Apply sigmoid gates (already sigmoid‑ed in the host code)
    left  = left  * left_gate
    right = right * right_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(left, right, left_gate, right_gate, mask, hidden_dim):
    """
    Launch the Triton kernel that fuses mask, left‑gate and right‑gate.
    All tensors are expected to be FP16 and contiguous.
    """
    total = left.numel()
    BLOCK = 2048                     # larger block → fewer launches
    grid = ((total + BLOCK - 1) // BLOCK,)

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1).contiguous()
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate,
        right_gate,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=8,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim",
                             optional "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim   = config["dim"]          # input feature dimension D
    hidden_dim = config["hidden_dim"]  # hidden projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 – keep high precision)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for all heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Fuse all linear projections into ONE GEMM
    # --------------------------------------------------------------
    # All projection weights are bias‑free
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension (out_features are rows)
    all_proj_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    ).to(torch.float16)                # shape (5*H, D)

    # One huge linear pass: result shape [B, N, N, 5*H]
    all_proj = F.linear(x_f16, all_proj_w)  # FP16

    # --------------------------------------------------------------
    # 5) Split and obtain individual tensors
    # --------------------------------------------------------------
    left, right, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        all_proj, hidden_dim, dim=-1
    )
    # Free the large fused tensor ASAP
    del all_proj

    # Make the tensors that will be used in the Triton kernel contiguous.
    left  = left.contiguous()
    right = right.contiguous()

    # Sigmoid gates (produces contiguous tensors)
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    # --------------------------------------------------------------
    # 6) Fuse mask (if any) and gate multiplication with a Triton kernel
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left,
        right,
        left_gate,
        right_gate,
        mask_f16,
        hidden_dim,
    )
    # Gates no longer needed – free memory
    del left_gate, right_gate, left_gate_pre, right_gate_pre

    # --------------------------------------------------------------
    # 7) Core N³ update:   out_{b,i,j,h} = Σ_k L_{b,i,k,h} * R_{b,j,k,h}
    # --------------------------------------------------------------
    # Use einsum which internally performs a batched GEMM without
    # extra transposes / copies.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16
    # Release the large intermediate tensors early
    del left, right

    # --------------------------------------------------------------
    # 8) Output LayerNorm (FP16), out‑gate multiplication and final proj.
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Apply out‑gate (already sigmoid‑ed)
    out = out * out_gate
    del out_gate

    # Final linear projection back to original dimension D
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16, shape [B, N, N, D]

    # --------------------------------------------------------------
    # 9) Cast back to FP32 for the public API
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    return out
```
