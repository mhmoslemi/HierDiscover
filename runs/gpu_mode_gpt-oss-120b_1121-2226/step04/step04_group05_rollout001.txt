"""
TriMul “outgoing” update – Triton‑accelerated forward pass.

The kernel fuses the O(N³) core of the TriMul module:
    out_{b,i,j,h} = Σ_k  left_{b,i,k,h} * right_{b,j,k,h}
which is a batched matrix multiplication of shape
    (B·H, N, N)  @  (B·H, N, N)ᵀ   → (B·H, N, N)

All other operations (LayerNorms, linear projections, gating) are performed in
PyTorch with mixed‑precision (FP32 for the first LayerNorm, FP16 for the rest)
to keep memory traffic low.  The final output is returned as FP32.

The Triton kernel tiles the work with (BLOCK_M×BLOCK_N×BLOCK_K) = (128×128×64)
for large N (≥256) and (64×64×64) for smaller N.  FP16 inputs are accumulated
in FP32 (tensor‑core friendly) and written back as FP16.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) -> C (M×N)
#   A : (batch, M, K)   fp16
#   B : (batch, N, K)   fp16   (will be interpreted as transposed)
#   C : (batch, M, N)   fp16
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                 # pointers
    M, N, K,                             # matrix dimensions
    stride_ah, stride_am, stride_ak,     # A strides (batch, M, K)
    stride_bh, stride_bn, stride_bk,     # B strides (batch, N, K)
    stride_ch, stride_cm, stride_cn,     # C strides (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (flattened B·H)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tile‑wise)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ----- Load tile of A (M×K) -----
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ----- Load tile of B, transposed (K×N) -----
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # ----- Dot product (FP32 accumulation) -----
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Store result (cast to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B must be CUDA fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) as fp16.
    """
    assert A.is_cuda and B.is_cuda, "Inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Inputs must be fp16"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, f"Mismatched inner dimensions {K} vs {Kb}"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in element units
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,
        num_stages=4,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing update)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the TriMul “outgoing” module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights      : dict with the exact keys used in the reference
                               model (all float32)
              - config       : dict containing at least
                               * dim (int) – input feature dimension D
                               * hidden_dim (int) – projection dimension H
                               * nomask (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32) – the module output.
    """
    # --------------------------------------------------------------
    # 1. Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2. Input LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  float32

    # --------------------------------------------------------------
    # 3. Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4. Linear projections (bias‑free) – FP16
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)    # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w) # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 5. Optional binary mask (broadcasted)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 6. Gates (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # --------------------------------------------------------------
    # 7. Apply gates element‑wise
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8. Core N³ batched mat‑mul (outgoing update)
    # --------------------------------------------------------------
    B, N, _, H = left.shape   # B batch, N sequence length, H = hidden_dim

    # Reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Heuristic block sizes
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B·H, N, N] fp16

    # Restore layout to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 9. Output LayerNorm (FP16), out‑gate multiplication,
    #    and final projection back to the original dimension.
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # [B, N, N, D]  fp16

    # --------------------------------------------------------------
    # 10. Cast back to FP32 (API contract)
    # --------------------------------------------------------------
    return out.to(torch.float32)
