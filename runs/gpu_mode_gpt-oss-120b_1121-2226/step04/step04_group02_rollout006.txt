"""
TriMul “outgoing” forward pass – Triton‑accelerated gating and mask fusion.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → float16 for later ops).
2. Fuse the two linear projections (left/right) with their
   corresponding sigmoid‑gate linear layers into a single GEMM each:
        [proj | gate] = Linear(x, [W_proj; W_gate])        # (B,N,N,2*H)
   and split the result.
3. Compute the out‑gate (sigmoid) separately.
4. One Triton kernel performs, for every hidden element:
        left  = left  * sigmoid(left_gate_raw)  * mask?
        right = right * sigmoid(right_gate_raw) * mask?
   The sigmoid is evaluated inside the kernel, avoiding extra memory traffic.
5. Core cubic update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched matmul on tensors of shape (B,H,N,N).
6. Output LayerNorm, out‑gate multiplication and final linear projection.
7. Cast back to float32.

Only the gate‑+‑mask fusion is written in Triton (requirement). All heavy
linear / batched‑matmul work uses cuBLAS via torch.nn.functional.linear and
torch.matmul, which on H100 are Tensor‑core bound and very fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr,          # FP16,   [total_elems]
    right_ptr,         # FP16,   [total_elems]
    left_gate_raw_ptr, # FP16,   [total_elems] – pre‑sigmoid
    right_gate_raw_ptr,# FP16,   [total_elems] – pre‑sigmoid
    mask_ptr,          # FP16,   [total_masks] (optional)
    total_elems: tl.int32,
    hidden_dim: tl.int32,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """In‑place fusion:
       left  = left  * sigmoid(left_gate_raw)  * (mask if present)
       right = right * sigmoid(right_gate_raw) * (mask if present)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load tensors
    left = tl.load(left_ptr + offs, mask=mask, other=0.0)
    right = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lgate_raw = tl.load(left_gate_raw_ptr + offs, mask=mask, other=0.0)
    rgate_raw = tl.load(right_gate_raw_ptr + offs, mask=mask, other=0.0)

    # Sigmoid (1/(1+exp(-x))) – computed in fp16
    lgate = 1.0 / (1.0 + tl.exp(-lgate_raw))
    rgate = 1.0 / (1.0 + tl.exp(-rgate_raw))

    if HAS_MASK:
        # One mask value per (b,i,k) – broadcast over hidden_dim
        mask_idx = offs // hidden_dim            # integer division
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
        left = left * lgate * m
        right = right * rgate * m
    else:
        left = left * lgate
        right = right * rgate

    # Store results back in‑place
    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def _apply_gates_and_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate_raw: torch.Tensor,
    right_gate_raw: torch.Tensor,
    mask: torch.Tensor | None,
    hidden_dim: int,
):
    """Launch the Triton kernel that fuses sigmoid‑gate, optional mask
    and element‑wise multiplication. Operates in‑place on ``left`` and ``right``."""
    total = left.numel()
    BLOCK = 1024  # good trade‑off between launch overhead & occupancy
    grid = ( (total + BLOCK - 1) // BLOCK, )

    if mask is None:
        dummy_mask = torch.empty(0, dtype=torch.float16, device=left.device)
        HAS_MASK = 0
    else:
        dummy_mask = mask.view(-1)  # flatten to 1‑D
        HAS_MASK = 1

    _gate_mask_kernel[grid](
        left,
        right,
        left_gate_raw,
        right_gate_raw,
        dummy_mask,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
        HAS_MASK=HAS_MASK,
        num_warps=4,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights       : dict of model parameters (float32)
              - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (high‑precision FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 4) Linear projections + gate pre‑sigmoid (fused per side)
    # --------------------------------------------------------------
    # left side: concatenate projection and gate weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    left_cat_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2*H, D)

    left_cat = F.linear(x_f16, left_cat_w)  # [B,N,N,2*H] fp16
    left = left_cat[..., :hidden_dim]                # projection
    left_gate_raw = left_cat[..., hidden_dim:]       # pre‑sigmoid gate

    # right side: same pattern
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_cat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    right_cat = F.linear(x_f16, right_cat_w)  # [B,N,N,2*H] fp16
    right = right_cat[..., :hidden_dim]
    right_gate_raw = right_cat[..., hidden_dim:]

    # --------------------------------------------------------------
    # 5) Out‑gate (separate sigmoid)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # fp16, shape [B,N,N,H]

    # --------------------------------------------------------------
    # 6) Fuse mask (if any) and sigmoid‑gate in a Triton kernel
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).contiguous()
    else:
        mask_f16 = None

    _apply_gates_and_mask(
        left,
        right,
        left_gate_raw,
        right_gate_raw,
        mask_f16,
        hidden_dim,
    )   # left/right are modified in‑place

    # --------------------------------------------------------------
    # 7) Core cubic update (batched matmul)
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # --------------------------------------------------------------
    B, N, _, H = left.shape

    # Shape to (B, H, N, N) – no extra copies, just a view with permuted stride
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2) # (B, H, N, N)

    # Batched matmul: (B,H,N,N) @ (B,H,N,N)^T -> (B,H,N,N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # Back to [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate multiplication and final linear
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # apply out‑gate (fp16)

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # [B, N, N, D]  fp16

    # --------------------------------------------------------------
    # 9) Cast back to FP32 for API compliance
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    return out
```
