"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Input LayerNorm (FP32 → FP16).
2. Fuse **all five** linear projections (left, left‑gate, right, right‑gate,
   out‑gate) into a single GEMM:   X_flat @ W_concatᵀ.
   The result is sliced into the five tensors (all FP16).
3. Apply the three sigmoid gates (left‑gate, right‑gate, out‑gate).
4. Fuse mask multiplication and gate‑post‑scale for *left* and *right*
   with a tiny Triton kernel (only required when a mask is present).
5. Core N³ update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication L @ Rᵀ for each head *h*.
   Implemented with a single torch.bmm on the strided‑batched tensors
   (still FP16 → FP16, accumulation in FP32).
6. Output LayerNorm (FP16) → element‑wise multiply by *out_gate* (FP16).
7. Final linear projection back to *dim* (FP16) and cast the result to FP32.

Only the mask‑gate fusion is performed in a custom Triton kernel; the rest
uses highly‑optimized Torch/cuBLAS kernels while staying in half‑precision
to minimise memory traffic.  The fused projection step removes redundant
loads of the large tensor X, dramatically cutting bandwidth and yielding
sub‑millisecond runtimes on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate multiplication (in‑place)
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,          # left tensors
    right_ptr, right_gate_ptr,        # right tensors
    mask_ptr,                         # broadcasted mask (same shape as left/right)
    total_elems: tl.int32,            # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,         # threads per program
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values (out‑of‑bounds loads are masked)
    l_val   = tl.load(left_ptr   + offs, mask=mask, other=0.0)
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_val   = tl.load(right_ptr  + offs, mask=mask, other=0.0)
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m_val   = tl.load(mask_ptr   + offs, mask=mask, other=1.0)

    # Fuse three element‑wise multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32, on CUDA)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32, on CUDA)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for the heavy work
    # ------------------------------------------------------------------
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse all five linear projections into one GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    total_items = B * N * N

    # flatten to (BNN, dim)
    x_flat = x.view(total_items, dim).contiguous()

    # concatenate the five weight matrices (all bias‑free)
    # order: left, left_gate, right, right_gate, out_gate
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # single fused linear -> (total_items, 5*hidden_dim)
    fused = F.linear(x_flat, w_concat)  # fp16
    fused = fused.view(B, N, N, 5 * hidden_dim)

    # split
    left      = fused[..., 0 * hidden_dim : 1 * hidden_dim]
    left_gate = fused[..., 1 * hidden_dim : 2 * hidden_dim]
    right     = fused[..., 2 * hidden_dim : 3 * hidden_dim]
    right_gate= fused[..., 3 * hidden_dim : 4 * hidden_dim]
    out_gate  = fused[..., 4 * hidden_dim : 5 * hidden_dim]

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5) Optional mask + gate fusion (via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, hidden_dim] (broadcasted)
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).repeat(1, 1, 1, hidden_dim).contiguous()

        # ensure contiguous views for 1‑D memory layout
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_fp16 = mask_fp16.contiguous()

        total = left.numel()                     # B·N·N·H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_fp16.data_ptr(),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just apply the gates
        left  = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core N³ batched matrix multiplication (L @ Rᵀ)
    #    All work stays in fp16; cuBLAS accumulates in fp32 internally.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B*H, N, N) for a strided‑batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N) fp16
    # reshape back to [B, N, N, H] (still fp16)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (FP16), out‑gate multiplication, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    out = out * out_gate  # element‑wise, fp16

    # final linear projection back to dim (still fp16)
    out_flat = out.view(total_items, hidden_dim).contiguous()
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out_flat, to_out_w)  # (BNN, dim) fp16
    out_proj = out_proj.view(B, N, N, dim).to(torch.float32)

    return out_proj
