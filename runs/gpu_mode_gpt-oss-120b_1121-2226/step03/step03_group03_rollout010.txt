"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The module implements the “outgoing” triangle multiplicative update from
AlphaFold3.  The heavy N³ reduction is performed with a custom Triton
batched GEMM kernel:

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}

where `left` and `right` are the linearly projected and gated pairwise
representations.  The kernel computes, for each hidden channel `h`,
a matrix‑multiply `L @ Rᵀ` (both of shape N×N) using fp16 inputs and
fp32 accumulation (Tensor‑Core friendly).  All remaining operations
(LayerNorm, linear projections, gating) are performed with PyTorch in
mixed‑precision.

The design goals are:
* correctness w.r.t. the reference PyTorch implementation,
* pure‑CUDA execution on an H100 (Triton 3.3.1),
* a small Triton kernel is used (the batched GEMM) as required,
* runtime is kept low by choosing tile sizes adaptively to the sequence
  length.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) * B (K×N) → C (M×N)
#   – A, B are fp16, C is fp32 (fp16·fp16 → fp32 accumulation)
#   – The third launch axis indexes the batch dimension (B·H)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bk, stride_bn,    # B strides: batch, K, N
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One program instance (i.e. one block) computes a tile C[pid_b,
    offs_m:offs_m+BLOCK_M, offs_n:offs_n+BLOCK_N] of the result.
    """
    # ---------------------------------------------------------------
    # 1) program ids
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # ---------------------------------------------------------------
    # 2) tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # 3) loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (already K×N, i.e. transposed on‑the‑fly): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # fused dot‑product (fp16·fp16 → fp32)
        acc += tl.dot(a_tile, b_tile)

    # ---------------------------------------------------------------
    # 4) write back the tile
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, K, N).
    Returns C of shape (batch, M, N) in fp32.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must reside on CUDA device"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16 inputs"

    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, f"Incompatible inner dimensions: {K} vs {K2}"

    # Allocate output
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides (in elements, not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()   # note: K‑stride then N‑stride
    stride_ch, stride_cm, stride_cn = C.stride()

    # Grid: (M‑tiles, N‑tiles, batch)
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    # Heuristic for number of warps (one warp per 32 rows of the tile)
    num_warps = max(1, BLOCK_M // 32)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,                # double‑buffered loads
    )
    return C


def _choose_tile_sizes(seqlen: int):
    """Select tile sizes that give good occupancy on an H100."""
    if seqlen >= 1024:
        return 256, 256, 64
    if seqlen >= 512:
        return 128, 128, 64
    if seqlen >= 256:
        return 64, 64, 32
    # fallback
    return 64, 64, 32


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) – may be None when nomask=True
            - weights: dict of model parameters (float32)
            - config: dict containing at least "dim", "hidden_dim",
                      optional "nomask"
    Returns:
        Tensor of shape [B, N, N, D] (float32) – the module output.
    """
    # ---------------------------------------------------------------
    # 0) unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # Cast to fp16 for the linear & gating steps (better bandwidth)
    x_f16 = x.to(torch.float16)

    # ---------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H]  fp16

    # ---------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    if (not nomask) and mask is not None:
        # mask ∈ {0,1}; expand to (B,N,N,1) then broadcast
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f16
        right = right * mask_f16

    # ---------------------------------------------------------------
    # 4) Gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # fp16 (B,N,N,H)

    # ---------------------------------------------------------------
    # 5) Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 6) Core N³ batched mat‑mul: left @ rightᵀ  (per hidden channel)
    B, N, _, H = left.shape  # batch size, sequence length, hidden dim

    # Reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Choose tile sizes based on sequence length
    BLOCK_M, BLOCK_N, BLOCK_K = _choose_tile_sizes(N)

    # Batched GEMM: left @ rightᵀ
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )   # [B·H, N, N]  (float32)

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ---------------------------------------------------------------
    # 7) Output LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    # Multiply by the out‑gate (broadcast automatically to fp32)
    out = out * out_gate

    # ---------------------------------------------------------------
    # 8) Final linear projection back to original dimension D
    out = F.linear(out, weights["to_out.weight"])

    return out
