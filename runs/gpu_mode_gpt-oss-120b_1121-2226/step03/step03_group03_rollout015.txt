"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The module processes a 4‑D tensor x ∈ ℝ^{B×N×N×D} and returns a tensor of the same
shape.  The heavy part of the computation is the “outgoing” update

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} · right_{b,j,k,h}      (1)

which is a batched matrix‑multiply L @ Rᵀ for every hidden channel h.
The implementation follows these steps

1️⃣  Layer‑norm over the last dimension (D).
2️⃣  Fuse the five bias‑free linear layers (left_proj, right_proj,
    left_gate, right_gate, out_gate) into a single GEMM:
    x̂ = x.view(-1,D)   (shape = B·N·N × D)
    Ŵ = [W_left_proj; W_right_proj; W_left_gate; W_right_gate; W_out_gate]
    ŷ = x̂ @ Ŵᵀ      → shape B·N·N × (5·H)
    Slice ŷ to obtain left, right, left_gate, right_gate, out_gate.
3️⃣  Optional binary mask M ∈ {0,1} (broadcast over H) applied to left/right.
4️⃣  Gate the projected tensors with sigmoid outputs.
5️⃣  Core N³ operation – a custom Triton kernel performs a batched GEMM
    (batch = B·H, M = N, K = N, N = N) in fp16·fp16 → fp32.
    The kernel tiles the problem (BLOCK_M × BLOCK_N × BLOCK_K) and
    uses tensor‑core dot products with double‑buffered loads.
6️⃣  Reshape the result back to (B,N,N,H).
7️⃣  Layer‑norm over the hidden dimension, element‑wise multiplication by out‑gate,
    and a final linear projection back to D.

Only the N³ fragment is written in Triton; the surrounding linear layers,
gating, and normalisation use PyTorch (which already runs on cuBLAS/​cuDNN).
Dynamic tile sizes are chosen from the sequence length to keep shared‑memory
use low while maximising occupancy on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) * B (K×N) → C (M×N)
#   – A, B are fp16, C is fp32 (fp16·fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,    # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile (BLOCK_M × BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (BLOCK_K × BLOCK_N) – B is stored as K‑major, N‑minor
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # fused dot‑product (fp16·fp16 → fp32)
        acc += tl.dot(a_tile, b_tile)

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
    num_warps: int = 8,
) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, K, N).
    Returns C with shape (batch, M, N) in fp32.
    """
    assert A.is_cuda and B.is_cuda, "tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) – may be None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------- unpack ------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------- 1) LayerNorm -----------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # [B, N, N, D]  float32

    # ----------------------------------------------------
    # 2️⃣ Fuse the five linear layers into a single GEMM
    # ----------------------------------------------------
    # cast once to fp16 (half‑precision)
    x_f16 = x.half()                     # [B, N, N, D]

    # convert all five weights to fp16 and concatenate
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    combined_weight = torch.cat(
        [left_proj_w,
         right_proj_w,
         left_gate_w,
         right_gate_w,
         out_gate_w],
        dim=0,
    )  # (5*H, D), fp16, contiguous

    B, N, _, _ = x_f16.shape
    x_flat = x_f16.view(-1, dim)                # (B·N·N, D)

    # fused linear → (B·N·N, 5*H)
    fused = torch.nn.functional.linear(x_flat, combined_weight)  # fp16
    fused = fused.view(B, N, N, 5 * hidden_dim)                 # [B, N, N, 5*H]

    # slice the fused result
    h = hidden_dim
    left      = fused[..., 0:h]               # (B,N,N,H)
    right     = fused[..., h:2*h]
    left_gate = torch.sigmoid(fused[..., 2*h:3*h])
    right_gate= torch.sigmoid(fused[..., 3*h:4*h])
    out_gate  = torch.sigmoid(fused[..., 4*h:5*h])

    # ------------------- 3) optional mask -----------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # ------------------- 4) apply gates -------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------- 5) core batched GEMM (outgoing) --------------------
    # reshape tensors to (B*H, N, N); right needs a transpose on the last two axes
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B·H,N,N) fp16
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)  # (B·H,N,N) fp16 (already transposed)

    # choose tile sizes adaptively
    if N <= 128:
        BLOCK_M = BLOCK_N = 64
        num_warps = 4
    elif N <= 256:
        BLOCK_M = BLOCK_N = 128
        num_warps = 8
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        num_warps = 8
    else:  # up to 1024
        BLOCK_M = BLOCK_N = 256
        num_warps = 16
    BLOCK_K = 64 if N >= 64 else 32

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )   # (B·H, N, N) fp32

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------- 6) output LayerNorm & final projection -------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    out = out * out_gate   # out_gate (fp16) will be up‑cast automatically

    # final linear back to original dimension D
    out = F.linear(out, weights["to_out.weight"])   # fp32, shape [B,N,N,D]

    return out
