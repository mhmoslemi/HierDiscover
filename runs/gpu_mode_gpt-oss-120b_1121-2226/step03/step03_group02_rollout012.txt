    device = input_tensor.device

    hidden_dim = config['hidden_dim']

    norm_bias = weights['norm.bias'].to(torch.float16)
    left_proj_weight = weights['left_proj.weight'].to(torch.float16)
    right_proj_weight = weights['right_proj.weight'].to(torch.float16)

    right_gate_weight = weights['right_gate.weight'].to(torch.float16)
    out_gate_weight = weights['out_gate.weight'].to(torch.float16)

    to_out_norm_bias = weights['to_out_norm.bias'].to(torch.float16)
    to_out_weight = weights['to_out.weight'].to(torch.float16)

import triton
import triton.language as tl

# Triton kernel: Apply mask and per‑channel gates in‑place.
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,

    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)

def custom_kernel(data):

    Triton‑accelerated forward pass for the “outgoing” TriMul op.
    The implementation fuses the two projection Linear layers into a single
    GEMM, fuses mask‑and‑gate application via a tiny Triton kernel,
    and uses half‑precision throughout (final result cast back to float32).
    """

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # --------------------------------------------------------------------
    # 0️⃣ Cast everything to half‑precision (fp16) upfront.
    # --------------------------------------------------------------------
    # Input
    x = input_tensor.to(torch.float16)

    # Layer‑norm parameters for the first LN
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    # --------------------------------------------------------------------
    # 1️⃣ First Layer‑Norm (over the feature dimension D)
    # --------------------------------------------------------------------
    x = F.layer_norm(x, normalized_shape=(dim,), weight=norm_weight,
                     bias=norm_bias, eps=eps)   # [B, N, N, D] fp16

    # --------------------------------------------------------------------
    # 2️⃣ One‑shot linear projection that produces left/right and all three gates.
    #    Shape of the weight matrix: (5*H, D)
    # --------------------------------------------------------------------
    # Gather weight matrices in the exact order expected later.
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],    # (H, D)
            weights["right_proj.weight"],   # (H, D)
            weights["left_gate.weight"],    # (H, D)
            weights["right_gate.weight"],   # (H, D)
            weights["out_gate.weight"],     # (H, D)
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, D)

    # Perform the fused GEMM.
    proj = F.linear(x, proj_weight)       # [B, N, N, 5*H] fp16

    # Split the fused tensor.
    left, right, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=-1
    )  # each [B,N,N,H]

    # Apply sigmoid to the three gate vectors.
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------------
    # 3️⃣ Optional mask + gate fusion via a tiny Triton kernel (in‑place).
    # --------------------------------------------------------------------

        # mask: [B, N, N]  →  fp16 scalar per position.
        mask_f16 = mask.to(torch.float16).contiguous()

        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total_elements = left.numel()                     # B*N*N*H
        BLOCK = 1024
        grid = ( (total_elements + BLOCK - 1) // BLOCK, )
        _apply_mask_gate_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_f16,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates are already applied inside the kernel.
    else:
        # No mask – just gate the projections (cheap element‑wise ops).
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------------
    # 4️⃣ Core N³ update – batched matrix multiplication.
    # --------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # Reshape to (B*H, N, N) for a single batched GEMM.
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # fp16

    # (B*H, N, N) @ (B*H, N, N)^T  →  (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # Reshape back to the original 4‑D view.
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # --------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, out‑gate, final linear projection.
    # --------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)          # fp16

    # Apply the out‑gate (already sigmoid‑ed) – element‑wise.
    out = out * out_gate

    # Final linear back to the original feature dimension.
    to_out_weight = weights["to_out.weight"].to(torch.float16)  # (dim, H)
    out = F.linear(out, to_out_weight)                         # fp16, shape [B,N,N,dim]

    # Cast to float32 for the public API contract.
    return out.to(torch.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _apply_mask_gate_kernel(left_ptr, right_ptr,
                            left_gate_ptr, right_gate_ptr,
                            mask_ptr,
                            total_elements: tl.constexpr,
                            hidden_dim: tl.constexpr,
                            BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors (masked loads)
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask)

    # mask has shape (total_elements/hidden_dim)
    mask_idx = offs // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply per‑channel gates and mask
    left = left * l_gate * mask_val
    right = right * r_gate * mask_val

    tl.store(left_ptr + offs, left, mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.
    The implementation fuses the two linear projections into a single
    mat‑mul, uses a tiny Triton kernel for mask‑gate fusion, and keeps
    the expensive N³ update on the GPU (fp16 mat‑mul).  Final result
    is returned as float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ----------------- 0️⃣ FP32 weights for FP32‑only ops -----------------
    norm_weight_fp32 = weights["norm.weight"]
    norm_bias_fp32 = weights["norm.bias"]
    to_out_norm_weight_fp32 = weights["to_out_norm.weight"]
    to_out_norm_bias_fp32 = weights["to_out_norm.bias"]
    to_out_weight_fp32 = weights["to_out.weight"]

    # ----------------- 1️⃣ First LayerNorm (FP32) -----------------
    # keep input in float32 for exactness
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight_fp32,
        bias=norm_bias_fp32,
        eps=eps,
    )   # [B, N, N, D]  FP32

    # ----------------- 2️⃣ Cast to FP16 for the heavy GEMM -----------------
    x_fp16 = x.to(torch.float16)

    # ----------------- 3️⃣ Fuse the two projections + 3 gates -----------------
    # Gather weights; all of them can be kept in half‑precision.
    proj_weight_fp16 = torch.cat(
        [
            weights["left_proj.weight"],    # (H, D)
            weights["right_proj.weight"],   # (H, D)
            weights["left_gate.weight"],    # (H, D)
            weights["right_gate.weight"],   # (H, D)
            weights["out_gate.weight"],     # (H, D)
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, D)

    # Single GEMM: (B*N*N, D) @ (5*H, D)^T → (B*N*N, 5*H)
    proj = F.linear(x_fp16, proj_weight_fp16)   # [B, N, N, 5*H]  fp16

    # Split back into tensors.
    left, right, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=-1
    )   # each [B, N, N, H] fp16

    # Apply sigmoid to gates (still fp16).
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ----------------- 4️⃣ Optional mask‑&‑gate fusion (Triton) -----------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).contiguous()  # [B, N, N]

        # Ensure tensors are contiguous for flattened view.
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                     # B*N*N*H
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _apply_mask_gate_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp16,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates have been applied in‑place, nothing else to do.
    else:
        # No mask – just apply the per‑channel gates (cheap element‑wise ops).
        left = left * left_gate
        right = right * right_gate

    # ----------------- 5️⃣ Core N³ update – batched fp16 mat‑mul -----------------
    B, N, _, H = left.shape
    # Reshape for batch‑GEMM: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # fp16

    # (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # Back‑to four‑dimensional tensor.
    out_fp16 = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # ----------------- 6️⃣ Output Layer‑Norm (FP32) -----------------
    # Cast to FP32 for the final layer‑norm to match the reference implementation.
    out_fp32 = out_fp16.to(torch.float32)

    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight_fp32,
        bias=to_out_norm_bias_fp32,
        eps=eps,
    )   # FP32

    # Apply the out‑gate (broadcast to FP32).
    out_norm = out_norm * out_gate.to(torch.float32)

    # ----------------- 7️⃣ Final projection back to `dim` (FP32) -----------------
    out = F.linear(out_norm, to_out_weight_fp32)   # [B,N,N,dim] FP32

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + per‑channel gate application.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,               # fp16*  (flattened left tensor)
    right_ptr,              # fp16*  (flattened right tensor)
    left_gate_ptr,          # fp16*  (flattened left‑gate tensor)
    right_gate_ptr,         # fp16*  (flattened right‑gate tensor)
    mask_ptr,               # fp16*  (flattened mask, no hidden dim)
    total_elements: tl.constexpr,   # B*N*N*H
    hidden_dim:    tl.constexpr,   # H   (known at compile time)
    BLOCK_SIZE:    tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # Load the data (masked loads avoid OOB reads)
    left      = tl.load(left_ptr      + offs, mask=active)
    right     = tl.load(right_ptr     + offs, mask=active)
    l_gate    = tl.load(left_gate_ptr + offs, mask=active)
    r_gate    = tl.load(right_gate_ptr+ offs, mask=active)

    # mask has shape (B,N,N); broadcast over the hidden dimension
    mask_idx = offs // hidden_dim               # index into mask view
    mask_val = tl.load(mask_ptr + mask_idx, mask=active)

    # Apply gates and mask in‑place
    left  = left  * l_gate * mask_val
    right = right * r_gate * mask_val

    tl.store(left_ptr  + offs, left,  mask=active)
    tl.store(right_ptr + offs, right, mask=active)

# Entry point required by the harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Strategy
    --------
    1️⃣  FP32 Layer‑Norm over the input feature dimension.
    2️⃣  Cast to FP16 and fuse the two projections (left/right) and the three
        gate linear layers into a *single* GEMM (shape (5·H, D)).
    3️⃣  Apply sigmoid to the three gate tensors.
    4️⃣  If a mask is present, fuse mask‑and‑gate multiplication via a tiny
        Triton kernel (in‑place on left/right).
    5️⃣  Core O(N³) update: batched half‑precision mat‑mul
        (B·H, N, N) @ (B·H, N, N)ᵀ using torch.bmm.
    6️⃣  Cast back to FP32, apply the output Layer‑Norm, the out‑gate,
        and the final linear projection (all in FP32 to retain reference‑level
        numerical fidelity).
    7️⃣  Return the result in float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # input feature dimension D
    hidden_dim = config["hidden_dim"]       # projection dimension H
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------
    # 0️⃣ FP32 Layer‑Norm on the raw input
    # ------------------------------------------------------------
    norm_w_fp32 = weights["norm.weight"]
    norm_b_fp32 = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w_fp32,
        bias=norm_b_fp32,
        eps=eps,
    )                     # [B, N, N, D]  float32

    # ------------------------------------------------------------
    # 1️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # ------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + three gates into one GEMM
    # ------------------------------------------------------------
    # Stack the five weight matrices (left, right, left_gate, right_gate, out_gate)
    proj_weight_fp16 = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5·H, D)

    # Single linear pass: (B·N·N, D) @ (5·H, D)ᵀ → (B·N·N, 5·H)
    proj = F.linear(x_fp16, proj_weight_fp16)   # [B, N, N, 5·H]  fp16

    left, right, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=-1
    )                                            # each [B, N, N, H] fp16

    # ------------------------------------------------------------
    # 3️⃣ Sigmoid the gate tensors (still fp16)
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------
    # 4️⃣ Optional mask & gate fusion via Triton (in‑place)
    # ------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).contiguous()   # [B, N, N]

        # Ensure contiguous layout for flattened pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                              # B·N·N·H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _apply_mask_gate_kernel[grid](
            left,
            right,
            left_gate,
            right_gate,
            mask_fp16,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # The kernel has already applied both per‑channel gates and the mask.
    else:
        # No mask – just apply the per‑channel gates (cheap element‑wise ops)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------
    # 5️⃣ Core N³ update – batched fp16 matrix multiplication
    # ------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # fp16

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # Restore the original 4‑D layout
    out_fp16 = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                            # [B, N, N, H] fp16

    # ------------------------------------------------------------
    # 6️⃣ Output Layer‑Norm, out‑gate, final projection (FP32 for fidelity)
    # ------------------------------------------------------------
    # Cast back to float32 for the remaining (smaller) ops
    out_fp32 = out_fp16.to(torch.float32)

    to_out_norm_w_fp32 = weights["to_out_norm.weight"]
    to_out_norm_b_fp32 = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w_fp32,
        bias=to_out_norm_b_fp32,
        eps=eps,
    )                                            # float32

    # Apply the out‑gate (cast to fp32 first)
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear back to the original feature dimension
    to_out_w_fp32 = weights["to_out.weight"]     # (dim, H)
    out = F.linear(out_norm, to_out_w_fp32)     # [B, N, N, dim] float32

    return out
