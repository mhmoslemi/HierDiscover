def _apply_mask_gate_and_sigmoid_kernel(
    left_ptr,           # fp16 * (size = total_elements)
    right_ptr,
    left_gate_ptr,      # fp16 raw (pre-sigmoid)
    right_gate_ptr,
    mask_ptr,           # fp16 * (B*N*N) optional
    total_elements,
    hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements
    # load
    l = tl.load(left_ptr + offs, mask=in_range, other=0.0)
    r = tl.load(right_ptr + offs, mask=in_range, other=0.0)
    lg_raw = tl.load(left_gate_ptr + offs, mask=in_range, other=0.0)
    rg_raw = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)
    # sigmoid:
    lg = 1.0 / (1.0 + tl.exp(-lg_raw))
    rg = 1.0 / (1.0 + tl.exp(-rg_raw))
    # mask
    if HAS_MASK:
        # calculate mask index
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)
    # apply
    l = l * lg * m
    r = r * rg * m
    # store
    tl.store(left_ptr + offs, l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)
```

    bs, seq_len, _, dim = input_tensor.shape # seq_len = N

    hidden_dim = config["hidden_dim"]

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32
    x_h = x.half()   # fp16

    # 2. Concatenated linear for all projections & gates
    # Gather weights (float32) and cast to half
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    # weight shapes: (hidden_dim, dim)
    # cat along output dim (dim 0)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    ).half()   # shape (5*hidden_dim, dim)

    # Compute combined linear
    combined = torch.nn.functional.linear(x_h, combined_weight)  # shape (B,N,N,5*H) fp16

    out_gate = torch.sigmoid(out_gate_raw)  # fp16

    HAS_MASK = 1 if mask is not None else 0
    if HAS_MASK:
        mask_h = mask.to(dtype=left_proj.dtype).contiguous()
        mask_flat = mask_h.view(-1)  # (B*N*N)
    else:

    total_elements = left_proj.numel()
    left_flat = left_proj.view(-1)
    right_flat = right_proj.view(-1)
    left_gate_raw_flat = left_gate_raw.view(-1)
    right_gate_raw_flat = right_gate_raw.view(-1)

"""
TriMul (outgoing) – highly‑optimized forward pass

Algorithm
---------
1️⃣ Layer‑norm over the last dimension (`dim`) of the input (FP32 → FP16).
2️⃣ Fuse **all five** linear layers (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a **single** matmul
   `x @ W_cat`.  This reads the huge [B,N,N,dim] tensor only once.
3️⃣ Split the concatenated result into the five tensors.
4️⃣ Apply sigmoid to the three gate tensors **and** fuse the optional
   mask, the left/right gates and the projections with a tiny
   Triton kernel (in‑place, FP16).
5️⃣ Convert the gated `left`/`right` tensors to the layout required
   for a batched GEMM:
       left  (B,N,N,H) → (B*H,N,N)   (row‑major)
       right (B,N,N,H) → (B*H,N,N)   (row‑major, transposed N‑axes)
6️⃣ Batched matrix multiplication (`torch.bmm`) – this is the
   N³ work and is executed by cuBLAS Tensor‑cores.
7️⃣ Second Layer‑norm over the hidden dimension (FP32).
8️⃣ Multiply by the `out_gate` (sigmoid) and a final FP32 linear
   projection back to `dim`.

Only the element‑wise mask‑gate‑sigmoid fusion is written in Triton;
the remaining heavy lifting is delegated to highly‑optimized PyTorch
primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel:  (left, right) ← (left * sigmoid(left_gate) * mask,
#                                   right * sigmoid(right_gate) * mask)
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_and_sigmoid_kernel(
    left_ptr,           # fp16 * (B·N·N·H)
    right_ptr,          # fp16 * (B·N·N·H)
    left_gate_ptr,      # fp16 * (B·N·N·H)  (pre‑sigmoid)
    right_gate_ptr,     # fp16 * (B·N·N·H)  (pre‑sigmoid)
    mask_ptr,           # fp16 * (B·N·N)    (optional, broadcast over H)
    total_elements,     # i64  = B·N·N·H
    hidden_dim,         # i64  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # ------------------------------------------------------------------
    # Load tensors (fp16)
    # ------------------------------------------------------------------
    l  = tl.load(left_ptr + offs,        mask=in_range, other=0.0)
    r  = tl.load(right_ptr + offs,       mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr + offs,   mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs,  mask=in_range, other=0.0)

    # ------------------------------------------------------------------
    # Sigmoid (in‑place)
    #   sigmoid(x) = 1 / (1 + exp(-x))
    # ------------------------------------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # each consecutive group of `hidden_dim` elements share the same mask entry
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)   # fp16
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Fuse:  left = left * sigmoid(left_gate) * mask
    #        right = right * sigmoid(right_gate) * mask
    # ------------------------------------------------------------------
    l = l * lg * m
    r = r * rg * m

    # ------------------------------------------------------------------
    # Store results back in‑place
    # ------------------------------------------------------------------
    tl.store(left_ptr  + offs, l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]        # called H in the paper

    device = input_tensor.device

    # --------------------------------------------------------------
    # 2️⃣ first LayerNorm (float32 → FP16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                          # [B, N, N, dim]  float32
    x_h = x.half()             # [B, N, N, dim]  float16

    # --------------------------------------------------------------
    # 3️⃣ fuse all linear layers into a single matmul
    #    (shape: dim → 5·hidden_dim)
    # --------------------------------------------------------------
    # concatenate weight matrices along the output dimension
    cat_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                # (5·H, dim)
    ).half().to(device)       # move to the correct device & cast to fp16

    # combined linear: shape -> [B, N, N, 5·H] (fp16)
    combined = F.linear(x_h, cat_weight)

    # split back into the five components
    left_proj, left_gate_raw, right_proj, right_gate_raw, out_gate_raw = combined.split(
        hidden_dim, dim=3
    )
    # free the large intermediate immediately
    del combined, cat_weight, x, x_h

    # --------------------------------------------------------------
    # 4️⃣ compute the output gate (sigmoid) – needed after GEMM
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)   # fp16, shape [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ fuse optional mask + left/right gates with a Triton kernel
    # --------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None) and mask.numel() > 0 else 0
    if HAS_MASK:
        mask_h = mask.to(dtype=left_proj.dtype).contiguous()   # fp16
        mask_flat = mask_h.view(-1)                            # (B·N·N)
    else:
        # dummy tensor – not accessed when HAS_MASK==0
        mask_flat = torch.empty(0, dtype=left_proj.dtype, device=device)

    # flatten everything to 1‑D for the kernel
    total_elements = left_proj.numel()               # B·N·N·H
    left_flat   = left_proj.view(-1)
    right_flat  = right_proj.view(-1)
    left_g_flat = left_gate_raw.view(-1)
    right_g_flat = right_gate_raw.view(-1)

    BLOCK_SIZE = 1024  # must be a multiple of 32 and ≤ max threads per block
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _apply_mask_gate_and_sigmoid_kernel[grid](
        left_flat,
        right_flat,
        left_g_flat,
        right_g_flat,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=HAS_MASK,
    )
    # The tensors `left_proj` and `right_proj` have been updated in‑place.
    del left_gate_raw, right_gate_raw, left_g_flat, right_g_flat, mask_h, mask_flat

    # --------------------------------------------------------------
    # 6️⃣ prepare tensors for the cubic contraction (batched GEMM)
    #    left: (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    #    right: (B, N, N, H) → (B, H, N, N) (swap the inner N‑axes) → (B·H, N, N)
    # --------------------------------------------------------------
    left_perm  = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_proj.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)  fp16
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)  fp16

    # --------------------------------------------------------------
    # 7️⃣ batched matrix multiplication (Tensor‑core accelerated)
    # --------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)    # (B·H, N, N)  fp16

    # restore original layout: [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ second LayerNorm over the hidden dimension (float32)
    # --------------------------------------------------------------
    out_fp32 = out.float()
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H] float32

    # --------------------------------------------------------------
    # 9️⃣ apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_gate_fp32 = out_gate.float()
    out = out_norm * out_gate_fp32
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] float32

    return out
```
