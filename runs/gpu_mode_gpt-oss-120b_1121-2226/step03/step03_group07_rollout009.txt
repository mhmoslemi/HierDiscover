@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2) # flattened batch = B * H
    pid_m = tl.program_id(0)   # row tile i
    pid_n = tl.program_id(1)   # column tile j
    
    # decode batch and hidden
    b = pid_batch // H
    h = pid_batch % H
    
    # offsets for rows and cols
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # base pointers for left and right for this (b,h)
    left_base = left_ptr + b * stride_left_batch + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    
    # compute row and column pointers:
    # left[b,i,k,h] = left_base + i * stride_left_i + k * stride_left_k
    # right[b,j,k,h] = right_base + j * stride_right_i + k * stride_right_k
    a_ptrs = left_base + offs_i[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_base + offs_j[None, :] * stride_right_i + tl.arange(0, BLOCK_K)[:, None] * stride_right_k
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_i[:, None] < N) & (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < N),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < N) & (offs_j[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)  # fp16->fp32 accumulation

        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # write out result
    out_base = out_ptr + b * stride_out_batch + h * stride_out_h
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    mask_c = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣  Input LayerNorm (FP32) → `x`.
2️⃣  Cast `x` to FP16 for all subsequent matmul‑heavy ops.
3️⃣  Six linear layers (two projections, three gates, final output) are
    executed with PyTorch’s highly‑optimised GEMM kernels (half‑precision).
4️⃣  Optional mask: a tiny Triton kernel casts the FP32 mask to FP16 and
    broadcasts it over the hidden dimension.
5️⃣  Gating (sigmoid) is performed in FP16.
6️⃣  Core N³ operation (outgoing TriMul) is a batched matrix‑multiply
    `L @ Rᵀ` per hidden channel.  The custom Triton kernel treats the
    combined batch dimension `B·H` as a single program‑id dimension,
    eliminates any materialised copies (no view/reshape), and
    accumulates in FP32 while keeping the arithmetic in FP16 – a perfect
    fit for H100 Tensor‑Cores.
7️⃣  Output LayerNorm over the hidden channel, element‑wise multiplication
    by the output gate, and a final FP32 linear projection back to `dim`.
8️⃣  The function `custom_kernel` follows the required signature and works
    for all test configurations (different batch sizes, sequence lengths,
    hidden dimensions, and optional masks).

The implementation fuses the most expensive step (the N³ reduction) into a
single Triton kernel and avoids the costly `view(...).contiguous()` copy
used in the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# tiny kernel: cast FP32 mask → FP16 (used only when a mask exists)
# --------------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,                # fp32 source
    dst_ptr,                # fp16 destination
    numel,                  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)


# --------------------------------------------------------------
# core outgoing TriMul kernel:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#   – left / right: FP16
#   – out          : FP32 (accumulated)
# --------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # base pointers
    B, N, K, H,                            # dimensions (K == N)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------
    # 0️⃣  program ids
    # ------------------------------------------------------
    pid_m = tl.program_id(0)            # row block   (i)
    pid_n = tl.program_id(1)            # col block   (j)
    pid_b = tl.program_id(2)            # combined batch id = B * H

    # ------------------------------------------------------
    # 1️⃣  decode combined batch into (b, h)
    # ------------------------------------------------------
    b = pid_b // H                      # original batch index
    h = pid_b % H                       # hidden‑channel index

    # ------------------------------------------------------
    # 2️⃣  compute offsets inside the tile
    # ------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------
    # 3️⃣  base pointers for this (b, h) pair
    # ------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # ------------------------------------------------------
    # 4️⃣  accumulator (FP32)
    # ------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------
    # 5️⃣  iterate over K dimension in BLOCK_K tiles
    # ------------------------------------------------------
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for t in range(num_tiles_k):
        k_offset = t * BLOCK_K
        # construct the K‑indices for this tile
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # masks for the current tile (out‑of‑bounds → zero)
        mask_k = offs_k < K

        # --------------------------------------------------
        # load A tile : left[b, i, k, h]
        # --------------------------------------------------
        a = tl.load(
            left_base
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # --------------------------------------------------
        # load B tile : right[b, j, k, h]
        # --------------------------------------------------
        b_tile = tl.load(
            right_base
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # --------------------------------------------------
        # FP16·FP16 → FP32 dot‑product
        # --------------------------------------------------
        acc += tl.dot(a, b_tile)   # acc stays FP32

    # ------------------------------------------------------
    # 6️⃣  write back the result (still FP32)
    # ------------------------------------------------------
    out_ptrs = out_base \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def _tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor,
                             BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 64) -> torch.Tensor:
    """
    Triton wrapper for the outgoing TriMul kernel.
    `left` and `right` must be FP16 tensors of shape (B, N, N, H).
    Returns a FP32 tensor of the same shape.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)   # (B, N, N, H)

    # strides in *elements*
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    # launch grid: (M‑tiles, N‑tiles, B·H)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        N,               # K == N
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim] (float32)
        - mask         : torch.Tensor [bs, seq_len, seq_len]   (float32)
        - weights      : dict of model weights (float32 tensors)
        - config       : dict with at least ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------
    # unpack
    # --------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)

    # --------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32)
    # --------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------
    # 2️⃣  Cast to FP16 for the heavy arithmetic
    # --------------------------------------------------
    x_h = x.to(torch.float16)

    # --------------------------------------------------
    # 3️⃣  Load (and half‑cast) linear weights
    # --------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    # output LayerNorm & final linear stay in FP32
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # (dim, hidden_dim), FP32

    # --------------------------------------------------
    # 4️⃣  Linear projections (FP16)
    # --------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H]  fp16

    # --------------------------------------------------
    # 5️⃣  Optional mask – broadcast over hidden dim
    # --------------------------------------------------
    if not nomask:
        # mask: FP32 → FP16 (tiny kernel)
        mask_flat = mask.view(-1)
        mask_fp16 = torch.empty_like(mask_flat, dtype=torch.float16, device=device)

        BLOCK = 1024
        grid = ((mask_flat.numel() + BLOCK - 1) // BLOCK,)
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_fp16,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK,
        )
        mask_h = mask_fp16.view(mask.shape + (1,))          # [B, N, N, 1] fp16
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------
    # 6️⃣  Gating (sigmoid) – still FP16
    # --------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, H] fp16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B, N, N, H] fp16

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------
    # 7️⃣  Core N³ operation – batched matmul L @ Rᵀ
    # --------------------------------------------------
    # left/right are (B, N, N, H) fp16, output (B, N, N, H) fp32
    out_intermediate = _tri_mul_outgoing_triton(left, right,
                                                BLOCK_M=64,
                                                BLOCK_N=64,
                                                BLOCK_K=64)

    # --------------------------------------------------
    # 8️⃣  Output LayerNorm over hidden dimension (FP32)
    # --------------------------------------------------
    out = F.layer_norm(
        out_intermediate,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # [B, N, N, H] fp32

    # --------------------------------------------------
    # 9️⃣  Multiply by output gate and final projection
    # --------------------------------------------------
    out = out * out_gate.to(torch.float32)   # element‑wise gate
    out = F.linear(out, to_out_w)            # FP32 → [B, N, N, dim]

    return out
