def _apply_gate_mask_kernel(out_left_ptr, out_right_ptr, left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr, total_elements, BLOCK_SIZE: tl.constexpr):

    pid = tl.program_id(0)

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)

# Triton kernels for fused elementwise multiplication
# 1. fuse left/right gating and mask
@triton.jit
def _fused_gate_mask_kernel(
    out_left_ptr, out_right_ptr,
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,  # mask broadcast over hidden_dim
    total_elements: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elements

    # Load values
    l = tl.load(left_ptr + offset, mask=mask, other=0.0)
    r = tl.load(right_ptr + offset, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offset, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offset, mask=mask, other=0.0)

    # mask broadcast (mask_ptr is B*N*N)

    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)  # shape broadcasted

    lo = l * lg * m
    ro = r * rg * m

    tl.store(out_left_ptr + offset, lo, mask=mask)
    tl.store(out_right_ptr + offset, ro, mask=mask)

"""
custom_kernel
--------------
Triton‑accelerated forward pass of the **outgoing** TriMul module.

Algorithm
---------
1. **Layer‑norm** the input (float32) – cheap compared to the cubic term.
2. Cast to fp16 and compute three linear projections:
      * left_proj, right_proj   – map (dim → hidden_dim)
      * left_gate, right_gate, out_gate – same map followed by sigmoid.
   These are performed with `torch.nn.functional.linear` (cuBLAS‑fast).
3. **Mask & gating** – when a mask is present we fuse the mask,
   left/right gates and the projected tensors with a tiny Triton kernel
   (`_fused_gate_mask_kernel`).  In the no‑mask case we use a simple
   element‑wise torch multiplication.
4. **Core cubic update**  
   `out[i,j] = Σ_k left[i,k] * right[j,k]`  ⇔  `out = left @ rightᵀ`.  
   We reshape the tensors from `[B,N,N,H]` → `[B·H,N,N]` and use the
   highly‑optimized cuBLAS batched GEMM (`torch.matmul`) which is far
   faster than a generic Triton matmul for the large N³ workloads.
5. **Second Layer‑norm** over the hidden dimension (float32).
6. **Output gating** – apply `out_gate` (fp16) to the normalized tensor
   with a second Triton kernel (`_apply_out_gate_kernel`) that casts the
   gate to fp32 on‑the‑fly.
7. **Final projection** back to the original feature dimension with a
   float32 linear layer.

Only the element‑wise gating steps are implemented in Triton; the heavy
matrix multiplication leverages cuBLAS for maximal throughput on an
H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
# Gate + mask fusion: out_left = left * left_gate * mask,
#                     out_right = right * right_gate * mask
@triton.jit
def _fused_gate_mask_kernel(
    out_left_ptr, out_right_ptr,
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,                     # [B*N*N] fp16, broadcast over hidden_dim
    total_elements: tl.constexpr,  # B*N*N*hidden_dim
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elements

    # Load projected tensors and gates
    l = tl.load(left_ptr + offset, mask=mask, other=0.0)
    r = tl.load(right_ptr + offset, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offset, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offset, mask=mask, other=0.0)

    # Broadcast mask: index = offset // hidden_dim
    mask_idx = offset // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # 1.0 = no‑mask

    lo = l * lg * m
    ro = r * rg * m

    tl.store(out_left_ptr + offset, lo, mask=mask)
    tl.store(out_right_ptr + offset, ro, mask=mask)


# Simple fp16 elementwise multiplication (used when nomask==True)
@triton.jit
def _elementwise_mul_kernel(
    out_ptr,
    a_ptr,
    b_ptr,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elements

    a = tl.load(a_ptr + offset, mask=mask, other=0.0)
    b = tl.load(b_ptr + offset, mask=mask, other=0.0)
    tl.store(out_ptr + offset, a * b, mask=mask)


# Apply the output gate (fp16) to a fp32 tensor in‑place
@triton.jit
def _apply_out_gate_kernel(
    out_ptr,            # fp32
    out_gate_ptr,       # fp16, same shape as out
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elements

    out_val = tl.load(out_ptr + offset, mask=mask, other=0.0)
    gate = tl.load(out_gate_ptr + offset, mask=mask, other=1.0)
    out_val = out_val * tl.cast(gate, tl.float32)
    tl.store(out_ptr + offset, out_val, mask=mask)


# ----------------------------------------------------------------------
# Helper wrappers for Triton kernels
# ----------------------------------------------------------------------
def _apply_gate_mask(
    left: torch.Tensor,
    right: torch.Tensor,
    left_gate: torch.Tensor,
    right_gate: torch.Tensor,
    mask: torch.Tensor,
) -> Tuple[torch.Tensor, torch.Tensor]:
    """Fuse mask, left/right gating with a Triton kernel."""
    total = left.numel()
    hidden_dim = left.shape[-1]
    BLOCK_SIZE = 1024
    grid = (total + BLOCK_SIZE - 1) // BLOCK_SIZE

    out_left = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # flatten mask to 1‑D (B*N*N) half tensor
    mask_flat = mask.contiguous().view(-1)

    _fused_gate_mask_kernel[grid](
        out_left_ptr=out_left,
        out_right_ptr=out_right,
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_flat,
        total_elements=total,
        hidden_dim=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out_left, out_right


def _apply_elementwise_mul(
    a: torch.Tensor, b: torch.Tensor
) -> torch.Tensor:
    """Element‑wise fp16 multiplication via Triton."""
    total = a.numel()
    BLOCK_SIZE = 1024
    grid = (total + BLOCK_SIZE - 1) // BLOCK_SIZE
    out = torch.empty_like(a)
    _elementwise_mul_kernel[grid](
        out_ptr=out,
        a_ptr=a,
        b_ptr=b,
        total_elements=total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out


def _apply_out_gate(out: torch.Tensor, out_gate: torch.Tensor) -> torch.Tensor:
    """Multiply fp32 tensor `out` by fp16 gate `out_gate` in‑place."""
    total = out.numel()
    BLOCK_SIZE = 1024
    grid = (total + BLOCK_SIZE - 1) // BLOCK_SIZE
    _apply_out_gate_kernel[grid](
        out_ptr=out,
        out_gate_ptr=out_gate,
        total_elements=total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float). May be None.
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim] – float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for heavy ops
    # ------------------------------------------------------------------
    x_h = x.half()  # fp16

    # ------------------------------------------------------------------
    # 3) Linear projections (fp16, no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left = F.linear(x_h, left_proj_w)   # [B,N,N,hidden_dim] – fp16
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = F.linear(x_h, left_gate_w).sigmoid()   # [B,N,N,hidden_dim] – fp16
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()    # [B,N,N,hidden_dim] – fp16

    # ------------------------------------------------------------------
    # 5) Apply mask and element‑wise gates (Triton‑fused)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N] – cast to half and keep as is (will be broadcast)
        mask_h = mask.to(dtype=torch.float16, device=device)
        left, right = _apply_gate_mask(
            left, right, left_gate, right_gate, mask_h
        )
    else:
        # No mask – just multiply by gates via a thin Triton kernel
        left = _apply_elementwise_mul(left, left_gate)
        right = _apply_elementwise_mul(right, right_gate)

    # ------------------------------------------------------------------
    # 6) Core cubic update – batched matmul via cuBLAS (torch.matmul)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape (B,N,N,H) -> (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Batched matmul: left @ rightᵀ   -> fp16, then up‑cast
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1)).float()  # [B*H,N,N] fp32

    # Restore original layout: (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()     # fp32

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension (fp32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 8) Output gating (Triton kernel)
    # ------------------------------------------------------------------
    out = _apply_out_gate(out, out_gate)

    # ------------------------------------------------------------------
    # 9) Final linear projection back to dim (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim] – float32

    return out
