"""
TriMul “outgoing” update – Triton‑accelerated forward pass
========================================================

The module takes a 4‑D tensor X ∈ ℝ^{B×N×N×D} and produces an update
Y ∈ ℝ^{B×N×N×D}.  The core of the operation is the N³ “outgoing”
multiplication

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}               (1)

where `left` and `right` are linear projections of the normalized input,
gated by sigmoid‑gates and (optionally) a binary mask.  Equation (1) can be
seen as a batched matrix multiplication of shape
(A,B) = (B·H,N,N) @ (B·H,N,N) where each hidden channel h is an
independent batch.  The heavy O(B·H·N³) work is performed by a custom
Triton kernel that computes

    C_{b,i,j} = Σ_k A_{b,i,k} * B_{b,j,k}                (2)

The kernel works on FP16 inputs and accumulates in FP32 for numerical
stability.  All remaining operations (LayerNorm, gating, final projection)
are done with PyTorch (mixed‑precision), keeping the final result in
FP32.

The implementation follows the reference PyTorch code but replaces the
inner N³ loop with the batched GEMM `_batched_matmul_triton`.  It works
for all test configurations (different batch sizes, sequence lengths,
feature dimensions, and optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A (M×K) × B (N×K) → C (M×N)
#   – A, B are fp16, C is fp32 (fp16·fp16 → fp32 accumulation)
#   – Batch dimension is the third launch axis.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions
    stride_ah, stride_am, stride_ak,         # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,         # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,         # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where B is stored as (N, K) and we compute
    C_{i,j} = Σ_k A_{i,k} * B_{j,k}.  The kernel is double‑buffered
    (num_stages=4) and uses 8 warps (256 threads) per block.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K‑loop
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        # ---- load A tile (M×K) ----
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_base + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # ---- load B tile (N×K) transposed on‑the‑fly -> (K×N) ----
        b_ptrs = b_batch + (k_base + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # fused dot product (fp16·fp16 → fp32)
        acc += tl.dot(a_tile, b_tile)

    # ---- store result ----
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Triton wrapper for the batched GEMM.
    A, B – fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) in fp32.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16 inputs"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    # Allocate output (FP32 for stability)
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps (256 threads) – good for 128×128 tiles on H100
        num_stages=4,         # double‑buffered loads
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) – may be None if nomask=True
            - weights: dict of model parameters (float32)
            - config: dict containing at least "dim", "hidden_dim",
                      optional "nomask" flag

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32) – the module output.
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # hidden projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2️⃣ Input LayerNorm (over last dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D], float32

    # Cast to fp16 for the heavy linear / gating work
    x_f16 = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) and gating weights
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # Project to hidden dimension H
    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H], fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H], fp16

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (binary 0/1) – broadcast over hidden dim
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) and element‑wise gate application
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H], fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))  # [B, N, N, H], fp16
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B, N, N, H], fp16

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Core N³ update – batched GEMM via Triton
    # --------------------------------------------------------------
    B, N_seq, _, H = left.shape  # B, N, N, H

    # Reshape to (B·H, N, N) so each hidden channel is an independent batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N_seq, N_seq)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N_seq, N_seq)  # fp16

    # Choose a K‑tile size that fits the sequence length
    block_k = 64 if N_seq >= 64 else 32

    # Batched GEMM: Σ_k left_{i,k} * right_{j,k}
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=block_k,
    )   # [B·H, N, N] – fp32

    # Restore original layout (B, N, N, H)
    out = out_mat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()   # fp32

    # --------------------------------------------------------------
    # 7️⃣ Output LayerNorm, gating and final projection back to D
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32, over hidden dimension

    # Apply out‑gate (cast to fp32 automatically)
    out = out * out_gate

    # Final linear projection to original dimension D
    out = F.linear(out, weights["to_out.weight"])

    return out
