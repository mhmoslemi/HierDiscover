@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr, mask_ptr, left_gate_ptr, right_gate_ptr,
    M, N, H,  # dims for left/right shape: [B,N,N,H]
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lgate_b, stride_lgate_i, stride_lgate_j,
    stride_rgate_b, stride_rgate_i, stride_rgate_j,
    BLOCK_B: tl.constexpr, BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    off_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    off_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    off_h = tl.arange(0, BLOCK_H)  # handle H dimension inside loop maybe

    # compute mask offset broadcasted across H dimension
    mask_off = pid_b * stride_mask_b + off_i[:, None] * stride_mask_i + off_j[None, :] * stride_mask_j
    mask_tile = tl.load(mask_ptr + mask_off, mask=(off_i[:, None] < M) & (off_j[None, :] < N), other=0.0)   # shape (I,J)

    # left_gate similarly shape [B,N,N,1] (expanded), left_gate broadcast across H
    left_gate_off = pid_b * stride_lgate_b + off_i[:, None] * stride_lgate_i + off_j[None, :] * stride_lgate_j
    left_gate_tile = tl.load(left_gate_ptr + left_gate_off, mask=(off_i[:, None] < M) & (off_j[None, :] < N), other=0.0)

    # right_gate
    right_gate_off = pid_b * stride_rgate_b + off_i[:, None] * stride_rgate_i + off_j[None, :] * stride_rgate_j
    right_gate_tile = tl.load(right_gate_ptr + right_gate_off, mask=(off_i[:, None] < M) & (off_j[None, :] < N), other=0.0)

    # compute scaling factors
    left_scale = mask_tile * left_gate_tile
    right_scale = mask_tile * right_gate_tile

    # Now process H dimension in chunks
    for offs_h in range(0, H, BLOCK_H):
        off_h = offs_h + tl.arange(0, BLOCK_H)
        # Load left and right tiles
        left_ptrs = left_ptr + pid_b * stride_left_b + off_i[:, None, None] * stride_left_i + off_j[None, :, None] * stride_left_j + off_h[None, None, :] * stride_left_h
        mask_left = (off_i[:, None, None] < M) & (off_j[None, :, None] < N) & (off_h[None, None, :] < H)
        left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)
        right_ptrs = right_ptr + pid_b * stride_right_b + off_i[:, None, None] * stride_right_i + off_j[None, :, None] * stride_right_j + off_h[None, None, :] * stride_right_h
        mask_right = (off_i[:, None, None] < M) & (off_j[None, :, None] < N) & (off_h[None, None, :] < H)
        right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # scaling
        left_tile = left_tile * left_scale[:, :, None]
        right_tile = right_tile * right_scale[:, :, None]

        # store back
        tl.store(left_ptrs, left_tile, mask=mask_left)
        tl.store(right_ptrs, right_tile, mask=mask_right)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1. Input LayerNorm (FP32)
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # 2. Cast to fp16
    x_f16 = x.to(torch.float16)

    # 3. Fuse linear projections
    B, N, _, _ = x_f16.shape

    # Flatten positions
    x_flat = x_f16.view(-1, dim)  # (B*N*N, dim)

    # Concatenate proj weights
    left_proj_w_fp16 = weights["left_proj.weight"].to(torch.float16)
    right_proj_w_fp16 = weights["right_proj.weight"].to(torch.float16)
    proj_w = torch.cat([left_proj_w_fp16, right_proj_w_fp16], dim=0)  # (2*hidden, dim)
    proj_out = F.linear(x_flat, proj_w)  # (B*N*N, 2*hidden)
    left, right = torch.split(proj_out, hidden_dim, dim=-1)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # Gate weights
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    gate_w = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*hidden, dim)
    gate_out = F.linear(x_flat, gate_w)  # (B*N*N, 3*hidden)
    left_gate, right_gate, out_gate = torch.split(gate_out, hidden_dim, dim=-1)

    right_gate = torch.sigmoid(right_gate).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate).view(B, N, N, hidden_dim)

    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]

    out = out.to(torch.float32)

"""
TriMul “outgoing” forward – optimized implementation.

Algorithm
---------
1. Input LayerNorm                (FP32)
2. Cast to FP16, flatten the [B,N,N] spatial dimensions.
3. Fuse the two projections (left/right) into a single
   linear layer by concatenating the weights → one matmul.
4. Fuse the three gating linear layers (left‑gate, right‑gate,
   out‑gate) similarly → one matmul + split + sigmoid.
5. Apply the (optional) binary mask together with the gates
   in a single element‑wise multiply (no extra memory pass).
6. Core cubic update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   Implemented as a batched mat‑mul L @ Rᵀ with a custom Triton
   kernel (FP16 inputs, FP32 accumulation, FP16 output).  The
   kernel tiles the work with (BLOCK_M×BLOCK_N×BLOCK_K) = (128×128×32)
   and uses 8‑16 warps per block (auto‑chosen based on N).
7. Output LayerNorm (FP16), out‑gate scaling, final linear
   projection back to the original dimension.
8. Cast to FP32 for the API return.

All heavy compute (steps 3, 4, 6, 7) stay in FP16 to minimise
memory bandwidth while preserving numerical quality with FP32
accumulation where needed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (batch, M, K)   fp16
#   B : (batch, N, K)   fp16  (will be accessed as K×N)
#   C : (batch, M, N)   fp16  (accumulated in FP32)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,    # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    #  Program IDs
    # --------------------------------------------------------------
    pid_b = tl.program_id(0)   # batch dimension
    pid_m = tl.program_id(1)   # block row   (M)
    pid_n = tl.program_id(2)   # block col   (N)

    # --------------------------------------------------------------
    #  Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # --------------------------------------------------------------
    #  Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over the K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B tile (as Bᵀ): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Dot product (FP32 accumulation)
        acc += tl.dot(a_tile, b_tile)

    # --------------------------------------------------------------
    #  Store result (cast to fp16)
    # --------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 32,
) -> torch.Tensor:
    """
    Batched matrix multiplication via Triton.

    Args:
        A (torch.Tensor): fp16 tensor of shape (batch, M, K)
        B (torch.Tensor): fp16 tensor of shape (batch, N, K)  (treated as Bᵀ)
        BLOCK_* (int): compile‑time tile sizes

    Returns:
        torch.Tensor: fp16 tensor (batch, M, N)
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # strides in element units
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # auto‑tune warp count: more warps for larger M/N
    num_warps = 8 if max(M, N) <= 256 else 16

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )
    return C


# ------------------------------------------------------------------
# Entry point required by the harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of TriMul (outgoing update).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to fp16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)          # [B, N, N, D] (fp16)
    B, N, _, _ = x_f16.shape
    flat_x = x_f16.view(-1, dim)         # (B·N·N, D)

    # --------------------------------------------------------------
    # 4) Fuse linear projections (left/right) into one matmul
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)    # (H, D)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)  # (H, D)
    proj_w = torch.cat([left_proj_w, right_proj_w], dim=0)       # (2H, D)

    proj_out = F.linear(flat_x, proj_w)        # (B·N·N, 2H)
    left, right = torch.split(proj_out, hidden_dim, dim=-1)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 5) Fuse gating linear layers (left_gate, right_gate, out_gate)
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)  # (H, D)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)     # (H, D)
    gate_w = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3H, D)

    gate_out = F.linear(flat_x, gate_w)         # (B·N·N, 3H)
    left_gate, right_gate, out_gate = torch.split(gate_out, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6) Apply mask (if present) and element‑wise gates
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * m * left_gate
        right = right * m * right_gate
    else:
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core cubic update – batched mat‑mul L @ Rᵀ
    # --------------------------------------------------------------
    # reshape to (B·H, N, N) batches
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N).contiguous()

    # Perform the heavy batched mat‑mul with Triton
    out_mat = _batched_matmul_triton(left_mat, right_mat)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 8) Output LayerNorm (FP16), out‑gate, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # gate post‑norm

    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_w)                          # fp16, shape (B,N,N,D)

    # --------------------------------------------------------------
    # 9) Cast to FP32 for the API return
    # --------------------------------------------------------------
    return out.to(torch.float32)
