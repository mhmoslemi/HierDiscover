import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (mirrors the PyTorch reference)
-----------------------------------------
1️⃣ Layer‑norm the 4‑D input (B,N,N,D).
2️⃣ Two bias‑free linear projections (left/right) → (B,N,N,H) and three
   sigmoid gates (left_gate, right_gate, out_gate) from the same normalized input.
3️⃣ Optional binary mask M∈{0,1} is applied to left/right.
4️⃣ Element‑wise gate multiplication:
        left  = left  * left_gate
        right = right * right_gate
5️⃣ Core N³ update (outgoing):
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix‑multiply L @ Rᵀ for each hidden channel.
   We reshape (B,N,N,H) → (B·H,N,N) and run a highly‑tuned Triton GEMM
   (fp16·fp16 → fp32, 8 warps, double‑buffered).
6️⃣ Reshape back to (B,N,N,H).
7️⃣ Output LayerNorm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to the original dimension D.
   The result is returned in float32.

The heavy N³ computation is the only part written in Triton;
all other operations use PyTorch with mixed‑precision (fp16 for the
linear / gating steps) to reduce memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A (M×K) * B (N×K) → C (M×N)
#   – A, B are fp16, C is fp32 (fp16·fp16 → fp32 accumulation)
#   – Batch dimension is the 3rd launch axis.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,    # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------- program ids -------------------
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # ------------------- tile offsets ------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- loop over K -------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (transposed on‑the‑fly): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # fused dot‑product (fp16·fp16 → fp32)
        acc += tl.dot(a_tile, b_tile)

    # ------------------- write back --------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(A: torch.Tensor,
                           B: torch.Tensor,
                           BLOCK_M: int = 128,
                           BLOCK_N: int = 128,
                           BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C with shape (batch, M, N) in fp32.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16 inputs"

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    # Output in fp32 to retain precision for the following LayerNorm
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps (256 threads) – good for 128×128 tiles
        num_stages=4,         # double‑buffered loads
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) – may be None when nomask=True
            - weights: dict of model parameters (float32)
            - config: dict containing at least "dim", "hidden_dim",
                      optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the module output.
    """
    # -------------------------- unpack -------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension D
    hidden_dim = config["hidden_dim"]  # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------- 1) Input LayerNorm -------------------------
    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D] – float32

    # Cast to fp16 for the heavy linear / gate work
    x_f16 = x.to(torch.float16)

    # ------------------- 2) Linear projections ---------------------
    # Convert weight tensors once to fp16
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H] – fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H] – fp16

    # ------------------- 3) Optional mask -------------------------
    if (not nomask) and mask is not None:
        # mask ∈ {0,1}; broadcast over the hidden dimension
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * m
        right = right * m

    # ------------------- 4) Gates (sigmoid) -----------------------
    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # fp16, shape [B, N, N, H]

    # ------------------- 5) Apply gates ---------------------------
    left  = left * left_gate
    right = right * right_gate

    # ------------------- 6) Core N³ batched mat‑mul ---------------
    B, N, _, H = left.shape  # B, seq_len, _, hidden_dim

    # Reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Choose a K‑tile that fits the sequence length (avoid very small tiles)
    block_k = 64 if N >= 64 else 32

    # Batched GEMM: left @ rightᵀ  (fp16·fp16 → fp32)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=block_k,
    )   # [B·H, N, N] – fp32

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------- 7) Output LayerNorm & final proj ---------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32, normalized over hidden dimension

    # Multiply by the out‑gate (broadcast)
    out = out * out_gate  # out_gate will be up‑cast to fp32 automatically

    # Final linear projection back to the original dimension D
    out = F.linear(out, weights["to_out.weight"])

    return out
