def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul outgoing update.

    ...
    """

    # Unpack
    input_tensor, mask, weights, config = data
    ...

    # Normalization
    x = F.layer_norm(...)

    # Cast to fp16 (or bfloat16)
    x_f16 = x.to(torch.float16)

    # Combine left_proj and left_gate weights
    left_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=1).to(torch.float16)
    right_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=1).to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # Flatten and linear
    B, N, _, D = x_f16.shape
    x_flat = x_f16.reshape(-1, D)   # (B*N*N, D)

    left_out = torch.nn.functional.linear(x_flat, left_w)  # (B*N*N, 2*H)
    right_out = torch.nn.functional.linear(x_flat, right_w) # (B*N*N, 2*H)
    out_gate = torch.nn.functional.linear(x_flat, out_gate_w)   # (B*N*N, hidden_dim)

    # split
    left = left_out[:, :hidden_dim].reshape(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_out[:, hidden_dim:].reshape(B, N, N, hidden_dim))
    right = right_out[:, :hidden_dim].reshape(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_out[:, hidden_dim:].reshape(B, N, N, hidden_dim))
    out_gate = torch.sigmoid(out_gate.reshape(B, N, N, hidden_dim))

    # mask
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # core batched GEMM
    left_mat = left.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*hidden_dim, N, N)

    # choose block sizes based on N
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64 if N >= 64 else 32

    out_mat = _batched_matmul_triton(left_mat, right_mat,
                                     BLOCK_M=BLOCK_M,
                                     BLOCK_N=BLOCK_N,
                                     BLOCK_K=BLOCK_K)

    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # (B,N,N,H)

    # output LayerNorm (fp32)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (mirrors the reference PyTorch implementation)
-------------------------------------------------------
1️⃣ Layer‑norm the 4‑D tensor (B, N, N, D).
2️⃣ Two bias‑free linear projections (left/right) and three sigmoid gates
   (left_gate, right_gate, out_gate).  The two projections and their
   corresponding gates are fused into two GEMMs (dim → 2·hidden_dim) to
   halve the number of matmul calls.
3️⃣ Optional binary mask (broadcast over the hidden dimension) applied
   to the projected tensors.
4️⃣ Element‑wise gating:   left  = left  * left_gate
                           right = right * right_gate
5️⃣ Core N³ update (outgoing):
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix‑multiply L @ Rᵀ for each hidden channel.
   We reshape (B, N, N, H) → (B·H, N, N) and run a custom Triton GEMM
   (fp16·fp16 → fp32 accumulation) with 8 warps and double‑buffering.
6️⃣ Reshape back to (B, N, N, H).
7️⃣ Output LayerNorm over the hidden dimension, element‑wise multiplication
   by the out‑gate, and a final linear projection back to dimension D.

The heavy N³ computation is performed by a Triton kernel; all other
operations use highly‑optimized PyTorch kernels (layer‑norm, linear,
sigmoid).  Mixed‑precision (fp16 for the bulk of the work) keeps memory
traffic low while the final output is retained in float32 for numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A (M×K) * B (N×K) → C (M×N)
#   – A, B are fp16, C is fp32 (fp16·fp16 → fp32 accumulation)
#   – The hidden dimension H is folded into the batch dimension.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,    # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ---- Load A tile (BLOCK_M × BLOCK_K) ---------------------------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # ---- Load B tile (BLOCK_K × BLOCK_N) ---------------------------------
        # B is accessed as Bᵀ on‑the‑fly
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # ---- Fused dot‑product (fp16·fp16 → fp32) ---------------------------
        # Explicit fp32 accumulation enables Tensor‑Core usage on H100.
        acc += tl.dot(a_tile, b_tile, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write back the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched GEMM (A @ Bᵀ) using the Triton kernel above.

    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C with shape (batch, M, N) in fp32.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16 inputs"

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block (good for 128×128 tiles)
        num_stages=4,         # double‑buffered loads
    )
    return C


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/int) – may be None when nomask=True
            - weights: dict of model parameters (float32)
            - config: dict containing at least "dim", "hidden_dim" and optionally "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the module output.
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack and configuration
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # input feature dimension D
    hidden_dim = config["hidden_dim"] # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (over D)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D] – float32

    # Work in fp16 for the heavy linear / gate steps
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left/right projection + corresponding gate into two GEMMs
    # ------------------------------------------------------------------
    # Shapes:
    #   left_proj.weight : (hidden_dim, dim)
    #   left_gate.weight : (hidden_dim, dim)
    # We concatenate on the output dimension so a single matmul yields
    # both projection and gate pre‑sigmoid values.
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*H, D)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*H, D)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)   # (H, D)

    B, N, _, _ = x_f16.shape
    x_flat = x_f16.view(-1, dim)                    # (B·N·N, D)

    # Combined linear calls (no bias)
    left_comb = F.linear(x_flat, left_w)            # (B·N·N, 2·H)
    right_comb = F.linear(x_flat, right_w)          # (B·N·N, 2·H)
    out_gate_pre = F.linear(x_flat, out_gate_w)    # (B·N·N, H)

    # Split projection & gate
    left = left_comb[:, :hidden_dim].view(B, N, N, hidden_dim)           # (B,N,N,H)
    left_gate = torch.sigmoid(left_comb[:, hidden_dim:].view(B, N, N, hidden_dim))

    right = right_comb[:, :hidden_dim].view(B, N, N, hidden_dim)        # (B,N,N,H)
    right_gate = torch.sigmoid(right_comb[:, hidden_dim:].view(B, N, N, hidden_dim))

    out_gate = torch.sigmoid(out_gate_pre.view(B, N, N, hidden_dim))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1]
        m = mask.to(torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates (element‑wise)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core N³ update – batched GEMM (left @ rightᵀ) per hidden channel
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) where each channel is a separate batch.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # fp16

    # Tuning of block sizes: larger K tiles for long sequences,
    # but keep a reasonable BLOCK_K for memory bandwidth.
    BLOCK_K = 64 if N >= 64 else 32

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=BLOCK_K,
    )   # (B·H, N, N) – fp32

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm, out‑gate multiplication, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32, normalized over hidden dimension

    # Broadcasted multiplication by out_gate (fp16 → fp32)
    out = out * out_gate

    # Final linear projection back to dimension D
    out = F.linear(out, weights["to_out.weight"])

    return out
