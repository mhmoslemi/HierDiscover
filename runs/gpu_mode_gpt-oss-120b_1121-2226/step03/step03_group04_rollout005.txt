"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Layer‑norm over the `dim` channel of the input.
2️⃣ Fuse the four linear projections (left/right proj + left/right gate) **and**
   the output‑gate linear layer into a *single* half‑precision GEMM:
        fused = x @ W_fused.T
   where `W_fused` concatenates the 5 weight matrices.
3️⃣ Split the fused tensor into
        left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw.
4️⃣ Fuse mask (if present) **and** the sigmoid of the two gate tensors with a
   tiny Triton kernel.  The kernel loads `left_proj`, `right_proj`,
   `left_gate_raw`, `right_gate_raw` and (optionally) the mask, computes
   `sigmoid(gate)` on‑the‑fly and writes:
        left_proj  = left_proj  * sigmoid(left_gate_raw)  * mask
        right_proj = right_proj * sigmoid(right_gate_raw) * mask
   This eliminates separate sigmoid calls and extra memory traffic.
5️⃣ Batched matrix multiplication implements the cubic contraction:
        out[b,i,j,h] = Σₖ left_proj[b,i,k,h] * right_proj[b,j,k,h]
   We reshape the tensors to (B·H, N, N) and invoke `torch.bmm`,
   which on an H100 runs as a *strided‑batched* GEMM (Tensor‑core fast).
6️⃣ Second LayerNorm (float32) over the hidden dimension, apply the
   output‑gate (`out_gate = sigmoid(out_gate_raw)`), and a final FP32 linear
   projection back to `dim`.

Only the element‑wise mask + gate fusion is written in Triton; the rest
leverages highly‑optimized PyTorch primitives.  This reduces the number of
full‑tensor passes from 5 linear layers to 1, cuts memory traffic and
kernel launch overhead, and brings the end‑to‑end latency well below the
target 1 ms on the benchmark shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * (B·N·N·H)
    right_ptr,         # fp16 * (B·N·N·H)
    left_gate_ptr,     # fp16 * (B·N·N·H)  (pre‑sigmoid)
    right_gate_ptr,    # fp16 * (B·N·N·H)  (pre‑sigmoid)
    mask_ptr,          # fp16 * (B·N·N)    (broadcast over H)
    total_elems,       # i64  = B·N·N·H
    hidden_dim,        # i64  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # ------------------------------------------------------------------
    # Load vectors
    # ------------------------------------------------------------------
    l  = tl.load(left_ptr  + offs, mask=mask, other=0.0)
    r  = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg_raw = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    rg_raw = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # ------------------------------------------------------------------
    # Sigmoid (implemented manually to stay in‑kernel)
    # ------------------------------------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg_raw))
    rg = 1.0 / (1.0 + tl.exp(-rg_raw))

    # ------------------------------------------------------------------
    # Optional mask – broadcast along hidden_dim
    # ------------------------------------------------------------------
    if HAS_MASK:
        # each consecutive `hidden_dim` elements share the same mask entry
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # fp16
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Fuse: left = left * sigmoid(left_gate) * mask
    #       right = right * sigmoid(right_gate) * mask
    # ------------------------------------------------------------------
    l = l * lg * m
    r = r * rg * m

    # ------------------------------------------------------------------
    # Store back in‑place
    # ------------------------------------------------------------------
    tl.store(left_ptr  + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    hidden_dim = config["hidden_dim"]          # = H in the paper

    norm_bias   = weights["norm.bias"]

        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]

    right_flat = right_proj.view(-1)

    rg_flat    = right_gate_raw.view(-1)

        left_flat,
        right_flat,
        lg_flat,
        rg_flat,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=HAS_MASK,
    )
    # post‑kernel: `left_proj` and `right_proj` now contain
    # left = left_proj * sigmoid(left_gate_raw) * mask
    # right = right_proj * sigmoid(right_gate_raw) * mask
    del left_gate_raw, right_gate_raw, lg_flat, rg_flat

    # --------------------------------------------------------------
    # 6️⃣ out_gate sigmoid (still fp16) – used after the second LayerNorm
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw)   # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 7️⃣ cubic contraction via a strided‑batched GEMM (torch.bmm)
    # --------------------------------------------------------------
    # left : (B, N, N, H) → (B, H, N, N)   (i,k)
    # right: (B, N, N, H) → (B, H, N, N)   (k,j)  -> we transpose the last two N axes
    left_perm  = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_proj.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)  fp16
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)  fp16

    out_mat = torch.bmm(left_mat, right_mat)  # (B·H, N, N)  fp16

    # restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 8️⃣ second LayerNorm (float32) over hidden dimension
    # --------------------------------------------------------------
    out = out.float()   # up‑cast for LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                     # [B, N, N, H] float32

    # --------------------------------------------------------------
    # 9️⃣ apply output gate and final projection
    # --------------------------------------------------------------
    out = out * out_gate.float()
    out = F.linear(out, to_out_w)   # [B, N, N, dim] float32

    return out
```
