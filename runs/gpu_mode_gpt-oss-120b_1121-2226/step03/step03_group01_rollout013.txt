"""
TriMul “outgoing” kernel (forward only)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D pairwise input (float32).
2️⃣ Cast to fp16 – all heavy linear work stays in half precision.
3️⃣ Fuse the five bias‑free linear layers
     (left_proj, left_gate, right_proj, right_gate, out_gate)
   into a single matrix multiplication (B·N·N × dim) × (dim × 5·hidden_dim).
   The result is split and reshaped to obtain the projected tensors and the
   three sigmoid‑gates.
4️⃣ Apply the optional pair‑wise mask and the element‑wise gates
   (still in fp16).
5️⃣ Reshape to a batched 3‑D view (B·H, N, N) and launch a Triton kernel
   that computes, for each hidden channel,
       out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]
   i.e. a batched GEMM A @ Bᵀ.  The kernel works on fp16 tiles,
   accumulates in fp32 and finally stores the result as fp16 (≈‑50 % bandwidth).
6️⃣ Cast the GEMM result back to float32, reshape to [B, N, N, H].
7️⃣ Layer‑norm over the hidden dimension (float32).
8️⃣ Multiply by the output‑gate, apply the final bias‑free linear projection
   back to the original dimension D.
9️⃣ Return a float32 tensor of shape [B, N, N, D].

The only non‑trivial operation (the N³ matrix multiplication) is performed
by the custom Triton kernel; everything else relies on highly‑optimized
torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B are fp16, C is fp16,
#                                            accumulation in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N] (fp16)
    b_ptr,               # right : [BH, N, N] (fp16) – interpreted as (BH, K, N)
    c_ptr,               # out   : [BH, N, N] (fp16)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)            # (batch * hidden)
    pid_m = tl.program_id(axis=0)            # tile row
    pid_n = tl.program_id(axis=1)            # tile col

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (better precision, no extra conversion per tile)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- load A tile (M × K) ------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ----- load Bᵀ tile (K × N) ------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # fp16

        # ----- accumulate ------------------------------------------------
        # tl.dot returns fp16; we cast to fp32 before accumulation.
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back: cast the accumulator to fp16 once per tile
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D]  (fp16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse 5 bias‑free linear layers (left_proj, left_gate,
    #    right_proj, right_gate, out_gate) into a single matmul.
    # ------------------------------------------------------------------
    # flatten to (B·N·N, dim)
    flat = x_fp16.view(-1, dim)                       # (B·N·N, D)

    # stack weights → (5·hidden_dim, dim)
    stacked_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                                          # fp16

    # one large linear
    lin_out = F.linear(flat, stacked_weight)          # (B·N·N, 5·H)

    # split back to the five tensors
    split = torch.split(lin_out, hidden_dim, dim=1)
    left_proj_out, left_gate_pre, right_proj_out, right_gate_pre, out_gate_pre = split

    # reshape to [B, N, N, H]
    left_proj_out = left_proj_out.view(bs, seq_len, seq_len, hidden_dim)
    left_gate = torch.sigmoid(left_gate_pre.view(bs, seq_len, seq_len, hidden_dim))
    right_proj_out = right_proj_out.view(bs, seq_len, seq_len, hidden_dim)
    right_gate = torch.sigmoid(right_gate_pre.view(bs, seq_len, seq_len, hidden_dim))
    out_gate = torch.sigmoid(out_gate_pre.view(bs, seq_len, seq_len, hidden_dim))

    # ------------------------------------------------------------------
    # 4️⃣ Apply optional pair‑wise mask and the element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj_out * left_gate
    right = right_proj_out * right_gate
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    # output buffer (fp16 to save bandwidth)
    out_flat = torch.empty(
        (BH, seq_len, seq_len), dtype=torch.float16, device=device
    )

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel launch (batched GEMM)
    # ------------------------------------------------------------------
    # tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    # use a larger K‑tile for large seq_len to cut loop overhead
    BLOCK_K = 128 if seq_len > 512 else 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # tiles over M
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # tiles over N
        BH,                                   # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right – note the transposed layout)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                # more warps → better tensor‑core utilisation
        num_stages=4,               # double‑buffering
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] and up‑cast to float32
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous().float()     # (B, N, N, H) fp32

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # stays float32

    # ------------------------------------------------------------------
    # 9️⃣ Apply output‑gate and final projection back to D
    # ------------------------------------------------------------------
    out = out * out_gate                               # broadcast fp16->fp32
    out = F.linear(out, weights["to_out.weight"])      # (bias‑free) → [B, N, N, D]

    return out
