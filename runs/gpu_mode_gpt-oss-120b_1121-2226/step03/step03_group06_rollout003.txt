"""
Triton‑accelerated forward pass for the **outgoing** TriMul module.

Algorithm
---------
1.  Layer‑norm the 4‑D input (FP32).
2.  Cast to FP16 for the heavy linear / gate ops.
3.  Apply three bias‑free linear projections (left, right) and three
    gating linear layers (sigmoid).  Optional mask is fused into the
    projection result.
4.  Compute the N³ “triangle” update
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched mat‑mul of shape (B·H,N,N) @ (B·H,N,N)^T.
    The kernel reads the original [B,N,N,H] tensors directly – no
    permute‑copy – by calculating the batch‑/hidden‑offset inside the
    kernel.  FP16·FP16 → FP32 accumulation (Tensor‑Core friendly) is used.
5.  Reshape the result back to (B,N,N,H) and apply a second LayerNorm.
6.  Multiply by the output gate and a final FP32 linear projection back
    to the original feature dimension (dim).

All operations are performed on‑device; only the tiny reshapes before
and after the Triton kernel are executed on the host.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ where A/B have shape
#   (B, N, N, H)   (i.e. [batch, seq_i, seq_k, hidden]).
# The hidden dimension is fused into the batch dimension without a
# memory copy – the kernel computes the correct base offset from the
# program id.
# ----------------------------------------------------------------------
@triton.jit
def _batched_trimul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes (all == seq_len)
    stride_batch_a, stride_i_a, stride_k_a,   # A strides: (batch, i, k)
    stride_batch_b, stride_n_b, stride_k_b,   # B strides: (batch, n, k)
    stride_batch_c, stride_i_c, stride_n_c,   # C strides: (batch, i, n)
    stride_hidden, hidden_dim,                # hidden‑stride (=1) and hidden size
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row index   (i)
    pid_n = tl.program_id(1)          # tile col index   (j)
    pid_hb = tl.program_id(2)         # combined (batch, hidden) index

    # ----------------------------------------------------------
    # Decode the combined batch/hidden index
    # ----------------------------------------------------------
    b = pid_hb // hidden_dim                     # batch index 0..B‑1
    h = pid_hb - b * hidden_dim                  # hidden index 0..H‑1

    # Base pointer for the selected (b,h) slice
    a_base = A_ptr + b * stride_batch_a + h * stride_hidden
    b_base = B_ptr + b * stride_batch_b + h * stride_hidden

    # ----------------------------------------------------------
    # Offsets for the current tile in the i‑ and n‑dimensions
    # ----------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the beginning of the tile (i‑,k) and (k‑,n)
    a_ptrs = a_base + offs_i[:, None] * stride_i_a + tl.arange(0, BLOCK_K)[None, :] * stride_k_a
    b_ptrs = b_base + tl.arange(0, BLOCK_K)[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

    # Accumulator in FP32 (mixed‑precision Tensor‑Core reduction)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over the k‑dimension in BLOCK_K‑sized tiles
    # ----------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # Masks for out‑of‑bounds loads
        mask_a = (offs_i[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # FP16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # FP16

        # Tensor‑core dot product (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b)

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

    # ----------------------------------------------------------
    # Write the tile into the output tensor
    # ----------------------------------------------------------
    c_base = C_ptr + b * stride_batch_c + h * stride_hidden
    c_ptrs = c_base + offs_i[:, None] * stride_i_c + offs_n[None, :] * stride_n_c
    mask_c = (offs_i[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_trimul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    hidden_dim: int,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Execute the core N³ update with the Triton kernel defined above.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape ``[B, N, N, H]`` with ``dtype=torch.float16``.
    hidden_dim : int
        Same as ``H``.
    BLOCK_* : int
        Tile sizes – tuned for the H100.

    Returns
    -------
    torch.Tensor
        Shape ``[B, N, N, H]`` (float32) – the result of the
        ``einsum('... i k h, ... j k h -> ... i j h')``.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    assert H == hidden_dim and right.shape == (B, N, N, H)

    # Output tensor (batched matmul produces fp32)
    out = torch.empty((B * H, N, N), dtype=torch.float32, device=left.device)

    # ----------------------------------------------------------
    # Strides (in element counts, not bytes)
    # ----------------------------------------------------------
    # left / right have identical stride patterns
    stride_batch_a, stride_i_a, stride_k_a, stride_hidden = left.stride()
    stride_batch_b, stride_n_b, stride_k_b, _ = right.stride()
    stride_batch_c, stride_i_c, stride_n_c = out.stride()

    # Grid: one block per (tile_i, tile_j, hidden‑batch)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    _batched_trimul_kernel[grid](
        left,
        right,
        out,
        N,              # M
        N,              # N
        N,              # K (summation dimension)
        stride_batch_a,
        stride_i_a,
        stride_k_a,
        stride_batch_b,
        stride_n_b,
        stride_k_b,
        stride_batch_c,
        stride_i_c,
        stride_n_c,
        stride_hidden,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to (B, N, N, H) – view is cheap because `out` is contiguous.
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for the heavy work
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] – FP16

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].half())   # [B, N, N, hidden]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4) Optional mask – fused before gating
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(left.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].half()).sigmoid()  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core cubic update – batched matmul via Triton
    # ------------------------------------------------------------------
    # Choose tile sizes. Larger tiles are beneficial when N > 384.
    if left.shape[1] <= 384:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 64

    # The Triton kernel works directly on the original layout – no permute copy.
    out = _batched_trimul_triton(
        left,
        right,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B, N, N, hidden] – FP32

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over the hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # FP32

    # ------------------------------------------------------------------
    # 9) Output gating + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.float()
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim] – FP32

    return out
```
