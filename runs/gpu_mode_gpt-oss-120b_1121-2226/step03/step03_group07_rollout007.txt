"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module but moves the
heavy N³ “pairwise‑multiply” into a custom Triton kernel.  The steps are:

1️⃣  Layer‑norm over the last dimension (FP32, same as reference).
2️⃣  Cast to fp16 – all subsequent linear‑algebra is performed in half‑precision.
3️⃣  Six linear layers (two projections, three gates, final projection).  The
    projection and gate weights are stored in fp16; the final projection stays
    in fp32.
4️⃣  Optional mask – a tiny Triton kernel casts the fp32 mask to fp16 and
    broadcasts it over the hidden dimension.
5️⃣  Core operation – for every hidden channel d we need

       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

    which is a batched matrix‑multiply  L @ Rᵀ  with shape
    (B·D, N, N).  The Triton kernel computes this with fp16 inputs,
    fp32 accumulation and Tensor‑Core friendly tile sizes (128×128×64).

6️⃣  Layer‑norm over the hidden dimension, sigmoid output‑gate, and final
    fp32 linear projection back to the original feature dimension.

The function `custom_kernel` conforms to the required signature:
    (input, mask, weights, config) → output.
Only the mask‑cast and the N³ multiplication are written in Triton; the
remaining ops leverage highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# 1️⃣  Tiny Triton kernel: cast FP32 mask → FP16
# -----------------------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,            # fp32 source pointer
    dst_ptr,            # fp16 destination pointer
    numel,              # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)


# -----------------------------------------------------------------------
# 2️⃣  Batched half‑precision GEMM with fp32 accumulation
#     Computes   C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]    (i.e. L @ Rᵀ)
# -----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides (batch, row, col)
    stride_bh, stride_bn, stride_bk,   # B strides (batch, col, row) – see notes below
    stride_ch, stride_cm, stride_cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Layout expectations
    -------------------
    * A : (batch, M, K)    – row‑major (row index = M, col index = K)
    * B : (batch, N, K)    – row‑major (first dim = N, second = K)
    * In the GEMM we need Bᵀ, i.e. element B_T[k, j] = B[j, k].
      The kernel loads B with the K index as the *row* dimension and the
      N index as the *column* dimension, which exactly yields B_T.
    """

    pid_b = tl.program_id(2)          # batch   (B * hidden_dim)
    pid_m = tl.program_id(0)          # output row block
    pid_n = tl.program_id(1)          # output col block

    # ---------------------------------------------------------------
    # Tile offsets
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ---------------------------------------------------------------
    # Accumulator (fp32)
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # k‑offset in the original K axis
        k_offset = k * BLOCK_K

        # --------------------------------------------------------------------
        # Load tiles of A and B (with out‑of‑bounds masking)
        # --------------------------------------------------------------------
        # A tile  : shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + pid_b * stride_ah
            + offs_m[:, None] * stride_am
            + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        )
        a_mask = (offs_m[:, None] < M) & ((k_offset + tl.arange(0, BLOCK_K))[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)

        # B tile  : shape (BLOCK_K, BLOCK_N)
        # note: we index B as B[j, k] → using stride_bn for the N‑index (j)
        # and stride_bk for the K‑index.
        b_ptrs = (
            B_ptr
            + pid_b * stride_bh
            + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b_mask = ((k_offset + tl.arange(0, BLOCK_K))[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # --------------------------------------------------------------------
        # Fused dot: fp16·fp16 → fp32 accumulation
        # --------------------------------------------------------------------
        acc += tl.dot(a, b)

    # --------------------------------------------------------------------
    # Write out the result (still in fp32)
    # --------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor,
                          BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B : fp16 tensors with shape (batch, M, K) and (batch, N, K)
    Returns C : fp32 tensor with shape (batch, M, N) = A @ Bᵀ
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    # Output buffer (fp32)
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element counts (needed by Triton)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# -----------------------------------------------------------------------
# Entry point required by the evaluation harness
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim]  (float32)
        - mask         : torch.Tensor [bs, seq_len, seq_len]      (float32)
        - weights      : dict of model weights (float32 tensors)
        - config       : dict containing at least ``dim``, ``hidden_dim`` and
                         optional ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device

    dim = config["dim"]                # input feature dimension
    hidden_dim = config["hidden_dim"]  # projection dimension (H)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32) – identical to the reference
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣  Cast to fp16 for the heavy linear algebra
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣  Load linear weights (cast to half where appropriate)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    # Output‑norm and final projection remain fp32 (as in the reference)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]      # (dim, hidden_dim) fp32

    # ------------------------------------------------------------------
    # 4️⃣  Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)    # [B, N, N, hidden]   fp16
    right = F.linear(x_h, right_proj_w)   # [B, N, N, hidden]   fp16

    # ------------------------------------------------------------------
    # 5️⃣  Optional mask – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask from fp32 to fp16 with a tiny Triton kernel
        mask_flat = mask.view(-1)
        mask_fp16_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)

        BLOCK = 1024
        grid = ((mask_flat.numel() + BLOCK - 1) // BLOCK,)
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_fp16_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK,
        )
        mask_h = mask_fp16_flat.view(mask.shape + (1,))   # [B, N, N, 1] fp16
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣  Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, hidden] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, hidden] fp16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B, N, N, hidden] fp16

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣  Core N³ operation – batched GEMM L @ Rᵀ (FP16→FP32)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # B, seq_len, seq_len, hidden
    batch = B * H

    # Rearrange tensors to (batch, M, K) / (batch, N, K) format expected
    # left: (B, N, N, H) → (B·H, N, N)  where the last two dims are (i,k)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    # right: same layout; the kernel treats the second dim as the “N” dimension
    # and the third dim as “K”.  The loading pattern inside the kernel yields
    # the required transpose (Rᵀ) automatically.
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    # Choose tile sizes adaptively – larger tiles for larger N give better
    # utilisation on H100 Tensor‑Cores.
    if N >= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64

    out_mat = _batched_matmul_triton(
        left_mat, right_mat,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )  # [B*H, N, N] fp32

    # Restore original shape: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 8️⃣  Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp32

    # ------------------------------------------------------------------
    # 9️⃣  Output gate (cast to fp32) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)      # element‑wise gate
    out = F.linear(out, to_out_w)               # fp32 → [B,N,N,dim]

    return out
