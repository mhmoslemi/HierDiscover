    # using weight and bias
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x_h = x.to(torch.float16)

    # Build combined weight for 5 linear transforms
    W_combined = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"]
        ],
        dim=0,
    ).to(torch.float16)

    # Flatten x_h for linear
    B, N, N2, D = x_h.shape
    X_flat = x_h.view(-1, D)   # shape (B*N*N, D)
    Y_flat = F.linear(X_flat, W_combined)  # shape (B*N*N, 5*H)

    # reshape to B,N,N,5*H
    Y = Y_flat.view(B, N, N, 5*hidden_dim)
    left = Y[..., :hidden_dim]
    right = Y[..., hidden_dim:2*hidden_dim]
    left_gate_pre = Y[..., 2*hidden_dim:3*hidden_dim]
    right_gate_pre = Y[..., 3*hidden_dim:4*hidden_dim]
    out_gate_pre = Y[..., 4*hidden_dim:]

    # mask
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    # gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left * left_gate
    right = right * right_gate

    # Core matmul
    B, N, N2, H = left.shape
    # reshape left/right to (B*H,N,N)
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

    out_mat = _batched_matmul_triton(left_mat, right_mat)   # fp16
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # output layernorm
    out = F.layer_norm(out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"].to(torch.float16),
            bias=weights["to_out_norm.bias"].to(torch.float16)
    )

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))
    out = out.to(torch.float32)
    return out

"""
TriMul “outgoing” Triton kernel (AlphaFold3)

Algorithm
---------
1. Input LayerNorm (FP32) → x      [B, N, N, D]
2. Cast to fp16 for the heavy work.
3. Fuse the five linear transforms (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single GEMM:
        X_flat (B·N·N, D) @ W_all.T (D, 5·H) → Y (B·N·N, 5·H)
   Split Y into left, right, left_gate, right_gate, out_gate.
4. Apply optional binary mask (broadcasted) to left/right,
   then apply sigmoid gates.
5. Core cubic update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched mat‑mul L @ Rᵀ with a custom
   Triton kernel (fp16 inputs, fp32 accumulation, fp16 output).
6. Output LayerNorm (fp16), multiply by out_gate and project back
   to the original dimension.
7. Cast the final result to FP32 (required by the API).

All expensive arithmetic (the N³ update and the large fused GEMM)
uses fp16 for bandwidth, with fp32 accumulation for numerical stability.
The Triton kernel tiles the work with (BLOCK_M×BLOCK_N×BLOCK_K) = (128×128×32)
and runs with 8 warps / 4 stages – a good baseline on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------------
# Triton kernel: batched mat‑mul A (M×K) * B (N×K)ᵀ → C (M×N)
#   A, B : fp16 (shape: batch, M, K) and (batch, N, K)
#   C    : fp16 (shape: batch, M, N) – result is cast from fp32 acc.
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                  # pointers
    M, N, K,                              # matrix dimensions
    stride_ah, stride_am, stride_ak,      # A strides: batch, row, col
    stride_bh, stride_bn, stride_bk,      # B strides: batch, row, col
    stride_ch, stride_cm, stride_cn,      # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # ------------------------------------------------------------------
    # Offsets of the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for the current batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # Load B tile (as Bᵀ): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate
        acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # Store the result (cast to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 32,
) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) (fp16).
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in element units
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C


# ------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data : tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (high‑precision FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to fp16 for all subsequent heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)  # [B, N, N, D]  (fp16)

    # --------------------------------------------------------------
    # 4) Fuse the five linear transformations (proj + gates)
    # --------------------------------------------------------------
    # Stack weights: each is (hidden_dim, dim) → we concatenate along out_features
    W_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*H, D)

    B, N, _, D = x_f16.shape

    # Flatten the spatial dimensions → (B·N·N, D)
    x_flat = x_f16.view(-1, D)  # no copy, just a view

    # Single GEMM → (B·N·N, 5·H)
    y_flat = F.linear(x_flat, W_all)  # fp16, bias‑free

    # Reshape to a 4‑D tensor to obtain the five slices
    y = y_flat.view(B, N, N, 5 * hidden_dim)

    left      = y[..., :hidden_dim]                       # [B,N,N,H]
    right     = y[..., hidden_dim:2 * hidden_dim]         # [B,N,N,H]
    left_g    = y[..., 2 * hidden_dim:3 * hidden_dim]    # pre‑sigmoid
    right_g   = y[..., 3 * hidden_dim:4 * hidden_dim]
    out_gate  = y[..., 4 * hidden_dim:]                   # pre‑sigmoid

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcasted)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 6) Sigmoid gates (still fp16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_g)
    right_gate = torch.sigmoid(right_g)
    out_gate   = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 7) Apply gates to the projected tensors
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8) Core cubic update (batched mat‑mul)
    # --------------------------------------------------------------
    # Rearrange to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched mat‑mul: L @ Rᵀ  (All in fp16, acc in fp32)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )  # [B·H, N, N] (fp16)

    # Restore the original 4‑D layout
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 9) Output LayerNorm (fp16), out‑gate multiplication, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )  # fp16

    out = out * out_gate  # element‑wise gating (fp16)

    out = F.linear(out, weights["to_out.weight"].to(torch.float16))  # back to D

    # --------------------------------------------------------------
    # 10) Cast back to FP32 for the API contract
    # --------------------------------------------------------------
    return out.to(torch.float32)
