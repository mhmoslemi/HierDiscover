def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm in FP32
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Cast to FP16
    x_f16 = x.half()

    # Projections
    left = torch.nn.functional.linear(x_f16, weights["left_proj.weight"].half())
    right = torch.nn.functional.linear(x_f16, weights["right_proj.weight"].half())

    # Mask
    if not nomask and mask is not None:
        left = left * mask.unsqueeze(-1).half()
        right = right * mask.unsqueeze(-1).half()

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_f16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_f16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_f16, weights["out_gate.weight"].half()))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape

    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Computation using Triton
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=256,
        BLOCK_N=256,
        BLOCK_K=64,
    )

    # Reshape output to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm (FP16)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"].half(),
        bias=weights["to_out_norm.bias"].half(),
    )

    out = out * out_gate

    # Final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"].half())

    # Return FP32
    return out.float()

TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated core cubic update.

The heavy N³ operation is a batched matrix multiply
    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
for every hidden channel *h* (hidden_dim).  We reshape the tensors so that each
pair (b,h) becomes an independent batch element and launch a custom Triton

All remaining work (LayerNorms, linear projections, sigmoid gates and the final
output projection) is done with PyTorch in fp16 to keep the memory bandwidth
low.  The final result is returned in fp32 as required by the API.

The implementation avoids any Python loops and uses only one Triton launch,
so the total runtime is dominated by the GEMM and comfortably fits the
target budget on an H100.
"""

import torch.nn.functional as F

import triton.language as tl

# ----------------------------------------------------------------------
@triton.jit

    A_ptr, B_ptr, C_ptr,                     # base pointers

    stride_ah, stride_am, stride_ak,         # A strides (batch, row, col)
    stride_bh, stride_bn, stride_bk,         # B strides (batch, row, col)
    stride_ch, stride_cm, stride_cn,         # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,

    """
    Classic tiled GEMM with fp16 inputs and fp32 accumulation.
    The kernel works for any (potentially non‑contiguous) layout as long as
    the strides are provided correctly.
    """
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # ------------------------------------------------------------------
    # Offsets in each dimension
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # Pointers to the start of the current batch
    # ------------------------------------------------------------------
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main GEMM loop over K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        cur_k = k_base + tl.arange(0, BLOCK_K)

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)          # fp16

        # Load B tile (as Bᵀ): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        mask_b = (cur_k[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)          # fp16

        # Dot product – fp32 accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 256,
    BLOCK_N: int = 256,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.  A and B must be fp16 tensors on the same
    CUDA device with shapes (batch, M, K) and (batch, N, K) respectively.
    Returns C of shape (batch, M, N) in fp16.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides are expressed in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,          # H100 prefers many warps per block
        num_stages=4,
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D]  (float32)
            - mask         : torch.Tensor [B, N, N]    (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (high‑precision FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.half()   # [B, N, N, D]  (fp16)

    # --------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w)   # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 5) Optional mask (broadcasted)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask ∈ {0,1} – convert once to fp16 and broadcast
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 6) Gates (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # --------------------------------------------------------------
    # 7) Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8) Core N³ batched mat‑mul (outgoing update)
    # --------------------------------------------------------------
    B, N, _, H = left.shape   # B, seq_len, _, hidden_dim

    # Reshape to (B·H, N, N) – each hidden channel becomes an independent batch.
    # The permutation moves the hidden dimension next to batch; we then make the
    # data contiguous so each (N × N) matrix is row‑major for the Triton GEMM.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # The GEMM expects Bᵀ, so we transpose the second matrix.
    right_mat_t = right_mat.permute(0, 2, 1).contiguous()

    # Perform the fused batched mat‑mul in FP16 (accumulated in FP32).
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat_t,
        BLOCK_M=256,
        BLOCK_N=256,
        BLOCK_K=64,
    )   # [B·H, N, N]  (fp16)

    # Reshape back to [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # --------------------------------------------------------------
    # 9) Output LayerNorm (fp16), out‑gate, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    out = out * out_gate    # element‑wise gate (fp16)

    to_out_w = weights["to_out.weight"].half()
    out = F.linear(out, to_out_w)   # fp16, result shape [B, N, N, D]

    # --------------------------------------------------------------
    # 10) Cast back to FP32 – required by the public API
    # --------------------------------------------------------------
    return out.float()
