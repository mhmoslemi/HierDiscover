"""
custom_kernel – Triton‑accelerated outgoing TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Layer‑norm on the input (float32) → cast to FP16.
2️⃣ Fuse *all* per‑position linear layers (left/right projections, their gates,
   and the output gate) into a **single** matmul:
        combined = x_h @ W_combined.T   (W_combined ∈ ℝ⁽⁵·H⁾ˣC)
   Afterwards slice the 5·H channels into
        left, left_gate, right, right_gate, out_gate.
3️⃣ Sigmoid the three gate tensors.
4️⃣ Fuse mask (if any) and the two gates into the projected tensors with a tiny
   Triton kernel (in‑place:  left = left * left_gate * mask,
                     right = right * right_gate * mask ).
5️⃣ Rearrange `left` and `right` so that the cubic contraction becomes a batch of
   matrix multiplications:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is performed by a single `torch.bmm` on tensors of shape (B·H, N, N).
6️⃣ Second LayerNorm over the hidden dimension, apply the output‑gate and a final
   linear projection back to `dim`.  The final output is float32.

Only the element‑wise mask + gate fusion is written in Triton; the heavy
computations (matmul, GEMM, LayerNorm) use highly‑optimized PyTorch kernels.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

def _mask_gate_fuse_kernel(

    right_gate_ptr,    # fp16 * (B·N·N·H)
    mask_ptr,          # fp16 * (B·N·N)  (broadcasted over H)
    total_elements,    # i64  = B·N·N·H
    hidden_dim,        # i64  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    """
    In‑place:  left  = left  * left_gate  * mask
              right = right * right_gate * mask
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # load tensors
    l  = tl.load(left_ptr + offs,        mask=in_range, other=0.0)
    r  = tl.load(right_ptr + offs,       mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)

    # optional mask (broadcasted H‑wise)
    if HAS_MASK:
        # each consecutive block of hidden_dim elements shares the same mask
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # fuse
    l = l * lg * m
    r = r * rg * m

    # store back in‑place
    tl.store(left_ptr + offs,  l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H

    # ------------------------------------------------------------------
    # 2️⃣ load weights (FP32 → FP16 where possible)
    # ------------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear maps (no bias) – cast to FP16
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # Output LayerNorm & final projection (remain FP32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                 # [B, N, N, dim]  float32
    x_h = x.half()                    # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Fuse *all* per‑position linear layers into a single matmul
    # ------------------------------------------------------------------
    # concat weights:  [left_proj, left_gate, right_proj, right_gate, out_gate]
    w_combined = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                 # (5*H, dim)  fp16
    # one big linear
    combined = F.linear(x_h, w_combined)    # [B, N, N, 5*H]  fp16

    H = hidden_dim
    # slice the five parts
    left      = combined[..., 0 * H : 1 * H]   # [B,N,N,H]
    left_gate = combined[..., 1 * H : 2 * H]
    right     = combined[..., 2 * H : 3 * H]
    right_gate= combined[..., 3 * H : 4 * H]
    out_gate  = combined[..., 4 * H : 5 * H]

    # sigmoid the three gate tensors (in‑place to save memory)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse mask + gates with Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        # mask to fp16 and flatten
        mask_h = mask.to(dtype=torch.float16).contiguous()
        mask_flat = mask_h.view(-1)                     # (B·N·N)

        # flatten tensors for the kernel
        left_flat   = left.view(-1)
        right_flat  = right.view(-1)
        lg_flat     = left_gate.view(-1)
        rg_flat     = right_gate.view(-1)

        total_elements = left_flat.numel()               # B·N·N·H
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            lg_flat,
            rg_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=HAS_MASK,
        )
        # left/right now contain the final masked&gated values
        del left_gate, right_gate
    else:
        # no mask → simple element‑wise fusion in PyTorch
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Prepare tensors for the cubic contraction (batched GEMM)
    # ------------------------------------------------------------------
    # left  : (B, N, N, H) → (B, H, N, N)   (axes i,k)
    # right : (B, N, N, H) → (B, H, N, N)   (axes k,j)   (swap last two N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)  fp16
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Batched matrix multiplication (leverages Tensor‑cores)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)          # (B·H, N, N) fp16

    # restore original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = out.float()  # up‑cast for numerically stable LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                 # [B, N, N, H] float32

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate (float32) and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.float()
    out = F.linear(out, to_out_w)      # [B, N, N, dim] float32

    return out
