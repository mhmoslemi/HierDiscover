def layernorm_kernel(x_ptr, out_ptr, weight_ptr, bias_ptr, M, D, stride_xm, stride_xd, stride_outm, stride_outd, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    if pid >= M: return
    col = tl.arange(0, BLOCK_D)
    mask = col < D
    # compute base pointers
    x_ptrs = x_ptr + pid * stride_xm + col * stride_xd
    # load
    x = tl.load(x_ptrs, mask=mask, other=0.0)
    # compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / D
    # compute variance: np.mean(x^2) - mean^2
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) / D
    inv_std = 1.0 / tl.sqrt(var + 1e-5)
    # normalize
    y = (x - mean) * inv_std
    # apply affine
    if weight_ptr is not None:
        w = tl.load(weight_ptr + col * 1, mask=mask, other=1.0)  # weight shape D
        y = y * w
    if bias_ptr is not None:
        b = tl.load(bias_ptr + col * 1, mask=mask, other=0.0)
        y = y + b
    # store
    out_ptrs = out_ptr + pid * stride_outm + col * stride_outd
    tl.store(out_ptrs, y, mask=mask)

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    M, D,
    stride_xm, stride_xd,
    stride_outm, stride_outd,
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    row = tl.program_id(0)
    # mask to ensure row < M
    if row >= M:
        return
    col = tl.arange(0, BLOCK_D)
    mask = col < D
    x_ptrs = x_ptr + row * stride_xm + col * stride_xd
    x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)
    # Compute sum and sum of squares
    sum1 = tl.sum(x, axis=0)   # scalar
    sum2 = tl.sum(x * x, axis=0)
    mean = sum1 / D
    var = sum2 / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    b = tl.load(bias_ptr + col, mask=mask, other=0.0).to(tl.float32)

```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(x_ptr, out_ptr, weight_ptr, bias_ptr,
                     M, D,
                     stride_xm, stride_xd,
                     stride_outm, stride_outd,
                     eps: tl.constexpr, BLOCK_D: tl.constexpr):
    row = tl.program_id(0)
    if row >= M:
        return
    # load a row of size D
    col = tl.arange(0, BLOCK_D)
    mask = col < D
    x_ptrs = x_ptr + row * stride_xm + col * stride_xd
    x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)

    # compute mean and variance (fs)
    sum1 = tl.sum(x, axis=0)       # scalar
    sum2 = tl.sum(x * x, axis=0)   # scalar
    mean = sum1 / D
    var = sum2 / D - mean * mean
    var = tl.maximum(var, 0.0)     # guard against negative due to fp16 rounding
    inv_std = 1.0 / tl.sqrt(var + eps)

    # normalize
    y = (x - mean) * inv_std

    # weight & bias (fp16 -> fp32)
    w = tl.load(weight_ptr + col, mask=mask, other=1.0).to(tl.float32)
    b = tl.load(bias_ptr + col, mask=mask, other=0.0).to(tl.float32)
    y = y * w + b

    # cast back to fp16
    y_fp16 = y.to(tl.float16)

    out_ptrs = out_ptr + row * stride_outm + col * stride_outd
    tl.store(out_ptrs, y_fp16, mask=mask)
```

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of TriMul (outgoing) module.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # cast weights to fp16 once
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)

    left_proj_w = torch.cat([weights["left_proj.weight"], weights["right_proj.weight"]], dim=0).to(torch.float16)  # (2H, D)
    gate_w = torch.cat([weights["left_gate.weight"], weights["right_gate.weight"], weights["out_gate.weight"]], dim=0).to(torch.float16)  # (3H, D)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (dim, H)

    B, N, _, _ = input_tensor.shape  # input shape [B,N,N,dim]

    # -------------------------------------------------
    # 1. Input layer‑norm (Triton)
    # -------------------------------------------------
    # flatten to (M, D)
    x_flat = input_tensor.view(-1, dim).to(torch.float16)  # cast to half after LN (still fp16)
    # allocate output
    x_norm_flat = torch.empty_like(x_flat)
    M_input = x_flat.shape[0]
    eps = 1e-5
    # launch kernel
    grid = (M_input,)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        norm_w,
        norm_b,
        M_input,
        dim,
        dim, 1,  # stride_xm, stride_xd
        dim, 1,  # stride_outm, stride_outd
        eps=eps,
        BLOCK_D=dim,
        num_warps=max(4, (dim // 32)),
    )
    # reshape back to [B, N, N, dim] (still fp16)
    x_norm = x_norm_flat.view(B, N, N, dim)

    # -------------------------------------------------
    # 2. Linear projections (fused)
    # -------------------------------------------------
    proj = F.linear(x_norm, left_proj_w)   # [B,N,N,2*H] fp16
    left, right = torch.split(proj, hidden_dim, dim=-1)

    gates = F.linear(x_norm, gate_w)       # [B,N,N,3*H] fp16
    left_gate, right_gate, out_gate = torch.split(gates, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # -------------------------------------------------
    # 3. Optional mask routing (element‑wise, no extra kernel)
    # -------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # Apply gates (in‑place)
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 4. Core N³ update: batched mat‑mul (torch.bmm)
    # -------------------------------------------------
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16, shape (B*H, N, N)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # -------------------------------------------------
    # 5. Hidden layer‑norm (Triton)
    # -------------------------------------------------
    out_flat = out.view(-1, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)
    M_hidden = out_flat.shape[0]
    grid_hidden = (M_hidden,)
    layernorm_kernel[grid_hidden](
        out_flat,
        out_norm_flat,
        to_out_norm_w,
        to_out_norm_b,
        M_hidden,
        hidden_dim,
        hidden_dim, 1,
        hidden_dim, 1,
        eps=eps,
        BLOCK_D=hidden_dim,
        num_warps=max(4, (hidden_dim // 32)),
    )
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)

    # -------------------------------------------------
    # 6. Out‑gate multiplication
    # -------------------------------------------------
    out_norm = out_norm * out_gate

    # -------------------------------------------------
    # 7. Final projection back to dim
    # -------------------------------------------------
    out_final = F.linear(out_norm.view(-1, hidden_dim), to_out_w)  # fp16
    out_final = out_final.view(B, N, N, dim)

    # cast to float32 for result
    return out_final.to(torch.float32)

"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Input LayerNorm (Triton) – operates on the original fp32 tensor,
   writes the normalized result directly as fp16.
2️⃣ Fuse the two projection Linear layers (left/right) and the three
   gate Linear layers (left_gate, right_gate, out_gate) into two
   matrix‑multiplications (dim → 2·hidden, dim → 3·hidden).
3️⃣ Apply the optional mask (element‑wise) and the per‑channel gates.
4️⃣ Core N³ update – a batched GEMM on half‑precision tensors
   (B·hidden, N, N) @ (B·hidden, N, N)ᵀ using torch.bmm (Tensor‑Cores).
5️⃣ Hidden LayerNorm (Triton) on the intermediate hidden dimension.
6️⃣ Multiply by the out‑gate, then a final Linear projection back to *dim*.
All heavy arithmetic is kept in fp16; only the final tensor is cast back
to fp32 for the required output format.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: Layer‑Norm (works for fp16 or fp32 input,
#                weight & bias are given in float32).
# ------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,                # fp16 / fp32  [M, D] input
    out_ptr,              # fp16         [M, D] output
    weight_ptr,           # fp32         [D]   (γ)
    bias_ptr,             # fp32         [D]   (β)
    M, D,                 # problem size
    stride_xm, stride_xd,  # strides of input   (row‑major → (D,1))
    stride_outm, stride_outd,  # strides of output (same as input)
    eps: tl.constexpr,    # numeric epsilon
    BLOCK_D: tl.constexpr,  # compile‑time block size = D (≤384)
):
    """Layer‑norm over the last dimension.

    Each program processes ONE row (i.e. one “*…*” element of the
    original 4‑D tensor).  All reductions are performed in float32
    for numerical stability and the result is written back as fp16.
    """
    row = tl.program_id(0)                # 0 ≤ row < M
    if row >= M:
        return

    # ----------------------------------------------------------------
    # 1) load one row (size D) – mask for the tail when D < BLOCK_D
    # ----------------------------------------------------------------
    col = tl.arange(0, BLOCK_D)           # thread index over D
    mask = col < D
    x_ptrs = x_ptr + row * stride_xm + col * stride_xd
    x = tl.load(x_ptrs, mask=mask, other=0.0).to(tl.float32)  # fp32 accumulator

    # ----------------------------------------------------------------
    # 2) mean & variance (float32)
    # ----------------------------------------------------------------
    sum1 = tl.sum(x, axis=0)                 # Σ x_i
    sum2 = tl.sum(x * x, axis=0)             # Σ x_i²
    mean = sum1 / D
    var = sum2 / D - mean * mean
    var = tl.maximum(var, 0.0)               # guard against tiny negatives
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ----------------------------------------------------------------
    # 3) normalize, apply affine (γ,β)
    # ----------------------------------------------------------------
    y = (x - mean) * inv_std
    w = tl.load(weight_ptr + col, mask=mask, other=1.0).to(tl.float32)   # γ
    b = tl.load(bias_ptr + col, mask=mask, other=0.0).to(tl.float32)   # β
    y = y * w + b

    # ----------------------------------------------------------------
    # 4) write result as fp16
    # ----------------------------------------------------------------
    y_fp16 = y.to(tl.float16)
    out_ptrs = out_ptr + row * stride_outm + col * stride_outd
    tl.store(out_ptrs, y_fp16, mask=mask)


# ------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Parameters
    ----------
    data : Tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor[float32] of shape [B, N, N, dim]
        - mask         : torch.Tensor[bool/int] of shape [B, N, N] (may be None)
        - weights      : dict containing the model parameters (float32)
        - config       : dict with keys “dim”, “hidden_dim”, optional “nomask”

    Returns
    -------
    torch.Tensor[float32] of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input feature dimension (D)
    hidden_dim = config["hidden_dim"]  # hidden projection size (H)
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------
    # 0️⃣ Cast static weights (only the large Linear‑weights) to fp16 once
    # --------------------------------------------------------------------
    # fused projection weight: [2*H, D]
    left_proj_w = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)
    # fused gate weight: [3*H, D]
    gate_w = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)

    # final projection weight: [dim, H]
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------------
    # 1️⃣ Input LayerNorm  (Triton, fp16 output)
    # --------------------------------------------------------------------
    # flatten to [M, D] where M = B·N·N
    B, N, _, _ = input_tensor.shape
    M_input = B * N * N
    x_flat = input_tensor.view(M_input, dim)                # fp32 view
    x_norm_flat = torch.empty((M_input, dim), dtype=torch.float16, device=device)

    eps = 1e-5
    grid = (M_input,)
    _layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        M_input,
        dim,
        dim, 1,                      # stride_xm, stride_xd
        dim, 1,                      # stride_outm, stride_outd
        eps=eps,
        BLOCK_D=dim,
        num_warps=max(4, dim // 32),
    )
    x_norm = x_norm_flat.view(B, N, N, dim)   # fp16 tensor ready for the projected MLPs

    # --------------------------------------------------------------------
    # 2️⃣ Linear projections (fused) – all in fp16
    # --------------------------------------------------------------------
    proj = F.linear(x_norm, left_proj_w)               # [B,N,N,2*H] fp16
    left, right = torch.split(proj, hidden_dim, dim=-1)  # each [B,N,N,H]

    gates = F.linear(x_norm, gate_w)                   # [B,N,N,3*H] fp16
    left_gate, right_gate, out_gate = torch.split(gates, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # --------------------------------------------------------------------
    # 3️⃣ Optional mask + per‑channel gating (pure torch, no extra copy)
    # --------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → broadcast over the hidden dimension
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f16
        right = right * mask_f16

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 4️⃣ Core N³ update – batched GEMM on half‑precision tensors
    # --------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N) fp16
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H] fp16

    # --------------------------------------------------------------------
    # 5️⃣ Hidden LayerNorm (Triton, still half‑precision)
    # --------------------------------------------------------------------
    M_hidden = B * N * N
    out_flat = out.view(M_hidden, hidden_dim)                 # fp16
    out_norm_flat = torch.empty_like(out_flat)                # fp16

    grid_hidden = (M_hidden,)
    _layernorm_kernel[grid_hidden](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        M_hidden,
        hidden_dim,
        hidden_dim, 1,
        hidden_dim, 1,
        eps=eps,
        BLOCK_D=hidden_dim,
        num_warps=max(4, hidden_dim // 32),
    )
    out_norm = out_norm_flat.view(B, N, N, hidden_dim)      # fp16

    # --------------------------------------------------------------------
    # 6️⃣ Apply out‑gate
    # --------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # --------------------------------------------------------------------
    # 7️⃣ Final projection back to the original dimension
    # --------------------------------------------------------------------
    out_final = F.linear(out_norm.view(-1, hidden_dim), to_out_w)  # fp16
    out_final = out_final.view(B, N, N, dim)

    # ------------------------------------------------------------
    # Cast back to float32 (required by the harness)
    # ------------------------------------------------------------
    return out_final.to(torch.float32)
