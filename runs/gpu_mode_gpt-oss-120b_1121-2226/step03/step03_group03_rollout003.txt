"""
custom_kernel – Triton‑accelerated outgoing Triangle‑Multiplicative update
=======================================================================

The implementation follows the PyTorch reference but moves the expensive
N³ “outgoing” update into a fused batched GEMM kernel written in Triton.
Key optimisations:

*   Fuse the left/right linear projections with their corresponding gates
    (4 matmuls → 2 matmuls) by concatenating the weight matrices.
*   Cast the heavy linear work to FP16 (Tensor‑core friendly) while keeping
    the surrounding LayerNorms and the final output in FP32.
*   Optional binary mask is applied in FP16 after the projections.
*   Reshape the tensors so that each hidden channel becomes an independent
    batch dimension (B·H, N, N) and run a highly‑tuned batched GEMM
    (fp16·fp16 → fp32) on the H100.  The kernel uses 8 warps, double‑buffered
    loads and exploits the hardware matrix‑multiply unit.
*   All other operations (LayerNorm, sigmoid, final Linear) are performed
    with PyTorch’s fast kernels.

The overall data flow:
    x      = LayerNorm(input)                           # FP32
    x_f16  = x.half()
    left, left_gate   = fused_linear(x_f16, left+gate)  # FP16
    right, right_gate = fused_linear(x_f16, right+gate) # FP16
    if mask:  left, right = left*mask, right*mask
    left  = left * left_gate
    right = right * right_gate
    out   = batched_matmul(left, right)                 # FP32, N³
    out   = LayerNorm(out)                              # FP32
    out   = out * out_gate (sigmoid)                    # FP32
    out   = Linear(out, to_out.weight)                  # FP32
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (batch, M, K)  FP16
#   B : (batch, N, K)  FP16  (transposed inside the kernel)
#   C : (batch, M, N)  FP32  (fp16·fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,    # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block column
    pid_b = tl.program_id(2)  # batch

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (transposed on‑the‑fly): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # fused fp16·fp16 → fp32 dot product
        acc += tl.dot(a_tile, b_tile)

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(A: torch.Tensor,
                           B: torch.Tensor,
                           BLOCK_M: int = 128,
                           BLOCK_N: int = 128,
                           BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A, B must be FP16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C with shape (batch, M, N) in FP32.
    """
    assert A.is_cuda and B.is_cuda, "Inputs must reside on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects FP16 inputs"

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # 8 warps (256 threads) – good for H100
        num_stages=4,          # double‑buffered loads
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict containing at least
                         "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] FP32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection + gate for left & right paths
    # ------------------------------------------------------------------
    #   left_proj  + left_gate  → weight shape (2*H, D)
    #   right_proj + right_gate → weight shape (2*H, D)
    left_fused_w = torch.cat([weights["left_proj.weight"],
                              weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_fused_w = torch.cat([weights["right_proj.weight"],
                               weights["right_gate.weight"]], dim=0).to(torch.float16)

    # Linear produces [B, N, N, 2*H]
    left_fused = F.linear(x_f16, left_fused_w)   # FP16
    right_fused = F.linear(x_f16, right_fused_w) # FP16

    # Split into projection and gate parts
    H = hidden_dim
    left = left_fused[..., :H]                     # FP16
    left_gate = torch.sigmoid(left_fused[..., H:])  # FP16
    right = right_fused[..., :H]                   # FP16
    right_gate = torch.sigmoid(right_fused[..., H:])  # FP16

    # ------------------------------------------------------------------
    # 5️⃣ Optional binary mask (apply in FP16)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        m = mask.to(torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 6️⃣ Apply per‑channel gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ batched matrix multiplication (outgoing update)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # original batch and sequence length

    # Reorder to (B, H, N, N) and flatten batch*H → (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # FP16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # FP16

    # Choose a K‑tile size that works well for all tested N
    BLOCK_K = 64 if N >= 64 else 32

    # Perform the batched GEMM: left @ rightᵀ
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=BLOCK_K,
    )  # [B·H, N, N] FP32

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP32

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # FP32

    # Out gate (sigmoid)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))  # FP16 → will be up‑cast

    # Multiply by out_gate (broadcast over hidden dim)
    out = out * out_gate  # FP32 * FP16 → FP32

    # Final linear projection back to original dimension D
    out = F.linear(out, weights["to_out.weight"])

    return out
