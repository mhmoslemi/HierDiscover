"""
TriMul “outgoing” forward pass (no backward).

Algorithm
---------
1. Input LayerNorm over the last channel (float32).
2. Cast to fp16 – the heavy linear work (projections + gates) stays in half‑precision.
3. Compute left / right projections (bias‑free linear) and three sigmoid gates.
   If a pair‑wise mask is provided, apply it before the gates.
4. Fuse the gates into the projected tensors.
5. Rearrange the tensors to a batched 3‑D view (B·H, N, N) and launch a
   Triton kernel that computes the batched GEMM
        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]
   i.e. `left @ rightᵀ`.  Inputs are fp16, accumulation is fp32, the result is
   stored in fp32.
6. Reshape the result back to [B, N, N, H].
7. Layer‑norm over the hidden dimension (float32).
8. Apply the output gate and a final bias‑free linear projection back to the
   original dimension D.
9. Return a float32 tensor of shape [B, N, N, D].

Only the N³ matrix multiplication (the true bottleneck) is executed in Triton;
all surrounding operations use highly‑optimized torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : fp16
#   C   : fp32 (accumulated)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N]  (fp16)
    b_ptr,               # right : [BH, N, N]  (fp16) – interpreted as (K,N) i.e. transposed
    c_ptr,               # out   : [BH, N, N]  (fp32)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiply C = A @ Bᵀ with fp16 inputs, fp32 accumulation."""
    pid_h = tl.program_id(axis=2)          # batch * hidden
    pid_m = tl.program_id(axis=0)          # tile row
    pid_n = tl.program_id(axis=1)          # tile column

    # ------------------------ tile offsets ---------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------ loop over K ---------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- A tile (M × K) ---------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ----- Bᵀ tile (K × N) ---------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ----- accumulate (fp16 dot -> fp32) -------------------------
        acc += tl.dot(a, b).to(tl.float32)

    # ----------------------- write back -----------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()                         # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) – fp16 → fp16
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].half())
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].half())
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].half())
    )

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to (B·H, N, N) for the batched GEMM
    # ------------------------------------------------------------------
    #  (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim
    left_flat = left.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right.view(BH, seq_len, seq_len) # fp16, contiguous

    # Allocate output buffer (fp32 accumulation)
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Launch the Triton batched GEMM kernel
    # ------------------------------------------------------------------
    # Tile sizes – chosen to give good occupancy on H100 while staying < 32 KB SMEM.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # tiles along M (i)
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # tiles along N (j)
        BH,                                   # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right – note the transpose)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # stays float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (cast to float32) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)          # [B, N, N, H]
    out = F.linear(out, weights["to_out.weight"])  # bias‑free, back to D (float32)

    return out
```
