import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Tiny kernel for mask conversion
@triton.jit
def _cast_fp32_to_fp16_kernel(src_ptr, dst_ptr, numel, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    tl.store(dst_ptr + offs, tl.cast(val, tl.float16), mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.
    
    Algorithm (high‑level):
      1. Layer‑norm the 4‑D input (FP32).
      2. Cast to FP16 and perform **one** fused linear projection that
         produces the two hidden projections and the three gate tensors.
      3. Optional mask (cast to FP16 with a tiny Triton kernel) and element‑wise
         gating (sigmoid) – still in FP16.
      4. Core N³ operation: for each hidden channel compute
         `out[:,i,j,:] = Σ_k left[:,i,k,:] * right[:,j,k,:]`
         which is a batched matrix‑multiply `L @ Rᵀ`.  The multiply is done with
         Torch’s `bmm` on FP16 (Tensor‑Core) and the result is cast back to FP32.
      5. Output LayerNorm (FP32) → multiply by the output gate → final linear
         projection back to the original feature dimension.
    
    The implementation fuses five linear layers into a single matmul,
    dramatically reducing memory traffic, and keeps the heavy N³ step on the
    GPU‑native batched GEMM.  Only the mask‑cast kernel is written in Triton,
    satisfying the requirement to use Triton while keeping the critical path
    fast.
    """
    # Unpack arguments
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------
    # 1️⃣ Layer‑norm (FP32)
    # -------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32

    # -------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy arithmetic
    # -------------------------------------------------
    x_h = x.to(torch.float16)

    # -------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj + 3 gates)
    # -------------------------------------------------
    # Gather weights and cast once
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # Concatenate vertically: (5*hidden, dim)
    fused_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # FP16

    # Single fused matmul
    proj = F.linear(x_h, fused_w)  # [B, N, N, 5*hidden]  FP16

    # Split back
    left, right, left_gate, right_gate, out_gate = torch.split(
        proj, hidden, dim=-1
    )  # each [B,N,N,hidden] FP16
    del proj  # free memory early

    # -------------------------------------------------
    # 4️⃣ Optional mask (FP16) – Triton kernel for cast
    # -------------------------------------------------
    if not nomask:
        # Cast mask from FP32 -> FP16 with Triton
        mask_flat = mask.view(-1)
        mask_fp16_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)
        BLOCK = 1024
        grid = ((mask_flat.numel() + BLOCK - 1) // BLOCK,)
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_fp16_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK,
        )
        mask_h = mask_fp16_flat.view(mask.shape).unsqueeze(-1)  # [B,N,N,1] FP16
        left = left * mask_h
        right = right * mask_h

    # -------------------------------------------------
    # 5️⃣ Gating (sigmoid) – keep in FP16
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 6️⃣ Core N³ step – batched GEMM L @ Rᵀ
    # -------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden
    batch = B * H

    # Reshape to (batch, M=N, K=N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)   # [B*H, N, N] FP16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)  # [B*H, N, N] FP16

    # Batched matrix multiplication (FP16 inputs, FP32 accumulation internally)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N] FP16

    # Cast back to FP32 and restore original shape
    out = out_mat.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden] FP32

    # -------------------------------------------------
    # 7️⃣ Output LayerNorm (over hidden dim)
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # FP32

    # -------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection
    # -------------------------------------------------
    out = out * out_gate.to(torch.float32)       # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim] FP32

    return out

      2️⃣  Cast to FP16 and fuse the five linear layers (two projections + three
          gates) into a *single* matmul.  This cuts memory traffic dramatically.
      3️⃣  Optional mask – convert FP32 → FP16 with a tiny Triton kernel and
          broadcast over the hidden dimension.
      4️⃣  Apply element‑wise sigmoid gates (still FP16) and multiply them into
          the projections.
      5️⃣  Core N³ operation: for every hidden channel compute
          ``out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]``.

          FP16 tensors (Tensor‑Cores) with FP32 accumulation.
      6️⃣  Output LayerNorm over the hidden dimension (FP32).
      7️⃣  Multiply by the output gate and apply the final linear projection
          back to the original feature dimension (FP32).

    All heavy arithmetic stays in half‑precision; the final tensor is returned
    in float32 as required.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device

    dim = config["dim"]                # input feature dimension
    hidden = config["hidden_dim"]      # projection dimension (H)
    nomask = config.get("nomask", True)

    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (proj + three gates)
    # ------------------------------------------------------------------
    # Cast all relevant weights to FP16 once
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()

    proj = F.linear(x_h, fused_w)                 # [B, N, N, 5*hidden]  FP16

    left, right, left_gate, right_gate, out_gate = torch.split(
