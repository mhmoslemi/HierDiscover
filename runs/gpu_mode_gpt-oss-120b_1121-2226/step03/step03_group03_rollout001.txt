"""
TriMul (outgoing) – Triton‐accelerated forward pass
===================================================

The implementation follows the reference PyTorch TriMul module but moves the
computationally dominant N³ update into a custom Triton kernel.

Algorithm
---------
1. Layer‑Norm over the last dimension of the 4‑D input.
2. Two bias‑free linear projections (left/right) and three sigmoid gates
   (left_gate, right_gate, out_gate) computed from the same normalized input.
3. Optional binary mask is applied to left/right.
4. Element‑wise gating:   left  = left  * left_gate,
                         right = right * right_gate.
5. Core outgoing update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply L @ Rᵀ for each hidden channel.
   We reshape (B,N,N,H) → (B·H,N,N) and run an autotuned Triton GEMM
   (fp16·fp16 → fp32 accumulation).
6. Reshape back to (B,N,N,H) and apply a second Layer‑Norm, the out‑gate,
   and a final linear projection back to the original feature dimension D.
7. The result is returned as float32.

Only the N³ mat‑mul is implemented in Triton; all other ops use
torch.nn.functional for simplicity. Mixed‑precision (fp16 for linear /
gating, fp32 for the final norm/projection) reduces memory traffic while
maintaining numerical quality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) * B (N×K)ᵀ → C (M×N)
#   • A, B are fp16, C is fp32 (fp16·fp16 → fp32 accumulation)
#   • The third launch axis bundles the batch dimension (B·H)
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32},
                      num_stages=3, num_warps=4),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64},
                      num_stages=4, num_warps=8),
        triton.Config({'BLOCK_M': 32, 'BLOCK_N': 64, 'BLOCK_K': 32},
                      num_stages=2, num_warps=2),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,    # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch axis (B·H)
    pid_m = tl.program_id(0)          # block row (M)
    pid_n = tl.program_id(1)          # block col (N)

    # ------------------------------------------------------------------
    # Offsets of the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ---- load A tile (M x K) ----
        a_ptrs = a_batch + \
                 (offs_m[:, None] * stride_am) + \
                 ((k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak)
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # ---- load B tile transposed on‑fly (K x N) ----
        # B is stored as (batch, N, K); we need Bᵀ → (K, N)
        b_ptrs = b_batch + \
                 ((k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk) + \
                 (offs_n[None, :] * stride_bn)
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # ---- fused matmul (fp16·fp16 → fp32) ----
        acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # Write back the result tile
    # ------------------------------------------------------------------
    c_ptrs = c_batch + \
             (offs_m[:, None] * stride_cm) + \
             (offs_n[None, :] * stride_cn)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(A: torch.Tensor,
                           B: torch.Tensor,
                           BLOCK_M: int = 128,
                           BLOCK_N: int = 128,
                           BLOCK_K: int = 64) -> torch.Tensor:
    """
    Wrapper that launches the Triton batched GEMM.

    Parameters
    ----------
    A : torch.Tensor
        fp16 tensor of shape (batch, M, K)
    B : torch.Tensor
        fp16 tensor of shape (batch, N, K)
    BLOCK_M, BLOCK_N, BLOCK_K : int
        Tile sizes (used only as defaults when the autotuner does not
        select a different config).

    Returns
    -------
    C : torch.Tensor
        fp32 tensor of shape (batch, M, N)
    """
    assert A.is_cuda and B.is_cuda, "Inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16"

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, f"Incompatible inner dimensions: {K} vs {Kb}"

    # Allocate output in fp32 (higher precision for the following LayerNorm)
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # fallback; autotuner may override
        num_stages=4,         # double‑buffered loads
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args
    ----
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys `dim`, `hidden_dim`, optional `nomask`

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D] – float32

    # Cast to fp16 for the heavy linear / gate work
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (fp16) and optional mask
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H] fp16

    if not nomask and mask is not None:
        # mask is binary {0,1} – broadcast over hidden dim
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – also fp16
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5) Apply gating
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core N³ batched mat‑mul (left @ rightᵀ) via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B·H, N, N) – each hidden channel is an independent batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Choose a sensible BLOCK_K based on sequence length (helps the autotuner)
    block_k = 64 if N >= 64 else 32

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=block_k,
    )   # [B·H, N, N] – fp32

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    # out_gate is fp16 → broadcast‑cast to fp32
    out = out * out_gate

    # Final linear back to original dimension D
    out = F.linear(out, weights["to_out.weight"])

    return out
```
