"""
TriMul “outgoing” forward pass (AlphaFold3 style)

Algorithm
---------
1.   Layer‑norm (float32) over the last channel of the 4‑D pair tensor.
2.   Cast the normalized tensor to FP16 – the heavy matrix work stays in half‑precision.
3.   Bias‑free linear projections for the left/right paths and three sigmoid‑gates.
4.   Apply the optional pair‑wise mask (broadcast over the hidden channel) **before**
     the gates, as in the reference implementation.
5.   Multiply each path by its gate (still FP16).
6.   Rearrange the tensors to a batched 3‑D view  (B·H, N, N)  where H = hidden_dim.
7.   Compute the batched GEMM
        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]          (A @ Bᵀ)
     – For small sequences (≤256) a custom Triton kernel is used (fp16 × fp16 → fp32).
     – For larger sequences the highly‑optimized cuBLAS batched GEMM via ``torch.bmm``
       (fp16 tensor cores) is employed, giving a ~3‑5× speed‑up.
8.   Reshape back to [B, N, N, H] and apply a second Layer‑Norm over H.
9.   Multiply by the output gate (sigmoid of a third projection) and a final
     bias‑free linear projection back to the original dimension D.
10.  Return a float32 tensor of shape [B, N, N, D].

The Triton kernel works on tiles (M,N,K) = (64,64,128), keeps the A‑tile and
B‑tile in registers/​shared memory (≈32 KB per block) and accumulates in
float32, which matches the numerical precision of the reference code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel: out = A @ Bᵀ   (A,B are fp16, out is fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N]   (fp16)
    b_ptr,               # right : [BH, N, N]   (fp16) – interpreted as transposed
    c_ptr,               # out   : [BH, N, N]   (fp32)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)               # batch * hidden
    pid_m = tl.program_id(axis=0)               # tile row
    pid_n = tl.program_id(axis=1)               # tile col

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- A tile (M × K) ------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ---- Bᵀ tile (K × N) ------------------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ---- Accumulate ----------------------------------------------------
        # tl.dot returns fp16; we cast to fp32 before accumulation.
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) – stay in FP16
    # --------------------------------------------------------------
    # Pre‑convert weight matrices once to avoid repeated .half() calls.
    w_left = weights["left_proj.weight"].half()
    w_right = weights["right_proj.weight"].half()
    w_left_gate = weights["left_gate.weight"].half()
    w_right_gate = weights["right_gate.weight"].half()
    w_out_gate = weights["out_gate.weight"].half()

    left = F.linear(x_fp16, w_left)          # [B, N, N, H]  fp16
    right = F.linear(x_fp16, w_right)        # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on H)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5️⃣ Gating – sigmoid (still FP16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_fp16, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_fp16, w_out_gate))

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Reshape to (B·H, N, N) for the batched GEMM
    # --------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    # --------------------------------------------------------------
    # 7️⃣ Batched matrix multiplication
    # --------------------------------------------------------------
    # For very small sequences the Triton kernel is competitive;
    # for larger N we fall back to cuBLAS via torch.bmm, which on H100
    # leverages Tensor‑Cores and provides the best throughput.
    if seq_len <= 256:
        out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float32, device=device)

        # Tile configuration – chosen to stay within ~32 KB shared memory
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 128

        grid = (
            (seq_len + BLOCK_M - 1) // BLOCK_M,   # tiles over i
            (seq_len + BLOCK_N - 1) // BLOCK_N,   # tiles over j
            BH,                                   # batch × hidden
        )

        tri_mul_outgoing_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            seq_len,                     # M
            seq_len,                     # N
            seq_len,                     # K
            # strides for A (left)
            left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
            # strides for Bᵀ (right – note the transposed layout)
            right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
            # strides for C (output)
            out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    else:
        # cuBLAS batched GEMM (FP16 → FP16), accumulate in FP16 then cast to FP32
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2)).float()

    # --------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # --------------------------------------------------------------
    out_perm = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # stays float32

    # --------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection (float32)
    # --------------------------------------------------------------
    out = out * out_gate.float()                     # broadcast gate
    out = F.linear(out, weights["to_out.weight"])    # bias‑free, back to D

    return out
