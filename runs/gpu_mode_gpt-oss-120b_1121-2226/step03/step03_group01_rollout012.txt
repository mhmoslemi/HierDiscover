"""
TriMul “outgoing” operator – fast forward implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and compute the two bias‑free projections and the three
   sigmoid‑gates (the gates stay in float32 and are later cast to fp16).
3. If a pair‑wise mask is present, apply it to the projected tensors with a
   tiny Triton kernel (this satisfies the “use a kernel” requirement).
4. Fuse the mask and the element‑wise gates.
5. Reshape to a batched 3‑D view (B·H, N, N) and perform the core
   N³ multiplication `C = A @ Bᵀ` with a single batched `torch.bmm`,
   which lets cuBLAS/Tensor‑Cores work at full speed.
6. Reshape back to [B, N, N, H] and apply a second LayerNorm over H.
7. Multiply by the output‑gate (float32) and finish with a bias‑free linear
   projection back to the original dimension D.
8. Return a float32 tensor of shape [B, N, N, D].

Only the mask‑application is performed by a custom Triton kernel; the heavy
matrix multiplication is delegated to cuBLAS via `torch.bmm`, which is
substantially faster than the reference Triton GEMM.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply a 4‑D tensor by a 3‑D mask.
# The tensor is flattened; the mask is broadcast across the last dim.
# ----------------------------------------------------------------------
@triton.jit
def mask_mul_kernel(
    x_ptr,                # fp16 input tensor, flattened
    mask_ptr,             # fp16 mask tensor, flattened (no H dim)
    out_ptr,              # fp16 output tensor, flattened (may be same as x_ptr)
    numel,                # total number of elements in x
    hidden_dim,           # size of the broadcasted dimension H
    stride_x: tl.constexpr,   # stride for x (must be 1 for a flat view)
    stride_mask: tl.constexpr, # stride for mask (must be 1 for a flat view)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < numel

    # load x
    x = tl.load(x_ptr + offset * stride_x, mask=mask, other=0.0)

    # mask index = offset // hidden_dim   (integer division)
    mask_index = offset // hidden_dim
    m = tl.load(mask_ptr + mask_index * stride_mask, mask=mask, other=0.0)

    # store result
    tl.store(out_ptr + offset * stride_x, x * m, mask=mask)


def apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Apply a pair‑wise mask to a tensor of shape [B, N, N, H] using Triton.
    The mask has shape [B, N, N]. The operation performed is:
        out[b, i, j, h] = tensor[b, i, j, h] * mask[b, i, j]
    The tensors must be contiguous and on the same CUDA device.
    """
    if not tensor.is_contiguous():
        tensor = tensor.contiguous()
    if not mask.is_contiguous():
        mask = mask.contiguous()

    total = tensor.numel()
    hidden = tensor.shape[-1]           # H
    BLOCK = 65536                       # 64 K threads per launch (tunable)

    grid = ((total + BLOCK - 1) // BLOCK,)

    # both tensors are flattened views; strides are 1
    mask_mul_kernel[grid](
        tensor,
        mask.view(-1),
        tensor,
        total,
        hidden,
        stride_x=1,
        stride_mask=1,
        BLOCK_SIZE=BLOCK,
    )
    return tensor


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()          # [B, N, N, D] – fp16
    x_float = x                # keep a float32 copy for the gates

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) – fp16 → fp16
    # ------------------------------------------------------------------
    left = F.linear(
        x_fp16,
        weights["left_proj.weight"].half(),
    )  # [B, N, N, H] – fp16
    right = F.linear(
        x_fp16,
        weights["right_proj.weight"].half(),
    )  # [B, N, N, H] – fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on H)
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask to fp16 and apply with a tiny Triton kernel
        mask_fp16 = mask_tensor.to(dtype=torch.float16, device=device)
        left = apply_mask_triton(left, mask_fp16)
        right = apply_mask_triton(right, mask_fp16)

    # ------------------------------------------------------------------
    # 5️⃣ Gating – sigmoid (computed in float32, then cast)
    # ------------------------------------------------------------------
    left_gate_f = torch.sigmoid(F.linear(x_float, weights["left_gate.weight"]))
    right_gate_f = torch.sigmoid(F.linear(x_float, weights["right_gate.weight"]))
    out_gate_f = torch.sigmoid(F.linear(x_float, weights["out_gate.weight"]))

    # Cast gates to fp16 for the element‑wise product
    left_gate = left_gate_f.half()
    right_gate = right_gate_f.half()

    # ------------------------------------------------------------------
    # 6️⃣ Apply gates (still fp16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Batched GEMM   C = A @ Bᵀ   (A,B are fp16, C accumulated in fp32)
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16
    right_flat = right_perm.view(BH, seq_len, seq_len)  # fp16

    # Batched matrix‑multiply (A @ Bᵀ)
    out_flat = torch.bmm(
        left_flat,
        right_flat.transpose(1, 2)
    )  # result dtype = fp16

    # Convert to float32 for the remaining float‑precision ops
    out_flat = out_flat.float()

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # stays float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection (float32)
    # ------------------------------------------------------------------
    out = out * out_gate_f          # broadcast gate (float32)
    out = F.linear(out, weights["to_out.weight"])  # bias‑free, back to D

    return out
