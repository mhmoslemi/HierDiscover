4. Compute three sigmoid gates and fuse them in‑place with the
   projected tensors (still fp16).
5. Reshape to a batched 3‑D view (B·H, N, N) and launch a custom
   Triton kernel that evaluates the batched GEMM

        out = left @ rightᵀ          (C = A @ Bᵀ)

   – tiles: (M,N,K) = (128,128,128) → only two K‑tiles per problem,
   – fp16 inputs, accumulation in fp32, final store in fp32.
6. Reshape back to [B, N, N, H] and apply a second LayerNorm.
7. Apply the output gate and a final bias‑free linear projection back
   to the original dimension D.
8. Return the result as float32.

Only the N³ matrix multiplication (the true bottleneck) runs in
Triton; all surrounding ops use highly‑optimised torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B fp16, C fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left   : [BH, N, N]  (fp16)
    b_ptr,               # right  : [BH, N, N]  (fp16) – interpreted as (K,N)
    c_ptr,               # out    : [BH, N, N]  (fp32)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication `C = A @ Bᵀ`.
    Each program (pid_h, pid_m, pid_n) computes a (BLOCK_M × BLOCK_N) tile
    of one hidden‑channel batch.
    """
    pid_h = tl.program_id(axis=2)          # batch * hidden
    pid_m = tl.program_id(axis=0)          # tile rows
    pid_n = tl.program_id(axis=1)          # tile cols

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiles of size BLOCK_K)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- A tile (M × K) -------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                                   # fp16

        # ---- Bᵀ tile (K × N) -------------------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                   # fp16

        # ---- Accumulate in fp32 -------------------------------------------
        # tl.dot returns fp16 → explicit cast to fp32
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point – “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 0️⃣ ensure we are not tracking gradients (forward‑only)
    # ------------------------------------------------------------------
    with torch.no_grad():
        # ------------------------------------------------------------------
        # 1️⃣ Input LayerNorm (float32)
        # ------------------------------------------------------------------
        x = F.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
        )  # [B, N, N, D] – float32

        # ------------------------------------------------------------------
        # 2️⃣ Cast to fp16 for the heavy linear work
        # ------------------------------------------------------------------
        x_fp16 = x.half()                         # [B, N, N, D] – fp16

        # ------------------------------------------------------------------
        # 3️⃣ Linear projections (bias‑free) – fp16 → fp16
        # ------------------------------------------------------------------
        left = F.linear(x_fp16, weights["left_proj.weight"].to(torch.float16))
        right = F.linear(x_fp16, weights["right_proj.weight"].to(torch.float16))

        # ------------------------------------------------------------------
        # 4️⃣ Optional pair‑wise mask (broadcast on hidden dim)
        # ------------------------------------------------------------------
        if not nomask:
            # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
            mask = mask_tensor.unsqueeze(-1).to(left.dtype)
            left.mul_(mask)      # in‑place to avoid extra allocation
            right.mul_(mask)

        # ------------------------------------------------------------------
        # 5️⃣ Gating – sigmoid (still fp16)
        # ------------------------------------------------------------------
        left_gate = torch.sigmoid(
            F.linear(x_fp16, weights["left_gate.weight"].to(torch.float16))
        )
        right_gate = torch.sigmoid(
            F.linear(x_fp16, weights["right_gate.weight"].to(torch.float16))
        )
        out_gate = torch.sigmoid(
            F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16))
        )

        # Fuse gates in‑place
        left.mul_(left_gate)
        right.mul_(right_gate)

        # ------------------------------------------------------------------
        # 6️⃣ Reshape to [B·H, N, N] for the batched GEMM
        # ------------------------------------------------------------------
        # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
        left = left.permute(0, 3, 1, 2).contiguous()
        right = right.permute(0, 3, 1, 2).contiguous()

        BH = bs * hidden_dim                     # batch × hidden
        left_flat = left.view(BH, seq_len, seq_len)   # fp16, contiguous
        right_flat = right.view(BH, seq_len, seq_len) # fp16, contiguous

        # Output buffer (fp32 accumulation)
        out_flat = torch.empty(
            (BH, seq_len, seq_len), dtype=torch.float32, device=device
        )

        # ------------------------------------------------------------------
        # 7️⃣ Triton GEMM launch (C = left @ rightᵀ)
        # ------------------------------------------------------------------
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 128                       # larger K tile → fewer loop iterations

        grid = (
            (seq_len + BLOCK_M - 1) // BLOCK_M,   # rows
            (seq_len + BLOCK_N - 1) // BLOCK_N,   # cols
            BH,                                   # batch*hidden
        )

        tri_mul_outgoing_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            seq_len,                # M
            seq_len,                # N
            seq_len,                # K
            # strides for A (left)  – (batch, i, k)
            left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
            # strides for Bᵀ (right) – (batch, k, j)
            right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
            # strides for C (out)   – (batch, i, j)
            out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

        # ------------------------------------------------------------------
        # 8️⃣ Reshape back to [B, N, N, H]
        # ------------------------------------------------------------------
        out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
        out = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

        # ------------------------------------------------------------------
        # 9️⃣ Second LayerNorm over the hidden dimension (float32)
        # ------------------------------------------------------------------
        out = F.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
        )  # stays float32

        # ------------------------------------------------------------------
        # 🔟 Apply output gate and final projection back to D
        # ------------------------------------------------------------------
        # out_gate is fp16; broadcasting automatically up‑casts to fp32
        out = out * out_gate
        out = F.linear(out, weights["to_out.weight"])   # bias‑free, float32

        return out
```
