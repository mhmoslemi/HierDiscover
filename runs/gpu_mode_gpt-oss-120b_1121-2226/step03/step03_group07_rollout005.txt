"""
custom_kernel – Triton‑accelerated outgoing TriMul (forward only)

Algorithm
---------
1️⃣ Input LayerNorm (FP32) on the 4‑D tensor.
2️⃣ Cast the normalized tensor to FP16.
3️⃣ Fuse the four linear projections (left_proj, left_gate, right_proj,
   right_gate) into a single GEMM by concatenating the weight matrices.
   Split the result and apply sigmoid to the two gate slices.
4️⃣ (Optional) Apply the pairwise mask – a tiny Triton kernel converts the
   FP32 mask → FP16 and broadcasts it over the hidden dimension.
5️⃣ Element‑wise multiplication of projections with their gates (still FP16).
6️⃣ Core N³ operation: for each hidden channel compute
   out_h[i,j] = Σₖ left_h[i,k]·right_h[j,k].
   Implemented as a batched half‑precision GEMM using `torch.bmm`,
   which on H100 utilizes Tensor‑Cores (FP16·FP16 → FP32 accumulation).
7️⃣ Reshape back to [B,N,N,hidden] and apply a second LayerNorm (FP32).
8️⃣ Multiply by the output gate (sigmoid of a linear projection) and
   a final FP32 linear layer to map back to the original `dim`.
9️⃣ Return the result in float32.

Only the mask‑cast uses a custom Triton kernel; all heavy arithmetic
remains in half‑precision for speed while the final result stays in FP32.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel: cast FP32 mask → FP16 (tiny, but satisfies “kernel” requirement)
# ---------------------------------------------------------------------------
@triton.jit
def _mask_cast_fp32_to_fp16_kernel(
    src_ptr,               # fp32 source
    dst_ptr,               # fp16 destination
    numel,                 # total number of elements
    BLOCK_SIZE: tl.constexpr,

    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)

# ---------------------------------------------------------------------------

    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple

        - mask         : torch.Tensor, shape [bs, seq_len, seq_len] (float32)
        - weights_dict : dict of model weights (float32 tensors)
        - config_dict  : dict containing at least ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)  # [B,N,N,dim]  FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse four linear projections (left/right + gates) into one GEMM
    # ------------------------------------------------------------------
    # Concatenate weight matrices along the output‑feature dimension.
    # Each weight: (hidden_dim, dim) because bias=False.
    fused_w = torch.cat(

            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).contiguous()  # (4*hidden_dim, dim)
    fused_w = fused_w.to(torch.float16)

    # One fused linear: result shape [B, N, N, 4*hidden_dim] (FP16)
    proj_all = F.linear(x_h, fused_w)  # FP16

    H = hidden_dim
    left_proj   = proj_all[..., :H]          # (B,N,N,H)
    left_gate_z = proj_all[..., H:2*H]      # (B,N,N,H)
    right_proj  = proj_all[..., 2*H:3*H]    # (B,N,N,H)
    right_gate_z = proj_all[..., 3*H:]      # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – stays in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_z)   # FP16

    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # FP16

    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # FP16

    # Batched GEMM using torch.bmm (leverages Tensor‑Cores on H100).
    # Result dtype is FP16; we immediately cast to FP32 for the following norm.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))            # (B*H,N,N) FP16

    # 9️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------

    out_fp32 = F.linear(out_fp32, weights["to_out.weight"])  # FP32, shape [B,N,N,dim]

    return out_fp32
