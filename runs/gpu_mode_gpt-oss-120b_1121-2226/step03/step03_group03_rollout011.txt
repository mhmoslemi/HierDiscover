"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (mirrors the PyTorch reference)
-----------------------------------------
1️⃣  Layer‑norm the 4‑D input (B,N,N,D) in float32.
2️⃣  Cast to FP16 and perform **all five** bias‑free linear projections
    (left_proj, right_proj, left_gate, right_gate, out_gate) with a *single*
    large GEMM:  X_flat @ W_concat.T  →  (B·N·N, 5·H).  This removes four extra
    kernel launches and reduces memory traffic.
3️⃣  Extract the five tensors, apply the optional binary mask to *left* and
    *right*, then multiply by the respective sigmoid gates.
4️⃣  Core N³ update (outgoing):
        out₍b,i,j,h₎ = Σₖ left₍b,i,k,h₎ · right₍b,j,k,h₎
    Implemented as a batched FP16·FP16 → FP32 GEMM using a small Triton kernel.
    Each hidden channel h becomes an independent batch (B·H, N, N).
5️⃣  Reshape back to (B,N,N,H), apply a second LayerNorm over H,
    multiply by the out‑gate and finish with a FP16→FP32 linear projection
    back to the original dimension D.

All heavy work (large fused linear and the N³ GEMM) runs in mixed‑precision
FP16 for maximal throughput on H100.  The final result is returned in
float32, matching the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A (M×K) * Bᵀ (N×K) → C (M×N)
#   – A, B are FP16, accumulation in FP32.
#   – Batch dimension is the third launch axis (B·H).
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_ah, stride_am, stride_ak,  # A strides: batch, row (M), col (K)
    stride_bh, stride_bn, stride_bk,  # B strides: batch, row (N), col (K)
    stride_ch, stride_cm, stride_cn,  # C strides: batch, row (M), col (N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid_b = tl.program_id(2)   # batch (B·H)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (transposed on‑the‑fly): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # fused dot‑product (FP16·FP16 → FP32)
        acc += tl.dot(a_tile, b_tile)

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(A: torch.Tensor,
                           B: torch.Tensor,
                           BLOCK_M: int = 128,
                           BLOCK_N: int = 128,
                           BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A,B – FP16 tensors of shape (batch, M, K) and (batch, N, K) respectively.
    Returns C of shape (batch, M, N) in FP32.
    """
    assert A.is_cuda and B.is_cuda, "inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "kernel expects FP16"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data : tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing at least "dim", "hidden_dim",
                             optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # -------------------------------------------------------------
    # unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (D) – float32
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] float32

    # -------------------------------------------------------------
    # 2️⃣ Cast to FP16 once – all heavy work runs in FP16
    # -------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # -------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections (left/right + 3 gates)
    #    into a single GEMM: X_flat @ W_concat.T
    # -------------------------------------------------------------
    # concatenate weight matrices (shape: (5*H, D))
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)  # (5*H, D)

    B, N, _, _ = x_f16.shape
    # flatten to (B*N*N, D)
    x_flat = x_f16.view(-1, dim)                # (B·N·N, D)
    proj_flat = F.linear(x_flat, w_concat)      # (B·N·N, 5·H)  FP16
    proj = proj_flat.view(B, N, N, 5 * hidden_dim)  # (B,N,N,5*H)

    # split results
    left = proj[..., 0:hidden_dim]                                 # (B,N,N,H)
    right = proj[..., hidden_dim:2 * hidden_dim]                   # (B,N,N,H)
    left_gate_raw = proj[..., 2 * hidden_dim:3 * hidden_dim]       # (B,N,N,H)
    right_gate_raw = proj[..., 3 * hidden_dim:4 * hidden_dim]      # (B,N,N,H)
    out_gate_raw = proj[..., 4 * hidden_dim:]                      # (B,N,N,H)

    # -------------------------------------------------------------
    # 4️⃣ Optional mask on left/right (mask broadcasted over H)
    # -------------------------------------------------------------
    if (not nomask) and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left = left * m
        right = right * m

    # -------------------------------------------------------------
    # 5️⃣ Sigmoid gates (still FP16)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)   # (B,N,N,H) FP16

    # -------------------------------------------------------------
    # 6️⃣ Apply element‑wise gates
    # -------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 7️⃣ Core N³ batched matrix multiplication (outgoing)
    #    left @ rightᵀ   →   (B·H, N, N)  FP32
    # -------------------------------------------------------------
    # reshape to (B*H, N, N) without copying data
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B·H, N, N) FP16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N) # (B·H, N, N) FP16

    # select a block_k that works for all supported N
    block_k = 64 if N >= 64 else 32

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=block_k,
    )   # (B·H, N, N) FP32

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) FP32

    # -------------------------------------------------------------
    # 8️⃣ Output LayerNorm over H (float32)
    # -------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # (B,N,N,H) float32

    # -------------------------------------------------------------
    # 9️⃣ Multiply by out‑gate (casted to float32) and final projection
    # -------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # broadcast multiplication

    # final linear back to D – run in FP16 for speed, cast to FP32 afterwards
    to_out_w_fp16 = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out_fp16 = out.to(torch.float16)
    out = F.linear(out_fp16, to_out_w_fp16)   # (B,N,N,D) FP16
    out = out.to(torch.float32)               # return in FP32

    return out
