"""
TriMul “outgoing” update – fused linear + mask‑gate Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ [B, N, N, D] (float32).
2. Fuse the five bias‑free linear projections (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single GEMM:
        Y = X @ Wᵀ   with   W ∈ ℝ^{5·H × D}
   and then slice Y into the five parts.  This removes four extra
   launches and four memory passes.
3. Apply the three sigmoid gates.  If a mask is present we fuse
   ``left = left * left_gate * mask`` and
   ``right = right * right_gate * mask`` with a tiny Triton kernel.
4. Core N³ update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is a batched GEMM (L @ Rᵀ) for each hidden channel h.
   We reshape to (B·H, N, N) and call a single torch.bmm – cuBLAS
   Tensor‑Core‐accelerated and therefore very fast.
5. Layer‑norm over the hidden dimension, multiply by the output‑gate,
   and a final linear projection back to D.  All heavy work stays in
   fp16; the final result is cast to fp32.

The only mandatory Triton component is the element‑wise mask‑gate fuse
kernel; the rest of the pipeline uses highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask and gate multiplication (in‑place)
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,          # left tensors
    right_ptr, right_gate_ptr,        # right tensors
    mask_ptr,                         # binary mask (0/1) broadcasted to H
    total_elems: tl.int32,            # total number of elements = B·N·N·H
    BLOCK_SIZE: tl.constexpr,         # threads per program
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values (masked loads return 0/1 for out‑of‑bounds elements)
    l_val   = tl.load(left_ptr      + offs, mask=mask, other=0.0)
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_val   = tl.load(right_ptr     + offs, mask=mask, other=0.0)
    r_gate  = tl.load(right_gate_ptr+ offs, mask=mask, other=0.0)
    m_val   = tl.load(mask_ptr      + offs, mask=mask, other=1.0)   # default 1 → no mask

    # Fuse the three element‑wise multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # If nomask is True we may receive ``mask`` as None.
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for all heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # concatenate weights: (5*H, D)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, D]

    # single linear projection
    # shape: [B, N, N, 5·H] (fp16)
    combined = F.linear(x_f16, combined_weight)

    # split the concatenated dimension
    (
        left,
        right,
        left_gate_pre,
        right_gate_pre,
        out_gate_pre,
    ) = torch.split(combined, hidden_dim, dim=-1)

    # gates (sigmoid, fp16)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)   # shape [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 5) Optional mask + gate fusion (via Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask → fp16 and broadcast to hidden dim
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]

        # ensure contiguous storage for the 1‑D view used by Triton
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_f16 = mask_f16.contiguous()

        total_elements = left.numel()                     # B·N·N·H
        BLOCK = 1024                                      # threads per block
        grid = ( (total_elements + BLOCK - 1) // BLOCK, )  # 1‑D launch grid

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total_elements,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # no mask – just apply the gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core N³ batched mat‑mul (L @ Rᵀ) – one torch.bmm call
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # [B, N, N, H]

    # (B, N, N, H) → (B·H, N, N)   (contiguous copy required for the transpose)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (fp16×fp16 → fp16 via Tensor Cores)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] (fp16)

    # Reshape back to [B, N, N, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, output‑gate, final projection (still fp16)
    # ------------------------------------------------------------------
    # Cast the LayerNorm parameters to fp16 to stay in the same dtype
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # out_gate is fp16 – broadcast automatically up‑casts to fp16
    out = out * out_gate

    # final linear projection back to D
    to_out_weight = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out = F.linear(out, to_out_weight)                         # fp16

    # ------------------------------------------------------------------
    # 8) Cast back to float32 (required output dtype)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
