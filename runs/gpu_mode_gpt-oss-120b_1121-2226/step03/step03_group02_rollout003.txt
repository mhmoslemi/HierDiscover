"""
custom_kernel – highly‑optimized Triton forward for the “outgoing” TriMul
operation (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑norm the [B,N,N,D] tensor (FP32 → FP16 afterwards).
2️⃣ Fuse the five linear layers (left‑proj, right‑proj, left‑gate,
   right‑gate, out‑gate) into a *single* GEMM:
      Y = X @ Wᵀ                     # X : (B·N·N, D)  FP16
      W = concat([left_proj, right_proj,
                  left_gate, right_gate, out_gate])   # (5·H, D)
   Split Y into the five tensors and apply sigmoid to the three gates.
3️⃣ If a mask is present, a tiny Triton kernel fuses
   “mask × left_gate” and “mask × right_gate” with the in‑place
   multiplication of the projected tensors – a single pass over the data.
4️⃣ Core N³ update:
      out₍b,i,j,h₎ = Σₖ left₍b,i,k,h₎·right₍b,j,k,h₎
   This is a batched matrix multiply over the hidden dimension:
      (B·H,N,N) @ (B·H,N,N)ᵀ   (half‑precision Tensor Cores).
5️⃣ Layer‑norm over the hidden dimension, multiply by the out‑gate,
   and a final FP16 linear projection back to D.
6️⃣ Cast the result back to FP32 (the format required by the caller).

All heavy linear work stays in half‑precision (Tensor‑Core friendly);
only the very first LayerNorm runs in FP32 for numerical stability.
The Triton kernel is required by the problem statement and fuses the
mask + gate multiplication to avoid extra launches.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: in‑place mask & gate application.
# ------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,               # fp16*   [B,N,N,H] (flattened)
    right_ptr,              # fp16*   same layout as left
    mask_ptr,               # fp16*   [B,N,N]   (flattened)
    left_gate_ptr,          # fp16*   [B,N,N,H] (flattened)
    right_gate_ptr,         # fp16*   [B,N,N,H] (flattened)
    total_elements: tl.constexpr,  # B*N*N*H
    hidden_dim:    tl.constexpr,  # H
    BLOCK_SIZE:    tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load the four tensors (masked to avoid OOB)
    left      = tl.load(left_ptr      + offs, mask=mask)
    right     = tl.load(right_ptr     + offs, mask=mask)
    l_gate    = tl.load(left_gate_ptr + offs, mask=mask)
    r_gate    = tl.load(right_gate_ptr+ offs, mask=mask)

    # mask is stored without the hidden dimension – broadcast it
    mask_idx = offs // hidden_dim          # index into [B,N,N] view
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates (in‑place)
    left  = left  * mask_val * l_gate
    right = right * mask_val * r_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)

    # -----------------------------------------------------------------
    # Layer‑norm over the hidden dimension (still FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Apply the out‑gate (already sigmoid‑ed, fp16)
    out_norm = out_norm * out_gate

    # Final linear projection back to D (no bias)
    to_out_w = weights["to_out.weight"].to(torch.float16)  # (D, H)
    out_final = F.linear(out_norm, to_out_w)               # fp16, [B,N,N,D]

    # Return in the required float32 format
    return out_final.to(torch.float32)
