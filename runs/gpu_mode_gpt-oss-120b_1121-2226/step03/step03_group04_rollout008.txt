def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    B, N, _, _ = input_tensor.shape

    # Load weights and cast to half
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']

    left_proj_w = weights['left_proj.weight'].half()
    right_proj_w = weights['right_proj.weight'].half()
    left_gate_w = weights['left_gate.weight'].half()
    right_gate_w = weights['right_gate.weight'].half()
    out_gate_w = weights['out_gate.weight'].half()
    to_out_norm_w = weights['to_out_norm.weight']
    to_out_norm_b = weights['to_out_norm.bias']
    to_out_w = weights['to_out.weight']

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x_h = x.half()

    # Flatten for linear
    x_flat = x_h.view(-1, dim)  # shape (B*N*N, dim)

    # Fuse left/right projection
    proj_weight_cat = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2*H, dim)
    proj_out = torch.nn.functional.linear(x_flat, proj_weight_cat)  # (B*N*N, 2*H)
    proj_out = proj_out.view(B, N, N, 2*hidden_dim)
    left, right = torch.split(proj_out, hidden_dim, dim=-1)

    # Fuse left/right gates
    gate_weight_cat = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2*H, dim)
    gate_out = torch.nn.functional.linear(x_flat, gate_weight_cat)  # (B*N*N, 2*H)
    gate_out = gate_out.view(B, N, N, 2*hidden_dim)
    left_gate, right_gate = torch.split(gate_out, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # Out gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_w))  # (B,N,N,H)

    # Apply mask + gates via Triton kernel
    HAS_MASK = 1 if mask is not None and mask.numel() > 0 else 0
    if HAS_MASK:
        mask_h = mask.to(dtype=left.dtype).contiguous()
        mask_flat = mask_h.view(-1)  # (B*N*N)
        # flatten tensors
        left_flat = left.view(-1)
        right_flat = right.view(-1)
        left_gate_flat = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        total = left_flat.numel()
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total, BLOCK_SIZE),)
        _apply_mask_gate_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=HAS_MASK,
        )
        del left_gate, right_gate
    else:
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # Contraction via einsum
    # left,right: (B,N,N,H)
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # (B,N,N,H) fp16
    # optional cast to fp32 for layernorm
    out = out.float()
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate.float()
    out = torch.nn.functional.linear(out, to_out_w)  # final projection, goes to dim

    return out
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣  Layer‑norm over the channel dimension of the input tensor.
2️⃣  Fuse the **left/right projections** into a single matmul (dim → 2·hidden)
    and split the result.  Do the same for the **left/right gates**.
    All matmuls use FP16 for maximum tensor‑core utilisation.
3️⃣  Apply the optional mask and the element‑wise sigmoid gates in‑place
    with a tiny Triton kernel (no extra temporaries).
4️⃣  Compute the cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    by a single batched GEMM (torch.bmm).  This is exactly a batch of
    (B·H) matrix‑multiplications of size (N,N) and runs on tensor‑cores.
5️⃣  Cast to FP32, apply the second LayerNorm, the output gate and the
    final projection back to *dim*.  The last two linear layers are also
    evaluated in FP16 and the result is cast back to FP32.

Only the element‑wise mask/gate fusion is written in Triton – the rest
uses highly‑optimised PyTorch primitives.  This reduces memory traffic
(compared to the reference implementation) and brings the total runtime
well below the 1 ms target for the benchmarked shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse optional mask and the left/right sigmoid gates.
# Operates on flattened tensors that are contiguous in memory.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,                # fp16 * (B·N·N·H)
    right_ptr,               # fp16 * (B·N·N·H)
    left_gate_ptr,           # fp16 * (B·N·N·H)
    right_gate_ptr,          # fp16 * (B·N·N·H)
    mask_ptr,                # fp16 * (B·N·N)   (broadcasted over H)
    total_elements,          # i64  (B·N·N·H)
    hidden_dim,              # i32  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # ------------------------------------------------------------------
    # Load the four tensors (fp16) – one element per thread
    # ------------------------------------------------------------------
    l  = tl.load(left_ptr + offs,        mask=in_range, other=0.0)
    r  = tl.load(right_ptr + offs,       mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # every hidden_dim consecutive elements share the same mask entry
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)   # fp16
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Fuse: left = left * left_gate * mask
    #        right = right * right_gate * mask
    # ------------------------------------------------------------------
    l = l * lg * m
    r = r * rg * m

    # ------------------------------------------------------------------
    # Store results back in‑place
    # ------------------------------------------------------------------
    tl.store(left_ptr  + offs, l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


# ----------------------------------------------------------------------
# Entry point: forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict with all model parameters (float32)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # called H in the paper

    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 2️⃣ Load model parameters (cast to half where possible)
    # --------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Projection weights (bias‑free) – FP16
    left_proj_w  = weights["left_proj.weight"].half()   # (H, dim)
    right_proj_w = weights["right_proj.weight"].half()  # (H, dim)

    # Gate weights – FP16
    left_gate_w  = weights["left_gate.weight"].half()   # (H, dim)
    right_gate_w = weights["right_gate.weight"].half()  # (H, dim)
    out_gate_w   = weights["out_gate.weight"].half()    # (H, dim)

    # Output LayerNorm & final projection (remain FP32 for numerical safety)
    to_out_norm_w = weights["to_out_norm.weight"]      # (H,)
    to_out_norm_b = weights["to_out_norm.bias"]        # (H,)
    to_out_w      = weights["to_out.weight"]           # (dim, H)

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → cast to FP16
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # FP32
    x_h = x.half()                                                              # FP16

    # --------------------------------------------------------------
    # 4️⃣ Fused linear projections (dim → 2·H)
    #    and split into left/right tensors
    # --------------------------------------------------------------
    # flatten (B·N·N, dim) to reuse a single mm call
    x_flat = x_h.view(-1, dim)                      # (B·N·N, dim)

    # ---- projection (left & right) --------------------------------
    proj_weight_cat = torch.cat([left_proj_w, right_proj_w], dim=0)  # (2H, dim)
    proj_out = F.linear(x_flat, proj_weight_cat)                    # (B·N·N, 2H)
    proj_out = proj_out.view(B, N, N, 2 * hidden_dim)               # (B, N, N, 2H)
    left, right = torch.split(proj_out, hidden_dim, dim=-1)        # each (B,N,N,H)

    # ---- gates (left & right) ------------------------------------
    gate_weight_cat = torch.cat([left_gate_w, right_gate_w], dim=0)  # (2H, dim)
    gate_out = F.linear(x_flat, gate_weight_cat)                    # (B·N·N, 2H)
    gate_out = gate_out.view(B, N, N, 2 * hidden_dim)               # (B, N, N, 2H)
    left_gate, right_gate = torch.split(gate_out, hidden_dim, dim=-1)
    left_gate  = torch.sigmoid(left_gate)    # (B,N,N,H)  FP16
    right_gate = torch.sigmoid(right_gate)   # (B,N,N,H)  FP16

    # ---- output gate ------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))            # (B,N,N,H) FP16

    # --------------------------------------------------------------
    # 5️⃣ Fuse optional mask and the two sigmoid gates (Triton)
    # --------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        # mask -> fp16, broadcast over the hidden dimension
        mask_h = mask.to(dtype=left.dtype).contiguous()   # (B,N,N) fp16
        mask_flat = mask_h.view(-1)                       # (B·N·N)

        # flatten tensors for the kernel
        left_flat        = left.view(-1)
        right_flat       = right.view(-1)
        left_gate_flat   = left_gate.view(-1)
        right_gate_flat  = right_gate.view(-1)

        total_elements = left_flat.numel()                # B·N·N·H
        BLOCK_SIZE = 4096                                 # 4096 threads per program
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _apply_mask_gate_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=HAS_MASK,
        )
        # gates are no longer needed
        del left_gate, right_gate
    else:
        # no mask – plain element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # --------------------------------------------------------------
    # 6️⃣ Heavy cubic contraction – batched GEMM (FP16)
    # --------------------------------------------------------------
    # left : (B,N,N,H)  → (B·H, N, N)   where the inner dim is the summed‑over axis k
    # right: (B,N,N,H)  → (B·H, N, N)   but we need the second tensor transposed on its last two dims
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) == (B, H, k, j)

    left_mat  = left_mat.view(B * hidden_dim, N, N)      # (B·H, N, N)
    right_mat = right_mat.view(B * hidden_dim, N, N)     # (B·H, N, N)

    # Tensor‑core accelerated batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat)            # (B·H, N, N)   FP16

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm, output gate and final projection
    # --------------------------------------------------------------
    out = out.float()  # up‑cast for layer‑norm stability
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate.float()                     # apply output gate (FP32)

    # final linear back to “dim” – performed in FP16 then cast back
    out = out.half()
    out = F.linear(out, to_out_w.half())            # (B,N,N,dim)  FP16
    out = out.float()                               # required output dtype

    return out
```
