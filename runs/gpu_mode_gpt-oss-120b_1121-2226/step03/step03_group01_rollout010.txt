def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    x_fp16 = x.half()

    # Concatenate projection weights
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_lgate = weights["left_gate.weight"]
    w_rgate = weights["right_gate.weight"]
    w_ogate = weights["out_gate.weight"]
    # each shape (hidden_dim, dim)
    combined_weight = torch.cat([w_left, w_right, w_lgate, w_rgate, w_ogate], dim=0).half()

    # Single linear for projections+gates
    proj = F.linear(x_fp16, combined_weight)  # shape [B,N,N,5*hidden_dim]
    # split
    split = torch.split(proj, hidden_dim, dim=-1)
    left, right, left_gate, right_gate, out_gate = split

    # apply optional mask before gating
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Sigmoid on gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Reshape for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)
    right_flat = right_perm.view(BH, seq_len, seq_len)

    # Output buffer
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float32, device=device)

    # Triton kernel launch

    BLOCK_N = 128
    BLOCK_K = 128

"""
TriMul “outgoing” forward pass (float32 output)

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → x.
2. Cast x to fp16 for all heavy linear work.
3. Fuse all five bias‑free linear projections (left, right, three gates)
   into a single matmul by concatenating the weight matrices along the
   output dimension.  This reads x only once from memory.
4. Split the concatenated result into left/right projections and the three
   sigmoid gates, apply the optional pair‑wise mask, and apply the
   element‑wise gates (still in fp16).
5. Reshape left/right to (B·H, N, N) (H = hidden_dim) and launch a custom
   Triton kernel that computes the batched GEMM
        out = left @ rightᵀ
   for every hidden channel, using fp16 tiles, fp32 accumulation and
   Tensor‑Core dot (BLOCK_M=N_BLOCK=128, BLOCK_K=128).  The kernel writes
   the result in float32.
6. Reshape the GEMM output back to (B, N, N, H) and apply a second
   Layer‑Norm over the hidden dimension (float32).
7. Multiply by the output gate, then a final bias‑free linear projection
   back to the original dimension D (float32).
8. Return the float32 tensor of shape [B, N, N, D].

All major memory‑bound stages (projections, gating, mask) are fused,
and the O(N³·H) GEMM is executed by an optimized Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B fp16, C fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N]   (fp16)
    b_ptr,               # right : [BH, N, N]   (fp16) – interpreted as (K,N)
    c_ptr,               # out   : [BH, N, N]   (fp32)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # batch * hidden
    pid_m = tl.program_id(axis=0)          # tile row
    pid_n = tl.program_id(axis=1)          # tile col

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile (M × K)
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Bᵀ tile (K × N)
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # Accumulate with Tensor‑Core dot, result in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Write back
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])

    # (B, N, N, H) -> (B, H, N, N) -> (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    # Output buffer for the GEMM (FP32 accumulation)
    out_flat = torch.empty(
        (BH, seq_len, seq_len), dtype=torch.float32, device=device
    )

    # --------------------------------------------------------------
    # 8️⃣ Triton batched GEMM: out_flat = left_flat @ right_flatᵀ
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # rows
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # cols
        BH,                                   # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                # M
        seq_len,                # N
        seq_len,                # K
        # strides for A (left)
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # strides for Bᵀ (right, interpreted as K × N)
        right_flat.stride(0),
        right_flat.stride(2),
        right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, H]
    # --------------------------------------------------------------
    out_perm = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)

    # --------------------------------------------------------------
    # 🔟 Second LayerNorm over the hidden dimension (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # stays float32

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Apply output gate and final projection back to D
    # --------------------------------------------------------------
    # out_gate is fp16; promote it to float32 for the multiplication
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])   # bias‑free, returns float32

    return out
