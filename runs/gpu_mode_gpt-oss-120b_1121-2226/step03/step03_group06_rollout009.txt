"""
custom_kernel – Outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32).
2.  Cast to fp16 and perform **one** fused linear projection that yields
    - left projection      (L)
    - right projection     (R)
    - left gate            (gL)
    - right gate           (gR)
    - output gate          (gO)
   All five heads are obtained by a single matrix‑multiply with a
   concatenated weight matrix (shape (5·hidden_dim, dim)).  The gates are
   passed through a sigmoid; the optional mask is applied to L and R
   before gating.
3.  Element‑wise gating:
       L ← L * gL
       R ← R * gR
4.  Core cubic update:
       out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
   This is a batched matrix‑multiply L @ Rᵀ for each hidden channel.
   For modest sequence lengths (N ≤ 384) a hand‑written Triton tiled
   matmul is used; for larger N we fall back to torch.bmm which invokes
   cuBLAS Tensor‑Core kernels (fp16·fp16 → fp32).
5.  Second LayerNorm over the hidden dimension (float32).
6.  Multiply by the output gate gO and finish with a linear projection
   back to the original feature dimension (float32).

All heavy arithmetic is performed in fp16 with fp32 accumulation,
producing a float32 output tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)  (computes A @ Bᵀ)
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # row tile
    pid_n = tl.program_id(1)          # col tile
    pid_b = tl.program_id(2)          # batch (= hidden‑channel) tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the tile in A and B
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator (fp32) – Tensor‑Core friendly
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Tensor‑core dot product (fp16·fp16 → fp32)
        acc += tl.dot(a, b)

        # Advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """Batched fp16·fp16 → fp32 matmul (A @ Bᵀ) used for the TriMul cubic update."""
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ------------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass for the **outgoing** TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] – fp16

    B, N, _, _ = x_h.shape

    # ------------------------------------------------------------------
    # 3) Fused linear projection + gating (single GEMM)
    #    Concatenate the five weight matrices:
    #    [left_proj; right_proj; left_gate; right_gate; out_gate] → (5·H, dim)
    # ------------------------------------------------------------------
    w_left_proj = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]

    # Concatenate along the out‑feature dimension and cast to half
    fused_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate], dim=0
    ).to(device).half()                         # shape (5*H, dim)

    # Collapse the first three dimensions → (B·N·N, dim)
    x_flat = x_h.view(-1, dim)  # [B·N·N, dim]

    # Single linear pass (fp16·fp16 → fp16)
    fused_out = F.linear(x_flat, fused_weight)   # [B·N·N, 5*H]

    # Split back into the five heads
    splits = torch.split(fused_out, hidden_dim, dim=1)
    left_proj = splits[0].view(B, N, N, hidden_dim)    # fp16
    right_proj = splits[1].view(B, N, N, hidden_dim)   # fp16
    left_gate = splits[2].view(B, N, N, hidden_dim).sigmoid()   # fp16
    right_gate = splits[3].view(B, N, N, hidden_dim).sigmoid()  # fp16
    out_gate = splits[4].view(B, N, N, hidden_dim).sigmoid()    # fp16

    # Release intermediate tensor early
    del fused_out

    # ------------------------------------------------------------------
    # 4) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(left_proj.dtype).unsqueeze(-1)  # [B, N, N, 1] in fp16
        left_proj = left_proj * m
        right_proj = right_proj * m

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates (still fp16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B, N, N, H] fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6) Core cubic update – batched matmul L @ Rᵀ
    #    For small seq‑len we use the Triton kernel; otherwise torch.bmm.
    # ------------------------------------------------------------------
    batch_h = B * hidden_dim
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(batch_h, N, N)   # [B·H, N, N] fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch_h, N, N)  # [B·H, N, N] fp16

    if N <= 384:
        # Tile sizes tuned for the “small‑N” regime
        BLOCK_M, BLOCK_N = 64, 64
        BLOCK_K = 64
        out_mat = _batched_matmul_triton(
            left_mat,
            right_mat,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )  # [B·H, N, N] fp32
    else:
        # cuBLAS‑accelerated batched mm (fp16·fp16 → fp32)
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).float()  # [B·H, N, N] fp32

    # Reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 8) Output gating and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.float()                # element‑wise gate (fp32)
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] fp32

    return out
