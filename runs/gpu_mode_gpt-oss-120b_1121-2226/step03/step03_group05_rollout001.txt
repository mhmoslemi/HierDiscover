"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. **Input LayerNorm** (float32 – more stable) on the 4‑D tensor
   `[B, N, N, D]`.

2. Cast to FP16 for the heavy work.

3. **Linear projections + gates** (all bias‑free):
   * `left  = x @ left_proj_weight`
   * `right = x @ right_proj_weight`
   * `left_gate  = sigmoid(x @ left_gate_weight)`
   * `right_gate = sigmoid(x @ right_gate_weight)`
   * `out_gate   = sigmoid(x @ out_gate_weight)`

4. **Mask & gate fusion** (when a mask is present) with a tiny
   Triton kernel – one pass that multiplies
   `left  *= left_gate  * mask` and
   `right *= right_gate * mask`.

   The mask is stored only once (`[B, N, N]`) and is looked‑up inside
   the kernel via integer division, avoiding a large broadcast.

5. **Core N³ update** – batched matrix multiplication
   `L @ Rᵀ` for every hidden channel.
   By reshaping to `(B·H, N, N)` we can reuse cuBLAS’ tensor‑core GEMM
   (`torch.bmm`) which runs entirely in FP16.

6. **Output LayerNorm + output‑gate** – still in FP16.

7. **Final linear projection** back to the model dimension `D`
   (`to_out.weight`).  The result is cast to float32 for the caller.

All matrix‑multiplications stay in FP16 (tensor‑core friendly),
only the two layer‑norms and the final cast touch FP32.  The only
explicit Triton kernel fuses the three element‑wise products required
by the mask‑gate step, dramatically reducing memory traffic.

The implementation works for arbitrary batch size (1 or 2), sequence
length up to 1024 and hidden dimensions 128/384, with or without a
mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gate application.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# The mask is stored only once with shape [B, N, N] and is broadcasted
# across the hidden dimension inside the kernel.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,            # fp16 * left
    left_gate_ptr,       # fp16 * left_gate
    right_ptr,           # fp16 * right
    right_gate_ptr,      # fp16 * right_gate
    mask_ptr,            # fp16 * mask (shape B*N*N)
    total_elems: tl.int32,        # B*N*N*H
    HEAD_DIM: tl.constexpr,       # hidden dimension H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Index of the mask element (broadcast over H)
    mask_idx = offs // HEAD_DIM

    # Load everything (other=0 for values, other=1 for mask → no‑mask case)
    l_val   = tl.load(left_ptr       + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr  + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m_val   = tl.load(mask_ptr       + mask_idx, mask=mask, other=tl.float16(1.0))

    # Fuse three element‑wise multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)

    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → fp16, no extra broadcast dimension
        mask_f16 = mask.to(torch.float16).contiguous()   # shape [B, N, N]

        # Ensure contiguous views for pointer arithmetic
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()

        total = left.numel()                     # B·N·N·H  (int32 fits)
        BLOCK = 1024                               # max threads per block on H100
        grid = ((total + BLOCK - 1) // BLOCK, )    # 1‑D launch grid

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            HEAD_DIM=hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,            # memory‑bound kernel – enough warps for latency hiding
        )
    else:
        # No mask – just apply the gates in‑place
        left  = left.mul_(left_gate)
        right = right.mul_(right_gate)

    # ------------------------------------------------------------------
    # 7) Core N³ update: batched GEMM  L @ Rᵀ  (all fp16 → tensor cores)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N).contiguous()

    # batched matrix multiplication, tensor‑core accelerated
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] (fp16)

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 8) Output LayerNorm + output‑gate (still fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate            # broadcasted element‑wise product

    # ------------------------------------------------------------------
    # 9) Final linear projection back to D (fp16 → fp32)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # result fp16
    out = out.to(torch.float32)     # required output dtype

    return out
```
