    right_out_ptrs = right_out_ptr + offs_pos[:, None] * stride_out_pos + tl.arange(0, hidden_dim)[None, :] * stride_out_dim

    right_gate_out_ptrs = right_gate_out_ptr + offs_pos[:, None] * stride_out_pos + tl.arange(0, hidden_dim)[None, :] * stride_out_dim
    out_gate_out_ptrs = out_gate_out_ptr + offs_pos[:, None] * stride_out_pos + tl.arange(0, hidden_dim)[None, :] * stride_out_dim

    mask_c = mask_pos[:, None] & (tl.arange(0, hidden_dim)[None, :] < hidden_dim)
    # cast to fp16
    tl.store(left_out_ptrs, left_acc.to(tl.float16), mask=mask_c)
    tl.store(right_out_ptrs, right_acc.to(tl.float16), mask=mask_c)
    tl.store(left_gate_out_ptrs, left_gate_acc.to(tl.float16), mask=mask_c)
    tl.store(right_gate_out_ptrs, right_gate_acc.to(tl.float16), mask=mask_c)

    norm_b = weights["norm.bias"].to(torch.float16)

    right_proj_w = weights["right_proj.weight"].t().contiguous().to(torch.float16)

    right_gate_w = weights["right_gate.weight"].t().contiguous().to(torch.float16)
    out_gate_w = weights["out_gate.weight"].t().contiguous().to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Flatten input and convert to fp16
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    B, N, _, D = input_tensor.shape  # D = dim
    total_pos = B * N * N
    x_flat = x_fp16.view(total_pos, dim).contiguous()
    # Allocate output buffers
    left = torch.empty((total_pos, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty((total_pos, hidden_dim), dtype=torch.float16, device=device)
    left_gate = torch.empty((total_pos, hidden_dim), dtype=torch.float16, device=device)
    right_gate = torch.empty((total_pos, hidden_dim), dtype=torch.float16, device=device)
    out_gate = torch.empty((total_pos, hidden_dim), dtype=torch.float16, device=device)

    # Launch fused kernel
    BLOCK_POS = 1   # each thread processes one position
    BLOCK_D = 32    # tile size for dim
    grid = (total_pos + BLOCK_POS - 1) // BLOCK_POS
    fused_ln_proj_gate_kernel[grid](
        x_flat, norm_w, norm_b,
        left_proj_w, right_proj_w,
        left_gate_w, right_gate_w,
        out_gate_w,
        left, right,
        left_gate, right_gate, out_gate,
        total_pos, dim, hidden_dim,
        # strides
        x_flat.stride(0), x_flat.stride(1),
        norm_w.stride(0), norm_b.stride(0),
        left_proj_w.stride(0), left_proj_w.stride(1),
        right_proj_w.stride(0), right_proj_w.stride(1),
        left_gate_w.stride(0), left_gate_w.stride(1),

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: fused LayerNorm + projections + gate linear layers
# ------------------------------------------------------------------
@triton.jit
def fused_ln_proj_gate_kernel(
    X_ptr,                     # input flattened: [P, D] fp16
    norm_w_ptr, norm_b_ptr,   # LayerNorm weight & bias: [D] fp16
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr,              # projection weight matrices (transposed) [D, H] fp16 each
    left_out_ptr, right_out_ptr,
    left_gate_out_ptr, right_gate_out_ptr, out_gate_out_ptr,
    total_pos, dim, hidden_dim,
    stride_x_batch, stride_x_dim,
    stride_norm_w, stride_norm_b,
    stride_left_proj_dim, stride_left_proj_out,
    stride_right_proj_dim, stride_right_proj_out,
    stride_left_gate_dim, stride_left_gate_out,
    stride_right_gate_dim, stride_right_gate_out,
    stride_out_gate_dim, stride_out_gate_out,
    stride_left_out_batch, stride_left_out_dim,
    stride_right_out_batch, stride_right_out_dim,
    stride_left_gate_out_batch, stride_left_gate_out_dim,
    stride_right_gate_out_batch, stride_right_gate_out_dim,
    stride_out_gate_out_batch, stride_out_gate_out_dim,
    BLOCK_POS: tl.constexpr,   # set to 1
    BLOCK_D: tl.constexpr,     # e.g., 32
    EPS: tl.constexpr = 1e-5,
):
    pid = tl.program_id(0)     # each program handles one position
    pos = pid * BLOCK_POS + tl.arange(0, BLOCK_POS)
    mask_pos = pos < total_pos

    # ------------------------------------------------------------------
    # 1) Compute mean and variance of the input vector (Dim D)
    # ------------------------------------------------------------------
    sum_fp32 = tl.zeros([BLOCK_POS], dtype=tl.float32)
    sumsq_fp32 = tl.zeros([BLOCK_POS], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d * BLOCK_D
        range_d = tl.arange(0, BLOCK_D)
        # mask for dim overflow
        mask_d = (d_start + range_d) < dim

        # load input slice (fp16) -> promote to fp32
        x_ptrs = X_ptr + pos[:, None] * stride_x_batch + (d_start + range_d)[None, :] * stride_x_dim
        x_tile = tl.load(x_ptrs, mask=mask_pos[:, None] & mask_d[None, :], other=0.0).to(tl.float32)  # (BLOCK_POS, BLOCK_D)

        sum_fp32 += tl.sum(x_tile, axis=1)
        sumsq_fp32 += tl.sum(x_tile * x_tile, axis=1)

    mean = sum_fp32 * (1.0 / dim)
    var = sumsq_fp32 * (1.0 / dim) - mean * mean
    var = tl.maximum(var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + EPS)   # (BLOCK_POS,)

    # ------------------------------------------------------------------
    # 2) Accumulators for linear outputs (fp32)
    # ------------------------------------------------------------------
    left_acc = tl.zeros([hidden_dim], dtype=tl.float32)
    right_acc = tl.zeros([hidden_dim], dtype=tl.float32)
    left_gate_acc = tl.zeros([hidden_dim], dtype=tl.float32)
    right_gate_acc = tl.zeros([hidden_dim], dtype=tl.float32)
    out_gate_acc = tl.zeros([hidden_dim], dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over D to apply LayerNorm scaling and perform projections
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d * BLOCK_D
        range_d = tl.arange(0, BLOCK_D)
        mask_d = (d_start + range_d) < dim

        # Load input slice
        x_ptrs = X_ptr + pos[:, None] * stride_x_batch + (d_start + range_d)[None, :] * stride_x_dim
        x_tile = tl.load(x_ptrs, mask=mask_pos[:, None] & mask_d[None, :], other=0.0).to(tl.float32)   # (BLOCK_POS, BLOCK_D)

        # Normalize: (x - mean) * inv_std
        # Broadcast mean & inv_std from (BLOCK_POS,) to (BLOCK_POS, BLOCK_D)
        x_norm = (x_tile - mean[:, None]) * inv_std[:, None]

        # Apply LayerNorm weight and bias
        w_norm = tl.load(norm_w_ptr + (d_start + range_d) * stride_norm_w, mask=mask_d, other=0.0).to(tl.float32)   # (BLOCK_D)
        b_norm = tl.load(norm_b_ptr + (d_start + range_d) * stride_norm_b, mask=mask_d, other=0.0).to(tl.float32)
        x_norm = x_norm * w_norm[None, :] + b_norm[None, :]

        # --------------------
        # left projection
        # --------------------
        left_w = tl.load(
            left_proj_ptr
            + (d_start + range_d)[:, None] * stride_left_proj_dim
            + tl.arange(0, hidden_dim)[None, :] * stride_left_proj_out,
            mask=mask_d[:, None],
        ).to(tl.float32)   # (BLOCK_D, H)
        left_acc += tl.dot(x_norm, left_w)[0]

        # --------------------
        # right projection
        # --------------------
        right_w = tl.load(
            right_proj_ptr
            + (d_start + range_d)[:, None] * stride_right_proj_dim
            + tl.arange(0, hidden_dim)[None, :] * stride_right_proj_out,
            mask=mask_d[:, None],
        ).to(tl.float32)
        right_acc += tl.dot(x_norm, right_w)[0]

        # --------------------
        # left gate
        # --------------------
        left_gate_w = tl.load(
            left_gate_ptr
            + (d_start + range_d)[:, None] * stride_left_gate_dim
            + tl.arange(0, hidden_dim)[None, :] * stride_left_gate_out,
            mask=mask_d[:, None],
        ).to(tl.float32)
        left_gate_acc += tl.dot(x_norm, left_gate_w)[0]

        # --------------------
        # right gate
        # --------------------
        right_gate_w = tl.load(
            right_gate_ptr
            + (d_start + range_d)[:, None] * stride_right_gate_dim
            + tl.arange(0, hidden_dim)[None, :] * stride_right_gate_out,
            mask=mask_d[:, None],
        ).to(tl.float32)
        right_gate_acc += tl.dot(x_norm, right_gate_w)[0]

        # --------------------
        # out gate
        # --------------------
        out_gate_w = tl.load(
            out_gate_ptr
            + (d_start + range_d)[:, None] * stride_out_gate_dim
            + tl.arange(0, hidden_dim)[None, :] * stride_out_gate_out,
            mask=mask_d[:, None],
        ).to(tl.float32)
        out_gate_acc += tl.dot(x_norm, out_gate_w)[0]

    # ------------------------------------------------------------------
    # 4) Apply sigmoid to gate tensors
    # ------------------------------------------------------------------
    left_gate_acc = tl.sigmoid(left_gate_acc)
    right_gate_acc = tl.sigmoid(right_gate_acc)
    out_gate_acc = tl.sigmoid(out_gate_acc)

    # ------------------------------------------------------------------
    # 5) Store results (cast back to fp16)
    # ------------------------------------------------------------------
    # left
    left_ptrs = left_out_ptr + pos[:, None] * stride_left_out_batch + tl.arange(0, hidden_dim)[None, :] * stride_left_out_dim
    tl.store(left_ptrs, left_acc.to(tl.float16), mask=mask_pos[:, None])

    # right
    right_ptrs = right_out_ptr + pos[:, None] * stride_right_out_batch + tl.arange(0, hidden_dim)[None, :] * stride_right_out_dim
    tl.store(right_ptrs, right_acc.to(tl.float16), mask=mask_pos[:, None])

    # left_gate
    left_gate_ptrs = left_gate_out_ptr + pos[:, None] * stride_left_gate_out_batch + tl.arange(0, hidden_dim)[None, :] * stride_left_gate_out_dim
    tl.store(left_gate_ptrs, left_gate_acc.to(tl.float16), mask=mask_pos[:, None])

    # right_gate
    right_gate_ptrs = right_gate_out_ptr + pos[:, None] * stride_right_gate_out_batch + tl.arange(0, hidden_dim)[None, :] * stride_right_gate_out_dim
    tl.store(right_gate_ptrs, right_gate_acc.to(tl.float16), mask=mask_pos[:, None])

    # out_gate
    out_gate_ptrs = out_gate_out_ptr + pos[:, None] * stride_out_gate_out_batch + tl.arange(0, hidden_dim)[None, :] * stride_out_gate_out_dim
    tl.store(out_gate_ptrs, out_gate_acc.to(tl.float16), mask=mask_pos[:, None])
```

"""
TriMul “outgoing” (AlphaFold3) – Triton‑accelerated forward pass.

Algorithm
---------
1. Fuse LayerNorm, the five linear projections (left, right,
   left_gate, right_gate, out_gate) and the three sigmoid gates into a
   single Triton kernel.  The kernel reads each (B,N,N,D) element only
   once, computes the LN statistics, applies the LN scale/bias and then
   performs two matrix‑vector products per D‑tile:
        * proj_weight  :  D×(2·H)  = [left_proj | right_proj]
        * gate_weight  :  D×(3·H)  = [left_gate | right_gate | out_gate]
   The results are written out as concatenated tensors, which are later
   split into the individual tensors.
2. Apply the binary mask (if present) and multiply the projected
   tensors by their gates.
3. Perform the cubic “tri‑multiply”:
        out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   implemented as a batched matrix‑multiply (B·H, N, N) @ (B·H, N, N)ᵀ
   using a custom Triton kernel (FP16 inputs, FP32 accumulation).
4. Layer‑norm the hidden‑dim output, apply the out‑gate, and project back
   to the original dimension with a bias‑free linear layer.
5. Return a float‑32 tensor of shape [B, N, N, D].

All heavy compute (the LN+projections and the cubic mat‑mul) is executed
on the GPU via Triton, dramatically reducing the number of memory passes
over the large [B,N,N,D] tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# 1. Fused LayerNorm + five linear layers + three sigmoid gates
# ------------------------------------------------------------------
@triton.jit
def fused_ln_proj_gate_kernel(
    X_ptr,                           # [P, D]  input flattened, fp16
    norm_w_ptr, norm_b_ptr,          # [D]    LayerNorm weight & bias, fp16
    proj_weight_ptr,                 # [D, 2*H]  concatenated left & right projection weights, fp16
    gate_weight_ptr,                 # [D, 3*H]  concatenated left_gate, right_gate, out_gate weights, fp16
    proj_out_ptr,                    # [P, 2*H] output of projections, fp16
    gate_out_ptr,                    # [P, 3*H] output of gates (pre‑sigmoid), fp16
    total_pos, dim,                  # runtime scalars
    HIDDEN_DIM: tl.constexpr,        # compile‑time hidden dimension (H)
    stride_x_batch, stride_x_dim,    # strides for X
    stride_norm_w, stride_norm_b,    # strides for LayerNorm params (always 1)
    stride_proj_dim, stride_proj_out,# strides for proj_weight (row‑major)
    stride_gate_dim, stride_gate_out,# strides for gate_weight (row‑major)
    stride_proj_out_batch, stride_proj_out_dim,   # strides for proj_out
    stride_gate_out_batch, stride_gate_out_dim,   # strides for gate_out
    BLOCK_POS: tl.constexpr,         # =1  (one position per program)
    BLOCK_D: tl.constexpr,           # tile size over the D dimension, e.g. 32
    EPS: tl.constexpr = 1e-5,        # epsilon for LayerNorm
):
    pid = tl.program_id(0)                 # one program per position
    pos = pid * BLOCK_POS + tl.arange(0, BLOCK_POS)   # (BLOCK_POS,)
    mask_pos = pos < total_pos

    # ------------------------------------------------------------------
    # 1a) Compute mean and variance of the D‑dim vector (single pass)
    # ------------------------------------------------------------------
    sum_fp32   = tl.zeros([BLOCK_POS], dtype=tl.float32)
    sumsq_fp32 = tl.zeros([BLOCK_POS], dtype=tl.float32)

    num_blocks = tl.cdiv(dim, BLOCK_D)
    for d in range(0, num_blocks):
        d_start = d * BLOCK_D
        range_d = tl.arange(0, BLOCK_D)
        mask_d = (d_start + range_d) < dim                     # (BLOCK_D,)

        # load a slice of the input vector
        x_ptrs = X_ptr + pos[:, None] * stride_x_batch + (d_start + range_d)[None, :] * stride_x_dim
        x_tile = tl.load(x_ptrs,
                         mask=mask_pos[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)          # (BLOCK_POS, BLOCK_D)

        sum_fp32   += tl.sum(x_tile, axis=1)
        sumsq_fp32 += tl.sum(x_tile * x_tile, axis=1)

    mean = sum_fp32 * (1.0 / dim)
    var  = sumsq_fp32 * (1.0 / dim) - mean * mean
    var  = tl.maximum(var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + EPS)          # (BLOCK_POS,)

    # ------------------------------------------------------------------
    # 1b) Accumulators for the two mat‑vec products
    # ------------------------------------------------------------------
    proj_acc = tl.zeros([2 * HIDDEN_DIM], dtype=tl.float32)   # left|right
    gate_acc = tl.zeros([3 * HIDDEN_DIM], dtype=tl.float32)   # left_gate|right_gate|out_gate

    # ------------------------------------------------------------------
    # 2) Loop over D‑tiles, apply LN scaling, then do the two mat‑vecs
    # ------------------------------------------------------------------
    for d in range(0, num_blocks):
        d_start = d * BLOCK_D
        range_d = tl.arange(0, BLOCK_D)
        mask_d = (d_start + range_d) < dim

        # load input slice
        x_ptrs = X_ptr + pos[:, None] * stride_x_batch + (d_start + range_d)[None, :] * stride_x_dim
        x_tile = tl.load(x_ptrs,
                         mask=mask_pos[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)           # (BLOCK_POS, BLOCK_D)

        # LayerNorm: (x - mean) * inv_std
        x_norm = (x_tile - mean[:, None]) * inv_std[:, None]   # (BLOCK_POS, BLOCK_D)

        # LN affine transform
        w_n = tl.load(norm_w_ptr + (d_start + range_d) * stride_norm_w,
                      mask=mask_d, other=0.0).to(tl.float32)    # (BLOCK_D,)
        b_n = tl.load(norm_b_ptr + (d_start + range_d) * stride_norm_b,
                      mask=mask_d, other=0.0).to(tl.float32)    # (BLOCK_D,)
        x_norm = x_norm * w_n[None, :] + b_n[None, :]          # (BLOCK_POS, BLOCK_D)

        # ---- projection (left & right) ---------------------------------
        proj_w = tl.load(
            proj_weight_ptr
            + (d_start + range_d)[:, None] * stride_proj_dim
            + tl.arange(0, 2 * HIDDEN_DIM)[None, :] * stride_proj_out,
            mask=mask_d[:, None]
        ).to(tl.float32)                                       # (BLOCK_D, 2*H)

        proj_acc += tl.dot(x_norm, proj_w)[0]                  # (2*H,)

        # ---- gates (left_gate, right_gate, out_gate) ------------------
        gate_w = tl.load(
            gate_weight_ptr
            + (d_start + range_d)[:, None] * stride_gate_dim
            + tl.arange(0, 3 * HIDDEN_DIM)[None, :] * stride_gate_out,
            mask=mask_d[:, None]
        ).to(tl.float32)                                       # (BLOCK_D, 3*H)

        gate_acc += tl.dot(x_norm, gate_w)[0]                  # (3*H,)

    # sigmoid for the three gates (the projection part stays linear)
    gate_acc = tl.sigmoid(gate_acc)

    # ------------------------------------------------------------------
    # 3) Store results (cast back to fp16)
    # ------------------------------------------------------------------
    # proj_out: [P, 2*H]
    proj_ptrs = proj_out_ptr + pos[:, None] * stride_proj_out_batch + tl.arange(0, 2 * HIDDEN_DIM)[None, :] * stride_proj_out_dim
    tl.store(proj_ptrs, proj_acc.to(tl.float16), mask=mask_pos[:, None])

    # gate_out: [P, 3*H]
    gate_ptrs = gate_out_ptr + pos[:, None] * stride_gate_out_batch + tl.arange(0, 3 * HIDDEN_DIM)[None, :] * stride_gate_out_dim
    tl.store(gate_ptrs, gate_acc.to(tl.float16), mask=mask_pos[:, None])


# ------------------------------------------------------------------
# 2. Batched mat‑mul kernel for the cubic update
# ------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B: batch, N, K   (will be read as K×N)
    stride_ch, stride_cm, stride_cn,   # C: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (B·H)

    # offsets for the tile that this program works on
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)   # (BLOCK_M, BLOCK_K)

        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_N)

    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor,
                         BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """Batched (batch, M, K) @ (batch, N, K)ᵀ → (batch, M, N) (FP16 inputs, FP32 acc)."""
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C


# ------------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Args:
        data: (input_tensor, mask, weights, config)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) or None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 0) unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Cast input to fp16 (contiguous) and flatten the first three axes
    # --------------------------------------------------------------
    x_fp16 = input_tensor.to(torch.float16).contiguous()
    B, N, _, _ = x_fp16.shape
    total_pos = B * N * N
    x_flat = x_fp16.view(total_pos, dim).contiguous()   # [P, D]

    # --------------------------------------------------------------
    # 2) Prepare all weight tensors (transpose + fp16 + contiguous)
    # --------------------------------------------------------------
    # LayerNorm
    norm_w = weights["norm.weight"].to(torch.float16).contiguous()
    norm_b = weights["norm.bias"].to(torch.float16).contiguous()

    # Projection weights: (hidden_dim, dim) -> transpose to (dim, hidden_dim)
    left_proj_w  = weights["left_proj.weight"].t().contiguous().to(torch.float16)
    right_proj_w = weights["right_proj.weight"].t().contiguous().to(torch.float16)

    # Gate weights (same shape as projection)
    left_gate_w  = weights["left_gate.weight"].t().contiguous().to(torch.float16)
    right_gate_w = weights["right_gate.weight"].t().contiguous().to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].t().contiguous().to(torch.float16)

    # Concatenate for a single mat‑vec per D‑tile
    proj_weight = torch.cat([left_proj_w, right_proj_w], dim=1).contiguous()          # [D, 2*H]
    gate_weight = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=1).contiguous()  # [D, 3*H]

    # --------------------------------------------------------------
    # 3) Allocate output buffers for the fused kernel
    # --------------------------------------------------------------
    proj_out = torch.empty((total_pos, 2 * hidden_dim), dtype=torch.float16, device=device)
    gate_out = torch.empty((total_pos, 3 * hidden_dim), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 4) Launch fused LayerNorm + projection + gates kernel
    # --------------------------------------------------------------
    BLOCK_POS = 1          # one position per program
    BLOCK_D   = 32         # tile size over the D dimension

    grid = (total_pos + BLOCK_POS - 1) // BLOCK_POS
    fused_ln_proj_gate_kernel[grid](
        x_flat,
        norm_w, norm_b,
        proj_weight,
        gate_weight,
        proj_out,
        gate_out,
        total_pos, dim,
        HIDDEN_DIM=hidden_dim,
        # strides
        x_flat.stride(0), x_flat.stride(1),
        norm_w.stride(0), norm_b.stride(0),
        proj_weight.stride(0), proj_weight.stride(1),
        gate_weight.stride(0), gate_weight.stride(1),
        # output strides
        proj_out.stride(0), proj_out.stride(1),
        gate_out.stride(0), gate_out.stride(1),
        BLOCK_POS=BLOCK_POS,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 5) Split concatenated results back to individual tensors
    # --------------------------------------------------------------
    left  = proj_out[:, :hidden_dim]                      # [P, H]
    right = proj_out[:, hidden_dim:]                      # [P, H]

    left_gate  = gate_out[:, :hidden_dim]                 # [P, H]
    right_gate = gate_out[:, hidden_dim:2*hidden_dim]    # [P, H]
    out_gate   = gate_out[:, 2*hidden_dim:]              # [P, H]

    # --------------------------------------------------------------
    # 6) Apply optional mask (broadcasted) – mask shape [B,N,N]
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.view(-1, 1).to(torch.float16)   # [P,1]
        left  = left * m
        right = right * m

    # --------------------------------------------------------------
    # 7) Apply the three sigmoid gates to left/right
    # --------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8) Reshape for the cubic mat‑mul (batched mat‑mul)
    # --------------------------------------------------------------
    # From [P, H] → [B, N, N, H]
    left = left.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()

    # Merge batch & hidden_dim for the kernel: (B*H, N, N)
    left_mat  = left.view(B * hidden_dim, N, N)
    right_mat = right.view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 9) Core cubic update (batched mat‑mul)
    # ------------------------------------------------------------------
    out_mat = batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )   # (B*H, N, N), fp16

    # Back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # --------------------------------------------------------------
    # 10) Output LayerNorm, out‑gate, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).contiguous()
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).contiguous()
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # reshape out_gate to [B,N,N,H] and apply
    out_gate = out_gate.view(B, N, N, hidden_dim).contiguous()
    out = out * out_gate

    # Final linear projection back to original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16).contiguous()
    out = F.linear(out, to_out_w)   # fp16

    # Cast to float32 as required by the API
    return out.to(torch.float32)
