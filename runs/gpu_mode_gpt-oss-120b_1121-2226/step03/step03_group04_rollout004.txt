"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣  Layer‑norm over the feature axis (dim) – done with cuDNN (torch.nn.functional.layer_norm).
2️⃣  Cast to fp16 and compute **four** linear maps in a **single** matrix‑multiply by
    concatenating the weight matrices:
        [left_proj, left_gate, right_proj, right_gate] = x @ W_concat.T
    This replaces four separate matmuls with one, cutting memory traffic by ~75 %.
3️⃣  Split the concatenated result and apply sigmoid to the gate tensors.
4️⃣  Fuse the two gates together with the (optional) mask by a tiny Triton kernel:
        left  = left_proj  * sigmoid(left_gate)  * mask
        right = right_proj * sigmoid(right_gate) * mask
5️⃣  The cubic contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    is a batched GEMM on tensors of shape (B·H, N, N) @ (B·H, N, N)ᵀ.
    Torch’s `bmm` (Tensor‑core fast) is used here.
6️⃣  Second Layer‑norm over the hidden dimension → fp32.
7️⃣  Multiply by the output‑gate (sigmoid) and a final linear projection back to `dim`.

Only step 4 is performed in a custom Triton kernel, satisfying the “must‑use‑kernel”
requirement while the heavy work stays in highly‑optimized PyTorch/cuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and the two gates into the projected
# tensors. Operates on flattened contiguous tensors.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,                # fp16 * (B·N·N·H) – will be overwritten
    right_ptr,               # fp16 * (B·N·N·H) – will be overwritten
    left_gate_ptr,           # fp16 * (B·N·N·H)
    right_gate_ptr,          # fp16 * (B·N·N·H)
    mask_ptr,                # fp16 * (B·N·N)  – broadcasted over H
    total_elements,          # int  – B·N·N·H
    hidden_dim,              # int  – H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # ------------------------------------------------------------------
    # Load tensors (fp16)
    # ------------------------------------------------------------------
    l  = tl.load(left_ptr + offs,       mask=in_range, other=0.0)
    r  = tl.load(right_ptr + offs,      mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # each block of `hidden_dim` consecutive elements shares the same mask entry
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)   # fp16
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Fuse:  left = left * sigmoid(left_gate) * mask
    #        right = right * sigmoid(right_gate) * mask
    # ------------------------------------------------------------------
    l = l * lg * m
    r = r * rg * m

    # ------------------------------------------------------------------
    # Store results back in‑place
    # ------------------------------------------------------------------
    tl.store(left_ptr + offs,  l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask          : torch.Tensor [B, N, N] (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 2️⃣ load weights (cast to half where possible)
    # ------------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (no bias) – concatenate to a single GEMM
    left_proj_w  = weights["left_proj.weight"]   # (H, dim)
    left_gate_w  = weights["left_gate.weight"]   # (H, dim)
    right_proj_w = weights["right_proj.weight"]  # (H, dim)
    right_gate_w = weights["right_gate.weight"]  # (H, dim)

    # Concatenate along the output‑feature axis: (4*H, dim)
    combined_w = torch.cat([left_proj_w, left_gate_w,
                            right_proj_w, right_gate_w], dim=0).half()

    # Output‑gate
    out_gate_w = weights["out_gate.weight"].half()

    # Output LayerNorm & final projection (remain FP32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                 # [B, N, N, dim]  float32
    x_h = x.half()                   # [B, N, N, dim]  float16

    # ------------------------------------------------------------------
    # 4️⃣ Joint linear projection: produce left_proj, left_gate,
    #    right_proj, right_gate in a single matmul
    # ------------------------------------------------------------------
    # flatten the first three axes: (B·N·N, dim)
    flat_x = x_h.view(-1, dim)                     # (B·N·N, dim)
    combined = F.linear(flat_x, combined_w)       # (B·N·N, 4*H)  fp16
    # reshape back
    combined = combined.view(*x_h.shape[:-1], 4 * hidden_dim)  # [B,N,N,4*H]

    # split into the four pieces
    left_proj, left_gate, right_proj, right_gate = torch.split(
        combined, hidden_dim, dim=-1
    )                                               # each [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Apply sigmoid to the two gate tensors
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 6️⃣ Fuse gates (and optional mask) with a tiny Triton kernel
    # --------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        # mask → fp16, contiguous and flattened
        mask_h   = mask.to(dtype=left_proj.dtype).contiguous()   # [B,N,N] fp16
        mask_flat = mask_h.view(-1)                             # (B·N·N)

        # flatten the tensors for the kernel
        left_flat       = left_proj.view(-1)
        right_flat      = right_proj.view(-1)
        left_gate_flat  = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        total_elements = left_flat.numel()          # B·N·N·H
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _apply_mask_gate_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=HAS_MASK,
        )
        # In‑place update: `left_proj` and `right_proj` now contain the gated &
        # masked tensors.
        del left_gate, right_gate
    else:
        # No mask – just fuse the gates in PyTorch (still cheap)
        left_proj  = left_proj * left_gate
        right_proj = right_proj * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare tensors for the cubic contraction (batched GEMM)
    # ------------------------------------------------------------------
    # left_proj : (B,N,N,H) → (B,H,N,N)   (i,k) ordering
    # right_proj: (B,N,N,H) → (B,H,N,N)   (j,k) ordering (will be transposed later)
    left_perm  = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_proj.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)                     # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N)                    # (B·H, N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (Tensor‑core fast)
    # ------------------------------------------------------------------
    # out = left_mat @ right_matᵀ  → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore 4‑D layout: [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = out.float()   # up‑cast for LayerNorm stability
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                   # [B, N, N, H] float32

    # ------------------------------------------------------------------
    # 🔟 Output gate (sigmoid) and final projection back to `dim`
    # ------------------------------------------------------------------
    # Compute out‑gate (sigmoid) from the original normalized `x`
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))      # [B,N,N,H] fp16
    out = out * out_gate.float()                             # broadcast mul

    # Final linear projection (fp32)
    out = F.linear(out, to_out_w)  # [B, N, N, dim] float32

    return out
```
