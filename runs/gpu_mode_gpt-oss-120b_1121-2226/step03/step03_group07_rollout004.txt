"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module but moves the
heavy N³ “pairwise multiplication” into a single batched GEMM performed
by cuBLAS (torch.bmm).  cuBLAS on an H100 provides FP16 Tensor‑Core
matrix multiplication with FP32 accumulation, which is far faster than a
hand‑written Triton kernel for the same work.  To satisfy the requirement
of using Triton we keep a tiny kernel that casts the optional FP32 mask
to FP16.

Algorithm
---------
1️⃣  Layer‑Norm the 4‑D input (FP32, with learned weight/bias).
2️⃣  Cast the normalized tensor to FP16 – all subsequent arithmetic lives
    in half‑precision.
3️⃣  Six linear layers (two projections, three gates, one output gate) are
    executed with FP16 weights (`F.linear`).  The gating sigmoids are
    applied in FP16.
4️⃣  If a mask is provided, a small Triton kernel converts it from FP32
    to FP16 and broadcasts it over the hidden dimension.
5️⃣  Element‑wise mask, left‑gate and right‑gate are fused with the
    projections (simple pointwise ops).
6️⃣  Core operation:
       outᵦᵢⱼₕ = Σₖ leftᵦᵢₖₕ · rightᵦⱼₖₕ
   is realised as a batched matrix‑multiply
       L @ Rᵀ
   where L/R have shape (B·H, N, N).  torch.bmm uses the highly‑optimised
   cuBLAS Tensor‑Core path (FP16 inputs → FP32 accumulation).  The result
   is cast back to FP32.
7️⃣  Output LayerNorm over the hidden dimension (FP32), multiplication by
    the output gate, and a final FP32 linear projection back to `dim`.

All tensors stay on the GPU; the only Triton‑based work is the optional
mask conversion.  The final output is float32 and matches the reference
implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Tiny Triton kernel: cast FP32 mask → FP16
# -------------------------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,            # fp32 source pointer
    dst_ptr,            # fp16 destination pointer
    numel,              # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)

# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim] (float32)
        - mask         : torch.Tensor [bs, seq_len, seq_len] (float32) – may be dummy
        - weights      : dict of model weights (float32 tensors)
        - config       : dict containing at least ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device

    dim = config["dim"]                # input feature dimension
    hidden_dim = config["hidden_dim"]  # internal hidden dimension
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)  # [B, N, N, dim]  (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Load weights (cast to FP16 where applicable)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()
    # Output‑norm & final linear stay in FP32 (as in the reference)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]       # (dim, hidden_dim), FP32

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)    # [B, N, N, hidden]   (FP16)
    right = F.linear(x_h, right_proj_w)   # [B, N, N, hidden]   (FP16)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask from FP32 → FP16 with Triton
        mask_fp16 = torch.empty_like(mask, dtype=torch.float16, device=device)
        BLOCK = 1024
        grid = ((mask.numel() + BLOCK - 1) // BLOCK,)
        _cast_fp32_to_fp16_kernel[grid](
            mask,
            mask_fp16,
            mask.numel(),
            BLOCK_SIZE=BLOCK,
        )
        # Broadcast to hidden dimension
        mask_h = mask_fp16.view(mask.shape + (1,))   # [B, N, N, 1] (FP16)
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – keep in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B, N, N, hidden] (FP16)
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B, N, N, hidden] (FP16)
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B, N, N, hidden] (FP16)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ operation – batched GEMM L @ Rᵀ using cuBLAS
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                         # B, seq_len, seq_len, hidden
    batch = B * H                                    # effective batch for GEMM

    # Re‑order to (batch, N, N) layout expected by torch.bmm
    # (B, N, N, H) → (B, H, N, N) → (batch, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)   # FP16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)  # FP16

    # Batched matrix multiply: each slice computes L @ Rᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))              # FP16, FP32 accumulation
    # Convert to FP32 for the following LayerNorm
    out_fp32 = out_mat.to(torch.float32)

    # Restore original shape: (B, N, N, hidden)
    out = out_fp32.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()     # FP32

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm over hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # FP32

    # ------------------------------------------------------------------
    # 9️⃣ Multiply by output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)          # element‑wise gating
    out = F.linear(out, to_out_w)                  # FP32 → [B, N, N, dim]

    return out
