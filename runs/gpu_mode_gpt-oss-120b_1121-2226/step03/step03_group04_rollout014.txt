def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    eps = 1e-5
    
    # Load weights
    # Norm weight & bias (first LN)
    norm_w = weights["norm.weight"].to(device).half()
    norm_b = weights["norm.bias"].to(device).half()
    # Proj weights
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()
    # Output LN
    out_norm_w = weights["to_out_norm.weight"].to(device).half()
    out_norm_b = weights["to_out_norm.bias"].to(device).half()
    # Final linear weight
    to_out_w = weights["to_out.weight"].to(device).half()
    
    # 1. First layer norm
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)   # float32 result?
    # Convert to half
    x_h = x.half()
    
    # 2. Left/right projections and gates (half)
    left_proj = F.linear(x_h, left_proj_w)    # (B,N,N,hidden)
    right_proj = F.linear(x_h, right_proj_w)
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))
    
    B, N, _, H = left_proj.shape
    
    # 3. Fuse mask, gates and reorder to (B*H,N,N) layout for both sides using a Triton kernel
    # Flatten and allocate outputs
    total_batch = B * H
    left_mat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)
    right_mat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)
    
    if mask is not None:
        mask_h = mask.to(device).half()
    else:
        mask_h = None
    
    # Launch kernel
    _fuse_reorder_kernel[
        (total_batch, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_K - 1) // BLOCK_K)
    ](
        left_proj, left_gate,
        right_proj, right_gate,
        mask_h,
        left_mat, right_mat,
        B, N, H,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        # mask strides (if mask_h is not None else 0)
        mask_h.stride(0) if mask_h is not None else 0,
        mask_h.stride(1) if mask_h is not None else 0,
        mask_h.stride(2) if mask_h is not None else 0,
        left_mat.stride(0), left_mat.stride(1), left_mat.stride(2),
        BLOCK_I=64, BLOCK_K=64, HAS_MASK=1 if mask_h is not None else 0
    )
    
    # 4. Batched GEMM (FP16) -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)
    
    # 5. Reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    
    # 6. Second layer norm (half)
    out = F.layer_norm(out, (hidden_dim,), weight=out_norm_w, bias=out_norm_b)
    
    # 7. Apply out_gate and final projection (half)
    out = out * out_gate
    out = F.linear(out, to_out_w)  # result half
    # Convert to fp32 per spec
    return out.float()

```
@triton.jit
def _fuse_reorder_kernel(
    left_proj_ptr, left_gate_ptr,
    right_proj_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, HAS_MASK: tl.constexpr
):
    pid_bh = tl.program_id(0)  # batch * hidden dimension
    pid_i = tl.program_id(1)
    pid_k = tl.program_id(2)

    # compute indices
    bh = pid_bh
    b = bh // H
    h = bh % H

    i_start = pid_i * BLOCK_I
    k_start = pid_k * BLOCK_K

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_k = k_start + tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_k = offs_k < N
    in_range = mask_i[:, None] & mask_k[None, :]   # shape (BLOCK_I, BLOCK_K)

    # Compute addresses
    # For left
    left_base = left_proj_ptr + b*stride_lp_b + h*stride_lp_h
    left = tl.load(
        left_base + offs_i[:, None]*stride_lp_i + offs_k[None, :]*stride_lp_k,
        mask=in_range, other=0.0
    )
    left_gate_base = left_gate_ptr + b*stride_lg_b + h*stride_lg_h
    left_gate = tl.load(
        left_gate_base + offs_i[:, None]*stride_lg_i + offs_k[None, :]*stride_lg_k,
        mask=in_range, other=0.0
    )
    # Right
    right_base = right_proj_ptr + b*stride_lp_b + h*stride_lp_h  # reuse stride as same shape
    right = tl.load(
        right_base + offs_i[:, None]*stride_lp_i + offs_k[None, :]*stride_lp_k,
        mask=in_range, other=0.0
    )
    right_gate_base = right_gate_ptr + b*stride_lg_b + h*stride_lg_h
    right_gate = tl.load(
        right_gate_base + offs_i[:, None]*stride_lg_i + offs_k[None, :]*stride_lg_k,
        mask=in_range, other=0.0
    )
    if HAS_MASK:
        mask_base = mask_ptr + b*stride_mask_b
        m = tl.load(
            mask_base + offs_i[:, None]*stride_mask_i + offs_k[None, :]*stride_mask_k,
            mask=in_range, other=1.0
        )
    else:
        m = tl.full((BLOCK_I, BLOCK_K), 1.0, dtype=tl.float16)

    # Apply gates and mask
    left_out = left * left_gate * m
    right_out = right * right_gate * m

    # Write out
    out_base = left_out_ptr + bh*stride_out_b
    tl.store(out_base + offs_i[:, None]*stride_out_i + offs_k[None, :]*stride_out_k,
             left_out, mask=in_range)
    out_base2 = right_out_ptr + bh*stride_out_b
    tl.store(out_base2 + offs_i[:, None]*stride_out_i + offs_k[None, :]*stride_out_k,
             right_out, mask=in_range)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module but moves the
expensive element‑wise mask/gate fusion and the layout transformation for the
cubic contraction into a single Triton kernel.  The heavy N³ work is still
performed by cuBLAS via a batched GEMM (torch.bmm) on FP16 data, while the
rest of the pipeline stays in FP16 to keep the memory traffic low.
A tiny Triton kernel is therefore required (the problem statement) and the
overall runtime is reduced by eliminating the two large ``permute(...).contiguous()``
copies that the naïve implementation needs.

Key steps
---------
1️⃣  Layer‑norm over the input feature dimension (FP16).
2️⃣  Linear projections and gating (FP16, bias‑free).
3️⃣  **Fusion kernel** – loads the two projections, the two gates and the optional
    mask, applies ``proj * gate * mask`` and writes the result directly into
    a contiguous layout ``(B·H, N, N)`` that is ready for a batched GEMM.
4️⃣  Batched matrix multiply (FP16) – replaces the explicit ``permute``/``view``.
5️⃣  Second Layer‑norm over the hidden dimension (FP16) + output‑gate.
6️⃣  Final linear projection back to ``dim`` (FP16) and cast to ``float32``.

The kernel works for any batch size (1‑2), any sequence length up to 1024 and
both masked / un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) + gates and reorder the tensors
# to a contiguous (batch*hidden, N, N) layout required by torch.bmm.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_reorder_kernel(
    left_proj_ptr,     # fp16 * (B, N, N, H)
    left_gate_ptr,     # fp16 * (B, N, N, H)
    right_proj_ptr,    # fp16 * (B, N, N, H)
    right_gate_ptr,    # fp16 * (B, N, N, H)
    mask_ptr,          # fp16 * (B, N, N)   (broadcasted over H)
    left_out_ptr,      # fp16 * (B*H, N, N)
    right_out_ptr,     # fp16 * (B*H, N, N)
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_k, stride_lp_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, HAS_MASK: tl.constexpr,
):
    # program ids
    pid_bh = tl.program_id(0)                 # (batch * hidden) dimension
    pid_i  = tl.program_id(1)                 # block row   inside a matrix
    pid_k  = tl.program_id(2)                 # block column inside a matrix

    # decode batch & hidden indices
    bh = pid_bh
    b = bh // H
    h = bh % H

    # block offsets in the N‑dimensional matrix
    i0 = pid_i * BLOCK_I
    k0 = pid_k * BLOCK_K
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_k = k0 + tl.arange(0, BLOCK_K)

    # mask for the boundaries (N may not be a multiple of BLOCK_*)
    i_valid = offs_i < N
    k_valid = offs_k < N
    mask_2d = i_valid[:, None] & k_valid[None, :]

    # ------------------------------------------------------------------
    # Load left projection + left gate
    # ------------------------------------------------------------------
    base_lp = left_proj_ptr + b * stride_lp_b + h * stride_lp_h
    left      = tl.load(base_lp + offs_i[:, None] * stride_lp_i + offs_k[None, :] * stride_lp_k,
                        mask=mask_2d, other=0.0)
    base_lg = left_gate_ptr + b * stride_lg_b + h * stride_lg_h
    left_gate = tl.load(base_lg + offs_i[:, None] * stride_lg_i + offs_k[None, :] * stride_lg_k,
                        mask=mask_2d, other=0.0)

    # ------------------------------------------------------------------
    # Load right projection + right gate
    # ------------------------------------------------------------------
    base_rp = right_proj_ptr + b * stride_lp_b + h * stride_lp_h
    right      = tl.load(base_rp + offs_i[:, None] * stride_lp_i + offs_k[None, :] * stride_lp_k,
                         mask=mask_2d, other=0.0)
    base_rg = right_gate_ptr + b * stride_lg_b + h * stride_lg_h
    right_gate = tl.load(base_rg + offs_i[:, None] * stride_lg_i + offs_k[None, :] * stride_lg_k,
                         mask=mask_2d, other=0.0)

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        base_m = mask_ptr + b * stride_mask_b
        m = tl.load(base_m + offs_i[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k,
                    mask=mask_2d, other=1.0)          # fp16
    else:
        m = tl.full((BLOCK_I, BLOCK_K), 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Apply:  out = proj * gate * mask   (both left and right)
    # ------------------------------------------------------------------
    left_out  = left  * left_gate  * m
    right_out = right * right_gate * m

    # ------------------------------------------------------------------
    # Store into the contiguous (B*H, N, N) buffers
    # ------------------------------------------------------------------
    out_base_l = left_out_ptr + bh * stride_out_b
    tl.store(out_base_l + offs_i[:, None] * stride_out_i + offs_k[None, :] * stride_out_k,
             left_out, mask=mask_2d)

    out_base_r = right_out_ptr + bh * stride_out_b
    tl.store(out_base_r + offs_i[:, None] * stride_out_i + offs_k[None, :] * stride_out_k,
             right_out, mask=mask_2d)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold‑3 style).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5
    device     = input_tensor.device
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 2️⃣ cast weights to the most efficient dtype (FP16 where possible)
    # --------------------------------------------------------------
    # LayerNorm 1
    norm_w = weights["norm.weight"].to(device).half()
    norm_b = weights["norm.bias"].to(device).half()

    # Linear projections (no bias)
    left_proj_w  = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()

    # Gates
    left_gate_w  = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w   = weights["out_gate.weight"].to(device).half()

    # Output LayerNorm & final projection
    to_out_norm_w = weights["to_out_norm.weight"].to(device).half()
    to_out_norm_b = weights["to_out_norm.bias"].to(device).half()
    to_out_w      = weights["to_out.weight"].to(device).half()   # hidden_dim -> dim

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32 -> half)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )                                   # [B, N, N, dim]  float32
    x = x.half()                        # keep everything in FP16 from now on

    # --------------------------------------------------------------
    # 4️⃣ Linear projections and gates (all FP16)
    # --------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_w)                 # [B, N, N, hidden_dim]
    right_proj = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 5️⃣ Fuse mask + gates + layout re‑ordering with a Triton kernel
    # --------------------------------------------------------------
    # prepare mask (optional) – broadcasted over the hidden dimension
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(device).half().contiguous()
        HAS_MASK = 1
    else:
        # dummy tensor; it will never be read because HAS_MASK==0
        mask_h = torch.empty((1,), dtype=torch.float16, device=device)
        HAS_MASK = 0

    # output buffers in the layout required by the batched GEMM
    total_batch = B * hidden_dim                # B·H
    left_mat  = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)
    right_mat = torch.empty((total_batch, N, N), dtype=torch.float16, device=device)

    # stride information (all tensors are contiguous)
    lp_s0, lp_s1, lp_s2, lp_s3 = left_proj.stride()   # (B, N, N, H)
    lg_s0, lg_s1, lg_s2, lg_s3 = left_gate.stride()
    rp_s0, rp_s1, rp_s2, rp_s3 = right_proj.stride()
    rg_s0, rg_s1, rg_s2, rg_s3 = right_gate.stride()
    mk_s0, mk_s1, mk_s2      = mask_h.stride() if HAS_MASK else (0, 0, 0)
    out_s0, out_s1, out_s2   = left_mat.stride()    # same for right_mat

    # block sizes – 64×64 tiles give good occupancy on H100
    BLOCK_I = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_K - 1) // BLOCK_K,
    )

    _mask_gate_fuse_reorder_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask_h,
        left_mat, right_mat,
        B, N, hidden_dim,
        lp_s0, lp_s1, lp_s2, lp_s3,
        lg_s0, lg_s1, lg_s2, lg_s3,
        mk_s0, mk_s1, mk_s2,
        out_s0, out_s1, out_s2,
        BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K, HAS_MASK=HAS_MASK,
    )
    # we can optionally free the now‑unused tensors
    del left_proj, right_proj, left_gate, right_gate, mask_h

    # --------------------------------------------------------------
    # 6️⃣ Batched GEMM (FP16, Tensor‑core accelerated)
    # --------------------------------------------------------------
    # left_mat / right_mat : (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)          # (B·H, N, N)  FP16

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )   # still FP16

    # --------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection (both FP16)
    # --------------------------------------------------------------
    out = out * out_gate            # element‑wise gate
    out = F.linear(out, to_out_w)   # -> [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 9️⃣ Cast to FP32 as required by the API
    # --------------------------------------------------------------
    return out.float()
