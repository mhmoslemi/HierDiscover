@triton.jit
def layer_norm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    eps,
    total_rows: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    row = tl.program_id(0)
    if row >= total_rows:
        return

    offs_h = tl.arange(0, BLOCK_H)
    mask = offs_h < hidden_dim
    # compute mean
    x = tl.load(x_ptr + row * hidden_dim + offs_h, mask=mask, other=0.0)
    # cast to fp32
    x_fp32 = x.to(tl.float32)
    sum_val = tl.sum(x_fp32, axis=0)
    mean = sum_val / hidden_dim

    # compute variance
    diff = x_fp32 - mean
    var = tl.sum(diff * diff, axis=0) / hidden_dim

    # compute inverse std
    inv_std = 1.0 / tl.sqrt(var + eps)

    # load weight and bias
    w = tl.load(weight_ptr + offs_h, mask=mask)
    b = tl.load(bias_ptr + offs_h, mask=mask)

    # normalize
    out = (x_fp32 - mean) * inv_std * w.to(tl.float32) + b.to(tl.float32)

    # store as fp16
    tl.store(out_ptr + row * hidden_dim + offs_h, out.to(tl.float16), mask=mask)

"""
custom_kernel – Triton‑accelerated forward pass for the outgoing TriMul
operation (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (over the last channel) – still in float16.
2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single GEMM by concatenating their weight matrices.  This yields
   tensors `left`, `right`, `left_gate`, `right_gate`, `out_gate` of shape
   [B, N, N, H] (H = hidden_dim) in one pass.
3️⃣ If a mask is present, apply it together with the left/right gates using a
   tiny Triton kernel (`_apply_mask_gate_kernel`).  The kernel works on the
   flattened tensors and updates `left` and `right` in‑place:
      left  ← left  * mask * left_gate
      right ← right * mask * right_gate
   When there is no mask we simply multiply the gates.
4️⃣ Core N³ update – compute for every hidden channel h
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is exactly a batched matrix‑multiply over the reshaped tensors
      (B·H, N, N) @ (B·H, N, N)ᵀ   performed with torch.bmm in fp16 (Tensor‑Cores).
5️⃣ Layer‑norm over the hidden dimension, element‑wise multiply by the
   output‑gate, and a final linear projection back to the original channel
   dimension.  All heavy work stays in fp16; the final result is cast to
   float32 as required.
Only a single Triton kernel is needed (step 3); the rest uses highly
optimised torch operations while keeping memory traffic to a minimum.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------
# Tiny Triton kernel: in‑place mask & gate fusion for the left/right tensors.
# --------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,            # fp16*  [B,N,N,H] flattened
    right_ptr,           # fp16*  same layout as left
    mask_ptr,            # fp16*  [B,N,N] flattened (no hidden dim)
    left_gate_ptr,       # fp16*  [B,N,N,H] flattened
    right_gate_ptr,      # fp16*  [B,N,N,H] flattened
    total_elements: tl.constexpr,   # B*N*N*H (int)
    hidden_dim:      tl.constexpr,   # H (compile‑time known)
    BLOCK_SIZE:      tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors (masked loads avoid OOB)
    left      = tl.load(left_ptr      + offs, mask=mask)
    right     = tl.load(right_ptr     + offs, mask=mask)
    l_gate    = tl.load(left_gate_ptr + offs, mask=mask)
    r_gate    = tl.load(right_gate_ptr+ offs, mask=mask)

    # Broadcast mask over the hidden dimension
    mask_idx = offs // hidden_dim               # index into [B,N,N] view
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply mask and gates (in‑place)
    left  = left  * mask_val * l_gate
    right = right * mask_val * r_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor: torch.Tensor [B, N, N, D] (float32)
              - mask: torch.Tensor [B, N, N] (bool/int) – may be None when nomask=True
              - weights: dict of model parameters (float32)
              - config: dict with keys "dim", "hidden_dim" and optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input channel dimension D
    hidden_dim = config["hidden_dim"]   # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (keep in fp16 thereafter)
    # ------------------------------------------------------------
    # nn.LayerNorm works in float32 for the statistics – acceptable.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, D]  float32
    x = x.to(torch.float16)            # switch to half precision

    # ------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps (proj + gates) into a single GEMM
    # ------------------------------------------------------------
    # Concatenate weight matrices (all are (H, D) in the reference implementation)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # (5*H, D)

    # Linear projection: shape -> [B, N, N, 5*H]
    fused_proj = F.linear(x, fused_weight)  # bias is None in all layers

    # Split back into the individual tensors
    left, right, left_gate, right_gate, out_gate = torch.split(
        fused_proj, hidden_dim, dim=-1
    )                                   # each [B, N, N, H]

    # Apply sigmoid to gate tensors
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------------------------------------------------
    # 3️⃣ Optional mask & gate fusion (Triton kernel)
    # ------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to fp16 (values 0.0 / 1.0) and flatten
        mask_f16 = mask.to(torch.float16).contiguous()  # [B, N, N]

        # Ensure all tensors are contiguous for pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_f16 = mask_f16.contiguous()

        total_elems = left.numel()                     # B * N * N * H
        BLOCK = 8192                                    # Tune as needed
        grid = ((total_elems + BLOCK - 1) // BLOCK, )

        # Kernel launches – in‑place updates of `left` and `right`
        _apply_mask_gate_kernel[grid](
            left,
            right,
            mask_f16,                     # mask without hidden dim
            left_gate,
            right_gate,
            total_elems,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
            num_stages=1,
        )
    else:
        # No mask → only apply per‑channel gates (cheap PyTorch ops)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------
    # 4️⃣ Core N³ update via batched half‑precision GEMM
    # ------------------------------------------------------------
    B, N, _, _ = left.shape               # B, seq_len, _, hidden_dim
    # reshape to (B*H, N, N) – each hidden channel becomes a separate batch
    left_mat = (
        left.permute(0, 3, 1, 2)          # [B, H, N, N]
        .reshape(B * hidden_dim, N, N)   # [B*H, N, N]
    )
    right_mat = (
        right.permute(0, 3, 1, 2)
        .reshape(B * hidden_dim, N, N)
    )

    # Batched matrix multiplication on Tensor‑Cores (fp16 × fp16 → fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # Restore the original 4‑D view: [B, N, N, H] (still fp16)
    out = (
        out_mat
        .view(B, hidden_dim, N, N)        # [B, H, N, N]
        .permute(0, 2, 3, 1)              # [B, N, N, H]
        .contiguous()
    )

    # ------------------------------------------------------------
    # 5️⃣ Output layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------
    # Layer‑norm over the hidden dimension H (weights are stored in fp32;
    # we keep the tensor in fp16 – PyTorch will cast internally).
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                   # fp16 → fp32 internally, result fp32

    # Apply the output gate (already sigmoid‑ed)
    out = out.to(torch.float16) * out_gate

    # Final linear projection back to original channel dimension D
    # The weight matrix is (D, H); we run in fp16.
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # Return float32 as required by the interface
    return out.to(torch.float32)
