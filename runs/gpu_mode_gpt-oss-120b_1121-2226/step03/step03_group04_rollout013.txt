"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

We implement the *outgoing* TriMul operator from AlphaFold‑3.
The heavy N³ work (Σₖ left[b,i,k,h]·right[b,j,k,h]) is performed with a
single `torch.einsum`, which internally maps to a highly‑optimised
batched GEMM and therefore avoids the costly permutations that a naïve
implementation would need.

To reduce memory traffic we:
1. Fuse each projection and its gate into one linear layer (weight
   concatenation).  This halves the number of passes over the large
   input tensor `x` (the original code performed four separate matmuls).
2. Fuse mask application, the gate‑sigmoid, and the element‑wise product
   into a tiny Triton kernel that works on the concatenated tensors
   directly, overwriting the projection slots in‑place.
3. Keep all heavy work in FP16 (Tensor‑core friendly) and only switch to
   FP32 for the two LayerNorms and the final projection.

The overall flow is:
   x    = LayerNorm(x)                         (FP32 → FP16)
   left_concat , right_concat = fused linears (FP16)
   out_gate  = sigmoid(fused linear)          (FP16)
   left , right = mask * sigmoid(gate) * proj (Triton kernel, FP16)
   out   = Σₖ left·right   (einsum → batched GEMM, FP16)
   out   = LayerNorm(out)                     (FP32)
   out   = out * out_gate                     (FP32)
   out   = Linear(out)                        (FP32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional), sigmoid(gate) and multiply with
# the projection.  Operates on the concatenated tensors of shape
# [B, N, N, 2*H] (contiguous in memory).  The kernel overwrites the
# projection part (first H elements) with the final gated value.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_sigmoid_kernel(
    left_ptr,                 # fp16 * (B·N·N·2*H)
    right_ptr,                # fp16 * (B·N·N·2*H)
    mask_ptr,                 # fp16 * (B·N·N)   (broadcasted over H)
    total_elements,           # i64  = B·N·N·H  (number of gated elements)
    hidden_dim,               # i32  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # ------------------------------------------------------------------
    # Compute location (b,i,k) and hidden index h from the flat offset.
    #   loc = floor(offs / H)   – index of the (b,i,k) element
    #   h   = offs % H          – hidden index inside that element
    # ------------------------------------------------------------------
    loc = offs // hidden_dim
    h   = offs % hidden_dim

    # ------------------------------------------------------------------
    # Offsets inside the concatenated (proj+gate) buffer.
    # Each location stores 2*H values: [proj(0:H), gate_raw(H:2H)].
    # ------------------------------------------------------------------
    base_offset = loc * (hidden_dim * 2)                # start of the pair
    l_proj_offset = base_offset + h                     # proj part
    l_gate_offset = base_offset + hidden_dim + h        # raw gate part
    r_proj_offset = base_offset + h
    r_gate_offset = base_offset + hidden_dim + h

    # ------------------------------------------------------------------
    # Load the four values.
    # ------------------------------------------------------------------
    l_proj = tl.load(left_ptr + l_proj_offset,  mask=in_range, other=0.0)
    l_gate_raw = tl.load(left_ptr + l_gate_offset, mask=in_range, other=0.0)
    r_proj = tl.load(right_ptr + r_proj_offset,  mask=in_range, other=0.0)
    r_gate_raw = tl.load(right_ptr + r_gate_offset, mask=in_range, other=0.0)

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # one mask value per (b,i,k) location
        mask_val = tl.load(mask_ptr + loc, mask=in_range, other=1.0)
    else:
        mask_val = 1.0

    # ------------------------------------------------------------------
    # Sigmoid: compute in FP32 for a little extra accuracy.
    # ------------------------------------------------------------------
    l_gate = 1.0 / (1.0 + tl.exp(-tl.float32(l_gate_raw)))
    r_gate = 1.0 / (1.0 + tl.exp(-tl.float32(r_gate_raw)))
    # Cast back to FP16 for the final multiplication
    l_gate = tl.float16(l_gate)
    r_gate = tl.float16(r_gate)

    # ------------------------------------------------------------------
    # Apply mask and gate, write the result back to the projection slot.
    # ------------------------------------------------------------------
    l_out = l_proj * l_gate * mask_val
    r_out = r_proj * r_gate * mask_val

    tl.store(left_ptr + l_proj_offset,  l_out, mask=in_range)
    tl.store(right_ptr + r_proj_offset, r_out, mask=in_range)

# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict with all model parameters (float32)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # model hyper‑parameter H

    # ------------------------------------------------------------------
    # 2️⃣ load weights (cast to half where possible)
    # ------------------------------------------------------------------
    # LayerNorm (first)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections + their gates (fused)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]

    # Out‑gate (separate)
    out_gate_w    = weights["out_gate.weight"]

    # Output LayerNorm & final projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                           # [B, N, N, dim]  float32
    x = x.half()                               # [B, N, N, dim]  float16

    # ------------------------------------------------------------------
    # 4️⃣ Fused linear: (proj , gate) for left and right side
    # ------------------------------------------------------------------
    left_fused_w  = torch.cat([left_proj_w, left_gate_w], dim=0).half()   # (2*H, dim)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0).half()

    # (B·N·N, 2*H) result, reshaped back to (B,N,N,2*H)
    left_concat  = F.linear(x, left_fused_w)    # fp16, shape [B,N,N,2*H]
    right_concat = F.linear(x, right_fused_w)   # fp16, shape [B,N,N,2*H]

    # ------------------------------------------------------------------
    # 5️⃣ Out‑gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(x, out_gate_w.half())                # [B,N,N,H] fp16
    out_gate = torch.sigmoid(out_gate_raw)                       # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask + sigmoid(gate) + element‑wise product (Triton)
    # ------------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        mask_h = mask.to(dtype=torch.float16, device=inp.device).contiguous()
        mask_flat = mask_h.view(-1)               # (B·N·N)
    else:
        # dummy tensor – never accessed when HAS_MASK == 0
        mask_flat = torch.empty(1, dtype=torch.float16, device=inp.device)

    # Ensure concatenated tensors are contiguous (required for pointer arithmetic)
    left_concat = left_concat.contiguous()
    right_concat = right_concat.contiguous()

    total_elements = left_concat.shape[0] * left_concat.shape[1] * left_concat.shape[2] * hidden_dim  # B·N·N·H
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _mask_gate_sigmoid_kernel[grid](
        left_concat,
        right_concat,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=HAS_MASK,
    )
    # ------------------------------------------------------------------
    # 7️⃣ Extract the gated projection tensors (first H values)
    # ------------------------------------------------------------------
    left  = left_concat[..., :hidden_dim]   # [B, N, N, H]   (non‑contiguous)
    right = right_concat[..., :hidden_dim]  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Cubic contraction: Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    `einsum` automatically maps to a batched GEMM and avoids the
    #    expensive permute+contiguous copies of the naive implementation.
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16, [B,N,N,H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm (float32) + output gate
    # ------------------------------------------------------------------
    out = out.float()   # up‑cast for stable LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                           # [B,N,N,H] float32
    out = out * out_gate.float()                # apply output gate

    # ------------------------------------------------------------------
    # 🔟 Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)               # [B,N,N,dim] float32
    return out
