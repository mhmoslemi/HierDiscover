def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_k, stride_h,
    N, H,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch and head indices
    b = pid_bh // H
    h = pid_bh % H

    # compute offsets
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    off = b * stride_b + h * stride_h

    # Block pointers
    a_ptr = tl.make_block_ptr(base=left_ptr + off,
                              shape=(N, N),
                              strides=(stride_i, stride_k),
                              offsets=(i_offset, 0),
                              block_shape=(BLOCK_M, BLOCK_K),
                              order=(0, 1))

    b_ptr = tl.make_block_ptr(base=right_ptr + off,
                              shape=(N, N),
                              strides=(stride_k, stride_i),   # Actually for right matrix we need (k stride, j stride) => (stride_k, stride_i)
                              offsets=(0, j_offset),
                              block_shape=(BLOCK_K, BLOCK_N),
                              order=(0, 1))

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):

@triton.jit
def trimul_outgoing_matmul_kernel(
    left_ptr,    # fp16
    right_ptr,   # fp16
    out_ptr,    # fp32
    stride_b,   # int
    stride_i,   # int
    stride_k,   # int
    stride_h,   # int
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ID dimensions
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_bh = tl.program_id(2)  # batch-head index

    # compute batch and head index
    b = pid_bh // H
    h = pid_bh % H

    # Row/col offsets within N
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # offset to beginning of the (b,h) matrix slice
    base = b * stride_b + h * stride_h

    # Create block pointers for A (left) and B (right)
    # A shape: (N, N) with row stride_i, col stride_k
    a_ptr = tl.make_block_ptr(
        base=left_ptr + base,
        shape=(N, N),
        strides=(stride_i, stride_k),
        offsets=(i_offset, 0),
        block_shape=(BLOCK_M, BLOCK_K),
        order=(0, 1)
    )
    # B shape: (N, N) with row stride_k, col stride_i (since we need transpose of right)
    b_ptr = tl.make_block_ptr(
        base=right_ptr + base,
        shape=(N, N),
        strides=(stride_k, stride_i),
        offsets=(0, j_offset),
        block_shape=(BLOCK_K, BLOCK_N),
        order=(0, 1)
    )

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        # Load current block
        a = tl.load(a_ptr, mask=(i_offset + tl.arange(0, BLOCK_M)[:,None] < N) &
                                (k + tl.arange(0, BLOCK_K)[None,:] < N), other=0.0).to(tl.float16)
        b = tl.load(b_ptr, mask=((k + tl.arange(0, BLOCK_K)[:,None] < N) &
                                 (j_offset + tl.arange(0, BLOCK_N)[None,:] < N)), other=0.0).to(tl.float16)

        # Perform matrix multiplication on the block
        # tl.dot does reduction over last dim of a and second-last dim of b
        acc += tl.dot(a, b)

        # Advance pointers
        a_ptr = tl.advance(a_ptr, (0, BLOCK_K))
        b_ptr = tl.advance(b_ptr, (BLOCK_K, 0))

    # Store result to out
    # Output stride for i = stride_i, for j = stride_k
    out_ptr_block = tl.make_block_ptr(
        base=out_ptr + base,
        shape=(N, N),
        strides=(stride_i, stride_k),
        offsets=(i_offset, j_offset),
        block_shape=(BLOCK_M, BLOCK_N),
        order=(0, 1)
    )
    # mask for storing
    mask_i = i_offset + tl.arange(0, BLOCK_M) < N
    mask_j = j_offset + tl.arange(0, BLOCK_N) < N
    mask = mask_i[:, None] & mask_j[None, :]

"""
TriMul “outgoing” forward pass – highly‑fused Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ [B, N, N, D] (float32).
2. Fuse the two projection linear layers (proj + gate) for the left and right
   tensors into a single mat‑mul per side:
        L_pre = X @ [W_proj ; W_gate]ᵀ   (shape …, 2·H)
        L_proj = L_pre[..., :H]          # projection
        L_gate = sigmoid(L_pre[..., H:]) # gate
   (same for R).  This halves the number of GEMMs.
3. If a mask is present, fuse mask, left‑gate and right‑gate multiplication in
   a tiny Triton element‑wise kernel (no extra memory copies).
4. Core N³ update:
        out[b,i,j,h] = Σ_k  L_proj[b,i,k,h] * R_proj[b,j,k,h]
   which is a batched matrix multiplication L @ Rᵀ for each (b, h).
   Implemented with a custom Triton kernel that works directly on the
   original [B, N, N, H] layout – avoids the permute‑contiguous copy needed
   by a naïve torch.bmm approach.
5. Multiply the result by the output‑gate (broadcasted), apply a LayerNorm
   over the hidden dimension, and a final linear projection back to D.
   All heavy work runs in FP16 (Tensor‑Core friendly); the N³ product
   accumulates in FP32 and the final LayerNorm / projection also use FP32.

The implementation meets the required API:
    custom_kernel(data) → torch.Tensor [B, N, N, D] (float32)
-----------------------------------------------------------------------
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) mask + gate fusion kernel (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,          # left tensors (FP16)
    right_ptr, right_gate_ptr,        # right tensors (FP16)
    mask_ptr,                         # mask broadcasted to shape [B,N,N,H] (FP16)
    total_elems: tl.int32,            # B·N·N·H
    BLOCK_SIZE: tl.constexpr,         # threads per program
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    within = offs < total_elems

    l_val   = tl.load(left_ptr   + offs, mask=within, other=0.0)
    l_gate  = tl.load(left_gate_ptr + offs, mask=within, other=0.0)
    r_val   = tl.load(right_ptr  + offs, mask=within, other=0.0)
    r_gate  = tl.load(right_gate_ptr + offs, mask=within, other=0.0)
    m_val   = tl.load(mask_ptr   + offs, mask=within, other=1.0)   # default 1 → no masking

    tl.store(left_ptr  + offs, l_val * l_gate * m_val, mask=within)
    tl.store(right_ptr + offs, r_val * r_gate * m_val, mask=within)


# ----------------------------------------------------------------------
# 2) core batched mat‑mul kernel (L @ Rᵀ) over the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_matmul_kernel(
    left_ptr,          # FP16  [B,N,N,H]
    right_ptr,         # FP16  [B,N,N,H]
    out_ptr,           # FP32  [B,N,N,H] – result
    stride_b, stride_i, stride_k, stride_h,   # strides of the 4‑D tensors
    N: tl.constexpr,   # sequence length
    H: tl.constexpr,   # hidden dimension
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D launch grid: (block_i, block_j, batch_head)
    pid_m = tl.program_id(0)          # block row (i)
    pid_n = tl.program_id(1)          # block col (j)
    pid_bh = tl.program_id(2)         # combined batch*head index

    # split batch and head
    b = pid_bh // H
    h = pid_bh - b * H                # pid_bh % H

    # starting coordinates of the current output tile
    i_offset = pid_m * BLOCK_M
    j_offset = pid_n * BLOCK_N

    # base offset for the (b, h) slice (the hidden dim is the innermost)
    base = b * stride_b + h * stride_h

    # ------------------------------------------------------------------
    # block pointers for the left (L) and right (R) matrices
    # L: shape (N, N) with row‑stride = stride_i, col‑stride = stride_k
    # R: we need Rᵀ, i.e. rows = k (stride_k), cols = j (stride_i)
    # ------------------------------------------------------------------
    a_ptr = tl.make_block_ptr(
        base=left_ptr + base,
        shape=(N, N),
        strides=(stride_i, stride_k),
        offsets=(i_offset, 0),
        block_shape=(BLOCK_M, BLOCK_K),
        order=(0, 1),
    )
    b_ptr = tl.make_block_ptr(
        base=right_ptr + base,
        shape=(N, N),
        strides=(stride_k, stride_i),
        offsets=(0, j_offset),
        block_shape=(BLOCK_K, BLOCK_N),
        order=(0, 1),
    )

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # indices used for masking
    i = tl.arange(0, BLOCK_M)
    j = tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # ------------------------------------------------------------------
    # loop over the reduction axis k
    # ------------------------------------------------------------------
    for kk in range(0, N, BLOCK_K):
        # mask for A and B loads
        a_mask = (i_offset + i < N)[:, None] & (kk + k < N)[None, :]
        b_mask = (kk + k < N)[:, None] & (j_offset + j < N)[None, :]

        a = tl.load(a_ptr, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K), FP16
        b = tl.load(b_ptr, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N), FP16

        acc += tl.dot(a, b)                          # (BLOCK_M, BLOCK_N) FP32

        # advance block pointers to the next k‑slice
        a_ptr = tl.advance(a_ptr, (0, BLOCK_K))
        b_ptr = tl.advance(b_ptr, (BLOCK_K, 0))

    # ------------------------------------------------------------------
    # store to the output tensor (FP32) – same layout as left/right
    # ------------------------------------------------------------------
    out_ptr_block = tl.make_block_ptr(
        base=out_ptr + base,
        shape=(N, N),
        strides=(stride_i, stride_k),
        offsets=(i_offset, j_offset),
        block_shape=(BLOCK_M, BLOCK_N),
        order=(0, 1),
    )

    store_mask_i = i_offset + i < N
    store_mask_j = j_offset + j < N
    store_mask = store_mask_i[:, None] & store_mask_j[None, :]
    tl.store(out_ptr_block, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None when nomask=True
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for all heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.half()   # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 4) Fused left/right projection + gate (single matmul per side)
    # ------------------------------------------------------------------
    # concatenate projection and gate weight matrices (dim, 2*hidden_dim)
    left_concat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()                               # (2*H, D)
    right_concat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()                               # (2*H, D)

    # single matmul per side
    left_concat = F.linear(x_f16, left_concat_weight)   # [B, N, N, 2*H] fp16
    right_concat = F.linear(x_f16, right_concat_weight)

    H = hidden_dim
    # split projection / gate
    left_proj = left_concat[..., :H]                     # [B, N, N, H]
    left_gate_pre = left_concat[..., H:]
    left_gate = torch.sigmoid(left_gate_pre)            # fp16

    right_proj = right_concat[..., :H]
    right_gate_pre = right_concat[..., H:]
    right_gate = torch.sigmoid(right_gate_pre)          # fp16

    # ------------------------------------------------------------------
    # 5) Output gate (separate linear)
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].half()
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_weight))   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6) Optional mask + gate fusion (via Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # expand mask to the full [B,N,N,H] layout (spares a stride‑0 broadcast)
        mask_f16 = mask.to(torch.float16).unsqueeze(-1).expand(-1, -1, -1, H).contiguous()
        # ensure contiguous tensors for the element‑wise kernel
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_f16 = mask_f16.contiguous()

        total = left_proj.numel()                        # B·N·N·H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left_proj.data_ptr(),
            left_gate.data_ptr(),
            right_proj.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # no mask → just apply the gates
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core N³ batched mat‑mul (L @ Rᵀ) – custom Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left_proj.shape               # [B, N, N, H]  (fp16)

    # allocate output tensor in FP32 (required by following LayerNorm)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # strides of the 4‑D tensors (identical for left/right)
    stride_b, stride_i, stride_k, stride_h = left_proj.stride()

    # launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_outgoing_matmul_kernel[grid](
        left_proj.data_ptr(),
        right_proj.data_ptr(),
        out.data_ptr(),
        stride_b,
        stride_i,
        stride_k,
        stride_h,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8) Apply output gate, output LayerNorm, final projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = F.linear(out, weights["to_out.weight"])   # back to D (float32)

    return out
