@triton.jit
def _apply_mask_gate_and_sigmoid_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # apply sigmoid
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))

    if HAS_MASK:
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)
```

    dim = config["dim"]; hidden_dim = config["hidden_dim"]

    norm_bias   = weights["norm.bias"]
    # Linear weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]
    # compute combined weight
    proj_combined_w = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0).half()
    out_gate_w = out_gate_w.half()  # half

    x_h = x.half()

@triton.jit
def _apply_mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    total_elements,
    # compile-time constants:
    BLOCK_SIZE: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    # Sigmoid
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    if HAS_MASK:
        mask_idx = offs // HIDDEN_DIM
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)
    l = l * lg * m
    r = r * rg * m
    tl.store(left_ptr + offs, l, mask=mask)
    tl.store(right_ptr + offs, r, mask=mask)
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Layer‑norm on the input (float32) → cast to FP16.
2️⃣ Fuse the four linear layers (left‑proj, right‑proj, left‑gate, right‑gate)
   into a **single** matrix multiplication:
        out = x @ W_combined            # shape …, 4·H
   Split the result into `left`, `right`, `left_gate`, `right_gate`.
3️⃣ Compute `out_gate` with a second linear layer (still in FP16).
4️⃣ Fuse mask (if present), the sigmoid of the two gates and the mask
   multiplication in a tiny Triton kernel (all in‑place, FP16).
5️⃣ Rearrange `left` and `right` → (B·H, N, N) and perform the cubic
   contraction as a batched GEMM (`torch.bmm`) which on H100 uses Tensor‑cores.
6️⃣ Second LayerNorm (float32) over the hidden dimension.
7️⃣ Apply the `out_gate` (sigmoid) and the final linear projection back to
   `dim`.  The final output is FP32.

All heavy arithmetic (matmuls, GEMM) stays in FP16; only the two LayerNorms
operate in FP32 for numerical stability.  The Triton kernel eliminates an
extra memory pass for the mask and the sigmoid gates.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16 * (B·N·N·H)
    right_ptr,              # fp16 * (B·N·N·H)
    left_gate_ptr,          # fp16 * (B·N·N·H)
    right_gate_ptr,         # fp16 * (B·N·N·H)
    mask_ptr,               # fp16 * (B·N·N)   (broadcasted over H)
    total_elems,            # int  (B·N·N·H)
    BLOCK_SIZE: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elems

    # ------------------------------------------------------------------
    # Load tensors (fp16)
    # ------------------------------------------------------------------
    l  = tl.load(left_ptr + offs,        mask=in_range, other=0.0)
    r  = tl.load(right_ptr + offs,       mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr + offs,   mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs,  mask=in_range, other=0.0)

    # ------------------------------------------------------------------
    # Sigmoid of gate values
    # ------------------------------------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # each group of `HIDDEN_DIM` consecutive elements shares the same mask entry
        mask_idx = offs // HIDDEN_DIM
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)   # fp16
    else:
        m = tl.full([BLOCK_SIZE], 1.0, tl.float16)

    # ------------------------------------------------------------------
    # Fuse: left = left * sigmoid(left_gate) * mask
    #       right = right * sigmoid(right_gate) * mask
    # ------------------------------------------------------------------
    l = l * lg * m
    r = r * rg * m

    # ------------------------------------------------------------------
    # Store results back in‑place
    # ------------------------------------------------------------------
    tl.store(left_ptr + offs,  l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict with model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]  # called H in the paper

    # --------------------------------------------------------------
    # 2️⃣ extract model weights (cast to half where possible)
    # --------------------------------------------------------------
    # LayerNorm 1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear weights (no bias) – keep as FP16
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Combine the four projection/gate matrices into one matmul
    proj_combined_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w], dim=0
    ).half()                              # (4·H, dim)

    out_gate_w = out_gate_w.half()        # (H, dim)

    # Output LayerNorm & final linear (remain FP32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]  # (dim, H)

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → cast to fp16
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                     # [B, N, N, dim]  float32
    x_h = x.half()       # [B, N, N, dim]  fp16

    # --------------------------------------------------------------
    # 4️⃣ Fused projection + gate linear layer (single matmul)
    # --------------------------------------------------------------
    #  (B, N, N, 4·H)
    combined = F.linear(x_h, proj_combined_w)

    # split into four tensors of shape (B, N, N, H)
    left, right, left_gate_raw, right_gate_raw = torch.split(
        combined, hidden_dim, dim=-1
    )

    # --------------------------------------------------------------
    # 5️⃣ Out‑gate (separate matmul) – keep in fp16 for now
    # --------------------------------------------------------------
    out_gate_raw = F.linear(x_h, out_gate_w)          # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 6️⃣ Fuse mask, sigmoid‑gates and multiplication (Triton)
    # --------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        # mask to half and flatten (broadcast over hidden_dim)
        mask_h = mask.to(dtype=left.dtype, device=left.device).contiguous()
        mask_flat = mask_h.view(-1)                     # (B·N·N,)

        # flatten tensors for the kernel
        left_flat       = left.view(-1)
        right_flat      = right.view(-1)
        left_gate_flat  = left_gate_raw.view(-1)
        right_gate_flat = right_gate_raw.view(-1)

        total_elements = left_flat.numel()               # B·N·N·H
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _mask_gate_fuse_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            BLOCK_SIZE=BLOCK_SIZE,
            HIDDEN_DIM=hidden_dim,
            HAS_MASK=HAS_MASK,
        )
        # clean up
        del left_gate_raw, right_gate_raw
    else:
        # No mask: apply sigmoid gates in PyTorch (still in‑place)
        left = left * torch.sigmoid(left_gate_raw)
        right = right * torch.sigmoid(right_gate_raw)
        del left_gate_raw, right_gate_raw

    # --------------------------------------------------------------
    # 7️⃣ Batched GEMM for the cubic contraction
    # --------------------------------------------------------------
    # left : (B, N, N, H) → (B, H, N, N)   (i,k)
    # right: (B, N, N, H) → (B, H, N, N)   (j,k)  (swap the two N axes)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)

    # Tensor‑core GEMM (fp16 → fp32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat)          # (B·H, N, N) fp16

    # Restore (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32)
    # --------------------------------------------------------------
    out = out.float()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                 # [B, N, N, H] float32

    # --------------------------------------------------------------
    # 9️⃣ Apply out‑gate (sigmoid) and final projection back to `dim`
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate_raw).float()   # [B,N,N,H] float32
    out = out * out_gate
    out = F.linear(out, to_out_w)                    # [B, N, N, dim] float32

    return out
