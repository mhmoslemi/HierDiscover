"""
Triton‑accelerated “outgoing” TriMul (Triangular Multiplicative Update).

Algorithm
---------
1.  Layer‑norm the input tensor X ∈ [B,N,N,D] in FP32 (high‑precision).
2.  Cast X to FP16 and apply two bias‑free linear projections (left/right)
    → L,R ∈ [B,N,N,H]   (H = hidden_dim).
3.  Optional binary mask M ∈ [B,N,N] is broadcast and applied to L,R.
4.  Compute three sigmoid gates from X and multiply:
        L ← L * sigmoid(left_gate(X))
        R ← R * sigmoid(right_gate(X))
        Gout ← sigmoid(out_gate(X))
5.  Core cubic operation (outgoing update):
        out_{b,i,j,h} = Σ_k L_{b,i,k,h} * R_{b,j,k,h}
   This is a batched matrix multiplication L @ Rᵀ for each hidden channel.
   We reshape (B,N,N,H) → (B·H, N, N) and run a custom Triton kernel that
   tiles the work with (BLOCK_M×BLOCK_N×BLOCK_K) = (256×256×64) (or a
   smaller variant for tiny N).  FP16 inputs are multiplied on Tensor‑Cores
   with FP32 accumulation.
6.  Reshape back to [B,N,N,H] (still FP16) and apply a second LayerNorm.
7.  Multiply by the output gate Gout, project back to the original dimension
    with a bias‑free linear layer and cast the result to FP32.

Only the N³ batched mat‑mul is performed in Triton; all other operations
use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bk, stride_bn,   # B strides: batch, K, N
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM:  C[b] = A[b] @ B[b]ᵀ   with FP16 × FP16 → FP32 accumulation."""
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (B·H)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch \
                 + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (already transposed): (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a_tile, b_tile)

    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    block_m: int,
    block_n: int,
    block_k: int,
) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A : (batch, M, K)  FP16
    B : (batch, K, N)  FP16   (already transposed, i.e. rightᵀ)
    Returns C : (batch, M, N) FP16
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb, "Inner dimensions must match"
    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=8,
        num_stages=4,
    )
    return C


def _choose_block_sizes(seq_len: int):
    """Select tile sizes that keep shared memory < 80 KB and provide good occupancy."""
    if seq_len <= 128:
        return 64, 64, 32
    if seq_len <= 256:
        return 128, 128, 32
    if seq_len <= 512:
        return 128, 128, 64
    # seq_len >= 1024
    return 256, 256, 64


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32 tensors)
            - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1) Unpack & basic config
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Layer‑norm on the input (FP32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 – the heavy work will be in FP16
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Optional binary mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is 0/1 – cast once to fp16 and broadcast
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 6) Gates (sigmoid) – all in FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # ------------------------------------------------------------------
    # 7) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8) Core cubic mat‑mul (outgoing update)
    #    left  : (B, N, N, H)  → (B·H, N, K) where K = N
    #    right : (B, N, N, H)  → (B·H, K, N) after transposition
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape: (B, H, N, N) → (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (M=N, K=N)
    # transpose the second & third axes to obtain (K=N, N) layout for the right operand
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (K=N, N)

    # Choose block sizes based on sequence length (aim for high occupancy)
    block_m, block_n, block_k = _choose_block_sizes(N)

    # Batched GEMM: left_mat @ right_matᵀ  → (B·H, N, N)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        block_m=block_m,
        block_n=block_n,
        block_k=block_k,
    )  # fp16

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # out_gate has shape [B, N, N, H] – already fp16
    out = out * out_gate

    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16, shape [B, N, N, D]

    # Convert to float32 for the API contract
    return out.to(torch.float32)
