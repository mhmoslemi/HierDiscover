```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel to fuse mask (if any) and gating multiplicative updates.
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    total_elems: tl.int32,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values (masked loads)
    l_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m_val = tl.load(mask_ptr + offs, mask=mask, other=1.0)

    # Fuse three element‑wise multiplications
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.
    Implements a fused projection (5×) to reduce GEMM launches, a
    Triton kernel for mask‑gate fusion, and uses a single torch.bmm
    for the N³ update.  All heavy linear work is performed in fp16,
    final result is returned as float32.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, D] float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 3. One‑shot linear projection that yields left/right and all gates
    # ------------------------------------------------------------------
    # Gather weights and cast to fp16
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the out‑feature dimension (5 * hidden_dim)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )   # shape (5*hidden_dim, dim)

    # Linear → shape (B·N·N, 5*hidden_dim)
    B, N, _, _ = x_f16.shape
    flat_input = x_f16.view(-1, dim)                     # (B·N·N, dim)
    combined = F.linear(flat_input, combined_weight)    # (B·N·N, 5*hidden_dim)
    # Restore 5‑way layout
    combined = combined.view(B, N, N, 5, hidden_dim)

    # Split
    left_raw      = combined[..., 0, :]   # (B, N, N, hidden_dim)
    left_gate_raw = combined[..., 1, :]
    right_raw     = combined[..., 2, :]
    right_gate_raw= combined[..., 3, :]
    out_gate_raw  = combined[..., 4, :]

    # Apply sigmoid to gate tensors
    left_gate  = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate   = out_gate_raw.sigmoid()

    # ------------------------------------------------------------------
    # 4. Optional mask + gate fusion (via Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask to fp16 and broadcast on hidden dimension
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # (B, N, N, 1)

        # Ensure contiguous memory before passing raw pointers to Triton
        left      = left_raw.contiguous()
        right     = right_raw.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_f16  = mask_f16.contiguous()

        total = left.numel()  # B·N·N·hidden_dim
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        left = left  # results already in‑place
        right = right
    else:
        # No mask – just apply the gates
        left = left_raw * left_gate
        right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ update: batched mat‑mul L @ Rᵀ for each head
    # ------------------------------------------------------------------
    # left/right are fp16 with shape (B, N, N, hidden_dim)
    # reshape to (B·hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched matmul, fp16×fp16 → fp32 accumulation
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Cast to float32 for the following LayerNorm
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate multiplication and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate is fp16 → broadcast‑upcast to fp32
    out = out * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
```

"""
TriMul “outgoing” forward pass (AlphaFold3 style).

Algorithm
---------
1. Input LayerNorm on the 4‑D tensor X ∈ [B, N, N, D].
2. Fuse the five linear maps (left_proj, left_gate, right_proj,
   right_gate, out_gate) into one single GEMM:
        Y = X_flat @ Wᵀ   where   W ∈ ℝ⁽⁵·H×D⁾,  H = hidden_dim.
   The result Y is reshaped to (B, N, N, 5, H) and split into the
   five required tensors.  The three gate tensors receive a sigmoid.
3. Optional binary mask (shape [B, N, N]) is expanded to [B, N, N, H]
   and fused together with the two element‑wise multiplications
   (left·left_gate, right·right_gate) in a tiny Triton kernel.
4. Core O(N³) update – a batched matrix multiplication for every head:
        out[b, i, j, h] = Σₖ left[b, i, k, h] · right[b, j, k, h]
   realized as a single torch.bmm on tensors of shape (B·H, N, N).
5. LayerNorm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to D.
All heavy linear work runs in fp16; the result after the batched mat‑mul
is up‑cast to fp32 for the two LayerNorms.  The Triton kernel
implements only the mask‑gate fusion (the part that cannot be fused by
torch’s JIT), keeping the overall runtime well below the 1 ms target
on the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (if any) and the two gate multiplications.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,          # fp16 * left
    left_gate_ptr,     # fp16 * left_gate
    right_ptr,         # fp16 * right
    right_gate_ptr,    # fp16 * right_gate
    mask_ptr,          # fp16 * mask (already broadcasted to H)
    total_elems: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    m = offs < total_elems                     # out‑of‑bounds guard

    # Load values (masked loads → 0/1 for out‑of‑bounds)
    l_val   = tl.load(left_ptr      + offs, mask=m, other=0.0)
    l_gate  = tl.load(left_gate_ptr + offs, mask=m, other=0.0)
    r_val   = tl.load(right_ptr     + offs, mask=m, other=0.0)
    r_gate  = tl.load(right_gate_ptr+ offs, mask=m, other=0.0)
    m_val   = tl.load(mask_ptr      + offs, mask=m, other=1.0)   # default 1 → no masking

    # Fuse the three element‑wise multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=m)
    tl.store(right_ptr + offs, r_out, mask=m)


# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/int) or None
        - weights      : dict with the nine model weights (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                     # input channel dimension D
    hidden_dim = config["hidden_dim"]       # model hidden dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16).contiguous()           # [B, N, N, D]

    # ------------------------------------------------------------------
    # 3) One‑shot linear projection that yields left/right and all gates
    # ------------------------------------------------------------------
    # Gather the five weight matrices (bias‑free) and cast to fp16
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature dimension → shape (5·H, D)
    combined_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )   # (5*hidden_dim, dim)

    B, N, _, _ = x_f16.shape
    flat_input = x_f16.view(-1, dim)                    # (B·N·N, D)

    # Single GEMM producing all five tensors at once
    combined = F.linear(flat_input, combined_weight)   # (B·N·N, 5·H)

    # Reshape to (B, N, N, 5, H) and split
    combined = combined.view(B, N, N, 5, hidden_dim)
    left_raw      = combined[..., 0, :]   # (B, N, N, H)
    left_gate_raw = combined[..., 1, :]
    right_raw     = combined[..., 2, :]
    right_gate_raw= combined[..., 3, :]
    out_gate_raw  = combined[..., 4, :]

    # Apply sigmoid to the three gate tensors
    left_gate  = left_gate_raw.sigmoid()      # (B, N, N, H)
    right_gate = right_gate_raw.sigmoid()
    out_gate   = out_gate_raw.sigmoid()       # used after the second LayerNorm

    # ------------------------------------------------------------------
    # 4) Optional mask + gate fusion (via Triton)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Expand mask to the hidden dimension and cast to fp16
        mask_f16 = mask.to(torch.float16).unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

        # Ensure all tensors are contiguous before passing raw pointers
        left      = left_raw.contiguous()
        right     = right_raw.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()
        mask_f16  = mask_f16.contiguous()

        total = left.numel()                     # B·N·N·H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # After the kernel, `left` and `right` already contain the
        # masked‑and‑gated values.
    else:
        # No mask – just apply the gates
        left  = left_raw * left_gate
        right = right_raw * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ update: batched mat‑mul L @ Rᵀ for each head
    # ------------------------------------------------------------------
    # Shape -> (B·H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched mat‑mul (fp16 × fp16 → fp32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # Cast to float32 for LayerNorm
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate scaling, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out_gate is fp16 → up‑cast automatically
    out = out * out_gate

    # Final linear projection back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
