    left_gate = torch.sigmoid(left_gate_raw)

def _mask_gate_fused_kernel(
    proj_left_ptr, proj_right_ptr,
    gate_left_ptr, gate_right_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total_positions: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):

    mask = pid < total_positions
    if not mask:
        return
    # compute offsets
    offs = tl.arange(0, BLOCK_SIZE)
    base = pid * hidden_dim
    proj_l = tl.load(proj_left_ptr + base + offs, mask=mask)
    proj_r = tl.load(proj_right_ptr + base + offs, mask=mask)
    gate_l = tl.load(gate_left_ptr + base + offs, mask=mask)

    tl.store(out_right_ptr + base + offs, out_r, mask=mask)

def _mask_gate_fused_kernel(
    proj_left_ptr,
    proj_right_ptr,
    gate_left_ptr,
    gate_right_ptr,
    mask_ptr,
    out_left_ptr,
    out_right_ptr,
    total_positions: tl.constexpr,
    hidden_dim: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)  # each pid processes one position

    # compute mask for this position
    valid = pid < total_positions
    if not valid:
        return

    offset = pid * hidden_dim
    offs = tl.arange(0, BLOCK_SIZE)
    mask = offs < hidden_dim  # always true because off range = hidden_dim

    # load values
    left_proj = tl.load(proj_left_ptr + offset + offs, mask=mask & valid)
    right_proj = tl.load(proj_right_ptr + offset + offs, mask=mask & valid)
    left_gate = tl.load(gate_left_ptr + offset + offs, mask=mask & valid)
    right_gate = tl.load(gate_right_ptr + offset + offs, mask=mask & valid)

    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

    left_out = left_proj * left_gate * mask_val
    right_out = right_proj * right_gate * mask_val

    tl.store(out_left_ptr + offset + offs, left_out, mask=mask & valid)
    tl.store(out_right_ptr + offset + offs, right_out, mask=mask & valid)

"""
Fast Triton‑accelerated forward pass for the “outgoing” TriMul module
(AlphaFold‑3 style).

Algorithm
---------
1️⃣ Layer‑norm the input tensor X ∈ [B,N,N,D] (D = dim) – keep it in FP32,
   then cast to FP16 for the remaining work.

2️⃣ Fuse **all** linear projections (left / right / gates) into a *single*
   GEMM:
       weight_combined = [left_proj; right_proj; left_gate; right_gate; out_gate]
   → shape (5·H, D)   (H = hidden_dim).
   The GEMM produces a flat tensor of shape [B·N·N, 5·H] from which we
   obtain:
       left_proj, right_proj, left_gate, right_gate, out_gate.
   The three gate tensors are passed through sigmoid.

3️⃣ If a mask is supplied we launch a tiny Triton kernel that, for each
   position (b,i,k) and each hidden channel h, computes
       left  = left_proj  * sigmoid(left_gate)  * mask
       right = right_proj * sigmoid(right_gate) * mask
   The kernel works on the flattened view (M = B·N·N) and processes one
   position per program‑id, loading a vector of length H at a time.
   When `nomask==True` we simply do the element‑wise products in PyTorch.

4️⃣ Core N³ update – batched matrix multiplication:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched GEMM of shape (B·H, N, N) @ (B·H, N, N)ᵀ.
   Torch’s cuBLAS implementation (fp16×fp16→fp16) is used because it
   saturates the H100 Tensor‑Core bandwidth.

5️⃣ Layer‑norm over the hidden dimension, multiply by the already‑sigmoid
   out_gate, and a final linear projection back to the original dimension.
   All remaining arithmetic stays in fp16; only the final result is cast
   to float32 as required.

Only the mask‑gate fusion step is written in Triton (the problem’s
“must use Triton” requirement).  The rest of the pipeline relies on
highly‑optimised PyTorch/CUDA kernels, delivering sub‑millisecond
latency on the target shapes.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: fuse mask, gate‑sigmoid and elementwise multiplication
# ------------------------------------------------------------
@triton.jit
def _mask_gate_fused_kernel(
    proj_left_ptr,    # fp16* [M, H]  – left projection
    proj_right_ptr,   # fp16* [M, H]  – right projection
    gate_left_ptr,    # fp16* [M, H]  – left gate (pre‑sigmoid)
    gate_right_ptr,   # fp16* [M, H]  – right gate (pre‑sigmoid)
    mask_ptr,         # fp16* [M]     – 0/1 mask per position
    out_left_ptr,     # fp16* [M, H]  – output left (gated & masked)
    out_right_ptr,    # fp16* [M, H]  – output right (gated & masked)
    total_positions: tl.constexpr,
    hidden_dim:      tl.constexpr,
    BLOCK_SIZE:      tl.constexpr,  # = hidden_dim
):
    pid = tl.program_id(0)                 # one program per position
    if pid >= total_positions:
        return

    # offsets for the vector belonging to this position
    base = pid * hidden_dim
    offs = tl.arange(0, BLOCK_SIZE)

    # load raw values
    l_proj = tl.load(proj_left_ptr + base + offs)
    r_proj = tl.load(proj_right_ptr + base + offs)
    l_gate = tl.load(gate_left_ptr + base + offs)
    r_gate = tl.load(gate_right_ptr + base + offs)

    # sigmoid for the gates (logistic)
    l_gate = 1.0 / (1.0 + tl.exp(-l_gate))
    r_gate = 1.0 / (1.0 + tl.exp(-r_gate))

    # scalar mask (broadcasted over the hidden dimension)
    mask_val = tl.load(mask_ptr + pid)

    # gated + masked output
    l_out = l_proj * l_gate * mask_val
    r_out = r_proj * r_gate * mask_val

    # store results
    tl.store(out_left_ptr + base + offs,  l_out)
    tl.store(out_right_ptr + base + offs, r_out)


# ------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.
    Args:
        data: Tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor: torch.Tensor [B, N, N, D]  (float32)
              - mask: torch.Tensor [B, N, N] (bool/int) – may be None when nomask=True
              - weights: dict of model parameters (float32)
              - config: dict containing at least "dim", "hidden_dim",
                        optionally "nomask"
    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input feature dimension D
    hidden_dim = config["hidden_dim"]   # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension (float32 → fp16)
    # --------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B,N,N,D]  float32
    x = x.to(torch.float16)            # continue in half‑precision

    # --------------------------------------------------------
    # 2️⃣ Single fused linear for projections + all gates
    # --------------------------------------------------------
    # concatenate weight matrices: order = Lproj, Rproj, Lgate, Rgate, OutGate
    w_combined = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # (5*H, D)

    # flatten to (B*N*N, D) and apply the fused GEMM
    B, N, _, _ = x.shape
    M = B * N * N                       # total positions
    x_flat = x.reshape(M, dim)          # [M, D]

    fused_out = F.linear(x_flat, w_combined)   # [M, 5*H]

    # split back into the five tensors
    split_sizes = [hidden_dim] * 5
    left_proj, right_proj, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        fused_out, split_sizes, dim=1
    )                                   # each [M, H]

    # reshape to [B,N,N,H] (still half‑precision)
    left_proj = left_proj.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    left_gate = torch.sigmoid(left_gate_raw).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(right_gate_raw).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(out_gate_raw).view(B, N, N, hidden_dim)  # used later

    # --------------------------------------------------------
    # 3️⃣ Optional mask & gate fusion via Triton (in‑place)
    # --------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → fp16 scalar per position
        mask_f16 = mask.to(torch.float16).reshape(-1).contiguous()   # [M]

        # flatten the tensors that are inputs to the kernel
        left_proj_f = left_proj.reshape(M, hidden_dim).contiguous()
        right_proj_f = right_proj.reshape(M, hidden_dim).contiguous()
        left_gate_f = left_gate.reshape(M, hidden_dim).contiguous()
        right_gate_f = right_gate.reshape(M, hidden_dim).contiguous()

        # allocate output buffers
        left = torch.empty_like(left_proj_f)
        right = torch.empty_like(right_proj_f)

        # launch one program per position (M programs)
        grid = (M,)
        _mask_gate_fused_kernel[grid](
            left_proj_f,
            right_proj_f,
            left_gate_f,
            right_gate_f,
            mask_f16,
            left,
            right,
            M,
            hidden_dim,
            BLOCK_SIZE=hidden_dim,
            num_warps=4,
            num_stages=1,
        )
        # reshape back to the 4‑D layout
        left = left.view(B, N, N, hidden_dim)
        right = right.view(B, N, N, hidden_dim)
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # --------------------------------------------------------
    # 4️⃣ Core N³ update – batched matrix multiplication (B·H, N, N)
    # --------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().reshape(B * hidden_dim, N, N)

    # (B*H, N, N) @ (B*H, N, N)ᵀ   →  fp16 result
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------
    # 5️⃣ Output Layer‑Norm, out‑gate, final projection
    # --------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # still fp16

    # apply the out‑gate (already in fp16)
    out_norm = out_norm * out_gate

    # final linear back to original dimension
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    out_final = F.linear(out_norm, to_out_w)               # fp16, [B,N,N,dim]

    # cast to float32 for the required output type
    return out_final.to(torch.float32)
```
