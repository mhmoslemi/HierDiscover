"""
TriMul (outgoing) – Fast Triton‑accelerated forward pass
=======================================================

Algorithm
---------
1️⃣ Layer‑norm on the input (float32 → float16).
2️⃣ Fuse the two linear projections (left/right) into one GEMM.
   Fuse the three gate linear layers (left‑gate, right‑gate, out‑gate)
   into one GEMM and split the result.
3️⃣ Apply sigmoid to the gates.
4️⃣ Fuse optional mask, left‑gate and right‑gate into a tiny Triton kernel
   (in‑place on `left` and `right`).  If no mask, use a simple in‑place
   element‑wise mul.
5️⃣ Reshape & permute `left`/`right` so that the cubic contraction becomes a
   batched matrix multiplication:
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   This is executed with a single `torch.bmm` on FP16 tensors.
6️⃣ Second LayerNorm over the hidden dimension (float32), apply the output
   gate and a final linear projection back to `dim` (float32).

Only the mask‑gate fusion is written in Triton; the remaining heavy
operations use the highly‑optimised cuBLAS kernels exposed via PyTorch.
Combined GEMMs (projection + gating) cut launch overhead, and a large
`BLOCK_SIZE` in the Triton kernel reduces the number of program launches.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and the left/right gates into the
# projected tensors.  Operates on flattened contiguous tensors.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # fp16 * (B·N·N·H)
    right_ptr,         # fp16 * (B·N·N·H)
    left_gate_ptr,     # fp16 * (B·N·N·H)
    right_gate_ptr,    # fp16 * (B·N·N·H)
    mask_ptr,          # fp16 * (B·N·N)  (broadcasted over H)
    total_elements,    # i64  (B·N·N·H)
    hidden_dim,        # i32  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # Load tensors (fp16)
    l = tl.load(left_ptr + offs,        mask=in_range, other=0.0)
    r = tl.load(right_ptr + offs,       mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)

    # Optional mask (broadcasted over the hidden dimension)
    if HAS_MASK:
        mask_idx = offs // hidden_dim          # each H‑wide chunk shares a mask element
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # Fuse: left = left * left_gate * mask
    #       right = right * right_gate * mask
    l = l * lg * m
    r = r * rg * m

    # Store results back in‑place
    tl.store(left_ptr + offs,  l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H in the paper

    # ------------------------------------------------------------------
    # 2️⃣ load model weights (cast to fp16 where possible)
    # ------------------------------------------------------------------
    # LayerNorm 1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (no bias) – fp16
    left_proj_w  = weights["left_proj.weight"].half()   # (H, dim)
    right_proj_w = weights["right_proj.weight"].half()  # (H, dim)

    # Gate linear layers – fp16
    left_gate_w  = weights["left_gate.weight"].half()   # (H, dim)
    right_gate_w = weights["right_gate.weight"].half()  # (H, dim)
    out_gate_w   = weights["out_gate.weight"].half()    # (H, dim)

    # Output LayerNorm & final projection – stay in fp32
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]           # (dim, H)

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                             # [B, N, N, dim]  fp32
    x_h = x.half()                               # [B, N, N, dim]  fp16

    B, N, _, _ = x_h.shape
    total_spatial = B * N * N                     # = B·N·N

    # ------------------------------------------------------------------
    # 4️⃣ Fuse left/right projections into a single GEMM
    # ------------------------------------------------------------------
    # Flatten input for linear layers: (B·N·N, dim)
    x_flat = x_h.view(-1, dim)                   # (total_spatial, dim)

    # Concatenate left/right projection weights → (2*H, dim)
    proj_weights = torch.cat([left_proj_w, right_proj_w], dim=0)  # fp16
    proj = F.linear(x_flat, proj_weights)       # (total_spatial, 2*H)

    # Split into left and right tensors
    left  = proj[:, :hidden_dim]                # (total_spatial, H)
    right = proj[:, hidden_dim:]                # (total_spatial, H)

    # ------------------------------------------------------------------
    # 5️⃣ Fuse the three gate linear layers into a single GEMM
    # ------------------------------------------------------------------
    gate_weights = torch.cat([left_gate_w, right_gate_w, out_gate_w], dim=0)  # (3*H, dim)
    gates = F.linear(x_flat, gate_weights)      # (total_spatial, 3*H)

    # Split gates and apply sigmoid
    left_gate  = torch.sigmoid(gates[:, :hidden_dim])
    right_gate = torch.sigmoid(gates[:, hidden_dim:2*hidden_dim])
    out_gate   = torch.sigmoid(gates[:, 2*hidden_dim:])   # (total_spatial, H)

    # ------------------------------------------------------------------
    # 6️⃣ Fuse optional mask with left/right gates (tiny Triton kernel)
    # ------------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        # mask → fp16, shape (B, N, N), broadcast over H
        mask_h = mask.to(dtype=left.dtype).contiguous()      # fp16
        mask_flat = mask_h.view(-1)                          # (total_spatial,)

        # Flatten everything for the kernel
        left_flat       = left.contiguous().view(-1)         # (total_spatial*H,)
        right_flat      = right.contiguous().view(-1)
        left_gate_flat  = left_gate.contiguous().view(-1)
        right_gate_flat = right_gate.contiguous().view(-1)

        total_elements = left_flat.numel()   # B·N·N·H

        # Choose a large BLOCK_SIZE to minimise launch overhead
        BLOCK_SIZE = 65536
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _apply_mask_gate_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=HAS_MASK,
        )
        # Reshape back to (total_spatial, H)
        left  = left_flat.view(total_spatial, hidden_dim)
        right = right_flat.view(total_spatial, hidden_dim)
        # Gates no longer needed
        del left_gate, right_gate
    else:
        # No mask → simple in‑place element‑wise multiplication
        left  = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare tensors for the cubic contraction (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to 4‑D tensors: (B, N, N, H)
    left  = left .view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # Permute so that the two N dimensions become (i,k) and (k,j)
    # left_perm:  (B, H, N, N)   indexes (i,k)
    # right_perm: (B, H, N, N)   indexes (k,j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Collapse the batch & hidden dimension for torch.bmm
    BH, N1, N2 = left_perm.shape[0] * left_perm.shape[1], left_perm.shape[2], left_perm.shape[3]
    left_mat  = left_perm.view(BH, N1, N2)   # (B*H, N, N)
    right_mat = right_perm.view(BH, N1, N2)  # (B*H, N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (Tensor‑core fast)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)  fp16

    # Restore original layout → (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = out.float()  # up‑cast for numerically stable mean/var
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                          # (B, N, N, H)  float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim).float()
    out = out * out_gate                       # gating (float32)

    out = F.linear(out, to_out_w)              # final projection → (B, N, N, dim) float32

    return out
