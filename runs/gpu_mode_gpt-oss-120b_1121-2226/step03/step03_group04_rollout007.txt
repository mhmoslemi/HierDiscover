"""
TriMul (outgoing) – high‑performance Triton‑accelerated forward pass
===================================================================

Algorithm (forward):
  1️⃣ Layer‑norm over the input feature (“dim”) – FP32.
  2️⃣ Cast to FP16 and perform **one** batched GEMM that simultaneously
     computes the two projections (left/right) and the three gate
     linear layers (left‑gate, right‑gate, out‑gate).  This replaces five
     separate `F.linear` calls and cuts the memory traffic for the large
     (B·N·N, dim) matrix by ~5×.
  3️⃣ Apply sigmoid to the three gate tensors.
  4️⃣ Fuse optional mask, left‑gate and right‑gate into the projected
     tensors with a tiny Triton kernel (in‑place, no extra temporaries).
  5️⃣ Rearrange the tensors so the cubic contraction becomes a **batched
     GEMM**:   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
     Implemented with a single `torch.bmm` on (B·H, N, N) matrices,
     which on an H100 runs on Tensor‑cores.
  6️⃣ Second Layer‑norm over the hidden dimension (still FP32).
  7️⃣ Multiply by the output‑gate, project back to `dim` with a final
     FP32 linear layer, and return the result (FP32).

Only the element‑wise mask‑gate fusion is written in Triton; everything
else uses highly‑optimized PyTorch (FP16) primitives.  The heavy N³ work
is fully delegated to cuBLAS/Tensor‑core GEMMs, while the linear‑layer
fusion cuts the dominant FP16 matrix‑multiply cost by ~5×.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # fp16 * (B·N·N·H)
    right_ptr,         # fp16 * (B·N·N·H)
    left_gate_ptr,     # fp16 * (B·N·N·H)
    right_gate_ptr,    # fp16 * (B·N·N·H)
    mask_ptr,          # fp16 * (B·N·N)  (broadcasted over H)
    total_elements,    # i64  B·N·N·H
    hidden_dim,        # i64  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    within = offs < total_elements

    # Load data (fp16)
    l  = tl.load(left_ptr      + offs, mask=within, other=0.0)
    r  = tl.load(right_ptr     + offs, mask=within, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=within, other=0.0)
    rg = tl.load(right_gate_ptr+ offs, mask=within, other=0.0)

    # Load mask (broadcasted over hidden_dim)
    if HAS_MASK:
        mask_idx = offs // hidden_dim          # each H consecutive elements share mask
        m = tl.load(mask_ptr + mask_idx, mask=within, other=1.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # Fuse:   left = left * left_gate * mask
    #         right = right * right_gate * mask
    l = l * lg * m
    r = r * rg * m

    # Store back in‑place
    tl.store(left_ptr  + offs, l, mask=within)
    tl.store(right_ptr + offs, r, mask=within)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]               # H in the paper

    # --------------------------------------------------------------
    # 2️⃣ load weights (cast to half where possible)
    # --------------------------------------------------------------
    # LayerNorm 1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (no bias) – cast to FP16.
    # We fuse all five linear layers into one large matmul.
    left_proj_w   = weights["left_proj.weight"]      # (H, dim)
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension (first dim)
    combined_weight = torch.cat([
        left_proj_w,   # H x dim
        right_proj_w,  # H x dim
        left_gate_w,   # H x dim
        right_gate_w,  # H x dim
        out_gate_w    # H x dim
    ], dim=0).half()                                 # (5*H, dim)

    # Output‑norm & final projection (remain FP32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]        # (dim, H)

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (FP32) → FP16
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                                # [B, N, N, dim]  float32
    x_h = x.half()                                   # half for the heavy math

    # --------------------------------------------------------------
    # 4️⃣ Fused linear projection (one GEMM → 5× hidden tensors)
    # --------------------------------------------------------------
    B, N, _, _ = x_h.shape
    x_flat = x_h.view(-1, dim)                       # (B·N·N, dim)
    combined = torch.nn.functional.linear(x_flat, combined_weight)  # (B·N·N, 5*H)
    # reshape back
    combined = combined.view(B, N, N, 5 * hidden_dim)   # [B, N, N, 5*H]

    # split
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        combined, hidden_dim, dim=-1
    )                                                # each [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Gates (sigmoid) – still FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)             # will be used later

    # --------------------------------------------------------------
    # 6️⃣ Fuse optional mask + left/right gates (tiny Triton kernel)
    # --------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None) and mask.numel() > 0 else 0
    if HAS_MASK:
        # mask → half, broadcast over H
        mask_h = mask.to(dtype=left_proj.dtype).contiguous()   # [B, N, N] fp16
        mask_flat = mask_h.view(-1)                             # (B·N·N)

        # flatten tensors for the kernel
        left_flat       = left_proj.view(-1)
        right_flat      = right_proj.view(-1)
        left_gate_flat  = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        total_elements = left_flat.numel()                 # B·N·N·H
        BLOCK_SIZE = 128                                   # = hidden_dim (fits nicely)
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _apply_mask_gate_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=HAS_MASK,
        )
        # `left_proj` and `right_proj` have been updated in‑place.
        # Gates can be freed now.
        del left_gate, right_gate
    else:
        # No mask → pure element‑wise fuse in PyTorch
        left_proj  = left_proj * left_gate
        right_proj = right_proj * right_gate
        del left_gate, right_gate

    # --------------------------------------------------------------
    # 7️⃣ Prepare for the cubic contraction (batched GEMM)
    # --------------------------------------------------------------
    # left : (B, N, N, H) → (B, H, N, N)   axes (i,k)
    # right: (B, N, N, H) → (B, H, N, N)   axes (j,k) -> need (k,j) order
    left_perm  = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_proj.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape

    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N) fp16
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N) fp16

    # --------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (Tensor‑core fast)
    # --------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # (B·H, N, N) fp16

    # Restore original layout: [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # --------------------------------------------------------------
    out = out.float()  # up‑cast for stable LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                               # [B, N, N, H] float32

    # --------------------------------------------------------------
    # 🔟 Apply output gate (float32) and final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate.float()                     # element‑wise gate
    out = F.linear(out, to_out_w)                    # [B, N, N, dim] float32

    return out
