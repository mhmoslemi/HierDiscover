def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # definitions
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16 for projector
    x_fp16 = x.to(torch.float16)

    # Combine weights
    combined_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)

    proj = F.linear(x_fp16, combined_weight)   # shape (B,N,N,5*H)

    # Split
    H = hidden_dim
    left = proj[..., :H]
    right = proj[..., H:2*H]
    left_gate = proj[..., 2*H:3*H]
    right_gate = proj[..., 3*H:4*H]
    out_gate = proj[..., 4*H:5*H]

    left_gate = left_gate.sigmoid_()
    right_gate = right_gate.sigmoid_()
    out_gate = out_gate.sigmoid_()

            num_warps=4

        left.mul_(left_gate)
        right.mul_(right_gate)

    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B*H2, N, N)

    )

-----------------------

Triton‑accelerated forward pass of the “outgoing” TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑Norm the 4‑D input (float32) → x.

   in a single pass (fp16 I/O, fp32 math for the sigmoid and mask
   multiplication).
   When there is no mask we simply multiply left/right with their sigmoids.
5. Core N³ update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication L @ Rᵀ for each head h.
   We reshape to (B·H, N, N) and call ``torch.bmm`` – cuBLAS/Tensor‑Core
   handles the heavy work efficiently.
6. Cast the result back to float32, apply a second LayerNorm over the hidden
   dimension, multiply by ``out_gate`` (broadcasted) and finally project back
   to the original ``dim`` with a linear layer.
7. Return the tensor of shape ``[B, N, N, dim]`` (float32).

Only the mask‑gate‑sigmoid fusion is written in Triton; all large GEMM /
mat‑mul work stays in PyTorch (cuBLAS) which is already optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid (gate) and element‑wise multiplication
#   left  = left  * sigmoid(left_gate)  * mask
#   right = right * sigmoid(right_gate) * mask
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,            # fp16 * (B·N·N·H)
    left_gate_ptr,       # fp16 * (B·N·N·H)   (pre‑sigmoid)
    right_ptr,           # fp16 * (B·N·N·H)
    right_gate_ptr,      # fp16 * (B·N·N·H)   (pre‑sigmoid)
    mask_ptr,            # fp16 * (B·N·N)   (broadcasted)
    total_elems: tl.int32,   # B·N·N·H
    hidden_dim: tl.int32,    # H  (used for broadcasting mask)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # bounds check
    mask = offs < total_elems

    # index for the mask (mask is one value per (b,i,j) → stride H)
    mask_idx = offs // hidden_dim

    # load values (fp16)
    l_val   = tl.load(left_ptr       + offs, mask=mask, other=0.0)
    lg_val  = tl.load(left_gate_ptr  + offs, mask=mask, other=0.0)
    r_val   = tl.load(right_ptr      + offs, mask=mask, other=0.0)
    rg_val  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m_val   = tl.load(mask_ptr       + mask_idx, mask=mask, other=1.0)   # 1 → no masking

    # compute sigmoid in fp32 for numerical stability
    lg_sig = 1.0 / (1.0 + tl.exp(-tl.float32(lg_val)))
    rg_sig = 1.0 / (1.0 + tl.exp(-tl.float32(rg_val)))

    # apply mask and write back as fp16
    l_out = tl.float16(tl.float32(l_val) * lg_sig * tl.float32(m_val))
    r_out = tl.float16(tl.float32(r_val) * rg_sig * tl.float32(m_val))

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask         : torch.Tensor, shape [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict containing keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (float32, aligns with reference implementation)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]   float32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for all heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # concatenate the weight matrices (bias‑free) along the output dimension
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, D)
            weights["right_proj.weight"],  # (H, D)
            weights["left_gate.weight"],   # (H, D)
            weights["right_gate.weight"],  # (H, D)
            weights["out_gate.weight"],    # (H, D)
        ],
        dim=0,
    ).to(torch.float16)                     # (5*H, D)

    # one big linear → shape [B, N, N, 5*H] (fp16)
    proj = F.linear(x_f16, combined_weight)   # bias=None

    # ------------------------------------------------------------------
    # 5️⃣ Partition the concatenated tensor
    # ------------------------------------------------------------------
    H = hidden_dim
    left_raw   = proj[..., :H]               # [B, N, N, H]
    right_raw  = proj[..., H:2*H]            # [B, N, N, H]
    left_gate_raw  = proj[..., 2*H:3*H]      # pre‑sigmoid
    right_gate_raw = proj[..., 3*H:4*H]      # pre‑sigmoid
    out_gate  = proj[..., 4*H:5*H].sigmoid_()   # sigmoid → fp16, used later

    # ------------------------------------------------------------------
    # 6️⃣ Apply mask & sigmoid to left/right in a fused Triton kernel
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask → fp16, shape [B, N, N, 1] (broadcastable)
        mask_f16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        total = left_raw.numel()                 # B·N·N·H
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        _gate_mask_fuse_kernel[grid](
            left_raw.data_ptr(),
            left_gate_raw.data_ptr(),
            right_raw.data_ptr(),
            right_gate_raw.data_ptr(),
            mask_f16.data_ptr(),
            total,
            H,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → just apply the sigmoids (in‑place, fp16)
        left_raw.mul_(left_gate_raw.sigmoid_())
        right_raw.mul_(right_gate_raw.sigmoid_())

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ update: batched mat‑mul L @ Rᵀ for each hidden channel
    # ------------------------------------------------------------------
    B, N, _, _ = left_raw.shape   # (B, N, N, H)

    # reshape to (B·H, N, N) – both tensors stay fp16
    left_mat  = left_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_raw.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched mat‑mul (fp16×fp16 → fp32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out_mat = out_mat.to(torch.float32)   # cast for the following LayerNorm

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # out_gate auto‑casts to fp32
    out = F.linear(out, weights["to_out.weight"])   # (D, H) → final dim

    return out
```
