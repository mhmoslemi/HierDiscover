def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # LayerNorm
    x = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Convert to fp16 for following heavy ops
    x_f16 = x.to(torch.float16)

    # Combine projection weights
    # Ensure all weights are on same device and dtype
    left_proj_w = weights['left_proj.weight']
    left_gate_w = weights['left_gate.weight']
    right_proj_w = weights['right_proj.weight']
    right_gate_w = weights['right_gate.weight']
    out_gate_w = weights['out_gate.weight']
    # shape D x H each
    combined_w = torch.cat([left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w], dim=1)  # D x (5*H)
    combined_w = combined_w.to(torch.float16)

    # Linear combine
    combined = F.linear(x_f16, combined_w)  # shape [B, N, N, 5*H]

    # Slice
    H = hidden_dim
    left = combined[..., :H]  # B,N,N,H
    right = combined[..., H:2*H]
    left_gate = combined[..., 2*H:3*H]
    right_gate = combined[..., 3*H:4*H]
    out_gate = combined[..., 4*H:]

    # Apply sigmoid to gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Mask
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # B,N,N,1
        left = left * m
        right = right * m

    # Apply gates to left/right
    left = left * left_gate
    right = right * right_gate

    # Core matmul - we need to compute left @ right.T for each hidden channel

    # Permute to (B, H, N, N) then reshape for kernel (B*H,N,N)
    left_mat = left.permute(0,3,1,2).contiguous().view(bs * H, seq_len, seq_len)
    right_mat = right.permute(0,3,1,2).contiguous().view(bs * H, seq_len, seq_len)

    # call batched kernel
    out_mat = _batched_matmul_triton(left_mat, right_mat,
                                     BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    out = out_mat.view(bs, H, seq_len, seq_len).permute(0,2,3,1).contiguous()  # B,N,N,H

    # Output layernorm (hid)
    to_out_norm_w = weights['to_out_norm.weight'].to(torch.float16)
    to_out_norm_b = weights['to_out_norm.bias'].to(torch.float16)
    out = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # Multiply out_gate (broadcast)
    out = out * out_gate

    # Final linear projection back to dim
    to_out_w = weights['to_out.weight'].to(torch.float16)
    out = F.linear(out, to_out_w)

    # Cast back to fp32
    out = out.to(torch.float32)
    return out

"""
TriMul “outgoing” forward pass – Triton‑accelerated implementation.

Algorithm
---------
1. Input LayerNorm (FP32) → x  [B, N, N, D].
2. Cast x to FP16 for the heavy work.
3. Fuse the five bias‑free linear projections (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single matmul:
        Y = x @ W   (W has shape D×(5·H) ).
   Slice Y to obtain
        left, right, left_gate, right_gate, out_gate  each [B,N,N,H].
4. Apply optional binary mask to left/right, then element‑wise
   sigmoid gates and multiplication:
        left  = left  * mask * sigmoid(left_gate)
        right = right * mask * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)
5. Core cubic update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   i.e. batch‑wise matrix multiply  left @ rightᵀ per hidden channel.
   We reshape (B,N,N,H) → (B·H, N, N) and run a custom Triton kernel
   with tile sizes (BLOCK_M,N) = (128,128) and BLOCK_K = 32.
   The kernel uses FP16 inputs, FP32 accumulation and writes FP16.
6. Reshape the result back to [B,N,N,H] (still FP16).
7. Output LayerNorm over the hidden dimension, multiply by out_gate
   and project back to the original dimension with a bias‑free linear
   layer.
8. Cast the final tensor to FP32 (required by the API) and return.

The heavy computation (steps 3‑5) is executed in fp16/Tensor‑Core
fashion, while lightweight LayerNorm / sigmoid steps use PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, row, col
    stride_bh, stride_bn, stride_bk,    # B strides: batch, row, col
    stride_ch, stride_cm, stride_cn,    # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    #  Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (combined B·H)

    # ------------------------------------------------------------------
    #  Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # ------------------------------------------------------------------
    #  Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load tile of A : (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # Load tile of Bᵀ : (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Dot product (FP32 accumulation)
        acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    #  Store result (cast to FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 32,
) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ using the Triton kernel above.
    A, B must be FP16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) as FP16.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects FP16 inputs"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C

    """

    """

    # 1) Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data

    dim = config["dim"]                # D

    # 2) Input LayerNorm (FP32 for numerical stability)
    # --------------------------------------------------------------

        input_tensor,
        normalized_shape=(dim,),

    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_fp16 = x.to(torch.float16)  # [B, N, N, D]  (fp16)

    # --------------------------------------------------------------
    # 4) Fuse the five bias‑free linear projections into one matmul
    # --------------------------------------------------------------
    # weight shapes: (hidden_dim, dim)
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Concatenate along the output‑feature dimension (dim=0)
    combined_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*H, D)

    # Single fused linear: result shape [B, N, N, 5*H]
    combined = F.linear(x_fp16, combined_w)

    H = hidden_dim
    left        = combined[..., :H]                 # [B, N, N, H]
    right       = combined[..., H:2 * H]            # [B, N, N, H]
    left_gate   = combined[..., 2 * H:3 * H]        # [B, N, N, H]
    right_gate  = combined[..., 3 * H:4 * H]        # [B, N, N, H]
    out_gate    = combined[..., 4 * H:]             # [B, N, N, H]

    # --------------------------------------------------------------
    # 5) Gates (sigmoid) – still in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # --------------------------------------------------------------
    # 6) Optional mask (binary)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 7) Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8) Core N³ batched mat‑mul (outgoing update)
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # batch, seq_len, _, hidden_dim

    # Reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Perform the fused batched mat‑mul in FP16 (accumulated in FP32)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )  # [B·H, N, N] (fp16)

    # Reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp16

    # --------------------------------------------------------------
    # 9) Output LayerNorm (over hidden dim), out‑gate, final projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    out = out * out_gate  # broadcasted multiplication, fp16

    # Final projection back to original dimension D
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16

    # Cast to FP32 for the final output as required by the API
    out = out.to(torch.float32)

    return out
```
