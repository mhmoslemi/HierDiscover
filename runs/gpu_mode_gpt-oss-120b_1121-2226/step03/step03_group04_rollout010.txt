    out_ptr,         # fp16*
    out_gate_ptr,    # fp16*
    out_res_ptr,     # fp32*

    # positions indices
    pos = pos_base + tl.arange(0, BLOCK_POS)
    mask_pos = pos < total_pos

    # compute sum and sum_sq
    sum_ = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    for d in range(0, hidden_dim):
        # offset for each position
        offset = pos * hidden_dim + d
        # load out and cast
        val = tl.load(out_ptr + offset, mask=mask_pos, other=0.0)
        val_f = val.to(tl.float32)

@triton.jit
def _ln_gate_kernel(
    out_ptr,                # * fp16
    out_gate_ptr,           # * fp16
    out_res_ptr,            # * fp32
    gamma_ptr,              # * fp32
    beta_ptr,               # * fp32
    total_positions,        # int
    HIDDEN_DIM: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_POS: tl.constexpr,
):
    # each program_id processes BLOCK_POS positions.
    pid = tl.program_id(0)
    pos = pid * BLOCK_POS + tl.arange(0, BLOCK_POS)
    mask = pos < total_positions

    # Compute sum and sumsq
    sum_ = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    for d in range(HIDDEN_DIM):
        # linear offset
        offset = pos * HIDDEN_DIM + d
        val = tl.load(out_ptr + offset, mask=mask, other=0.0)
        val_f = val.to(tl.float32)
        sum_ += val_f
        sum_sq += val_f * val_f

    mean = sum_ * (1.0 / HIDDEN_DIM)
    var = sum_sq * (1.0 / HIDDEN_DIM) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # second loop: compute normalized, apply gamma/beta, multiply gate, write out.
    for d in range(HIDDEN_DIM):
        offset = pos * HIDDEN_DIM + d
        val = tl.load(out_ptr + offset, mask=mask, other=0.0)
        gate = tl.load(out_gate_ptr + offset, mask=mask, other=0.0)
        val_f = val.to(tl.float32)
        gate_f = gate.to(tl.float32)
        gamma = tl.load(gamma_ptr + d)
        beta = tl.load(beta_ptr + d)

        norm = (val_f - mean) * inv_std
        out_val = norm * gamma + beta
        out_val = out_val * gate_f
        tl.store(out_res_ptr + offset, out_val, mask=mask)
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣  Layer‑norm on the input (float32).
2️⃣  Cast to FP16 and compute three linear projections:
        * left  (dim  → hidden)
        * right (dim  → hidden)
        * out_gate (dim → hidden)  – followed by a sigmoid
   Two additional linear layers (left_gate, right_gate) are also computed
   and sigmoid‑ed.
3️⃣  Fuse *mask* (if present) together with the left/right gates in a tiny
   Triton kernel – everything stays in FP16.
4️⃣  Perform the cubic contraction as a batch of matrix multiplications:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is realized as a batched GEMM on tensors of shape
   (B·H, N, N) @ (B·H, N, N) using tensor‑cores (FP16 → FP32 accumulation).
5️⃣  Fuse the second Layer‑Norm (over the hidden dimension) and the
   out‑gate multiplication into a single Triton kernel.  The kernel
   computes per‑position mean/variance, normalises, applies the learned
   scale / bias and finally multiplies by the out‑gate.  The result is a
   FP32 tensor `out_ln` of shape [B, N, N, hidden].
6️⃣  Project back to the model dimension with a regular `torch.nn.functional.linear`
   (FP32 × FP32 → FP32).  This is a single large GEMM (batch = B·N·N).

Only the element‑wise mask‑gate fusion and the Layer‑Norm + out‑gate fusion
are written in Triton – the remainder uses highly‑optimised PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask (optional) and the left/right gates.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,                # fp16 * (B·N·N·H)
    right_ptr,               # fp16 * (B·N·N·H)
    left_gate_ptr,           # fp16 * (B·N·N·H)
    right_gate_ptr,          # fp16 * (B·N·N·H)
    mask_ptr,                # fp16 * (B·N·N)   (broadcasted over H)
    total_elements,          # int  (B·N·N·H)
    hidden_dim,              # int  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # load tensors (fp16)
    l  = tl.load(left_ptr  + offs, mask=in_range, other=0.0)
    r  = tl.load(right_ptr + offs, mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr  + offs, mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)

    if HAS_MASK:
        # each group of `hidden_dim` entries shares the same mask value
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # fuse: left = left * left_gate * mask,  right = right * right_gate * mask
    l = l * lg * m
    r = r * rg * m

    tl.store(left_ptr  + offs, l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


# ----------------------------------------------------------------------
# Kernel 2: Layer‑Norm over the hidden dimension followed by the out‑gate.
#          Operates on flattened tensors (shape = total_pos * hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def _ln_gate_kernel(
    out_ptr,            # fp16 * (total_pos * H)   – result of the cubic contraction
    out_gate_ptr,       # fp16 * (total_pos * H)
    out_res_ptr,        # fp32 * (total_pos * H)   – destination (norm + gate)
    gamma_ptr,          # fp32 * (H)                – to_out_norm.weight
    beta_ptr,           # fp32 * (H)                – to_out_norm.bias
    total_positions,    # int  = B·N·N
    HIDDEN_DIM: tl.constexpr,
    EPS: tl.constexpr,
    BLOCK_POS: tl.constexpr,
):
    pid = tl.program_id(0)
    # each program instance processes BLOCK_POS different positions
    pos = pid * BLOCK_POS + tl.arange(0, BLOCK_POS)
    mask_pos = pos < total_positions

    # ------------------------------------------------------------------
    # 1) compute mean and variance for each position
    # ------------------------------------------------------------------
    sum_   = tl.zeros((BLOCK_POS,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_POS,), dtype=tl.float32)

    for d in range(HIDDEN_DIM):
        idx = pos * HIDDEN_DIM + d                     # linear index
        v = tl.load(out_ptr + idx, mask=mask_pos, other=0.0)
        v_f = v.to(tl.float32)
        sum_   += v_f
        sum_sq += v_f * v_f

    mean = sum_ * (1.0 / HIDDEN_DIM)
    var  = sum_sq * (1.0 / HIDDEN_DIM) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ------------------------------------------------------------------
    # 2) normalise, apply LN weight/bias, multiply by out_gate and store
    # ------------------------------------------------------------------
    for d in range(HIDDEN_DIM):
        idx = pos * HIDDEN_DIM + d
        v  = tl.load(out_ptr      + idx, mask=mask_pos, other=0.0)
        gv = tl.load(out_gate_ptr + idx, mask=mask_pos, other=0.0)
        v_f  = v .to(tl.float32)
        gv_f = gv.to(tl.float32)

        gamma = tl.load(gamma_ptr + d)   # LN scale
        beta  = tl.load(beta_ptr  + d)   # LN bias

        norm = (v_f - mean) * inv_std
        out_val = norm * gamma + beta
        out_val = out_val * gv_f           # multiply by out‑gate

        tl.store(out_res_ptr + idx, out_val, mask=mask_pos)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5

    # ------------------------------------------------------------------
    # 2️⃣ load weights (cast to half where possible)
    # ------------------------------------------------------------------
    # LayerNorm 1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (no bias) – keep FP32 for better accuracy, cast later
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    # Output LayerNorm and final projection (remain FP32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                   # [B, N, N, dim]  float32
    x_h = x.half()                     # cast to FP16 for the heavy ops

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)      # [B, N, N, hidden_dim]  fp16
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Fuse optional mask + gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        # mask → half, broadcast over hidden_dim
        mask_h = mask.to(dtype=left.dtype).contiguous()   # [B,N,N] fp16
        mask_flat = mask_h.view(-1)                       # (B·N·N)

        # flatten tensors for the kernel
        left_flat       = left.view(-1)
        right_flat      = right.view(-1)
        left_gate_flat  = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        total_elements = left_flat.numel()                # B·N·N·H
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _apply_mask_gate_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=HAS_MASK,
        )
        # `left` and `right` have been updated in‑place.
        del left_gate, right_gate
    else:
        # No mask → simple element‑wise fuse in PyTorch
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare for the cubic contraction (batched GEMM)
    # ------------------------------------------------------------------
    # left  : (B, N, N, H) → (B, H, N, N)   with axes (i,k)
    # right : (B, N, N, H) → (B, H, N, N)   with axes (j,k) (swap last two N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)   fp16
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)   fp16

    # ------------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (Tensor‑core fast)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # (B·H, N, N)  fp16

    # Restore original layout: [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Fuse second LayerNorm and the out‑gate (Triton)
    # ------------------------------------------------------------------
    total_positions = B * N * N
    out_flat      = out.view(-1)                 # (total_positions * H)   fp16
    out_gate_flat = out_gate.view(-1)            # same shape
    out_ln_flat = torch.empty_like(out_flat, dtype=torch.float32)  # destination

    BLOCK_POS = 128        # number of positions processed per program instance
    grid = (triton.cdiv(total_positions, BLOCK_POS),)

    _ln_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_ln_flat,
        to_out_norm_w,
        to_out_norm_b,
        total_positions,
        hidden_dim,                # HIDDEN_DIM (constexpr)
        EPS=eps,
        BLOCK_POS=BLOCK_POS,
    )
    # reshape back to [B, N, N, hidden_dim] (float32)
    out_ln = out_ln_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (FP32 GEMM)
    # ------------------------------------------------------------------
    # torch.nn.functional.linear expects input (..., in_features)
    # and weight of shape (out_features, in_features).
    output = F.linear(out_ln, to_out_w)    # [B, N, N, dim]  float32

    return output
