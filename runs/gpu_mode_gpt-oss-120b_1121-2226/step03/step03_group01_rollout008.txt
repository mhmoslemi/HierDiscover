"""
TriMul “outgoing” forward pass (high‑performance Triton)

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to fp16 – all heavy linear work stays in half‑precision.
3. **Fused projections + gates** – concatenate the projection and gate
   weight matrices of each side (left/right) and run a single
   `torch.nn.functional.linear`.  The result is split into a
   projection and a gate; the gate is passed through `sigmoid` and
   multiplied element‑wise.  This halves the number of linear passes
   and reduces memory traffic.
4. Optional pair‑wise mask is applied (broadcast on the hidden dim).
5. Reshape to a batched 3‑D view `[B·H, N, N]` and launch a custom
   Triton kernel that computes a batched GEMM
   `C = A @ Bᵀ` (half‑precision inputs, float32 accumulation, float32
   output).  The kernel uses a 128×128×64 tile, loads tiles in fp16,
   accumulates directly in fp32 via `tl.dot(..., out_dtype=tl.float32)`
   and stores the result.
6. Reshape back to `[B, N, N, H]`.
7. Layer‑norm over the hidden dimension (float32).
8. Apply the output gate (sigmoid of a third linear) and the final
   bias‑free projection back to the original dimension.
9. Return a float32 tensor of shape `[B, N, N, D]`.

The heavy N³ work is performed by the Triton kernel, while the
surrounding linear + gating work is fused in PyTorch to minimise
memory passes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B : fp16   (shape [BH, N, N])
#   C    : fp32   (shape [BH, N, N])
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # ptr to left  : [BH, N, N]   (fp16)
    b_ptr,               # ptr to right : [BH, N, N]   (fp16) – interpreted as transposed
    c_ptr,               # ptr to out   : [BH, N, N]   (fp32)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)   # batch * hidden index
    pid_m = tl.program_id(axis=0)   # tile row
    pid_n = tl.program_id(axis=1)   # tile column

    # ------------------------------------------------------------------
    # Compute tile offsets and masks
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M x K) ------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                               # fp16

        # ----- Load Bᵀ tile (K x N) ------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                               # fp16

        # ----- Accumulate ------------------------------------------------
        # Direct fp32 accumulation via out_dtype argument (TensorCore friendly)
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                       # [B, N, N, D] – FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()      # [B, N, N, D] – FP16

    # ------------------------------------------------------------------
    # 3️⃣ Fused left projection + left gate
    # ------------------------------------------------------------------
    # Concatenate weights: [proj; gate] → shape (2*H, D)
    left_cat_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()
    left_cat = F.linear(x_fp16, left_cat_weight)          # [B, N, N, 2*H] – FP16
    left_proj = left_cat[..., :hidden_dim]                # [B,N,N,H]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:]) # [B,N,N,H]
    left = left_proj * left_gate                          # [B,N,N,H] – FP16

    # ------------------------------------------------------------------
    # 4️⃣ Fused right projection + right gate
    # ------------------------------------------------------------------
    right_cat_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()
    right_cat = F.linear(x_fp16, right_cat_weight)        # [B,N,N,2*H] – FP16
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])
    right = right_proj * right_gate                       # [B,N,N,H] – FP16

    # ------------------------------------------------------------------
    # 5️⃣ Optional pair‑wise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] → broadcast
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # fp16
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 6️⃣ Prepare batched GEMM inputs (BH, N, N)
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16, contiguous

    # Output buffer (FP32 accumulator)
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Launch Triton kernel for the batched GEMM
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # rows (i‑tiles)
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # cols (j‑tiles)
        BH,                                   # batch × hidden
    )

    # Strides are in *elements* (not bytes)
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # A strides (batch, i, k)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # Bᵀ strides (batch, k, j) – note the transposition
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # C strides (batch, i, j)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out_perm = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dim (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # stays FP32

    # ------------------------------------------------------------------
    # 🔟 Output gate (sigmoid) and final projection back to D
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].half()
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))   # [B,N,N,H] – fp16
    out = out * out_gate.float()                                 # broadcast, FP32

    # Final linear projection (bias‑free) back to original dimension D
    out = F.linear(out, weights["to_out.weight"])                # FP32

    return out
