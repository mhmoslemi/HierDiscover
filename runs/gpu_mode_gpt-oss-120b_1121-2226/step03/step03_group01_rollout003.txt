"""
TriMul “outgoing” kernel (forward only)

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (float32).
2. Cast to fp16 for the heavy linear work.
3. Compute the two linear projections (left / right) and the three
   sigmoid‑gates (left_gate, right_gate, out_gate) – all bias‑free.
4. Apply the optional pair‑wise mask and the two element‑wise gates.
5. Reshape to a batched 3‑D view (B·H, N, N) and launch a Triton kernel
   that computes the batched matrix multiplication

        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]

   i.e. `left @ rightᵀ`.  The kernel works on fp16 tiles, accumulates in
   fp32 and stores the result in fp32.
6. Reshape back to [B, N, N, H] and apply a second Layer‑norm over the
   hidden dimension H.
7. Multiply by the output gate and apply the final bias‑free linear
   projection back to the original dimension D.
8. Return a float32 tensor of shape [B, N, N, D].

Only the O(N³) matrix multiplication is executed in Triton; the rest uses
highly‑optimized torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B fp16 → C fp32)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, N, N] (fp16)
    b_ptr,               # right : [BH, N, N] (fp16) – interpreted as (K,N) = (N,N)ᵀ
    c_ptr,               # out   : [BH, N, N] (fp32)
    M, N, K,             # problem size (all equal to seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM `C = A @ Bᵀ`.
    Each thread block computes a TILE of size (BLOCK_M, BLOCK_N).
    Accumulation is done in fp32.
    """
    pid_h = tl.program_id(axis=2)          # batch * hidden channel
    pid_m = tl.program_id(axis=0)          # tile row
    pid_n = tl.program_id(axis=1)          # tile column

    # --------------------------------------------------------------
    # Compute tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- Load A tile ------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                                   # fp16

        # ---- Load Bᵀ tile ------------------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                   # fp16

        # ---- Accumulate --------------------------------------------------
        # dot returns fp16, cast to fp32 before accumulation
        acc += tl.dot(a, b).to(tl.float32)

    # --------------------------------------------------------------
    # Write result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # flag whether a pair‑wise mask is present
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Project to hidden dimension (bias‑free linear, fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(dtype=torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(dtype=torch.float16, device=device)

    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(dtype=left.dtype)      # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Compute three sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(dtype=torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(dtype=torch.float16, device=device)
    out_gate_w = weights["out_gate.weight"].to(dtype=torch.float16, device=device)

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape for batched GEMM: (B·H, N, N)
    # ------------------------------------------------------------------
    # From [B, N, N, H] → [B, H, N, N] → [B·H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim

    left_flat = left.view(BH, seq_len, seq_len)   # fp16, contiguous
    right_flat = right.view(BH, seq_len, seq_len)  # fp16, contiguous

    # Output buffer for the GEMM (fp32 accumulation)
    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Triton GEMM launch (C = A @ Bᵀ)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # tiles over i (rows)
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # tiles over j (cols)
        BH,                                   # batch*hidden
    )

    _tri_mul_outgoing_kernel[grid](
        # pointers
        left_flat,
        right_flat,
        out_flat,
        # problem size
        seq_len,                  # M
        seq_len,                  # N
        seq_len,                  # K
        # A (left) strides: batch, i, k
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        # Bᵀ (right) strides: batch, k, j
        right_flat.stride(0),
        right_flat.stride(2),   # stride for k (original column)
        right_flat.stride(1),   # stride for j (original row)
        # C (output) strides: batch, i, j
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        # compile‑time tile sizes
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()              # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # stays float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (cast to fp32) and final projection (float32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(dtype=torch.float32)            # broadcast gate
    out = F.linear(out, weights["to_out.weight"])           # bias‑free, back to D

    return out
