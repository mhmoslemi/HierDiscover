"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Cast the input to FP16 and apply the first LayerNorm in FP16.
2️⃣ Four FP16 linear projections (left, right, left‑gate, right‑gate, out‑gate).
   * The two projection tensors (left/right) are multiplied element‑wise
     by their respective gates and by the optional mask.
   * This fusion is performed by a tiny Triton kernel that works on a
     1‑D view of the tensors, avoiding an extra Python launch per element.
3️⃣ The cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is exactly a batched GEMM: for every hidden dimension *h* we need a
   (N×N) @ (N×N) matrix multiplication.  We reshape the tensors to
   (B·H, N, N) and call a single `torch.bmm`, which on H100 uses
   Tensor‑cores (FP16 matmul → FP32 accumulation) and is far faster than a
   Python loop.
4️⃣ A second LayerNorm (still in FP16), element‑wise multiplication by the
   output gate and a final FP16 linear projection back to `dim`.
5️⃣ Cast the result to FP32 for the model output.

All heavy element‑wise work (mask + gate fusion) is done in Triton, while
the O(N³) contraction leverages highly‑optimized cuBLAS batched GEMM.
Mixed‑precision (FP16) is used throughout except for the final cast,
giving a large speed‑up over the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # fp16 * (B·N·N·H)
    right_ptr,         # fp16 * (B·N·N·H)
    left_gate_ptr,     # fp16 * (B·N·N·H)
    right_gate_ptr,    # fp16 * (B·N·N·H)
    mask_ptr,          # fp16 * (B·N·N)   (broadcast over H)
    total_elements,    # i64  (B·N·N·H)
    hidden_dim,        # i64  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # Load tensors (fp16)
    l  = tl.load(left_ptr       + offs, mask=in_range, other=0.0)
    r  = tl.load(right_ptr      + offs, mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr  + offs, mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)

    # Optional mask (broadcasted over hidden dim)
    if HAS_MASK:
        # each `hidden_dim` consecutive elements share one mask entry
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)   # fp16
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # Fuse: left = left * left_gate * mask
    #       right = right * right_gate * mask
    l = l * lg * m
    r = r * rg * m

    # Store back in‑place
    tl.store(left_ptr  + offs, l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H in the paper

    # ------------------------------------------------------------------
    # 2️⃣ cast weights to the precision we will use (FP16 wherever possible)
    # ------------------------------------------------------------------
    norm_weight      = weights["norm.weight"].half()
    norm_bias        = weights["norm.bias"].half()

    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()

    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()

    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()

    # ------------------------------------------------------------------
    # 3️⃣ first LayerNorm in FP16
    # ------------------------------------------------------------------
    x = input_tensor.half()                                   # [B,N,N,dim] fp16
    x = F.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )                                                          # fp16

    # ------------------------------------------------------------------
    # 4️⃣ linear projections (no bias) – all in FP16
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)      # [B,N,N,H] fp16
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 5️⃣ gate values (sigmoid) – FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Fuse mask (if any) and the two gates with a tiny Triton kernel.
    #    We always launch the kernel; when there is no mask `HAS_MASK`
    #    is a compile‑time constant and the mask load is omitted.
    # ------------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0

    if HAS_MASK:
        mask_h = mask.to(dtype=left.dtype).contiguous()          # [B,N,N] fp16
        mask_flat = mask_h.view(-1)                               # (B·N·N)
    else:
        # Dummy tensor – never read because HAS_MASK==0
        mask_flat = left.new_empty((1,))

    # Flatten everything to 1‑D views for the kernel
    left_flat       = left.view(-1)
    right_flat      = right.view(-1)
    left_gate_flat  = left_gate.view(-1)
    right_gate_flat = right_gate.view(-1)

    total_elements = left_flat.numel()          # B·N·N·H
    BLOCK_SIZE = 8192                           # tuning parameter
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    _apply_mask_gate_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total_elements,
        hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=HAS_MASK,
    )
    # `left` and `right` have been updated in‑place.
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare tensors for the cubic contraction (batched GEMM)
    # ------------------------------------------------------------------
    # left : (B, N, N, H) → (B, H, N, N)   – rows = i, cols = k
    # right: (B, N, N, H) → (B, H, N, N)   – rows = j, cols = k (transpose last two N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape

    # Collapse (B·H) into the batch dimension for torch.bmm
    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)   fp16
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)   fp16

    # ------------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (Tensor‑core accelerated)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # (B·H, N, N)   fp16

    # Restore the original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm (still in FP16) + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )                                            # fp16
    out = out * out_gate                         # fp16

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (FP16) and cast to FP32
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)                # [B, N, N, dim] fp16
    return out.float()                           # cast back to float32
