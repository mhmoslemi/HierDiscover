"""
custom_kernel
--------------
Triton‑accelerated forward pass for the **outgoing** TriMul module.

Algorithm
---------
1. Layer‐norm the 4‑D input (float32).
2. Cast to fp16 and fuse the *four* linear projections/gates
   (left_proj, left_gate, right_proj, right_gate) into a single
   matrix multiplication to minimise memory traffic.
3. Compute the third gate (out_gate) with a separate linear.
4. Optional mask is applied (broadcasted) in fp16.
5. Element‑wise gate application.
6. Core cubic update
       out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   is a batched matrix multiplication L @ Rᵀ with the hidden dimension
   flattened into the batch axis.  For N ≤ 256 we use a handcrafted
   Triton kernel; for larger N we delegate to cuBLAS via ``torch.bmm``,
   which on an H100 is far faster than the custom kernel.
7. Second LayerNorm over the hidden dimension, followed by the output
   gate and a final linear projection back to the original feature
   dimension (float32).

All heavy arithmetic is performed in fp16 with fp32 accumulation,
while the final result is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ------------------------------------------------------------------
# Triton batched GEMM (used for very small N where launch overhead
# dominates).  It computes C = A @ Bᵀ for each batch entry.
# ------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row index
    pid_n = tl.program_id(1)          # tile col index
    pid_b = tl.program_id(2)          # batch index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop, tiled
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        acc += tl.dot(a, b)                           # fp16·fp16 → fp32

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Triton implementation of a batched GEMM (A @ Bᵀ).

    A, B : (batch, M, K) and (batch, N, K) – fp16 tensors.
    Returns C : (batch, M, N) – fp32 tensor.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = ((M + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            batch)

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bn, stride_bk,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the *outgoing* TriMul module.

    Arguments
    ----------
    data : (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : Dict of all linear / norm parameters (float32)
        - config_dict  : contains ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` flag.

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 3) Cast to fp16 for the heavy ops
    # ------------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim] – fp16
    B_, N_, _, _ = x_h.shape

    # ------------------------------------------------------------------
    # 4) Fuse the four linear projections/gates (left/right + gates)
    # ------------------------------------------------------------------
    # Flatten the 4‑D tensor to (B·N·N, dim)
    x_flat = x_h.view(-1, dim)         # [B·N·N, dim]

    # Retrieve and cast the five weight matrices
    left_proj_w   = weights["left_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_proj_w  = weights["right_proj.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # Combine the four “pairwise” weights (proj & gate for left/right)
    # Shape after cat: (4·hidden_dim, dim)
    combined_w = torch.cat(
        [
            left_proj_w,
            left_gate_w,
            right_proj_w,
            right_gate_w,
        ],
        dim=0,
    ).half()   # fp16

    # One fused matmul → (B·N·N, 4·hidden_dim)
    combined_out = F.linear(x_flat, combined_w)   # fp16
    combined_out = combined_out.view(B_, N_, N_, 4 * hidden_dim)

    # Split the fused result
    left      = combined_out[..., 0 * hidden_dim : 1 * hidden_dim]   # [B,N,N,H]
    left_gate = combined_out[..., 1 * hidden_dim : 2 * hidden_dim]   # [B,N,N,H]
    right     = combined_out[..., 2 * hidden_dim : 3 * hidden_dim]   # [B,N,N,H]
    right_gate= combined_out[..., 3 * hidden_dim : 4 * hidden_dim]   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5) Compute the output‑gate (stand‑alone linear)
    # ------------------------------------------------------------------
    out_gate = F.linear(x_h, out_gate_w.half()).sigmoid()   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6) Optional mask (broadcasted)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * m
        right = right * m

    # ------------------------------------------------------------------
    # 7) Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate  = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8) Core cubic update – batch matmul L @ Rᵀ
    # ------------------------------------------------------------------
    # Reshape to (B·H, N, N) – contiguous layout required by GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B_ * hidden_dim, N_, N_)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B_ * hidden_dim, N_, N_)

    # Choose execution path:
    #   • For tiny N (≤256) the handcrafted Triton kernel beats cuBLAS.
    #   • Otherwise we let cuBLAS (torch.bmm) handle the large GEMMs.
    if N_ <= 256:
        out_mat = _batched_matmul_triton(
            left_mat,
            right_mat,
            BLOCK_M=64,
            BLOCK_N=64,
            BLOCK_K=64,
        )  # fp32
    else:
        # torch.bmm uses the highly‑optimized cuBLAS/cuBLASLt kernels on H100
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).float()  # fp32

    # Restore layout → [B, N, N, hidden_dim]
    out = out_mat.view(B_, hidden_dim, N_, N_).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 10) Output gate & final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.float()                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])    # [B,N,N,dim] – float32

    return out
