def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1 layer norm: float32
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # convert to fp16
    x_fp16 = x.half()

    # flatten for linear ops
    x_flat = x_fp16.view(-1, dim)  # (bs*N*N, dim)

    # pre-convert weights to half
    left_proj_w = weights["left_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    # concatenate projection+gate for fused matmul
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2*hidden, dim)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # compute concatenated linear outputs
    left_concat = F.linear(x_flat, left_concat_w)  # (bs*N*N, 2*hidden)
    right_concat = F.linear(x_flat, right_concat_w)

    # split
    left_proj = left_concat[:, :hidden_dim]   # (bs*N*N, hidden)
    left_gate = left_concat[:, hidden_dim:]   # (bs*N*N, hidden)
    right_proj = right_concat[:, :hidden_dim]
    right_gate = right_concat[:, hidden_dim:]

    # apply mask if needed (mask shape [bs,N,N,1] after unsqueeze)
    if not nomask:
        mask = mask_tensor.to(left_proj.dtype).view(-1, 1)   # flatten to (bs*N*N, 1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # apply sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # reshape to (bs, hidden, N, N) for kernel
    left = left.view(bs, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()   # (bs, hidden, N, N)
    right = right.view(bs, N, N, hidden_dim).permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim
    left_flat = left.view(BH, N, N)
    right_flat = right.view(BH, N, N)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # launch triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )
    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back to [bs, N, N, hidden]
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # second layer norm over hidden dim
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # compute out_gate
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # shape (bs,N,N,hidden)
    # broadcast and apply gate (cast to float32)
    out = out * out_gate.to(out.dtype)

    # final projection back to dim
    out = F.linear(out, weights["to_out.weight"])

"""
TriMul “outgoing” implementation (forward only)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input tensor (float32).
2️⃣ Cast to fp16 – all heavy linear work stays in half‑precision.
3️⃣ Fuse the two projections and the two gate‑linear layers by concatenating
   their weights, so that each of the left/right branches is computed with a
   single matmul:
        left_concat = [left_proj | left_gate] @ x
        right_concat = [right_proj | right_gate] @ x
   Afterwards we split the result, apply the optional pair‑wise mask only to
   the projection part and apply sigmoid to the gate part.
4️⃣ Reshape to a batched 3‑D view (B·H, N, N) and launch a Triton kernel that
   computes the batched GEMM
        out = left @ rightᵀ    (A,B are fp16, accumulation in fp32)
   The kernel works on tiles (128×128×64), accumulating in float32 and
   storing the result in float32.
5️⃣ Reshape back to [B, N, N, H] and apply a second LayerNorm over the hidden
   dimension.
6️⃣ Compute the output gate from the normalized input, multiply it with the
   normalized result and finish with a bias‑free linear projection back to the
   original dimension D.
7️⃣ Return a float32 tensor of shape [B, N, N, D].

Only the O(N³) matrix multiplication is performed in Triton; all surrounding
operations use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B fp16, C fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # ptr to left  : [BH, N, N] (fp16)
    b_ptr,               # ptr to right : [BH, N, N] (fp16), interpreted as (K,N)
    c_ptr,               # ptr to out   : [BH, N, N] (fp32)
    M, N, K,             # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix‑multiply `C = A @ Bᵀ`.  Accumulates in fp32 and writes fp32.
    """
    pid_h = tl.program_id(axis=2)      # batch × hidden
    pid_m = tl.program_id(axis=0)      # tile row
    pid_n = tl.program_id(axis=1)      # tile column

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- A tile (M × K) ------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ----- Bᵀ tile (K × N) ------------------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # ----- Accumulate in FP32 -------------------------------------------
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.half()                                   # [B, N, N, D] – fp16
    x_flat = x_fp16.view(-1, dim)                       # (B·N·N, D)

    # --------------------------------------------------------------
    # 3️⃣ Fuse projection + gate linear layers (bias‑free) – fp16
    # --------------------------------------------------------------
    # Convert weights once
    left_proj_w  = weights["left_proj.weight"].half()   # (H, D)
    left_gate_w  = weights["left_gate.weight"].half()   # (H, D)
    right_proj_w = weights["right_proj.weight"].half()  # (H, D)
    right_gate_w = weights["right_gate.weight"].half()  # (H, D)
    out_gate_w   = weights["out_gate.weight"].half()    # (H, D)

    # Concatenate projection and gate weights (2*H, D)
    left_concat_w  = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, D)
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0) # (2H, D)

    # One fused matmul per side
    left_concat  = F.linear(x_flat, left_concat_w)   # (B·N·N, 2H)  fp16
    right_concat = F.linear(x_flat, right_concat_w)  # (B·N·N, 2H)  fp16

    # Split back into projection / gate tensors
    left_proj = left_concat[:, :hidden_dim]          # (B·N·N, H)
    left_gate = left_concat[:, hidden_dim:]          # (B·N·N, H)
    right_proj = right_concat[:, :hidden_dim]        # (B·N·N, H)
    right_gate = right_concat[:, hidden_dim:]        # (B·N·N, H)

    # --------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(left_proj.dtype).view(-1, 1)   # (B·N·N, 1)
        left_proj  = left_proj  * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 5️⃣ Gating – sigmoid (still fp16)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # --------------------------------------------------------------
    # 6️⃣ Apply element‑wise gates
    # --------------------------------------------------------------
    left  = left_proj * left_gate   # (B·N·N, H)  fp16
    right = right_proj * right_gate # (B·N·N, H)  fp16

    # --------------------------------------------------------------
    # 7️⃣ Reshape to [B·H, N, N] for the batched GEMM
    # --------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left = left.view(bs, seq_len, seq_len, hidden_dim).permute(0, 3, 1, 2).contiguous()
    right = right.view(bs, seq_len, seq_len, hidden_dim).permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim                                   # batch × hidden
    left_flat = left.view(BH, seq_len, seq_len)           # fp16, contiguous
    right_flat = right.view(BH, seq_len, seq_len)         # fp16, contiguous

    out_flat = torch.empty((BH, seq_len, seq_len), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # 8️⃣ Launch the Triton kernel (batched GEMM)
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                                   # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right – transposed view)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 9️⃣ Reshape back to [B, N, N, H]
    # --------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # fp32

    # --------------------------------------------------------------
    # 🔟 Second LayerNorm over the hidden dimension (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # stays float32

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Compute output gate (still fp16) and apply it
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))  # (B, N, N, H) fp16
    out = out * out_gate.to(out.dtype)                     # broadcast, back‑to‑fp32

    # --------------------------------------------------------------
    # 1️⃣2️⃣ Final linear projection back to the original dimension D
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])          # bias‑free, float32

    return out
```
