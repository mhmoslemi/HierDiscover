"""
TriMul “outgoing” forward (no backward) – Triton‑accelerated core

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 – all heavy work (linear layers, gemm) runs in half precision.
3. Compute the two bias‑free projections (left, right) and the three
   sigmoid‑gates (left_gate, right_gate, out_gate) with torch.nn.functional.linear.
   The optional pair‑wise mask is applied to left/right before gating.
4. Fuse the hidden dimension into the batch dimension:
        (B, N, N, H) → (B·H, N, N)
   and launch a Triton kernel that evaluates, for every batch element,
        out = left @ rightᵀ
   i.e. a batched matrix multiplication on fp16 tiles with fp32 accumulation.
5. Reshape back to (B, N, N, H).
6. Layer‑norm over the hidden dimension (float32).
7. Apply the output gate and a final bias‑free linear projection back to the
   original dimensionality (float32).
8. Return the result (float32).

Only the N³ matrix multiplication (the true bottleneck) is executed in Triton;
all surrounding ops use highly‑optimised torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for batched GEMM: C = A @ Bᵀ
#   A, B : fp16, shape (BH, N, N)
#   C    : fp32, shape (BH, N, N)
#   Each block computes a (BLOCK_M × BLOCK_N) tile.
#   Accumulation is performed in fp32 via `tl.dot(..., out_dtype=tl.float32)`.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,  # left  : [BH, N, N]   (fp16)
    b_ptr,  # right : [BH, N, N]   (fp16) – interpreted as (K, N)
    c_ptr,  # out   : [BH, N, N]   (fp32)
    M, N, K,                 # all equal to seq_len
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)            # batch * hidden
    pid_m = tl.program_id(axis=0)            # tile row
    pid_n = tl.program_id(axis=1)            # tile col

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # fp32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- A tile (M × K) -------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # fp16

        # ---- Bᵀ tile (K × N) -------------------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # fp16

        # ---- Accumulate ----------------------------------------------------
        # `out_dtype=tl.float32` makes the dot use tensor‑cores and accumulate
        # directly in FP32, avoiding a separate cast.
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write the tile back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point – the forward pass of TriMul “outgoing”
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul outgoing operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping of weight names to torch tensors (float32)
        - config_dict  : {"dim": D, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack ---------------------------------------------------------
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) ---------------------------------------
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work -------------------------
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) – fp16 → fp16 --------------------
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on hidden dim) ---------------
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N] → [B, N, N, 1] and cast to fp16
        m = mask_tensor.unsqueeze(-1).to(left.dtype)
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 5️⃣ Gating – sigmoid (still fp16) -----------------------------------
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # Apply the element‑wise gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape for the batched GEMM ------------------------------------
    # ------------------------------------------------------------------
    # From (B, N, N, H) → (B, H, N, N) → (B·H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    BH = B * H
    left_flat  = left.view(BH, N, N)   # fp16, contiguous
    right_flat = right.view(BH, N, N)  # fp16, contiguous

    # Output buffer (fp32 accumulation)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7️⃣ Launch Triton kernel (batched GEMM) -----------------------------
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        BH,                             # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        # A strides
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # Bᵀ strides (right interpreted as transposed)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # C strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to (B, N, N, H) ------------------------------------
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (float32) ----------------
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # remains float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (broadcast) and final linear projection --------
    # ------------------------------------------------------------------
    # out_gate is fp16, cast to float32 for the multiplication
    out = out * out_gate.to(out.dtype)
    out = F.linear(out, weights["to_out.weight"])  # bias‑free, back to D (float32)

    return out
