import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1️⃣ Layer‑norm over the `dim` axis of the input tensor.
2️⃣ Cast to FP16 and compute four linear layers:
   * left  projection  (dim → hidden)
   * right projection  (dim → hidden)
   * left  gate       (dim → hidden) + sigmoid
   * right gate       (dim → hidden) + sigmoid
   * out   gate       (dim → hidden) + sigmoid
3️⃣ Fuse *mask* (if present) with the two gates in a tiny Triton kernel.
4️⃣ Rearrange the tensors so that the cubic contraction becomes a batch of
   matrix multiplications:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batched GEMM   (B·H, N, N) @ (B·H, N, N).
   The heavy N³ work is delegated to `torch.bmm`, which on an H100 uses
   Tensor‑cores (FP16 → FP32 accumulation) and is far faster than a naïve
   PyTorch loop.
5️⃣ Second Layer‑norm over the hidden dimension, apply the output gate
   and a final FP32 linear projection back to `dim`.

Only the element‑wise mask‑and‑gate fusion is written in Triton – the rest
uses highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask (optional) and the left/right gates into the
# projected tensors.  Operates on flattened tensors that are contiguous.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,                # fp16 * (B·N·N·H)
    right_ptr,               # fp16 * (B·N·N·H)
    left_gate_ptr,           # fp16 * (B·N·N·H)
    right_gate_ptr,          # fp16 * (B·N·N·H)
    mask_ptr,                # fp16 * (B·N·N)   (broadcasted over H)
    total_elements,          # int  (B·N·N·H)
    hidden_dim,              # int  = H
    BLOCK_SIZE: tl.constexpr,
    HAS_MASK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # ------------------------------------------------------------------
    # Load the four tensors (fp16)
    # ------------------------------------------------------------------
    l = tl.load(left_ptr + offs,        mask=in_range, other=0.0)
    r = tl.load(right_ptr + offs,       mask=in_range, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=in_range, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=in_range, other=0.0)

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        # each group of `hidden_dim` consecutive elements shares the same mask
        mask_idx = offs // hidden_dim
        m = tl.load(mask_ptr + mask_idx, mask=in_range, other=1.0)   # fp16
    else:
        m = tl.full((BLOCK_SIZE,), 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Fuse:  left = left * left_gate * mask
    #        right = right * right_gate * mask
    # ------------------------------------------------------------------
    l = l * lg * m
    r = r * rg * m

    # ------------------------------------------------------------------
    # Store results back in-place
    # ------------------------------------------------------------------
    tl.store(left_ptr + offs,  l, mask=in_range)
    tl.store(right_ptr + offs, r, mask=in_range)


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict with all model parameters (float32)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]      # called `hidden_dim` in the paper

    # ------------------------------------------------------------------
    # 2️⃣ load weights (cast to half where possible)
    # ------------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (no bias) – cast to FP16
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    # Gate linear layers – FP16
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # Output LayerNorm & final projection (remain FP32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                   # [B, N, N, dim]  float32
    x_h = x.half()                     # [B, N, N, dim]  float16

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)      # [B, N, N, hidden_dim] fp16
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 5️⃣ Gates (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6️⃣ Fuse optional mask + gates with a tiny Triton kernel
    # ------------------------------------------------------------------
    HAS_MASK = 1 if mask is not None and mask.numel() > 0 else 0
    if HAS_MASK:
        # mask → half, broadcast over hidden_dim
        mask_h = mask.to(dtype=left.dtype).contiguous()   # [B,N,N] fp16
        mask_flat = mask_h.view(-1)                       # (B·N·N)

        # Flatten everything so the kernel can work on a 1‑D view
        left_flat       = left.view(-1)
        right_flat      = right.view(-1)
        left_gate_flat  = left_gate.view(-1)
        right_gate_flat = right_gate.view(-1)

        total_elements = left_flat.numel()                # B·N·N·H
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

        _apply_mask_gate_kernel[grid](
            left_flat,
            right_flat,
            left_gate_flat,
            right_gate_flat,
            mask_flat,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
            HAS_MASK=HAS_MASK,
        )
        # The tensors `left` and `right` have been updated in‑place.
        # Gates are no longer needed.
        del left_gate, right_gate
    else:
        # No mask → simple element‑wise fuse in PyTorch
        left = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare for the cubic contraction (batched GEMM)
    # ------------------------------------------------------------------
    # left  : (B, N, N, H) → (B, H, N, N)   with axes (i,k)
    # right : (B, N, N, H) → (B, H, N, N)   with axes (k,j) (swap last two N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape

    # Collapse (B·H) into the batch dimension for torch.bmm
    left_mat  = left_perm.view(B * H, N, N)   # (B·H, N, N)  fp16
    right_mat = right_perm.view(B * H, N, N)  # (B·H, N, N)  fp16

    # ------------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (Tensor‑core fast)
    # ------------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)   # (B·H, N, N)  fp16

    # Restore original layout: [B, N, N, H] (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = out.float()  # up‑cast for the LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                   # [B, N, N, hidden_dim] float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (float32) and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.float()
    out = F.linear(out, to_out_w)       # [B, N, N, dim] float32

    return out
