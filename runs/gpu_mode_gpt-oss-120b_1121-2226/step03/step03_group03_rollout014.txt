import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel
@triton.jit
def _tri_mul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_ac, stride_am, stride_ak,
    stride_bh, stride_bc, stride_bn, stride_bk,
    stride_ch, stride_cc, stride_cm, stride_cn,
    H,                     # hidden dimension (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # row tile
    pid_n = tl.program_id(1)   # col tile
    pid   = tl.program_id(2)   # (batch, hidden) flattened

    # decode batch and channel indices
    batch_id = pid // H
    chan_id  = pid - batch_id * H   # pid % H

    # base pointers for this (batch, channel) pair
    a_batch = A_ptr + batch_id * stride_ah + chan_id * stride_ac
    b_batch = B_ptr + batch_id * stride_bh + chan_id * stride_bc
    c_batch = C_ptr + batch_id * stride_ch + chan_id * stride_cc

    # tile offsets (row and column)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # A tile (M x K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (cur_k + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile, transposed on‑the‑fly (K x N)
        b_ptrs = b_batch + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (cur_k + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # fused fp16·fp16 → fp32 dot
        acc += tl.dot(a_tile, b_tile, out_dtype=tl.float32)

    # write‑back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_tri_mul_outgoing(A: torch.Tensor,
                              B: torch.Tensor,
                              hidden_dim: int,
                              BLOCK_M: int = 128,
                              BLOCK_N: int = 128,
                              BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched GEMM for the outgoing TriMul update:
        out_{b,i,j,h} = Σ_k A_{b,i,k,h} * B_{b,j,k,h}
    A and B are fp16 tensors of shape (B, N, N, H).
    Returns C in fp32 of the same shape.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    B_dim = hidden_dim
    batch, N, _, H_check = A.shape
    assert H_check == hidden_dim
    assert B.shape == (batch, N, N, hidden_dim)

    # allocate output in fp32
    C = torch.empty_like(A, dtype=torch.float32)

    # strides in element units
    a_s_batch, a_s_row, a_s_col, a_s_chan = A.stride()
    b_s_batch, b_s_row, b_s_col, b_s_chan = B.stride()
    c_s_batch, c_s_row, c_s_col, c_s_chan = C.stride()

    grid = (

        (N + BLOCK_N - 1) // BLOCK_N,

        B,
        C,

        N,                 # N

        c_s_batch, c_s_chan, c_s_row, c_s_col,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    The implementation follows the reference PyTorch code but moves the
    N³ pairwise multiplication to a custom Triton kernel that works on
    the (B, N, N, H) layout without any extra copying.  Input LayerNorm,
    linear projections, gating and the final LayerNorm + linear projection
    are performed with PyTorch (float16 where possible) to keep the
    overall memory traffic low.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the feature dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] float32

    # cast to fp16 for the heavy linear / gate work
    x_f16 = x.half()

    # --------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) and three gates
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    left  = F.linear(x_f16, left_proj_w)    # [B, N, N, H] fp16
    right = F.linear(x_f16, right_proj_w)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional binary mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise gate application
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core outgoing TriMul (batched GEMM) using Triton
    # --------------------------------------------------------------
    # left/right have shape (B, N, N, H)
    out = _batched_tri_mul_outgoing(left, right, hidden_dim)

    # --------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gate, and final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # out_gate will be up‑cast to fp32 automatically
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul – outgoing update (Triton‑accelerated core)

The implementation follows the reference PyTorch module but moves the
N³ pairwise multiplication

    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}

to a custom Triton kernel.  The kernel works directly on the
[B, N, N, H] layout (no extra permutes or copies) by treating the
hidden dimension as an extra batch axis (total batches = B·H).  All
other operations (LayerNorm, linear projections and sigmoid gates) are
performed with PyTorch in fp16/fp32 as appropriate.  The final output
has shape [B, N, N, D] and dtype torch.float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: outgoing TriMul (batched GEMM)
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#   left/right : fp16     (B, N, N, H)
#   out        : fp32     (B, N, N, H)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    A_ptr, B_ptr, C_ptr,                 # tensors
    M, N, K,                             # matrix sizes (all = seq_len)
    stride_ah, stride_ac, stride_am, stride_ak,   # A strides: batch, channel, row, col
    stride_bh, stride_bc, stride_bn, stride_bk,   # B strides
    stride_ch, stride_cc, stride_cm, stride_cn,   # C strides
    H,                                   # hidden dimension (int, not constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block column
    pid   = tl.program_id(2)   # flattened (batch, hidden)

    # decode batch and hidden‑channel indices
    batch_id = pid // H
    chan_id  = pid % H

    # base pointers for this (batch, channel) pair
    a_batch = A_ptr + batch_id * stride_ah + chan_id * stride_ac
    b_batch = B_ptr + batch_id * stride_bh + chan_id * stride_bc
    c_batch = C_ptr + batch_id * stride_ch + chan_id * stride_cc

    # tile offsets for rows and columns
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K

        # ---- left tile (M × K) -------------------------------------------------
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (cur_k + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # ---- right tile, transposed on‑the‑fly (K × N) -------------------------
        b_ptrs = b_batch + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (cur_k + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # fused half·half → fp32 dot (uses Tensor Cores)
        acc += tl.dot(a_tile, b_tile, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # write back the result
    # ------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_tri_mul_outgoing(
    left: torch.Tensor,
    right: torch.Tensor,
    hidden_dim: int,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left / right : [B, N, N, H]  (fp16)
    returns      : [B, N, N, H]  (fp32)
    """

    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, N2, H = left.shape
    assert N == N2 and H == hidden_dim
    assert right.shape == (B, N, N, hidden_dim)

        right,

        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # good balance on H100 for 128×128 tiles
        num_stages=4,     # double‑buffered loads
    )
    return out

    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args
    ----
    data : tuple (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor : [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm over the feature dimension D
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # [B, N, N, D]  float32
    x_f16 = x.half()                # fp16 for the heavy linear work

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) and three sigmoid gates
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    left  = F.linear(x_f16, left_proj_w)     # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w)    # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional binary mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:

        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid) and element‑wise gate application
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core N³ update – Triton kernel
    # ------------------------------------------------------------------
    # left/right : [B, N, N, H]  fp16
    out = _batched_tri_mul_outgoing(left, right, hidden_dim)

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm, gate multiplication, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                               # fp32, over hidden dimension

    out = out * out_gate            # out_gate auto‑casts to fp32
    out = F.linear(out, weights["to_out.weight"])

    return out
