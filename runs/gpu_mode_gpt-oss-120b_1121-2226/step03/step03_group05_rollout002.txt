@triton.jit
def _gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.int32,
    total_elems: tl.int32,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load left_val, left_gate_val, right_val, right_gate_val
    l_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Compute mask index:
    mask_idx = offs // hidden_dim
    # Load mask value (broadcast)
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse
    l_out = l_val * l_gate * m
    r_out = r_val * r_gate * m

    tl.store(left_ptr + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)

@triton.jit
def _gate_fuse_kernel(
    left_ptr,
    left_gate_ptr,
    right_ptr,
    right_gate_ptr,
    mask_ptr,
    hidden_dim: tl.constexpr,
    total_elems: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    l_val = tl.load(left_ptr + offs, mask=mask, other=tl.float16(0.0))
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val = tl.load(right_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Determine mask index (broadcast across hidden_dim)
    mask_idx = offs // hidden_dim
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)

"""
Highly‑optimized forward pass for the “outgoing” TriMul operator.

Algorithm
---------
1. Input LayerNorm (FP32).
2. Cast to FP16 and perform **one** fused linear projection that produces
   the 5 tensors needed for the update (left, right, left_gate, right_gate,
   out_gate).
3. Apply sigmoid to gate tensors.
4. Fuse element‑wise multiplication of left/right with their gates and the
   optional binary mask using a small Triton kernel (single launch).
5. Core N³ update: for each hidden channel h compute
        out[...,h] = left[...,h] @ right[...,h].T
   implemented as a single batched GEMM (torch.bmm) over the reshaped
   tensors (leveraging cuBLAS Tensor‑Cores on H100).
6. Output LayerNorm (FP32) → multiply by out_gate → final linear projection
   back to the model dimension (FP32).

All heavy linear work runs in FP16; only the two LayerNorms and the final
projection run in FP32.  The only Triton kernel is the element‑wise fuse,
which removes separate PyTorch kernels for masking and gating and therefore
reduces launch overhead and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse gate multiplication and optional mask.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# mask is broadcast over the hidden dimension.
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,                     # [B, N, N]  (fp16, 0/1)
    hidden_dim: tl.constexpr,    # size of the hidden dimension H
    total_elems: tl.int32,       # B * N * N * H
    BLOCK_SIZE: tl.constexpr,    # threads per program
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load values
    l_val   = tl.load(left_ptr      + offs, mask=mask, other=tl.float16(0.0))
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val   = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # Compute mask index (broadcast over hidden_dim)
    mask_idx = offs // hidden_dim
    m_val   = tl.load(mask_ptr + mask_idx, mask=mask, other=tl.float16(1.0))

    # Fuse the three multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Triton kernel: fuse gate multiplication when no mask is present.
# ----------------------------------------------------------------------
@triton.jit
def _gate_fuse_no_mask_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    total_elems: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    l_val  = tl.load(left_ptr      + offs, mask=mask, other=tl.float16(0.0))
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    r_val  = tl.load(right_ptr      + offs, mask=mask, other=tl.float16(0.0))
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    l_out = l_val * l_gate
    r_out = r_val * r_gate

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]      (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]   FP32

    # ------------------------------------------------------------------
    # 3) Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) ONE fused projection that yields left, right, gates, out_gate
    # ------------------------------------------------------------------
    # Convert model weights to FP16 once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Stack all five weight matrices → shape (5*H, dim)
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*hidden_dim, dim)

    # Combined linear (FP16→FP16)
    fused_proj = F.linear(x_f16, fused_weight)  # [B, N, N, 5*H]  FP16

    # Split back into the five tensors
    left, right, left_gate_pre, right_gate_pre, out_gate_pre = torch.split(
        fused_proj, hidden_dim, dim=-1
    )
    # Apply sigmoid to gates (still FP16)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)  # shape [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 5) Fuse gate multiplication (and optional mask) using Triton
    # ------------------------------------------------------------------
    total = left.numel()                         # B * N * N * H
    BLOCK = 256                                  # 256 threads per block → 8 warps
    grid = ( (total + BLOCK - 1) // BLOCK, )     # 1‑D launch grid

    if not nomask and mask is not None:
        # mask: [B, N, N] → fp16, broadcast over hidden dim
        mask_f16 = mask.to(torch.float16).contiguous()
        # Ensure all tensors are contiguous before passing raw pointers
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        _gate_fuse_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )
    else:
        # No mask – just apply the gates
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        _gate_fuse_no_mask_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=8,
        )

    # ------------------------------------------------------------------
    # 6) Core N³ update: batched mat‑mul L @ Rᵀ  (FP16 → FP32)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # [B, N, N, H]

    # Reshape to (B*H, N, N) for a single torch.bmm call
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched mat‑mul (uses Tensor‑Cores on H100)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # FP16
    out_mat = out_mat.to(torch.float32)                       # cast for LayerNorm

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP32

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # FP32

    # out_gate is FP16 → automatic up‑cast in the multiplication
    out = out * out_gate

    # Final linear projection back to model dimension (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
