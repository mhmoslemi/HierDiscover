def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # layernorm
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])
    # cast to fp16
    x_half = x.half()
    
    # linear projections
    left = F.linear(x_half, weights["left_proj.weight"].half())
    right = F.linear(x_half, weights["right_proj.weight"].half())
    
    if not nomask:
        m = mask.to(left.dtype).unsqueeze(-1) # (B,N,N,1)
        left = left * m
        right = right * m
    
    # gates
    left_gate = F.linear(x_half, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_half, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_half, weights["out_gate.weight"].half()).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    B,N,_,H = left.shape
    
    # reshape to (B*H, N, N) contiguous. this copies due to permute+contiguous.
    left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)
    
    # Choose multiplication method
    if N >= 512:
        # torch.bmm is highly optimized on Hopper
        out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
        out_mat = out_mat.float()
    else:
        # Triton kernel
        if N <= 384:
            BLOCK_M = BLOCK_N = 64
        else:
            BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        out_mat = _batched_matmul_triton(
            left_mat,
            right_mat,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()
    
    # second layernorm over hidden dim
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    
    out = out * out_gate.float()  # gating
    
    # final linear projection
    out = F.linear(out, weights["to_out.weight"])
    
    return out

"""
custom_kernel
--------------
Triton‑accelerated forward pass for the **outgoing** TriMul module (AlphaFold3).

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32).
2.  Cast to fp16 and apply three linear projections (left/right) and three
    gating linear layers (no bias).  The optional mask and sigmoid gates are
    fused in half‑precision.
3.  The cubic update
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    is a batched matrix multiplication L @ Rᵀ for each hidden channel.  We
    reshape the tensors to (B·H,N,N) where H = hidden_dim and use a tiled
    Triton kernel (mixed‑precision fp16·fp16 → fp32) for modest sequence lengths.
    For larger N (≥ 512) we fall back to cuBLAS via ``torch.bmm`` which is
    heavily optimized on H100.
4.  Reshape back to (B,N,N,H) and apply a second LayerNorm over the hidden
    dimension.
5.  Multiply by the output gate and finish with a linear projection back to
    the original feature dimension (float32).

The implementation keeps all heavy arithmetic on the GPU, uses only a
single Triton kernel (required by the problem statement), and switches to
the highly‑optimized cuBLAS routine for the biggest workloads to meet the
target runtime.
"""

import torch
import torch.nn.functional as F
from typing import Tuple, Dict

import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
# Computes C = A @ Bᵀ for each batch entry.
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # global pointers
    M, N, K,                                 # matrix dimensions
    stride_ah, stride_am, stride_ak,         # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,         # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,         # C strides: batch, M, N
    H: tl.constexpr,                         # hidden dimension (used for batch split)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # combined batch index (b * H + h)

    # ----------------------------------------------------------
    # Decompose pid_b into outer batch (b) and hidden index (h)
    # ----------------------------------------------------------
    b_idx = pid_b // H
    h_idx = pid_b % H

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # -----------------------------------------------------------------
    # Pointers to the start of the tile in A and B.
    # A is interpreted as (batch, M, K) where K = N and M = N.
    # B is interpreted as (batch, K, N) (i.e. transposed on‑the‑fly).
    # -----------------------------------------------------------------
    a_ptrs = (
        A_ptr
        + b_idx * stride_ah               # move to the batch
        + h_idx * stride_ak               # move to the hidden channel inside the batch
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + b_idx * stride_bh
        + h_idx * stride_bk
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator in fp32 (mixed‑precision Tensor‑Core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over K dimension in tiles
    # ----------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # Out‑of‑bounds masks
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulation

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----------------------------------------------------------
    # Write out the tile
    # ----------------------------------------------------------
    c_ptrs = (
        C_ptr
        + b_idx * stride_ch
        + h_idx * stride_cn               # hidden stride of output (normally 1)
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matmul for the TriMul update.
    A, B : (batch, N, N) – fp16 tensors.
    Returns C : (batch, N, N) – fp32 tensor.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    # Output tensor (fp32)
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # hidden dimension (required for batch split)
    # In our usage batch == B * H, therefore H = batch // original_B
    # We recover H by inspecting the stride of the innermost dimension of A
    # (which is 1 for a contiguous layout after permute+contiguous).
    H = stride_ak // 1  # this yields H because after the view stride_ak == 1

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        H=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for heavy compute
    # ------------------------------------------------------------------
    x_h = x.half()      # [B, N, N, dim] – fp16

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].half())   # [B, N, N, H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4) Optional mask (fp16) – applied before gating
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].half()).sigmoid()

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core cubic update – batched matmul
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # Re‑order to (B, H, N, N) and make it contiguous so that the inner
    # dimension (K) is laid out consecutively.  This copy is cheap compared
    # to the O(N³) work.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Choose the fastest matmul implementation based on sequence length.
    if N >= 512:
        # cuBLAS batched GEMM (torch.bmm) is heavily tuned on H100.
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).float()   # [B*H, N, N]
    else:
        # Triton‑based tiled matmul.
        if N <= 384:
            BLOCK_M, BLOCK_N = 64, 64
        else:
            BLOCK_M, BLOCK_N = 128, 128
        BLOCK_K = 64
        out_mat = _batched_matmul_triton(
            left_mat,
            right_mat,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )  # fp32

    # Reshape back to original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over hidden dimension (fp32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 9) Output gating and final projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate.float()                     # element‑wise gated output
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] (float32)

    return out
