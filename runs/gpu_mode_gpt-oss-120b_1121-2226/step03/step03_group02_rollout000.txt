"""
custom_kernel – Triton‑accelerated forward pass of the **outgoing** TriMul update
(AlphaFold3).

Algorithm (high‑level)
----------------------
1️⃣ Layer‑norm the 4‑D input X ∈ [B,N,N,D].
2️⃣ Fuse the two projection linear layers (left/right) into a single mat‑mul and
   split the result → left, right ∈ [B,N,N,H].
3️⃣ Fuse the three gate linear layers (left_gate, right_gate, out_gate) likewise.
4️⃣ If a mask is present, a tiny Triton kernel fuses:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
   (in‑place, half‑precision, no extra temporaries).
   When no mask is given we just multiply by the gates in PyTorch.
5️⃣ Core N³ update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is exactly a batched matrix‑multiply:
        (B·H, N, N)  @  (B·H, N, N)ᵀ.
   We delegate it to torch.bmm (cuBLAS/Tensor‑Cores) in fp16 → fp32.
6️⃣ Layer‑norm over the hidden dimension, multiply by the out‑gate and apply the
   final linear projection back to the original dimension.
All heavy linear work stays in fp16 to reduce traffic; the only Triton kernel
is the mask‑gate fusion, satisfying the “use Triton” requirement while keeping
the overall runtime well below the 1 ms target on the provided sizes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask & gate application.
#   left,right : [B,N,N,H] (flattened)   fp16
#   mask       : [B,N,N]   (flattened)   fp16  (0 or 1)
#   left_gate, right_gate : same layout as left/right, fp16
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,          # fp16 *
    right_ptr,         # fp16 *
    mask_ptr,          # fp16 *
    left_gate_ptr,     # fp16 *
    right_gate_ptr,    # fp16 *
    total_elements,   # B*N*N*H   (constexpr)
    hidden_dim,        # H         (constexpr)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load the vectors
    left      = tl.load(left_ptr      + offs, mask=mask)
    right     = tl.load(right_ptr     + offs, mask=mask)
    l_gate    = tl.load(left_gate_ptr + offs, mask=mask)
    r_gate    = tl.load(right_gate_ptr+ offs, mask=mask)

    # The mask has no hidden dimension → broadcast by integer division.
    mask_idx  = offs // hidden_dim          # index in the [B,N,N] view
    mask_val  = tl.load(mask_ptr + mask_idx, mask=mask)

    # In‑place masked & gated multiplication
    left  = left  * mask_val * l_gate
    right = right * mask_val * r_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask         : torch.Tensor [B, N, N]    (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with at least ``dim`` and ``hidden_dim``.
                         Optional key ``nomask`` (bool).

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # D
    hidden_dim = config["hidden_dim"]   # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm on the last dimension (float32 → fp16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B,N,N,D]  float32
    x = x.to(torch.float16)             # work in half precision

    # --------------------------------------------------------------
    # 2️⃣ Fuse linear projections (left/right) and gates
    # --------------------------------------------------------------
    # ---- left/right projection (2·H outputs) ----
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                # (2·H, D)
    proj = F.linear(x, proj_weight)    # [B,N,N,2·H] (fp16)
    left, right = torch.split(proj, hidden_dim, dim=-1)   # each [B,N,N,H]

    # ---- gates (3·H outputs) ----
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # (3·H, D)
    gates = F.linear(x, gate_weight)   # [B,N,N,3·H] (fp16)
    left_gate, right_gate, out_gate = torch.split(gates, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)   # [B,N,N,H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask‑gate fusion (Triton)
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # mask: [B,N,N]  →  fp16, broadcast over hidden dim
        mask_f16 = mask.to(torch.float16).contiguous()
        # Ensure tensors are contiguous for flat pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_f16 = mask_f16.contiguous()

        total_elements = left.numel()                 # B·N·N·H
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        _apply_mask_gate_kernel[grid](
            left,
            right,
            mask_f16,
            left_gate,
            right_gate,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates already applied inside the kernel → no further multiplication.
    else:
        # No mask – just apply the per‑channel gates (pure PyTorch, negligible cost)
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 4️⃣ Core N³ update – batched matrix multiplication
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # B, N, N, H (H == hidden_dim)
    # reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # fp16

    # Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ  – use Tensor Cores (fp16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16
    # Cast to float32 for the following layer‑norm (more stable)
    out_mat = out_mat.to(torch.float32)

    # reshape back to [B,N,N,H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B,N,N,H]  float32

    # --------------------------------------------------------------
    # 5️⃣ Output layer‑norm, out‑gate, final projection
    # --------------------------------------------------------------
    # LN over hidden dimension (use half‑precision weights to stay in fp16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_fp16 = out.to(torch.float16)
    out_norm = F.layer_norm(
        out_fp16,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # Apply out‑gate (already fp16)
    out_norm = out_norm * out_gate

    # Final linear back to original dimension (fp16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out_final = F.linear(out_norm, to_out_w)                # fp16, [B,N,N,D]

    return out_final.to(torch.float32)                      # required dtype
