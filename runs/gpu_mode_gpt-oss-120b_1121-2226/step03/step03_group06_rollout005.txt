    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)
    # 1. LN
    x = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # 2. Cast to half
    x_h = x.half()
    
    # 3. Linear projections and gates (fused in one pass)
    # Using torch.nn.functional.linear with concatenated weights
    # Create fused weight matrix
    # On first call, we could precompute fused weight, but we will compute each call.
    left_proj_w = weights['left_proj.weight'].half()
    right_proj_w = weights['right_proj.weight'].half()
    left_gate_w = weights['left_gate.weight'].half()
    right_gate_w = weights['right_gate.weight'].half()
    out_gate_w = weights['out_gate.weight'].half()
    
    fused_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)  # (5*hidden_dim, dim)
    
    # Flatten input
    B, N, _, _ = x_h.shape
    flat_input = x_h.view(-1, dim)  # (B*N*N, dim)
    
    # Linear (half precision)
    fused_out = torch.nn.functional.linear(flat_input, fused_weight)  # (B*N*N, 5*hidden_dim)
    
    # Split
    hidden = hidden_dim
    left_proj = fused_out[:, 0:hidden]
    right_proj = fused_out[:, hidden:2*hidden]
    left_gate = fused_out[:, 2*hidden:3*hidden]
    right_gate = fused_out[:, 3*hidden:4*hidden]
    out_gate = fused_out[:, 4*hidden:5*hidden]

    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()

"""
Triton‑accelerated outgoing TriMul (AlphaFold3 style)

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32).
2.  Cast to fp16 for the heavy path.
3.  Fuse the five linear layers (left_proj, right_proj, left_gate,
    right_gate, out_gate) into a single mat‑mul with a weight matrix of
    shape (5·hidden_dim, dim).  This reduces the number of passes over the
    large tensor [B, N, N, dim] from five to one and therefore cuts memory
    traffic dramatically.
4.  Apply sigmoid to the three gate outputs, optionally mask the two
    projection outputs and fuse the gates:
        left  = (x·W_left)   * mask * sigmoid(x·W_left_gate)
        right = (x·W_right)  * mask * sigmoid(x·W_right_gate)
        out_gate = sigmoid(x·W_out_gate)
5.  Reshape left/right to (B·H, N, N) (H = hidden_dim) and perform the
    cubic update
        out(i,j) = Σ_k left(i,k) * right(j,k)
    i.e. a batched matrix‑multiply A @ Bᵀ.  For N ≤ 256 a small Triton
    kernel is used; otherwise we fall back to cuBLAS’s highly‑optimised
    batched GEMM via ``torch.bmm`` (Tensor‑Core fp16 → fp32 accumulation).
6.  Reshape back to [B, N, N, H] (float32) and apply a second LayerNorm.
7.  Multiply by the output gate and finish with the final linear projection
    back to the original feature dimension (float32).

All heavy arithmetic is performed in fp16 with fp32 accumulation,
the final tensor is float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Small‑N batched matmul kernel (A @ Bᵀ) – kept for completeness.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch (here = hidden channel)

    # ------------------------------------------------------------------
    # Offsets for the current M/N tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulate

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 64,
    BLOCK_N: int = 64,
    BLOCK_K: int = 32,
) -> torch.Tensor:
    """
    Batched matmul A @ Bᵀ via Triton.
    A, B : (batch, M, K) and (batch, N, K) – dtype torch.float16
    Returns C : (batch, M, N) – dtype torch.float32
    """
    batch, M, K = A.shape
    _, N, _ = B.shape

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return C

# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : mapping from layer names to weight tensors (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy path
    # ------------------------------------------------------------------
    x_h = x.half()                        # [B, N, N, dim] – fp16
    B, N, _, _ = x_h.shape                # B, seq_len, seq_len, dim

    # ------------------------------------------------------------------
    # 3) Fuse the five linear layers (left/right projections + 3 gates)
    #    into a single mat‑mul: (B·N·N, dim) @ (dim, 5·hidden_dim)ᵀ
    # ------------------------------------------------------------------
    # Convert all required weights to fp16 once
    w_left   = weights["left_proj.weight"].half()
    w_right  = weights["right_proj.weight"].half()
    w_lgate  = weights["left_gate.weight"].half()
    w_rgate  = weights["right_gate.weight"].half()
    w_outgate = weights["out_gate.weight"].half()

    # Concatenate along the output‑feature axis (dim 0)
    fused_w = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0)  # (5*H, dim)

    # Flatten the input to (B·N·N, dim)
    x_flat = x_h.view(-1, dim)               # (B*N*N, dim)

    # Perform the fused linear (fp16 → fp16)
    fused_out = torch.nn.functional.linear(x_flat, fused_w)   # (B*N*N, 5*H) in fp16

    # ------------------------------------------------------------------
    # 4) Split and apply sigmoid to gates
    # ------------------------------------------------------------------
    H = hidden_dim
    left_proj   = fused_out[:, 0:H]
    right_proj  = fused_out[:, H:2*H]
    left_gate   = fused_out[:, 2*H:3*H].sigmoid()
    right_gate  = fused_out[:, 3*H:4*H].sigmoid()
    out_gate    = fused_out[:, 4*H:5*H].sigmoid()

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flattened → [B·N·N, 1] for broadcasting
        m = mask.to(x_h.dtype).view(-1, 1)     # (B*N*N, 1)
        left_proj  = left_proj * m
        right_proj = right_proj * m

    # ------------------------------------------------------------------
    # 6) Fuse element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate            # (B*N*N, H)   fp16
    right = right_proj * right_gate          # (B*N*N, H)   fp16

    # Reshape back to 4‑D tensors
    left  = left.view(B, N, N, H)            # [B, N, N, H] fp16
    right = right.view(B, N, N, H)           # [B, N, N, H] fp16
    out_gate = out_gate.view(B, N, N, H)    # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7) Core cubic update: Batched matmul A @ Bᵀ  (A=left, B=right)
    # ------------------------------------------------------------------
    # Prepare (B·H, N, N) tensors for the batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # For relatively small N we keep the Triton implementation; otherwise
    # cuBLAS (via torch.bmm) is faster.
    if N <= 256:
        # Triton kernel (fp16·fp16 → fp32)
        out_mat = _batched_matmul_triton(
            left_mat,
            right_mat,
            BLOCK_M=64,
            BLOCK_N=64,
            BLOCK_K=32,
        )  # shape (B*H, N, N), dtype float32
    else:
        # cuBLAS batched GEMM – tensor‑core accelerated
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).float()  # (B*H, N, N)

    # Restore the original 4‑D layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] float32

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # float32

    # ------------------------------------------------------------------
    # 9) Output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.float()                     # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])    # [B, N, N, dim] float32

    return out
