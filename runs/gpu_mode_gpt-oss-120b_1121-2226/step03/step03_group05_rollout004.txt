@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(2)
    block_m = tl.program_id(0)
    block_n = tl.program_id(1)

    # Compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to each batch base
    A_base = A_ptr + batch_id * stride_Ab
    B_base = B_ptr + batch_id * stride_Bb
    C_base = C_ptr + batch_id * stride_Cb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        # Offsets for K dimension
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A sub-tile (BLOCK_M, BLOCK_K)
        a_ptrs = A_base + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B sub-tile (BLOCK_K, BLOCK_N) because B is transposed (shape K x N)
        # B original shape (batch, N, K) but we treat as (K,N) via transposed stride order.
        b_ptrs = B_base + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product under the hood
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # Store result
    c_ptrs = C_base + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) forward with Triton acceleration
-------------------------------------------------
The kernel performs the heavy N³ update:

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

as a batched GEMM (left @ rightᵀ) over the hidden dimension *h*.
All cheaper element‑wise work (mask & gate fusion) is also done in
Triton.  The rest of the pipeline (LayerNorm, linear projections,
sigmoid gates, final projection) uses PyTorch but runs in fp16 to
minimise memory traffic.  The implementation is fully mixed‑precision:
inputs are fp32, the bulk of the work is in fp16 with fp32 accumulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Fuse mask, left‑gate and right‑gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,    # fp16*
    left_gate_ptr,
    right_ptr,
    right_gate_ptr,
    mask_ptr,    # fp16* (0.0 / 1.0)
    total_elems: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    l_val   = tl.load(left_ptr   + offs, mask=mask, other=0.0, dtype=tl.float16)
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    r_val   = tl.load(right_ptr  + offs, mask=mask, other=0.0, dtype=tl.float16)
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    m_val   = tl.load(mask_ptr   + offs, mask=mask, other=1.0, dtype=tl.float16)

    tl.store(left_ptr  + offs, l_val * l_gate * m_val, mask=mask)
    tl.store(right_ptr + offs, r_val * r_gate * m_val, mask=mask)


# ----------------------------------------------------------------------
# 2️⃣ Batched GEMM kernel: out = left @ rightᵀ  (fp16 × fp16 → fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    stride_Ab, stride_Am, stride_Ak,          # A batch/row/col strides
    stride_Bb, stride_Bk, stride_Bn,          # B batch/row/col strides (B is accessed transposed)
    stride_Cb, stride_Cm, stride_Cn,          # C batch/row/col strides
    M, N, K,                                  # problem size (all equal to seq_len)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_z = tl.program_id(2)          # batch index  (0 … B*H‑1)
    pid_m = tl.program_id(0)          # tile over M dimension
    pid_n = tl.program_id(1)          # tile over N dimension

    # ---- compute offsets within the tile ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ---- base pointers for this batch ------------------------------------
    a_base = A_ptr + pid_z * stride_Ab
    b_base = B_ptr + pid_z * stride_Bb
    c_base = C_ptr + pid_z * stride_Cb

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension -------------------------------------------
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)

        # Bᵀ tile: (BLOCK_K, BLOCK_N)   (note the transposed layout)
        b_ptrs = b_base + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        # fp16 × fp16 → fp32 dot product (Tensor‑Core friendly)
        acc += tl.dot(a, b)

    # ---- store the result -------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]      (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → fp32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for all heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections & gates (fp16)
    # ------------------------------------------------------------------
    left = F.linear(x_f16, weights["left_proj.weight"].half())          # [B,N,N,H]
    right = F.linear(x_f16, weights["right_proj.weight"].half())

    left_gate = torch.sigmoid(F.linear(x_f16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_f16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_f16, weights["out_gate.weight"].half()))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask (if present) with the element‑wise gate products
    # ------------------------------------------------------------------
    if not nomask:
        # mask → fp16 and broadcast on the hidden dimension
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]

        # make everything contiguous for the 1‑D view used by the kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_f16 = mask_f16.contiguous()

        total = left.numel()                     # B·N·N·H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK, )
        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # no mask → just multiply by the gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Reshape to (B·H, N, N) and launch the batched GEMM kernel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # left.shape = [B,N,N,H]
    # (B, H, N, N) → (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # output buffer (fp32)
    out_mat = torch.empty((B * H, N, N), dtype=torch.float32, device=device)

    # choose tile sizes (tuned for H100)
    if N <= 256:
        BLOCK_M = 32
        BLOCK_N = 32
    else:
        BLOCK_M = 64
        BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,         # tiles over M
        (N + BLOCK_N - 1) // BLOCK_N,         # tiles over N
        B * H,                                 # batch dimension
    )

    _batched_matmul_kernel[grid](
        left_mat.data_ptr(),
        right_mat.data_ptr(),
        out_mat.data_ptr(),
        # strides (elements, not bytes)
        left_mat.stride(0),   # batch stride for A
        left_mat.stride(1),   # row stride  (M)
        left_mat.stride(2),   # col stride  (K)
        right_mat.stride(0),  # batch stride for B
        right_mat.stride(2),  # Bᵀ row stride → original col stride (1)
        right_mat.stride(1),  # Bᵀ col stride → original row stride (N)
        out_mat.stride(0),    # batch stride for C
        out_mat.stride(1),    # row stride
        out_mat.stride(2),    # col stride
        N, N, N,              # M, N, K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # reshape back to the original 4‑D layout
    out = (
        out_mat
        .view(B, H, N, N)                      # [B, H, N, N]
        .permute(0, 2, 3, 1)                    # [B, N, N, H]
        .contiguous()
    )   # fp32

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + out‑gate + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate is fp16; broadcasting/up‑casting to fp32 is automatic
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # project back to dim (float32)

    return out
