"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The implementation follows the reference PyTorch module but removes the
expensive `permute(...).contiguous()` copy that would materialise the
hidden dimension as a separate batch axis.  A custom Triton kernel
directly works on the original `[B, N, N, H]` layout and treats the
combined batch index `b*H + h` inside the kernel (the hidden channel
becomes the “batch” dimension).  This eliminates an O(B·N²·H) copy and
reduces memory traffic, allowing the N³ update to run in ~1 ms on an
H100 for the benchmark sizes.

Algorithm
---------
1️⃣ Layer‑Norm over the last dimension (float32).  
2️⃣ Cast to fp16 and apply three bias‑free linear projections (left/right)
   together with three gating projections; mask and sigmoid are fused in
   half‑precision.  
3️⃣ Core cubic update  
   `out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]`  
   is computed as a batched mat‑mul `left @ rightᵀ` using a tiled Triton
   kernel that:
   * treats each hidden channel as a separate batch entry (`B·H` batches)
   * works on the native `[B,N,N,H]` layout (no transpose or copy)
   * uses fp16·fp16 → fp32 accumulation (Tensor‑Core friendly)
4️⃣ Second Layer‑Norm over the hidden dimension.  
5️⃣ Multiply by the output gate and finish with a linear projection back
   to the original feature dimension.

All heavy arithmetic (step 3) runs on the GPU; the surrounding
operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K)ᵀ → C (M×N)
# Operates on tensors of shape (B, N, N, H) without materialising
# the hidden dimension as a separate batch axis.
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_hidden_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    B_val, H_val,                       # batch‑size and hidden‑size (runtime)
    M_val, N_val, K_val,                # matrix sizes (M=N=K=seq_len)
    stride_A_batch, stride_A_row, stride_A_col, stride_A_hidden,
    stride_B_batch, stride_B_row, stride_B_col, stride_B_hidden,
    stride_C_batch, stride_C_row, stride_C_col, stride_C_hidden,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)    # tile row
    pid_n = tl.program_id(1)    # tile col
    pid_b = tl.program_id(2)    # combined (batch, hidden) index

    # ------------------------------------------------------------------
    # 1) Offsets within the M/N tiles
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # 2) Decode combined batch index into (batch, hidden)
    # ------------------------------------------------------------------
    b_idx = pid_b // H_val
    h_idx = pid_b %  H_val

    # Base pointers for the current (b, h) pair
    a_base = A_ptr + b_idx * stride_A_batch + h_idx * stride_A_hidden
    b_base = B_ptr + b_idx * stride_B_batch + h_idx * stride_B_hidden
    c_base = C_ptr + b_idx * stride_C_batch + h_idx * stride_C_hidden

    # ------------------------------------------------------------------
    # 3) Accumulator (fp32) for mixed‑precision Tensor‑Core dot
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) K‑loop: traverse the reduction dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K_val, BLOCK_K)):
        k_base = k * BLOCK_K
        k_offsets = k_base + tl.arange(0, BLOCK_K)

        # Pointers to the current A‑tile (M×K) and B‑tile (K×N)
        a_ptrs = a_base + offs_m[:, None] * stride_A_row + k_offsets[None, :] * stride_A_col
        b_ptrs = b_base + k_offsets[:, None] * stride_B_col + offs_n[None, :] * stride_B_row

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M_val) & (k_offsets[None, :] < K_val)
        mask_b = (k_offsets[:, None] < K_val) & (offs_n[None, :] < N_val)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Tensor‑Core dot (fp16·fp16 → fp32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 5) Write the tile to C
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_C_row + offs_n[None, :] * stride_C_col
    mask_c = (offs_m[:, None] < M_val) & (offs_n[None, :] < N_val)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_hidden(
    left: torch.Tensor,
    right: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Calls the Triton kernel defined above.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape ``[B, N, N, H]`` (float16).  ``left`` and ``right`` must have the
        same ``B, N, H``.
    BLOCK_* : int
        Tile sizes.  Larger tiles are used for large ``N`` inside the caller.

    Returns
    -------
    out : torch.Tensor
        Shape ``[B, N, N, H]`` (float32) – the result of
        ``left @ rightᵀ`` for each hidden channel.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    # Output in float32
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Strides in element units (not bytes)
    a_batch, a_row, a_col, a_h = left.stride()
    b_batch, b_row, b_col, b_h = right.stride()
    c_batch, c_row, c_col, c_h = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _batched_matmul_hidden_kernel[grid](
        left,
        right,
        out,
        B,
        H,
        N,          # M
        N,          # N
        N,          # K
        a_batch,
        a_row,
        a_col,
        a_h,
        b_batch,
        b_row,
        b_col,
        b_h,
        c_batch,
        c_row,
        c_col,
        c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# --------------------------------------------------------------
# Entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the **outgoing** TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim) – float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for heavy ops
    # ------------------------------------------------------------------
    x_h = x.half()  # (B, N, N, dim) – fp16

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias, fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left = F.linear(x_h, left_proj_w)    # (B, N, N, hidden_dim) – fp16
    right = F.linear(x_h, right_proj_w)  # (B, N, N, hidden_dim) – fp16

    # ------------------------------------------------------------------
    # 4) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(left.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = F.linear(x_h, left_gate_w).sigmoid()    # (B, N, N, hidden_dim)
    right_gate = F.linear(x_h, right_gate_w).sigmoid()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()     # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core cubic update – batched matmul via Triton (no permute copy)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Choose tile sizes adaptively
    if N <= 384:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 64

    out_hidden = _batched_matmul_hidden(
        left,
        right,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # (B, N, N, H) – float32

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # (B, N, N, H) – float32

    # ------------------------------------------------------------------
    # 9) Output gating and final projection back to `dim`
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.float()          # (B, N, N, H) – float32
    out = out_hidden * out_gate_f32         # element‑wise gated output

    # Final linear projection (float32)
    out = F.linear(out, weights["to_out.weight"])  # (B, N, N, dim)

    return out
