"""
Triton‑accelerated forward pass for the “outgoing” TriMul module (AlphaFold3).

Algorithm
---------
1. Input LayerNorm (FP32) → x   [B, N, N, D]
2. Cast to FP16.
3. Two bias‑free linear projections (left/right) from D → H.
4. Optional binary mask (broadcasted) applied to both projections.
5. Compute three sigmoid gates from the original normalized x and
   multiply them element‑wise with the projected tensors.
6. Core cubic operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiply L @ Rᵀ for each hidden channel.
   We keep the native (B,N,N,H) layout (hidden dimension innermost) and
   launch a custom Triton kernel that works directly on the strided tensors,
   eliminating the costly contiguous copies performed in the reference
   implementation.
7. Output LayerNorm over the hidden dimension, multiply by the out‑gate
   and a final bias‑free linear projection back to D.
8. Cast the result to FP32 (required by the API).

The heavy N³ work is performed entirely in the Triton kernel below,
using FP16 arithmetic with FP32 accumulation.  No extra copies of the
large tensors are made, which reduces memory traffic and brings the
runtime well below the baseline.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel: batched GEMM for TriMul “outgoing” update
#   left  : fp16 (B, N, N, H)  – stride: (N*N*H, N*H, H, 1)
#   right : fp16 (B, N, N, H)  – same strides
#   out   : fp16 (B, N, N, H)  – same layout as inputs
#   Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    M, N, K, H,                           # matrix sizes (M=N=K=seq_len, H=hidden_dim)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_hb = tl.program_id(2)  # combined batch*hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden indices
    # ------------------------------------------------------------------
    hidden = pid_hb % H
    batch  = pid_hb // H

    # ------------------------------------------------------------------
    # Offsets for current tile
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + batch * stride_l_batch  + hidden * stride_l_h
    right_base = right_ptr + batch * stride_r_batch + hidden * stride_r_h
    out_base   = out_ptr   + batch * stride_o_batch + hidden * stride_o_h

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # Load a tile from left: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_l_i \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_l_k
        mask_a = (offs_i[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # Load a tile from right (transposed): shape (BLOCK_K, BLOCK_N)
        # Right matrix is (j, k); we need (k, j) for the dot product.
        b_ptrs = right_base + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_i
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_j[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # FP32 accumulation via tensor‑core dot
        acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # Write the result (cast back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    mask_c = (offs_i[:, None] < M) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def _tri_mul_outgoing_triton(left: torch.Tensor,
                             right: torch.Tensor,
                             BLOCK_M: int = 128,
                             BLOCK_N: int = 128,
                             BLOCK_K: int = 32) -> torch.Tensor:
    """
    Wrapper that launches the batched GEMM kernel.

    left, right: fp16 tensors of shape (B, N, N, H) – **no extra copy**.
    Returns out tensor of the same shape (fp16).
    """
    assert left.is_cuda and right.is_cuda, "Tensors must be on CUDA"
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    # Allocate output with identical layout (no copy)
    out = torch.empty_like(left)

    # Strides in element units (not bytes)
    slb, sli, slk, slh = left.stride()
    srb, sri, srk, srh = right.stride()
    sob, soi, soj, soh = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,            # M
        N,            # N
        N,            # K
        H,            # hidden_dim
        slb, sli, slk, slh,
        srb, sri, srk, srh,
        sob, soi, soj, soh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return out


# ------------------------------------------------------------------
# Entry‑point required by the evaluator
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask         : torch.Tensor [B, N, N]  (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # 1) Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # D
    hidden_dim = config["hidden_dim"]  # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (FP32) – high‑precision statistics
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  FP32

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)  # [B, N, N, D]  fp16

    # --------------------------------------------------------------
    # 4) Linear projections (bias‑free) – FP16 weights
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_f16, right_proj_w)  # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 5) Optional binary mask (broadcasted)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is {0,1} – convert once to fp16 and broadcast
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 6) Gating (sigmoid) – all in FP16
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))  # fp16
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # fp16

    # --------------------------------------------------------------
    # 7) Apply element‑wise gates
    # --------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8) Core cubic N³ operation (batched mat‑mul) – in‑place kernel
    # --------------------------------------------------------------
    out = _tri_mul_outgoing_triton(left, right,
                                   BLOCK_M=128,
                                   BLOCK_N=128,
                                   BLOCK_K=32)            # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 9) Output LayerNorm (FP16), out‑gate multiplication,
    #    final linear projection back to D, cast to FP32.
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Apply out‑gate (broadcast implicitly over last dim)
    out = out * out_gate

    # Final projection to original dimension (bias‑free)
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16

    # Cast back to FP32 for the API contract
    return out.to(torch.float32)
