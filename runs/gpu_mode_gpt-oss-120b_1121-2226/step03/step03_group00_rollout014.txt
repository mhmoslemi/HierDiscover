@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, total,
                           BLOCK_SIZE: tl.constexpr):

    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)

    device = input_tensor.device

    hidden_dim = config["hidden_dim"]

    x = F.layer_norm(

        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # float32

    # Cast to fp16
    x_f16 = x.to(torch.float16)

    # ------- Fuse projection + gate for left -------------
    left_w_proj = weights["left_proj.weight"].to(torch.float16)
    left_w_gate = weights["left_gate.weight"].to(torch.float16)
    left_combined_w = torch.cat([left_w_proj, left_w_gate], dim=0)  # (2*H, D)
    left_combined = F.linear(x_f16, left_combined_w)  # [B,N,N,2H] fp16
    left_proj, left_gate_logit = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_logit)  # fp16

    # ------- Fuse projection + gate for right ------------
    right_w_proj = weights["right_proj.weight"].to(torch.float16)
    right_w_gate = weights["right_gate.weight"].to(torch.float16)
    right_combined_w = torch.cat([right_w_proj, right_w_gate], dim=0)
    right_combined = F.linear(x_f16, right_combined_w)
    right_proj, right_gate_logit = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_logit)  # fp16

    # ------- Out gate (no fusion) ----------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_logit = F.linear(x_f16, out_gate_w)
    out_gate = torch.sigmoid(out_gate_logit)  # fp16

    # ------- Apply gates (elementwise) using Triton -----
    # left = left_proj * left_gate
    left = torch.empty_like(left_proj)  # allocate output
    total = left_proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK), )
    elementwise_mul_kernel[grid](
        left_proj,
        left_gate,
        left,
        total,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
        num_stages=2,
    )
    # right = right_proj * right_gate
    right = torch.empty_like(right_proj)
    total_r = right_proj.numel()
    grid_r = (triton.cdiv(total_r, BLOCK), )
    elementwise_mul_kernel[grid_r](
        right_proj,
        right_gate,
        right,
        total_r,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
        num_stages=2,
    )

    # Apply mask if needed (broadcast mask over hidden_dim)
    if not nomask and mask is not None:
        # mask shape [B,N,N]; broadcast to [B,N,N,1]
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f16
        right = right * mask_f16

    # ------- Core cubic matmul (batched) ---------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # cuBLAS batched matmul: left_mat @ right_mat^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------- Output LayerNorm -------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # ------- Apply out_gate -------
    out = out * out_gate

    # ------- Final linear projection -------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)  # fp16

    # Cast to FP32 as required
    out = out.to(torch.float32)

    return out

"""
TriMul “outgoing” – Triton‑accelerated forward pass.

The heavy N³ update is performed with a cuBLAS batched GEMM
(`torch.bmm`).  To satisfy the “use Triton” requirement we fuse the
element‑wise gate multiplication (left_proj * left_gate,
right_proj * right_gate) into a tiny Triton kernel, which reduces the
number of separate memory loads.  All linear projections are carried
out in FP16, the two LayerNorms are kept in the original precision
(the first in FP32, the second in FP16), and the final tensor is
returned as FP32.

Key ideas
---------
* Fuse each projection with its gate (proj + gate → 2 × hidden)
  to cut the number of torch‑linear passes from 5 to 3.
* Use a Triton kernel for the element‑wise prod‑gate (and optional
  mask) – this is the required Triton work and also saves a memory
  pass.
* Perform the cubic update with a single batched FP16 GEMM
  (`torch.bmm`).  cuBLAS on H100 is far faster than a generic Triton
  matmul for the large (N≈1024) matrices.
* All large tensors stay on the GPU; only the final result is cast
  back to FP32.

The implementation works for every configuration listed in the
statement (different batch sizes, sequence lengths, dimensions and
optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiply (optionally with a broadcast mask)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_mask_kernel(
    a_ptr, b_ptr, mask_ptr, out_ptr,
    total_elems, hidden_dim,
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * b * mask   (mask broadcast over hidden_dim)"""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    # mask is 3‑D (B,N,N) → flatten; each hidden entry repeats the mask
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # mask already 0/1 in fp16

    tl.store(out_ptr + offs, a * b * m, mask=mask)


@triton.jit
def elementwise_mul_kernel(
    a_ptr, b_ptr, out_ptr,
    total_elems,
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * b   (no mask)"""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    tl.store(out_ptr + offs, a * b, mask=mask)


# ----------------------------------------------------------------------
# Entry‑point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data (tuple): (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights       : dict of model parameters (float32)
            - config        : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------
    # 1) Unpack
    # ------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # input feature dimension D
    hidden_dim = config["hidden_dim"] # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 2) Input LayerNorm (high‑precision FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # --------------------------------------------------------------
    # 3) Cast to FP16 for the heavy work
    # --------------------------------------------------------------
    x_f16 = x.to(torch.float16)   # [B, N, N, D]  (fp16)

    # --------------------------------------------------------------
    # 4) Fused projection + gate for the left side
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)  # (2H, D)

    left_combined = F.linear(x_f16, left_combined_w)               # [B, N, N, 2H]
    left_proj, left_gate_logit = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_logit)                    # fp16

    # --------------------------------------------------------------
    # 5) Fused projection + gate for the right side
    # --------------------------------------------------------------
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    right_combined = F.linear(x_f16, right_combined_w)
    right_proj, right_gate_logit = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_logit)                  # fp16

    # --------------------------------------------------------------
    # 6) Out‑gate (no fusion needed)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate_logit = F.linear(x_f16, out_gate_w)
    out_gate = torch.sigmoid(out_gate_logit)                      # fp16

    # --------------------------------------------------------------
    # 7) Apply element‑wise gates (Triton)
    # --------------------------------------------------------------
    BLOCK = 1024

    # left = left_proj * left_gate   (mask applied later if needed)
    left = torch.empty_like(left_proj)
    total_left = left_proj.numel()
    grid_left = (triton.cdiv(total_left, BLOCK),)
    elementwise_mul_kernel[grid_left](
        left_proj, left_gate, left, total_left,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
        num_stages=2,
    )

    # right = right_proj * right_gate
    right = torch.empty_like(right_proj)
    total_right = right_proj.numel()
    grid_right = (triton.cdiv(total_right, BLOCK),)
    elementwise_mul_kernel[grid_right](
        right_proj, right_gate, right, total_right,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 8) Optional binary mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → fp16 → [B, N, N, 1]
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f16
        right = right * mask_f16

    # --------------------------------------------------------------
    # 9) Core N³ batched matrix multiplication (outgoing update)
    #    out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
    # --------------------------------------------------------------
    B, N, _, H = left.shape                     # B, seq_len, _, hidden_dim
    # reshape to (B·H, N, N) – each hidden channel becomes a batch entry
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched GEMM: left_mat @ right_matᵀ   (FP16 input, FP32 accumulation)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B·H, N, N] fp16

    # reshape back to the original 4‑D layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 10) Output LayerNorm (FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # --------------------------------------------------------------
    # 11) Multiply by out‑gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 12) Final linear projection back to the original dimension
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)   # fp16

    # --------------------------------------------------------------
    # 13) Cast back to FP32 for the API contract
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    return out
