def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # norm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to half for heavy part
    x_h = x.half()

    # optional mask
    if not nomask:
        mask_h = mask.unsqueeze(-1).half()
    else:
        mask_h = None

    # Concatenated projection+gate weights
    left_cat_w = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).half()
    right_cat_w = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).half()
    out_gate_w = weights["out_gate.weight"].half()
    # Linear concatenated
    left_cat = F.linear(x_h, left_cat_w)  # shape (..., 2*hidden_dim)
    right_cat = F.linear(x_h, right_cat_w)
    # split
    left_proj = left_cat[..., :hidden_dim]
    left_gate = torch.sigmoid(left_cat[..., hidden_dim:])
    right_proj = right_cat[..., :hidden_dim]
    right_gate = torch.sigmoid(right_cat[..., hidden_dim:])
    # apply mask if present
    if mask_h is not None:
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h
    # apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # out gate
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))  # shape (..., hidden_dim)
    # core TriMul: batched gemm left @ right^T per hidden channel
    B, N, _, H = left.shape
    # permute to (B, H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # batched matmul via Triton
    out_mat = _batched_matmul_triton(left_mat, right_mat,
                                     BLOCK_M=128, BLOCK_N=128, BLOCK_K=64)
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # output layernorm over hidden dim
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])
    # gating
    out = out * out_gate.to(torch.float32)
    # final linear
    out = F.linear(out, weights["to_out.weight"])  # fp32
    return out

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3 style)

Algorithm (high‑level)
----------------------
1️⃣  FP32 Layer‑Norm on the last dimension (dim) →   x : [B,N,N,dim].
2️⃣  Cast to FP16 for all heavy arithmetic.
3️⃣  **Fused projection + gating**: a single large FP16 Linear computes
    * left_proj, left_gate, right_proj, right_gate and out_gate* in one
    pass over `x`.  This cuts the memory traffic of three separate Linear
    layers (≈3× reduction) while keeping the exact mathematical form.
4️⃣  Apply the optional mask (broadcast over the hidden dimension) **before**
    the sigmoid‑gates, then fuse the element‑wise gate multiplications.
5️⃣  Core N³ operation – for every hidden channel `h` compute
    `out_h[i,j] = Σ_k left[i,k,h]·right[j,k,h]`.  This is a batched GEMM
    `L @ Rᵀ` with FP16 inputs and FP32 accumulation, implemented with a
    custom Triton kernel (`_batched_matmul_kernel`).  Block sizes
    (128,128,64) are tuned for the H100 Tensor‑Core pipeline.
6️⃣  FP32 Layer‑Norm over the hidden dimension.
7️⃣  Multiply by the sigmoid out‑gate (cast to FP32) and apply the final
    FP32 Linear projection back to `dim`.

Only the batch‑matmul is written in Triton – the rest uses highly‑optimised
PyTorch kernels.  The whole pipeline returns a float32 tensor of shape
`[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# 2️⃣  Batched matmul kernel (FP16·FP16 → FP32 accumulation)
# ---------------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A: (batch, M, K)
    stride_bh, stride_bn, stride_bk,   # B: (batch, N, K)
    stride_ch, stride_cm, stride_cn,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)               # batch = B * hidden
    pid_m = tl.program_id(0)               # row block
    pid_n = tl.program_id(1)               # column block

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initial pointers for this block
    a_ptrs = A_ptr + pid_b * stride_ah + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_b * stride_bh + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) &
                         (tl.arange(0, BLOCK_K)[None, :] + k * BLOCK_K < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K)[:, None] + k * BLOCK_K < K) &
                         (offs_n[None, :] < N),
                    other=0.0)
        # fp16·fp16 → fp32 dot
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back result (still FP32)
    c_ptrs = C_ptr + pid_b * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ
    A, B : fp16 tensors of shape (batch, M, K) and (batch, N, K)
    Returns C : fp32 tensor of shape (batch, M, N)
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in element counts
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ---------------------------------------------------------------------------
# 1️⃣  Entry point required by the evaluation harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [bs, N, N, dim] (float32)
        - mask         : torch.Tensor [bs, N, N] (float32) – may be dummy
        - weights      : dict of model weights (float32 tensors)
        - config       : dict containing at least ``dim``, ``hidden_dim`` and optionally ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device

    dim = config["dim"]                # input feature dimension
    hidden_dim = config["hidden_dim"]  # projection dimension (H)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2️⃣  Cast to FP16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.half()

    # ------------------------------------------------------------------
    # 3️⃣  Optional mask – broadcast over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    mask_h = None
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] (fp16) and will be applied before gating
        mask_h = mask.unsqueeze(-1).half()

    # ------------------------------------------------------------------
    # 4️⃣  Fused projection + gating (single Linear)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (left_proj, left_gate,
    # right_proj, right_gate, out_gate) → shape (5*H, dim)
    weight_cat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # fp16

    # Single linear pass – output shape [B,N,N,5*H] (fp16)
    fused = F.linear(x_h, weight_cat)

    # Slice the fused tensor
    H = hidden_dim
    left_proj = fused[..., :H]                         # [B,N,N,H]
    left_gate = torch.sigmoid(fused[..., H : 2 * H])   # [B,N,N,H]
    right_proj = fused[..., 2 * H : 3 * H]             # [B,N,N,H]
    right_gate = torch.sigmoid(fused[..., 3 * H : 4 * H])  # [B,N,N,H]
    out_gate = torch.sigmoid(fused[..., 4 * H : 5 * H])    # [B,N,N,H]

    # Apply mask *before* the sigmoid gates (as in the reference)
    if mask_h is not None:
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # Apply element‑wise sigmoid gates
    left = left_proj * left_gate    # fp16
    right = right_proj * right_gate # fp16

    # ------------------------------------------------------------------
    # 5️⃣  Core N³ operation – batched matmul L @ Rᵀ (FP16 → FP32)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) for the Triton kernel
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # batched GEMM with FP32 accumulation
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # [B*H, N, N]  fp32

    # restore original layout → [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # fp32

    # ------------------------------------------------------------------
    # 6️⃣  Output LayerNorm (over hidden dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 7️⃣  Multiply by out‑gate (cast to FP32) and final projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)           # fp32 element‑wise
    out = F.linear(out, weights["to_out.weight"])   # fp32 → [B,N,N,dim]

    return out
