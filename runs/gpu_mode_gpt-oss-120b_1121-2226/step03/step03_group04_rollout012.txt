    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)  # float32

    # Cast to half
    x_h = x_norm.half()

    # Flatten

    # get weights from dict and half
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    all_weight = torch.cat([left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w], dim=0)   # shape (5*hidden_dim, dim)

    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

```python
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.
    The algorithm:
    1) Apply LayerNorm on the channel dimension (float32).
    2) Cast to fp16 and flatten -> (B·N·N, dim).
    3) Perform a single fused linear projection that computes the five branches
       (left_proj, right_proj, left_gate, right_gate, out_gate) at once.
    4) Apply sigmoid to the three gate tensors.
    5) A tiny Triton kernel fuses the optional mask, the two gates and
       simultaneously writes the projected tensors into the layout required for the
       cubic contraction: (B, hidden, N, N). The output gate is also written in this
       layout.
    6) Batched GEMM (torch.bmm) carries out the O(N³) contraction.
    7) A second LayerNorm over the hidden dimension (float32) plus output‑gate
       scaling.
    8) Final linear projection back to `dim` (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)  # (B,N,N,dim) float32

    # ------------------------------------------------------------------
    # 2) Prepare flatten input
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    M = B * N * N
    x_flat = x.view(M, dim).contiguous()  # float32

    # ------------------------------------------------------------------
    # 3) Single fused Linear (5 * hidden_dim) in fp16
    # ------------------------------------------------------------------
    # concat weight matrices (out_features = 5*hidden_dim, in_features = dim)
    lp_w = weights["left_proj.weight"].half()
    rp_w = weights["right_proj.weight"].half()
    lg_w = weights["left_gate.weight"].half()
    rg_w = weights["right_gate.weight"].half()
    og_w = weights["out_gate.weight"].half()
    all_weight = torch.cat([lp_w, rp_w, lg_w, rg_w, og_w], dim=0)  # (5*H, dim) fp16

    # cast input to fp16 for the matmul
    x_flat_h = x_flat.half()
    # fused linear
    all_out = F.linear(x_flat_h, all_weight)  # (M, 5*H) fp16

    # ------------------------------------------------------------------
    # 4) Sigmoid on gate tensors (still fp16)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Compute offsets for each section in the fused tensor.
    # We'll keep the fused tensor untouched and let the Triton kernel read from it.
    # Create an auxiliary view that adds the sigmoid for the three gate blocks.
    # The kernel will read all_out directly, so we need the sigmoid applied.
    # To avoid extra memory, we can compute the sigmoid on‑the‑fly in the kernel,
    # but a small torch call is cheaper given the size.
    # We therefore write three temporary tensors that contain the sigmoid‑ed gates.
    # Keeping everything in fp16.
    # out_gate_raw = all_out[:, 4*H:5*H]   # not yet sigmoid
    # left_gate_raw = all_out[:, 2*H:3*H]
    # right_gate_raw = all_out[:, 3*H:4*H]
    # Compute sigmoid
    left_gate = torch.sigmoid(all_out[:, 2*H:3*H])
    right_gate = torch.sigmoid(all_out[:, 3*H:4*H])
    out_gate = torch.sigmoid(all_out[:, 4*H:5*H])

    # ------------------------------------------------------------------
    # 5) Triton kernel: mask + gate fusion + layout permutation
    # ------------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        mask_h = mask.to(dtype=torch.float16).contiguous().view(-1)  # (M,)
    else:
        # dummy tensor – never read because HAS_MASK==0
        mask_h = torch.empty(0, dtype=torch.float16, device=x.device)

    # Allocate output buffers in (B, H, N, N) layout
    left_buf = torch.empty((B, H, N, N), dtype=torch.float16, device=x.device)
    right_buf = torch.empty((B, H, N, N), dtype=torch.float16, device=x.device)
    out_gate_buf = torch.empty((B, H, N, N), dtype=torch.float16, device=x.device)

    total_elems = M * H
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    _mask_gate_permute_kernel[grid](
        all_out,                # fused tensor (M,5H)
        mask_h,
        left_buf,
        right_buf,
        out_gate_buf,
        M, N, H,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 6) Batched GEMM (B*H, N, N) @ (B*H, N, N)
    # ------------------------------------------------------------------
    BH = B * H
    left_mat = left_buf.view(BH, N, N)
    right_mat = right_buf.view(BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N) fp16

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over hidden dim (float32)
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and cast to float32
    out_f = out_mat.view(B, H, N, N).float()

    # compute mean/var over hidden dim (dim=1)
    mean = out_f.mean(dim=1, keepdim=True)
    var = out_f.var(dim=1, unbiased=False, keepdim=True)
    inv_std = torch.rsqrt(var + eps)

    # to_out_norm parameters (float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # reshape for broadcasting
    scale = to_out_norm_w.view(1, H, 1, 1)
    bias = to_out_norm_b.view(1, H, 1, 1)

    out_norm = (out_f - mean) * inv_std * scale + bias

    # Apply output gate (fp16->float)
    out_gate_f = out_gate_buf.float()
    out_norm = out_norm * out_gate_f

    # ------------------------------------------------------------------
    # 8) Final projection (float32)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) for linear
    out_perm = out_norm.permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    to_out_w = weights["to_out.weight"]
    out = F.linear(out_perm, to_out_w)  # (B,N,N,dim) float32

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module but fuses the
expensive parts into custom kernels:

1️⃣  Layer‑Norm over the last channel dimension (float32).
2️⃣  Cast to fp16 and flatten to (B·N·N, dim).
3️⃣  **One** fused linear projection computes the five branches
    (left_proj, right_proj, left_gate, right_gate, out_gate) in a
    single GEMM.
4️⃣  A tiny Triton kernel
    – loads the fused tensor,
    – applies the three *sigmoid* gates,
    – (optionally) multiplies by the mask,
    – stores the gated tensors directly into the layout required for the
      cubic contraction: (B, hidden, N, N).  The output‑gate is stored in
      the same layout.
5️⃣  Batched GEMM (torch.bmm) carries out the O(N³) contraction.
6️⃣  Second Layer‑Norm over the hidden dimension (float32) together with
    the learned scale/bias, followed by element‑wise multiplication with
    the output‑gate.
7️⃣  Final linear projection back to `dim` (float32).

Only the mask‑and‑gate fusion is written in Triton – the remaining
operations use highly‑optimised PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_permute_kernel(
    all_out_ptr,               # fp16 * (M, 5*H)
    mask_ptr,                  # fp16 * (M)    (optional)
    left_out_ptr,              # fp16 * (B, H, N, N)
    right_out_ptr,             # fp16 * (B, H, N, N)
    out_gate_out_ptr,          # fp16 * (B, H, N, N)
    M, N, H,                   # runtime scalars
    total_elements,            # M*H
    BLOCK_SIZE: tl.constexpr,  # compile‑time block size
    HAS_MASK: tl.constexpr,    # 1 → mask present, 0 → no mask
):
    """Fuse mask, three sigmoid gates and permute to (B,H,N,N)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    in_range = offs < total_elements

    # ------------------------------------------------------------------
    # Decode linear index -> (m, h)  where   m ∈ [0, M) , h ∈ [0, H)
    # ------------------------------------------------------------------
    h = offs % H
    m = offs // H

    # ------------------------------------------------------------------
    # Decode m -> (b, i, k)  with  N² = N * N
    # ------------------------------------------------------------------
    N2 = N * N
    b = m // N2
    rem = m % N2
    i = rem // N
    k = rem % N

    # ------------------------------------------------------------------
    # Mask value (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + m, mask=in_range, other=1.0)  # fp16 0/1
    else:
        mask_val = 1.0

    # ------------------------------------------------------------------
    # Offsets inside the fused tensor (shape M × 5H, row‑major)
    # ------------------------------------------------------------------
    stride = 5 * H                     # distance between rows (in elements)
    row_base = m * stride              # start of row m

    lp = tl.load(all_out_ptr + row_base + h,                mask=in_range, other=0.0)   # left proj
    rp = tl.load(all_out_ptr + row_base + H + h,            mask=in_range, other=0.0)   # right proj
    lg_raw = tl.load(all_out_ptr + row_base + 2 * H + h,    mask=in_range, other=0.0)   # left gate (pre‑sigmoid)
    rg_raw = tl.load(all_out_ptr + row_base + 3 * H + h,    mask=in_range, other=0.0)   # right gate (pre‑sigmoid)
    og_raw = tl.load(all_out_ptr + row_base + 4 * H + h,    mask=in_range, other=0.0)   # out gate (pre‑sigmoid)

    # --------------------------------------------------------------
    # Sigmoid in fp16 (1/(1+exp(-x))) – cheap compared to the big GEMM
    # --------------------------------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg_raw))
    rg = 1.0 / (1.0 + tl.exp(-rg_raw))
    og = 1.0 / (1.0 + tl.exp(-og_raw))

    # ------------------------------------------------------------------
    # Apply gates and mask
    # ------------------------------------------------------------------
    l_val = lp * lg * mask_val
    r_val = rp * rg * mask_val

    # ------------------------------------------------------------------
    # Destination offset in layout (B, H, N, N) – row‑major
    # dest = ((b*H + h) * N + i) * N + k
    # ------------------------------------------------------------------
    dest = ((b * H + h) * N + i) * N + k

    tl.store(left_out_ptr + dest,      l_val, mask=in_range)
    tl.store(right_out_ptr + dest,     r_val, mask=in_range)
    tl.store(out_gate_out_ptr + dest,  og,    mask=in_range)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask         : torch.Tensor  [B, N, N]   (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                               # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Flatten + cast to fp16
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    M = B * N * N                     # total (b,i,k) triples
    x_flat = x.view(M, dim).contiguous()   # (M, dim)  float32
    x_flat_h = x_flat.half()          # (M, dim)  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Single fused linear (5 * hidden_dim) in fp16
    # ------------------------------------------------------------------
    # collect all projection / gate weights (fp16)
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    all_weight = torch.cat([
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
    ], dim=0)                         # shape (5*H, dim)  fp16

    # fused projection
    all_out = F.linear(x_flat_h, all_weight)   # (M, 5*H) fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask handling
    # ------------------------------------------------------------------
    HAS_MASK = 1 if (mask is not None and mask.numel() > 0) else 0
    if HAS_MASK:
        mask_h = mask.to(dtype=torch.float16, device=input_tensor.device).contiguous()
        mask_flat = mask_h.view(-1)   # (M,)
    else:
        # dummy tensor – never accessed because HAS_MASK == 0
        mask_flat = torch.empty(0, dtype=torch.float16, device=input_tensor.device)

    # ------------------------------------------------------------------
    # 5️⃣ Triton kernel: apply mask, sigmoid‑gates and rewrite layout
    # ------------------------------------------------------------------
    # allocate the three buffers in the layout required for the O(N³) step
    left_buf      = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=input_tensor.device)
    right_buf     = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=input_tensor.device)
    out_gate_buf  = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=input_tensor.device)

    total_elems = M * hidden_dim
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

    _mask_gate_permute_kernel[grid](
        all_out,                   # fused (M,5H)
        mask_flat,
        left_buf,
        right_buf,
        out_gate_buf,
        M, N, hidden_dim,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        HAS_MASK=HAS_MASK,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Batched matrix multiplication (Tensor‑core fast)
    # ------------------------------------------------------------------
    BH = B * hidden_dim
    left_mat  = left_buf.view(BH, N, N)      # (B*H, N, N) fp16
    right_mat = right_buf.view(BH, N, N)     # (B*H, N, N) fp16
    out_mat   = torch.bmm(left_mat, right_mat)   # (B*H, N, N) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out_fp32 = out_mat.view(B, hidden_dim, N, N).float()   # (B, H, N, N) float32

    # compute mean / variance over dimension 1 (the hidden axis)
    mean = out_fp32.mean(dim=1, keepdim=True)
    var  = out_fp32.var(dim=1, unbiased=False, keepdim=True)
    inv_std = torch.rsqrt(var + eps)

    # LayerNorm weight / bias (float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    scale = to_out_norm_w.view(1, hidden_dim, 1, 1)    # (1,H,1,1)
    bias  = to_out_norm_b.view(1, hidden_dim, 1, 1)    # (1,H,1,1)

    out_norm = (out_fp32 - mean) * inv_std * scale + bias

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (already sigmoid‑ed) – still in fp16
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate_buf.float()   # (B, H, N, N) float32

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to ``dim`` (float32)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) for the last linear layer
    out_pre = out_norm.permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)
    to_out_w = weights["to_out.weight"]                  # (dim, H) float32
    out = F.linear(out_pre, to_out_w)                    # (B, N, N, dim) float32

    return out
