"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1.  Layer‑norm the 4‑D input (FP32).
2.  Cast to FP16 and compute **four** linear projections in a *single* fused
    matmul:
        left_proj , left_gate , right_proj , right_gate
   The three gating tensors are passed through a sigmoid.
3.  Apply the optional mask and element‑wise gates.
4.  Core cubic update:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication
        L @ Rᵀ  (B·H independent N×N products).
   Implemented with a tiled Triton kernel using fp16 inputs,
   fp32 accumulation (Tensor‑Core friendly) and configurable tile sizes.
5.  Reshape back to (B,N,N,H) and apply a second LayerNorm over H.
6.  Multiply by the output gate and finish with a linear projection back
    to the original feature dimension (FP32).

The kernel is deliberately written to minimize launches and memory traffic
while keeping the overall API identical to the reference PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) × B (N×K)ᵀ → C (M×N)
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch (hidden‑channel)

    # ----------------------------------------------------------
    # Offsets within the current tile
    # ----------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointer to the start of the tile in A (M×K) and B (K×N)
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator in fp32 (mixed‑precision Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------------------
    # Loop over K in tiles
    # ----------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # Load with bounds checking
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Tensor‑core dot (fp16·fp16 → fp32)
        acc += tl.dot(a, b)   # fp32 accumulation

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----------------------------------------------------------
    # Write back the tile
    # ----------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Perform a batched matmul C = A @ Bᵀ with fp16 inputs and fp32 output.
    Shapes:
        A : (batch, M, K)   – fp16
        B : (batch, N, K)   – fp16
    Returns:
        C : (batch, M, N)   – fp32
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Entry point required by the evaluation harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul module.

    Arguments
    ---------
    data : (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be ignored
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – FP32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] – fp16

    # ------------------------------------------------------------------
    # 3) Fuse left/right projections and their gates into a single linear
    # ------------------------------------------------------------------
    #    Shapes (out_features, in_features)
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]

    # Concatenate all four weight matrices along the out_feature dimension:
    # order -> left_proj, left_gate, right_proj, right_gate
    combined_w = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w], dim=0
    ).half()  # shape (4*hidden_dim, dim)

    # Perform the fused linear transformation
    combined = F.linear(x_h, combined_w)  # [B, N, N, 4*hidden_dim] – fp16

    # Slice the result
    h = hidden_dim
    left = combined[..., :h]                 # left_proj
    left_gate_pre = combined[..., h : 2 * h]     # left_gate pre‑sigmoid
    right = combined[..., 2 * h : 3 * h]     # right_proj
    right_gate_pre = combined[..., 3 * h :]  # right_gate pre‑sigmoid

    # ------------------------------------------------------------------
    # 4) Optional input mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp16 = mask.to(dtype=left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 5) Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate    # [B,N,N,hidden_dim] – fp16
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].half()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()  # [B,N,N,hidden_dim] – fp16

    # ------------------------------------------------------------------
    # 8) Core cubic update – batched matmul via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B, seq_len, seq_len, hidden_dim

    # Reshape each hidden channel into a separate batch element:
    # (B, N, N, H) → (B·H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Choose tile sizes based on N (larger tiles for larger sequences)
    if N <= 384:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 64

    # Batched matmul: left @ rightᵀ  (B*H independent N×N products)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B*H, N, N] – fp32

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 9) Second LayerNorm over hidden dimension (fp32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, hidden_dim] – fp32

    # ------------------------------------------------------------------
    # 10) Apply output gate (cast to fp32) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.float()                     # gated output
    out = F.linear(out, weights["to_out.weight"])    # [B,N,N,dim] – fp32

    return out
