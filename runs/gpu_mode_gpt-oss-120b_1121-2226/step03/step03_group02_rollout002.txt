"""
custom_kernel – Triton‑accelerated TriMul “outgoing” update (AlphaFold3).

Algorithm
---------
1️⃣ Layer‑norm the input tensor X ∈ [B, N, N, D] (D = dim).
2️⃣ Fuse the two linear projections (left/right) into a single mat‑mul
   (2·H → left|right) and the three gate‑lines (left_gate, right_gate,
   out_gate) into another mat‑mul (3·H → gates).  All work in fp16.
3️⃣ If a mask is present, apply mask × gate to the left/right tensors with a
   tiny Triton kernel that works on the flattened view (no extra copies).
   Otherwise only the per‑channel gates are applied (pure PyTorch).
4️⃣ Core N³ update:
      out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication (B·H, N, N) @ (B·H, N, N)ᵀ
   performed with torch.bmm (cuBLAS/Tensor‑Cores).
5️⃣ Layer‑norm over the hidden dimension H, multiply by the out‑gate,
   and finally linearly project back to the original dimension D.
   The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Tiny Triton kernel: in‑place mask & gate fusion.
# ------------------------------------------------------------
@triton.jit
def _apply_mask_gate_kernel(
    left_ptr,                # fp16* [B,N,N,H] flattened
    right_ptr,               # fp16* same layout as left
    mask_ptr,                # fp16* [B,N,N]   flattened
    left_gate_ptr,           # fp16* [B,N,N,H] flattened
    right_gate_ptr,          # fp16* [B,N,N,H] flattened
    total_elements: tl.constexpr,   # B*N*N*H  (number of fp16 elements)
    hidden_dim:    tl.constexpr,    # H (known at compile time)
    BLOCK_SIZE:    tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load tensors
    left      = tl.load(left_ptr      + offs, mask=mask)
    right     = tl.load(right_ptr     + offs, mask=mask)
    l_gate    = tl.load(left_gate_ptr + offs, mask=mask)
    r_gate    = tl.load(right_gate_ptr+ offs, mask=mask)

    # Index into the 3‑D mask (broadcast over the hidden dim)
    mask_idx  = offs // hidden_dim          # integer division → index in [B,N,N] view
    mask_val  = tl.load(mask_ptr + mask_idx, mask=mask)

    # Apply:  left = left * mask * l_gate
    #         right = right * mask * r_gate
    left  = left  * mask_val * l_gate
    right = right * mask_val * r_gate

    tl.store(left_ptr  + offs, left,  mask=mask)
    tl.store(right_ptr + offs, right, mask=mask)


# ------------------------------------------------------------
# Triton‑compatible entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) – may be None
            - weights      : dict of model parameters (float32)
            - config       : dict with at least "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------
    # 0️⃣ Unpack
    # --------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input feature dimension D
    hidden_dim = config["hidden_dim"]   # projection dimension H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------
    # 1️⃣ Input Layer‑norm (float32 → fp16)
    # --------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B, N, N, D]  float32
    x_f16 = x.to(torch.float16)        # switch to half precision

    # --------------------------------------------------------
    # 2️⃣ Fuse linear projections and gates
    # --------------------------------------------------------
    # ----- left / right projection (2·H) -----
    proj_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                # (2·H, D)
    proj = F.linear(x_f16, proj_weight)        # [B, N, N, 2·H]  fp16
    left, right = torch.split(proj, hidden_dim, dim=-1)   # each [B,N,N,H]

    # ----- gate projections (3·H) -----
    gate_weight = torch.cat(
        [
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # (3·H, D)
    gates = F.linear(x_f16, gate_weight)          # [B, N, N, 3·H]  fp16
    left_gate, right_gate, out_gate = torch.split(gates, hidden_dim, dim=-1)
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # --------------------------------------------------------
    # 3️⃣ Optional mask & gate fusion (Triton)
    # --------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → fp16, broadcast over hidden_dim
        mask_f16 = mask.to(torch.float16).contiguous()   # [B,N,N] fp16

        # Ensure contiguous layout for all tensors used in the kernel
        left      = left.contiguous()
        right     = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate= right_gate.contiguous()

        total_elements = left.numel()                 # B·N·N·H
        BLOCK = 1024
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        _apply_mask_gate_kernel[grid](
            left,
            right,
            mask_f16,
            left_gate,
            right_gate,
            total_elements,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
            num_stages=1,
        )
        # Gates have already been applied inside the kernel → nothing more to do.
    else:
        # No mask – just apply the per‑channel gates (tiny overhead)
        left  = left  * left_gate
        right = right * right_gate

    # --------------------------------------------------------
    # 4️⃣ Core N³ update: batched mat‑mul (B·H, N, N) @ (B·H, N, N)ᵀ
    # --------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    # reshape to (B·H, N, N) where the second dim is the “k” axis
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # fp16

    # Batched mat‑mul: (B·H, N, N) @ (B·H, N, N)ᵀ  → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16
    out_mat = out_mat.to(torch.float32)                         # back to fp32 for LN

    # Restore the original 4‑D layout [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # [B, N, N, H]  float32

    # --------------------------------------------------------
    # 5️⃣ Output Layer‑norm, out‑gate, final projection
    # --------------------------------------------------------
    # Layer‑norm over hidden dimension (use half‑precision weights)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out_fp16 = out.to(torch.float16)
    out_norm = F.layer_norm(
        out_fp16,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # fp16

    # Apply the out‑gate (already fp16)
    out_norm = out_norm * out_gate

    # Final linear projection back to original dimension D
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (D, H)
    out_final = F.linear(out_norm, to_out_w)               # fp16, [B,N,N,D]

    # Return float32 as required by the harness
    return out_final.to(torch.float32)
