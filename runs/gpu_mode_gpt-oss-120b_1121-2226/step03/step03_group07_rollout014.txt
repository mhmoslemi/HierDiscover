"""
TriMul (outgoing) – Triton‑accelerated forward pass
==================================================================

The TriMul op computes for each hidden channel **h** a dense
matrix‑multiplication

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are the linear projections of the
4‑D input `[B, N, N, C]`.  The computation is the only N³‑heavy
part of the module; everything else (LayerNorm, gating,
final projection) is performed with highly‑optimised PyTorch
kernels.

Key ideas
---------
1️⃣  Cast the FP32 input to FP16 – the heavy arithmetic lives in half‑precision.
2️⃣  Apply the three linear projections (left / right) and three
    sigmoid gates in FP16.
3️⃣  **Kernel**: a custom Triton kernel that treats the hidden
    dimension as an extra batch axis, i.e. it computes a batched
    GEMM `A @ Bᵀ` for **B·H** independent (N×N) matrix‑multiplies.
    The kernel works on FP16 operands, accumulates in FP32 and
    writes the result back in FP32.
4️⃣  Output LayerNorm (over the hidden dimension) → FP32.
5️⃣  Multiply by the output‑gate and apply the final FP32 linear
    projection back to the original `dim`.

Only the core N³ GEMM is written in Triton – this gives a massive
speed‑up on H100 (Tensor‑Core enabled FP16 × FP16 → FP32).  The
kernel avoids any extra copies: the hidden dimension is fused
into the batch index, so the original `[B,N,N,H]` layout is used
directly.

The entry point `custom_kernel` follows the required signature:
    (input_tensor, mask, weights_dict, config_dict) → output_tensor
and returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiply (A @ Bᵀ) over the hidden dimension.
#   left : (B, N, N, H)   – rows = i, reduction dim = k, hidden = h
#   right: (B, N, N, H)   – rows = j, reduction dim = k, hidden = h
#   out  : (B, N, N, H)   – result of Σ_k left[i,k] * right[j,k] per h
# -------------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                # tensor pointers
    B, N, H,                                     # batch, seq_len, hidden
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -------------------------------------------------------------
    # program ids
    # -------------------------------------------------------------
    pid_m = tl.program_id(0)          # output‑row block   (i)
    pid_n = tl.program_id(1)          # output‑col block   (j)
    pid_bh = tl.program_id(2)         # combined batch‑hidden axis (B * H)

    # -------------------------------------------------------------
    # decode combined batch‑hidden index
    # -------------------------------------------------------------
    b = pid_bh // H                   # batch id
    h = pid_bh % H                    # hidden channel id

    # -------------------------------------------------------------
    # tile offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # -------------------------------------------------------------
    # base pointers for this (b, h) pair
    # -------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # -------------------------------------------------------------
    # initialise pointers to the beginning of the tile
    # -------------------------------------------------------------
    a_ptrs = left_base + offs_m[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_base + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + offs_n[None, :] * stride_right_i

    # -------------------------------------------------------------
    # accumulator (FP32)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # iterate over the reduction dimension K (= N)
    # -------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k_iter * BLOCK_K
        k_idx = tl.arange(0, BLOCK_K) + k_off
        mask_k = k_idx < N

        # load A tile (shape BLOCK_M x BLOCK_K) – mask out‑of‑bounds
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load B tile (shape BLOCK_K x BLOCK_N)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # fp16·fp16 → fp32 dot‑product (uses Tensor‑Cores on H100)
        acc += tl.dot(a, b)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # -------------------------------------------------------------
    # write back the result (still FP32)
    # -------------------------------------------------------------
    out_ptrs = out_base + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    mask_out = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def _tri_mul_triton(left: torch.Tensor,
                    right: torch.Tensor,
                    block_m: int = 128,
                    block_n: int = 128,
                    block_k: int = 32) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel defined above.
    Arguments
    ---------
    left, right : torch.HalfTensor of shape (B, N, N, H)
    Returns
    -------
    out : torch.FloatTensor of shape (B, N, N, H)  (FP32 accumulation)
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    # allocate output tensor (FP32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # strides (in elements, not bytes)
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    grid = (
        (N + block_m - 1) // block_m,          # rows
        (N + block_n - 1) // block_n,          # cols
        B * H,                                 # batch * hidden
    )
    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_ri,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return out


# -------------------------------------------------------------------------
# Entry point required by the evaluation harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim] (float32)
        - mask         : torch.Tensor [bs, seq_len, seq_len] (float32) – may be dummy
        - weights      : dict of model weights (float32 tensors)
        - config       : dict containing at least ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------- unpack -------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]                 # input feature dimension
    hidden_dim = config["hidden_dim"]   # projection dimension (H)
    nomask = config.get("nomask", True)

    # --------------------- 1️⃣ LayerNorm (FP32) -----------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # (B, N, N, dim)  FP32

    # --------------------- 2️⃣ Cast to FP16 for heavy compute --------------------
    x_h = x.to(torch.float16)

    # --------------------- 3️⃣ Load weights (cast to FP16 where possible) -------
    left_proj_w   = weights["left_proj.weight"].half()
    right_proj_w  = weights["right_proj.weight"].half()
    left_gate_w   = weights["left_gate.weight"].half()
    right_gate_w  = weights["right_gate.weight"].half()
    out_gate_w    = weights["out_gate.weight"].half()

    # output‑norm / final linear stay in FP32 (as in the reference)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # (dim, hidden), FP32

    # --------------------- 4️⃣ Linear projections (FP16) ------------------------
    left  = F.linear(x_h, left_proj_w)    # (B, N, N, H)  FP16
    right = F.linear(x_h, right_proj_w)   # (B, N, N, H)  FP16

    # --------------------- 5️⃣ Optional mask ------------------------------------
    if not nomask:
        # mask: [B, N, N] → FP16 and broadcast over hidden dimension
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask_h
        right = right * mask_h

    # --------------------- 6️⃣ Gating (sigmoid) ---------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # (B, N, N, H)  FP16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # (B, N, N, H)  FP16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # (B, N, N, H)  FP16

    left  = left * left_gate
    right = right * right_gate

    # --------------------- 7️⃣ Core N³ operation: batched GEMM -----------------
    # out = Σ_k left[i,k,h] * right[j,k,h]  (per hidden channel)
    out = _tri_mul_triton(left, right,
                          block_m=128, block_n=128, block_k=32)   # (B, N, N, H)  FP32

    # --------------------- 8️⃣ Output LayerNorm (over hidden) ------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # (B, N, N, H)  FP32

    # --------------------- 9️⃣ Multiply by output gate & final linear ----------
    out = out * out_gate.to(torch.float32)         # (B, N, N, H)  FP32
    out = F.linear(out, to_out_w)                  # (B, N, N, dim) FP32

    return out
