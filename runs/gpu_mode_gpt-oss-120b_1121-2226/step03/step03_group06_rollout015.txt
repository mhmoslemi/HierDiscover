"""Triton‑accelerated outgoing TriMul (AlphaFold‑3 style).

Algorithm
---------
1. Layer‑Norm the 4‑D input (float32).
2. Cast to fp16 and apply three bias‑free linear projections
   (left, right) together with three sigmoid gates (left/ right/ out).
3. Optional mask is fused in half‑precision.
4. Core cubic update
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication L @ Rᵀ where each hidden channel
   `h` becomes a separate batch element.  The kernel works directly on the
   original layout (B,N,N,H) – no intermediate copies or permutes are made.
   It uses fp16·fp16 → fp32 accumulation (Tensor‑Core friendly) with
   configurable tile sizes (64/128).
5. Second LayerNorm over the hidden dimension, element‑wise out‑gate,
   and a final linear projection back to the original feature dimension
   (float32).

The heavy part (step 4) is implemented in Triton; the remaining
operations use PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ with hidden‑channel fused batch.
#   A, B : (B, N, N, H)   – fp16
#   C    : (B, H, N, N)   – fp32
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (M=N=K=seq_len)
    stride_ah, stride_am, stride_ak,         # A strides: batch, row(i), col(k)
    stride_bh, stride_bn, stride_bk,         # B strides: batch, row(j), col(k)
    stride_ch, stride_ch_hidden, stride_cm, stride_cn,  # C strides
    HIDDEN_DIM: tl.constexpr,                # hidden dimension (H)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row (i)
    pid_n = tl.program_id(1)          # tile col (j)
    pid_hb = tl.program_id(2)         # combined batch*hidden index

    # ------------------------------------------------------------------
    # De‑compose combined index into batch (b) and hidden channel (h)
    # ------------------------------------------------------------------
    b = pid_hb // HIDDEN_DIM
    h = pid_hb % HIDDEN_DIM

    # Base offset for the current (b, h) slice.
    # A/B layout: ((b*N + i)*N + k)*H + h
    #   → batch stride = N*N*H, row stride = N*H, col stride = H, hidden stride = 1
    offset = b * stride_ah + h          # offset = b*batch_stride + h*1

    # --------------------------------------------------------------
    # Offsets for the current M/N tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the tile in A and B.
    # A tile: (BLOCK_M, BLOCK_K)   – rows i, cols k
    a_ptrs = (
        A_ptr
        + offset
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    # B tile: (BLOCK_K, BLOCK_N)   – rows k, cols j
    b_ptrs = (
        B_ptr
        + offset
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator in fp32 (mixed‑precision Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Mask to avoid out‑of‑bounds accesses
        mask_a = (offs_m[:, None] < M) & ((k * BLOCK_K + tl.arange(0, BLOCK_K)[None, :]) < K)
        mask_b = ((k * BLOCK_K + tl.arange(0, BLOCK_K)[:, None]) < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulation

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Write out the computed tile
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + b * stride_ch
        + h * stride_ch_hidden
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    left: torch.Tensor,
    right: torch.Tensor,
    hidden_dim: int,
    *,
    BLOCK_M: int,
    BLOCK_N: int,
    BLOCK_K: int,
) -> torch.Tensor:
    """
    Batched matmul for the TriMul update.
    left / right: (B, N, N, H) – fp16 tensors.
    Returns: (B, N, N, H) – fp32 tensor.
    """
    assert left.is_cuda and right.is_cuda
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    assert H == hidden_dim
    # (B, H, N, N) output – fp32
    out = torch.empty((B, hidden_dim, N, N), dtype=torch.float32, device=left.device)

    # Strides (in element units)
    stride_ah, stride_am, stride_ak, _ = left.stride()
    stride_bh, stride_bn, stride_bk, _ = right.stride()
    stride_ch, stride_ch_hidden, stride_cm, stride_cn = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        N,          # M
        N,          # N
        N,          # K
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_ch_hidden,
        stride_cm,
        stride_cn,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Convert back to original layout (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry‑point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the **outgoing** TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights_dict : dict with all linear/norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # --------------------------------------------------------------
    # 2) Cast to fp16 for heavy work
    # --------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] – fp16

    # --------------------------------------------------------------
    # 3) Linear projections (fp16, bias‑free)
    # --------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim] – fp16
    right = F.linear(x_h, right_proj_w)   # fp16

    # --------------------------------------------------------------
    # 4) Optional mask (fp16) – applied before gating
    # --------------------------------------------------------------
    if not nomask:
        # mask is broadcast over the hidden dimension
        m = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------
    # 5) Gating (sigmoid) – still in fp16
    # --------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = F.linear(x_h, left_gate_w).sigmoid()    # [B, N, N, hidden_dim]
    right_gate = F.linear(x_h, right_gate_w).sigmoid()  # fp16
    out_gate = F.linear(x_h, out_gate_w).sigmoid()      # fp16

    # --------------------------------------------------------------
    # 6) Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7) Core cubic update – batched matmul via Triton (no copies)
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # Choose tile sizes adaptively
    if N <= 384:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 64 if N <= 768 else 128

    out = _batched_matmul_triton(
        left,
        right,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B, N, N, hidden_dim] – fp32

    # --------------------------------------------------------------
    # 8) Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # --------------------------------------------------------------
    # 9) Output gating & final projection back to dim
    # --------------------------------------------------------------
    out = out * out_gate.float()                     # element‑wise gated
    out = F.linear(out, weights["to_out.weight"])    # [B, N, N, dim] – fp32

    return out
