    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    b = batch_idx // H
    h = batch_idx % H

    # Compute block coordinates for i and j:
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Determine offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # left[i,k]

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ [B, N, N, D] (D = dim).
2. Fuse the five linear projections (left, right, left_gate, right_gate,
   out_gate) into a single matrix‑multiply:
        Y = X @ Wᵀ ,   W ∈ ℝ^{5·H × D}
   Slice Y → left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw.
3. Apply sigmoid to the three gate tensors.
4. Fuse gating (and optional binary mask) into the tensors:
        left  = left_raw  * left_gate  * mask
        right = right_raw * right_gate * mask
5. Core N³ update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton batched‑mat‑mul kernel that computes all
   H heads for every batch without any data reshaping/copies.
6. Layer‑norm over the hidden dimension, multiply by out_gate and a final
   linear projection back to D.
All heavy work runs in fp16 (accumulates in fp32), the final result is cast
to float32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: Batched mat‑mul for the N³ update.
# Computes, for each (batch, head) pair:
#   out_{i,j} = Σ_k left_{i,k} * right_{j,k}
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # problem size
    stride_lb, stride_li, stride_lk, stride_lh, # left strides
    stride_rb, stride_ri, stride_rk, stride_rh, # right strides
    stride_ob, stride_oi, stride_oj, stride_oh, # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------
    # program ids: (batch_head, tile_m, tile_n)
    # ------------------------------
    batch_head = tl.program_id(0)            # 0 … B*H-1
    tile_m = tl.program_id(1)                # i‑tile
    tile_n = tl.program_id(2)                # j‑tile

    # decompose batch_head → batch b and head h
    b = batch_head // H
    h = batch_head % H

    # ------------------------------
    # tile offsets in the M/N dimensions (i and j)
    # ------------------------------
    offs_i = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base address for this (b, h) slice
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------
    # K loop
    # ------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  → shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[j, k] transposed → shape (BLOCK_K, BLOCK_N)
        # (k as rows, j as columns)
        b_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # dot: (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b)

    # ------------------------------
    # write result
    # ------------------------------
    c_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D]      (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None when nomask=True
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ---------------------------------------------------------------
    # 1) unpack arguments
    # ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------
    # 2) Input LayerNorm (float32)
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ---------------------------------------------------------------
    # 3) Cast to fp16 for the heavy linear work
    # ---------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ---------------------------------------------------------------
    # 4) Fuse the five linear layers into a single matmul
    # ---------------------------------------------------------------
    # Gather the five weight matrices (bias‑free) and cast once
    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    # each weight: (hidden_dim, dim)
    combined_w = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16)  # shape (5*H, dim)

    # single fused linear
    fused = F.linear(x_f16, combined_w)           # [B, N, N, 5*H] (fp16)

    # split tensors
    H = hidden_dim
    left_raw      = fused[..., 0 * H : 1 * H]      # [B,N,N,H]
    right_raw     = fused[..., 1 * H : 2 * H]
    left_gate_raw = fused[..., 2 * H : 3 * H]
    right_gate_raw= fused[..., 3 * H : 4 * H]
    out_gate_raw  = fused[..., 4 * H : 5 * H]

    # ---------------------------------------------------------------
    # 5) Gates (sigmoid) and optional mask fusion
    # ---------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_raw)     # fp16
    right_gate = torch.sigmoid(right_gate_raw)    # fp16
    out_gate   = torch.sigmoid(out_gate_raw)      # fp16, used later

    if not nomask:
        # mask shape [B, N, N] -> broadcast over hidden dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left_raw  * left_gate  * mask_fp16
        right = right_raw * right_gate * mask_fp16
    else:
        left  = left_raw  * left_gate
        right = right_raw * right_gate

    # ---------------------------------------------------------------
    # 6) Core N³ batched mat‑mul (L @ Rᵀ) via custom Triton kernel
    # ---------------------------------------------------------------
    B, N, _, _ = left.shape  # [B, N, N, H]
    # output tensor (fp16)
    out = torch.empty_like(left, dtype=torch.float16)

    # strides for left/right/out (all share the same layout)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # block sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # launch grid: (batch*head, n_tiles_M, n_tiles_N)
    grid = (
        B * H,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _trimul_matmul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ---------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final projection
    # ---------------------------------------------------------------
    # Cast weights for the output norm & final linear to fp16 once
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)   # (dim, H)

    # layer‑norm over hidden dimension (H)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # still fp16

    # multiply by the output gate (sigmoid)
    out = out * out_gate

    # final linear back to original dimension D
    out = F.linear(out, to_out_w)   # fp16 → fp16

    # Cast to float32 for the final result
    return out.to(torch.float32)
