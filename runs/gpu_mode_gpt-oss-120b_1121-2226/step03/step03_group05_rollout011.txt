    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_ni, stride_nk, stride_hl,
    stride_br, stride_nj, stride_nk_r, stride_hr,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    batch_head = pid
    b = batch_head // H
    h = batch_head % H

    # offsets for i and j tile
    offs_i = tl.arange(0, BLOCK_M) + tl.program_id(1) * BLOCK_M
    offs_j = tl.arange(0, BLOCK_N) + tl.program_id(2) * BLOCK_N

    # mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # left base = left_ptr + b*stride_bl + h*stride_hl

    left_base = left_ptr + b*stride_bl + h*stride_hl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_g_b, stride_g_i, stride_g_k, stride_g_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_rg_b, stride_rg_j, stride_rg_k, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program id dimension: 0 is batch*head, 1 is i tile, 2 is j tile
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and head indices
    b = pid // H
    h = pid % H

    # compute global row/column offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers base for left and right derived from (b,h)
    # left_ptr base index for left[i,k] values: L(b,i,k,h)
    # We will compute offset for each (i,k) pair.
    # Compute base offset for left of (b,0,0,h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    left_gate_base = left_gate_ptr + b * stride_g_b + h * stride_g_h

    # similarly for right
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    right_gate_base = right_gate_ptr + b * stride_rg_b + h * stride_rg_h

    # mask base
    mask_base = mask_ptr + b * stride_mask_b

    # allocate accumulator in fp32

"""
Triton‑accelerated forward pass for the “outgoing” Triangle Multiplicative Update
(TriMul) used in AlphaFold‑3.

Algorithm
---------
1. Layer‑norm the 4‑D input X ∈ [B, N, N, D] (float32).
2. Cast X to fp16 and apply four bias‑free linear projections:
   * left_proj , right_proj      → [B, N, N, H]
   * left_gate , right_gate , out_gate → [B, N, N, H] (sigmoid)
3. Fuse the optional binary mask with the two gates in‑place via a tiny
   element‑wise Triton kernel (left/ right ← left·gate·mask, etc.).
4. Core N³ update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched mat‑mul L @ Rᵀ for each head *h*.  The computation is
   performed by a custom Triton kernel that works directly on the
   [B,N,N,H] layout, thereby avoiding the costly permute+contiguous copies
   required for `torch.bmm`.  The kernel loads 64×64 tiles of the output,
   iterates over K in 64‑element blocks and uses `tl.dot` (Tensor‑Core on H100)
   with fp16 inputs and fp32 accumulation.
5. Layer‑norm over the hidden dimension H, multiply by the output‐gate and
   apply the final linear projection back to the original dimensionality D.
All heavy arithmetic is performed in fp16, the accumulation in the N³ step
is in fp32, and the final result is returned as float32.

The implementation satisfies the required signature `custom_kernel(data)`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Element‑wise mask + gate fusion (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,                     # 0/1 mask broadcasted to hidden dim
    total_elems: tl.int32,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elems

    l   = tl.load(left_ptr       + offs, mask=valid, other=0.0)
    lg  = tl.load(left_gate_ptr  + offs, mask=valid, other=0.0)
    r   = tl.load(right_ptr      + offs, mask=valid, other=0.0)
    rg  = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)
    m   = tl.load(mask_ptr       + offs, mask=valid, other=1.0)  # 1 → no masking

    l_out = l * lg * m
    r_out = r * rg * m

    tl.store(left_ptr  + offs, l_out, mask=valid)
    tl.store(right_ptr + offs, r_out, mask=valid)


# ----------------------------------------------------------------------
# Core N³ update: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# Works on the original [B,N,N,H] layout, avoiding any permute.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left (same layout as right)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (identical to left)
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    # strides for output (identical to left/right)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 3‑D launch: (batch*head, i‑tile, j‑tile)
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch index and head index
    b = pid // H
    h = pid % H

    # global i / j coordinates for this tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        start_k = kk * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile ------------------------------------------------
        left_ptrs = left_base + \
                    (offs_i[:, None] * stride_l_i) + \
                    (offs_k[None, :] * stride_l_k)
        l = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                      # fp16

        # ----- load right tile (note shape N x K) ---------------------------
        right_ptrs = right_base + \
                     (offs_j[:, None] * stride_r_j) + \
                     (offs_k[None, :] * stride_r_k)
        r = tl.load(right_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0)                      # fp16

        # fp16 × fp16 → fp32 accumulation via Tensor Cores
        acc += tl.dot(l, r, trans_b=True)  # (M,K)·(K,N)ᵀ → (M,N)

    # write out the tile
    out_ptrs = out_base + \
               (offs_i[:, None] * stride_o_i) + \
               (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, D] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor  shape [B, N, N, D] (float32)
    """
    # ---------------------- unpack ----------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------- 1) input LayerNorm ----------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # [B, N, N, D] (float32)

    # ---------------------- 2) cast to fp16 for heavy work ----------------------
    x_f16 = x.to(torch.float16)

    # ---------------------- 3) linear projections & gates (fp16) ----------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left  = F.linear(x_f16, left_proj_w)   # [B, N, N, H] fp16
    right = F.linear(x_f16, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x_f16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_f16, out_gate_w))   # fp16

    # ---------------------- 4) optional mask + gate fusion (in‑place) ----------
    if not nomask and mask is not None:
        # mask → fp16, shape [B, N, N, 1] to match hidden dim broadcasting
        mask_f16 = mask.to(torch.float16).unsqueeze(-1).contiguous()
        # ensure contiguous tensors for 1‑D view used by the tiny kernel
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask_f16 = mask_f16.contiguous()

        total = left.numel()                       # B·N·N·H
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # simple element‑wise multiplication when no mask is present
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ batched mat‑mul (L @ Rᵀ) – Triton implementation
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                      # [B, N, N, H]
    out_hidden = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # compute strides (in elements, not bytes)
    ls = left.stride()   # (b, i, k, h)
    rs = right.stride()
    os = out_hidden.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch * head dimension
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _trimul_outgoing_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_hidden.data_ptr(),
        B, N, H,
        ls[0], ls[1], ls[2], ls[3],             # left strides
        rs[0], rs[1], rs[2], rs[3],             # right strides (identical layout)
        os[0], os[1], os[2], os[3],             # output strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                            # good occupancy on H100
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                           # still fp32

    # out_gate is fp16 → will be up‑cast automatically
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # back to [B,N,N,D] (fp32)

    return out
