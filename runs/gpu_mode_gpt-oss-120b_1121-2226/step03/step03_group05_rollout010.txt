"""
TriMul “outgoing” forward (no backward) – Triton‑accelerated version.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Fuse all five bias‑free linear projections (left, right, left‑gate,
   right‑gate, out‑gate) into a single mat‑mul with a concatenated weight
   matrix → one GEMM instead of five.
3. Split the result, apply sigmoid to the three gate tensors.
4. Fuse mask (if present) and gate multiplications into a tiny Triton
   kernel that works in‑place on the half‑precision tensors.
5. Core N³ update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed with a single ``torch.einsum`` which avoids the costly
   permutation / copy required for a batched ``torch.bmm``.
6. Cast to float32, apply the output LayerNorm, scale by the output gate
   and project back to the original ``dim`` with a final linear layer.

All heavy linear work runs in fp16 (Tensor‑Core friendly); the only
auxiliary Triton kernel fuses mask and gate element‑wise work.  The
implementation supports both masked and un‑masked inputs and works for
the full range of shapes used in the evaluation harness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, left‑gate and right‑gate multiplication.
#   left  = left  * left_gate  * mask
#   right = right * right_gate * mask
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,          # left tensors
    right_ptr, right_gate_ptr,        # right tensors
    mask_ptr,                         # 1‑D mask (B·N·N) – fp16
    total_elems: tl.int32,            # B·N·N·H
    hidden_dim: tl.constexpr,         # H
    BLOCK_SIZE: tl.constexpr,         # threads per program
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load the four values
    l_val   = tl.load(left_ptr   + offs, mask=mask, other=0.0)
    l_gate  = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_val   = tl.load(right_ptr  + offs, mask=mask, other=0.0)
    r_gate  = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # The mask is stored without the hidden dimension – broadcast it.
    # ``offs // hidden_dim`` gives the index into the 1‑D mask tensor.
    m_idx = offs // hidden_dim
    m_val   = tl.load(mask_ptr + m_idx, mask=mask, other=1.0)  # default 1 → no masking

    # Fuse the three element‑wise multiplies
    l_out = l_val * l_gate * m_val
    r_out = r_val * r_gate * m_val

    tl.store(left_ptr  + offs, l_out, mask=mask)
    tl.store(right_ptr + offs, r_out, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update implemented with a
    fused linear layer and a small Triton kernel for mask/gate fusion.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 1) unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2) input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 3) cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) fused linear projection for left/right and three gates
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (each of shape [hidden_dim, dim])
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*hidden_dim, dim)

    # single GEMM: (B·N·N, dim) @ (dim, 5*hidden_dim).T -> (B·N·N, 5*hidden_dim)
    all_proj = F.linear(x_f16, proj_weight)   # [B, N, N, 5*hidden_dim]

    # split the concatenated result
    left, right, left_gate_raw, right_gate_raw, out_gate_raw = torch.split(
        all_proj, hidden_dim, dim=-1
    )   # each -> [B, N, N, hidden_dim]

    # gate activations
    left_gate = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate = torch.sigmoid(out_gate_raw)

    # ------------------------------------------------------------------
    # 5) optional mask + gate fusion (via Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N]  → 1‑D fp16 tensor of length B·N·N
        mask_f16 = mask.to(torch.float16).contiguous().view(-1)

        # ensure contiguous tensors for linear memory indexing
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        total = left.numel()                       # B·N·N·H
        BLOCK = 1024                               # ≤ 65536 threads per block
        grid = ( (total + BLOCK - 1) // BLOCK, )

        _mask_gate_fuse_kernel[grid](
            left.data_ptr(),
            left_gate.data_ptr(),
            right.data_ptr(),
            right_gate.data_ptr(),
            mask_f16.data_ptr(),
            total,
            hidden_dim,          # constexpr hidden dimension for division
            BLOCK_SIZE=BLOCK,
            num_warps=4,        # modest occupancy – enough for elementwise work
        )
    else:
        # no mask – just apply the gates (in‑place to save memory)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core N³ update via einsum (no intermediate permute/copy)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Result is fp16
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 7) Output layer‑norm, out‑gate scaling and final projection
    # ------------------------------------------------------------------
    out = out.to(torch.float32)   # layer‑norm works in fp32

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate is fp16 – promote to fp32 for the scaling
    out = out * out_gate.to(torch.float32)

    # final linear projection back to `dim` (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
