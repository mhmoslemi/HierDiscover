"""
TriMul “outgoing” (forward‑only) implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor [B, N, N, D] in FP32.
2. Cast to FP16 – all heavy linear work stays in half‑precision.
3. Compute the two bias‑free projections (left/right) and the three
   sigmoid‑gates (left_gate, right_gate, out_gate) with      torch.nn.functional.linear.
4. (Optional) apply the pair‑wise mask to left/right before gating.
5. Multiply left/right by their respective gates.
6. Reshape to a batched 3‑D view (B·H, N, N) and launch a fused
   Triton kernel that computes the batched matrix product

        out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k]

   i.e. left @ rightᵀ.  The kernel works on FP16 tiles, accumulates
   in FP32 and stores the result in FP32.  Tile size (64,64,64) with
   4 warps and 3 pre‑fetch stages gives high occupancy on H100.
7. Reshape back to [B, N, N, H].
8. Layer‑norm over the hidden dimension H (FP32).
9. Apply the output gate and the final bias‑free linear projection
   back to the original dimension D.
10. Return a FP32 tensor of shape [B, N, N, D].

Only the N³ matrix multiplication (the true bottleneck) is executed
in Triton; all surrounding ops use highly‑optimized torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ  (A,B are fp16, C is fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,               # left  : [BH, M, K]   (fp16)
    b_ptr,               # right : [BH, K, N]   (fp16) – interpreted as transposed
    c_ptr,               # out   : [BH, M, N]   (fp32)
    M, N, K,             # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, k, n)
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr = 3,
    num_warps: tl.constexpr = 4,
):
    """
    Batched matrix multiply `C = A @ Bᵀ` where each batch element corresponds
    to one hidden channel.  The kernel works on half‑precision inputs,
    accumulates in FP32 and writes FP32.
    """
    pid_h = tl.program_id(axis=2)               # batch * hidden
    pid_m = tl.program_id(axis=0)               # tile row
    pid_n = tl.program_id(axis=1)               # tile col

    # ------------------------------------------------------------------
    # Tile offsets & masks
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile (M × K)
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                                          # fp16

        # Bᵀ tile (K × N) – note swapped strides for transposition
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                          # fp16

        # Accumulate: tl.dot gives fp16, cast to fp32 before adding
        acc += tl.dot(a, b).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, D] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Bias‑free linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4️⃣ Optional pair‑wise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating – sigmoid on FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_fp16, weights["left_gate.weight"].half())
    )
    right_gate = torch.sigmoid(
        F.linear(x_fp16, weights["right_gate.weight"].half())
    )
    out_gate = torch.sigmoid(
        F.linear(x_fp16, weights["out_gate.weight"].half())
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Reshape to [B·H, N, N] for the batched GEMM
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N) -> (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = bs * hidden_dim                      # batch × hidden
    left_flat = left_perm.view(BH, seq_len, seq_len)   # fp16
    right_flat = right_perm.view(BH, seq_len, seq_len) # fp16

    # Output buffer for the GEMM (FP32 accumulation)
    out_flat = torch.empty(
        (BH, seq_len, seq_len), dtype=torch.float32, device=device
    )

    # ------------------------------------------------------------------
    # 7️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # rows
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # cols
        BH,                                   # batch*hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,                     # M
        seq_len,                     # N
        seq_len,                     # K
        # strides for A (left)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # strides for Bᵀ (right, transposed view)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # strides for C (output)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out_perm = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()             # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # stays float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (broadcast) and final linear projection (FP32)
    # ------------------------------------------------------------------
    # out_gate is still in FP16; promote to FP32 for the multiplication
    out = out * out_gate.to(out.dtype)          # [B,N,N,H] × [B,N,N,H]
    out = F.linear(out, weights["to_out.weight"])   # bias‑free, back to D

    return out
