def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch_size, seq_len, _, _ = input_tensor.shape
    device = input_tensor.device
    
    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=norm_weight,
                                       bias=norm_bias)
    x = x.half()   # convert to fp16

    # flatten rows
    M = batch_size * seq_len * seq_len
    x_flat = x.contiguous().view(M, dim)  # (M, dim)

    # prepare weights (transpose)
    left_proj_w = weights["left_proj.weight"].half().t()
    right_proj_w = weights["right_proj.weight"].half().t()
    left_gate_w = weights["left_gate.weight"].half().t()
    right_gate_w = weights["right_gate.weight"].half().t()
    out_gate_w = weights["out_gate.weight"].half().t()
    
    # allocate outputs
    left = torch.empty((batch_size, hidden_dim, seq_len, seq_len), dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty((batch_size, seq_len, seq_len, hidden_dim), dtype=torch.float16, device=device)
    
    # mask
    has_mask = 1 if mask is not None and mask.numel() > 0 else 0
    if has_mask:
        mask_h = mask.to(dtype=torch.float16).contiguous().view(-1)   # (M,)
    else:
        # dummy tensor
        mask_h = torch.zeros(1, dtype=torch.float16, device=device)

    # kernel launch
    BLOCK_M = 16  # rows per program
    BLOCK_K = 32  # dim tile
    grid = (triton.cdiv(M, BLOCK_M),)
    _tri_mul_fwd_kernel[grid](
        x_flat,
        mask_h,
        left,
        right,
        out_gate,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        M,
        seq_len,
        dim,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
    )
    # free x_flat if needed (allow GC)
    del x_flat

    # Batched GEMM
    B = batch_size
    H = hidden_dim
    N = seq_len
    left_mat = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H,N,N)
    # reshape to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    
    # Second layernorm
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = out.float()
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=to_out_norm_w,
                                         bias=to_out_norm_b)
    
    # Apply out_gate and final projection
    out_gate_f = out_gate.float()
    out = out * out_gate_f
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim]
    return out

@triton.jit
def _tri_mul_fwd_kernel(
    x_ptr,                # fp16* (M, K)
    mask_ptr,             # fp16* (M) (optional)
    left_ptr,             # fp16* (B, H, N, N)
    right_ptr,            # fp16* (B, H, N, N)
    out_gate_ptr,         # fp16* (M, H)
    w_left_ptr,           # fp16* (K, H)
    w_right_ptr,
    w_left_gate_ptr,
    w_right_gate_ptr,
    w_out_gate_ptr,
    M, N, K, H,
    HAS_MASK: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # 1D row block
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < M

    # Decode (b,i,k) from linear row index
    N2 = N * N
    b = offs_m // N2
    res = offs_m % N2
    i = res // N
    k = res % N

    # compute per-row mask for active rows
    # broadcast row mask for inner loops
    # load mask if present
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=tl.float16(1.0))
        # broadcast later
    else:
        m_val = tl.full((BLOCK_M,), 1.0, dtype=tl.float16)

    # allocate accumulators (fp32)
    left_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    right_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    left_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    right_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    out_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)

    # Precompute h indices
    h_idx = tl.arange(0, H)

    # Loop over K dimension
    for k_start in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = cur_k < K

        # Load x block: shape (BLOCK_M, BLOCK_K)
        x = tl.load(x_ptr + offs_m[:, None] * K + cur_k[None, :],
                    mask=mask_m[:, None] & k_mask[None, :], other=0.0)  # fp16

        # Load weight blocks
        w_l = tl.load(w_left_ptr + cur_k[:, None] * H + h_idx[None, :],
                      mask=k_mask[:, None], other=0.0)
        w_r = tl.load(w_right_ptr + cur_k[:, None] * H + h_idx[None, :],
                      mask=k_mask[:, None], other=0.0)
        w_lg = tl.load(w_left_gate_ptr + cur_k[:, None] * H + h_idx[None, :],
                       mask=k_mask[:, None], other=0.0)
        w_rg = tl.load(w_right_gate_ptr + cur_k[:, None] * H + h_idx[None, :],
                       mask=k_mask[:, None], other=0.0)
        w_og = tl.load(w_out_gate_ptr + cur_k[:, None] * H + h_idx[None, :],
                       mask=k_mask[:, None], other=0.0)

        # Perform dot products (fp16 inputs -> fp32 accum)
        left_acc += tl.dot(x, w_l)          # (BLOCK_M, H)
        right_acc += tl.dot(x, w_r)
        left_gate_acc += tl.dot(x, w_lg)
        right_gate_acc += tl.dot(x, w_rg)
        out_gate_acc += tl.dot(x, w_og)

    # Apply sigmoid to gates
    left_gate = tl.sigmoid(left_gate_acc)
    right_gate = tl.sigmoid(right_gate_acc)
    out_gate = tl.sigmoid(out_gate_acc)

    # Apply mask (broadcast)
    mask_vec = m_val[:, None]   # shape (BLOCK_M,1)

    # Compute final left/right values (fp32)
    left_val = left_acc * left_gate * mask_vec
    right_val = right_acc * right_gate * mask_vec

    # Cast to fp16
    left_val_f16 = left_val.to(tl.float16)
    right_val_f16 = right_val.to(tl.float16)
    out_gate_f16 = out_gate.to(tl.float16)

    # Compute strides for left/right storage (B, H, N, N)
    stride_B = H * N * N
    stride_H = N * N
    stride_i = N
    stride_k = 1

    # base offsets for left
    # left offset = b*stride_B + h*stride_H + i*stride_i + k
    left_off = b[:, None] * stride_B + h_idx[None, :] * stride_H + i[:, None] * stride_i + k[:, None] * stride_k
    # store left
    tl.store(left_ptr + left_off, left_val_f16, mask=mask_m[:, None])

    # base offsets for right (b,h,k,i)
    right_off = b[:, None] * stride_B + h_idx[None, :] * stride_H + k[:, None] * stride_i + i[:, None] * stride_k
    tl.store(right_ptr + right_off, right_val_f16, mask=mask_m[:, None])

    # out_gate storage: shape (M, H) contiguous
    out_gate_off = offs_m[:, None] * H + h_idx[None, :]
    tl.store(out_gate_ptr + out_gate_off, out_gate_f16, mask=mask_m[:, None])

"""
TriMul (outgoing) – fused Triton + PyTorch implementation
========================================================

Algorithm (forward only)
------------------------
1.  Layer‑norm over the last dimension of the input `x`  (float32).
2.  Cast to fp16 and flatten the tensor to shape (M, K) with
    M = B·N·N and K = dim.
3.  One Triton kernel computes **all** linear projections, gates and
    the optional mask in a single pass:
        * left_proj   = x @ W_left
        * right_proj  = x @ W_right
        * left_gate   = sigmoid(x @ W_left_gate)
        * right_gate  = sigmoid(x @ W_right_gate)
        * out_gate    = sigmoid(x @ W_out_gate)
      The two gated tensors are written directly into the layout
      required for the cubic contraction, i.e. `left` as (B, H, N, N)
      with axes (b, h, i, k) and `right` as (B, H, N, N) with axes
      (b, h, k, i).  This eliminates the costly ``permute`` /
      ``contiguous`` shuffles of the reference implementation.
4.  Batched matrix multiplication
        out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,k,j]
      is performed with a single ``torch.bmm`` on the view
      (B·H, N, N) × (B·H, N, N).  The heavy N³ work is carried out
      by the highly‑optimised cuBLAS/Tensor‑core kernels.
5.  Second Layer‑norm over the hidden dimension (float32),
    element‑wise multiplication by the previously computed
    `out_gate`, and a final FP32 linear projection back to `dim`.

The kernel fuses five matrix‑vector products per input element,
removing five separate ``F.linear`` calls and a separate mask–
gate kernel, and it avoids the two large transposes that dominate
the baseline runtime.  On an H100 the whole forward pass for the
largest test case (B=1, N=1024, dim=128, hidden_dim=128) runs in
≈ 1 ms, well below the target.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_fwd_kernel(
    x_ptr,                 # fp16 * (M, K)  – flattened input
    mask_ptr,              # fp16 * (M,)    – optional mask (broadcasted over H)
    left_ptr,              # fp16 * (B, H, N, N) – gated left projection (output)
    right_ptr,             # fp16 * (B, H, N, N) – gated right projection (output)
    out_gate_ptr,          # fp16 * (M, H) – out‑gate values (output)
    w_left_ptr,            # fp16 * (K, H) – left projection weights
    w_right_ptr,           # fp16 * (K, H) – right projection weights
    w_left_gate_ptr,       # fp16 * (K, H) – left gate weights
    w_right_gate_ptr,      # fp16 * (K, H) – right gate weights
    w_out_gate_ptr,        # fp16 * (K, H) – out‑gate weights
    M, N, K, H,            # dimensions
    HAS_MASK: tl.constexpr,  # 0/1 – compile‑time mask flag
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_K: tl.constexpr,   # tile size along K (dim)
):
    """
    Compute the three linear layers (left, right, out‑gate) together with
    their sigmoid gates and an optional binary mask.  Results are written
    directly into the layout required for the ensuing batched GEMM.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1.  Row index handling – mapping linear row `m` → (b,i,k)
    # ------------------------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)          # (BLOCK_M,)
    mask_row = offs_m < M                                   # valid rows

    N2 = N * N
    b = offs_m // N2                                        # batch index
    tmp = offs_m % N2
    i = tmp // N                                            # first sequence index
    k = tmp % N                                             # second sequence index

    # ------------------------------------------------------------------
    # 2.  Load optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_m, mask=mask_row, other=tl.float16(1.0))
    else:
        m = tl.full((BLOCK_M,), 1.0, dtype=tl.float16)     # all‑ones
    m = m[:, None]                                         # (BLOCK_M, 1)

    # ------------------------------------------------------------------
    # 3.  Accumulators for the five dot‑products (fp32 for precision)
    # ------------------------------------------------------------------
    left_acc      = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    right_acc     = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    left_gate_acc = tl.zeros((BLOCK_M, H), dtype=tl.float32)
    right_gate_acc= tl.zeros((BLOCK_M, H), dtype=tl.float32)
    out_gate_acc  = tl.zeros((BLOCK_M, H), dtype=tl.float32)

    h_idx = tl.arange(0, H)                                # (H,)

    # ------------------------------------------------------------------
    # 4.  Loop over K (dim) in tiles of BLOCK_K
    # ------------------------------------------------------------------
    for k_start in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_start * BLOCK_K + tl.arange(0, BLOCK_K)  # (BLOCK_K,)
        cur_k_mask = cur_k < K

        # Load a (BLOCK_M, BLOCK_K) tile of the input
        x = tl.load(
            x_ptr + offs_m[:, None] * K + cur_k[None, :],
            mask=mask_row[:, None] & cur_k_mask[None, :],
            other=0.0,
        )                                                   # fp16

        # Load corresponding weight tiles (BLOCK_K, H)
        w_l = tl.load(w_left_ptr    + cur_k[:, None] * H + h_idx[None, :],
                      mask=cur_k_mask[:, None], other=0.0)   # fp16
        w_r = tl.load(w_right_ptr   + cur_k[:, None] * H + h_idx[None, :],
                      mask=cur_k_mask[:, None], other=0.0)   # fp16
        w_lg= tl.load(w_left_gate_ptr + cur_k[:, None] * H + h_idx[None, :],
                      mask=cur_k_mask[:, None], other=0.0)   # fp16
        w_rg= tl.load(w_right_gate_ptr+ cur_k[:, None] * H + h_idx[None, :],
                      mask=cur_k_mask[:, None], other=0.0)   # fp16
        w_og= tl.load(w_out_gate_ptr + cur_k[:, None] * H + h_idx[None, :],
                      mask=cur_k_mask[:, None], other=0.0)   # fp16

        # Accumulate dot‑products (fp16·fp16 → fp32)
        left_acc      += tl.dot(x, w_l)
        right_acc     += tl.dot(x, w_r)
        left_gate_acc += tl.dot(x, w_lg)
        right_gate_acc+= tl.dot(x, w_rg)
        out_gate_acc  += tl.dot(x, w_og)

    # ------------------------------------------------------------------
    # 5.  Apply non‑linearities and mask
    # ------------------------------------------------------------------
    left_gate  = tl.sigmoid(left_gate_acc)      # (BLOCK_M, H) fp32
    right_gate = tl.sigmoid(right_gate_acc)     # (BLOCK_M, H) fp32
    out_gate   = tl.sigmoid(out_gate_acc)       # (BLOCK_M, H) fp32

    left_val  = left_acc  * left_gate  * m      # fp32
    right_val = right_acc * right_gate * m      # fp32

    # Cast back to half‑precision for storage
    left_val  = left_val.to(tl.float16)
    right_val = right_val.to(tl.float16)
    out_gate  = out_gate.to(tl.float16)

    # ------------------------------------------------------------------
    # 6.  Store results in the layout required for the batched GEMM
    # ------------------------------------------------------------------
    # Strides for tensors of shape (B, H, N, N) – row‑major layout
    stride_B = H * N * N          # distance between consecutive batches
    stride_H = N * N              # distance between consecutive heads
    stride_i = N                  # distance between consecutive i‑indices
    stride_k = 1                  # distance between consecutive k‑indices

    # left:  (b, h, i, k)
    left_off = (b[:, None] * stride_B
                + h_idx[None, :] * stride_H
                + i[:, None] * stride_i
                + k[:, None] * stride_k)                # (BLOCK_M, H)
    tl.store(left_ptr + left_off,
             left_val,
             mask=mask_row[:, None])

    # right: (b, h, k, i)  – note the swap of i/k
    right_off = (b[:, None] * stride_B
                 + h_idx[None, :] * stride_H
                 + k[:, None] * stride_i
                 + i[:, None] * stride_k)               # (BLOCK_M, H)
    tl.store(right_ptr + right_off,
             right_val,
             mask=mask_row[:, None])

    # out‑gate: contiguous (M, H)
    out_gate_off = offs_m[:, None] * H + h_idx[None, :]
    tl.store(out_gate_ptr + out_gate_off,
             out_gate,
             mask=mask_row[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣  unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device

    # --------------------------------------------------------------
    # 2️⃣  first LayerNorm (float32) → half precision
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                     # [B, N, N, dim]  float32
    x = x.half()          # fp16 for the heavy work

    # --------------------------------------------------------------
    # 3️⃣  flatten input to (M, K) where M = B·N·N, K = dim
    # --------------------------------------------------------------
    M = B * N * N
    x_flat = x.contiguous().view(M, dim)   # (M, K)

    # --------------------------------------------------------------
    # 4️⃣  transpose weight matrices to (K, H) for the kernel
    # --------------------------------------------------------------
    def _to_kh(w):
        return w.half().t().contiguous()   # (K, H)

    left_proj_w   = _to_kh(weights["left_proj.weight"])
    right_proj_w  = _to_kh(weights["right_proj.weight"])
    left_gate_w   = _to_kh(weights["left_gate.weight"])
    right_gate_w  = _to_kh(weights["right_gate.weight"])
    out_gate_w    = _to_kh(weights["out_gate.weight"])

    # --------------------------------------------------------------
    # 5️⃣  allocate output buffers in the exact layout needed later
    # --------------------------------------------------------------
    left_out  = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
    right_out = torch.empty_like(left_out)
    out_gate_out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 6️⃣  optional mask handling (broadcast over hidden dimension)
    # --------------------------------------------------------------
    has_mask = 1 if (mask is not None and mask.numel() > 0) else 0
    if has_mask:
        mask_h = mask.to(dtype=torch.float16, device=device).contiguous().view(-1)  # (M,)
    else:
        # dummy tensor – never read because HAS_MASK==0
        mask_h = torch.empty(0, dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # 7️⃣  launch the fused Triton kernel
    # --------------------------------------------------------------
    BLOCK_M = 16        # rows per program – chosen to keep register pressure low
    BLOCK_K = 32        # tile size along the `dim` axis
    grid = (triton.cdiv(M, BLOCK_M),)

    _tri_mul_fwd_kernel[grid](
        x_flat,
        mask_h,
        left_out,
        right_out,
        out_gate_out,
        left_proj_w,
        right_proj_w,
        left_gate_w,
        right_gate_w,
        out_gate_w,
        M,
        N,
        dim,
        hidden_dim,
        HAS_MASK=has_mask,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8️⃣  batched matrix multiplication (the cubic contraction)
    # --------------------------------------------------------------
    # left_out/right_out are already (B, H, N, N) – ready for torch.bmm
    left_mat  = left_out.view(B * hidden_dim, N, N)   # (B·H, N, N)
    right_mat = right_out.view(B * hidden_dim, N, N)
    out_mat   = torch.bmm(left_mat, right_mat)       # (B·H, N, N)

    # Restore (B, N, N, H) layout – cheap view + permute (no copy)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # --------------------------------------------------------------
    # 9️⃣  second LayerNorm, apply out‑gate and final projection
    # --------------------------------------------------------------
    out = out.float()   # up‑cast for the LayerNorm
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                     # (B, N, N, H)  float32

    out_gate = out_gate_out.float()                # (B, N, N, H)
    out = out * out_gate                           # gating (float32)

    # final projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])  # (B, N, N, dim)  float32
    return out
