"""
Triton‑accelerated outgoing TriMul (forward only)

Algorithm
---------
1. Layer‑norm over the input channel dimension (float‑32, PyTorch).
2. Cast to FP16 and apply four linear projections (no bias):
      - left_proj, right_proj   → hidden_dim
      - left_gate, right_gate,
        out_gate                → hidden_dim
   Gating values are passed through a sigmoid and multiplied in‑place.
   If a mask is provided it is broadcast and applied before gating.
   All heavy work is done in FP16 to halve memory traffic.
3. Contraction (core O(N³) work):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (A @ B) where each hidden
   channel `h` and batch `b` is a separate matrix of shape (N,N).
   The implementation
   * avoids any memory copies by flattening the hidden dimension into the
     batch dimension with `permute(...).flatten(0,1)`.
   * uses a custom Triton kernel that tiles the GEMM with
     BLOCK_M = BLOCK_N = 128, BLOCK_K = 64 and accumulates in FP32
     (Tensor‑core friendly FP16×FP16→FP32).
4. Reshape the result back to (B,N,N,hidden_dim) and convert to FP32.
5. Second LayerNorm over the hidden dimension (float‑32).
6. Multiply by the output gate (broadcasted, FP32).
7. Final linear projection back to the original channel dimension (float‑32).

Only the N³ contraction is executed in Triton; everything else uses
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched FP16 GEMM C = A @ B
#   A: (batch, M, K)  FP16
#   B: (batch, K, N)  FP16
#   C: (batch, M, N)  FP16 (accumulated in FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # matrix dimensions
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # block decomposition (batch, block_m, block_n)
    # ------------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # ------------------------------------------------------------------
    # tile offsets and masks
    # ------------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical safety
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch, offs_m, offs_k]
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # B[batch, offs_k, offs_n]
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Tensor‑core dot (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write back
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched FP16 GEMM using the Triton kernel above.

    Parameters
    ----------
    a : torch.Tensor
        Shape (B, M, K), dtype=torch.float16, contiguous.
    b : torch.Tensor
        Shape (B, K, N), dtype=torch.float16, contiguous.

    Returns
    -------
    torch.Tensor
        Shape (B, M, N), dtype=torch.float16, contiguous.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Tuned tile sizes for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_m * num_n,)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps (256 threads) works well with 128×128 tiles
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator, fully GPU‑accelerated.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]      (float32)
        - mask         : torch.Tensor [B, N, N] or None  (float32 / bool)
        - weights      : dict of torch.Tensor containing all model parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # model parameters (convert to the precision used in each stage)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    # Linear projection / gate weights are FP16 (no bias)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # Output‑norm and final linear stay in FP32 for numerical stability
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # (dim, hidden_dim)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over input channel dim (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for projection/gating (dram bandwidth cut in half)
    # --------------------------------------------------------------
    x_h = x.half()                                     # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – FP16
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)                 # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask may be bool; cast to same dtype as projections
        mask_h = mask.unsqueeze(-1).to(left.dtype)    # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – stay in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply gates in‑place to reduce temporary allocations
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched contraction (core O(N³) part)
    #    Treat each (batch, hidden) pair as an independent matrix‑mul.
    #    No extra copies: permute + flatten merely changes strides.
    # --------------------------------------------------------------
    # A: (B, hidden, N, N)   →   (B*hidden, N, N)
    A = left.permute(0, 3, 1, 2).flatten(0, 1)          # [B*H, N, N]  FP16
    # B: (B, hidden, N, N) with swapped axes k↔j → (B*hidden, N, N)
    B = right.permute(0, 3, 2, 1).flatten(0, 1)         # [B*H, N, N]  FP16

    # Batched matmul using the Triton kernel
    C = _batched_matmul_fp16(A, B)                     # [B*H, N, N]  FP16

    # --------------------------------------------------------------
    # 7️⃣ Restore the original 4‑D layout: (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    B_, H_, N_, _ = left.shape                         # B_, H_, N_ = B, hidden_dim, N
    out_h = C.view(B_, H_, N_, N_).permute(0, 2, 3, 1) # [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    # 8️⃣ Cast back to float32 for the second LayerNorm
    # --------------------------------------------------------------
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # --------------------------------------------------------------
    # 🔟 Apply the output gate (broadcasted)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)                      # [B, N, N, dim]  float32

    return out
