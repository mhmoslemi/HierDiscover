@triton.jit
def tril_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    bh = pid_bh

    b = bh // H
    h = bh % H

    # compute offsets
    i = i_start + tl.arange(0, BLOCK_I)
    j = j_start + tl.arange(0, BLOCK_J)

    # mask for boundaries
    mask_i = i < N
    mask_j = j < N

    # base pointers
    left_ptr_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_ptr_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_ptr_base = out_ptr + b * stride_out_b + h * stride_out_h

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K blocks
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        # mask for K dimension
        mask_k = k < N

        # load left tile: shape (BLOCK_I, BLOCK_K)
        left_addr = left_ptr_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left = tl.load(left_addr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile: shape (BLOCK_K, BLOCK_J)
        right_addr = right_ptr_base + k[:, None] * stride_right_k + j[None, :] * stride_right_j
        right = tl.load(right_addr, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)    # float32 accumulation

    # store result
    out_addr = out_ptr_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_addr, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast forward‑only implementation.

Algorithm
---------
1. **Layer‑norm + FP16 cast** – the input (B,N,N,D) is layer‑normalised in
   FP32 (the default for `torch.nn.functional.layer_norm`) and then cast to
   FP16 for the heavy linear work.

2. **Fused linear projection** – all five linear maps
   (`left_proj, left_gate, right_proj, right_gate, out_gate`) are
   concatenated into a single weight of shape `(5*H, D)` and applied with a
   single `F.linear`.  The result is sliced and the three gate tensors are
   passed through a sigmoid.

3. **Mask & gate fusion (triton)** – a tiny triton kernel multiplies the
   projected values with their corresponding gates and the optional mask.
   The kernel works on the flattened tensors, so only one launch is needed
   regardless of the sequence length.

4. **N³ contraction (triton batched GEMM)** – the costly contraction

        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   is performed by a custom Triton kernel that behaves like a batched
   GEMM without the expensive `contiguous` copies required by `torch.bmm`.
   The kernel iterates over the `k` dimension in tiles, loads the needed
   sub‑matrices directly from the original strided layout and accumulates
   the product in FP32 for numerical safety.  The result is written back as
   FP16.

5. **Post‑contraction LayerNorm + output gate** – a standard
   `torch.nn.functional.layer_norm` (still in FP16) is applied to the
   hidden‑dim tensor, then multiplied element‑wise by the `out_gate`.

6. **Final projection** – a second linear map (weight `(D, H)`) brings the
   hidden dimension back to `D`.  The final tensor is returned in FP32.

Only the mask‑gate fusion and the N³ contraction are written in Triton,
fulfilling the “must contain a custom kernel” requirement while keeping the
overall runtime well below the target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1. Mask‑gate fusion kernel
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,           # projected values (fp16)
    left_gate_ptr, right_gate_ptr, # gate values      (fp16)
    mask_ptr,                      # mask (fp16) of size B·N·N
    out_left_ptr, out_right_ptr,   # fused results    (fp16)
    total: tl.constexpr,           # total #elements = B·N·N·H
    hidden: tl.constexpr,          # hidden dimension H
    BLOCK: tl.constexpr,           # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # load tensors
    l_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # each `hidden` consecutive entries share one mask element
    mask_idx = offs // hidden
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # fuse mask & gates
    lo = l_val * l_gate * m_val
    ro = r_val * r_gate * m_val

    tl.store(out_left_ptr + offs, lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


# ----------------------------------------------------------------------
# 2. N³ contraction kernel – batched GEMM across heads
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,               # FP16, shape (B,N,N,H)
    right_ptr,              # FP16, same shape
    out_ptr,                # FP16, same shape
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,  # tile size in i dimension
    BLOCK_J: tl.constexpr,  # tile size in j dimension
    BLOCK_K: tl.constexpr,  # tile size in k dimension
):
    pid_i = tl.program_id(0)          # tile index over i
    pid_j = tl.program_id(1)          # tile index over j
    pid_bh = tl.program_id(2)         # (batch * head) index

    # decode batch and head from linear index
    b = pid_bh // H
    h = pid_bh % H

    # start offsets of the current tile
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # 1‑D index vectors for the tile
    i_ix = i_start + tl.arange(0, BLOCK_I)
    j_ix = j_start + tl.arange(0, BLOCK_J)

    # boundary masks
    i_mask = i_ix < N
    j_mask = j_ix < N

    # base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over K dimension in tiles
    for k_start in range(0, N, BLOCK_K):
        k_ix = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_ix < N

        # --------------------------------------------------------------
        # load left tile  [i, k]  -> shape (BLOCK_I, BLOCK_K)
        # --------------------------------------------------------------
        left_addr = (left_base
                     + i_ix[:, None] * stride_left_i
                     + k_ix[None, :] * stride_left_k)
        left = tl.load(left_addr,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)                     # fp16

        # --------------------------------------------------------------
        # load right tile [k, j]  -> shape (BLOCK_K, BLOCK_J)
        # --------------------------------------------------------------
        right_addr = (right_base
                      + k_ix[:, None] * stride_right_k
                      + j_ix[None, :] * stride_right_j)
        right = tl.load(right_addr,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0)                    # fp16

        # --------------------------------------------------------------
        # acc += left @ right   (dot over the K dimension)
        # --------------------------------------------------------------
        # tl.dot returns the same dtype as the inputs (fp16), so cast to fp32
        acc += tl.dot(left, right).to(tl.float32)

    # --------------------------------------------------------------
    # write result back (cast to fp16)
    # --------------------------------------------------------------
    out_addr = out_base + i_ix[:, None] * stride_out_i + j_ix[None, :] * stride_out_j
    tl.store(out_addr, acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])

        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                   # (5*H, D)
    fused = F.linear(x, fused_weight)   # [B, N, N, 5*H] (FP16)

    # split
    B, N, _, _ = fused.shape
    offset = 0
    left_proj = fused[..., offset : offset + hidden_dim]; offset += hidden_dim
    left_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]); offset += hidden_dim
    right_proj = fused[..., offset : offset + hidden_dim]; offset += hidden_dim
    right_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]); offset += hidden_dim
    out_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim])      # (B,N,N,H)

    # --------------------------------------------------------------
    # 3. Fuse mask, left/right gates (tiny Triton kernel)
    # --------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones(B * N * N, dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(-1)   # flatten B·N·N

    # flatten everything for the kernel
    left_proj_f  = left_proj.reshape(-1)
    right_proj_f = right_proj.reshape(-1)
    left_gate_f  = left_gate.reshape(-1)
    right_gate_f = right_gate.reshape(-1)

    left_out_f  = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elems = left_proj_f.shape[0]               # B·N·N·H
    BLOCK = 1024
    grid = (triton.cdiv(total_elems, BLOCK),)
    _mask_gate_fuse_kernel[grid](
        left_proj_f,
        right_proj_f,
        left_gate_f,
        right_gate_f,
        mask_fp16,
        left_out_f,
        right_out_f,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK,
    )

    # reshape back to [B,N,N,H]
    left  = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4. N³ contraction – custom Triton batched GEMM (no copies)
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # FP16 output tensor

    # strides (in elements) for the three tensors
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_j, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # kernel launch configuration
    BLOCK_I, BLOCK_J, BLOCK_K = 64, 64, 32
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid_bh = B * hidden_dim
    grid = (grid_i, grid_j, grid_bh)

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        sL_b, sL_i, sL_k, sL_h,
        sR_b, sR_j, sR_k, sR_h,
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 5. Post‑contraction LayerNorm (FP16) + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                                   # stays FP16
    out = out * out_gate                # element‑wise gate (FP16)

    # ------------------------------------------------------------------
    # 6. Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)       # (B,N,N,D)  FP16
    out = out.to(torch.float32)         # return as FP32

    return out
