"""
custom_kernel – Triton‑accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1️⃣ Apply LayerNorm on the last channel dimension (float32).
2️⃣ Cast to FP16 and compute four linear projections (left/right and three gates)
   with no bias, using cuBLAS (torch.nn.functional.linear).  All projections
   use the same input tensor, so the heavy matmul over the channel dimension is
   performed by the highly‑optimized cuBLAS kernels.
3️⃣ If a mask is supplied, broadcast‑multiply it (FP16) into the projected tensors.
4️⃣ Fuse the gating values (sigmoid) with the projected tensors.
5️⃣ Rearrange tensors to (B, H, N, N) where H = hidden_dim:
      left  → (B, H, i, k)
      right → (B, H, k, j)  (the inner‑dimension `k` is shared).
   The N³ contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication for each (b, h) pair.
   – For small sequence lengths (N ≤ 256) we launch a custom Triton GEMM
     (`_batched_matmul_fp16`) that tiles the problem and uses tt.dot.
   – For larger N we fall back to `torch.bmm`, i.e. cuBLAS’s batched GEMM,
     which is faster on an H100 for big matrices.
6️⃣ Convert the result back to FP32, apply a second LayerNorm over the hidden
   dimension, multiply by the output gate and finish with a linear projection
   back to the original `dim`.
The implementation uses mixed‑precision (FP16 for the large N³ work) while
preserving FP32 for the final output, satisfying both accuracy and speed
requirements.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (FP16 → FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ B  where
       A : (batch, M, K)
       B : (batch, K, N)
    All tensors are fp16, accumulator is fp32."""
    pid_batch = tl.program_id(2)   # merged batch * hidden
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile pointer: (batch, row=m, col=k)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        # B tile pointer: (batch, row=k, col=n)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot‑product over K
        acc += tl.dot(a, b)

    # store result (cast back to fp16)
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication in FP16 using the custom Triton kernel.
    Arguments:
        a : (B, M, K)  torch.float16, contiguous
        b : (B, K, N)  torch.float16, contiguous
    Returns:
        c : (B, M, N)  torch.float16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    # Heuristic block sizes (tuned for H100)
    if max(M, N) <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N

    # 3‑D grid: (tiles_m, tiles_n, batch)
    grid = (num_m, num_n, B)

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

    # 5️⃣ Gating (sigmoid) – also in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))

    # ------------------------------------------------------------------
    # 6️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask shape -> [B, N, N, 1] and cast to fp16
        mask_h = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 7️⃣ Apply gates (fused element‑wise)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 8️⃣ Prepare tensors for the N³ contraction
    #    left  : (B, N, N, H) → (B, H, N, N)   (i, k)
    #    right : (B, N, N, H) → (B, H, N, N)   (k, j)   (transpose of last two axes)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    # Release memory that is no longer needed
    del left, right, left_gate, right_gate, x_h, x

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions for a (batch, M, K) view
    A = left_perm.view(B * H, N, N)        # (B*H, i, k)
    B_mat = right_perm.view(B * H, N, N)   # (B*H, k, j)

    # ------------------------------------------------------------------
    # 9️⃣ Compute the contraction:
    #      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    #    Use Triton for small N, otherwise cuBLAS via torch.bmm.
    # ------------------------------------------------------------------
    if N <= 256:
        C = _batched_matmul_fp16(A, B_mat)          # (B*H, N, N)  fp16
    else:
        C = torch.bmm(A, B_mat)                      # (B*H, N, N)  fp16 (cuBLAS)

    # Restore original layout: (B, N, N, H)
    out_fp16 = (
        C.view(B, H, N, N)          # (B, H, N, N)
        .permute(0, 2, 3, 1)        # (B, N, N, H)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 10️⃣ Second LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out = out_fp16.float()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 11️⃣ Apply output gate (cast to float32 for stability)
    # ------------------------------------------------------------------
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 12️⃣ Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```
