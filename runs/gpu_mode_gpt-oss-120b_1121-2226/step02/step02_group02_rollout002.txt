"""
TriMul (outgoing) – Triton‑fused forward pass
================================================

This implementation follows the reference PyTorch version but
accelerates the two compute‑heavy parts with tiny Triton kernels:

1. **Mask conversion** – a simple kernel that casts a flat fp32 mask
   to fp16 (required because the rest of the arithmetic is performed
   in half precision).

2. **Batched matrix multiplication** – computes the core N³ operation
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   as a batched GEMM `L @ Rᵀ` where each hidden‑channel `d` is treated as
   an independent batch.  The kernel loads tiles of size
   (BLOCK_M × BLOCK_K) and (BLOCK_K × BLOCK_N), accumulates in fp32,
   and stores the result as fp32.  This eliminates the large
   `torch.bmm` call and lets us tune tile sizes (default:
   BLOCK_M = BLOCK_N = 128, BLOCK_K = 64) for H100 tensor‑core
   performance.

All other lightweight operations (layer‑norm, linear layers,
sigmoid gating, final projection) are kept in PyTorch, which incurs
minimal overhead compared to the N³ step.

The final output is a float‑32 tensor of shape
`[bs, seq_len, seq_len, dim]`.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Kernel 1: fp32 -> fp16 mask conversion (used only when mask is present)
# ---------------------------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,       # fp32 source pointer
    dst_ptr,       # fp16 destination pointer
    numel,         # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)


# ---------------------------------------------------------------------------
# Kernel 2: batched matmul C = A @ Bᵀ  (A,B half‑prec, C fp32)
#   A: (batch, M, K)    fp16
#   B: (batch, N, K)    fp16   (we read it as K x N)
#   C: (batch, M, N)    fp32
# ---------------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_ah, stride_am, stride_ak,   # A strides
    stride_bh, stride_bn, stride_bk,   # B strides (original B shape (batch,N,K))
    stride_ch, stride_cm, stride_cn,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col
    pid_b = tl.program_id(2)   # batch (including hidden dim)

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            A_ptr
            + pid_b * stride_ah
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=mask_a,
            other=0.0,
            dtype=tl.float16,
        )
        # Load B tile (as transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            B_ptr
            + pid_b * stride_bh
            + offs_k[:, None] * stride_bk          # k dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bn,         # n dimension (col of Bᵀ)
            mask=mask_b,
            other=0.0,
            dtype=tl.float16,
        )
        # Dot product: half·half → fp32 accumulation
        acc += tl.dot(a, b)

    # Write back result tile (fp32)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn,
        acc,
        mask=mask_c,
    )


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor,
                          BLOCK_M: int = 128, BLOCK_N: int = 128,
                          BLOCK_K: int = 64) -> torch.Tensor:
    """
    Compute C = A @ Bᵀ for 3‑D tensors using the Triton kernel above.

    Arguments
    ---------
    A : torch.Tensor (batch, M, K)   fp16
    B : torch.Tensor (batch, N, K)   fp16
    Returns
    -------
    C : torch.Tensor (batch, M, N)   fp32
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element counts (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ---------------------------------------------------------------------------
# Main entry point required by the harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim] (float32)
        - mask         : torch.Tensor [bs, seq_len, seq_len] (float32)
        - weights      : dict of model weights (float32 tensors)
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.half()   # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 3) Cast linear weights to fp16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()
    # Output‑norm and final linear stay in fp32 (as in reference)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # fp32

    # ------------------------------------------------------------------
    # 4) Left / Right linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B,N,N,hidden_dim] fp16
    right = F.linear(x_h, right_proj_w)  # [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcasted over hidden_dim)
    #    Convert mask to fp16 via a tiny Triton kernel.
    # ------------------------------------------------------------------
    if not nomask:
        # Flatten fp32 mask -> fp16 buffer
        mask_flat = mask.view(-1)
        mask_h_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)

        BLOCK_SIZE = 1024
        grid = ((mask_flat.numel() + BLOCK_SIZE - 1) // BLOCK_SIZE, )
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_h_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Reshape to [B,N,N,1] and broadcast over hidden_dim
        mask_h = mask_h_flat.view(mask.shape + (1,))
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6) Gating – sigmoid on half‑precision tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))    # [B,N,N,hidden_dim] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))   # [B,N,N,hidden_dim] fp16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B,N,N,hidden_dim] fp16

    # Apply element‑wise gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core N³ operation: batched matmul L @ Rᵀ
    #    Each hidden channel is treated as an independent batch.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # B, seq_len, seq_len, hidden_dim
    # Move hidden dimension into batch and make tensors contiguous
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Triton batched GEMM: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    # Tuned tile sizes work well on H100.
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )  # fp32 output, shape (B*H, N, N)

    # Reshape back to [B,N,N,hidden_dim] (float32)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim] fp32

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp32

    # ------------------------------------------------------------------
    # 9) Output gating & final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # element‑wise gating
    out = F.linear(out, to_out_w)            # fp32 → (B,N,N,dim)

    return out
