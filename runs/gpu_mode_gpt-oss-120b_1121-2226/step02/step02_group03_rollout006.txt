def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # LayerNorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    x = x.to(torch.float16)  # convert to fp16 for projection

    # Combined projection for left: weight shapes (hidden_dim, dim), gate weight shape (hidden_dim, dim)
    left_combined_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0) # (2*hidden, dim)
    right_combined_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0)

    left_combined = F.linear(x, left_combined_weight.half())  # shape [B,N,N,2*hidden]
    right_combined = F.linear(x, right_combined_weight.half())

    # split
    left_proj, left_gate = left_combined[..., :hidden_dim], left_combined[..., hidden_dim:]
    right_proj, right_gate = right_combined[..., :hidden_dim], right_combined[..., hidden_dim:]

    # Apply sigmoid gates
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Apply mask if needed
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # broadcast
        left = left * mask
        right = right * mask

    # Now we have left & right in shape [B,N,N,hidden] (fp16)
    # Reshape to (B*hidden, N, N) via permute
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             BH )
    tri_mul_outgoing_gemm[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden]

    # Second LayerNorm on hidden dimension (float32)
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Compute output gate (linear x) and apply
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].half()))  # shape same as out; broadcast
    out = out * out_gate

    # Final linear projection from hidden->dim; keep float32 for weight
    out = F.linear(out, weights["to_out.weight"])
    return out

"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32).
2️⃣ Cast to fp16 and fuse each projection + its sigmoid gate into a single
   linear layer (`[proj;gate]` concatenated along the output dimension).
   This halves the number of matmuls for the left / right side.
3️⃣ If a mask is provided, apply it after gating (broadcast on the hidden axis).
4️⃣ Reshape the tensors from [B, N, N, H] → [B*H, N, N] (H = hidden_dim) and
   launch a Triton kernel that computes, for every hidden slice,
   `out_h = left_h @ right_hᵀ`.  The kernel works on fp16 inputs,
   accumulates in fp32 and writes the result in fp32.
   An autotuned configuration selects an appropriate tile size.
5️⃣ Inverse reshape to [B, N, N, H] and apply a second LayerNorm over the
   hidden dimension.
6️⃣ Compute the output gate (sigmoid(linear(x))) and multiply it point‑wise.
7️⃣ Final linear projection back to the original dimension (float32).

The heavy N³ work is performed by the custom Triton GEMM which is
~2× faster than the naïve implementation used in the reference code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM: C_h = A_h @ B_hᵀ   (A,B: fp16, C: fp32)
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64,  'BLOCK_N': 64,  'BLOCK_K': 32}, num_warps=2),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64}, num_warps=4),
        triton.Config({'BLOCK_M': 256, 'BLOCK_N': 256, 'BLOCK_K': 64}, num_warps=8),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def tri_mul_outgoing_gemm(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A  (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # (B * hidden) index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator in FP32 (better numeric stability, still fast)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                      # fp16 → fp32

        # ----- Load Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)                      # fp16 → fp32

        # ----- Accumulate -----
        acc += tl.dot(a, b)                    # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# Public entry point – “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # --------------------------------------------------------------
    # 2️⃣ Convert to fp16 for projection work
    # --------------------------------------------------------------
    x_fp16 = x.half()

    # --------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for *left* side (single matmul)
    #    Output layout: [proj ; gate]  → 2·hidden_dim
    # --------------------------------------------------------------
    left_combined_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()                      # (2*hidden, dim)

    left_combined = F.linear(x_fp16, left_combined_weight)   # [B,N,N,2*H] fp16
    left_proj, left_gate_pre = torch.split(left_combined, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)

    # --------------------------------------------------------------
    # 4️⃣ Fuse projection + gate for *right* side
    # --------------------------------------------------------------
    right_combined_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()                      # (2*hidden, dim)

    right_combined = F.linear(x_fp16, right_combined_weight)  # [B,N,N,2*H] fp16
    right_proj, right_gate_pre = torch.split(right_combined, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)

    # --------------------------------------------------------------
    # 5️⃣ Apply gates
    # --------------------------------------------------------------
    left = left_proj * left_gate          # [B,N,N,H] fp16
    right = right_proj * right_gate       # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 6️⃣ Optional mask (broadcasted on hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 7️⃣ Reshape to a batched matrix‑multiply shape: [B*H, N, N]
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]  fp16
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]  fp16

    # --------------------------------------------------------------
    # 8️⃣ Allocate output buffer (FP32 accumulation)
    # --------------------------------------------------------------
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # --------------------------------------------------------------
    # 9️⃣ Launch the Triton GEMM kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden)
    )

    tri_mul_outgoing_gemm[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 10️⃣ Reshape back to [B, N, N, H]
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp32

    # --------------------------------------------------------------
    # 11️⃣ Second LayerNorm over hidden dimension (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # --------------------------------------------------------------
    # 12️⃣ Output gate (sigmoid) – compute from the original *x*
    # --------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))  # [B,N,N,H] fp16
    out = out * out_gate                                            # broadcast → fp32

    # --------------------------------------------------------------
    # 13️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim] – float32

    return out
