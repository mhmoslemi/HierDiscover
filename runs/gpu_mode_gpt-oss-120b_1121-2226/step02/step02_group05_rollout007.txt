"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the last dimension (dim) of the input (FP32 → FP16).
2️⃣  Linear projections (left/right) and three gating projections (FP16).
    An optional mask is applied after the projections.
3️⃣  Fuse gates into the projected tensors (still FP16).
4️⃣  Compute the cubic contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a *batched* matrix multiplication.
   The kernel works directly on the original 4‑D layout without any
   permute/contiguous copies:
      left  : (b,i,k,h) → matrix A (rows=i, cols=k)
      right : (b,j,k,h) → matrix Bᵀ (rows=k, cols=j)
   Each (b,h) pair is treated as an independent batch element.
5️⃣  Layer‑norm over the hidden dimension (FP32), apply the output gate,
    and a final linear projection back to `dim` (FP32).

The heavy N³ work is performed by a custom Triton kernel that:
- uses FP16 loads with FP32 accumulation,
- tiles the M/N dimensions (i/j) and the reduction K (k),
- merges the original batch and hidden dimensions into a single program‑id
  dimension (batch*hidden) and computes the hidden index on‑the‑fly,
- avoids any extra memory copies or transposes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A: (B, M, K), B: (B, N, K) → C: (B, M, N))
# The kernel works on the original 4‑D layout (B, N, N, H) without
# permutation – strides are passed explicitly.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr,                 # left  tensor pointer  (B, N, N, H)
    b_ptr,                 # right tensor pointer  (B, N, N, H)
    c_ptr,                 # output tensor pointer (B, N, N, H)
    N,                     # sequence length
    hidden_dim,            # H
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_batch = tl.program_id(2)        # merged (batch * hidden) dimension
    pid_m = tl.program_id(0)            # tile row (i)
    pid_n = tl.program_id(1)            # tile col (j)

    # hidden index inside the merged batch dimension
    hid = pid_batch % hidden_dim
    batch = pid_batch // hidden_dim

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointer for this (batch, hidden) pair (the hidden stride in the
    # original layout is 1, therefore we simply add `hid`).
    base_a = batch * stride_a_batch + hid
    base_b = batch * stride_b_batch + hid
    base_c = batch * stride_c_batch + hid

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K (= N) in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A(i,k) : left  tensor, rows=i, cols=k
        a_ptrs = a_ptr + base_a \
                + offs_m[:, None] * stride_a_row \
                + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B(k,j) : right tensor transposed, rows=k, cols=j
        b_ptrs = b_ptr + base_b \
                + offs_k[:, None] * stride_b_row \
                + offs_n[None, :] * stride_b_col
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) --> (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write out the tile
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + base_c \
             + offs_m[:, None] * stride_c_row \
             + offs_n[None, :] * stride_c_col
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication for the TriMul contraction.
    left/right: shape (B, N, N, H)  (FP16, contiguous)
    Returns: (B, N, N, H)  (FP16)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # FP16 output

    # Choose tile sizes – simple heuristic
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Number of tiles
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N

    # Strides (in elements) for the original layout (B, N, N, H)
    stride_a_batch = left.stride(0)
    stride_a_row   = left.stride(1)   # stride for dimension i
    stride_a_col   = left.stride(2)   # stride for dimension k

    # Right is read in transposed order (k, j)
    stride_b_batch = right.stride(0)
    stride_b_row   = right.stride(2)   # stride for k (original 3rd dim)
    stride_b_col   = right.stride(1)   # stride for j (original 2nd dim)

    # Output layout identical to left
    stride_c_batch = out.stride(0)
    stride_c_row   = out.stride(1)
    stride_c_col   = out.stride(2)

    grid = (num_m, num_n, B * H)

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        N,
        H,
        stride_a_batch,
        stride_a_row,
        stride_a_col,
        stride_b_batch,
        stride_b_row,
        stride_b_col,
        stride_c_batch,
        stride_c_row,
        stride_c_col,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point – fully differentiable (forward only) TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 2️⃣ load model parameters (cast to fp16 where safe)
    # ------------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (no bias)
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    # Gating projections
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # Output LayerNorm + linear
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # keep FP32 for stability

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → half‑precision for heavy ops
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                               # [B, N, N, dim]  float32
    x_h = x.half()                  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden] fp16
    right = F.linear(x_h, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B,N,N,1] fp16
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Cubic contraction via Triton (batched GEMM)
    # ------------------------------------------------------------------
    # left/right shape: (B, N, N, H) – already in the layout required by the kernel
    out_fp16 = _batched_matmul_triton(left, right)      # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 8️⃣ Post‑processing (float32)
    # ------------------------------------------------------------------
    out = out_fp16.float()                              # cast back to FP32

    # Second LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # Apply output gate (casted to FP32 for stability)
    out = out * out_gate.float()

    # Final projection back to `dim` (float32)
    out = F.linear(out, to_out_w)   # [B, N, N, dim]  float32

    return out
