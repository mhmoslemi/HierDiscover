block_n = block_idx % num_n

# compute block offset for i, j

offs_j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

mask_j = offs_j < N

"""
custom_kernel – Triton‑accelerated outgoing TriMul

Algorithm
---------
1️⃣ Layer‑norm over the input channel `dim` (FP32).
2️⃣ Cast to fp16 and apply five bias‑free linear projections:
   left_proj, right_proj, left_gate, right_gate, out_gate.
   The gates are passed through a sigmoid.
3️⃣ Optional mask (broadcasted on the hidden dim) is applied before gating.
4️⃣ The core N³ contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is implemented as a batched fp16 GEMM (A @ Bᵀ) using a custom Triton
   kernel.  The hidden dimension is folded into the batch dimension
   (“batch_h = B·H”) to avoid any data layout transformations.
   The kernel tiles the multiplication with BLOCK_M/N/K = 64 and
   accumulates in FP32 before casting back to FP16.
5️⃣ Convert the result to FP32, apply a second LayerNorm over `hidden_dim`,
   multiply by the output gate, and finally project back to `dim`
   with a bias‑free linear layer (FP32).

Only the contraction (step 4) is written in Triton; all other ops
use highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul contraction
#   left : (B, N, N, H)  fp16
#   right: (B, N, N, H)  fp16 – used transposed on the second N axis
#   out  : (B, N, N, H)  fp16  (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # problem size
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 3‑D grid decomposition: (batch_h, block_m, block_n)
    # --------------------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch_h = pid // (num_m * num_n)               # 0 .. B*H-1
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # ------------------------------------------------------------------
    # Offsets for the M and N dimensions
    # ------------------------------------------------------------------
    offs_i = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Derive the concrete batch (b) and hidden (h) indices
    # ------------------------------------------------------------------
    b = batch_h // H
    h = batch_h % H

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (the summed axis)
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b,i,k,h]  → shape (BLOCK_M, BLOCK_K)
        a_ptrs = (left_base
                  + offs_i[:, None] * stride_left_i
                  + offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[b,j,k,h] – we need it as (BLOCK_K, BLOCK_N)
        b_ptrs = (right_base
                  + offs_k[:, None] * stride_right_k
                  + offs_j[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # GEMM tile
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N), FP32

    # ------------------------------------------------------------------
    # Write back the FP16 result
    # ------------------------------------------------------------------
    c_ptrs = (out_base
              + offs_i[:, None] * stride_out_i
              + offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _batched_trimul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction for the outgoing TriMul:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left, right : torch.Tensor
        Shape (B, N, N, H), dtype=torch.float16, contiguous.

    Returns
    -------
    torch.Tensor
        Shape (B, N, N, H), dtype=torch.float16, contiguous.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, N2, H = left.shape
    assert N == N2 and right.shape == (B, N, N, H)

    out = torch.empty_like(left)

    # Tuned tile sizes for Hopper (fp16×fp16→fp32 Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batch = B * H
    grid = (total_batch * num_m * num_n,)

    # Strides (in elements) for each tensor dimension
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,   # j‑dimension stride is the same as the first N‑axis
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensors (see skeleton)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Model weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]           # keep FP32

    # --------------------------------------------------------------
    # 1️⃣ Layer‑norm over the input channel dimension (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim], fp16

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) in FP16
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted on hidden dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched contraction via Triton (fp16 → fp32 accumulation)
    # --------------------------------------------------------------
    out_h = _batched_trimul_fp16(left, right)   # [B, N, N, hidden_dim], fp16

    # --------------------------------------------------------------
    # 7️⃣ Back to FP32 for the second LayerNorm
    # --------------------------------------------------------------
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Multiply by the output gate (broadcasted)
    # --------------------------------------------------------------
    out = out * out_gate   # out_gate is fp16 -> upcasted automatically to fp32

    # --------------------------------------------------------------
    # 🔟 Final projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim], FP32

    return out
