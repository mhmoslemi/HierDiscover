def tri_mul_outgoing_kernel(
    left_ptr,
    right_ptr,
    out_ptr,
    B, N, H,
    stride_batch, stride_i, stride_k,
    stride_batch_r, stride_k_r, stride_j,
    stride_batch_out, stride_i_out, stride_j_out,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # pid for block in M,N
    pid = tl.program_id(0)
    # pid for batch*h dimension
    batch_h_pid = tl.program_id(1)
    batch = batch_h_pid // H
    h = batch_h_pid % H

    # compute tile indices
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    block_m = pid // num_n
    block_n = pid % num_n

    # offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers for this batch and hidden offset
    a_base = left_ptr + batch * stride_batch + h
    b_base = right_ptr + batch * stride_batch_r + h
    # out base
    c_base = out_ptr + batch * stride_batch_out + h

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: rows offs_m, cols offs_k
        a_ptrs = a_base + offs_m[:, None] * stride_i + offs_k[None, :] * stride_k

    offs_j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer for this (batch, h)
    a_base = left_ptr + batch * stride_batch_l + h
    b_base = right_ptr + batch * stride_batch_r + h
    c_base = out_ptr + batch * stride_batch_o + h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load A tile: (i, k)
        a_ptrs = a_base + offs_i[:, None] * stride_i_l + offs_k[None, :] * stride_k_l
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load B tile: (k, j) – note B is used as transposed
        b_ptrs = b_base + offs_k[:, None] * stride_k_r + offs_j[None, :] * stride_j_r
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product (FP16×FP16 → FP32 accumulator)
        acc += tl.dot(a, b)

    c_ptrs = c_base + offs_i[:, None] * stride_i_o + offs_j[None, :] * stride_j_o
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

    """
    Triton-accelerated outgoing TriMul forward pass.
       1. LayerNorm on input (float32).
    2. FP16 linear projections + optional mask.
    3. Sigmoid gates.
    4. Batched contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       performed by a custom Triton kernel that reads directly from the
       (B,N,N,H) layout without any reshaping or copies.
    5. FP32 LayerNorm on hidden dimension.
    6. Multiply by out_gate.
    7. Final linear projection back to `dim`.
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ---- weights -------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ---- 1. layer norm -------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---- 2. cast to fp16 ------------------------------------------------
    x_h = x.half()   # [B, N, N, dim] in fp16

    # ---- 3. linear projections (no bias) -------------------------------
    left = F.linear(x_h, left_proj_w)   # [B,N,N,hidden] fp16
    right = F.linear(x_h, right_proj_w)

    # ---- 4. mask (optional) -------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ---- 5. gating -------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ---- 6. contraction -------------------------------------------------
    B = left.shape[0]
    N = left.shape[1]
    H = hidden_dim

    # allocate output tensor in fp16
    out_fp16 = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # strides (in elements, not bytes)
    stride_batch_l = left.stride(0)
    stride_i_l     = left.stride(1)    # N*H
    stride_k_l     = left.stride(2)    # H

    stride_batch_r = right.stride(0)
    stride_k_r     = right.stride(2)   # H
    stride_j_r     = right.stride(1)   # N*H

    stride_batch_o = out_fp16.stride(0)
    stride_i_o     = out_fp16.stride(1)
    stride_j_o     = out_fp16.stride(2)

    # tile sizes (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m * num_n, B * H)

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out_fp16,
        B,
        N,
        H,
        stride_batch_l,
        stride_i_l,
        stride_k_l,
        stride_batch_r,
        stride_k_r,
        stride_j_r,
        stride_batch_o,
        stride_i_o,
        stride_j_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,   # 16 warps = 512 threads, good for 128×128 tiles on H100
    )

    # ---- 7. convert back to fp32 ----------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ---- 9. apply output gate --------------------------------------------
    out = out * out_gate

    # ---- 10. final linear projection ------------------------------------

    return out

"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑norm on the input (float32).
2. Cast to fp16 and apply four bias‑free linear projections:
     left, right, left_gate, right_gate, out_gate.
   An optional mask (B,N,N) is broadcasted and applied before the gates.
   All linear work stays in fp16.
3. Elementwise sigmoid gating and in‑place multiplication.
4. Core contraction  (the N³ part):
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that reads the native
   (B,N,N,H) layout (no transposes or copies).  The kernel treats each
   (batch, hidden) pair as an independent “batch” and performs a tiled
   fp16×fp16→fp32 GEMM with 128×128×64 tiles (16 warps per block).
5. Convert the result to fp32, Layer‑norm over the hidden dimension,
   multiply by the previously‑computed out_gate, and finally apply the
   output linear projection back to `dim`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#   left  : (B, N, N, H)   fp16
#   right : (B, N, N, H)   fp16
#   out   : (B, N, N, H)   fp16   (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_batch_l, stride_i_l, stride_k_l,
    stride_batch_r, stride_k_r, stride_j_r,
    stride_batch_o, stride_i_o, stride_j_o,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 2‑D program ids: (tile over i/j), (batch × hidden)
    # ------------------------------------------------------------------
    pid_tile   = tl.program_id(0)      # (block_m, block_n) tile
    pid_batchh = tl.program_id(1)      # combined batch * hidden index

    # decode batch and hidden index
    batch = pid_batchh // H
    h     = pid_batchh % H

    # ------------------------------------------------------------------
    # Tile coordinates in the output matrix (i‑j plane)
    # ------------------------------------------------------------------
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    block_m = pid_tile // num_n
    block_n = pid_tile %  num_n

    # --------------------------------------------------------------
    # Offsets and bounds for each tile
    # --------------------------------------------------------------
    offs_i = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, hidden) pair
    # ------------------------------------------------------------------
    a_base = left_ptr  + batch * stride_batch_l + h
    b_base = right_ptr + batch * stride_batch_r + h
    c_base = out_ptr   + batch * stride_batch_o + h

    # accumulator in FP32 (numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k in tiles of size BLOCK_K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: left[batch, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_i_l + offs_k[None, :] * stride_k_l
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # B tile: right[batch, j, k, h] used as transposed (k, j)
        #        -> shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_k_r + offs_j[None, :] * stride_j_r
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # FP16 dot, accumulate in FP32
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result back to out[batch, i, j, h]
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_i_o + offs_j[None, :] * stride_j_o
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass for the *outgoing* TriMul operator, fully fused with
    a Triton kernel for the N³ contraction.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor            shape [B, N, N, dim]
        - mask         : torch.Tensor or None    shape [B, N, N]
        - weights      : dict of torch.Tensor       (see keys below)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Model weights (cast to the precision used in the heavy path)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # kept FP32

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input dimension `dim` (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.half()                       # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – all in fp16
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ The N³ contraction via Triton (out = Σ_k left * right)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    H = hidden_dim

    out_fp16 = torch.empty((B, N, N, H), dtype=torch.float16, device=device)

    # Strides (in *elements*, not bytes)
    stride_batch_l = left.stride(0)     # B‑stride
    stride_i_l     = left.stride(1)     # i‑stride = N * H
    stride_k_l     = left.stride(2)     # k‑stride = H

    stride_batch_r = right.stride(0)
    stride_k_r     = right.stride(2)    # row‑stride in transposed view = H
    stride_j_r     = right.stride(1)    # col‑stride = N * H

    stride_batch_o = out_fp16.stride(0)
    stride_i_o     = out_fp16.stride(1)
    stride_j_o     = out_fp16.stride(2)

    # Tile sizes – tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N

    # 2‑D launch grid:
    #   axis 0 -> tiles over (i, j)
    #   axis 1 -> combined batch × hidden index
    grid = (num_m * num_n, B * H)

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out_fp16,
        B,
        N,
        H,
        stride_batch_l,
        stride_i_l,
        stride_k_l,
        stride_batch_r,
        stride_k_r,
        stride_j_r,
        stride_batch_o,
        stride_i_o,
        stride_j_o,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,          # 16 warps = 512 threads, good for 128×128 tiles on H100
    )

    # ------------------------------------------------------------------
    # 7️⃣ Convert back to FP32 and second LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate (broadcasted) and final projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_w)          # [B, N, N, dim]   FP32

    return out
