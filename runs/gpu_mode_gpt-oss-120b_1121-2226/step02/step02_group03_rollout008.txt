"""
TriMul “outgoing” implementation (forward only)

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input tensor (float32).
2️⃣ Cast to fp16 and fuse the two linear projections (proj & gate) for the
   left and right branches:
        [proj ; gate] = x @ [W_proj ; W_gate]^T
   The concatenated matmul replaces two separate linear layers.
3️⃣ Split the concatenated result, apply sigmoid to the gate part and
   multiply:  left = proj_left * sigmoid(gate_left)
              right = proj_right * sigmoid(gate_right)
   If a mask is provided, broadcast‑multiply `left` and `right`.
4️⃣ Rearrange to (B·H, N, N) (contiguous) and launch a custom Triton kernel
   that computes, for each (b, h) pair:

        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

   i.e. a batched GEMM A @ Bᵀ with A,B in fp16 and the accumulator in fp32.
   Block sizes (128×128×64) are chosen for H100 Tensor‑Core friendly
   FP16×FP16→FP32 execution.
5️⃣ Reshape the result back to (B, N, N, H) (float32).
6️⃣ Apply a second LayerNorm over the hidden dimension.
7️⃣ Compute the output gate (sigmoid of a linear on the normalized input)
   and multiply it with the tensor from step 6.
8️⃣ Final linear projection back to the original dimension `dim`.

The heavy N³ work is performed entirely inside the Triton kernel, while all
other operations stay in PyTorch (layer‑norm, linear, sigmoid, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ   (A,B are fp16, C fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers (half for A/B, float for C)
    M, N, K,                           # problem size (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A  (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ Bᵀ for many independent (batch*head) problems."""
    pid_h = tl.program_id(axis=2)          # (batch * hidden) index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # --------------------------------------------------------------
    # Tile offsets & masks
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)  # fp16 → fp32

        # ----- Load Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)  # fp16 → fp32

        # ----- Accumulate -----
        acc += tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write the result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # input / output feature dim
    hidden_dim = config["hidden_dim"]   # inner hidden dimension
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for left and right branches
    # ------------------------------------------------------------------
    # concatenate weights: [proj ; gate]  → (2*hidden, dim)
    left_w_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()
    right_w_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()

    # matmul: (B,N,N,2*hidden) in fp16
    left_cat = F.linear(x_fp16, left_w_concat)   # → [B, N, N, 2*hidden]
    right_cat = F.linear(x_fp16, right_w_concat) # → [B, N, N, 2*hidden]

    # split
    left_proj = left_cat[..., :hidden_dim]               # [B,N,N,hidden]
    left_gate_pre = left_cat[..., hidden_dim:]           # [B,N,N,hidden]
    right_proj = right_cat[..., :hidden_dim]
    right_gate_pre = right_cat[..., hidden_dim:]

    # sigmoid gates (still fp16)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    # apply gates
    left = left_proj * left_gate      # fp16
    right = right_proj * right_gate   # fp16

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Output gate (sigmoid) – independent of the heavy GEMM
    # ------------------------------------------------------------------
    out_gate_weight = weights["out_gate.weight"].half()
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_weight))  # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Rearrange for the batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B,N,N,hidden) → (B,hidden,N,N) → contiguous → (B*hidden,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]  fp16, contiguous
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]  fp16, contiguous

    # output buffer (fp32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Launch the Triton kernel (batched GEMM)
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # batch × hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                         # M
        N,                         # N
        N,                         # K
        left_flat.stride(0),  # stride_ah (batch*hidden)
        left_flat.stride(1),  # stride_ai (i)
        left_flat.stride(2),  # stride_ak (k)
        right_flat.stride(0), # stride_bh (batch*hidden)
        right_flat.stride(2), # stride_bk (k)   – note the transposed access
        right_flat.stride(1), # stride_bn (j)
        out_flat.stride(0),   # stride_ch
        out_flat.stride(1),   # stride_ci
        out_flat.stride(2),   # stride_cj
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H] (float32)

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection
    # ------------------------------------------------------------------
    # out_gate is fp16 → upcast automatically
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # → [B, N, N, dim] (float32)

    return out
```
