@triton.jit
def _trimul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute total tiles per batch
    tiles_m = tl.cdiv(N, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_bn = tiles_m * tiles_n

    batch_hidden = pid // tiles_per_bn
    tile_id = pid % tiles_per_bn

    block_m = tile_id // tiles_n
    block_n = tile_id % tiles_n

    # compute batch indices
    h = batch_hidden % H
    b = batch_hidden // H

    # pointer base offsets
    a_base = a_ptr + b * stride_a_batch + h * 1
    b_base = b_ptr + b * stride_b_batch + h * 1
    c_base = c_ptr + b * stride_c_batch + h * 1

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel – Triton‑accelerated *outgoing* TriMul

Algorithm
---------
1. Layer‑norm on the input tensor (float‑32).
2. Cast to FP16 and apply four bias‑free linear projections:
     left, right, left_gate, right_gate, out_gate.
   The three gates are passed through a sigmoid.
   An optional mask (B,N,N) is broadcasted and applied to left/right
   before gating.
3. Compute the contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication of shape
        (B·D, N, N) @ (B·D, N, N) → (B·D, N, N)
   where D = hidden_dim.
   The kernel accesses the original [B,N,N,D] layout directly,
   avoiding the huge (B·D·N·N) temporary created by a permute‑copy.
   It tiles the multiplication with BLOCK_M = BLOCK_N = BLOCK_K = 64
   and uses Tensor‑cores (FP16×FP16→FP32) with 8 warps per block.
4. Convert the result back to FP32 and apply a second LayerNorm
   over the hidden dimension.
5. Multiply by the output‑gate (broadcasted) and project back to
   the original dimension with a final FP32 linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched contraction for outgoing TriMul
#   A: left  tensor, shape (B, N, N, D)   (FP16)
#   B: right tensor, shape (B, N, N, D)   (FP16)
#   C: output tensor, shape (B, N, N, D) (FP16, acc in FP32)
#   The contraction is:
#       C[b,i,j,d] = Σ_k A[b,i,k,d] * B[b,j,k,d]
#   We treat each (b,d) as a separate batch and use custom strides
#   to read the original layout without any transpose/copy.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                    # tensor pointers
    B, N, D,                                 # dimensions
    stride_a_batch, stride_a_i, stride_a_k,  # left strides
    stride_b_batch, stride_b_k, stride_b_j,  # right strides (transposed view)
    stride_c_batch, stride_c_i, stride_c_j,  # output strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    #  Tile decomposition: each program handles a (batch*d) tile (i,j)
    # --------------------------------------------------------------
    tiles_m = tl.cdiv(N, BLOCK_M)
    tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_bn = tiles_m * tiles_n               # tiles for one (b,d)

    batch_hidden = pid // tiles_per_bn            # index over B*D
    tile_id       = pid % tiles_per_bn

    block_m = tile_id // tiles_n
    block_n = tile_id % tiles_n

    # --------------------------------------------------------------
    #  Recover original batch (b) and hidden index (d)
    # --------------------------------------------------------------
    d = batch_hidden % D
    b = batch_hidden // D

    # Base pointers for this (b,d)
    a_base = a_ptr + b * stride_a_batch + d * 1
    b_base = b_ptr + b * stride_b_batch + d * 1
    c_base = c_ptr + b * stride_c_batch + d * 1

    # --------------------------------------------------------------
    #  Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 (Tensor‑core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over the reduction dimension K (= N) in BLOCK_K steps
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load a tile of left:  A[b,i,k,d]
        a_ptrs = (a_base
                  + offs_m[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load a tile of right (transposed view): B[b,k,j,d]
        b_ptrs = (b_base
                  + offs_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)        # FP16×FP16 → FP32

    # --------------------------------------------------------------
    #  Store the result tile
    # --------------------------------------------------------------
    c_ptrs = (c_base
              + offs_m[:, None] * stride_c_i
              + offs_n[None, :] * stride_c_j)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _trimul_outgoing_fp16(left: torch.Tensor,
                          right: torch.Tensor,
                          hidden_dim: int) -> torch.Tensor:
    """
    Batched contraction for the outgoing TriMul.
    left, right – tensors of shape [B, N, N, D] (FP16, contiguous)
    returns tensor of the same shape (FP16).
    """
    B, N, _, D = left.shape
    assert D == hidden_dim
    assert left.is_contiguous() and right.is_contiguous()
    out = torch.empty_like(left)

    # Block sizes tuned for H100 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    tiles_m = (N + BLOCK_M - 1) // BLOCK_M
    tiles_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * D * tiles_m * tiles_n

    # Strides (in elements, not bytes)
    stride_a_batch = left.stride(0)   # B‑stride = N*N*D
    stride_a_i     = left.stride(1)   # i‑stride = N*D
    stride_a_k     = left.stride(2)   # k‑stride = D

    # Right tensor will be read as transposed (k, j)
    stride_b_batch = right.stride(0)  # same as left
    stride_b_k     = right.stride(2)  # row‑stride = D
    stride_b_j     = right.stride(1)  # col‑stride = N*D

    stride_c_batch = out.stride(0)
    stride_c_i     = out.stride(1)
    stride_c_j     = out.stride(2)

    _trimul_outgoing_kernel[(total_tiles,)](
        left,
        right,
        out,
        B,
        N,
        D,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_j,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor            shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None    shape [B, N, N]   (bool/float)
        - weights      : dict of torch.Tensor with the keys used below
        - config       : dict with entries "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # model weights (cast to the precision used in each stage)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # FP32 (final projection)

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the input `dim` (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_h = x.half()                                         # [B, N, N, dim]

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) in FP16
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over the hidden dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)          # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply the gates
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched contraction: Σ_k left[...,k,:] * right[...,k,:]
    #    Performed by a custom Triton kernel that works on the
    #    original [B,N,N,hidden] layout (no permute+copy).
    # --------------------------------------------------------------
    out_h = _trimul_outgoing_fp16(left, right, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Convert to FP32 for the second LayerNorm
    # --------------------------------------------------------------
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # --------------------------------------------------------------
    # 9️⃣ Apply the output gate (broadcasted)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)      # [B, N, N, dim] (float32)

    return out
