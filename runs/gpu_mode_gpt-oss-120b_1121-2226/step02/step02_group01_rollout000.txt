"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. **First LayerNorm** on the 4‑D input (float32).
2. Cast to fp16 and apply three linear projections (left/right) and three
   gating linear layers (no bias).  The optional mask and the sigmoid gates
   are fused in half‑precision.
3. **Core cubic update**:
      out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
   is a batched matrix multiplication `L @ Rᵀ` where each hidden channel is an
   independent batch element.  We reshape `(B,N,N,H)` → `(B·H,N,N)` and run a
   custom Triton kernel that computes a batch of `(M×K)·(N×K)ᵀ` products with
   fp16 inputs and fp32 accumulation (Tensor‑Core friendly).
4. Reshape back to `(B,N,N,H)` and apply a second LayerNorm.
5. Multiply by the output gate and finish with a linear projection back to
   the original feature dimension (float32).

The Triton kernel is a tiled matmul with configurable block sizes.
Larger tiles (128×128) are selected for big sequence lengths to improve
arithmetic intensity on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers
    M, N, K,                      # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of matrices using tiled GEMM."""
    pid_m = tl.program_id(0)  # tile index in M dimension
    pid_n = tl.program_id(1)  # tile index in N dimension
    pid_b = tl.program_id(2)  # batch (i.e. hidden‑channel) index

    # ------------------------------------------------------------------
    # Offsets for the M and N tiles
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator in fp32 (Tensor‑Core mixed‑precision accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # Guard against out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Mixed‑precision dot: fp16 * fp16 → fp32 accumulation
        acc += tl.dot(a, b)

        # Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper around the Triton kernel for batched matrix multiplication.

    Arguments
    ---------
    A, B : torch.Tensor
        Shapes (batch, M, K) and (batch, N, K) respectively.
        Must be on CUDA, dtype=torch.float16.
    BLOCK_M, BLOCK_N, BLOCK_K : int
        Tiling parameters.

    Returns
    -------
    C : torch.Tensor
        Shape (batch, M, N), dtype=torch.float32.
    """
    assert A.is_cuda and B.is_cuda, "Inputs must be CUDA tensors"
    assert A.dtype == torch.float16 and B.dtype == torch.float16

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) – may be
                         ignored when ``config["nomask"]`` is True.
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : must contain ``"dim"``, ``"hidden_dim"``,
                         optionally ``"nomask"`` (bool).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Cast to fp16 for the heavy computation path
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left = F.linear(x_h, weights["left_proj.weight"].half())   # [B,N,N,H]
    right = F.linear(x_h, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 3) Optional masking (still fp16)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(x_h.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – fp16
    # ------------------------------------------------------------------
    left_gate = F.linear(x_h, weights["left_gate.weight"].half()).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"].half()).sigmoid()
    out_gate = F.linear(x_h, weights["out_gate.weight"].half()).sigmoid()

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core cubic update – batched matmul via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # Collapse the hidden dimension into the batch dimension:
    # (B, N, N, H) -> (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).view(B * H, N, N)  # fp16

    # Choose tile sizes based on sequence length (larger tiles for larger N)
    if N <= 384:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 64

    # Batched matmul: left_mat @ right_matᵀ  →  (B*H, N, N)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # fp32

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 8) Output gating and final projection back to the original dim
    # ------------------------------------------------------------------
    out = out * out_gate.float()                    # broadcast over dim
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim] (float32)

    return out
