def gating_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    B, N, H,
    stride0_lp, stride1_lp, stride2_lp, stride3_lp,
    stride0_rp, stride1_rp, stride2_rp, stride3_rp,
    stride0_lg, stride1_lg, stride2_lg, stride3_lg,
    stride0_rg, stride1_rg, stride2_rg, stride3_rg,
    stride0_og, stride1_og, stride2_og, stride3_og,
    stride0_m, stride1_m, stride2_m,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(axis=0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    total_elements = B * N * N * H
    mask = offsets < total_elements

    # Compute multi-dim indices
    # Convert offset to (b,i,j,h) using strides: we can compute using division and modulo.

    # Use integer division:
    t = offsets
    h = t % H
    t = t // H
    j = t % N
    t = t // N
    i = t % N
    b = t // N

    # Now compute linear offsets for each pointer.
    lp_off = b * stride0_lp + i * stride1_lp + j * stride2_lp + h * stride3_lp
    rp_off = b * stride0_rp + i * stride1_rp + j * stride2_rp + h * stride3_rp
    lg_off = b * stride0_lg + i * stride1_lg + j * stride2_lg + h * stride3_lg
    rg_off = b * stride0_rg + i * stride1_rg + j * stride2_rg + h * stride3_rg
    og_off = b * stride0_og + i * stride1_og + j * stride2_og + h * stride3_og

    # load half

    right_proj = tl.load(right_proj_ptr + rp_off, mask=mask, other=0.0).to(tl.float32)

    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))

Implementation details for gating kernel:

```
@triton.jit
def gating_kernel_nomask(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr,
    B, N, H,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elements

    # load
    left = tl.load(left_ptr + offset, mask=mask, other=0.0).to(tl.float32)
    right = tl.load(right_ptr + offset, mask=mask, other=0.0).to(tl.float32)
    left_gate = tl.load(left_gate_ptr + offset, mask=mask, other=0.0).to(tl.float32)
    right_gate = tl.load(right_gate_ptr + offset, mask=mask, other=0.0).to(tl.float32)
    out_gate = tl.load(out_gate_ptr + offset, mask=mask, other=0.0).to(tl.float32)

    # sigmoid
    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

    # apply gating (no mask)
    left = left * left_gate
    right = right * right_gate

    # store back
    tl.store(left_ptr + offset, left.to(tl.float16), mask=mask)
    tl.store(right_ptr + offset, right.to(tl.float16), mask=mask)
    tl.store(out_gate_ptr + offset, out_gate.to(tl.float16), mask=mask)
```

@triton.jit
def gating_kernel_mask(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    out_gate_ptr,
    mask_ptr,
    B, N, H,
    total_elements: tl.constexpr,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < total_elements

    left = tl.load(left_ptr + offset, mask=mask, other=0.0).to(tl.float32)
    right = tl.load(right_ptr + offset, mask=mask, other=0.0).to(tl.float32)
    left_gate = tl.load(left_gate_ptr + offset, mask=mask, other=0.0).to(tl.float32)
    right_gate = tl.load(right_gate_ptr + offset, mask=mask, other=0.0).to(tl.float32)
    out_gate = tl.load(out_gate_ptr + offset, mask=mask, other=0.0).to(tl.float32)

    left_gate = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
    out_gate = 1.0 / (1.0 + tl.exp(-out_gate))

    # mask index = offset // H
    mask_idx = offset // H

    tl.store(left_ptr + offset, left.to(tl.float16), mask=mask)
    tl.store(right_ptr + offset, right.to(tl.float16), mask=mask)
    tl.store(out_gate_ptr + offset, out_gate.to(tl.float16), mask=mask)

"""
TriMul (outgoing) – fast Triton‑augmented implementation
========================================================

The forward pass consists of four stages:

1️⃣ Input Layer‑Norm (fp32 → fp16).
2️⃣ **Fused projection** – a single half‑precision GEMM computes the five
   linear maps (left_proj, right_proj, left_gate, right_gate, out_gate)
   in one shot.  The result is reshaped to [B,N,N,5,H].
3️⃣ **Element‑wise gating & optional masking** – a tiny Triton kernel
   fuses the three sigmoid operations, the mask broadcast and the
   multiplication `proj * gate`.  This reduces the element‑wise work
   from several separate kernels to a single memory‑bound pass.
4️⃣ **Cubic update** – the heavy `∑_k left[...,k] * right[...,k]`
   is performed by a batched GEMM (`torch.bmm`) which is executed by
   cuBLAS/Tensor‑Core on the H100 and therefore far faster than a
   handcrafted Triton matmul.  The result is layer‑normed, gated by
   `out_gate` and projected back to `dim`.

All heavy compute (the two GEMMs) runs in fp16 with fp32 accumulation,
final output is fp32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse sigmoid, mask and element‑wise scaling
# ----------------------------------------------------------------------
@triton.jit
def _gate_and_mask_kernel(
    left_ptr,          # [B,N,N,H]  – projection (will be overwritten)
    right_ptr,         # [B,N,N,H]  – projection (will be overwritten)
    left_gate_ptr,     # [B,N,N,H]  – raw gate logits
    right_gate_ptr,    # [B,N,N,H]  – raw gate logits
    out_gate_ptr,      # [B,N,N,H]  – raw gate logits (output after sigmoid)
    mask_ptr,          # [B,N,N]   – optional mask (broadcast over H)
    B, N, H,
    total_elem: tl.constexpr,
    BLOCK: tl.constexpr,
    HAS_MASK: tl.constexpr,   # 1 if mask is present, else 0
):
    pid = tl.program_id(0)
    off = pid * BLOCK + tl.arange(0, BLOCK)                # linear offset
    valid = off < total_elem

    # ----- load tensors -------------------------------------------------
    left = tl.load(left_ptr + off, mask=valid, other=0.0).to(tl.float32)
    right = tl.load(right_ptr + off, mask=valid, other=0.0).to(tl.float32)
    lg = tl.load(left_gate_ptr + off, mask=valid, other=0.0).to(tl.float32)
    rg = tl.load(right_gate_ptr + off, mask=valid, other=0.0).to(tl.float32)
    og = tl.load(out_gate_ptr + off, mask=valid, other=0.0).to(tl.float32)

    # ----- sigmoid -------------------------------------------------------
    lg = 1.0 / (1.0 + tl.exp(-lg))
    rg = 1.0 / (1.0 + tl.exp(-rg))
    og = 1.0 / (1.0 + tl.exp(-og))

    # ----- optional mask (broadcast over H) -----------------------------
    if HAS_MASK:
        mask_idx = off // H                     # each H‑slice shares the same mask entry
        m = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0).to(tl.float32)
        left = left * lg * m
        right = right * rg * m
    else:
        left = left * lg
        right = right * rg

    # ----- store back ----------------------------------------------------
    tl.store(left_ptr + off, left.to(tl.float16), mask=valid)
    tl.store(right_ptr + off, right.to(tl.float16), mask=valid)
    tl.store(out_gate_ptr + off, og.to(tl.float16), mask=valid)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (fp32) → fp16 for the heavy work
    # --------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B,N,N,dim]  fp32
    x_h = x.half()                               # fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5×hidden_dim outputs)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device)    # (5*H, dim)

    fused = F.linear(x_h, fused_w)               # [B,N,N,5*H] fp16
    B, N, _, _ = fused.shape

    # reshape to separate the five streams
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H] fp16
    left_proj  = fused[..., 0, :]                # [B,N,N,H]
    right_proj = fused[..., 1, :]
    left_gate_raw = fused[..., 2, :]
    right_gate_raw = fused[..., 3, :]
    out_gate_raw = fused[..., 4, :]

    # make all tensors contiguous for the Triton kernel
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate_raw = left_gate_raw.contiguous()
    right_gate_raw = right_gate_raw.contiguous()
    out_gate_raw = out_gate_raw.contiguous()

    # --------------------------------------------------------------
    # 3️⃣ Fuse sigmoid, mask and scaling (Triton)
    # --------------------------------------------------------------
    total_elem = B * N * N * hidden_dim
    BLOCK = 1024                         # ≤ 1024 threads / block
    grid = ( (total_elem + BLOCK - 1) // BLOCK, )

    if nomask:
        _gate_and_mask_kernel[grid](
            left_proj,
            right_proj,
            left_gate_raw,
            right_gate_raw,
            out_gate_raw,
            mask_ptr=mask,                # dummy pointer – never dereferenced
            B, N, hidden_dim,
            total_elem,
            BLOCK,
            HAS_MASK=0,
            num_warps=4,
            num_stages=2,
        )
    else:
        # mask → fp16, contiguous, broadcasted over hidden dimension
        mask_h = mask.to(dtype=torch.float16, device=device).contiguous()
        _gate_and_mask_kernel[grid](
            left_proj,
            right_proj,
            left_gate_raw,
            right_gate_raw,
            out_gate_raw,
            mask_h,
            B, N, hidden_dim,
            total_elem,
            BLOCK,
            HAS_MASK=1,
            num_warps=4,
            num_stages=2,
        )
    # after the kernel:
    #   left_proj  = left_proj * sigmoid(left_gate_raw) * mask (if any)
    #   right_proj = right_proj * sigmoid(right_gate_raw) * mask (if any)
    #   out_gate_raw = sigmoid(out_gate_raw)          (kept for later)

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM A @ Bᵀ   (torch.bmm → Tensor‑Core)
    # --------------------------------------------------------------
    # reshape for batched matmul: [B,H,N,N] → [B*H,N,N]
    left_flat = left_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_proj.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # (B*H,N,N) @ (B*H,N,N)ᵀ  →  (B*H,N,N)  (fp16, fp32 acc internally)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # --------------------------------------------------------------
    # 5️⃣ Reshape back to [B,N,N,H] (still fp16)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fp32)
    # --------------------------------------------------------------
    out_fp = out.float()
    out_fp = F.layer_norm(
        out_fp,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp = out_fp * out_gate_raw.float()        # apply output‑gate

    # --------------------------------------------------------------
    # 7️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(dtype=torch.float16, device=device)
    out_proj = F.linear(out_fp.half(), to_out_w)  # fp16
    out_final = out_proj.float()                  # fp32

    return out_final
```
