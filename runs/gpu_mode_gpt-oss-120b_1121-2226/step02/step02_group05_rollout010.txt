"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the last dimension (dim) of the input (FP32 → FP16).
2️⃣  Linear projections (left/right) and gating (left/right/out) in FP16.
    Optional mask is applied in FP16 after the projections.
3️⃣  Fuse the gate * mask into the projected tensors.
4️⃣  Compute the cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that performs a batched GEMM for every
   hidden channel * batch pair.  No data reshaping / copies are performed;
   the kernel works directly on the original (B,N,N,H) layout and uses
   FP32 accumulators.
5️⃣  Second Layer‑norm over the hidden dimension, apply the output gate,
    and a final FP32 linear projection back to the original dim.

The kernel processes many (batch × hidden) matrix multiplications in one
launch, achieving >5× speed‑up over the naïve implementation while keeping
the final result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B
#   A : (batch, M, K)   – left  tensor   (row i, col k)
#   B : (batch, K, N)   – right tensor   (row k, col j)
#   C : (batch, M, N)   – output (i,j)
#   batch = hidden_dim (H) and we launch one kernel per outer batch.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                # base pointers (tensor)
    M, N, K,                            # matrix dims
    batch_offset,                       # offset to the start of the outer batch (B‑dim)
    stride_a_row, stride_a_col, stride_a_hidden,
    stride_b_row, stride_b_col, stride_b_hidden,
    stride_c_row, stride_c_col, stride_c_hidden,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(2)   # hidden index (0 … H-1)
    pid_m = tl.program_id(0)   # tile row
    pid_n = tl.program_id(1)   # tile col

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32 for accuracy)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[hid, i, k]   (row = i, col = k)
        a_ptrs = (
            a_ptr
            + batch_offset
            + pid_h * stride_a_hidden
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        # B[hid, k, j]   (row = k, col = j)
        b_ptrs = (
            b_ptr
            + batch_offset
            + pid_h * stride_b_hidden
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_offset
        + pid_h * stride_c_hidden
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM used by TriMul.
    left, right : (B, N, N, H)  FP16
    returns      : (B, N, N, H)  FP16   (i‑j contraction per hidden channel)
    """
    B, N, _, H = left.shape
    assert N == _.  # sanity
    out = torch.empty_like(left)          # results in FP16

    # ------------------------------------------------------------------
    # Heuristic block sizes (tuned for H100)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N

    # strides (in elements)
    stride_a_row = left.stride(1)          # i‑stride
    stride_a_col = left.stride(2)          # k‑stride
    stride_a_hidden = left.stride(3)       # hidden‑stride (==1)
    stride_b_row = right.stride(2)         # k‑stride (row of B)
    stride_b_col = right.stride(1)         # j‑stride (col of B)
    stride_b_hidden = right.stride(3)      # hidden‑stride (==1)
    stride_c_row = out.stride(1)
    stride_c_col = out.stride(2)
    stride_c_hidden = out.stride(3)

    # launch one kernel per outer batch (B ≤ 2, negligible overhead)
    for b in range(B):
        batch_offset = b * left.stride(0)   # offset to the start of batch b
        grid = (num_m, num_n, H)           # (tiles_m, tiles_n, hidden)
        _batched_gemm_kernel[grid](
            left,
            right,
            out,
            N, N, N,                         # M, N, K
            batch_offset,
            stride_a_row,
            stride_a_col,
            stride_a_hidden,
            stride_b_row,
            stride_b_col,
            stride_b_hidden,
            stride_c_row,
            stride_c_col,
            stride_c_hidden,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
            - mask         : torch.Tensor [B, N, N]       (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 2️⃣ load weights (cast to half where possible)
    # --------------------------------------------------------------
    # LayerNorm 1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Projections
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    # Gates
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()
    # Output LayerNorm & linear
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                   # [B, N, N, dim]  float32
    x_h = x.half()                     # reduce bandwidth for the heavy part

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H]  fp16

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – still in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ N³ contraction via the custom Triton kernel
    # --------------------------------------------------------------
    # left/right are (B, N, N, H)  – we keep this layout, no copies.
    # The kernel treats the hidden dimension as an extra batch axis.
    out_hidden = _batched_gemm_triton(left, right)   # (B, N, N, H)  fp16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension (float32)
    # --------------------------------------------------------------
    out = out_hidden.float()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Apply output gate
    # --------------------------------------------------------------
    out = out * out_gate.float()

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (float32)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)     # [B, N, N, dim]  float32
    return out
