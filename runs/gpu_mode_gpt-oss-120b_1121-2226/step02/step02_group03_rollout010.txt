@triton.jit
def layernorm_fuse_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    eps: tl.constexpr,
    total_slices: tl.constexpr, hidden_dim: tl.constexpr,
    stride_xb, stride_xn, stride_xh,
    stride_ob, stride_on, stride_oh,
    str_w, str_b,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)   # slice index
    if pid >= total_slices: return
    # compute base pointers
    # compute coordinates: b,i,j -> can be computed using division/mod
    b = pid // (seq_len * seq_len)
    rem = pid % (seq_len * seq_len)
    i = rem // seq_len
    j = rem % seq_len

    # base pointer offset: base = x_ptr + b*stride_xb + i*stride_xn + j*stride_xh
    base_x = x_ptr + b*stride_xb + i*stride_xn + j*stride_xh
    base_out = out_ptr + b*stride_ob + i*stride_on + j*stride_oh

    # Compute sum and sum_sq across hidden dim
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    for start in range(0, hidden_dim, BLOCK_H):
        cur_h = start + tl.arange(0, BLOCK_H)
        mask = cur_h < hidden_dim
        x = tl.load(base_x + cur_h * stride_xh, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x)
        sum_sq += tl.sum(x * x)

    # compute mean and variance
    mean = sum_val / hidden_dim
    var = sum_sq / hidden_dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass to store normalized values
    for start in range(0, hidden_dim, BLOCK_H):
        cur_h = start + tl.arange(0, BLOCK_H)
        mask = cur_h < hidden_dim
        x = tl.load(base_x + cur_h * stride_xh, mask=mask, other=0.0).to(tl.float32)
        # normalize
        normalized = (x - mean) * inv_std
        # scale and bias
        w = tl.load(weight_ptr + cur_h * str_w, mask=mask, other=1.0).to(tl.float32)
        b = tl.load(bias_ptr + cur_h * str_b, mask=mask, other=0.0).to(tl.float32)
        y = normalized * w + b
        # apply out_gate: need out_gate pointer as well
        gate = tl.load(out_gate_ptr + cur_h * stride_gate, mask=mask, other=0.0)
        # out_gate is fp16; convert to float
        y = y * gate.to(tl.float32)
        tl.store(base_out + cur_h * stride_oh, y, mask=mask)
```

"""
Optimized TriMul “outgoing” operator (AlphaFold‑3 style).

Algorithm
---------
1️⃣  Input LayerNorm (float32) on the last channel (dim).
2️⃣  Cast to fp16 and apply four bias‑free linear projections:
        left, right, left_gate, right_gate, out_gate.
    The gate projections are followed by a sigmoid.
3️⃣  Optional mask (broadcast on the hidden dimension) is applied
    element‑wise to the projected tensors.
4️⃣  The core O(N³·H) operation is a batched matrix multiplication
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h] .
    • For very small sequences (seq_len ≤ 256) we use a custom
      Triton GEMM kernel that loads tiles of size 128×128×64 and
      accumulates in FP32 (full Tensor‑Core utilization).
    • For all other sizes we fall back to torch.bmm, which maps to
      cuBLAS Batched GEMM and is heavily optimized for H100.
5️⃣  The result (still fp16) is cast to FP32, reshaped back to
    [B, N, N, H].
6️⃣  Second LayerNorm over the hidden dimension (float32), an
    output gate (out_gate) multiplication and a final linear
    projection back to the original channel size (dim).
All heavy linear work stays in half‑precision, while the two
LayerNorms and the final projection use FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel:  batched GEMM  A @ Bᵀ  (both A and B are fp16, C is fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A  (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16×fp16→fp32 GEMM using Tensor‑Cores."""
    pid_h = tl.program_id(axis=2)          # (batch * hidden) index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # --------------------------------------------------------------
    # Tile offsets & masks
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # ----- Load Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # ----- Accumulate -----
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)        # True ⇒ mask is ignored

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.half()

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) – fp16 → fp16
    # --------------------------------------------------------------
    left      = F.linear(x_fp16, weights["left_proj.weight"].half())
    right     = F.linear(x_fp16, weights["right_proj.weight"].half())
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate= torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate  = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5️⃣ Apply per‑hidden gates
    # --------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Reshape for the batched GEMM  (B·H, N, N)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape                     # N == seq_len
    # (B, N, N, H) → (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Transpose the last two axes of right so that we can do A @ Bᵀ
    right_t   = right_perm.transpose(2, 3).contiguous()

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)      # [BH, N, N]  fp16
    right_flat = right_t.view(BH, N, N)       # [BH, N, N]  fp16

    # --------------------------------------------------------------
    # 7️⃣ Batched matrix multiplication
    # --------------------------------------------------------------
    # Small sequences (≤256) benefit from the hand‑written Triton kernel,
    # larger sequences are faster with cuBLAS via torch.bmm.
    use_triton = (N <= 256)

    if use_triton:
        # Output buffer (FP32 accumulator)
        out_flat = torch.empty_like(left_flat, dtype=torch.float32)

        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
            (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
            BH,                             # (batch * hidden)
        )

        tri_mul_outgoing_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N,            # M
            N,            # N
            N,            # K
            left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
            right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
            out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    else:
        # Torch batched GEMM (fp16×fp16→fp16).  The result is cast to FP32.
        out_flat = torch.bmm(left_flat, right_flat).float()

    # --------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # --------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # still FP32

    # --------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection
    # --------------------------------------------------------------
    out = out * out_gate.to(out.dtype)                    # broadcast gate
    out = F.linear(out, weights["to_out.weight"])         # → [B, N, N, dim]

    return out
