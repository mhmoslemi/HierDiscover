"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operator
used in AlphaFold‑3 and related protein‑structure models.

Algorithm
---------
1️⃣ Layer‑Norm over the last channel (float32) → half‑precision.
2️⃣ Fuse the five linear projections/gates (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a single GEMM (half‑precision).
   The result is split, sigmoid is applied to the three gates and an
   optional binary mask is broadcast‑multiplied onto left/right.
3️⃣ Apply the gating tensors (still half‑precision).
4️⃣ Rearrange tensors so that the hidden dimension becomes a batch dimension
   (BH = B·H).  Compute the cubic update
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   as a batched matrix‑multiply `A @ Bᵀ` via a fused Triton kernel.
   The kernel works in FP16 with FP32 accumulation (tf32 tensor‑cores).
5️⃣ Second Layer‑Norm (float32) + output‑gate, then final linear projection
   back to the original channel dimension (half → float32 output).

The heavy work (step 4) is done by the Triton kernel; all other
operations use standard PyTorch kernels but are fused or cast to
half‑precision to minimise memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ (A,B half → C half, accumulate fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers to left, right, out
    M, N, K,                        # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM with fp32 accumulation."""
    pid_h = tl.program_id(axis=2)          # hidden‑batch dimension (BH)
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # ----------------- tile offsets & masks -----------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32 (to avoid overflow)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------
    # loop over K dimension (tiles of length BLOCK_K)
    # -------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- load A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)  # fp16 → fp32

        # ----- load Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)  # fp16 → fp32

        # ----- inner product (BLOCK_M × BLOCK_N) -----
        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ----------------- write result (half) -----------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),                # convert back to fp16
        mask=mask_m[:, None] & mask_n[None, :],
    )

    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, dim]  float32
    x_h = x.half()                               # half for heavy maths

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 × hidden_dim outputs) – half precision
    # ------------------------------------------------------------------
    # concatenate the five projection/gate weight matrices
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5*H, dim)

    # single GEMM: [B,N,N,dim] @ (5*H,dim)^T → [B,N,N,5*H]
    fused = F.linear(x_h, fused_weight)          # half
    B, N, _, _ = fused.shape

    # split the fused result into projections and gates
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H] half
    left_proj  = fused[..., 0, :]                # [B,N,N,H] half
    right_proj = fused[..., 1, :]                # [B,N,N,H] half
    left_gate  = torch.sigmoid(fused[..., 2, :])  # half
    right_gate = torch.sigmoid(fused[..., 3, :])  # half
    out_gate   = torch.sigmoid(fused[..., 4, :])  # half

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # ------------------------------------------------------------------
    left  = left_proj * left_gate   # [B,N,N,H] half
    right = right_proj * right_gate # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 5️⃣ Prepare for batched GEMM (BH = B·H)
    # ------------------------------------------------------------------
    # bring hidden dim to batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N] half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N] half

    # ------------------------------------------------------------------
    # 6️⃣ Triton batched GEMM (A @ Bᵀ) -> half output
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Tile sizes – tuned for H100 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden) slices
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),                       # A strides
        right_flat.stride(0),
        right_flat.stride(2),                      # Bᵀ strides (k)
        right_flat.stride(1),                      # Bᵀ strides (j)
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),                        # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)         # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous() # [B, N, N, H] half

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output‑gate
    # ------------------------------------------------------------------
    out = out_half.float()  # cast to fp32 for stable layer‑norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                       # [B,N,N,H] float32

    out = out * out_gate.float()                     # apply output gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out.half(), to_out_weight_h)   # half → (B,N,N,dim)
    out_final = out_proj.float()                       # cast back to fp32

    return out_final
