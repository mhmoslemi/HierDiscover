"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Layer‑norm over the last dimension of the input tensor.
2️⃣ FP16 linear projections (left/right) and three gating projections.
   Optional mask is applied in FP16 and fused with the projected tensors.
3️⃣ Fuse the gates (sigmoid) into the projected tensors.
4️⃣ Compute the cubic contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched matrix multiplication for each (batch, hidden) pair.
   This heavy N³ work is performed by a custom Triton kernel that
   processes many (batch*hidden) GEMMs in a single launch, using FP16
   arithmetic with a FP32 accumulator for accuracy.
5️⃣ Second Layer‑norm over the hidden dimension, apply the output gate,
   and a final FP32 linear projection back to the original dim.

The Triton kernel uses a 3‑D launch grid (tile‑row, tile‑col, batch*hidden)
and a small autotuned set of tile sizes, delivering large speed‑ups
while keeping the final result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel
# Computes C = A @ B   where
#   A : (B, M, K) , B : (B, K, N) , C : (B, M, N)
#   Data type : FP16 (accumulator in FP32)
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 64, 'BLOCK_K': 32}, num_warps=4),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 64}, num_warps=8),
        triton.Config({'BLOCK_M': 64, 'BLOCK_N': 128, 'BLOCK_K': 32}, num_warps=4),
        triton.Config({'BLOCK_M': 128, 'BLOCK_N': 64, 'BLOCK_K': 64}, num_warps=8),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # dimensions
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 0️⃣ Program IDs
    # ------------------------------------------------------------------
    pid_batch = tl.program_id(2)          # merged (batch * hidden)
    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile column

    # ------------------------------------------------------------------
    # 1️⃣ Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 2️⃣ Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3️⃣ Loop over K in BLOCK_K chunks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles of A and B
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        # Load with masks – out‑of‑bounds elements are zeroed
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Accumulate dot product (FP16 inputs → FP32 accumulator)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # 4️⃣ Store the result tile (convert back to FP16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    Both inputs must be FP16, contiguous and have shapes
        a : (B, M, K)
        b : (B, K, N)
    Returns:
        c : (B, M, N)   FP16
    """
    assert a.dtype == torch.float16 and b.dtype == torch.float16
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "inner dimensions must match"

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Compute launch grid
    num_m = (M + 128 - 1) // 128   # over‑estimate, kernel will mask excess
    num_n = (N + 128 - 1) // 128
    grid = (num_m, num_n, B)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):

            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1️⃣ Load model parameters – cast to half where appropriate
    # --------------------------------------------------------------
    # LayerNorm 1 (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections – store in half to reduce bandwidth
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    # Gate linear layers
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # Output LayerNorm & final linear – keep in FP32 (final precision)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 2️⃣ First LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                 # [B, N, N, dim], float32
    x_h = x.half()    # cast to FP16 for the heavy work

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim], FP16
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to same dtype
        mask_h = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – also FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # keep for later

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Prepare tensors for the N³ contraction
    # --------------------------------------------------------------
    # left : (B, N, N, H)  → (B, H, N, N)   (i,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right : (B, N, N, H) → (B, H, N, N)   (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    del left, right, left_gate, right_gate, x_h  # free memory

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions so the kernel sees (BH, N, N)
    A = left_perm.view(B * H, N, N)           # (B*H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)      # (B*H, K=N, N)

    # --------------------------------------------------------------
    # 7️⃣ Batched matrix multiplication via Triton (FP16 → FP32 accum)
    # --------------------------------------------------------------
    C = _batched_matmul_fp16(A, B_mat)        # (B*H, N, N)  FP16

    # Restore original layout: [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over hidden dimension (convert to FP32)
    # --------------------------------------------------------------
    out = out.float()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 9️⃣ Apply output gate (cast to FP32 for stability)
    # --------------------------------------------------------------
    out = out * out_gate.float()

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to “dim” (float32)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim], float32
    return out
```
