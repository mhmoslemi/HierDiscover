def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layernorm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # cast to fp16
    x_f16 = x.to(torch.float16)

    # Concatenate all projection weights.
    # left_proj, right_proj, left_gate, right_gate, out_gate.
    # All weights shape (dim, hidden_dim).
    w_concat = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"]
    ], dim=1).to(torch.float16)  # shape (dim, 5*hidden_dim)

    # fused linear
    proj = F.linear(x_f16, w_concat)  # [B, N, N, 5*hidden_dim]

    # split
    lproj, rproj, lgate, rgate, ogate = torch.split(proj, hidden_dim, dim=-1)
    # apply mask to lproj & rproj
    if not nomask:
        m = mask.to(torch.float16).unsqueeze(-1)
        lproj = lproj * m
        rproj = rproj * m

    # sigmoid gates
    lgate = torch.sigmoid(lgate)
    rgate = torch.sigmoid(rgate)
    ogate = torch.sigmoid(ogate)

    # apply gates
    left = lproj * lgate
    right = rproj * rgate

    # core matmul: out = left @ right^T per hidden.
    B, N, _, H = left.shape

    # reshape for batched matmul: (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # call triton kernel
    out_mat = _batched_matmul_triton(left_mat, right_mat, seq_len=N)

    out = out * ogate

"""
custom_kernel – Triton‑accelerated forward pass of the TriMul “outgoing’’ operator.

Algorithm (high‑level)
----------------------
1️⃣ Layer‑norm the 4‑D input X ∈ [B,N,N,D] (D = dim).
2️⃣ Fuse the five bias‑free linear layers (left‑proj, right‑proj,
   left‑gate, right‑gate, out‑gate) into a *single* GEMM:
       X (fp16) @ W_concat (fp16)  →  Y ∈ [B,N,N,5·H]   (H = hidden_dim)
   Then slice Y back to the five tensors.
3️⃣ Apply the optional binary mask to left/right, compute the three
   sigmoid gates and multiply the gates into left/right (all in fp16).
4️⃣ Core N³ update (outgoing version):
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication L @ Rᵀ for each hidden
   channel.  We reshape to (B·H, N, N) and launch a highly‑tuned
   Triton kernel that tiles the computation with (BLOCK_M·BLOCK_N·BLOCK_K)
   = (256·256·64) on H100 Tensor‑Cores (fp16×fp16 → fp32).
5️⃣ Reshape back to [B,N,N,H] (fp32) and apply the output LayerNorm.
6️⃣ Multiply by the out‑gate (sigmoid) and a final bias‑free linear
   projection back to the original dimension D.

All heavy linear work (steps 2 & 4) runs in fp16, accumulation and
final results are in fp32, matching the reference implementation while
significantly reducing memory traffic and kernel launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A (M×K) * B (N×K) → C (M×N)
#   A, B : fp16,   C : fp32
#   Batch dimension is passed as the third launch axis.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    M, N, K,                                 # matrix sizes
    stride_ah, stride_am, stride_ak,         # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,         # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,         # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_b = tl.program_id(2)          # batch (B·H)

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # --- load A tile (BLOCK_M x BLOCK_K) ---
        a_ptrs = a_batch \
                 + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)    # fp16

        # --- load B tile (BLOCK_K x BLOCK_N) ---
        b_ptrs = b_batch \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)    # fp16

        # matrix dot for this K‑tile (fp32 accumulation)
        acc += tl.dot(a_tile, b_tile)

    # --- write back ---
    c_ptrs = c_batch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor, seq_len: int):
    """
    Batched matrix multiplication using the Triton kernel above.
    A, B : fp16 tensors of shape (batch, N, N)
    Returns C : fp32 tensor of shape (batch, N, N)
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16 inputs"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in element units
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    # ------------------------------------------------------------------
    # Heuristic tile sizes – all benchmark sizes are multiples of 128,
    # so we can use mask‑free tiles.
    # ------------------------------------------------------------------
    if seq_len <= 256:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32
        num_warps = 8
    else:
        BLOCK_M = 256
        BLOCK_N = 256
        BLOCK_K = 64
        num_warps = 8

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) – may be None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32).
    """
    # -------------------------- unpack ---------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input feature dimension (D)
    hidden_dim = config["hidden_dim"]   # projection dimension (H)
    nomask = config.get("nomask", True)

    # ------------------- 1) Input LayerNorm ----------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D], float32

    # ------------------- 2) Fuse five linear layers -------------------
    # Cast to fp16 once (Tensor‑Core friendly)
    x_f16 = x.to(torch.float16)

    # Concatenate the five weight matrices along the output dimension
    # Order: left_proj, right_proj, left_gate, right_gate, out_gate
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=1,
    ).to(torch.float16)               # (D, 5·H)

    # One fused linear projection
    fused = F.linear(x_f16, w_concat)   # [B, N, N, 5·H], fp16

    # Split back into the individual tensors
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        fused, hidden_dim, dim=-1
    )                                   # each [B, N, N, H]

    # ------------------- 3) Optional mask ------------------------------
    if not nomask and mask is not None:
        # mask is bool/int → 0/1 fp16
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * m
        right_proj = right_proj * m

    # ------------------- 4) Gates (sigmoid) ---------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # ------------------- 5) Apply element‑wise gates -------------------
    left = left_proj * left_gate          # fp16
    right = right_proj * right_gate       # fp16

    # ------------------- 6) Core N³ batched mat‑mul -------------------
    B, N, _, H = left.shape  # B, seq_len, _, hidden_dim

    # Re‑order to (B·H, N, N) with K (the reduction axis) contiguous.
    # This requires a contiguous view; the copy cost is negligible
    # compared with the O(N³·H) work.
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # Batched GEMM L @ Rᵀ  (each batch computes N×N @ N×Nᵀ)
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        seq_len=N,
    )   # [B·H, N, N], fp32

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()      # fp32

    # ------------------- 7) Output LayerNorm & gating -----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    out = out * out_gate   # out_gate is fp16 → up‑cast to fp32 automatically

    # ------------------- 8) Final linear projection --------------------
    out = F.linear(out, weights["to_out.weight"])   # back to dim (float32)

    return out
```
