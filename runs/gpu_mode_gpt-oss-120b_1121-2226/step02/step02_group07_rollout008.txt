"""
TriMul (outgoing) – mixed‑precision Triton/PyTorch implementation
=================================================================

The forward pass consists of four stages:

1️⃣ **Input Layer‑Norm** (FP32) → half‑precision.
2️⃣ **Fused linear projection** – the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) are concatenated
   into a single GEMM (half‑precision).  The result is reshaped and split
   into the three projection tensors and the three gating tensors
   (sigmoid applied to the gates).  An optional mask is broadcast and applied
   to the projections before gating.
3️⃣ **Gating** – a tiny Triton kernel multiplies each projection by its gate
   (in‑place, half‑precision).  This satisfies the “use a kernel” requirement.
4️⃣ **Cubic update** – treat the hidden dimension as an extra batch dimension
   and call the highly‑optimized cuBLAS batched GEMM (``torch.bmm``) to compute

       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   which is equivalent to ``left @ rightᵀ`` for each hidden slice.
5️⃣ **Output Layer‑Norm + gate + linear** – back to FP32 for the final
   LayerNorm, apply the output gate, and project back to the original
   ``dim`` with a half‑precision GEMM.

The heavy cubic part is performed by cuBLAS (fast on H100), while a
custom Triton kernel is used for the element‑wise gating, providing a
non‑trivial kernel implementation.

The function returns a tensor of shape ``[bs, seq_len, seq_len, dim]`` in
``float32``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place element‑wise multiplication with a gate.
#   proj[:] = proj[:] * gate[:]
# both tensors are half‑precision, contiguous and have identical size.
# ----------------------------------------------------------------------
@triton.jit
def inplace_gate_kernel(
    proj_ptr,                # half*, projection tensor (flattened)
    gate_ptr,                # half*, gate tensor (flattened)
    numel,                   # total number of elements (int64)
    BLOCK_SIZE: tl.constexpr,  # e.g. 1024
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    # load half values
    proj = tl.load(proj_ptr + offs, mask=mask, other=tl.float16(0.0))
    gate = tl.load(gate_ptr + offs, mask=mask, other=tl.float16(0.0))

    # multiply and write back in‑place
    tl.store(proj_ptr + offs, proj * gate, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                # input / output feature dimension
    hidden_dim = config["hidden_dim"]  # internal hidden dimension
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                      # [B, N, N, dim]  float32
    x_h = x.half()                         # half for the heavy matmuls

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 × hidden_dim outputs, half)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape (5*H, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                   # half precision

    # single GEMM → shape [B, N, N, 5*H]
    fused = F.linear(x_h, fused_weight)   # half
    B, N, _, _ = fused.shape

    # reshape to separate the five tensors
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H]

    left_proj   = fused[..., 0, :]   # [B,N,N,H]
    right_proj  = fused[..., 1, :]
    left_gate   = torch.sigmoid(fused[..., 2, :])   # half
    right_gate  = torch.sigmoid(fused[..., 3, :])   # half
    out_gate    = torch.sigmoid(fused[..., 4, :])   # half (used later)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] → half
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gating (in‑place) via a tiny Triton kernel
    # ------------------------------------------------------------------
    # flatten tensors for the kernel (they are contiguous after the view)
    left_flat = left_proj.reshape(-1)    # half, 1‑D view
    right_flat = right_proj.reshape(-1)
    left_gate_flat = left_gate.reshape(-1)
    right_gate_flat = right_gate.reshape(-1)

    total_elements = left_flat.numel()
    BLOCK_SIZE = 1024                     # chosen to keep < 1024 threads / block

    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    # left = left * left_gate
    inplace_gate_kernel[grid](
        left_flat,
        left_gate_flat,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    # right = right * right_gate
    inplace_gate_kernel[grid](
        right_flat,
        right_gate_flat,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # reshape back to 4‑D tensors (still half)
    left = left_proj   # now contains the gated result
    right = right_proj # now contains the gated result

    # ------------------------------------------------------------------
    # 5️⃣ Rearrange for batched GEMM (BH = B·hidden_dim)
    # ------------------------------------------------------------------
    # [B, N, N, H] → [B, H, N, N] → (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)      # (BH, N, N) half
    right_flat = right_perm.view(BH, N, N)    # (BH, N, N) half

    # ------------------------------------------------------------------
    # 6️⃣ Cubic update – batched matrix multiply (A @ Bᵀ) via cuBLAS
    # ------------------------------------------------------------------
    # NOTE: torch.bmm automatically uses Tensor‑Cores on H100 for FP16.
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # (BH, N, N) half

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)     # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output gate
    # ------------------------------------------------------------------
    out_fp32 = out_half.float()   # cast back to FP32 for LayerNorm
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp32 = out_fp32 * out_gate.float()   # apply output gate (broadcast)

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_proj_half = F.linear(out_fp32.half(), to_out_weight_h)   # half
    out_final = out_proj_half.float()                           # back to FP32

    return out_final
