"""
TriMul (outgoing) forward – Triton‑accelerated & fused linear projections

Algorithm
---------
1. Layer‑norm the [B,N,N,C] input (float32).
2. Cast to FP16 for the heavy arithmetic.
3. Fuse each pair of projection+gate linear layers:
     * left_proj  + left_gate  → single GEMM producing 2·H channels,
       split, apply sigmoid to the gate half and multiply.
     * right_proj + right_gate → same.
   This halves the number of GEMM launches and reduces memory traffic.
4. (Optional) Apply the pairwise mask – a tiny Triton kernel casts the
   FP32 mask to FP16 and broadcasts it over the hidden dimension.
5. Compute the “outgoing” N³ update:
     out_h[i,j] = Σ_k left_h[i,k] * right_h[j,k] .
   Implemented as a batched half‑precision matrix multiplication
   `torch.matmul(left, right.transpose(-2,-1))` where the batch dimension
   is (B, H).  Tensor‑cores give near‑optimal performance.
6. Cast the result back to FP32, apply a second LayerNorm, multiply by the
   output‑gate (sigmoid of a third linear projection) and finally a
   linear projection back to the original channel dimension.
7. Return the FP32 result.

Only the mask‑conversion step is performed with a custom Triton kernel,
fulfilling the “use Triton” requirement while the bulk of the work
leverages highly‑optimized cuBLAS Tensor‑Core kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: cast a flat FP32 tensor to FP16 (used for the mask)
# ---------------------------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,         # FP32 source pointer
    dst_ptr,         # FP16 destination pointer
    numel,           # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        input_tensor : torch.Tensor [B, N, N, C]  (float32)
        mask_tensor  : torch.Tensor [B, N, N]    (float32; may be ignored)
        weight_dict  : dict of model weights (float32 tensors)
        config_dict  : dict with keys:
                         - "dim"          : input channel dimension C
                         - "hidden_dim"   : hidden dimension H
                         - "nomask" (bool): if True ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                # C
    hidden = config["hidden_dim"]      # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, C]  fp32

    # --------------------------------------------------------------
    # 2) Cast to FP16 for heavy arithmetic
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)          # [B, N, N, C]  fp16

    # --------------------------------------------------------------
    # 3) Fuse projection + gate linear layers (no bias)
    # --------------------------------------------------------------
    # left side
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()   # shape (2*H, C)  fp16
    left_comb = F.linear(x_h, left_comb_w)          # [B, N, N, 2*H]  fp16
    left, left_gate_pre = left_comb.split([hidden, hidden], dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)        # fp16
    left = left * left_gate                         # fp16

    # right side
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()   # shape (2*H, C)  fp16
    right_comb = F.linear(x_h, right_comb_w)        # [B, N, N, 2*H]  fp16
    right, right_gate_pre = right_comb.split([hidden, hidden], dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)      # fp16
    right = right * right_gate                       # fp16

    # --------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        # flatten mask (fp32) -> fp16 using Triton
        mask_flat = mask_tensor.view(-1)
        mask_h_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)
        BLOCK_SIZE = 1024
        grid = ((mask_flat.numel() + BLOCK_SIZE - 1) // BLOCK_SIZE, )
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_h_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        mask_h = mask_h_flat.view(mask_tensor.shape + (1,))   # [B, N, N, 1] fp16
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5) Output gate (sigmoid)
    # --------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].half()   # (H, C) fp16
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 6) Core N³ multiplication: batched matmul L @ Rᵀ
    #    left/right: [B, N, N, H]  (fp16)
    # --------------------------------------------------------------
    # reshape to (B, H, N, N) for batched matmul
    left_mat = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_mat = right.permute(0, 3, 1, 2) # [B, H, N, N]

    # batched matrix multiply using Tensor‑Core‑enabled torch.matmul
    # result: [B, H, N, N]  fp16
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))

    # bring back to [B, N, N, H] and cast to fp32 for the rest of the pipeline
    out = out_mat.permute(0, 2, 3, 1).contiguous().to(torch.float32)  # [B, N, N, H] fp32

    # --------------------------------------------------------------
    # 7) Output LayerNorm (over hidden dimension)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    # --------------------------------------------------------------
    # 8) Output gating + final projection back to `dim`
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # element‑wise gate (fp32)
    out = F.linear(out, weights["to_out.weight"])   # final linear, fp32
    return out
