def custom_kernel(data):

    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    # Input LayerNorm

    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    x_h = x.half()

    # fused linear

    fused_weight = torch.cat([...], dim=0).to(torch.float16)

    fused = torch.nn.functional.linear(x_h, fused_weight) # shape [B,N,N,5*hidden_dim]

    B, N, _, _ = fused.shape

    fused = fused.view(B, N, N, 5, hidden_dim)

    left_proj = fused[...,0,:]
    right_proj = fused[...,1,:]
    left_gate = torch.sigmoid(fused[...,2,:])
    right_gate = torch.sigmoid(fused[...,3,:])
    out_gate = torch.sigmoid(fused[...,4,:])

    # mask
    if nomask:
        mask_f = torch.ones((B,N,N,1), device=device, dtype=torch.float16)
    else:
        mask_f = mask_tensor.unsqueeze(-1).to(torch.float16)

    # flatten for kernel gating

    left_proj_flat = left_proj.contiguous().view(-1)
    right_proj_flat = right_proj.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat = mask_f.contiguous().view(-1)

    total = left_proj_flat.numel()
    BLOCK = 1024

    @triton.jit
    def gating_kernel(
        left_proj_ptr, right_proj_ptr,
        left_gate_ptr, right_gate_ptr,
        mask_ptr,
        left_out_ptr, right_out_ptr,
        total_size: tl.constexpr,
        BLOCK: tl.constexpr
    ):
        pid = tl.program_id(0)
        idx = pid * BLOCK + tl.arange(0, BLOCK)
        mask_enabled = idx < total_size

        # Load
        lp = tl.load(left_proj_ptr + idx, mask=mask_enabled, other=0.0)
        rp = tl.load(right_proj_ptr + idx, mask=mask_enabled, other=0.0)
        lg = tl.load(left_gate_ptr + idx, mask=mask_enabled, other=0.0)
        rg = tl.load(right_gate_ptr + idx, mask=mask_enabled, other=0.0)
        m  = tl.load(mask_ptr + idx, mask=mask_enabled, other=1.0)

        left_out = lp * lg * m
        right_out = rp * rg * m

        tl.store(left_out_ptr + idx, left_out, mask=mask_enabled)
        tl.store(right_out_ptr + idx, right_out, mask=mask_enabled)

    grid = lambda meta: ( (total + meta['BLOCK'] - 1) // meta['BLOCK'], )
    left_out = torch.empty_like(left_proj_flat)
    right_out = torch.empty_like(right_proj_flat)

    gating_kernel[grid](
        left_proj_flat,
        right_proj_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        left_out,
        right_out,
        total,
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=2,
    )

    # reshape back
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)

    # Rearrange for batched matmul
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)

    # batched GEMM
    # Use torch.bmm; left * right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))

    # Reshape back to B,N,N,H
    out_half = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # Second layernorm and output gate
    out = out_half.float()
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float() # out_gate broadcast

    # Final linear
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_proj = torch.nn.functional.linear(out.half(), to_out_weight_h)
    out_final = out_proj.float() # cast back

    return out_final

"""
TriMul (outgoing) – fast Triton‑accelerated implementation
---------------------------------------------------------
The forward pass consists of four stages:

1️⃣  Input LayerNorm (float32) → half‑precision.
2️⃣  **Fused linear projection** – the five weight matrices
    (left_proj, right_proj, left_gate, right_gate, out_gate) are
    concatenated into a single matrix (5·H × dim) and multiplied with
    the input in a single half‑precision GEMM.  The result is reshaped
    and split into three projections and three gates (sigmoid applied
    to the gates).
3️⃣  **Mask + gating fusion** – a tiny Triton kernel multiplies the
    raw projections by the mask and their respective gates in one
    pass, avoiding extra memory traffic.
4️⃣  **Cubic update** – for each hidden slice we need
    `out[i,j] = Σ_k left[i,k] * right[j,k]`.  This is exactly a batched
    matrix‑multiply `A @ Bᵀ`.  We let cuBLAS (torch.bmm) execute the
    GEMM in half‑precision with FP32 accumulation, which is far faster
    than a hand‑written Triton kernel on the H100.
5️⃣  Second LayerNorm + output‑gate (float32) → final linear projection
    back to `dim`.

All heavy linear work is performed in FP16; only the two LayerNorms,
the output‑gate and the final projection use FP32 to keep numerical
stability.  The Triton kernel is required (mask + gate fusion) but the
expensive N³ part is delegated to cuBLAS for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask multiplication and gating (half → half)
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    proj_ptr,               # raw projection (left or right)
    gate_ptr,               # sigmoid‑gate
    mask_ptr,               # broadcast mask (already expanded to H)
    out_ptr,                # output after mask & gate
    total_size: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear indices
    mask = offs < total_size

    # load inputs
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m    = tl.load(mask_ptr + offs, mask=mask, other=1.0)

    # apply mask and gate (all half‑precision)
    out = proj * gate * m

    # store result
    tl.store(out_ptr + offs, out, mask=mask)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (optional)
        - weights      : dict of model weights (see code)
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                   # [B,N,N,dim]  float32
    x_h = x.half()                      # heavy work in fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear stage (5·hidden_dim outputs)
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # (5·H, dim)

    #    [B,N,N,5·H]   (fp16)
    fused = F.linear(x_h, fused_weight)

    # split into components
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H]
    left_proj  = fused[..., 0, :]                # [B,N,N,H]
    right_proj = fused[..., 1, :]
    left_gate  = torch.sigmoid(fused[..., 2, :])   # half‑precision sigmoid
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :])   # later used after 2nd LN

    # --------------------------------------------------------------
    # 3️⃣ Mask & gate fusion (Triton)
    # --------------------------------------------------------------
    if nomask:
        # mask of ones – same shape as the projections
        mask_exp = torch.ones_like(left_proj, dtype=torch.float16, device=device)
    else:
        # broadcast mask over the hidden dimension
        mask_exp = mask_tensor.unsqueeze(-1).to(torch.float16)          # [B,N,N,1]
        mask_exp = mask_exp.expand(-1, -1, -1, hidden_dim).contiguous()  # [B,N,N,H]

    # flatten everything for a 1‑D kernel launch
    lp_flat = left_proj.contiguous().view(-1)        # (B·N·N·H,)
    rp_flat = right_proj.contiguous().view(-1)
    lg_flat = left_gate.contiguous().view(-1)
    rg_flat = right_gate.contiguous().view(-1)
    mk_flat = mask_exp.contiguous().view(-1)

    total_elems = lp_flat.numel()
    BLOCK = 1024
    grid = ((total_elems + BLOCK - 1) // BLOCK,)

    # left side
    left_out = torch.empty_like(lp_flat)
    mask_gate_kernel[grid](
        lp_flat,
        lg_flat,
        mk_flat,
        left_out,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=2,
    )

    # right side (reuse same kernel)
    right_out = torch.empty_like(rp_flat)
    mask_gate_kernel[grid](
        rp_flat,
        rg_flat,
        mk_flat,
        right_out,
        total_elems,
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=2,
    )

    # reshape back to [B,N,N,H]
    left = left_out.view(B, N, N, hidden_dim)
    right = right_out.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 4️⃣ Cubic update – batched matmul A @ Bᵀ  (cuBLAS)
    # --------------------------------------------------------------
    # bring hidden dimension in front of the two spatial axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B,H,N,N]

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)                # [BH,N,N]  fp16
    right_flat = right_perm.view(BH, N, N)              # [BH,N,N]  fp16

    # batched GEMM:  (i,k) @ (k,j)  →  (i,j)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # fp16

    # back to [B,N,N,H]
    out_half = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output gate (float32)
    # --------------------------------------------------------------
    out = out_half.float()   # cast to fp32 for layer‑norm stability
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()          # broadcasted output gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.half(), to_out_w)  # fp16 matmul
    out = out.float()                     # final dtype float32

    return out
