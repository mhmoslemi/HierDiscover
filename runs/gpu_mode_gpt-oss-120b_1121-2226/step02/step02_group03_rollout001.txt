"""
TriMul “outgoing” kernel – a highly‑optimized back‑to‑back fused implementation.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input (float32) – only the last dimension is normalized.
2️⃣ Cast to fp16 and compute the two linear projections (left / right) together
   with their sigmoid gates (still fp16).  An optional mask is applied in‑place.
3️⃣ Reshape to (B·H, N, N) where H = hidden_dim and launch a custom Triton GEMM
   that computes  out[b·H, i, j] = Σₖ left[b·H, i, k] * right[b·H, j, k] .
   The kernel uses tile sizes (M,N,K) = (BLOCK_M,BLOCK_N,BLOCK_K) with
   fp16×fp16→fp32 accumulation (Tensor‑Core friendly).  Two‑level pre‑fetch
   (double‑buffer) reduces latency.
4️⃣ Reshape back to [B, N, N, H] and apply a second Layer‑Norm (float32).
5️⃣ Multiply by the output gate (sigmoid) and a final linear projection back to
   the original channel dimension (float32).

The heavy O(N³·H) part (step 3) is the only part written in Triton; everything
else stays in PyTorch for simplicity.  Block sizes adapt to the sequence length
to keep occupancy high while respecting shared‑memory limits.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton GEMM kernel for the “outgoing” TriMul.
# Computes C = A @ Bᵀ   (A,B: fp16, C: fp32)
#   A shape: (BH, M, K)   – M = N, K = N
#   B shape: (BH, N, K)   – we interpret B as transposed inside the kernel
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides: (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides: (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides: (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ.
    A,B are fp16, C is fp32.  The kernel is tiled with double‑buffered loads.
    """
    pid_h = tl.program_id(axis=2)          # batch × hidden index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Two accumulator buffers for double‑buffering
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Prefetch first tile of K dimension
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    cur_k = 0 * BLOCK_K + tl.arange(0, BLOCK_K)   # first tile start
    mask_k = cur_k < K

    a_ptrs = (
        a_ptr
        + pid_h * stride_ah
        + offs_m[:, None] * stride_ai
        + cur_k[None, :] * stride_ak
    )
    b_ptrs = (
        b_ptr
        + pid_h * stride_bh
        + cur_k[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )
    a_tile = tl.load(
        a_ptrs,
        mask=mask_m[:, None] & mask_k[None, :],
        other=0.0,
    ).to(tl.float32)               # fp16 → fp32
    b_tile = tl.load(
        b_ptrs,
        mask=mask_k[:, None] & mask_n[None, :],
        other=0.0,
    ).to(tl.float32)               # fp16 → fp32

    # ------------------------------------------------------------------
    # Main K‑loop with double buffering
    # ------------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        # Compute dot product for the *previous* tile (except first iteration)
        if k_tile > 0:
            acc += tl.dot(a_tile, b_tile)   # (M,K) @ (K,N)

        # Prefetch next tile (if any)
        next_k = (k_tile + 1) * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_next_k = next_k < K

        a_ptrs_next = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + next_k[None, :] * stride_ak
        )
        b_ptrs_next = (
            b_ptr
            + pid_h * stride_bh
            + next_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        a_tile = tl.load(
            a_ptrs_next,
            mask=mask_m[:, None] & mask_next_k[None, :],
            other=0.0,
        ).to(tl.float32)
        b_tile = tl.load(
            b_ptrs_next,
            mask=mask_next_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

    # The loop leaves the last loaded tile not yet multiplied
    acc += tl.dot(a_tile, b_tile)

    # ------------------------------------------------------------------
    # Store result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry‑point for the outgoing TriMul.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul “outgoing”.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 and compute linear projections & gates
    # ------------------------------------------------------------------
    x_fp16 = x.half()                               # [B,N,N,dim] fp16

    # Linear projections (no bias)
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # Optional mask (broadcasted on hidden dim)
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))  # [B,N,N,hidden]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Reshape to (B·H, N, N) – ready for the batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, N, N, H) → (B, H, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left.view(BH, N, N)    # fp16
    right_flat = right.view(BH, N, N)  # fp16

    # Output buffer – fp32 accumulation
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 4️⃣ Choose tile size based on sequence length (heuristic)
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64   # keep K modest for shared‑mem fit

    # Grid dimensions
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # batch × hidden
    )

    # ------------------------------------------------------------------
    # 5️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    tri_mul_outgoing_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),  # Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # reasonable default for H100
        num_stages=2,              # double‑buffering
    )

    # ------------------------------------------------------------------
    # 6️⃣ Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N] fp32
    out = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate (fp16 → fp32) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.float()                     # broadcast gate
    out = F.linear(out, weights["to_out.weight"])    # [B, N, N, dim] float32

    return out
```
