"""
TriMul (outgoing) – fast Triton / cuBLAS hybrid implementation
----------------------------------------------------------------
The forward pass consists of four stages:

1️⃣ Input LayerNorm (float32) → half‑precision.
2️⃣ **Fused linear stage** – the five weight matrices (left_proj,
   right_proj, left_gate, right_gate, out_gate) are concatenated into a
   single (5·H × C) matrix and multiplied with the input in one half‑
   precision GEMM.  The result is reshaped and split into the three
   projection tensors and the three gating tensors (sigmoid applied to
   the gates).  An optional mask is broadcast and applied before gating.
3️⃣ **Cubic update** – for each hidden slice we need
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   This is a batched matrix‑multiply `A @ Bᵀ`.  For small sequence lengths
   we launch a hand‑written Triton kernel (`tri_mul_outgoing_kernel`);
   for larger `N` we fall back to cuBLAS via `torch.bmm`,
   which is heavily tuned for the H100 tensor‑cores.
4️⃣ Second LayerNorm + output‑gate (float32) → final linear projection
   back to the original channel dimension.  The final tensor has shape
   `[B, N, N, C]` and dtype ``float32``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B half → C half, acc in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM (`A @ Bᵀ`) where each batch corresponds to one
    (batch, hidden) slice.  All tensors are half‑precision, accumulation
    is performed in FP32 for numerical stability."""
    pid_h = tl.program_id(axis=2)      # (batch * hidden) index
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # -------- A tile (M × K) ------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)               # FP16 → FP32

        # -------- Bᵀ tile (K × N) ----------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)               # FP16 → FP32

        # -------- inner product ------------------------------------
        acc += tl.dot(a, b)                # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # write out as half (store half‑precision to save bandwidth)
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, dim]  float32
    x_h = x.half()                               # half for the heavy ops

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 × hidden_dim outputs)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices once (shape (5*H, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # half

    # single half‑precision GEMM → shape [B,N,N,5*H]
    fused = F.linear(x_h, fused_weight)          # half
    B, N, _, _ = fused.shape

    # split the fused result
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H] (half)
    left_proj  = fused[..., 0, :]                # [B,N,N,H]
    right_proj = fused[..., 1, :]
    left_gate  = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :])   # will be used later

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # ------------------------------------------------------------------
    left  = left_proj  * left_gate   # [B,N,N,H] half
    right = right_proj * right_gate  # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 5️⃣ Rearrange for batched GEMM (BH = B·H)
    # ------------------------------------------------------------------
    # bring the hidden dimension to the second axis and make the tensor contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N]  half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]  half

    # ------------------------------------------------------------------
    # 6️⃣ Cubic update – batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    # For very small N we keep the handcrafted Triton kernel, otherwise
    # rely on cuBLAS via torch.bmm which is faster on H100 for N≥256.
    if N <= 256:
        out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

        # tiling parameters – these work well for N ≤ 256
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
            (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
            BH,                             # one slice per (batch, hidden)
        )

        tri_mul_outgoing_kernel[grid](
            left_flat,
            right_flat,
            out_flat,
            N, N, N,                                   # M, N, K
            left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
            right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
            out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=8,
            num_stages=3,
        )
    else:
        # Use cuBLAS batched GEMM (half → fp32 accumulation automatically)
        out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output gate
    # ------------------------------------------------------------------
    out_fp32 = out_half.float()
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                                   # [B,N,N,H] float32

    out_fp32 = out_fp32 * out_gate.float()              # apply output gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out_fp32.half(), to_out_weight_h)   # half
    out_final = out_proj.float()                           # cast back to fp32

    return out_final
```
