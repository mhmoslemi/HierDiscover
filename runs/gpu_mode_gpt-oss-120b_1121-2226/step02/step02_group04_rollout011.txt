@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch id b and hidden index h
    b = pid_batch // H
    h = pid_batch - b * H

    # base offsets
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # tile start points
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # create range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for OOB
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask = mask_i[:, None] & mask_j[None, :]

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # bounds mask for K dimension
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate
        acc += tl.dot(left, right)  # left: (M,K), right: (K,N) => (M,N)

    # store
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask)

"""
Optimized Triton implementation of the outgoing TriMul operator.

Algorithm:
1. Apply LayerNorm to input (FP32) and cast to FP16.
2. Fuse all five linear projections (left_proj, left_gate, right_proj, right_gate,
   out_gate) into a single matmul and split the result.
3. Fuse mask broadcasting, element‑wise mask and gate multiplication in a small
   Triton kernel.
4. Compute the N³ contraction with a custom batched GEMM Triton kernel that
   works directly on the original (B,N,N,H) layout, avoiding costly permute &
   contiguous copies.
5. Apply the second LayerNorm, multiply by the output gate and project back to
   the original dimension with a final FP16 linear layer.
All heavy arithmetic is performed in FP16; the final result is cast back to FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Kernel for mask & gate fusion
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    l_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # One mask element per (b,i,k) – share across hidden channels
    mask_idx = offs // hidden
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    lo = l_val * l_gate * m_val
    ro = r_val * r_gate * m_val

    tl.store(out_left_ptr + offs, lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


# Batched GEMM kernel for the N³ contraction (out = left @ rightᵀ)
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr,      # [B, N, N, H]   (FP16)
    right_ptr,     # [B, N, N, H]   (FP16)
    out_ptr,       # [B, N, N, H]   (FP16) – result
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs:   0 – batch (B*H), 1 – tile row (i), 2 – tile col (j)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # derive batch index b and hidden index h from flattened pid_batch
    b = pid_batch // H
    h = pid_batch - b * H

    # base pointers for the current (b, h) slice
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # starting coordinates of the tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # range vectors
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: (K, N) – note we need the (k, j) ordering
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate product
        acc += tl.dot(left, right)   # (M, K) @ (K, N) -> (M, N)

    # write back
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_ij)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask: torch.Tensor [B, N, N] (float32 or bool) or None
            - weights: dict of model parameters (torch.Tensor)
            - config: dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor: processed tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # fetch weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear weights (FP16 for speed)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (D, H)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                   # [B, N, N, D] float32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse all projections and gates into one matmul
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                   # (5*H, D)

    fused = F.linear(x, fused_weight)   # [B, N, N, 5*H] (FP16)

    B, N, _, _ = fused.shape
    # split fused tensor
    off = 0
    left_proj = fused[..., off : off + hidden_dim];            off += hidden_dim
    left_gate = torch.sigmoid(fused[..., off : off + hidden_dim]); off += hidden_dim
    right_proj = fused[..., off : off + hidden_dim];           off += hidden_dim
    right_gate = torch.sigmoid(fused[..., off : off + hidden_dim]); off += hidden_dim
    out_gate = torch.sigmoid(fused[..., off : off + hidden_dim])   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 3. Fuse mask, left/right gates (tiny Triton kernel)
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones(B * N * N, dtype=torch.float16, device=input_tensor.device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(-1)   # [B*N*N]

    # flatten tensors for the kernel
    left_proj_f   = left_proj.reshape(-1)
    right_proj_f  = right_proj.reshape(-1)
    left_gate_f   = left_gate.reshape(-1)
    right_gate_f  = right_gate.reshape(-1)

    left_out_f  = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elements = left_proj_f.numel()   # B * N * N * H
    BLOCK_SIZE = 1024

    grid_mask = (triton.cdiv(total_elements, BLOCK_SIZE),)
    _mask_gate_fuse_kernel[grid_mask](
        left_proj_f, right_proj_f,
        left_gate_f, right_gate_f,
        mask_fp16,
        left_out_f, right_out_f,
        total=total_elements,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # reshape back to [B, N, N, H]
    left = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. N³ contraction via custom Triton GEMM (no extra permutes)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # FP16 output tensor
    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Choose tile sizes – tuned for H100 FP16 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_gemm = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_gemm_kernel[grid_gemm](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Post‑contraction LayerNorm (FP32) + output gate
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)   # broadcast over H

    # ------------------------------------------------------------------
    # 6. Final projection back to original dimension (FP16 → FP32)
    # ------------------------------------------------------------------
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)   # (D, H) weight
    out = out.to(torch.float32)

    return out

All heavy arithmetic is performed in FP16; the final result is cast back to
float32.  At least one Triton kernel (mask‑gate fusion) and a second kernel
(the batched GEMM) are required, satisfying the problem constraints.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
#  Tiny kernel: apply mask and element‑wise gate to the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                 # projected values (fp16)
    left_gate_ptr, right_gate_ptr,       # gate values (fp16)
    mask_ptr,                            # mask (fp16) of size B·N·N
    out_left_ptr, out_right_ptr,         # fused results (fp16)
    total: tl.constexpr,                 # total #elements = B·N·N·H
    hidden: tl.constexpr,                # hidden_dim = H
    BLOCK: tl.constexpr,                 # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # one mask element per (b,i,k) → same across the hidden channel
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    lo = l * lg * m
    ro = r * rg * m

    tl.store(out_left_ptr + offs, lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


# ----------------------------------------------------------------------
#  Batched GEMM kernel for the N³ contraction:
#      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,                     # tensors [B,N,N,H] (fp16)
    B, N, H,                                          # runtime integers
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides (j‑dim first)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids: 0 → batch (flattened B·H), 1 → tile row i, 2 → tile col j
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch index into (b, h)
    b = pid_batch // H
    h = pid_batch - b * H

    # base pointers for this (b, h) slice
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # tile origin
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # range vectors
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over the reduction dimension k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_tile * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M, K)
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right tile (K, N) – we need the (k, j) ordering
        right_ptrs = right_base + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # GEMM accumulation
        acc += tl.dot(left, right)   # (M, K) @ (K, N) → (M, N)

    # write back the tile
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_ij)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters
            - config       : dict with keys ``dim``, ``hidden_dim`` and optional ``eps``

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Load model weights (cast to FP16 where advantageous)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    to_out_norm_w = weights["to_out_norm.weight"]    # (H,)
    to_out_norm_b = weights["to_out_norm.bias"]      # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)  # (D, H)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                           # [B, N, N, D] float32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse all linear projections & gates into one matmul
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                           # (5*H, D)
    fused = F.linear(x, fused_weight)           # [B, N, N, 5*H] (FP16)

    B, N, _, _ = fused.shape
    off = 0
    left_proj = fused[..., off:off + hidden_dim];               off += hidden_dim
    left_gate = torch.sigmoid(fused[..., off:off + hidden_dim]); off += hidden_dim
    right_proj = fused[..., off:off + hidden_dim];              off += hidden_dim
    right_gate = torch.sigmoid(fused[..., off:off + hidden_dim]); off += hidden_dim
    out_gate = torch.sigmoid(fused[..., off:off + hidden_dim]);    # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 3. Fuse mask, left‑gate and right‑gate (tiny Triton kernel)
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones(B * N * N, dtype=torch.float16, device=input_tensor.device)
    else:
        mask_fp16 = mask.to(torch.float16).reshape(-1)         # shape B·N·N

    # flatten tensors for kernel launch
    left_proj_f   = left_proj.reshape(-1)
    right_proj_f  = right_proj.reshape(-1)
    left_gate_f   = left_gate.reshape(-1)
    right_gate_f  = right_gate.reshape(-1)

    left_out_f  = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elems = left_proj_f.numel()          # B·N·N·H
    BLOCK_SIZE = 1024
    grid_mask = (triton.cdiv(total_elems, BLOCK_SIZE),)
    _mask_gate_fuse_kernel[grid_mask](
        left_proj_f, right_proj_f,
        left_gate_f, right_gate_f,
        mask_fp16,
        left_out_f, right_out_f,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # reshape back to [B, N, N, H]
    left  = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. N³ contraction via custom batched GEMM (no extra permutes)
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # FP16 output tensor

    # strides (in elements) of the three tensors
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Tile sizes – tuned for H100 Tensor‑Core FP16 performance
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_gemm = (
        B * hidden_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_gemm_kernel[grid_gemm](
        left, right, out,
        B, N, hidden_dim,
        s_l_b, s_l_i, s_l_k, s_l_h,
        s_r_b, s_r_i, s_r_k, s_r_h,
        s_o_b, s_o_i, s_o_j, s_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Post‑contraction LayerNorm (FP32) + output gate
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)      # broadcast over H

    # ------------------------------------------------------------------
    # 6. Final projection back to the original dimension (FP16 → FP32)
    # ------------------------------------------------------------------
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)               # (D, H) weight, FP16 matmul
    out = out.to(torch.float32)

    return out
