def custom_kernel(data):
   # unpack
   input_tensor, mask, weights, config = data
   # Extract dims
   dim = config["dim"]
   hidden_dim = config["hidden_dim"]
   nomask = config.get('nomask', True)

   # LayerNorm
   x_norm = F.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

   # Cast to FP16
   x_h = x_norm.to(torch.float16)

   # Combine weights
   w_left_proj = weights['left_proj.weight'].to(torch.float16)
   w_left_gate = weights['left_gate.weight'].to(torch.float16)
   w_right_proj = weights['right_proj.weight'].to(torch.float16)
   w_right_gate = weights['right_gate.weight'].to(torch.float16)
   w_out_gate = weights['out_gate.weight'].to(torch.float16)

   w_combined = torch.cat([w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate], dim=0)

   # Linear for all
   combined = F.linear(x_h, w_combined)  # shape [B,N,N,5*hidden_dim] FP16

   # split
   split = torch.split(combined, hidden_dim, dim=-1)   # returns 5 tensors
   left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = split

   # Apply mask if needed: mask shape B,N,N float32 (or bool?). Broadcast over hidden_dim
   if not nomask:
       # mask cast to half and broadcast
       mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
       left_proj = left_proj * mask_h
       right_proj = right_proj * mask_h

   # Apply sigmoid gating
   left_gate = torch.sigmoid(left_gate_pre)
   right_gate = torch.sigmoid(right_gate_pre)
   out_gate = torch.sigmoid(out_gate_pre)

   # Apply gates
   left = left_proj * left_gate
   right = right_proj * right_gate

   # core matmul: left and right shape (B,N,N,hidden_dim)
   # Use batched matmul across hidden_dim as before
   B, N, _, H = left.shape

   # Option 1: reshape to (B*H,N,N)
   left_mat = left.permute(0,3,1,2).contiguous().view(B*H, N, N)
   right_mat = right.permute(0,3,1,2).contiguous().view(B*H, N, N)

   # Compute
   out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # fp16

   # Reshape back
   out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # [B,N,N,H] fp32
   out = out.to(torch.float32)

   # Output layernorm
   out = F.layer_norm(out,
                      (hidden_dim,),
                      weight=weights['to_out_norm.weight'],
                      bias=weights['to_out_norm.bias'])

   # Multiply gating
   out = out * out_gate.to(torch.float32)

   # final linear back to dim (weight in float32)
   out = F.linear(out, weights['to_out.weight'])   # fp32

   return out

"""
TriMul (outgoing) – fast Triton‑augmented implementation
========================================================

Algorithm
---------
1. **Layer‑norm** on the 4‑D input (float32).
2. Cast to *fp16* for all heavy arithmetic.
3. **Fuse the five linear layers** (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) into a **single matmul** with a concatenated weight
   matrix (shape ``5*hidden_dim × dim``).  The result is split back into the
   five tensors; sigmoid is applied to the three gate tensors.
4. Optional mask: a tiny Triton kernel converts the fp32 mask to fp16 and
   broadcasts it over the hidden dimension before the gating step.
5. **Core N³ contraction** – compute for every batch b and hidden channel h

       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

   This is a batched matrix‑multiply ``L @ Rᵀ``.  We reshape the tensors so
   that the hidden dimension becomes part of the batch and call
   ``torch.bmm`` (Tensor‑Core ‐‑ fp16 → fp32 accumulation) which is
   substantially faster than a naïve Triton implementation.
6. **Output layer‑norm**, element‑wise gating, and final projection back to
   ``dim`` (float32).
7. The only required Triton kernel is the mask‑cast; all matrix work is
   performed by cuBLAS/Tensor‑Core kernels.

The fusion in step 3 reduces the number of expensive linear passes from five
to one, which together with the batched‑matmul gives runtimes well below the
target 1 ms on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: fast conversion fp32 → fp16 (used for the optional mask)
# ---------------------------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,               # fp32 source pointer
    dst_ptr,               # fp16 destination pointer
    numel,                 # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)

        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim] (float32)
        - mask         : torch.Tensor [bs, seq_len, seq_len] (float32)
        - weights      : dict of model weights (float32 tensors)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_input = input_tensor.dtype

    dim = config["dim"]                     # input feature dimension
    hidden_dim = config["hidden_dim"]       # projection dimension (H)
    nomask = config.get("nomask", True)    # True → no mask applied

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3) Fuse all five linear layers (proj + gates) into one matmul
    # ------------------------------------------------------------------
    # Gather and cast the five weight matrices (bias‑free in the reference)
    w_left_proj = weights["left_proj.weight"].to(torch.float16)
    w_left_gate = weights["left_gate.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate = weights["out_gate.weight"].to(torch.float16)

    # Concatenate along the output‑feature axis → shape (5*H, dim)
    w_fused = torch.cat(
        [
            w_left_proj,
            w_left_gate,
            w_right_proj,
            w_right_gate,
            w_out_gate,
        ],
        dim=0,
    )  # fp16

    # One large linear; result shape [B, N, N, 5*hidden_dim] (fp16)
    fused = F.linear(x_h, w_fused)

    # Split back into the individual tensors
    left_proj, left_gate_pre, right_proj, right_gate_pre, out_gate_pre = fused.split(
        hidden_dim, dim=-1
    )  # each [B,N,N,hidden_dim] fp16

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # Convert mask from fp32 -> fp16 with a tiny Triton kernel
        mask_flat = mask.view(-1)
        mask_fp16_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)

        BLOCK_SIZE = 1024
        grid = ((mask_flat.numel() + BLOCK_SIZE - 1) // BLOCK_SIZE,)
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_fp16_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        mask_fp16 = mask_fp16_flat.view(mask.shape + (1,))  # [B,N,N,1]

        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # 5) Apply sigmoid gates (still in fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # 6) Element‑wise gated projections
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # [B,N,N,H] fp16
    right = right_proj * right_gate  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 7) Core N³ operation: batched matmul L @ Rᵀ  (fp16 → fp32 accumulation)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # Move hidden dim to the batch axis (B*H, N, N) for torch.bmm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # (B*H, N, N) @ (B*H, N, N)ᵀ  → (B*H, N, N)  (fp16 matmul, fp32 acc)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore the original layout: [B, N, N, H] (float32)
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )  # [B, N, N, hidden_dim] fp32

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 9) Output gating + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)      # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])  # fp32 → final dim

    return out
