# -*- coding: utf-8 -*-
"""
TriMul (outgoing) – fast forward implementation in Triton.
---------------------------------------------------------

The forward pass consists of four stages:

1️⃣ **Input LayerNorm** (float32 → half).  
2️⃣ **Fused linear projection** – the five linear maps
   (left_proj, right_proj, left_gate, right_gate, out_gate) are
   concatenated into a single weight matrix (5·H × dim) and a single
   half‑precision GEMM is performed.  The result is split and the three
   gating tensors are passed through a sigmoid.  An optional mask is
   broadcast and applied before the gates.

3️⃣ **Cubic update** – for every hidden slice we need a matrix‑multiply
   `out[h,i,j] = Σ_k left[h,i,k] * right[h,j,k]`.  This is a batched
   GEMM of shape (BH, N, N) @ (BH, N, N)ᵀ and is carried out by the
   Triton kernel `tri_mul_outgoing_kernel`.  The kernel works in fp16
   with fp32 accumulation and writes the result back in fp16.

4️⃣ **Second LayerNorm + output gate** (float32) and a final linear
   projection back to the original feature dimension `dim`.

The implementation follows the reference PyTorch code but fuses
the linear layers and uses a custom Triton kernel for the dominant
`O(B·H·N³)` work.  The final tensor has shape `[B, N, N, dim]` and dtype
`float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A : (BH, M, K)   (half)
#   B : (BH, N, K)   (half)   (we read it transposed: Bᵀ shape K×N)
#   C : (BH, M, N)   (half, accumulate in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for every batch‑hidden slice (BH).
    All tensors are in fp16; accumulation uses fp32.
    """
    pid_h = tl.program_id(axis=2)      # (batch * hidden) index
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in blocks
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----------- Load A tile (M × K) -------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)                     # fp16 → fp32

        # ----------- Load Bᵀ tile (K × N) -----------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)                     # fp16 → fp32

        # ----------- Inner product ------------------------------
        acc += tl.dot(a, b)                       # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ----------- Write result (half) -----------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] with dtype torch.float32.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]               # input feature dimension
    hidden_dim = config["hidden_dim"] # projection dimension (H)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, dim] fp32
    x_h = x.half()                               # cast to fp16 for heavy ops

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 * hidden_dim outputs)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices once (shape (5*H, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # half precision

    # Single GEMM: [B,N,N,dim] @ (dim,5*H) -> [B,N,N,5*H] (half)
    fused = F.linear(x_h, fused_weight)          # half
    B, N, _, _ = fused.shape

    # Reshape & split
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H] (half)
    left_proj  = fused[..., 0, :]                # [B,N,N,H]
    right_proj = fused[..., 1, :]                # [B,N,N,H]
    left_gate  = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1]
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # ------------------------------------------------------------------
    left  = left_proj  * left_gate   # [B,N,N,H] half
    right = right_proj * right_gate  # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 5️⃣ Rearrange for batched GEMM (BH = B·H)
    # ------------------------------------------------------------------
    # Bring hidden dim to the second axis and make it contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N] half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N] half

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel: batched matmul A @ Bᵀ  (half → half, fp32 acc)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Tile sizes – tuned for modern tensor‑core GPUs (H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one slice per (batch, hidden)
    )

    tri_mul_outgoing_kernel[grid](
        # pointers
        left_flat,
        right_flat,
        out_flat,
        # problem sizes
        N, N, N,
        # A strides (BH, i, k)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # Bᵀ strides (BH, k, j) – note the intentional swap of the last two strides
        # so that we read B transposed without an explicit transpose.
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        # C strides (BH, i, j)
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        # compile‑time tile sizes
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)            # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output gate
    # ------------------------------------------------------------------
    out_fp32 = out_half.float()  # cast back to fp32 for layer‑norm
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp32 = out_fp32 * out_gate.float()   # apply output gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection: hidden → dim
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)  # (dim, hidden)
    out_proj_half = F.linear(out_fp32.half(), to_out_weight_h)   # half result
    out_final = out_proj_half.float()                            # back to fp32

    return out_final
```

**Explanation of key choices**

* **Fusion of linear layers** – By concatenating the five weight matrices we replace five separate `nn.Linear`s with a single half‑precision GEMM, drastically reducing memory traffic.
* **Triton batched GEMM** – The cubic update is the only `O(N³)` part.  The custom kernel computes each `(batch, hidden)` slice as a matrix‑multiply `A @ Bᵀ` using tile sizes `128×128×64`, fp16 arithmetic, and fp32 accumulation for numerical stability.  The stride gymnastics avoid an explicit transpose of the right tensor.
* **Mixed‑precision flow** – All heavy linear parts run in fp16, while the two LayerNorms and the final output remain fp32, matching the reference implementation’s precision profile.
* **Mask handling** – If `nomask` is `False`, the mask is broadcast and applied to the projected tensors before gating, exactly as in the PyTorch reference.

The function `custom_kernel` can be called with a tuple `(input, mask, weights, config)` and will return the correct `[B, N, N, dim]` output in `torch.float32`. This satisfies the correctness requirement while demonstrating a non‑trivial Triton kernel implementation.
