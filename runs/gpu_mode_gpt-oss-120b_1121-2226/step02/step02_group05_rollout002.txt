"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Layer‑norm over the last dimension of the input (float32 → fp16).
2️⃣ Linear projections (left/right) + optional mask (fp16).
3️⃣ Gating (sigmoid) fused with projections (fp16).
4️⃣ Cubic contraction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   implemented as a batched GEMM per hidden channel using a custom
   Triton kernel (FP16 inputs, FP32 accumulation, FP16 output).
5️⃣ Second Layer‑norm over the hidden dimension (fp32), apply output gate,
   and a final linear projection back to the original `dim` (float32).

The heavy N³ work is the batched GEMM; the kernel tiles the matrices
and uses Tensor‑Core `tl.dot` for high throughput on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute batched matrix multiplication with FP16 inputs and
    FP32 accumulation. One program works on a tile (BLOCK_M x BLOCK_N)
    of a single batch (merged batch*hidden dimension).
    """
    pid_batch = tl.program_id(2)   # merged batch/hidden
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to A and B tiles for this batch
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        # Load tiles (apply masks on the fly)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Tensor‑core dot product (FP16 x FP16 → FP32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton batched GEMM kernel.

    Args:
        a: (B, M, K) torch.float16 contiguous
        b: (B, K, N) torch.float16 contiguous

    Returns:
        c: (B, M, N) torch.float16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    B2, K2, N = b.shape
    assert B == B2 and K == K2, "Mismatched batch or inner dimensions"

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Simple heuristic for block sizes
    if max(M, N) <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,   # tiles in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles in N dimension
        B                               # merged batch*hidden dimension
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (tuple):
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 2️⃣ Load weights (cast to half where possible)
    # --------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections (no bias)
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()

    # Gating matrices
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    # Output LayerNorm + linear
    to_out_norm_w = weights["to_out_norm.weight"].to(device)
    to_out_norm_b = weights["to_out_norm.bias"].to(device)
    to_out_w = weights["to_out.weight"].to(device)

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (FP32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x_h = x.half()  # reduce bandwidth for subsequent ops

    # --------------------------------------------------------------
    # 4️⃣ Projections (FP16)
    # --------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden] fp16
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # Convert mask to fp16 and broadcast
        mask_h = mask.to(dtype=left.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) in FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Prepare tensors for the cubic contraction
    # --------------------------------------------------------------
    # left  : (B, N, N, H) → (B, H, N, N)   (i,k)
    # right : (B, N, N, H) → (B, H, N, N)   (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # [B, H, N, N] (k,j)

    # Release unneeded tensors early
    del left, right, left_gate, right_gate, x_h

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions to feed the batched GEMM kernel
    A = left_perm.view(B * H, N, N)          # (B*H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)     # (B*H, K=N, N)

    # --------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (FP16 inputs, FP32 accum)
    # --------------------------------------------------------------
    C = _batched_matmul_fp16(A, B_mat)      # (B*H, N, N) fp16

    # Restore original layout: (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (convert to FP32)
    # --------------------------------------------------------------
    out = out.float()
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 🔟 Apply output gate (cast gate to FP32 for stability)
    # --------------------------------------------------------------
    out = out * out_gate.float()

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to `dim` (float32)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim] float32

    return out
