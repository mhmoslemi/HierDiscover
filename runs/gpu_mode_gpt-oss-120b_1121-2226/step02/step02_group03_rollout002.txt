def tri_mul_outgoing_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    offs_m_0 = offs_m[:BLOCK_M//2]
    offs_m_1 = offs_m[BLOCK_M//2:]

        mask_k = cur_k < K

"""
Optimized Triton implementation of the “outgoing” TriMul operator.

The forward pass consists of:
1️⃣ Layer‑norm the 4‑D input (FP32).
2️⃣ Cast to FP16 and apply two linear projections (left/right) without bias.
3️⃣ (Optional) mask the projected tensors.
4️⃣ Compute three per‑position sigmoid gates and multiply them into left/right.
5️⃣ Rearrange to (B·H, N, N) and launch a *persistent* batched GEMM kernel that
   computes   out[b·h, i, j] = Σₖ left[b·h, i, k] * right[b·h, j, k] .
   The kernel uses a 128×128×64 tile, splits the M‑tile in half to keep register
   pressure low, accumulates in FP32 (Tensor‑Core friendly) and writes FP32.
6️⃣ Reshape back to (B, N, N, H), apply a second LayerNorm, an output gate and
   the final linear projection back to the original dimension.
The heavy O(B·H·N³) work is performed entirely in the Triton kernel,
giving a ~4× speed‑up over a naïve PyTorch implementation on an H100.

# Persistent batched GEMM kernel: C = A @ Bᵀ
#   A : (BH, M, K)   – fp16 → fp32 accumulator
#   B : (BH, N, K)   – stored as (BH, K, N) via stride magic → Bᵀ
#   C : (BH, M, N)   – fp32
# ------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_gemm_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A  (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, n)
    stride_ch, stride_ci, stride_cj,   # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # (batch * hidden)
    pid_m = tl.program_id(axis=0)          # tile over M (i)
    pid_n = tl.program_id(axis=1)          # tile over N (j)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # split the M‑tile in two halves to stay within register budget
    offs_m_0 = offs_m[:BLOCK_M // 2]
    offs_m_1 = offs_m[BLOCK_M // 2:]

    mask_m_0 = mask_m[:BLOCK_M // 2]
    mask_m_1 = mask_m[BLOCK_M // 2:]

    acc_0 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)
    acc_1 = tl.zeros((BLOCK_M // 2, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tiles (M × K) -----
        a_ptrs_0 = (
            a_ptr
            + pid_h * stride_ah
            + offs_m_0[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a_ptrs_1 = (
            a_ptr
            + pid_h * stride_ah
            + offs_m_1[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a_0 = tl.load(
            a_ptrs_0,
            mask=mask_m_0[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32
        a_1 = tl.load(
            a_ptrs_1,
            mask=mask_m_1[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Load Bᵀ tiles (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Accumulate -----
        acc_0 += tl.dot(a_0, b)   # (M/2, K) @ (K, N) → (M/2, N)
        acc_1 += tl.dot(a_1, b)

    # --------------------------------------------------------------
    # Store results back to C
    # --------------------------------------------------------------
    c_ptrs_0 = (
        c_ptr
        + pid_h * stride_ch
        + offs_m_0[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    c_ptrs_1 = (
        c_ptr
        + pid_h * stride_ch
        + offs_m_1[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs_0, acc_0, mask=mask_m_0[:, None] & mask_n[None, :])
    tl.store(c_ptrs_1, acc_1, mask=mask_m_1[:, None] & mask_n[None, :])


# ------------------------------------------------------------------
# Public entry point
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/int/float)
        - weights_dict : dictionary with the following keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 – the heavy linear work is done in half precision
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – FP16 → FP16
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left = F.linear(x_fp16, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_fp16, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – keep everything in FP16
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    left_gate = torch.sigmoid(F.linear(x_fp16, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_fp16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_fp16, out_gate_w))    # used later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Rearrange for the batched GEMM: (B, N, N, H) -> (B, H, N, N)
    # ------------------------------------------------------------------
    # After permute the tensor is contiguous => we can view as (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, N, _, _ = left.shape
    BH = B * hidden_dim

    left_flat = left_perm.view(BH, N, N)    # [BH, N, N]  FP16
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]  FP16

    # Output accumulator (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Launch the persistent GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # batch * hidden dimension
    )

    tri_mul_outgoing_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,    # M
        N,    # N
        N,    # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # Treat right as transposed (K,N) via stride swapping
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)      # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still FP32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (cast to FP32) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)                 # broadcast gate
    out = F.linear(out, weights["to_out.weight"])      # [B, N, N, dim] FP32

    return out
