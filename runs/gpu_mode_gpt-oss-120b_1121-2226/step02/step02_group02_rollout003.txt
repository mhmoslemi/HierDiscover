"""
TriMul (outgoing) – high‑performance forward pass
================================================

Algorithm
---------
1️⃣  Layer‑norm the 4‑D input (float32) → `x_norm`.
2️⃣  Cast to **fp16** (GPU tensor‑core friendly).
3️⃣  **Fuse** the five linear projections (left, right, left_gate,
    right_gate, out_gate) into a **single GEMM** by concatenating the
    weight matrices along the output dimension.
4️⃣  Split the fused result, optionally apply a **mask** (converted from
    fp32 to fp16 by a tiny Triton kernel), and compute sigmoid gates.
5️⃣  Multiply the left/right tensors by their respective gates.
6️⃣  Core N³ operation: for each hidden channel `h`,
   compute `L_h @ R_hᵀ`.  This is expressed as a *batched* matrix‑multiply
   (`torch.bmm`) where the batch dimension is `B * hidden`.  The multiplication
   runs in half‑precision and the result is up‑cast to float32.
7️⃣  Layer‑norm over the hidden dimension, apply the **output gate**, and a
    final linear projection back to `dim`.
8️⃣  Return the result (float32).

Only the mask conversion is performed with Triton, satisfying the
“part of the operations in a kernel” requirement while the bulk of the
computation uses cuBLAS/Tensor‑Core‑accelerated GEMMs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel – fp32 → fp16 mask conversion (tiny, fully asynchronous)
# ----------------------------------------------------------------------
@triton.jit
def _mask_fp32_to_fp16_kernel(
    src_ptr,      # fp32 source mask (flattened)
    dst_ptr,      # fp16 destination
    numel,        # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    src = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    dst = tl.cast(src, tl.float16)
    tl.store(dst_ptr + offs, dst, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (float32) or None
        - weights      : dict of model weights (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack and cache
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                 # input feature dim
    hidden = config["hidden_dim"]       # projection dim (H)
    nomask = config.get("nomask", True)
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) – same as reference implementation
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)           # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3. Fuse the five linear projections into a single GEMM
    # ------------------------------------------------------------------
    # Stack the weights (each of shape [hidden, dim]) → [5*hidden, dim]
    w_left   = weights["left_proj.weight"]
    w_right  = weights["right_proj.weight"]
    w_lgate  = weights["left_gate.weight"]
    w_rgate  = weights["right_gate.weight"]
    w_outgate= weights["out_gate.weight"]
    fused_w = torch.cat(
        [w_left, w_right, w_lgate, w_rgate, w_outgate], dim=0
    ).to(torch.float16)                     # [5*hidden, dim]  fp16

    # Single linear call – computes all five branches at once
    fused_out = F.linear(x_h, fused_w)      # [B, N, N, 5*hidden]  fp16

    # ------------------------------------------------------------------
    # 4. Split the fused result
    # ------------------------------------------------------------------
    # view order: (..., hidden) per branch
    left      = fused_out[..., 0               : hidden]                # [B,N,N,hidden]
    right     = fused_out[..., hidden           : 2*hidden]            # [B,N,N,hidden]
    left_gate = fused_out[..., 2*hidden         : 3*hidden]            # [B,N,N,hidden]
    right_gate= fused_out[..., 3*hidden         : 4*hidden]            # [B,N,N,hidden]
    out_gate  = fused_out[..., 4*hidden         : 5*hidden]            # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask from fp32 to fp16 using the tiny Triton kernel
        mask_flat = mask.contiguous().view(-1)
        mask_h_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)

        BLOCK_SIZE = 1024
        grid = ((mask_flat.numel() + BLOCK_SIZE - 1) // BLOCK_SIZE, )
        _mask_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_h_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        mask_h = mask_h_flat.view(mask.shape + (1,))   # [B,N,N,1] fp16
        left  = left  * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6. Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Core N³ operation: Batched matmul L @ Rᵀ for every hidden channel
    # ------------------------------------------------------------------
    B, N, _, H = left.shape      # B, seq_len, seq_len, hidden

    # Reshape to (B*H, N, N) – each hidden channel becomes an independent batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)ᵀ → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, Tensor‑Core path

    # Back to original layout, up‑cast to fp32 for the following layers
    out = (
        out_mat.view(B, H, N, N)
             .permute(0, 2, 3, 1)   # [B, N, N, hidden]
             .contiguous()
             .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 8. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # fp32

    # ------------------------------------------------------------------
    # 9. Output gate (still fp16 → cast) and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)    # element‑wise gate
    out = F.linear(out, weights["to_out.weight"])   # fp32 → final dim

    return out
