```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_gemm(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    B, H,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul "outgoing" operation.

    Computes for each batch b and hidden head h:
        out[b, i, j, h] = sum_{k=0}^{K-1} left[b, i, k, h] * right[b, j, k, h]

    left/right are expected in fp16, accumulation in fp32, output fp32.
    """
    pid = tl.program_id(axis=2)          # combined batch*head index
    pid_m = tl.program_id(axis=0)        # i‑tile
    pid_n = tl.program_id(axis=1)        # j‑tile

    # Derive batch and head indices
    b = pid // H
    h = pid % H

    # Base pointers for this (b,h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of tiles in the K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # -------- load a tile from left (M × K) --------
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)    # fp16 -> fp32

        # -------- load a tile from rightᵀ (K × N) --------
        # right is accessed as right[b, j, k, h] which corresponds to the
        # transposed layout required by the einsum.
        b_ptrs = right_base + cur_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # -------- accumulate --------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # -------- write the result --------
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_k
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    The implementation fuses:
        * input layer‑norm,
        * six linear projections (left/right + their gates + out_gate) into a single
          matmul,
        * the core N³ pairwise multiplication using a custom Triton batched GEMM.
    The final layer‑norm, output‑gate and projection are performed with
    torch.nn.functional for simplicity (they are cheap compared to the N³ work).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear projections (proj + three gates) into one matmul
    # ------------------------------------------------------------------
    # weight shapes: (hidden_dim, dim)
    weight_comb = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # (5*hidden_dim, dim)

    # Single fused matmul (batched)
    fused = F.linear(x_fp16, weight_comb)  # [B, N, N, 5*hidden_dim] (fp16)

    # Split back into the original tensors
    left, left_gate_pre, right, right_gate_pre, out_gate_pre = torch.split(
        fused, hidden_dim, dim=-1
    )

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – keep everything in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core TriMul: batched GEMM left @ rightᵀ  (fp16×fp16→fp32)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)  # [B, N, N, H] fp32

    # Triton kernel launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides (in elements, not bytes)
    stride_l_b = left.stride(0)
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    stride_l_h = left.stride(3)

    stride_r_b = right.stride(0)
    stride_r_i = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_k = out.stride(2)
    stride_o_h = out.stride(3)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden dimension
    )

    tri_mul_outgoing_gemm[grid](
        left,
        right,
        out,
        N,            # M
        N,            # N
        N,            # K
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        B, H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension, output gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    out = out * out_gate  # broadcast gate (fp16→fp32)

    out = F.linear(out, weights["to_out.weight"])   # -> [B, N, N, dim]

    return out

"""
TriMul “outgoing” forward pass (AlphaFold‑3 style)

Algorithm
---------
1. Input LayerNorm (float32).
2. Cast to fp16 and fuse the five linear maps
   (left_proj, left_gate, right_proj, right_gate, out_gate) into a single matmul.
3. Split the fused result, apply the optional mask and the three sigmoid gates.
4. Core N³ pairwise update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched fp16×fp16→fp32 GEMM with a custom Triton kernel.
5. Second LayerNorm over the hidden dimension, multiply by out_gate and a final
   linear projection back to the original channel dimension (float32).

The implementation removes the costly permute+contiguous copy present in the
reference, merges five cheap linear layers into one kernel launch and uses a
high‑performance Triton GEMM for the cubic‑cost step.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Batched GEMM  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_gemm(
    left_ptr, right_ptr, out_ptr,
    M, N, K,                     # all equal to seq_len
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    B, H,                        # batch size and hidden dimension (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM used by the “outgoing” TriMul.

    Computes for each (b, h):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    left/right are fp16, accumulation is fp32, result is fp32.
    """
    pid_h = tl.program_id(axis=2)          # combined batch*head index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # --------------------------------------------------------------
    # Decode batch and head indices
    # --------------------------------------------------------------
    b = pid_h // H
    h = pid_h % H

    # Base pointers for the selected (b,h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- left tile (M × K) -----
        a_ptrs = (
            left_base
            + offs_i[:, None] * stride_left_i
            + cur_k[None, :] * stride_left_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)        # fp16 → fp32

        # ----- rightᵀ tile (K × N) -----
        # right is accessed as right[b, j, k, h] (notice the swap i↔j)
        b_ptrs = (
            right_base
            + cur_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Accumulate -----
        acc += tl.dot(a, b)       # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write the result back to global memory
    # --------------------------------------------------------------
    c_ptrs = (
        out_base
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_k
    )
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the cheap linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear maps into one matmul
    # ------------------------------------------------------------------
    # Each weight has shape (hidden_dim, dim)
    weight_comb = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()                                           # (5*hidden_dim, dim)

    fused = F.linear(x_fp16, weight_comb)               # [B, N, N, 5*hidden_dim] (fp16)

    left, left_gate_pre, right, right_gate_pre, out_gate_pre = torch.split(
        fused, hidden_dim, dim=-1
    )                                                    # each [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – stay in fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Core TriMul (batched GEMM) – fp16×fp16 → fp32
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # H == hidden_dim
    out = torch.empty_like(left, dtype=torch.float32)   # [B, N, N, H] fp32

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # Strides in element units (not bytes)
    stride_l_b = left.stride(0)
    stride_l_i = left.stride(1)   # i‑axis
    stride_l_k = left.stride(2)   # k‑axis
    stride_l_h = left.stride(3)

    stride_r_b = right.stride(0)
    stride_r_i = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_k = out.stride(2)
    stride_o_h = out.stride(3)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch * hidden dimension
    )

    tri_mul_outgoing_gemm[grid](
        left,
        right,
        out,
        N,            # M
        N,            # N
        N,            # K
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_k, stride_o_h,
        B, H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm, output‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                   # still fp32

    out = out * out_gate                # broadcast, fp16→fp32

    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
