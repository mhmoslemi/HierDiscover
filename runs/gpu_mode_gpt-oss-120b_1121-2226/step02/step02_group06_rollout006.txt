@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   pid = tl.program_id(0)
   # total number of blocks: B*H*ceil(N/BLOCK_M)*ceil(N/BLOCK_N)
   num_i = tl.cdiv(N, BLOCK_M)
   num_j = tl.cdiv(N, BLOCK_N)
   batch_head = pid // (num_i * num_j)
   block_idx = pid % (num_i * num_j)
   block_i = block_idx // num_j
   block_j = block_idx % num_j

   b = batch_head // H
   h = batch_head % H

   offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
   mask_i = offs_i < N
   mask_j = offs_j < N

   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for kk in range(0, tl.cdiv(N, BLOCK_K)):
       offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # left[b, i, k, h]
       left_ptrs = left_ptr + b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_h
       # right[b, j, k, h]
       right_ptrs = right_ptr + b * stride_right_b + offs_j[:, None] * stride_right_j + offs_k[None, :] * stride_right_k + h * stride_right_h

       a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
       b_ = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
       acc += tl.dot(a, b_)

   out_ptrs = out_ptr + b * stride_out_b + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + h * stride_out_h
   tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul outgoing (AlphaFold3) – Triton‐accelerated forward pass.

Algorithm
---------
1. LayerNorm over the input channel dimension (float32).
2. Cast to FP16 and apply four linear projections without bias:
   left_proj, right_proj, left_gate, right_gate, out_gate.
   Optional mask (B,N,N) is broadcasted and multiplied before gating.
3. Gating: sigmoid on the gate projections and element‑wise multiplication.
4. Contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that treats each (b,h) pair as a
   separate batch and performs a tiled FP16×FP16 → FP32 GEMM with
   BLOCK_M = BLOCK_N = 128, BLOCK_K = 32 (8 warps per block).
   No tensor reshapes or copies are performed – the kernel accesses the
   original (B,N,N,H) layout directly.
5. Convert the result back to FP32, apply a second LayerNorm over the
   hidden dimension, multiply by the output gate, and finally project back
   to the original channel dimension with a float32 linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched contraction for the outgoing TriMul update.
#   left  : (B, N, N, H)  FP16
#   right : (B, N, N, H)  FP16
#   out   : (B, N, N, H)  FP16   (accumulated in FP32)
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # logical dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    #  block decomposition: (batch*head, block_i, block_j)
    # --------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    blocks_per_head = num_i * num_j

    head_idx = pid // blocks_per_head                # 0 .. B*H-1
    block_idx = pid % blocks_per_head

    block_i = block_idx // num_j
    block_j = block_idx % num_j

    # --------------------------------------------------------------
    #  recover batch and hidden indices
    # --------------------------------------------------------------
    b = head_idx // H
    h = head_idx % H

    # --------------------------------------------------------------
    #  tile offsets
    # --------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #  loop over the reduction axis k
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]
        left_tile_ptr = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        # right[b, j, k, h]
        right_tile_ptr = (
            right_ptr
            + b * stride_right_b
            + offs_j[:, None] * stride_right_j
            + offs_k[None, :] * stride_right_k
            + h * stride_right_h
        )

        a = tl.load(left_tile_ptr, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b_ = tl.load(right_tile_ptr, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        acc += tl.dot(a, b_)          # FP16×FP16 → FP32 accumulation

    # --------------------------------------------------------------
    #  write the output tile
    # --------------------------------------------------------------
    out_tile_ptr = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_tile_ptr,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _tri_mul_outgoing_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched contraction for outgoing TriMul.

    Parameters
    ----------
    left, right : torch.Tensor
        Shape (B, N, N, H), dtype=torch.float16, contiguous.

    Returns
    -------
    torch.Tensor
        Shape (B, N, N, H), dtype=torch.float16, contiguous.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, N2, H = left.shape
    assert N == N2
    out = torch.empty_like(left)       # FP16, same shape

    # Tile sizes tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # launch grid: one block per (batch*head, tile_i, tile_j)
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * grid_i * grid_j
    grid = (total_blocks,)

    # strides (int64)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_left[0],
        s_left[1],
        s_left[2],
        s_left[3],
        s_right[0],
        s_right[1],
        s_right[2],
        s_right[3],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]            # keep FP32 for final proj

    # ------------------------------------------------------------------
    # 1️⃣ Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.half()   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]  FP16
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1] FP16
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating tensors (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply gates in‑place
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched contraction (kernel = Σ_k left[b,i,k,h] * right[b,j,k,h])
    # ------------------------------------------------------------------
    out_h = _tri_mul_outgoing_fp16(left, right)   # [B, N, N, hidden_dim] FP16

    # ------------------------------------------------------------------
    # 7️⃣ Second layer‑norm (float32)
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 8️⃣ Output gate & final projection
    # ------------------------------------------------------------------
    out = out * out_gate                                 # broadcasted multiplication
    out = F.linear(out, to_out_w)                        # [B, N, N, dim] FP32

    return out
```
