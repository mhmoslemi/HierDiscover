@triton.jit
def _batched_matmul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak, stride_ah,
    stride_bb, stride_bk, stride_bn, stride_bh,
    stride_cb, stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    H: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute total blocks per batch
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # compute batch*hidden index and block indices
    batch_h = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # unpack batch and hidden
    batch = batch_h // H
    hidden = batch_h % H

    # compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers for this batch+hidden
    a_base = a_ptr + batch * stride_ab + hidden * stride_ah
    b_base = b_ptr + batch * stride_bb + hidden * stride_bh
    # loops over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # compute output pointer with offsets for batch and hidden
    c_ptrs = c_ptr + batch * stride_cb + hidden * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel – Triton‑accelerated outgoing TriMul (forward only)

Algorithm
---------
1️⃣ Layer‑norm over the input `dim` (FP32, torch implementation).
2️⃣ Cast to FP16 and apply four bias‑free linear projections:
      left, right, left_gate, right_gate, out_gate.
   The gates are passed through a sigmoid and multiplied in‑place.
   An optional mask (B,N,N) is broadcasted and applied before gating.
3️⃣ Batched contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a custom Triton kernel that performs a batched GEMM
   with arbitrary strides (no intermediate permutes or copies).  The
   hidden dimension is fused into the batch index, i.e. a combined
   batch size of B·H.  Accumulation is FP32 → output FP16.
4️⃣ Convert the result back to FP32 for the second LayerNorm.
5️⃣ Second LayerNorm over the hidden dimension (FP32).
6️⃣ Multiply by the output gate (broadcasted, FP32).
7️⃣ Final linear projection back to `dim` (FP32).

The kernel avoids the costly `permute(...).contiguous()` copies required
by the naïve implementation and therefore reduces memory traffic
dramatically, giving a large speed‑up on large sequence lengths.
"""

# Triton kernel: batched GEMM for the outgoing TriMul contraction.
#   A: (B, H, N, N)   → interpreted as (B*H, M=N, K=N)
#   B: (B, H, N, N)   → interpreted as (B*H, K=N, N=N) (row‑major k, col‑major j)
#   C: (B, H, N, N)   → (B*H, M=N, N=N)
# Strides are passed explicitly, thus no layout transposes are required.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                      # pointers
    M, N, K,                                  # matrix sizes
    stride_ab, stride_am, stride_ak, stride_ah,  # A strides (batch, row, col, hidden)
    stride_bb, stride_bk, stride_bn, stride_bh,  # B strides (batch, row, col, hidden)
    stride_cb, stride_cm, stride_cn, stride_ch,  # C strides (batch, row, col, hidden)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,                      # hidden dimension (static)
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    #  Block decomposition in the (batch*hidden) × M × N space
    # ------------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    # combined batch*hidden index
    batch_hidden = pid // (num_m * num_n)
    block_idx    = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # recover original batch and hidden indices
    batch  = batch_hidden // HIDDEN
    hidden = batch_hidden % HIDDEN

    # ------------------------------------------------------------------
    #  Tile offsets & masks
    # ------------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32 for numerical safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Base pointers for this (batch, hidden) slice
    # ------------------------------------------------------------------
    a_base = a_ptr + batch * stride_ab + hidden * stride_ah
    b_base = b_ptr + batch * stride_bb + hidden * stride_bh

    # ------------------------------------------------------------------
    #  Loop over the reduction dimension K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch, hidden, i, k]   (M×K tile)
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B[batch, hidden, k, j]   (K×N tile)   – note switched axes
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix‑multiply tile (Tensor‑core friendly)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    #  Write result tile C[batch, hidden, i, j]
    # ------------------------------------------------------------------
    c_ptrs = (c_ptr + batch * stride_cb + hidden * stride_ch
              + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)

    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _trimul_outgoing_gemm_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision GEMM for the outgoing TriMul contraction.
    Parameters
    ----------
    left : torch.Tensor
        Shape (B, N, N, H), dtype=torch.float16, contiguous.
    right : torch.Tensor
        Same shape as ``left``.
    Returns
    -------
    torch.Tensor
        Shape (B, N, N, H), dtype=torch.float16, contiguous.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    assert right.shape == (B, N, N, H)

    out = torch.empty_like(left)

    # -------------------- strides --------------------
    stride_ab = left.stride(0)   # batch stride
    stride_am = left.stride(1)   # i (row) stride
    stride_ak = left.stride(2)   # k (col) stride
    stride_ah = left.stride(3)   # hidden stride (usually 1)

    stride_bb = right.stride(0)  # batch stride
    # B is accessed as (k, j) → row = k (dim2), col = j (dim1)
    stride_bk = right.stride(2)  # k stride
    stride_bn = right.stride(1)  # j stride
    stride_bh = right.stride(3)  # hidden stride

    stride_cb = out.stride(0)    # batch stride
    stride_cm = out.stride(1)    # i stride
    stride_cn = out.stride(2)    # j stride
    stride_ch = out.stride(3)    # hidden stride

    # -------------------- launch parameters --------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_batches = B * H
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total_batches * num_m * num_n,)

    _trimul_outgoing_gemm_kernel[grid](
        left,
        right,
        out,
        N,                 # M
        N,                 # N
        N,                 # K
        stride_ab,
        stride_am,
        stride_ak,
        stride_ah,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cb,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=H,
        num_warps=8,                     # empirical good value on H100
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor            shape [B, N, N, dim]
        - mask         : torch.Tensor or None    shape [B, N, N]
        - weights      : dict of torch.Tensor with the model parameters
        - config       : dict with entries "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype=torch.float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # model weights (cast to the precision we will use)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]               # keep FP32

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the input `dim` (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_h = x.half()                 # [B, N, N, dim]   FP16

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (bias‑free) in FP16
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]  FP16
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask – broadcast over the hidden dimension
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1] FP16
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating tensors (sigmoid) – still in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply gates in‑place
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched contraction using the custom Triton kernel
    # --------------------------------------------------------------
    out_h = _trimul_outgoing_gemm_fp16(left, right)   # [B, N, N, hidden_dim]  FP16

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32 for the second LayerNorm
    # --------------------------------------------------------------
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # --------------------------------------------------------------
    # 9️⃣ Apply the output gate (broadcasted)
    # --------------------------------------------------------------
    out = out * out_gate.to(torch.float32)

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]   FP32

    return out
