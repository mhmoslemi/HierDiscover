@triton.jit
def tri_mul_outgoing_kernel(a_ptr, b_ptr, c_ptr,
                            M, N, K,
                            stride_ah, stride_ai, stride_ak,
                            stride_bh, stride_bk, stride_bn,
                            stride_ch, stride_ci, stride_cj,
                            BLOCK_M: tl.constexpr,
                            BLOCK_N: tl.constexpr,
                            BLOCK_K: tl.constexpr):
    pid_h = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile
        a_ptrs = a_ptr + pid_h * stride_ah + offs_m[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B transposed tile (B^T)
        b_ptrs = b_ptr + pid_h * stride_bh + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = c_ptr + pid_h * stride_ch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
Optimized outgoing TriMul implementation.

Key ideas
---------
1️⃣ Fuse the four linear projections (left_proj, left_gate, right_proj,
   right_gate) **and** the out‑gate into a single half‑precision GEMM.
   This reads the large input tensor only once and reduces memory traffic
   dramatically.

2️⃣ Perform the expensive “left @ rightᵀ” (the N³ kernel) with a custom
   Triton GEMM.  The kernel works on half‑precision inputs and accumulates
   in float‑32, exploiting H100 Tensor‑Cores.

3️⃣ Keep the two LayerNorms, the final linear projection and the final
   gating in FP32 for numerical stability.  These operations are cheap
   compared to the GEMM‑heavy stages.

The function `custom_kernel` follows the exact API required by the test
harness: it receives a tuple `(input_tensor, mask_tensor, weights_dict,
config_dict)` and returns a float‑32 tensor of shape
`[bs, seq_len, seq_len, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ   (A,B are fp16, C is fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A  (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # batch*hidden index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # --------------------------------------------------------------
    # Tile offsets & masks
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # fp16 → fp32

        # ----- Load Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)                     # fp16 → fp32

        # ----- Accumulate -----
        acc += tl.dot(a, b)                  # (BLOCK_M,K) @ (K,BLOCK_N)

    # --------------------------------------------------------------
    # Write result
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                # original channel dimension (c or c_z)
    hidden = config["hidden_dim"]      # projection dimension (c)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, dim] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse all per‑position linear transforms (left/right proj &
    #    gates + out‑gate) into a single GEMM.
    # ------------------------------------------------------------------
    #   ordering: 0..h-1            left_proj
    #            h..2h-1            left_gate
    #            2h..3h-1           right_proj
    #            3h..4h-1           right_gate
    #            4h..5h-1           out_gate
    weight_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device).half()               # (5*hidden, dim)

    # Linear: shape -> [B, N, N, 5*hidden] (fp16)
    all_proj = F.linear(x_fp16, weight_all)

    # ------------------------------------------------------------------
    # 4️⃣ Split the fused tensor
    # ------------------------------------------------------------------
    h = hidden
    left_proj = all_proj[..., 0:h]                     # [B,N,N, h]
    left_gate_pre = all_proj[..., h:2 * h]             # [B,N,N, h]
    right_proj = all_proj[..., 2 * h:3 * h]            # [B,N,N, h]
    right_gate_pre = all_proj[..., 3 * h:4 * h]        # [B,N,N, h]
    out_gate_pre = all_proj[..., 4 * h:5 * h]          # [B,N,N, h]

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 6️⃣ Apply sigmoid gates (still half‑precision)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)    # [B,N,N, h]  fp16
    right_gate = torch.sigmoid(right_gate_pre)  # [B,N,N, h]  fp16
    out_gate = torch.sigmoid(out_gate_pre)      # [B,N,N, h]  fp16

    # ------------------------------------------------------------------
    # 7️⃣ Gate the projections
    # ------------------------------------------------------------------
    left = left_proj * left_gate                # fp16
    right = right_proj * right_gate             # fp16

    # ------------------------------------------------------------------
    # 8️⃣ Reshape for the batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, N, N, h) -> (B, h, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden                     # batch * hidden
    left_flat = left_perm.view(BH, N, N)    # [BH, N, N]  fp16
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]  fp16

    # ------------------------------------------------------------------
    # 9️⃣ Batched GEMM with Triton (A @ Bᵀ)
    # ------------------------------------------------------------------
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # fp32 accumulator

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # batch*hidden
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10️⃣ Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden, N, N)          # [B, h, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, h]  fp32

    # ------------------------------------------------------------------
    # 11️⃣ Second LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # ------------------------------------------------------------------
    # 12️⃣ Apply the final output gate (cast to float32) and projection
    # ------------------------------------------------------------------
    out = out * out_gate.float()
    out = F.linear(out, weights["to_out.weight"])  # → [B, N, N, dim] (float32)

    return out
