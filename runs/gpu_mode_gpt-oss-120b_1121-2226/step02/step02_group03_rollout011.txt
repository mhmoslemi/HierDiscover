"""
TriMul “outgoing” implementation (AlphaFold3)

Algorithm
---------
1.  Layer‑norm the 4‑D input (float32).
2.  Cast to fp16 and fuse the two linear projections (left/right) into a
    single matmul:   X @ [W_left; W_right]ᵀ   →  shape [B,N,N,2·H].
3.  If a mask is supplied, broadcast‑multiply it on the projected tensors.
4.  Fuse the three linear gates (left, right, out) into a single matmul:
    X @ [W_lgate; W_rgate; W_ogate]ᵀ   →  shape [B,N,N,3·H],
    then apply `sigmoid`.
5.  Apply the left/right gates to the projected tensors.
6.  Reshape to [B·H, N, N] and run a custom Triton GEMM
    `out = left @ rightᵀ` (fp16 × fp16 → fp32) using a 128×128×64
    tiling schedule.
7.  Reshape back to [B,N,N,H] and apply a second LayerNorm.
8.  Apply the output gate and the final linear projection back to the
    original dimension.

The heavy linear work (steps 2 & 4) is reduced from 5 matmuls to 2,
and the O(N³) multiplication is performed by a fused Triton kernel
(step 6).  The final result is a float‑32 tensor
`[bs, seq_len, seq_len, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B are fp16, C is fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A  (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # (batch * hidden) index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # ------------------------------------------------------------------
    # Tile offsets & bounds
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile (M × K)
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # Bᵀ tile (K × N)
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16 → fp32

        # Accumulate
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for heavy linear work
    # ------------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, dim] – fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse left & right projections into a single matmul
    # ------------------------------------------------------------------
    left_w = weights["left_proj.weight"].half()   # [hidden, dim]
    right_w = weights["right_proj.weight"].half() # [hidden, dim]
    proj_weight = torch.cat([left_w, right_w], dim=0)  # [2*hidden, dim]

    proj = F.linear(x_fp16, proj_weight)           # [B, N, N, 2*hidden]
    left = proj[..., :hidden]                     # [B, N, N, hidden]
    right = proj[..., hidden:]                    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → broadcast to hidden dim
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # fp16
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5️⃣ Fuse three gate linear layers (left, right, out) then sigmoid
    # ------------------------------------------------------------------
    lg_w = weights["left_gate.weight"].half()
    rg_w = weights["right_gate.weight"].half()
    og_w = weights["out_gate.weight"].half()
    gate_weight = torch.cat([lg_w, rg_w, og_w], dim=0)  # [3*hidden, dim]

    gates = F.linear(x_fp16, gate_weight)               # [B, N, N, 3*hidden]
    left_gate = torch.sigmoid(gates[..., :hidden])
    right_gate = torch.sigmoid(gates[..., hidden:2 * hidden])
    out_gate = torch.sigmoid(gates[..., 2 * hidden:])

    # ------------------------------------------------------------------
    # 6️⃣ Apply left/right gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Batched GEMM: out = left @ rightᵀ  (fp16 → fp32)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, N, N, H) → (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)    # [BH, N, N]   fp16
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]   fp16

    # Output buffer (accumulate in fp32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden) dimension
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        # For Bᵀ we swap the K and N strides (see kernel comment)
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden, N, N)            # [B, H, N, N]  fp32
    out = out.permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # still float32

    # ------------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate.float()                     # broadcast gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] – float32

    return out
