"""
Highly‑optimized Triton implementation of the “outgoing” TriMul operation
from AlphaFold‑3.

Algorithm
---------
1️⃣  Layer‑norm over the last dimension (float32) → half.
2️⃣  Fuse the five linear projections (left, right, left_gate, right_gate,
    out_gate) into a single GEMM (half precision).  The result is reshaped
    to obtain the three projections and three sigmoid gates.
3️⃣  (Optional) apply the pairwise mask to the two projection tensors.
4️⃣  Apply the element‑wise gates.
5️⃣  Rearrange tensors so that each (batch, hidden) slice can be processed
    as a matrix‑multiply:  left[b,h,i,k] * right[b,h,j,k]  →  out[b,h,i,j].
    This is a batched mat‑mul A @ Bᵀ and is computed with a custom Triton
    kernel that uses FP16 inputs, FP32 accumulation and Tensor‑Core
    `tl.dot`.  Tile sizes (128×128×64) were chosen for H100.
6️⃣  Reshape back to [B, N, N, hidden_dim] (half), apply a second
    LayerNorm (float32), scale by the output gate and perform the final
    linear projection back to `dim`.
7️⃣  Return a float32 tensor of shape [B, N, N, dim].

The kernel works for any batch size (B∈{1,2}), sequence length N∈{128,…,1024},
and hidden dimension = `hidden_dim`.  Masking is optional.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ   (A,B half → C half, acc in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem dimensions (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision mat‑mul with FP32 accumulation.
    Computes   C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k] .
    """
    pid_h = tl.program_id(axis=2)        # (batch * hidden) index
    pid_m = tl.program_id(axis=0)        # i‑tile
    pid_n = tl.program_id(axis=1)        # j‑tile

    # ------------------------ tile offsets ----------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------ accumulator -----------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------ K‑loop ---------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- load a tile of A (M × K) -----------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)            # fp16 → fp32

        # ----- load a tile of Bᵀ (K × N) ---------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk   # K dimension (stride over k)
            + offs_n[None, :] * stride_bn   # N dimension (stride over j)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)            # fp16 → fp32

        # ----- inner product (dot) ---------------------------------
        acc += tl.dot(a, b)             # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ------------------------ store result -----------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor  [B, N, N] (bool/float)
        - weights_dict : dict with the following keys
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : dict containing
            * "dim" (int)
            * "hidden_dim" (int)
            * "nomask" (bool, optional, default=True)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                           # [B, N, N, dim]  float32
    x_h = x.half()                              # half precision for heavy ops

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear stage (produces left/right projections + gates)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices: (5*H, dim)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # half

    # Single GEMM:   [B,N,N,dim] @ [dim,5*H] → [B,N,N,5*H]
    fused = F.linear(x_h, fused_weight)        # half
    B, N, _, _ = fused.shape

    # Split fused output
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H]
    left_proj  = fused[..., 0, :]                # [B,N,N,H]
    right_proj = fused[..., 1, :]
    left_gate  = torch.sigmoid(fused[..., 2, :])  # sigmoid → half
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :])  # later used after to_out_norm

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → half
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply per‑position gating (still half)
    # ------------------------------------------------------------------
    left  = left_proj * left_gate                # [B,N,N,H] half
    right = right_proj * right_gate              # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 5️⃣ Prepare tensors for the batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    #   left  : [B,N,N,H] → permute → [B,H,N,N]
    #   right : [B,N,N,H] → permute → [B,H,N,N] (will be transposed inside kernel)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    BH = B * hidden_dim                       # total number of (batch, hidden) slices
    left_flat  = left_perm.view(BH, N, N)     # [BH, N, N] half
    right_flat = right_perm.view(BH, N, N)    # [BH, N, N] half

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel: batched GEMM A @ Bᵀ  →  out_flat  (half)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Tile sizes – empirically tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # each (batch, hidden) slice
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K (all = seq_len)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,        # 8 warps → 256 threads per program
        num_stages=3,       # triple buffering
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, hidden_dim] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)      # [B,H,N,N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output‑gate
    # ------------------------------------------------------------------
    out_fp32 = out_half.float()  # cast for layer‑norm
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                           # [B,N,N,H]  float32
    out_fp32 = out_fp32 * out_gate.float()  # apply output gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w_h = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out_fp32.half(), to_out_w_h)   # half → [B,N,N,dim]
    out_final = out_proj.float()                       # back to float32

    return out_final
```
