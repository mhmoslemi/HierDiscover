"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy N³ contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix‑multiply (A @ B) per (batch, hidden) pair
using a custom Triton kernel.  All inexpensive operations (layer‑norm,
linear projections, gating, final projection) are kept in FP16 where possible
and only the final output is cast back to FP32.

Key ideas
---------
* First LayerNorm in FP32 → fp16 to cut bandwidth.
* Left/right projections and their gates are computed with torch.nn.functional.
* Optional mask is fused into the projections in‑place.
* The contraction is a batched GEMM on tensors of shape (B·H, N, N) with
  FP16 inputs / FP32 accumulators.  Tile sizes are chosen heuristically.
* Second LayerNorm, output‑gate and final linear projection restore the
  original hidden dimension (dim) in FP32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched FP16 GEMM with FP32 accumulation.

    One program instance processes a (BLOCK_M, BLOCK_N) tile of the output
    for a single (batch * hidden) slice.
    """
    pid_batch = tl.program_id(2)       # merged batch*hidden dimension
    pid_m = tl.program_id(0)           # row tile id
    pid_n = tl.program_id(1)           # col tile id

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K) and B tile (K x N)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # FP16 dot → FP32 accumulation
        acc += tl.dot(a, b)

    # Store the tile to C (convert back to FP16)
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    FP16 batched GEMM with FP32 accumulation via the Triton kernel above.

    Args:
        a: (B, M, K) torch.float16, contiguous
        b: (B, K, N) torch.float16, contiguous

    Returns:
        c: (B, M, N) torch.float16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    assert b.shape[0] == B and b.shape[1] == K, "Shapes mismatch"

    N = b.shape[2]
    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Simple heuristic for tile sizes – tuned for H100.
    if max(M, N) <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif max(M, N) <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Number of tiles
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B,
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of named parameters (float32)
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 2️⃣ Load weights – cast to half where possible
    # ------------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections (bias‑free)
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    # Gating projections
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()

    # Output LayerNorm + projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # (B,N,N,dim) float32
    x_h = x.half()  # switch to fp16 for the heavy ops

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim) – in‑place mul
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask is bool/int; cast to fp16 and add singleton dim
        mask_h = mask.to(dtype=left.dtype, device=left.device).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare tensors for the N³ contraction
    #    Permute to (B, H, N, N) where the two N axes are (i,k) and (k,j)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    # Release intermediate tensors early to keep memory pressure low
    del left, right, left_gate, right_gate, x_h

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions for the kernel
    A = left_perm.view(B * H, N, N)          # (B*H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)     # (B*H, K=N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Batched GEMM (FP16 → FP32 accumulation)
    # ------------------------------------------------------------------
    C = _batched_matmul_fp16(A, B_mat)      # (B*H, N, N) fp16

    # Restore original layout: (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (convert to FP32 first)
    # ------------------------------------------------------------------
    out_float = out.float()
    out_float = F.layer_norm(
        out_float,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 10️⃣ Apply output gate (cast gate to FP32 for stability)
    # ------------------------------------------------------------------
    out_float = out_float * out_gate.float()

    # ------------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out_float = F.linear(out_float, to_out_w)   # (B,N,N,dim) float32

    return out_float
