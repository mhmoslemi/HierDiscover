def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layer norm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # cast to fp16
    x_h = x.half()

    # weights half
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]   # float32

    # linear projections
    left = F.linear(x_h, left_proj_w)   # (B,N,N,hidden)
    right = F.linear(x_h, right_proj_w)

    # apply mask if present
    if not nomask:
        # cast mask to fp16 via Triton kernel
        mask_flat = mask.view(-1)
        mask_fp16 = torch.empty_like(mask_flat, dtype=torch.float16, device=device)
        BLOCK = 1024
        grid = ((mask_flat.numel() + BLOCK - 1)//BLOCK,)
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_fp16,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK,
        )
        mask_fp16 = mask_fp16.view(mask.shape + (1,))
        left = left * mask_fp16
        right = right * mask_fp16

    # gating
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    B, N, _, H = left.shape

    # reshape for batched gemm
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # compute batched GEMM using Triton
    out_mat = _batched_gemm_triton(left_mat, right_mat)

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # output layer norm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # output gating and final linear
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)

    return out
```

def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor, BLOCK_M=128, BLOCK_N=128, BLOCK_K=64):
    # A: (batch, M, K) fp16
    # B: (batch, N, K) fp16, we will compute C = A @ B^T
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bn, stride_Bk = B.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    grid = ( (M + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             batch )

    _batched_gemm_kernel[grid](
        A, B, C,
        M, N, K,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bn, stride_Bk,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

"""
TriMul (outgoing) – Triton‑accelerated forward pass
--------------------------------------------------

Algorithm
~~~~~~~~~
1. Layer‑norm the input tensor (float32).
2. Cast to fp16 and apply the two linear projections (left/right) and
   the three gating projections.  All weights are also cast to fp16.
3. Optional mask: a tiny Triton kernel converts the fp32 mask to fp16
   and broadcasts it over the hidden dimension.
4. Apply sigmoid gates (still in fp16) and multiply the masked projections.
5. Core N³ update: for each hidden channel compute a batched matrix‑multiply
   `out_h = left_h @ right_hᵀ`.  This is implemented with a custom Triton
   kernel that fuses the entire batch of GEMMs, using fp16 multiplies and
   fp32 accumulation (Tensor‑Core friendly).
6. Layer‑norm over the hidden dimension, element‑wise multiplication by the
   output gate, and a final linear projection back to the original `dim`.
7. The final result is returned as float32.

Only the mask cast and the N³ batched GEMM are executed in Triton; the rest
uses standard PyTorch functional calls (which are cheap compared to the
O(N³) step).  The kernel is written for Triton 3.3.1 and targets an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: cast fp32 → fp16 (used for the optional mask)
# ---------------------------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,      # fp32 source pointer
    dst_ptr,      # fp16 destination pointer
    numel,        # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    # Load fp32, cast to fp16, store
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)


# ---------------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A,B are fp16, C is fp32)
# ---------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bn, stride_Bk,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)   # batch (including hidden) dimension
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Load A block (M x K)
        a = tl.load(
            A_ptr
            + pid_z * stride_Ab
            + offs_m[:, None] * stride_Am
            + offs_k[None, :] * stride_Ak,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
            dtype=tl.float16,
        )
        # Load B block (K x N) – we read B in transposed order
        b = tl.load(
            B_ptr
            + pid_z * stride_Bb
            + offs_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
            dtype=tl.float16,
        )
        # fp16·fp16 → fp32 dot product
        acc += tl.dot(a, b)

    # Store result (float32)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(
        C_ptr
        + pid_z * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn,
        acc,
        mask=mask_c,
    )


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor,
                         BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 64) -> torch.Tensor:
    """
    Wrapper for the Triton batched GEMM kernel.
    A: (batch, M, K) fp16
    B: (batch, N, K) fp16   (will be interpreted as Bᵀ inside the kernel)
    Returns:
        C: (batch, M, N) fp32
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in element counts (not bytes)
    stride_Ab, stride_Am, stride_Ak = A.stride()
    stride_Bb, stride_Bn, stride_Bk = B.stride()
    stride_Cb, stride_Cm, stride_Cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_Ab,
        stride_Am,
        stride_Ak,
        stride_Bb,
        stride_Bn,
        stride_Bk,
        stride_Cb,
        stride_Cm,
        stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ---------------------------------------------------------------------------
# Entry point required by the evaluation harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim] (float32)
        - mask         : torch.Tensor [bs, seq_len, seq_len] (float32)
        - weights      : dict of model weights (float32)
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]                 # input feature dimension
    hidden_dim = config["hidden_dim"]   # projection dimension (H)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm  (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3) Cast linear weights once to fp16 (no bias in any projection)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # fp32

    # ------------------------------------------------------------------
    # 4) Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B,N,N,hidden] fp16
    right = F.linear(x_h, right_proj_w)  # [B,N,N,hidden] fp16

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dimension)
    #    Convert mask to fp16 with a tiny Triton kernel.
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(-1)
        mask_fp16 = torch.empty_like(mask_flat, dtype=torch.float16, device=device)
        BLOCK_SIZE = 1024
        grid = ((mask_flat.numel() + BLOCK_SIZE - 1) // BLOCK_SIZE,)
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_fp16,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        mask_fp16 = mask_fp16.view(mask.shape + (1,))
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 6) Gating (sigmoid) – still in fp16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B,N,N,hidden] fp16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B,N,N,hidden] fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core N³ operation: batched matrix multiply L @ Rᵀ
    #    Use the custom Triton kernel (fused over the hidden/channel batch)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (B*H, N, N) for the kernel
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16

    # Compute C = left_mat @ right_matᵀ  → (B*H, N, N) in fp32
    out_mat = _batched_gemm_triton(left_mat, right_mat,
                                   BLOCK_M=128, BLOCK_N=128, BLOCK_K=64)

    # reshape back to [B, N, N, hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp32

    # ------------------------------------------------------------------
    # 9) Output gating + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # element‑wise gate (fp32)
    out = F.linear(out, to_out_w)            # fp32 → final dim

    return out
