"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy N³ contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched matrix multiplication
    (B·H, N, N) @ (B·H, N, N) → (B·H, N, N)
using a custom Triton kernel that accumulates in fp32
while keeping data in fp16 for bandwidth efficiency.

Optimisations
-------------
* Fuse the projection and its gate (two linear layers) into a single matmul
  per side (`left_proj + left_gate`, `right_proj + right_gate`).
* Apply the mask, gating and projection in place (no extra temporaries).
* Use a 3‑D launch grid (tile‑row, tile‑col, batch·hidden) and
  tile sizes that adapt to the sequence length.
* All heavy ops are in fp16; the final LayerNorms and linear layers stay
  in fp32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dims
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication with fp16 inputs and fp32 accumulation.
    Each program instance processes one tile (BLOCK_M × BLOCK_N) of a
    particular batch element (identified by pid_batch = program_id(2)).
    """
    pid_batch = tl.program_id(2)   # = batch_idx * hidden_idx
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile col

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to A and B tiles
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        # Load with masks (out‑of‑bounds elements are zero)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot product (fp16 × fp16 → fp32 accumulator)
        acc += tl.dot(a, b)

    # Write‑back
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel for batched fp16 matmul.
    a : (B, M, K)  fp16, contiguous
    b : (B, K, N)  fp16, contiguous
    Returns:
        c : (B, M, N) fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Heuristics for block sizes
    if max(M, N) <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif max(M, N) <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B,
    )
    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward needed).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask: torch.Tensor [B, N, N] (bool/int) or None
            - weights: dict of model parameters (float32)
            - config: dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ Load weights (cast to half where possible)
    # ------------------------------------------------------------------
    # LayerNorm1 (float32)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections (no bias) – cast to fp16
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()

    # Output LayerNorm & linear (float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → half for heavy ops
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x_h = x.half()  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection + gate for left and right sides
    #    The fused weight has shape (2*hidden_dim, dim)
    # ------------------------------------------------------------------
    left_fused_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, dim)
    right_fused_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # Linear with fused weights
    left_and_gate = F.linear(x_h, left_fused_w)   # [B, N, N, 2H] fp16
    right_and_gate = F.linear(x_h, right_fused_w)

    # Split projection / gate
    left = left_and_gate[..., :hidden_dim]               # [B, N, N, H]
    left_gate = left_and_gate[..., hidden_dim:]          # [B, N, N, H]
    right = right_and_gate[..., :hidden_dim]
    right_gate = right_and_gate[..., hidden_dim:]

    # ------------------------------------------------------------------
    # 5️⃣ Apply optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1] fp16
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Apply gating (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    # Out gate is needed later (after second LayerNorm)
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))  # [B,N,N,H] fp16

    # Fuse gate into projections
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare tensors for the cubic contraction (batched GEMM)
    #    left : (B, H, N, N)   (i,k)
    #    right: (B, H, N, N)   (k,j)  → transpose last two dims
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k,j)

    # Release no‑longer‑needed tensors
    del left, right, left_gate, right_gate, left_and_gate, right_and_gate, x_h

    B, H, N, _ = left_perm.shape
    batch_hidden = B * H

    # Merge batch & head dimensions for the kernel
    A = left_perm.view(batch_hidden, N, N)   # (B·H, M=N, K=N)
    B_mat = right_perm.view(batch_hidden, N, N)   # (B·H, K=N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (FP16 inputs, FP32 accumulate)
    # ------------------------------------------------------------------
    C = _batched_matmul_fp16(A, B_mat)   # (B·H, N, N) fp16

    # Restore original [B, N, N, H] layout
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm (float32) over hidden dimension
    # ------------------------------------------------------------------
    out = out.float()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (cast gate to fp32 for stability)
    # ------------------------------------------------------------------
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim] float32
    return out
```
