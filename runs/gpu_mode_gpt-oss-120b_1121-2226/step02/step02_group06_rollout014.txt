@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_m, stride_c_n,
    HIDDEN_DIM: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_pid = tl.program_id(0)  # combined batch*hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute actual batch and hidden
    batch = batch_pid // HIDDEN_DIM
    hid = batch_pid - batch * HIDDEN_DIM   # faster than % perhaps

    # Offsets for inner loops
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base offsets for pointers
    a_base = a_ptr + batch * stride_a_batch + hid * stride_a_h
    b_base = b_ptr + batch * stride_b_batch + hid * stride_b_h
    c_base = c_ptr + (batch * HIDDEN_DIM + hid) * stride_c_batch  # combined index for C

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Loads A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Loads B_T: shape (BLOCK_K, BLOCK_N) => consumed as (k,j) from original B
        # Row index uses offs_k (original dim2), column index uses offs_n (original dim1)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write to C
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Triton‑accelerated outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1. Layer‑norm on the last dimension `dim` (FP32).
2. Cast to FP16 and apply the four linear projections
   (left_proj, right_proj, left_gate, right_gate, out_gate).  No bias.
3. Optional mask (B,N,N) is broadcast to the hidden dimension.
4. Gates are sigmoid‑activated and multiplied in‑place.
5. Core cubic contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication A @ Bᵀ with
   A = left, B = right.  The hidden dimension is fused into the batch
   axis, so the kernel works on (B·H) independent (N×N) matmuls.
   The kernel is tiled (128×128) with a K‑tile of 128 and uses
   Tensor‑cores (FP16 × FP16 → FP32) with an FP32 accumulator.
6. Convert the result back to FP32, apply a second LayerNorm
   over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.

Only the N³ contraction is executed in Triton; the surrounding
operations stay in PyTorch (high‑throughput cuBLAS kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # tensors
    M, N, K,                                 # matrix sizes (all = seq_len)
    stride_a_batch, stride_a_m, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_m, stride_c_n,
    HIDDEN_DIM: tl.constexpr,                # hidden dimension (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matmul C = A @ Bᵀ where A/B have shape (B, N, N, H)
    but are interpreted as (B*H, N, N) inside the kernel.
    """
    # -------------- program IDs -------------------------------------------------
    pid_batch = tl.program_id(0)          # combined (batch, hidden) index
    pid_m = tl.program_id(1)              # block‑row index
    pid_n = tl.program_id(2)              # block‑col index

    # -------------- decode batch / hidden ---------------------------------------
    batch = pid_batch // HIDDEN_DIM
    hid   = pid_batch - batch * HIDDEN_DIM          # pid_batch % HIDDEN_DIM

    # -------------- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -------------- base pointers for this (batch, hidden) ----------------------
    a_base = a_ptr + batch * stride_a_batch + hid * stride_a_h
    b_base = b_ptr + batch * stride_b_batch + hid * stride_b_h
    c_base = c_ptr + pid_batch   * stride_c_batch   # one output matrix per (b,h)

    # -------------- accumulation (FP32) -----------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------- K‑tile loop -------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)  -> left[b,i,k,h]
        a_ptrs = a_base + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Bᵀ tile: (BLOCK_K, BLOCK_N) -> right[b,j,k,h] accessed as (k,j)
        # row index = k (original dim‑2), column index = j (original dim‑1)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Tensor‑core dot (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # -------------- write back --------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Helper that runs the above Triton kernel.
    left/right: (B, N, N, H)  FP16, contiguous.
    Returns C of shape (B*H, N, N)  FP16.
    """
    B, N, _, H = left.shape
    # Output buffer
    out = torch.empty((B * H, N, N), dtype=torch.float16, device=left.device)

    # Strides of the input tensors (contiguous layout)
    stride_a_batch, stride_a_m, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    # Tile sizes – hand‑tuned for H100 Tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128

    total_batch = B * H
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (total_batch, num_m, num_n)

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_a_h,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_b_h,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        HIDDEN_DIM=H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # 8 warps = 256 threads per block (good for 128×128 tiles)
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None  [B, N, N] (bool / 0‑1)
        - weights      : dict of torch.Tensor (model parameters)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # model weights (cast to the precision we will use)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # keep FP32 for the final proj

    # --------------------------------------------------------------
    # 1️⃣ LayerNorm over the input `dim` (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_h = x.half()                                   # [B, N, N, dim] FP16

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) in FP16
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)               # [B, N, N, hidden_dim] FP16
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1] FP16
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 5️⃣ Gating tensors (sigmoid) – stay in FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply the gates
    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Batched contraction using the fused Triton kernel
    # --------------------------------------------------------------
    # The kernel works directly on the native layout (B, N, N, H),
    # no permute/copy is required.
    # Output is (B*H, N, N) FP16.
    c = _batched_matmul_triton(left, right)   # -> (B*H, N, N) FP16

    # Restore the original 4‑D layout: (B, N, N, H)
    out_fp16 = c.view(input_tensor.shape[0], hidden_dim, input_tensor.shape[1], input_tensor.shape[2]) \
                 .permute(0, 2, 3, 1).contiguous()                     # [B, N, N, H]

    # --------------------------------------------------------------
    # 7️⃣ Convert back to FP32 for the second LayerNorm
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # --------------------------------------------------------------
    # 9️⃣ Apply the output gate (broadcasted)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 🔟 Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)                     # [B, N, N, dim] FP32

    return out
