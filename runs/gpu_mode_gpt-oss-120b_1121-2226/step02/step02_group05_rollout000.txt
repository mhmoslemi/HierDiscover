"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy N³ contraction is expressed as a batched matrix multiplication:

    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

For each hidden channel `h` we multiply an (N × N) matrix with the
transpose of another (N × N) matrix.  A custom Triton kernel performs a
large 3‑D launch (tile‑M, tile‑N, batch*hidden) where each program
instance computes a BLOCK_M × BLOCK_N tile of the result using FP16
operands and FP32 accumulation.  All other operations (LayerNorm,
linear projections, gating, final projection) are kept in PyTorch but
use half‑precision where possible.

The implementation supports the “no mask” case, arbitrary batch size
(1 or 2), sequence lengths up to 1024 and hidden dimensions 128.
The output is returned in float32 as required by the model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A : (B, M, K)   – fp16
#   B : (B, K, N)   – fp16
#   C : (B, M, N)   – fp16 (stored), accumulation in fp32
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # tensor pointers
    M, N, K,                            # static matrix dimensions
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: (B, M, K) @ (B, K, N) → (B, M, N)."""
    pid_batch = tl.program_id(2)      # batch * hidden dimension
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 for higher precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile pointer (batch, row=offs_m, col=offs_k)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        # B tile pointer (batch, row=offs_k, col=offs_n)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # GEMM on the tile – result accumulated in FP32
        acc += tl.dot(a, b)

    # Store the tile back to C (cast back to FP16)
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel.

    Args:
        a: (B, M, K) FP16, contiguous
        b: (B, K, N) FP16, contiguous

    Returns:
        c: (B, M, N) FP16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Heuristic block sizes – tuned for H100 (large shared memory)
    if max(M, N) <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_m, num_n, B)  # (M tiles, N tiles, batch*hidden)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/int) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor[B, N, N, dim]  (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 2️⃣ load weights – cast linear weights to fp16 for speed
    # --------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Projections (bias=False)
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    # Gates (bias=False)
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    # Output LayerNorm & final projection
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → half precision for heavy ops
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] float32
    x_h = x.half()  # down‑cast once

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(dtype=left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – stay in FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Prepare tensors for the N³ contraction
    #   left  : (B, N, N, H) → (B, H, N, N)   (i, k)
    #   right : (B, N, N, H) → (B, H, N, N)   (k, j) via transpose
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    # Release temporary tensors early to reduce peak memory
    del left, right, left_gate, right_gate, x_h

    B, H, N, _ = left_perm.shape

    # Merge batch & hidden dimensions → (B*H, N, N)
    A = left_perm.view(B * H, N, N)      # (batch*hidden, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N) # (batch*hidden, K=N, N)

    # --------------------------------------------------------------
    # 8️⃣ Batched GEMM via Triton (FP16 -> FP32 accumulation)
    # --------------------------------------------------------------
    C = _batched_matmul_fp16(A, B_mat)  # (B*H, N, N) fp16

    # Restore original layout: [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm (over hidden dimension) + output gate
    # --------------------------------------------------------------
    out = out.float()  # convert for layer‑norm stability
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate.float()  # broadcast gate (B,N,N,H)

    # --------------------------------------------------------------
    # 🔟 Final projection back to original `dim` (float32)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)  # [B, N, N, dim] float32

    return out
