"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The forward pass consists of four stages:

1️⃣ **Input LayerNorm** – performed in FP32 (torch.nn.functional.layer_norm)
   and cast to FP16 for the heavy parts.

2️⃣ **Fused projection & gating** – the five linear layers
   (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`)
   are concatenated into a single weight matrix of shape
   `(5·hidden_dim, dim)`.  A single FP16 GEMM produces all five tensors.
   The three gating tensors are passed through a sigmoid, and an optional
   mask is applied before element‑wise gating.

3️⃣ **Cubic update** – for each hidden slice we need
   `out[i,j] = Σ_k left[i,k] * right[j,k]`,
   i.e. a batched matrix‑multiply `A @ Bᵀ` where
   `A = left` and `B = right`.  A custom Triton kernel computes this
   with tensor‑core‑accelerated `tl.dot`.  The kernel works in FP16
   with FP32 accumulation and writes the result back in FP16.

4️⃣ **Output LayerNorm, output‑gate & final linear projection** – the
   intermediate result (`[B, N, N, hidden_dim]`) is normalized in FP32,
   multiplied by the output gate, projected back to `dim` with a final
   FP16 GEMM and cast to FP32 for the final output.

The kernel tile size `128×128×128` (M, N, K) together with
`num_warps=8, num_stages=4` gives good occupancy on an H100.
All heavy arithmetic is performed in FP16 (except accumulators) to
minimize memory traffic while retaining the numerical accuracy required
by AlphaFold‑3.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B : fp16 → C : fp16, acc in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,                 # left  : [BH, M, K]  (row‑major)
    b_ptr,                 # right : [BH, N, K]  (row‑major, we treat as Bᵀ)
    c_ptr,                 # out   : [BH, M, N]
    M, N, K,               # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides (batch, i, k)
    stride_bh, stride_bn, stride_bk,   # Bᵀ strides (batch, j, k) → note order of j/k
    stride_ch, stride_ci, stride_cj,   # C strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for each hidden slice.
    A  : (BH, M, K)
    Bᵀ : (BH, K, N)   – we pass B as (BH, N, K) and index with swapped strides.
    """
    pid_h = tl.program_id(axis=2)      # batch * hidden slice
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- Load A tile (M × K) ------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)

        # ---- Load Bᵀ tile (K × N) ----------------------------------
        # B is stored as (BH, N, K); we treat it as transposed by swapping
        # the stride of the N and K dimensions.
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk          # K dimension (contiguous)
            + offs_n[None, :] * stride_bn         # N dimension
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)

        # ---- Inner product -----------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ------------------------------------------------------------------
    # Write result back (store as FP16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/int) or None
        - weights_dict : mapping from weight name to torch.Tensor
        - config_dict  : contains ``dim`` and ``hidden_dim`` (int) and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                          # [B, N, N, dim] fp32
    x_h = x.half()                             # fp16 for heavy mats

    # --------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 × hidden_dim outputs)
    # --------------------------------------------------------------
    # concatenate five weight matrices (shape: (5*H, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # fp16

    # single GEMM → shape [B, N, N, 5*H] (fp16)
    fused = F.linear(x_h, fused_weight)        # fp16

    B, N, _, _ = fused.shape

    # ------------------------------------------------------------------
    # split fused result into the three projections and three gates
    # ------------------------------------------------------------------
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B, N, N, 5, H]
    left_proj  = fused[..., 0, :]                # [B, N, N, H]
    right_proj = fused[..., 1, :]                # [B, N, N, H]
    left_gate  = torch.sigmoid(fused[..., 2, :])   # [B, N, N, H] fp16
    right_gate = torch.sigmoid(fused[..., 3, :])   # [B, N, N, H] fp16
    out_gate   = torch.sigmoid(fused[..., 4, :])   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → [B, N, N, 1] → fp16
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 4️⃣ Apply gating (still fp16)
    # --------------------------------------------------------------
    left  = left_proj * left_gate   # [B, N, N, H]
    right = right_proj * right_gate # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Rearrange for batched GEMM (BH = B·H)
    # --------------------------------------------------------------
    #   left  : [B, N, N, H] → [B, H, N, N] → [BH, N, N]
    #   right : same layout
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N]  fp16
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]  fp16

    # --------------------------------------------------------------
    # 6️⃣ Triton kernel: batched matmul A @ Bᵀ (fp16 → fp16, acc fp32)
    # --------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 128   # larger K‑tile halves the loop count

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one launch per (batch, hidden) slice
    )

    tri_mul_outgoing_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        # A strides (BH, i, k)
        stride_ah=left_flat.stride(0),
        stride_ai=left_flat.stride(1),
        stride_ak=left_flat.stride(2),
        # Bᵀ strides (BH, k, j)   note the swap of N/K strides
        stride_bh=right_flat.stride(0),
        stride_bk=right_flat.stride(2),   # K dimension (contiguous)
        stride_bn=right_flat.stride(1),   # N dimension
        # C strides (BH, i, j)
        stride_ch=out_flat.stride(0),
        stride_ci=out_flat.stride(1),
        stride_cj=out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still fp16)
    # --------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)   # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output gate
    # --------------------------------------------------------------
    out_fp32 = out_half.float()   # cast to fp32 for layer‑norm
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H] fp32

    # Apply the output gate (originally fp16 → cast to fp32)
    out_fp32 = out_fp32 * out_gate.float()

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    # Linear in FP16, then cast back to FP32
    out_proj_half = F.linear(out_fp32.half(), to_out_weight_h)   # fp16
    out_final = out_proj_half.float()                           # fp32

    return out_final
