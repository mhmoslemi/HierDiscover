"""
TriMul (outgoing) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Layer‑norm over the last dimension (dim) of the input (FP32).
2️⃣  Cast to FP16 and compute left/right projections + gating (bias‑free linear + sigmoid).
3️⃣  Apply optional mask and fuse it with the gating in‑place (FP16).
4️⃣  Compute the cubic contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    as a batched matrix multiplication per (batch, hidden) pair.
    • For small sequence lengths (N ≤ 256) a tiny Triton kernel is used.
    • For larger N we fall back to cuBLAS via torch.bmm, which is heavily tuned on H100.
5️⃣  Second Layer‑norm over the hidden dimension, apply the output‑gate,
    and a final FP32 linear projection back to `dim`.

Only the heavy O(N³) part is split between Triton (small N) and cuBLAS (large N);
all other operations stay in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dims
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # batch * hidden
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (more accurate)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # store result
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16_tiny(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM for small N using a custom Triton kernel.
    a : (B, M, K)   FP16, contiguous
    b : (B, K, N)   FP16, contiguous
    Returns c : (B, M, N) FP16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    # Choose block sizes – tuned for small matrices
    if max(M, N) <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m, num_n, B)

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data : tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 2️⃣ Load weights – cast to FP16 where beneficial
    # --------------------------------------------------------------
    # LayerNorm1
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Linear projections (bias‑free)
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    # Gating matrices (bias‑free)
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # Output LayerNorm & final projection (remain FP32 for stability)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → FP16 for the heavy part
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                 # [B, N, N, dim]  FP32
    x_h = x.half()                    # [B, N, N, dim]  FP16

    # --------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden] FP16
    right = F.linear(x_h, right_proj_w)  # same shape

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim) – in‑place
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1] FP16
        left  = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – all FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Prepare tensors for the cubic contraction
    #    left : (B, N, N, H) → (B, H, N, N)   (i,k)
    #    right: (B, N, N, H) → (B, H, N, N)   (k,j) via axis swap
    # --------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()  # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    del left, right, left_gate, right_gate, x_h  # free memory

    B, H, N, _ = left_perm.shape

    # Flatten batch & hidden so we can run a batched GEMM
    A = left_perm.view(B * H, N, N)   # (B*H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)  # (B*H, K=N, N)

    # --------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication
    #    Small N → tiny Triton kernel, otherwise cuBLAS via torch.bmm
    # --------------------------------------------------------------
    if N <= 256:
        # Triton kernel for the small‑matrix regime
        C = _batched_matmul_fp16_tiny(A, B_mat)          # (B*H, N, N) FP16
    else:
        # cuBLAS – highly optimized on H100
        C = torch.bmm(A, B_mat)                          # (B*H, N, N) FP16

    # Restore the original layout: (B, H, N, N) → (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] FP16

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (FP32)
    # --------------------------------------------------------------
    out = out.float()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 🔟 Apply output gate (cast to FP32 for stability)
    # --------------------------------------------------------------
    out = out * out_gate.float()

    # --------------------------------------------------------------
    # 11️⃣ Final linear projection back to `dim` (float32)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]  FP32

    return out
```
