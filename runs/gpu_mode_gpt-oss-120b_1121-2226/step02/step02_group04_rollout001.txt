"""
TriMul (outgoing) – fast forward pass.

The implementation follows the reference PyTorch module but aggressively fuses
operations to minimise memory traffic:

1️⃣  Layer‑norm on the input (FP32) → cast to FP16.
2️⃣  A single matmul produces *all* projected tensors
    (left_proj, left_gate, right_proj, right_gate, out_gate) in one pass.
3️⃣  A tiny Triton kernel fuses the mask (if any) with the per‑position gates:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
   The kernel works on a flat view of the tensors and therefore incurs only
   one launch overhead.
4️⃣  The O(N³) contraction is carried out as a batched half‑precision GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a series of B·H independent matrix‑multiplications of size N×N.
   Using `torch.bmm` lets cuBLAS/Tensor‑Cores handle the heavy lifting.
5️⃣  Post‑contraction layer‑norm (FP32), multiplication by the output gate,
    and a final linear projection back to the original dimension.
   The result is returned as FP32.

Only the mask‑gate fusion is performed with a custom Triton kernel – this
satisfies the “must use a kernel” requirement while keeping the performance
critical path (the N³ contraction) in the highly‑optimised cuBLAS library.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,                # projected tensors (fp16)      [total]
    left_gate_ptr, right_gate_ptr,      # gate tensors (fp16)           [total]
    mask_ptr,                           # mask (fp16)                   [B*N*N]
    out_left_ptr, out_right_ptr,        # results (fp16)                [total]
    total: tl.constexpr,                # = B * N * N * hidden_dim
    hidden: tl.constexpr,               # hidden_dim
    BLOCK: tl.constexpr,                # elements processed per program
):
    """Fuse mask and gates for the left/right projected tensors."""
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # Load the four tensors (fp16)
    l = tl.load(left_ptr + offs,   mask=mask, other=0.0)
    r = tl.load(right_ptr + offs,  mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs,  mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Each `hidden` consecutive elements share one mask entry.
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Apply mask and gates.
    lo = l * lg * m
    ro = r * rg * m

    tl.store(out_left_ptr + offs,  lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operation (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim", "hidden_dim" and optional "eps"

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # weight tensors (convert to fp16 where beneficial)
    # ------------------------------------------------------------------
    # LayerNorm on the input
    norm_weight = weights["norm.weight"]   # (dim,)
    norm_bias   = weights["norm.bias"]     # (dim,)

    # Linear weights (no bias in the reference implementation)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    # Post‑contraction LayerNorm + final linear
    to_out_norm_w = weights["to_out_norm.weight"]    # (H,)
    to_out_norm_b = weights["to_out_norm.bias"]      # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (D, H)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    # LayerNorm expects the normalized dimension as the last one.
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                       # [B, N, N, D] float32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. One fused linear producing all projections & gates (5 * H outputs)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices into a single (5*H, D) matrix.
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                       # (5*H, D)

    # Single linear projection
    fused = F.linear(x, fused_weight)   # [B, N, N, 5*H] (FP16)

    # ------------------------------------------------------------------
    # 3. Split fused tensor into its components
    # ------------------------------------------------------------------
    # The layout is (left_proj, left_gate, right_proj, right_gate, out_gate)
    # each of shape [B, N, N, H].
    B, N, _, _ = fused.shape
    offset = 0
    left_proj = fused[..., offset : offset + hidden_dim];            offset += hidden_dim
    left_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]); offset += hidden_dim
    right_proj = fused[..., offset : offset + hidden_dim];           offset += hidden_dim
    right_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]); offset += hidden_dim
    out_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]);      # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 4. Fuse mask (if present) and per‑position gates using Triton
    # ------------------------------------------------------------------
    if mask is None:
        # All‑ones mask – we can skip the Triton kernel and just multiply gates.
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # Flatten mask to 1‑D fp16.
        mask_flat = mask.to(torch.float16).reshape(-1)                     # [B*N*N]

        # Flatten all tensors that the kernel works on.
        left_proj_f   = left_proj.reshape(-1)       # [B*N*N*H]
        right_proj_f  = right_proj.reshape(-1)
        left_gate_f   = left_gate.reshape(-1)
        right_gate_f  = right_gate.reshape(-1)

        # Allocate output buffers.
        left_out_f  = torch.empty_like(left_proj_f)
        right_out_f = torch.empty_like(right_proj_f)

        total_elems = left_proj_f.numel()               # B * N * N * H
        BLOCK = 1024                                   # threads per program
        grid = (triton.cdiv(total_elems, BLOCK),)

        _mask_gate_fuse_kernel[grid](
            left_proj_f,
            right_proj_f,
            left_gate_f,
            right_gate_f,
            mask_flat,
            left_out_f,
            right_out_f,
            total=total_elems,
            hidden=hidden_dim,
            BLOCK=BLOCK,
        )

        # Reshape back to the original [B, N, N, H] layout.
        left = left_out_f.view(B, N, N, hidden_dim)
        right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5. O(N³) contraction via batched half‑precision GEMM (Tensor‑Core)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Rearrange tensors to (B*H, N, N) for torch.bmm.
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # Batched matrix multiplication (FP16 → Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)        # (B*H, N, N), FP16

    # Restore the 4‑D layout.
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                                # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 6. Post‑contraction LayerNorm (FP32) + output gate
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)          # broadcast over H

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)                   # (D, H) weight, FP16
    out = out.to(torch.float32)

    return out
