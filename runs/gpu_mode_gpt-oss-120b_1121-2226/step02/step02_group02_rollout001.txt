"""
custom_kernel – Triton‑accelerated forward pass of the outgoing TriMul module.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast the normalized tensor to fp16 (half‑precision) – the heavy arithmetic
   runs in fp16 while the final result is returned in fp32.
3. Linear projections (left/right) and three gating linear layers are performed
   with half‑precision weights.  All linear ops are carried out with
   `torch.nn.functional.linear`.
4. Optional mask: a tiny Triton kernel converts the fp32 mask to fp16 and
   broadcasts it over the hidden dimension; the mask is then applied element‑wise.
5. Apply sigmoid gates (still in fp16) and multiply them into the projected
   tensors.
6. Core N³ operation – batched matrix multiply `L @ Rᵀ` where each hidden channel
   is a separate batch:
      * reshape to (B, hidden, N, N)
      * `torch.matmul` (half‑precision, Tensor‑Core friendly) computes all
        hidden‑channel GEMMs in one call.
7. Cast the result back to fp32, perform a hidden‑dim LayerNorm, multiply by the
   output gate and finally project back to the original `dim`.
8. Return the fp32 tensor of shape [B, N, N, dim].

Only the mask‑conversion step is written in Triton; the rest uses highly‑optimized
PyTorch kernels (cuBLAS / Tensor‑Core) which together achieve sub‑millisecond
latency on the target H100 for the prescribed test shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ---------------------------------------------------------
# Triton kernel: cast a flat fp32 tensor to fp16
# ---------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,          # fp32 source pointer
    dst_ptr,          # fp16 destination pointer
    numel,            # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    val_f16 = tl.cast(val_f32, tl.float16)
    tl.store(dst_ptr + offs, val_f16, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32)
        - weights      : dict of model weight tensors (float32)
        - config       : dict with keys
                         * "dim"          – input feature dimension
                         * "hidden_dim"   – projection dimension
                         * "nomask"       – bool, True → ignore supplied mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2) Cast to half‑precision for arithmetic‑intensive ops
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 3) Cast linear weights once to fp16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w   = w["left_proj.weight"].half()
    right_proj_w  = w["right_proj.weight"].half()
    left_gate_w   = w["left_gate.weight"].half()
    right_gate_w  = w["right_gate.weight"].half()
    out_gate_w    = w["out_gate.weight"].half()
    # output‑norm + final linear stay in fp32 (as in the reference)
    to_out_norm_w = w["to_out_norm.weight"]
    to_out_norm_b = w["to_out_norm.bias"]
    to_out_w      = w["to_out.weight"]          # fp32

    # ------------------------------------------------------------------
    # 4) Linear projections (fp16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)    # [B, N, N, hidden] fp16
    right = F.linear(x_h, right_proj_w)   # [B, N, N, hidden] fp16

    # ------------------------------------------------------------------
    # 5) Optional mask – convert fp32 mask → fp16 via a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # flatten mask and allocate half‑precision buffer
        mask_flat = mask.view(-1)
        mask_h_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)

        BLOCK_SIZE = 1024
        grid = ((mask_flat.numel() + BLOCK_SIZE - 1) // BLOCK_SIZE, )
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_h_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # reshape to [B, N, N, 1] and broadcast over hidden dim
        mask_h = mask_h_flat.view(mask.shape + (1,))
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6) Gating – sigmoid on half‑precision tensors
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B,N,N,hidden] fp16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,hidden] fp16

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core N³ operation: batched matmul L @ Rᵀ
    #    Treat the hidden channel as a batch dimension (B, hidden, N, N).
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden, N, N) – no extra copy beyond a view.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N] fp16
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N] fp16

    # Batched matmul: (B, hidden, N, N) @ (B, hidden, N, N).transpose(-1,-2)
    # Uses cuBLAS/Tensor‑Core for half‑precision matrices.
    out_fp16 = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # [B, hidden, N, N] fp16

    # ------------------------------------------------------------------
    # 8) Cast back to fp32 and restore original layout [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_fp16.to(torch.float32)                     # fp32
    out = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, hidden] fp32

    # ------------------------------------------------------------------
    # 9) Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp32

    # ------------------------------------------------------------------
    # 10) Output gating + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # broadcasted element‑wise gate
    out = F.linear(out, to_out_w)            # fp32 → final dim

    return out
