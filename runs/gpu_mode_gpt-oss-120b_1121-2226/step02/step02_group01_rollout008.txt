"""
TriMul (outgoing) – Triton‑accelerated forward pass

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (float32).
2. Cast to FP16, optionally mask the tensor **before** the heavy projections.
3. Fuse each pair (proj, gate) into a single linear layer:
      left  = x @ [W_left_proj;  W_left_gate]ᵀ
      right = x @ [W_right_proj; W_right_gate]ᵀ
   Split the result, apply sigmoid to the gate halves and multiply.
4. The cubic update
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiply L @ Rᵀ where each hidden channel h (and each batch
   element) is an independent batch element.  We view the tensors as
   (B·H, N, N) with *strided* memory (no copy) and run a tiled Triton kernel
   that uses fp16 inputs and fp32 accumulation (Tensor‑Core friendly).
5. Reshape the result back to (B,N,N,H), apply a second LayerNorm.
6. Multiply by the output gate and finish with a linear projection back to the
   original feature dimension (float32).

The implementation keeps the heavy N³ work on the GPU while reducing the
overhead of data reshaping and extra element‑wise passes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * B (N×K) → C (M×N)
#   (note: B is NOT transposed; we compute Σ_k A[i,k] * B[j,k])
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A: batch, M, K strides
    stride_bh, stride_bn, stride_bk,   # B: batch, N, K strides
    stride_ch, stride_cm, stride_cn,   # C: batch, M, N strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile over M
    pid_n = tl.program_id(1)          # tile over N
    pid_b = tl.program_id(2)          # batch*hidden index

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current tile
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Masks for out‑of‑bounds loads
        k_base = k * BLOCK_K
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        acc += tl.dot(a, b)   # fp16·fp16 → fp32 accumulate

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matmul for the TriMul cubic update.
    A, B : (batch, M, K) and (batch, N, K) – fp16 tensors.
    Returns C : (batch, M, N) – fp32 tensor.
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides in element units (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Helper: view (B,N,N,H) as (B*H,N,N) **without** copying memory.
# --------------------------------------------------------------
def _as_batch_hidden(t: torch.Tensor) -> torch.Tensor:
    """
    Convert a tensor of shape (B,N,N,H) to a view of shape (B*H,N,N)
    using strided indexing.  No data copy is performed.
    """
    B, N, _, H = t.shape
    # Original strides: (N*N*H, N*H, H, 1)
    s0, s1, s2, s3 = t.stride()
    # New strides: (1, N*H, H)  – batch dimension walks over the hidden axis.
    new_strides = (s3, s1, s2)
    new_shape = (B * H, N, N)
    return t.as_strided(new_shape, new_strides)


# --------------------------------------------------------------
# Entry point required by the harness
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – may be None
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all heavy ops
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 3) Optional mask – fuse it into the projection input
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(x_h.dtype).unsqueeze(-1)  # [B, N, N, 1]
        x_h = x_h * m                         # masked input for proj/gate

    # ------------------------------------------------------------------
    # 4) Fuse (proj + gate) for left and right sides
    # ------------------------------------------------------------------
    # combine left projection and its gate
    left_w_proj = weights["left_proj.weight"]
    left_w_gate = weights["left_gate.weight"]
    left_w_comb = torch.cat([left_w_proj, left_w_gate], dim=0).half()  # (2H, dim)

    # combine right projection and its gate
    right_w_proj = weights["right_proj.weight"]
    right_w_gate = weights["right_gate.weight"]
    right_w_comb = torch.cat([right_w_proj, right_w_gate], dim=0).half()  # (2H, dim)

    # linear pass (project + gate) – we reuse the same input x_h
    left_comb = F.linear(x_h, left_w_comb)   # [B,N,N,2*H] fp16
    right_comb = F.linear(x_h, right_w_comb)  # [B,N,N,2*H] fp16

    # split into proj and gate halves
    left_proj = left_comb[..., :hidden_dim]                # [B,N,N,H]
    left_gate = left_comb[..., hidden_dim:].sigmoid()      # [B,N,N,H]

    right_proj = right_comb[..., :hidden_dim]              # [B,N,N,H]
    right_gate = right_comb[..., hidden_dim:].sigmoid()    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5) Output gate (separate linear) – still fp16
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].half()
    out_gate = F.linear(x_h, out_gate_w).sigmoid()  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates (still fp16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 7) Core cubic update – batched matmul via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # View tensors as (B*H, N, N) without copying.
    left_mat = _as_batch_hidden(left)   # fp16, shape (B*H, N, N)
    right_mat = _as_batch_hidden(right) # fp16, shape (B*H, N, N)

    # Choose tile sizes – larger tiles for larger sequences.
    if N <= 384:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # [B*H, N, N]  (fp32)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 8) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 9) Output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.float()            # broadcast * (fp32)
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim] (float32)

    return out
