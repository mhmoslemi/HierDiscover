"""
TriMul – “outgoing” variant (AlphaFold‑3)

Algorithm
---------
1️⃣  Layer‑norm the input tensor (float32).
2️⃣  Cast to fp16 and fuse the two projection linear layers (proj+gate) into a
    single matmul per side:   [B,N,N,dim] → [B,N,N,2*hidden].
    The result is split into (proj, pre‑gate) and the gate is passed through
    sigmoid.
3️⃣  Apply the optional input mask (broadcasted on the hidden dimension) and
    combine projection + gate → tensors *left* and *right* of shape
    [B,N,N,hidden] (fp16).
4️⃣  Reshape to [B·hidden, N, N] and run a custom Triton GEMM that computes
    a batched matrix product  left @ rightᵀ  for every (batch, hidden) pair.
    The kernel uses an autotuned tile size and accumulates in FP32.
5️⃣  Reshape back to [B,N,N,hidden] (float32) and apply a second Layer‑Norm.
6️⃣  Multiply by the output gate, then a final linear projection back to the
    original dimension.

All heavy work (projections, GEMM) is performed in fp16; the accumulated
tri‑multiplicative result and the final output are kept in fp32 for numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul  C = A @ Bᵀ
#   A, B : fp16  (shape: [BH, N, N])
#   C    : fp32  (shape: [BH, N, N])
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config(
            {"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 64},
            num_warps=4,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64},
            num_warps=8,
        ),
        triton.Config(
            {"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 128},
            num_warps=8,
        ),
        triton.Config(
            {"BLOCK_M": 256, "BLOCK_N": 128, "BLOCK_K": 64},
            num_warps=8,
        ),
        triton.Config(
            {"BLOCK_M": 256, "BLOCK_N": 256, "BLOCK_K": 64},
            num_warps=16,
        ),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A   (batch, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ  (batch, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C   (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)          # batch * hidden index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                       # fp16
        a = a.to(tl.float32)    # promote for the dot

        # ----- Load Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk   # k‑dimension (row of Bᵀ)
            + offs_n[None, :] * stride_bn   # n‑dimension (col of Bᵀ)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                       # fp16
        b = b.to(tl.float32)

        # ----- Accumulate -----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write out C tile
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point – forward pass of the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------
    # 2️⃣  Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.half()

    # ------------------------------------------------------------------
    # 3️⃣  Fuse projection + gate linear layers (fp16, no bias)
    #     left:  [B,N,N,2*hidden]  → split → proj & gate
    #     right: same
    # ------------------------------------------------------------------
    left_comb_weight = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # (2*hidden, dim)
    right_comb_weight = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # (2*hidden, dim)

    left_comb = F.linear(x_h, left_comb_weight)   # [B, N, N, 2*hidden]
    right_comb = F.linear(x_h, right_comb_weight)  # [B, N, N, 2*hidden]

    left_proj, left_pre_gate = left_comb.split([hidden_dim, hidden_dim], dim=-1)
    right_proj, right_pre_gate = right_comb.split([hidden_dim, hidden_dim], dim=-1)

    # ------------------------------------------------------------------
    # 4️⃣  Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(left_proj.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 5️⃣  Gate sigmoids (still fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_pre_gate)
    right_gate = torch.sigmoid(right_pre_gate)

    # Apply gates
    left = left_proj * left_gate        # [B, N, N, hidden]  fp16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Compute the output gate (fp16)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].half())
    )  # [B, N, N, hidden] – fp16

    # ------------------------------------------------------------------
    # 7️⃣  Reshape to [B·hidden, N, N] for the batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, N, N, H) → (B, H, N, N) – make the hidden dim the second axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]  fp16
    right_flat = right_perm.view(BH, N, N) # [BH, N, N]  fp16

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 8️⃣  Launch the Triton GEMM kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # (batch * hidden) dimension
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0),   # stride_ah
        left_flat.stride(1),   # stride_ai
        left_flat.stride(2),   # stride_ak
        right_flat.stride(0),  # stride_bh
        right_flat.stride(2),  # stride_bk  (col ↔ K)
        right_flat.stride(1),  # stride_bn  (row ↔ N)
        out_flat.stride(0),    # stride_ch
        out_flat.stride(1),    # stride_ci
        out_flat.stride(2),    # stride_cj
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9️⃣  Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out = out.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 🔟  Second LayerNorm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 1️⃣1️⃣  Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)                # broadcast gate
    out = F.linear(out, weights["to_out.weight"])     # → [B, N, N, dim] (float32)

    return out
