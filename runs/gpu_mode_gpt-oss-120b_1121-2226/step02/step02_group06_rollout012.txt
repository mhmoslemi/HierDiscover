"""
Optimized Triton implementation of the *outgoing* TriMul operator.

Algorithm
---------
1. Layer‑norm over the input channel dimension `dim` (FP32, torch).
2. Cast to FP16 and apply five bias‑less linear projections:
   - left_proj, right_proj   → hidden feature maps
   - left_gate, right_gate, out_gate → gating maps (sigmoid)
   An optional mask (B,N,N) is broadcast and applied to `left`/`right`
   before gating.  All linear work stays in FP16.
3. Fuse the N³ contraction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM with the hidden dimension treated as an
   extra batch axis.  The kernel works directly on the original
   (B,N,N,H) layout, i.e. **no permutes or copies** are performed.
   The kernel tiles the multiplication with BLOCK_M/N/K = 64 and
   accumulates in FP32 (Tensor‑core friendly).
4. Convert the contracted result back to FP32.
5. Layer‑norm over the hidden dimension `hidden_dim`.
6. Multiply by the output gate (`out_gate`).
7. Final linear projection back to the original `dim` (FP32).

Only the contraction (step 3) is written in Triton; everything else
uses highly‑optimized PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with an extra hidden‑dim batch axis.
#   A: (B, H, M, K)    FP16   (reads as rows = i, cols = k)
#   B: (B, H, K, N)    FP16   (reads as rows = k, cols = j)
#   C: (B, H, M, N)    FP16   (accumulated in FP32)
# Strides are passed explicitly so no tensor copies are needed.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem size (all = seq_len)
    stride_ab, stride_ai, stride_ak, stride_ah,   # A strides: batch, i, k, hidden
    stride_bb, stride_bk, stride_bj, stride_bh,   # B strides: batch, k, j, hidden
    stride_cb, stride_ci, stride_cj, stride_ch,   # C strides: batch, i, j, hidden
    hidden_dim: tl.constexpr,          # hidden dimension (runtime constant)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    #   Tile decomposition: (combined_batch, block_i, block_j)
    # --------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_m * num_n

    # combined batch = outer batch * hidden_dim
    combined_batch = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    block_i = tile_idx // num_n
    block_j = tile_idx % num_n

    # Decompose combined batch -> (batch, hidden)
    batch_id = combined_batch // hidden_dim
    hidden_id = combined_batch % hidden_dim

    # Base pointers for this (batch, hidden) pair
    a_base = a_ptr + batch_id * stride_ab + hidden_id * stride_ah
    b_base = b_ptr + batch_id * stride_bb + hidden_id * stride_bh
    c_base = c_ptr + batch_id * stride_cb + hidden_id * stride_ch

    # --------------------------------------------------------------
    #   Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    #   Loop over K dimension in chunks of BLOCK_K
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile : rows = i, cols = k
        a_ptrs = (
            a_base
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # B tile : rows = k, cols = j
        b_ptrs = (
            b_base
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bj
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)        # FP16×FP16 → FP32 accumulation

    # --------------------------------------------------------------
    #   Write the result back (FP16)
    # --------------------------------------------------------------
    c_ptrs = (
        c_base
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        - mask         : torch.Tensor or None    shape [B, N, N] (float32/bool)
        - weights      : dict of torch.Tensor, keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config       : dict with keys "dim" and "hidden_dim"

    -------
    torch.Tensor
        Output tensor shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------
    # 2️⃣ model parameters (prepare precision)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]      # keep FP32

    # 3️⃣ LayerNorm over input channel dimension (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 4️⃣ Cast to FP16 for heavy linear work
    # --------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim]  (FP16)

    # --------------------------------------------------------------
    # 5️⃣ Linear projections (bias‑less) – FP16
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)      # [B, N, N, hidden_dim]

    # --------------------------------------------------------------

        right = right * mask_h

    # 7️⃣ Gating tensors (sigmoid) – FP16
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    # inplace multiplication to save memory
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 8️⃣ Batched contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    #    Implemented as a custom Triton GEMM that works on the
    #    native layout (B,N,N,hidden) – no permutes / copies.
    # --------------------------------------------------------------
    B, N, _, H = left.shape                # B, seq_len, seq_len, hidden_dim
    assert H == hidden_dim

    # output tensor (FP16) – same layout as left/right
    out_fp16 = torch.empty_like(left)

    # strides (in elements) for the three tensors
    a_stride_ab, a_stride_ai, a_stride_ak, a_stride_ah = left.stride()
    b_stride_bb, b_stride_ai, b_stride_ak, b_stride_ah = right.stride()
    # for the right matrix we need row‑stride = k, col‑stride = j
    #   row (k) -> original dim 2, column (j) -> original dim 1
    b_stride_bk = right.stride(2)   # k
    b_stride_bj = right.stride(1)   # j
    b_stride_bh = right.stride(3)   # hidden

    c_stride_cb, c_stride_ci, c_stride_cj, c_stride_ch = out_fp16.stride()

    # block sizes tuned for H100 Tensor‑cores (fp16×fp16 → fp32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = B * hidden_dim
    grid = (total_batches * num_m * num_n,)

    _tri_mul_outgoing_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out_fp16,
        M=N,
        N=N,
        K=N,
        stride_ab=a_stride_ab,
        stride_ai=a_stride_ai,
        stride_ak=a_stride_ak,
        stride_ah=a_stride_ah,
        stride_bb=b_stride_bb,
        stride_bk=b_stride_bk,
        stride_bj=b_stride_bj,
        stride_bh=b_stride_bh,
        stride_cb=c_stride_cb,
        stride_ci=c_stride_ci,
        stride_cj=c_stride_cj,
        stride_ch=c_stride_ch,
        hidden_dim=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                # 8 warps = 256 threads per block
    )

    # --------------------------------------------------------------
    # 9️⃣ Convert back to FP32 for the second LayerNorm
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32)       # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 1️⃣0️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 1️⃣1️⃣ Apply the output gate (broadcasted, FP32)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 1️⃣2️⃣ Final linear projection back to `dim` (FP32)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)      # [B, N, N, dim]  (float32)

    return out
