"""
custom_kernel – outgoing TriMul (AlphaFold‑3 style)

Algorithm
---------
1. **Layer‑Norm** on the 4‑D input (float32) → cast to fp16.
2. **Fused linear projection**:
   * Combine the five independent linear layers (left_proj, left_gate,
     right_proj, right_gate, out_gate) into a single matmul with weight
     shape (5·H, D).  The result is split back into the five tensors and
   * apply `sigmoid` to the three gate tensors.
3. **Mask & gate fusion** (tiny Triton kernel):
   * `left_out = left_proj * left_gate * mask`
   * `right_out = right_proj * right_gate * mask`
   The kernel works on flattened tensors; each mask entry is shared by
   `hidden_dim` consecutive values.
4. **N³ contraction**:
   * Reshape to `(B·H, N, N)` and perform a batched half‑precision GEMM
     (`torch.bmm`) which maps to cuBLAS Tensor‑Core kernels:
       `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
5. **Post‑contraction LayerNorm** (float32) → multiply by `out_gate`.
6. **Final projection** back to the original dimension `D` (fp16 linear
   → cast to fp32).

All heavy linear algebra uses half‑precision for speed, while the two
LayerNorms and the final output are kept in fp32 for numerical stability.
The Triton kernel supplies the required custom‑kernel component.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny Triton kernel: fuse mask, left/right gates and the mask broadcast.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,               # [total] fp16
    left_gate_ptr, right_gate_ptr,     # [total] fp16
    mask_ptr,                          # [B·N·N] fp16
    out_left_ptr, out_right_ptr,       # [total] fp16
    total: tl.constexpr,               # B·N·N·H
    hidden: tl.constexpr,              # hidden_dim = H
    BLOCK: tl.constexpr,               # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # Load values (masked loads return 0.0 for out‑of‑range threads)
    l_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # Each `hidden` consecutive entries share one mask element.
    mask_idx = offs // hidden
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # Fuse mask and gates
    lo = l_val * l_gate * m_val
    ro = r_val * r_gate * m_val

    # Store results
    tl.store(out_left_ptr + offs, lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, D] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32/float16) or
                         ``None`` for the “nomask” case.
        - weights      : dict mapping parameter names to tensors.
        - config       : dict with keys ``dim``, ``hidden_dim`` and optional
                         ``eps`` (LayerNorm epsilon).

    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0. Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1. Load model parameters (convert to fp16 where beneficial)
    # ------------------------------------------------------------------
    # LayerNorm before projection
    norm_weight = weights["norm.weight"]          # (dim,)
    norm_bias   = weights["norm.bias"]            # (dim,)

    # Linear weights (bias=False in the reference implementation)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    # Post‑contraction LayerNorm + final projection
    to_out_norm_w = weights["to_out_norm.weight"]   # (H,)
    to_out_norm_b = weights["to_out_norm.bias"]     # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)  # (D, H)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (float32) → fp16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, D]  float32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Fuse the five linear projections (5·H → D) into one matmul
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )  # (5*H, D)

    fused = F.linear(x, fused_weight)  # [B, N, N, 5*H]  fp16

    # Split the fused tensor
    B, N, _, _ = fused.shape
    offset = 0
    left_proj = fused[..., offset : offset + hidden_dim]; offset += hidden_dim
    left_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]); offset += hidden_dim
    right_proj = fused[..., offset : offset + hidden_dim]; offset += hidden_dim
    right_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim]); offset += hidden_dim
    out_gate = torch.sigmoid(fused[..., offset : offset + hidden_dim])          # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 4. Fuse mask, left/right gates with a Triton kernel
    # ------------------------------------------------------------------
    if mask is None:
        # All‑ones mask (no masking)
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=input_tensor.device)
    else:
        # Ensure mask is fp16 and flatten; shape = [B·N·N]
        mask_flat = mask.to(torch.float16).reshape(-1)

    # Flatten tensors (contiguous view)
    left_proj_f = left_proj.reshape(-1)
    right_proj_f = right_proj.reshape(-1)
    left_gate_f = left_gate.reshape(-1)
    right_gate_f = right_gate.reshape(-1)

    left_out_f  = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elems = left_proj_f.shape[0]                     # B·N·N·H
    BLOCK_SIZE = 1024                                      # tuneable

    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    _mask_gate_fuse_kernel[grid](
        left_proj_f,
        right_proj_f,
        left_gate_f,
        right_gate_f,
        mask_flat,
        left_out_f,
        right_out_f,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # Reshape back to [B, N, N, H]
    left  = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5. N³ contraction via batched half‑precision GEMM (torch.bmm)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B·H, N, N) matrices
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # (fp16) batched matmul uses Tensor‑Core kernels on H100
    out_mat = torch.bmm(left_mat, right_mat)               # (B*H, N, N) fp16

    # Reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)  # [B, N, N, H]
        .contiguous()
    )  # fp16

    # ------------------------------------------------------------------
    # 6. Post‑contraction LayerNorm (float32) + output gate
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)   # broadcast over H

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim D
    # ------------------------------------------------------------------
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)           # (D, H) weight, fp16 output
    out = out.to(torch.float32)

    return out
