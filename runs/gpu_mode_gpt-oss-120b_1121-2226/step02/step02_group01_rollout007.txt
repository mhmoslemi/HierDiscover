"""
TriMul (outgoing) – optimized Triton‑accelerated forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and fuse the five linear projections (left, right, three gates)
   into a single GEMM (size (B·N², dim) × (dim, 5·hidden_dim)).
3. Slice the fused result into `left`, `right`, `left_gate`,
   `right_gate` and `out_gate`.  Apply a tiny Triton kernel that
   fuses the element‑wise mask (if present) and the gate sigmoids:
         left  = left  * sigmoid(left_gate)  * mask
         right = right * sigmoid(right_gate) * mask
   The kernel works on a 1‑D view of the tensors, avoiding extra
   memory traffic.
4. Core cubic update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication L @ Rᵀ where L and R are
   reshaped to (B·hidden_dim, N, N) and executed with `torch.bmm`,
   which on an H100 leverages highly‑optimized Tensor‑Core kernels.
5. Second LayerNorm over the hidden dimension.
6. Multiply by the output gate (`out_gate`) and finish with a linear
   projection back to the original feature dimension.
All heavy arithmetic is performed in fp16 with fp32 accumulation where
needed; the final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, gate sigmoid and element‑wise multiplication
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,                     # flattened mask (B·N·N) – fp16
    left_out_ptr, right_out_ptr, # outputs – fp16
    N_ELEMENTS: tl.constexpr,    # total number of elements = B·N·N·H
    H: tl.constexpr,             # hidden dimension
    HAS_MASK: tl.constexpr,      # 0 → no mask, 1 → apply mask
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # 1‑D index
    mask = offs < N_ELEMENTS

    # Load operands (masked loads return 0.0 for out‑of‑bounds)
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)          # fp16
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)   # fp16
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)        # fp16
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0) # fp16

    if HAS_MASK:
        # mask is stored without the hidden dimension, so we divide by H
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)   # fp16
        out_l = l * tl.math.sigmoid(lg) * m
        out_r = r * tl.math.sigmoid(rg) * m
    else:
        out_l = l * tl.math.sigmoid(lg)
        out_r = r * tl.math.sigmoid(rg)

    tl.store(left_out_ptr + offs, out_l, mask=mask)
    tl.store(right_out_ptr + offs, out_r, mask=mask)


# ----------------------------------------------------------------------
# Main entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float) – optional
        - weights_dict : mapping of all linear / norm parameters (float32)
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Fuse all five linear maps into a single GEMM (fp16)
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim]  (fp16)

    # Concatenate the five weight matrices (out_features stacked)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).half()  # shape (5*hidden_dim, dim)

    fused = F.linear(x_h, w_concat)  # [B, N, N, 5*hidden_dim]  (fp16)

    # ------------------------------------------------------------------
    # 3) Slice fused tensor
    # ------------------------------------------------------------------
    h = hidden_dim
    left_raw   = fused[..., 0:h]
    right_raw  = fused[..., h:2 * h]
    left_gate  = fused[..., 2 * h:3 * h]
    right_gate = fused[..., 3 * h:4 * h]
    out_gate   = fused[..., 4 * h : 5 * h]  # (fp16)

    # ------------------------------------------------------------------
    # 4) Fuse mask, sigmoid gates and element‑wise multiplication via Triton
    # ------------------------------------------------------------------
    total_elems = left_raw.numel()                     # B·N·N·H
    BLOCK = 1024                                        # threads per block
    grid = ( (total_elems + BLOCK - 1) // BLOCK, )

    # Prepare mask (flattened) – broadcast automatically across H
    if nomask:
        # dummy mask of ones – never used when HAS_MASK=0
        mask_fp16 = torch.tensor([1.0], dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(dtype=torch.float16).contiguous().view(-1)

    # Allocate output tensors
    left = torch.empty_like(left_raw)
    right = torch.empty_like(right_raw)

    # Launch kernel (mask load is compiled out when HAS_MASK==0)
    _mask_gate_fuse_kernel[grid](
        left_ptr=left_raw,
        left_gate_ptr=left_gate,
        right_ptr=right_raw,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp16,
        left_out_ptr=left,
        right_out_ptr=right,
        N_ELEMENTS=total_elems,
        H=hidden_dim,
        HAS_MASK=0 if nomask else 1,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5) Core cubic update – batched matmul L @ Rᵀ  (fp16 → fp32 accumulator)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiplication (Tensor‑Core friendly)
    # result is fp16; we immediately cast to fp32 for the following ops
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).float()  # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 7) Output gate (sigmoid) and final linear projection
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(out_gate)          # fp16
    out = out * out_gate.float()                # broadcast, fp32

    out = F.linear(out, weights["to_out.weight"])  # final projection, fp32

    return out
