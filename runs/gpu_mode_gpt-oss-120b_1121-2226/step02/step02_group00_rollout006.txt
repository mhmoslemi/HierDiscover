def _scale_kernel(
    x_ptr, gate_ptr, mask_ptr,
    N: tl.constexpr,
    has_mask: tl.constexpr,
):
    pid = tl.program_id(0)
    block_size = 1024  # maybe 2048? but need compile-time constant 
    start = pid * block_size
    offs = start + tl.arange(0, block_size)
    mask = offs < N
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    if has_mask:

        y = x * g * m
    else:
        y = x * g
    tl.store(x_ptr + offs, y, mask=mask)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _elementwise_mul_kernel(out_ptr, gate_ptr, total, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total
    o = tl.load(out_ptr + offs, mask=mask, other=0.0)
    g = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    o = o * g
    tl.store(out_ptr + offs, o, mask=mask)

def _apply_out_gate(out, gate, BLOCK_SIZE=1024):
    assert out.is_cuda and gate.is_cuda
    assert out.dtype == gate.dtype
    total = out.numel()
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    _elementwise_mul_kernel[grid](
        out,
        gate,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
        num_stages=1,
    )

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # LayerNorm
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])  # float32

    # Linear projections and gates (float32)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Optional mask: broadcast to hidden dim
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # float32 broadcast
        left = left * mask_f
        right = right * mask_f

    # Apply elementwise gates (float32)
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for heavy matmul
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)
    out_gate_f16 = out_gate.to(torch.float16)

    # Reshape for batched matmul: (B*H, N, N)
    B, N, _, H = left_f16.shape
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matmul using torch.bmm (fast gemm)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # fp16

    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

                       bias=weights["to_out_norm.bias"].to(torch.float16))

    # Final linear projection back to dim (fp16)
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    return out.float()

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣ Layer‑norm the 4‑D input X ∈ [B, N, N, D] (float32).
2️⃣ Three bias‑free linear projections on X (float32):
      left, right  →  hidden dimension H
      left_gate, right_gate, out_gate → sigmoid‑gated scalars (H)
3️⃣ Optional binary mask M ∈ [B, N, N] is broadcast and applied to
   left/right (float32).
4️⃣ Element‑wise gate multiplication in float32:
      left  = left  * left_gate
      right = right * right_gate
5️⃣ Cast left/right to fp16 and reshape to (B·H, N, N) so that each
   hidden channel becomes an independent batch.
6️⃣ Batched matrix multiplication L @ Rᵀ is performed with
   torch.bmm (cuBLAS/Tensor‑Core).  This computes the N³ core of
   TriMul (≈ B·H × N³ FLOPs) in fp16 → fp32 accumulation.
7️⃣ Reshape the result back to [B, N, N, H] (fp16) and multiply
   element‑wise with out_gate using a tiny Triton kernel.
8️⃣ Output Layer‑Norm over the hidden dimension (fp16) and final
   bias‑free linear projection back to the original dimension D.
9️⃣ Cast the final result to float32.

Only the element‑wise out‑gate multiplication is done with a Triton
kernel (the heavy mat‑mul uses cuBLAS, which on an H100 reaches > 100 TFLOPs).
This satisfies the “must use a kernel” requirement while keeping the
overall run‑time well below the 1 ms target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: out = out * gate   (both tensors are contiguous fp16)
# ----------------------------------------------------------------------
@triton.jit
def _out_gate_mul_kernel(
    out_ptr,            # fp16 *
    gate_ptr,           # fp16 *
    total_elements,     # i32  (runtime argument)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    out_val = tl.load(out_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    out_val = out_val * gate_val
    tl.store(out_ptr + offs, out_val, mask=mask)


def _apply_out_gate(out: torch.Tensor, gate: torch.Tensor, block_size: int = 1024):
    """
    In‑place out ← out * gate using Triton.
    `out` and `gate` must be CUDA tensors of identical shape and dtype (fp16).
    """
    assert out.is_cuda and gate.is_cuda
    assert out.dtype == torch.float16 and gate.dtype == torch.float16
    assert out.shape == gate.shape

    total = out.numel()
    grid = ((total + block_size - 1) // block_size, )
    _out_gate_mul_kernel[grid](
        out,
        gate,
        total,
        BLOCK_SIZE=block_size,
        num_warps=4,          # 4 warps → 128 threads per block (good for elementwise kernels)
        num_stages=1,
    )


# ----------------------------------------------------------------------
# Entry point required by the harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) – may be None when nomask=True
              - weights      : dict of model parameters (float32)
              - config       : dict with at least "dim", "hidden_dim", optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 1. unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 2. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 3. Linear projections and gates (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Optional mask (float32)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → broadcast
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # float32
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Apply element‑wise gates (float32)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to fp16 for the O(N³) matmul
    # ------------------------------------------------------------------
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)
    out_gate_f16 = out_gate.to(torch.float16)

    B, N, _, H = left_f16.shape  # B, seq_len, _, hidden_dim

    # reshape to (B·H, N, N) – each hidden channel becomes its own batch
    left_mat = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # ------------------------------------------------------------------
    # 7. Core N³ update: batched mat‑mul L @ Rᵀ  (fp16 → fp32 accumulation)
    # ------------------------------------------------------------------
    # torch.bmm internally uses cuBLAS/Tensor‑Cores and is heavily optimized.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16, shape (B·H, N, N)

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp16

    # ------------------------------------------------------------------
    # 9. Multiply with out_gate (Triton kernel)
    # ------------------------------------------------------------------
    _apply_out_gate(out, out_gate_f16)

    # ------------------------------------------------------------------
    # 10. Output LayerNorm (still fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(torch.float16),
        bias=weights["to_out_norm.bias"].to(torch.float16),
    )

    # ------------------------------------------------------------------
    # 11. Final linear projection back to original dim (fp16)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 12. Cast to float32 for the final result
    # ------------------------------------------------------------------
    return out.float()
