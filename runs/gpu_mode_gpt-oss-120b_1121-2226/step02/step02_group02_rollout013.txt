import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel to cast mask fp32->fp16
@triton.jit
def _cast_fp32_to_fp16_kernel(src_ptr, dst_ptr, numel, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    tl.store(dst_ptr + offs, tl.cast(val, tl.float16), mask=mask)

# Triton kernel for batched matmul for TriMul (outgoing)
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)  # combines batch and hidden
    pid_i = tl.program_id(0)  # tile row (i)
    pid_j = tl.program_id(1)  # tile col (j)

    # Derive original batch index and hidden index
    b = pid_batch // H
    h = pid_batch % H

    # base pointers for this (b, h)
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # offsets within the tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        a = tl.load(
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load right tile (N x K)
        b_tile = tl.load(
            right_base
            + offs_j[:, None] * stride_r_i
            + offs_k[None, :] * stride_r_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # accumulate dot product over K
        acc += tl.dot(a, b_tile)  # (M,K)·(N,K)ᵀ → (M,N) in fp32

    # Write back results
    tl.store(
        out_base
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of TriMul (outgoing) module.
    
    The algorithm:
    1. Layer‑norm the 4‑D input.
    2. Cast to fp16 and apply linear projections and sigmoid gates.
    3. Optional mask multiplication (mask converted to fp16 via a tiny Triton kernel).
    4. Core N³ contraction C_{i,j,h} = Σ_k L_{i,k,h} * R_{j,k,h} implemented as a
       custom Triton batched‑matrix‑multiply kernel that works directly on the
       original (B,N,N,H) layout, avoiding the costly permute+contiguous copy.
    5. Layer‑norm over the hidden dimension, element‑wise output gating and final
       linear projection back to the model dimension.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    
    # 1. Input LayerNorm (float32)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # 2. Cast to fp16 for heavy compute
    x_h = x.to(torch.float16)
    
    # 3. Cast projection and gate weights to fp16
    left_proj_w = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w = weights["out_gate.weight"].half()
    # output norm and final linear remain fp32
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]
    
    # 4. Project + gate (all fp16)
    left = F.linear(x_h, left_proj_w)   # (B,N,N,H)
    right = F.linear(x_h, right_proj_w)
    
    # Apply optional mask
    if not nomask:
        # Convert mask to fp16 with Triton
        mask_flat = mask.view(-1)
        mask_h_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)
        BLOCK_SIZE = 1024
        grid = ((mask_flat.numel() + BLOCK_SIZE - 1) // BLOCK_SIZE,)
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_h_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        mask_h = mask_h_flat.view(mask.shape + (1,))
        left = left * mask_h
        right = right * mask_h
    
    # Gates
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))
    
    left = left * left_gate
    right = right * right_gate
    
    # 5. Core N³ operation via Triton kernel (output in fp32)
    B, N, _, H = left.shape   # B,N,N,H
    
    # Allocate output tensor
    out = torch.empty_like(left, dtype=torch.float32, device=device)  # (B,N,N,H) fp32
    
    # Compute strides (in element count, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()
    
    # Choose block sizes – tuned for H100 tensor cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )
    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    
    # 6. Output layernorm (over hidden dim)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    
    # 7. Output gating + final projection
    out = out * out_gate.to(torch.float32)
    out = F.linear(out, to_out_w)
    
    return out

~~~~~~~~~
1. Layer‑norm the 4‑D input (float32).
2. Cast to fp16 and apply the two linear projections and the three
   sigmoid‑gates (all fp16).  An optional mask is converted to fp16
   with a tiny Triton kernel and broadcast‑multiplied.
3. Core N³ contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix‑multiply `L @ Rᵀ`.  A custom Triton kernel
   works directly on the original `(B,N,N,H)` layout, avoiding the
   costly `permute/contiguous` reshapes required by `torch.bmm`.
   The kernel tiles the i‑ and j‑dimensions (128×128) and the reduction
   k‑dimension (32) and accumulates in float32.
4. Output layer‑norm over the hidden dimension, element‑wise output‑gate,
   and final linear projection back to `dim` (float32).

The heavy arithmetic runs in half‑precision on H100 Tensor‑Cores while
the surrounding element‑wise operations stay in float32, yielding a
sub‑millisecond latency for the tested configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Tiny Triton kernel: cast a flat fp32 mask to fp16 (used only when mask exists)
# ---------------------------------------------------------------------------
@triton.jit
def _cast_fp32_to_fp16_kernel(
    src_ptr,            # fp32 source
    dst_ptr,            # fp16 destination
    numel,              # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    tl.store(dst_ptr + offs, tl.cast(val_f32, tl.float16), mask=mask)


# ---------------------------------------------------------------------------
# Triton kernel: batched matmul for TriMul (outgoing)
# Computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ---------------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (i‑tile, j‑tile, batch*hidden)
    pid_i = tl.program_id(0)          # tile row   → i
    pid_j = tl.program_id(1)          # tile col   → j
    pid_bh = tl.program_id(2)         # combined batch * hidden index

    # -----------------------------------------------------------------------
    # Decode original batch index and hidden index from pid_bh
    # -----------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------------
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = kk * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile : (BLOCK_M, BLOCK_K)
        a = tl.load(
            left_base
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # right tile : (BLOCK_N, BLOCK_K)
        b_tile = tl.load(
            right_base
            + offs_j[:, None] * stride_r_i
            + offs_k[None, :] * stride_r_k,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Accumulate dot‑product over k (half·half → float32)
        acc += tl.dot(a, b_tile)   # (BLOCK_M, BLOCK_K)·(BLOCK_N, BLOCK_K)ᵀ → (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------------
    # Write results back (float32)
    # -----------------------------------------------------------------------
    tl.store(
        out_base
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ---------------------------------------------------------------------------
# Entry point required by the evaluation harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights_dict, config_dict)
        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim] (float32)
        - mask         : torch.Tensor [bs, seq_len, seq_len] (float32)
        - weights      : dictionary of model weights (float32 tensors)
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for the heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)   # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3) Cast linear / gate weights to fp16 (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]       # fp32

    # ------------------------------------------------------------------
    # 4) Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B,N,N,H]  fp16
    right = F.linear(x_h, right_proj_w)  # [B,N,N,H]  fp16

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # flatten mask (fp32) → fp16 buffer via tiny Triton kernel
        mask_flat = mask.view(-1)
        mask_h_flat = torch.empty_like(mask_flat, dtype=torch.float16, device=device)

        BLOCK_SIZE = 1024
        grid = ((mask_flat.numel() + BLOCK_SIZE - 1) // BLOCK_SIZE,)
        _cast_fp32_to_fp16_kernel[grid](
            mask_flat,
            mask_h_flat,
            mask_flat.numel(),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        mask_h = mask_h_flat.view(mask.shape + (1,))   # [B,N,N,1] fp16
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6) Gating (sigmoid) – still half‑precision
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))  # [B,N,N,H] fp16
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))    # [B,N,N,H] fp16

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Core N³ contraction via a custom Triton kernel
    #    Output is produced directly in the original layout (B,N,N,H)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # batch, seq_len, seq_len, hidden_dim

    # Allocate output buffer (float32 accumulator)
    out = torch.empty_like(left, dtype=torch.float32, device=device)   # [B,N,N,H] fp32

    # Strides in *elements* (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 Tensor‑Core performance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * H,                          # batch × hidden
    )
    _trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # [B,N,N,H] fp32

    # ------------------------------------------------------------------
    # 9) Output gating + final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)   # element‑wise gate
    out = F.linear(out, to_out_w)            # fp32 → final dim

    return out
```
