"""
Fast Triton implementation of the “outgoing” TriMul operator (AlphaFold3).

The forward pass consists of four stages:

1️⃣ Input LayerNorm (float32) → half‑precision.
2️⃣ Fused linear projection: the five weight matrices 
   (left_proj, right_proj, left_gate, right_gate, out_gate)
   are concatenated into a single GEMM (half → half).  The result is
   reshaped and split into three projection tensors and three gate tensors
   (the gates pass through a sigmoid).  An optional mask is broadcast and
   applied before gating.
3️⃣ Cubic update: for each hidden slice h we need
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply A @ Bᵀ.  The kernel
   `tri_mul_outgoing_kernel` computes this in FP16 with FP32 accumulation,
   using a 128×128×64 tiling (8 warps, 3 stages) that fully utilizes H100
   tensor‑cores.
4️⃣ Second LayerNorm + output‑gate (float32) → final linear projection
   back to the original model dimension.

The final tensor has shape `[B, N, N, dim]` and dtype ``float32``.
All heavy operations are performed in half‑precision; only the two
LayerNorms and the output gate are kept in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ (A, B half → C half, acc in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C[b,i,j] = Σ_k A[b,i,k] * B[b,j,k]   (i.e. A @ Bᵀ)."""
    pid_h = tl.program_id(axis=2)      # hidden‑batch index (B·H)
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- load A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)               # FP16 → FP32

        # ----- load Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)               # FP16 → FP32

        # ----- inner product -----
        acc += tl.dot(a, b)                # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ----- write result (back to FP16) -----
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]               # model dimension (C)
    hidden_dim = config["hidden_dim"] # intermediate dimension (C_z)
    # ``nomask`` = True → mask should be ignored (the mask tensor may still be present)
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half‑precision for the heavy ops
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, dim]  float32
    x_h = x.half()                               # half

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 × hidden_dim outputs)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (all have shape [hidden_dim, dim])
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # half‑precision weight

    # Single GEMM → shape [B, N, N, 5·hidden_dim]
    fused = F.linear(x_h, fused_weight)          # half
    B, N, _, _ = fused.shape

    # Reshape and split
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B, N, N, 5, H]
    left_proj  = fused[..., 0, :]                # [B, N, N, H]
    right_proj = fused[..., 1, :]
    left_gate  = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] (half)
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates (still half‑precision)
    # ------------------------------------------------------------------
    left = left_proj * left_gate                 # [B,N,N,H] half
    right = right_proj * right_gate              # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 5️⃣ Rearrange tensors for the batched matmul (A @ Bᵀ)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim                               # batch × hidden
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N] half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N] half

    # ------------------------------------------------------------------
    # 6️⃣ Triton kernel: compute out = left @ rightᵀ per hidden slice
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Tile sizes tuned for H100 tensor‑cores; they work well for all
    # supported N (256‑1024) and hidden_dim ≤ 128.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one launch per (batch, hidden) slice
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        # Bᵀ strides: (BH stride, K stride, N stride) → need to swap last two
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output gate
    # ------------------------------------------------------------------
    out = out_half.float()                                # cast to fp32
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                                      # [B,N,N,H] float32

    out = out * out_gate.float()                          # apply output gate

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out.half(), to_out_weight_h)      # half → half
    out_final = out_proj.float()                          # cast back to fp32

    return out_final
