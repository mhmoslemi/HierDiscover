"""
TriMul (outgoing) forward pass – highly‑optimised Triton implementation.

Algorithm
---------
1. Layer‑norm the input X ∈ [B, N, N, D].
2. Fuse the five per‑position linear layers (left_proj, left_gate,
   right_proj, right_gate, out_gate) into a single matmul:
        Y = X @ W_fused.T      # shape (B, N, N, 5·H)   (fp16)
   where W_fused stacks the five weight matrices (each H×D).
3. Split Y → (Lproj, Lgate, Rproj, Rgate, Ogate) and apply sigmoids.
   Compute gated tensors:
        left  = Lproj * sigmoid(Lgate)
        right = Rproj * sigmoid(Rgate)
   (optional binary mask M is applied after gating).
4. Core N³ update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix multiplication for each hidden channel.
   The implementation uses a custom Triton kernel that tiles the
   computation (BLOCK_M×BLOCK_N×BLOCK_K = 128×128×64) and accumulates
   in fp32.
5. Reshape back to [B, N, N, H] and apply an output LayerNorm.
6. Multiply by the out‑gate (sigmoid of Ogate) and project back to
   the original dimension with a final linear layer.
7. Return the result (float32).

The heavy memory traffic of the original PyTorch version is reduced
by fusing the five linear projections into a single GEMM, cutting the
number of reads of the large tensor X from five to one.  The
matrix‑multiplication kernel is also tuned (larger BLOCK_K, out‑param
for `tl.dot`, and 16 warps) to approach the theoretical limits on an
H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A (M×K) * B (N×K) → C (M×N)
#   - A, B: fp16, C: fp32 (fp16·fp16 → fp32 accumulation)
#   - batch dimension is the 3rd launch axis.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,    # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,    # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program ids
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (B·H)

    # --------------------------------------------------------------
    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16

        # B tile: (BLOCK_K, BLOCK_N)  (note: B is stored as (batch, N, K))
        b_ptrs = b_batch + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # fused dot‑accumulate
        tl.dot(a_tile, b_tile, out=acc)   # acc += a_tile @ b_tile (fp32)

    # --------------------------------------------------------------
    # write back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched fp16·fp16→fp32 mat‑mul using the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C with shape (batch, M, N) in fp32.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16 inputs"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Strides in element units
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=16,          # 16 warps → 512 threads per block
        num_stages=4,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated forward pass of the TriMul “outgoing” update.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor: torch.Tensor [B, N, N, D] (float32)
              - mask: torch.Tensor [B, N, N] (bool/int) – may be None when nomask=True
              - weights: dict of model parameters (float32)
              - config: dict containing at least "dim", "hidden_dim", and optional "nomask"

    Returns:
        Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2) Fuse all per‑position linear projections + gates into one matmul
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)   # cast once

    # Concatenate the five weight matrices (left_proj, left_gate,
    # right_proj, right_gate, out_gate) -> shape (5·H, D)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*H, D)

    # One big linear projection
    fused_y = F.linear(x_f16, fused_weight)   # [B, N, N, 5·H]   fp16

    # Split the fused tensor
    H = hidden_dim
    left_proj   = fused_y[..., 0                :H]          # (B,N,N,H)
    left_gate_z = fused_y[..., H                :2*H]        # (B,N,N,H)
    right_proj  = fused_y[..., 2*H              :3*H]        # (B,N,N,H)
    right_gate_z= fused_y[..., 3*H              :4*H]        # (B,N,N,H)
    out_gate_z  = fused_y[..., 4*H              :5*H]        # (B,N,N,H)

    # Apply sigmoid to gate logits
    left_gate = torch.sigmoid(left_gate_z)   # fp16
    right_gate = torch.sigmoid(right_gate_z) # fp16
    out_gate = torch.sigmoid(out_gate_z)    # fp16

    # Gated tensors
    left = left_proj * left_gate            # fp16
    right = right_proj * right_gate         # fp16

    # Optional mask (binary mask is broadcasted over the hidden dim)
    if not nomask:
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 3) Core N³ batched mat‑mul (outgoing update)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    # For the right side we need the transpose over the K‑dimension;
    # the Triton kernel expects B stored as (batch, N, K) and loads it as (K,N)
    # which implements the required transpose internally.
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16

    # Batched mat‑mul via Triton
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )   # [B·H, N, N]  fp32

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 4) Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # out_gate is fp16 → broadcasted to fp32 automatically
    out = out * out_gate

    # Final linear projection back to the original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
