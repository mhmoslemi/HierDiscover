Implementation details:

Define kernel:

```
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    hidden_dim: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # get pid_h = program_id(axis=2) which indexes over BH slices (b*hidden + h)
    pid_h = tl.program_id(axis=2)
    # compute b and h indices
    b_idx = pid_h // hidden_dim
    h_idx = pid_h % hidden_dim

    # tile offsets for i and j
    pid_m = tl.program_id(axis=0) # i tile
    pid_n = tl.program_id(axis=1) # j tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base offsets
    a_base = b_idx * stride_a_batch + h_idx * stride_a_h
    b_base = b_idx * stride_b_batch + h_idx * stride_b_h
    c_base = b_idx * stride_c_batch + h_idx * stride_c_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        a_ptrs = a_ptr + a_base + offs_m[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        b_ptrs = b_ptr + b_base + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_ptr + c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)  # flattened BH idx
    batch_idx = pid_h // H
    hidden_idx = pid_h % H

    # offsets for i and j
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_base = batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_base = batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_base = batch_idx * stride_c_batch + hidden_idx * stride_c_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # A tile (i,k)
        a_ptrs = a_ptr + a_base + offs_m[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # B transposed tile (k,j)
        b_ptrs = b_ptr + b_base + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)
    # Store
    c_ptrs = c_ptr + c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Input layernorm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.half()  # half

    # Fuse weights
    fused_weight = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(device).half()

    # fused linear
    fused = F.linear(x_h, fused_weight)  # (B,N,N,5*hidden)
    B, N, _, _ = fused.shape

    fused = fused.view(B, N, N, 5, hidden_dim)  # (B,N,N,5,H)

    left_proj = fused[..., 0, :]  # (B,N,N,H)
    right_proj = fused[..., 1, :]
    left_gate = torch.sigmoid(fused[..., 2, :]).half()
    right_gate = torch.sigmoid(fused[..., 3, :]).half()
    out_gate = torch.sigmoid(fused[..., 4, :]).half()

    # Apply mask if any
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)  # shape (B,N,N,1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gating
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Allocate output tensor
    out = torch.empty_like(left)  # half, shape (B,N,N,H)

    # Compute strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,

"""
TriMul (outgoing) – Triton implementation
------------------------------------------------
*   Input Layer‑Norm (float32) → half.
*   Five linear layers are fused into a single GEMM (half‑precision).
*   Gating + optional mask is applied element‑wise.
*   Batched matrix‑multiply `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
    is performed by a custom Triton kernel that works directly on the
    `[B, N, N, H]` layout – no permutes or copies are required.
    The kernel treats the hidden dimension as part of the batch (BH = B·H)
    and computes `A @ Bᵀ` with FP16 inputs, FP32 accumulation and FP16 output.
*   Second Layer‑Norm + output‑gate (float32) → final linear projection
    back to `dim`.
The design eliminates the expensive transposes/copies of the original
reference implementation, which yields a large speed‑up on large
sequences (N≈1024) on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all == seq_len)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,   # A strides
    stride_b_batch, stride_b_j, stride_b_k, stride_b_h,   # B strides (Bᵀ layout)
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,   # C strides
    HIDDEN_DIM: tl.constexpr,          # hidden dimension (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiply for TriMul (outgoing).

    Computes, for every (batch, hidden) slice:
        out[i, j] = Σ_k left[i, k] * right[j, k]
    All tensors are half‑precision, accumulation is FP32.
    """
    pid_h = tl.program_id(axis=2)               # flattened (batch * hidden)
    batch_idx = pid_h // HIDDEN_DIM
    hidden_idx = pid_h % HIDDEN_DIM

    pid_m = tl.program_id(axis=0)               # i‑tile
    pid_n = tl.program_id(axis=1)               # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer for this (batch, hidden) slice
    a_base = batch_idx * stride_a_batch + hidden_idx * stride_a_h
    b_base = batch_idx * stride_b_batch + hidden_idx * stride_b_h
    c_base = batch_idx * stride_c_batch + hidden_idx * stride_c_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- load A tile (i, k) -----
        a_ptrs = a_ptr + a_base + offs_m[:, None] * stride_a_i + cur_k[None, :] * stride_a_k
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)               # FP16 → FP32

        # ----- load Bᵀ tile (k, j) -----
        b_ptrs = b_ptr + b_base + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)               # FP16 → FP32

        # ----- inner product -----
        acc += tl.dot(a, b)                # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ----- store result (FP16) -----
    c_ptrs = c_ptr + c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                             # [B, N, N, dim]  float32
    x_h = x.half()                               # half for heavy work

    # --------------------------------------------------------------
    # 2️⃣ Fuse the five linear layers into one GEMM
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(device).half()                         # (5*H, dim) in half

    # [B, N, N, 5*H]  (half)
    fused = F.linear(x_h, fused_weight)
    B, N, _, _ = fused.shape

    # reshape to separate the five heads
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H] (half)

    left_proj = fused[..., 0, :]                 # [B,N,N,H]
    right_proj = fused[..., 1, :]
    left_gate = torch.sigmoid(fused[..., 2, :]).half()
    right_gate = torch.sigmoid(fused[..., 3, :]).half()
    out_gate = torch.sigmoid(fused[..., 4, :]).half()

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)   # [B,N,N,1] half
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # --------------------------------------------------------------
    left = left_proj * left_gate      # [B,N,N,H] half
    right = right_proj * right_gate   # [B,N,N,H] half

    # --------------------------------------------------------------
    # 5️⃣ Batched GEMM kernel A @ Bᵀ  (no transposes / copies)
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # half, shape [B,N,N,H]

    # strides for the kernel (in elements, not bytes)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,      # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,      # j‑tiles
        B * hidden_dim,                    # (batch * hidden) slices
    )

    tri_mul_outgoing_kernel[grid](
        left,                                 # a_ptr
        right,                                # b_ptr
        out,                                  # c_ptr
        N, N, N,                              # M, N, K
        stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
        stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
        stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
        HIDDEN_DIM=hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate
    # --------------------------------------------------------------
    out_fp32 = out.float()   # cast to fp32 for layer‑norm stability
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp32 = out_fp32 * out_gate.float()   # apply output gate

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(device).half()   # (dim, hidden)
    out_proj = F.linear(out_fp32.half(), to_out_weight_h)          # half output
    out_final = out_proj.float()                                   # back to fp32

    return out_final
```
