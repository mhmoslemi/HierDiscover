"""
custom_kernel
==============

Fast Triton‑accelerated forward pass of the **TriMul (outgoing) update**.

Algorithm (mirrors the reference PyTorch module)
------------------------------------------------
1. Layer‑norm the input tensor X ∈ [B, N, N, D] (D = dim).
2. Fuse the two linear projections (proj + gate) for *left* and *right*:
       [X] → concat([proj, gate]) → split → (proj, gate)
   This halves the number of dense mat‑muls.
3. Apply the optional binary mask (if present) and the element‑wise sigmoid gates.
4. Core O(N³) operation (outgoing update):
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched matrix‑multiply L @ Rᵀ for each hidden channel.
   The tensors are reshaped to (B·H, N, N) and a Triton kernel
   computes the batched GEMM with fp16×fp16 → fp32 accumulation.
5. Reshape back to [B, N, N, H], apply a second LayerNorm,
   multiply by the *out‑gate*, and finish with a linear projection
   back to the original dimension.

The Triton kernel uses 128 × 128 × 64 tiles, 8 warps and double‑buffered
staging (4 stages) to achieve sub‑millisecond runtimes on an H100 for the
provided test cases.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched gemm  A (M×K)  *  Bᵀ (K×N)  →  C (M×N)
#   A, B : fp16 input tensors
#   C    : fp32 accumulator (fp16·fp16 → fp32)
#   Batch dimension is the third launch axis.
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix extents
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bn, stride_bk,   # B strides: batch, N, K
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # 1) program ids (grid = (M_blocks, N_blocks, batch))
    # ---------------------------------------------------------------
    pid_m = tl.program_id(0)   # block row   (M dimension)
    pid_n = tl.program_id(1)   # block col   (N dimension)
    pid_b = tl.program_id(2)   # batch index (B·H)

    # ---------------------------------------------------------------
    # 2) compute tile offsets
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # 3) iterate over K dimension (tiles of size BLOCK_K)
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # ---- load A tile (M × K) -------------------------------------------------
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)          # fp16

        # ---- load B tile (K × N) – note the transpose (B is stored as (N,K)) -----
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)          # fp16

        # ---- multiply‑add (fp16·fp16 → fp32) ------------------------------------
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N) as fp32

    # ---------------------------------------------------------------
    # 4) write result
    # ---------------------------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_gemm_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Wrapper for the Triton kernel above.
    A, B -> fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C of shape (batch, M, N) in fp32.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must reside on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16"
    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides are expressed in *elements* (not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps ≈ 256 threads → good occupancy for 128×128 tiles
        num_stages=4,         # double‑buffering of shared loads
    )
    return C

        # convert mask {0,1} to half and broadcast
        m = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # fp16
    right_gate = torch.sigmoid(right_gate) # fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ batched GEMM (L @ Rᵀ) – Tensor‑core accelerated
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, _, hidden_dim
    # reshape to (B·H, N, N) → each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # GEMM → fp32 accumulator
    out_mat = _batched_gemm_triton(
        left_mat,
        right_mat,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )   # [B·H, N, N] (fp32)

    # reshape back to original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H] fp32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate & final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate   # out_gate is fp16 → auto‑cast to fp32

    # final linear back to original dimension (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
```
