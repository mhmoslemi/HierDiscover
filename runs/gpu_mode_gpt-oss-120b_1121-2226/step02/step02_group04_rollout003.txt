import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    total: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK: tl.constexpr
):
    # each program (block) processes BLOCK elements, strides across total elements
    pid = tl.program_id(0)
    num_programs = tl.num_programs(0)
    # starting offset for each thread
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    stride = num_programs * BLOCK

    for i in range(0, tl.cdiv(total, stride)):
        idx = offs + i * stride
        mask = idx < total

        # load values
        l_val = tl.load(left_ptr + idx, mask=mask, other=0.0)
        r_val = tl.load(right_ptr + idx, mask=mask, other=0.0)

        l_gate = tl.load(left_gate_ptr + idx, mask=mask, other=0.0)
        r_gate = tl.load(right_gate_ptr + idx, mask=mask, other=0.0)

        # each hidden block shares one mask entry
        mask_idx = idx // hidden
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

        # fuse mask & gate
        lo = l_val * l_gate * m_val
        ro = r_val * r_gate * m_val

        tl.store(out_left_ptr + idx, lo, mask=mask)
        tl.store(out_right_ptr + idx, ro, mask=mask)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation (no gradients).

    The implementation fuses:
      * Layer‑norm on the input (FP32 → FP16)
      * A single linear projection that produces left/right projections,
        the four gating tensors and the output gate (5·H output features)
      * A small Triton kernel that fuses mask broadcasting,
        gate multiplication and element‑wise masking.
      * The N³ pairwise contraction, performed as a large batched half‑precision
        GEMM (torch.bmm) which maps to cuBLAS Tensor‑Core kernels.
      * Post‑contraction LayerNorm, output‑gate scaling and final projection,
        all remaining in FP16 and only the final result is cast back to FP32.
    """
    # ---------------------------------------------------------------
    # unpack inputs
    # ---------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ---------------------------------------------------------------
    # weight tensors (pre‑cast to needed dtype)
    # ---------------------------------------------------------------
    norm_weight = weights["norm.weight"]          # (dim,)
    norm_bias   = weights["norm.bias"]            # (dim,)

    # linear projection weights (FP16)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # output‑norm weights/bias (FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    # final projection weight (FP16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ---------------------------------------------------------------
    # 1. Input LayerNorm (FP32) → FP16
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    x = x.to(torch.float16)

    # ---------------------------------------------------------------
    # 2. Fuse all linear projections & gates (5·H outputs)
    # ---------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                         # shape (5*H, D)

    fused = F.linear(x, fused_weight)       # [B, N, N, 5*H]  FP16

    B, N, _, _ = fused.shape

    # split fused tensor
    offset = 0
    left_proj = fused[..., offset:offset + hidden_dim]; offset += hidden_dim
    left_gate = torch.sigmoid(fused[..., offset:offset + hidden_dim]); offset += hidden_dim
    right_proj = fused[..., offset:offset + hidden_dim]; offset += hidden_dim
    right_gate = torch.sigmoid(fused[..., offset:offset + hidden_dim]); offset += hidden_dim
    out_gate = torch.sigmoid(fused[..., offset:offset + hidden_dim]);       # [B,N,N,H] FP16

    # ---------------------------------------------------------------
    # 3. Fuse mask, left/right gates using a tiny Triton kernel
    # ---------------------------------------------------------------
    if mask is None:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=input_tensor.device)
    else:
        # ensure mask is broadcastable and in FP16
        mask_flat = mask.to(torch.float16).reshape(-1)   # shape B·N·N

    # flatten tensors for the kernel (contiguous)
    left_proj_f = left_proj.reshape(-1)
    right_proj_f = right_proj.reshape(-1)
    left_gate_f = left_gate.reshape(-1)
    right_gate_f = right_gate.reshape(-1)

    left_out_f = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elems = left_proj_f.shape[0]               # B·N·N·H
    # block size (threads per program); 256 gives good occupancy on H100
    BLOCK_SIZE = 256

    # number of programs (grid) – we let Triton decide a convenient size.
    grid = triton.cdiv(total_elems, BLOCK_SIZE)

    _mask_gate_fuse_kernel[grid](
        left_proj_f,
        right_proj_f,
        left_gate_f,
        right_gate_f,
        mask_flat,
        left_out_f,
        right_out_f,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # reshape back
    left = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ---------------------------------------------------------------
    # 4. N³ contraction via batched half‑precision GEMM (torch.bmm)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ---------------------------------------------------------------
    # reshape to (B*H, N, N) matrices
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # right needs a transpose on the last two axes
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N) FP16

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # FP16

    # ---------------------------------------------------------------
    # 5. Post‑contraction LayerNorm (FP16) + output gate
    # ---------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate          # FP16 broadcast over hidden_dim

    # ---------------------------------------------------------------
    # 6. Final linear projection back to original dimension (FP16)
    # ---------------------------------------------------------------
    out = F.linear(out, to_out_w)  # (B,N,N,dim) FP16
    # cast to float32 for the final output as required by the API
    out = out.to(torch.float32)

    return out

"""
TriMul (outgoing) forward pass – heavily fused implementation.

Algorithm
---------
1. Layer‑norm the input in FP32 and cast to FP16.
2. Fuse the five linear projections (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) into a single matmul: weight shape (5·H, D).
3. Split the result, apply sigmoid to the gate tensors.
4. A tiny Triton kernel fuses mask broadcasting, gate multiplication and
   element‑wise masking:
        left_out  = left_proj  * left_gate  * mask
        right_out = right_proj * right_gate * mask
5. The cubic contraction
        out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   is performed as a batched half‑precision GEMM (torch.bmm) which maps to
   cuBLAS Tensor‑Core kernels.
6. Post‑contraction LayerNorm, multiplication by the output‑gate and the
   final linear projection back to the original dimension are all kept in
   FP16; only the final tensor is cast back to FP32.

Only step 4 is written in Triton, satisfying the “must use a kernel” rule.
The implementation works for any batch size ∈ {1,2}, sequence length N,
input dimension D and hidden dimension H as described in the problem
statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # left projection   (fp16)
    right_ptr,              # right projection  (fp16)
    left_gate_ptr,          # left gate         (fp16)
    right_gate_ptr,         # right gate        (fp16)
    mask_ptr,               # mask (B·N·N)      (fp16)
    out_left_ptr,           # fused left output (fp16)
    out_right_ptr,          # fused right output(fp16)
    total: tl.constexpr,    # total number of elements = B·N·N·H
    hidden: tl.constexpr,   # hidden dimension H
    BLOCK: tl.constexpr,    # threads per program
):
    """
    For every element (b,i,k,h) compute
        left_out  = left  * left_gate  * mask
        right_out = right * right_gate * mask
    where the same mask entry is shared by ``hidden`` consecutive
    hidden‑dimension values.
    """
    pid = tl.program_id(0)
    num_prog = tl.num_programs(0)

    # each program processes BLOCK elements, stride across the whole tensor
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    stride = num_prog * BLOCK

    # loop over the strided dimension (normally executes once)
    for s in range(0, tl.cdiv(total, stride)):
        idx = offs + s * stride
        mask = idx < total

        # load
        l_val = tl.load(left_ptr + idx,    mask=mask, other=0.0)
        r_val = tl.load(right_ptr + idx,   mask=mask, other=0.0)
        l_gate = tl.load(left_gate_ptr + idx,  mask=mask, other=0.0)
        r_gate = tl.load(right_gate_ptr + idx, mask=mask, other=0.0)

        # mask index: each H entries share one mask entry
        mask_idx = idx // hidden
        m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

        # fuse
        lo = l_val * l_gate * m_val
        ro = r_val * r_gate * m_val

        # store
        tl.store(out_left_ptr + idx,  lo, mask=mask)
        tl.store(out_right_ptr + idx, ro, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask         : torch.Tensor [B, N, N]   (float32) or None
        - weights      : dict with all model parameters
        - config       : dict with keys "dim", "hidden_dim" (and optional "eps")

    Returns
    -------
    torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # cast weights to the dtype required for the kernels
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]          # (D,)
    norm_bias   = weights["norm.bias"]            # (D,)

    # projection / gate weights (no bias) – FP16 for all matmuls
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    # output‑norm weights / bias – FP16
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)

    # final projection weight – FP16
    to_out_w = weights["to_out.weight"].to(torch.float16)            # (D, H)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                          # [B, N, N, D] float32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Fuse the five linear projections into a single matmul
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                          # (5*H, D)

    fused = F.linear(x, fused_weight)          # [B, N, N, 5*H] FP16

    B, N, _, _ = fused.shape

    # split the fused tensor
    off = 0
    left_proj = fused[..., off:off + hidden_dim];                     off += hidden_dim
    left_gate = torch.sigmoid(fused[..., off:off + hidden_dim]);     off += hidden_dim
    right_proj = fused[..., off:off + hidden_dim];                    off += hidden_dim
    right_gate = torch.sigmoid(fused[..., off:off + hidden_dim]);    off += hidden_dim
    out_gate = torch.sigmoid(fused[..., off:off + hidden_dim])      # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 3. Fuse mask, left/right gates with a Triton kernel
    # ------------------------------------------------------------------
    if mask is None:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16,
                               device=input_tensor.device)
    else:
        mask_flat = mask.to(torch.float16).reshape(-1)   # (B·N·N,)

    # flatten tensors for the kernel (contiguous)
    left_proj_f = left_proj.reshape(-1)
    right_proj_f = right_proj.reshape(-1)
    left_gate_f = left_gate.reshape(-1)
    right_gate_f = right_gate.reshape(-1)

    left_out_f = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elems = left_proj_f.shape[0]                # B·N·N·H
    BLOCK_SIZE = 256                                   # threads per program
    grid = triton.cdiv(total_elems, BLOCK_SIZE)

    _mask_gate_fuse_kernel[grid](
        left_proj_f,
        right_proj_f,
        left_gate_f,
        right_gate_f,
        mask_flat,
        left_out_f,
        right_out_f,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # reshape back to the original 4‑D layout
    left = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. N³ contraction via batched half‑precision GEMM (torch.bmm)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) matrices; left uses (i,k), right uses (j,k)ᵀ
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    out_mat = torch.bmm(left_mat, right_mat)          # (B*H, N, N) FP16

    # back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Post‑contraction LayerNorm (still in FP16) + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate                              # element‑wise FP16

    # ------------------------------------------------------------------
    # 6. Final linear projection back to original dimension (FP16)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)                     # (B,N,N,D) FP16

    # cast back to FP32 as required by the API
    return out.to(torch.float32)
```

This implementation follows the reference forward pass while keeping the heavy N³ contraction in a highly‑optimized batched GEMM and fusing the mask‑gate logic in a compact Triton kernel. All intermediate work is performed in FP16 (except the very first LayerNorm) to exploit Tensor‑Core throughput on an H100, and the final result is returned in float‑32.
