"""
TriMul (outgoing) – fast Triton‐based implementation
====================================================

Forward pass:
1️⃣ Input LayerNorm (FP32) → half.
2️⃣ Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM, producing a tensor
   [B, N, N, 5, H] in FP16.
3️⃣ Apply sigmoid to the three gate slices, optionally mask the
   two projection slices and multiply by the gates – still in FP16.
4️⃣ Rearrange to (BH, N, N) where BH = B·H and launch a tiled Triton
   kernel that computes a batched matrix multiplication
   `out[b·H, i, j] = Σ_k left[b·H, i, k] * right[b·H, j, k]`.
   The kernel works completely in FP16 (tensor‑core dot) with
   FP16 accumulation, which reduces the extra FP32 conversion
   overhead of the reference implementation.
5️⃣ Reshape back to [B, N, N, H] (FP16) → FP32.
6️⃣ Second LayerNorm (FP32) → apply the output gate (FP32) → final
   linear projection back to `dim` (FP16 → FP32).

Only the cubic update (step 4) is performed in a custom Triton kernel;
all other (light‑weight) ops stay in PyTorch for simplicity.

The implementation targets the H100 (Triton 3.3.1) and is heavily
optimised for the typical AlphaFold‑3 configuration
(B∈{1,2}, N∈{128…1024}, H=hidden_dim=128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B half → C half, accumulation in half)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM A @ Bᵀ with FP16 arithmetic.

    A: (BH, M, K), B: (BH, N, K) → C: (BH, M, N)
    """
    pid_h = tl.program_id(axis=2)      # hidden‑slice dimension
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP16 (tensor‑core friendly)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Loop over K dimension in tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- A tile (M × K) -----------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            cache_modifier="ca",
        )          # FP16, cached in L2 → L1

        # ---- Bᵀ tile (K × N) ----------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            cache_modifier="ca",
        )          # FP16

        # ---- Inner product (tensor‑core dot) -------------------------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---- Write result (FP16) -----------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → half)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, dim]  FP32
    x_h = x.half()                               # FP16 for heavy work

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear layers (left_proj, right_proj, left_gate,
    #    right_gate, out_gate) into a single GEMM
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5*H, dim)  FP16

    # Shape after GEMM: [B, N, N, 5*H]  (FP16)
    fused = F.linear(x_h, fused_weight)

    B, N, _, _ = fused.shape

    # Split back into the five tensors: shape [B, N, N, H]
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B, N, N, 5, H]
    left_proj  = fused[..., 0, :]                # [B, N, N, H]
    right_proj = fused[..., 1, :]                # [B, N, N, H]
    left_gate  = torch.sigmoid(fused[..., 2, :]) # [B, N, N, H]  FP16
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :]) # gating for final output

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)  # [B, N, N, 1] FP16
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gate scalings (still FP16)
    # ------------------------------------------------------------------
    left  = left_proj * left_gate    # [B, N, N, H] FP16
    right = right_proj * right_gate  # [B, N, N, H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Rearrange tensors for batched GEMM (BH = B·H)
    # ------------------------------------------------------------------
    # After permute the hidden dimension becomes the second axis:
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N]  FP16
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]  FP16

    # ------------------------------------------------------------------
    # 6️⃣ Triton batched matmul: out = left @ rightᵀ  (FP16)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Tile sizes – empirically good for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one slice per (batch, hidden) pair
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H]  (still FP16)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (FP32) + output‑gate
    # ------------------------------------------------------------------
    out = out_half.float()   # cast to FP32 for numerically stable LN
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                       # [B, N, N, H]  FP32

    out = out * out_gate.float()   # broadcasted output gate (FP32)

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)  # (dim, H) FP16
    out_proj = F.linear(out.half(), to_out_weight_h)             # (B,N,N,dim) FP16
    out_final = out_proj.float()                                 # back to FP32

    return out_final
