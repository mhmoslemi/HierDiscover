"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The operator works on a 4‑D tensor [B, N, N, C] and computes

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are linear projections of the input,
optionally masked, and modulated by sigmoid gates.
The heavy N³ contraction is performed as a large batch of matrix
multiplications (one per hidden channel).  For small sequence lengths
we use a custom Triton kernel; for larger sizes we fall back to
torch.bmm (cuBLAS/Tensor‑Core) which is heavily tuned on H100.

All compute‑intensive kernels are run in FP16 with FP32 accumulators.
The final result is returned in FP32 as required by the model.

The implementation also fuses several linear layers:
`left_proj` + `left_gate` and `right_proj` + `right_gate` are merged
into two concatenated linear operations, reducing memory traffic.
LayerNorms and the output gate are applied in FP16 as well.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for modest sequence lengths
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dims
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)   # merged batch * hidden
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)          # FP32 accumulation

    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication (A @ B) using the custom Triton kernel.
    Arguments must be FP16, contiguous.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "inner dimensions must match"

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Heuristic block sizes
    if max(M, N) <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif max(M, N) <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64  # keep reg usage safe for large N

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_m, num_n, B)   # (tile_m, tile_n, batch*hidden)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # increase parallelism on H100
        num_stages=3,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 1️⃣ unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]      # H in the paper

    # --------------------------------------------------------------
    # 2️⃣ load and cast weights (FP16 where possible)
    # --------------------------------------------------------------
    # LayerNorm on the input
    norm_weight = weights["norm.weight"].half()
    norm_bias   = weights["norm.bias"].half()

    # Concatenate projection + gate weights for left and right streams
    left_proj_w = weights["left_proj.weight"]          # (H, dim)
    left_gate_w = weights["left_gate.weight"]          # (H, dim)
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0).half()   # (2H, dim)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0).half()# (2H, dim)

    # Output gate (dim -> hidden)
    out_gate_w = weights["out_gate.weight"].half()      # (H, dim)

    # Output LayerNorm + final projection
    to_out_norm_w = weights["to_out_norm.weight"].half()
    to_out_norm_b = weights["to_out_norm.bias"].half()
    to_out_w      = weights["to_out.weight"].half()    # (dim, H)

    # --------------------------------------------------------------
    # 3️⃣ First LayerNorm (FP16) – reduces bandwidth for the heavy ops
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)            # [B, N, N, dim] float32
    x = x.half()                                   # FP16

    # --------------------------------------------------------------
    # 4️⃣ Fused linear + gate for left and right streams
    # --------------------------------------------------------------
    # left + left_gate
    left_combined = F.linear(x, left_combined_w)   # [B, N, N, 2H] FP16
    left = left_combined[..., :hidden_dim]         # [B, N, N, H]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])   # [B, N, N, H]

    # right + right_gate
    right_combined = F.linear(x, right_combined_w) # [B, N, N, 2H] FP16
    right = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:])

    # --------------------------------------------------------------
    # 5️⃣ Optional mask (broadcast over hidden dim)
    # --------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(left.dtype).unsqueeze(-1)     # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 6️⃣ Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7️⃣ Contract over the sequence dimension (the N³ work)
    # --------------------------------------------------------------
    # Re‑order to (B, H, N, N) with the K‑dimension as the inner axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    B, H, N, _ = left_perm.shape

    # Collapse batch + hidden into a single leading dimension
    A = left_perm.view(B * H, N, N)      # (B*H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N) # (B*H, K=N, N)

    # Small sequences → custom Triton kernel; otherwise use cuBLAS via torch.bmm
    if N <= 256:   # heuristic threshold
        C = _batched_matmul_fp16(A, B_mat)          # (B*H, N, N) FP16
    else:
        C = torch.bmm(A, B_mat)                     # (B*H, N, N) FP16

    # Restore the original layout: [B, N, N, H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # FP16

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (still FP16) + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)      # FP16

    # Output gate (digested from the original input)
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B, N, N, H] FP16
    out = out * out_gate                                 # FP16

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to model dimension (FP16)
    # --------------------------------------------------------------
    out = F.linear(out, to_out_w)        # [B, N, N, dim] FP16

    # Cast to float32 for downstream consumption
    return out.float()
