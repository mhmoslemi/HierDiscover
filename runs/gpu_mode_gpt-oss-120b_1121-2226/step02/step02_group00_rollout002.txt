"""
TriMul (outgoing) forward pass with Triton‑accelerated N³ update.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ [B, N, N, D] (D = dim).
2. Two bias‑free linear projections → left/right ∈ [B, N, N, H] (H = hidden_dim)
   and three sigmoid gates (left_gate, right_gate, out_gate) from the same X.
3. (Optional) apply binary mask M ∈ [B, N, N] to left/right.
4. Element‑wise gate multiplication:
        left  = left  * left_gate
        right = right * right_gate
5. Core N³ update:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched mat‑mul L @ Rᵀ for each hidden channel.
   We reshape (B,N,N,H) → (B·H, N, N) and launch a Triton kernel that
   tiles the work using (BLOCK_M×BLOCK_N×BLOCK_K) = (256×256×64).
   The kernel uses fp16 inputs, fp32 accumulation and Tensor‑Core
   dot‑product (tl.dot) → maximal H100 throughput.
6. Reshape the result back to [B, N, N, H] (float32).
7. Output layer‑norm over H, element‑wise multiplication with out_gate,
   and final linear projection back to the original dimension.

All heavy linear layers are performed with torch.nn.functional (cuBLAS‑backed)
and the only custom Triton part is the batched mat‑mul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A (M×K) * B (N×K) → C (M×N)
#   - A, B: fp16, C: fp32 (fp16·fp16 → fp32 accumulation)
#   - batch dimension is the 3rd launch axis.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix sizes
    stride_ah, stride_am, stride_ak,    # A strides: batch, row, col(K)
    stride_bh, stride_bn, stride_bk,    # B strides: batch, row, col(K)
    stride_ch, stride_cm, stride_cn,    # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ----- program ids (grid = (Mblocks, Nblocks, batch)) -----
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (B·H)

    # ----- offsets for the current tile ------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # ----- base pointers for this batch ------------------------
    a_batch_ptr = A_ptr + pid_b * stride_ah
    b_batch_ptr = B_ptr + pid_b * stride_bh
    c_batch_ptr = C_ptr + pid_b * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over K dimension -------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # A tile (BLOCK_M × BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        mask_a = (offs_m[:, None] < M) & (k_offset + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # B tile (BLOCK_K × BLOCK_N)
        b_ptrs = b_batch_ptr + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        mask_b = (k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # matrix multiplication for this K‑tile
        acc += tl.dot(a_tile, b_tile)   # (BLOCK_M, BLOCK_N) in fp32

    # ----- write back -------------------------------------------
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    BLOCK_M: int = 256,
    BLOCK_N: int = 256,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched mat‑mul using the Triton kernel above.
    A, B must be fp16 tensors of shape (batch, M, K) and (batch, N, K).
    Returns C with shape (batch, M, N) in fp32.
    """
    assert A.is_cuda and B.is_cuda, "Tensors must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "Kernel expects fp16 inputs"

    batch, M, K = A.shape
    _, N, Kb = B.shape
    assert K == Kb, "Inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # strides (in elements, not bytes)
    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bn, stride_bk = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bn,
        stride_bk,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps → 256 threads per block (good for 256×256 tiles)
        num_stages=4,         # double‑buffering of shared loads
    )
    return C


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, D] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32) with keys:
                  "norm.weight", "norm.bias",
                  "left_proj.weight", "right_proj.weight",
                  "left_gate.weight", "right_gate.weight", "out_gate.weight",
                  "to_out_norm.weight", "to_out_norm.bias",
                  "to_out.weight"
            - config: dict containing at least:
                  "dim", "hidden_dim", and optional "nomask" (bool)

    Returns:
        Tensor of shape [B, N, N, D] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                 # input feature dimension D
    hidden_dim = config["hidden_dim"]   # projection dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  (float32)

    # Cast to fp16 for all heavy linear work (lower bandwidth)
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (all bias‑free)
    # ------------------------------------------------------------------
    # Convert weights once to fp16
    left_proj_w = weights["left_proj.weight"].to(torch.float16)       # (H, D)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)     # (H, D)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)       # (H, D)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)     # (H, D)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)         # (H, D)

    # Projections
    left = F.linear(x_f16, left_proj_w)      # [B, N, N, H] fp16
    right = F.linear(x_f16, right_proj_w)    # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 3) Optional binary mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] -> fp16
        m = mask.to(torch.float16).unsqueeze(-1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_f16, left_gate_w))   # fp16
    right_gate = torch.sigmoid(F.linear(x_f16, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Apply gates element‑wise
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core N³ batched mat‑mul (outgoing update)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape   # B, seq_len, _, hidden_dim

    # Rearrange to (B·H, N, K) where K = N for the mat‑mul.
    #   left_mat[b*H + h, i, k] = left[b, i, k, h]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # fp16

    # Use the tuned Triton kernel (fp16 * fp16 → fp32)
    # Block sizes are chosen to hit Tensor‑Core efficiency on H100.
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat,
        BLOCK_M=256,
        BLOCK_N=256,
        BLOCK_K=64,
    )   # [B·H, N, N] fp32

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    # Multiply by out_gate (cast to fp32 automatically)
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
