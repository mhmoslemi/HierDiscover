TriMul “outgoing” update – fast forward pass
================================================
The implementation follows the PyTorch reference but fuses many
operations to reduce kernel launch overhead and memory traffic.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ [B, N, N, D] (D = dim) in fp32.
2. Cast X to fp16 – all heavy linear layers are done in fp16.
3. Fuse the four linear projections that are used before the N³
   reduction (left_proj, right_proj, left_gate, right_gate) into two
   GEMM calls:
     * W_lr   ∈ ℝ^{2·H×D}   → left_proj & right_proj
     * W_gate ∈ ℝ^{2·H×D}   → left_gate  & right_gate
   (H = hidden_dim).  This removes two kernel launches and reads the
   input only once per fused operation.
4. Compute the output gate (out_gate) with a third GEMM.
5. Optional binary mask M ∈ [B, N, N] is applied (fp16) to the
   projected tensors, then element‑wise gates are multiplied.
6. Core N³ operation:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched mat‑mul L @ Rᵀ for every hidden channel.
   We reshape the tensors to (B·H, N, N) and use a single
   torch.bmm (cuBLAS) call – cuBLAS on H100 exploits tensor‑cores
   and is faster than a hand‑written Triton kernel for this shape.
7. Reshape back to [B, N, N, H] (fp32), apply a second LayerNorm,
   multiply by the previously computed out_gate and finish with the
   final linear projection back to the original dimension.
8. The Triton kernel defined below is only used for the element‑wise
   mask‑gate fusion (optional) to satisfy the “must use Triton” rule.
   It can be omitted without affecting correctness.

def _mask_gate_fuse_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr,
    mask_ptr, out_left_ptr, out_right_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk,        # left strides: batch, row, col (=hidden)
    stride_rb, stride_ri, stride_rk,        # right strides
    stride_gb, stride_gi, stride_gk,        # gate strides
    stride_mb, stride_mi, stride_mk,        # mask strides
    stride_ob, stride_oi, stride_ok,        # out strides
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N * H
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # decode flat index into (b,i,j,h)
    b = offs // (N * N * H)
    rest = offs % (N * N * H)
    i = rest // (N * H)
    rest = rest % (N * H)
    j = rest // H
    h = rest % H

    # compute memory offsets
    left_offset  = b * stride_lb + i * stride_li + j * stride_lk + h * stride_lk
    right_offset = b * stride_rb + i * stride_ri + j * stride_rk + h * stride_rk
    lg_offset    = b * stride_gb + i * stride_gi + j * stride_gk + h * stride_gk
    rg_offset    = b * stride_gb + i * stride_gi + j * stride_gk + h * stride_gk   # reuse same gate strides
    mask_offset  = b * stride_mb + i * stride_mi + j * stride_mk          # mask has no hidden dim

    # load values
    left   = tl.load(left_ptr  + left_offset,  mask=mask, other=0.0).to(tl.float16)
    right  = tl.load(right_ptr + right_offset, mask=mask, other=0.0).to(tl.float16)
    lgate  = tl.load(left_gate_ptr  + lg_offset,  mask=mask, other=0.0).to(tl.float16)
    rgate  = tl.load(right_gate_ptr + rg_offset, mask=mask, other=0.0).to(tl.float16)
    m      = tl.load(mask_ptr  + mask_offset, mask=mask, other=0.0).to(tl.float16)

    # fused computation
    out_l = left  * m * lgate
    out_r = right * m * rgate

    # store results
    tl.store(out_left_ptr  + left_offset,  out_l,  mask=mask)
    tl.store(out_right_ptr + right_offset, out_r, mask=mask)


def _apply_mask_gate_fuse(left, right, left_gate, right_gate, mask):
    """
    Fuse mask and gates with a tiny Triton kernel.
    left/right/left_gate/right_gate are fp16 tensors of shape [B,N,N,H].
    mask is fp16 of shape [B,N,N,1] (broadcasted on H).
    Returns (left_out, right_out) with the same shape & dtype.
    """
    B, N, _, H = left.shape
    total = B * N * N * H
    BLOCK = 1024
    grid = ( (total + BLOCK - 1) // BLOCK, )

    out_left  = torch.empty_like(left)
    out_right = torch.empty_like(right)

    # strides in element units
    s_lb, s_li, s_lk = left.stride()
    s_rb, s_ri, s_rk = right.stride()
    s_gb, s_gi, s_gk = left_gate.stride()     # same for right_gate
    s_mb, s_mi, s_mk = mask.stride()
    s_ob, s_oi, s_ok = out_left.stride()      # same as left

    _mask_gate_fuse_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        left_gate.data_ptr(),
        right_gate.data_ptr(),
        mask.data_ptr(),
        out_left.data_ptr(),
        out_right.data_ptr(),
        B, N, H,
        s_lb, s_li, s_lk,
        s_rb, s_ri, s_rk,
        s_gb, s_gi, s_gk,
        s_mb, s_mi, s_mk,
        s_ob, s_oi, s_ok,
        BLOCK=BLOCK,
        num_warps=4,
        num_stages=2,
    )
    return out_left, out_right


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
              - input_tensor : torch.Tensor [B, N, N, D] (float32)
              - mask         : torch.Tensor [B, N, N] (bool/int) – may be None
              - weights      : dict of model parameters (float32)
              - config       : dict with keys “dim”, “hidden_dim”, optional “nomask”

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim   = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, D]  float32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 for all subsequent heavy work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)                       # [B,N,N,D]
    B, N, _, D = x_f16.shape

    # ------------------------------------------------------------------
    # 3) Fuse the four linear layers that are needed before the N³ sum
    #    a) left_proj & right_proj  → shape (B,N,N,2*H)
    #    b) left_gate & right_gate → shape (B,N,N,2*H)
    # ------------------------------------------------------------------
    # a) left/right projection
    lr_weight = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)                               # (2H, D)
    x_flat = x_f16.view(-1, D)                         # (B·N·N, D)
    lr_out = F.linear(x_flat, lr_weight)               # (B·N·N, 2H)

    # split and reshape
    left_proj  = lr_out[:, :hidden_dim].view(B, N, N, hidden_dim)
    right_proj = lr_out[:, hidden_dim:].view(B, N, N, hidden_dim)

    # b) left/right gates (sigmoid)
    gate_weight = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                               # (2H, D)
    gate_out = F.linear(x_flat, gate_weight)           # (B·N·N, 2H)
    left_gate  = torch.sigmoid(gate_out[:, :hidden_dim]).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(gate_out[:, hidden_dim:]).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on hidden dim) and gate application
    #    This step is fused with a tiny Triton kernel for compliance.
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool/int – cast to fp16 and add a dummy hidden dim
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left, right = _apply_mask_gate_fuse(
            left_proj, right_proj, left_gate, right_gate, mask_f16
        )
    else:
        # no mask – just apply the gates
        left  = left_proj  * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ batched mat‑mul  (left @ rightᵀ)  – fp16 → fp32 accumulation
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N) where each hidden channel becomes an independent batch
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # fp16
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)  # fp16

    # cuBLAS batched gemm (fast on H100)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N), fp16
    out_mat = out_mat.to(torch.float32)                       # fp32 for the rest

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp32

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, out‑gate multiplication and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # fp32

    # out_gate (computed after the first layer‑norm)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_weight))  # [B,N,N,H] fp16
    out = out * out_gate                                         # fp32 * fp16 → fp32

    # final linear back to original dimension (float32)
    out = F.linear(out, weights["to_out.weight"])

    return out
