def _batched_matmul_hidden_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_hidden, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_hidden, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_hidden, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # for (i_block, j_block)
    pid_h = tl.program_id(1) # hidden channel
    pid_b = tl.program_id(2) # batch

    # Determine block indices
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    block_m = pid // num_n
    block_n = pid % num_n

    # Offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base pointer offset for current batch and hidden
    a_base = a_ptr + pid_b * stride_a_batch + pid_h * stride_a_hidden
    b_base = b_ptr + pid_b * stride_b_batch + pid_h * stride_b_hidden
    c_base = c_ptr + pid_b * stride_c_batch + pid_h * stride_c_hidden

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write out
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel – Triton‑accelerated outgoing TriMul (AlphaFold3)

Algorithm
---------
1.  Layer‑norm over the input channel dimension `dim` (FP32).
2.  Cast to FP16 and apply the four linear projections (no bias):
        left_proj, right_proj, left_gate, right_gate, out_gate.
    Gating uses a sigmoid in FP16.  An optional mask (B,N,N) is broadcasted
    and applied before the gates.
3.  Contraction (the N³ step):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched matrix multiplication for every hidden channel `h`.
    The kernel treats the batch dimension `B` and the hidden dimension `H`
    as separate axes (grid‑dim 2 and 1) and tiles the multiplication with
    BLOCK_M/N/K = 64, accumulating in FP32 and storing FP16.  No extra
    contiguous copies of the tensors are required – we pass the original
    strides after a cheap `permute` view.
4.  Permute back to (B,N,N,H) and convert to FP32.
5.  Layer‑norm over the hidden dimension `hidden_dim` (FP32).
6.  Multiply by the output gate (broadcasted, FP32).
7.  Final linear projection back to `dim` (FP32).

Only the N³ contraction is executed in Triton; all other ops leverage
high‑performance PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ B   (A: (B,H,M,K),  B:(B,H,K,N))
#   A, B, C are FP16 (accumulated in FP32)
#   The hidden dimension H and batch dimension B are handled by separate
#   program‑id axes, thus no extra copies are needed.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_hidden_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix sizes (all equal to seq_len)
    stride_a_batch, stride_a_hidden,
    stride_a_i, stride_a_k,
    stride_b_batch, stride_b_hidden,
    stride_b_k, stride_b_j,
    stride_c_batch, stride_c_hidden,
    stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # program ids:
    #   pid0 -> tile over (i, j)   (grid dim 0)
    #   pid1 -> hidden channel    (grid dim 1)
    #   pid2 -> batch index       (grid dim 2)
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid_hidden = tl.program_id(1)
    pid_batch = tl.program_id(2)

    # ------------------------------------------------------------------
    # Decompose pid0 into block coordinates (block_m, block_n)
    # ------------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    block_m = pid0 // num_blocks_n
    block_n = pid0 % num_blocks_n

    # ------------------------------------------------------------------
    # Offsets for the current tile and masks for out‑of‑bounds handling
    # ------------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Base pointers for the selected batch & hidden channel
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_batch * stride_a_batch + pid_hidden * stride_a_hidden
    b_base = b_ptr + pid_batch * stride_b_batch + pid_hidden * stride_b_hidden
    c_base = c_ptr + pid_batch * stride_c_batch + pid_hidden * stride_c_hidden

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension K in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch, hidden, i, k]
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        # B[batch, hidden, k, j]
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_j

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Tensor‑core dot (FP16×FP16→FP32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result (store as FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_hidden(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched FP16 matmul for the TriMul contraction.
    a : (B, H, M, K)   fp16
    b : (B, H, K, N)   fp16
    Returns c : (B, H, M, N)   fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, H, M, K = a.shape
    _, _, K2, N = b.shape
    assert K == K2 and M == K and K == N, "TriMul expects square sequence dims"

    out = torch.empty((B, H, M, N), dtype=torch.float16, device=a.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_blocks_m * num_blocks_n, H, B)  # (i/j tiles, hidden, batch)

    _batched_matmul_hidden_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),  # batch
        a.stride(1),  # hidden
        a.stride(2),  # i (M)
        a.stride(3),  # k (K)
        b.stride(0),  # batch
        b.stride(1),  # hidden
        b.stride(2),  # k (K)
        b.stride(3),  # j (N)
        out.stride(0),  # batch
        out.stride(1),  # hidden
        out.stride(2),  # i (M)
        out.stride(3),  # j (N)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None shape [B, N, N] (bool/float)
        - weights      : dict of tensors (see code for required keys)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Model parameters (cast to the precision we will actually use)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]   # kept FP32 for the final projection

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input `dim` (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for heavy linear work
    # ------------------------------------------------------------------
    x_h = x.half()                       # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)    # [B, N, N, hidden]
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over the hidden channel)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Gating: sigmoid + element‑wise multiplication (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))          # [B, N, N, hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Contraction C = Σ_k left[..., k, :] * right[..., k, :].
    #               Implemented as a batched GEMM over the hidden axis.
    # ------------------------------------------------------------------
    # Permute to [B, H, N, N] without copying data
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1)        # (B, H, k, j)

    # Allocate output tensor in the same layout
    out_perm = torch.empty_like(left_perm)       # (B, H, N, N)  FP16

    # Batched FP16 GEMM using the Triton kernel defined above
    out_perm = _batched_matmul_hidden(left_perm, right_perm)  # (B, H, N, N) FP16

    # ------------------------------------------------------------------
    # 7️⃣ Restore original layout and convert to FP32 for the next LayerNorm
    # ------------------------------------------------------------------
    # shape (B, N, N, H)
    out_fp16 = out_perm.permute(0, 2, 3, 1)
    out = out_fp16.float()

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 9️⃣ Apply the output gate (broadcasted, FP32)
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.float()
    out = out * out_gate_f32

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to `dim` (FP32)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]  FP32

    return out
