"""
TriMul (outgoing) – fast Triton+PyTorch implementation
-----------------------------------------------------

The forward pass is split into four stages:

1️⃣  Input LayerNorm (FP32) → half‑precision.
2️⃣  **Fused linear projection + gates** – the five weight matrices
    (left_proj, right_proj, left_gate, right_gate, out_gate) are
    concatenated into a single (5·H × dim) matrix and multiplied with the
    input in one half‑precision GEMM.  The result is reshaped and the
    three projection tensors and the three gating tensors are obtained
    (sigmoid applied to the gates).  An optional mask is applied before
    gating.

3️⃣  **Cubic update** – a batched half‑precision GEMM
    `out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.  The hidden
    dimension is merged with the batch dimension (BH = B·H) and a
    Triton kernel computes `A @ Bᵀ` for each slice.  The kernel works in
    FP16 with FP32 accumulation and stores the result in FP16.

4️⃣  Second LayerNorm (FP32) + output‑gate → final linear projection back
    to `dim`.  The final tensor has shape `[B, N, N, dim]` and dtype
    ``float32`` as required.

The kernel uses a modest tile size (64×64×64) that fits H100 tensor‑core
register pressure while keeping occupancy high.  For very small
sequences (N≤256) we switch to a 128×128×64 tile to reduce the number of
launches.

Only the cubic update (the most expensive O(N³) part) is written in
Triton; all other operations are performed with highly‑optimized
PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B half → C half, acc in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM `C = A @ Bᵀ` with FP32 accumulation.

    Grid:
        (i_tile, j_tile, batch_hidden) = (axis0, axis1, axis2)
    """
    pid_h = tl.program_id(axis=2)          # batch * hidden slice index
    pid_m = tl.program_id(axis=0)          # i‑tile
    pid_n = tl.program_id(axis=1)          # j‑tile

    # ------------------------------------------------------------------
    # Tile offsets and masks
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension (tiles of size BLOCK_K)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ------- load A tile (M × K) ---------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)               # fp16 → fp32 for the dot

        # ------- load Bᵀ tile (K × N) --------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)

        # ------- inner product ----------------------------------------
        acc += tl.dot(a, b)                # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ------------------------------------------------------------------
    # Store result in FP16
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    # PyTorch LayerNorm works in FP32; we immediately cast to FP16.
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                            # [B, N, N, dim]  float32
    x_h = x.half()                               # half for later GEMMs

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 * hidden_dim outputs)
    # ------------------------------------------------------------------
    # Concatenate the five projection/gate weight matrices once.
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                         # (5*H, dim)

    # Single half‑precision GEMM → [B, N, N, 5*H]
    fused = F.linear(x_h, fused_weight)          # half
    B, N, _, _ = fused.shape

    # Split and apply sigmoid to gate slices.
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H] (half)
    left_proj = fused[..., 0, :]                 # [B,N,N,H]
    right_proj = fused[..., 1, :]                # [B,N,N,H]
    left_gate = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate = torch.sigmoid(fused[..., 4, :])   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → half
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # ------------------------------------------------------------------
    left = left_proj * left_gate                 # [B,N,N,H] half
    right = right_proj * right_gate              # [B,N,N,H] half

    # ------------------------------------------------------------------
    # 5️⃣ Rearrange for batched GEMM (BH = B·H)
    # ------------------------------------------------------------------
    # From (B,N,N,H) to (BH, N, N) where each slice corresponds to a
    # distinct hidden channel.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)      # [BH, N, N] half
    right_flat = right_perm.view(BH, N, N)    # [BH, N, N] half

    # ------------------------------------------------------------------
    # 6️⃣ Triton batched GEMM (A @ Bᵀ) → half output
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Choose tile size depending on N (larger tiles for small N to reduce
    # kernel launch overhead).
    if N <= 256:
        BLOCK_M = BLOCK_N = 128
    else:
        BLOCK_M = BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # batch*hidden slices
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        # Bᵀ: note the transposition – we pass (BH, K, N) strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # optimal for 64×64 tiles on H100
        num_stages=2,       # double‑buffering
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)     # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output gate
    # ------------------------------------------------------------------
    out_fp32 = out_half.float()   # cast back to FP32 for LayerNorm
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                       # [B,N,N,H] float32

    # Apply the output gate (converted to float32)
    out_fp32 = out_fp32 * out_gate.float()

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    # Linear in half‑precision, cast result back to FP32.
    out_proj = F.linear(out_fp32.half(), to_out_weight_h)   # half
    out_final = out_proj.float()                           # float32, shape [B,N,N,dim]

    return out_final
