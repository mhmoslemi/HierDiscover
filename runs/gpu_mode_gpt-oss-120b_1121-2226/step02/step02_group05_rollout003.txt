"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy N³ contraction
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
is computed as a batched matrix multiplication (B·H independent GEMMs)
using a custom Triton kernel that works in FP16 (inputs) with FP32
accumulation for numerical stability.  All other operations (layer‑norms,
linear projections, gating, masking) are performed with PyTorch but in
mixed‑precision (FP16 where possible) to keep bandwidth low.

The implementation follows the reference PyTorch module:
1️⃣ LN on the last dim.
2️⃣ Linear projections + mask‑fusion + sigmoid gating (FP16).
3️⃣ Permute tensors to (B, H, N, N) and view as (B·H, N, N).
4️⃣ Batched GEMM via Triton → (B·H, N, N) (still FP16).
5️⃣ LN on the hidden dimension, apply output gate, final FP32 linear.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute many independent GEMMs in one launch.
    A is (batch, M, K), B is (batch, K, N), C is (batch, M, N).
    All tensors are stored in row‑major order.
    """
    pid_batch = tl.program_id(2)                # which GEMM (merged batch*hidden)
    pid_m = tl.program_id(0)                    # tile row
    pid_n = tl.program_id(1)                    # tile col

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 (H100 can accumulate in TF32, but FP32 is safe)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension.
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile pointer (batch, row=m, col=k)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        # B tile pointer (batch, row=k, col=n)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        # load tiles (apply masks to avoid OOB reads)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate (FP16 * FP16 → FP32)
        tl.dot(a, b, out=acc)

    # ------------------------------------------------------------------
    # Store result tile (cast back to FP16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using the Triton kernel above.
    Arguments:
        a : (B, M, K) torch.float16, contiguous
        b : (B, K, N) torch.float16, contiguous
    Returns:
        c : (B, M, N) torch.float16
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Choose block sizes based on sequence length (M≈N≈seq_len)
    if max(M, N) <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # grid dimensions
    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B,
    )

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data : tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # called H in the paper

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 2️⃣ cast weights to the most appropriate dtype
    # ------------------------------------------------------------------
    # LayerNorm weights stay in FP32 (more stable)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear projection weights (FP16, no bias)
    left_proj_w = weights["left_proj.weight"].to(device).half()
    right_proj_w = weights["right_proj.weight"].to(device).half()
    # Gating linear weights (FP16)
    left_gate_w = weights["left_gate.weight"].to(device).half()
    right_gate_w = weights["right_gate.weight"].to(device).half()
    out_gate_w = weights["out_gate.weight"].to(device).half()
    # Output LayerNorm (FP32) and final projection (FP32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) → half for the remaining heavy ops
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim], dtype float32
    x_h = x.to(torch.float16)  # half precision for projections

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)    # [B, N, N, hidden_dim], fp16
    right = F.linear(x_h, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Apply mask (if present) – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # ensure mask dtype matches half, add trailing dim for broadcasting
        mask_h = mask.to(dtype=left.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))    # used after second LN

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Rearrange tensors for the N³ contraction
    #    left_perm : (B, H, N, N) where axis 2 is i, 3 is k
    #    right_perm: (B, H, N, N) where axis 2 is k, 3 is j
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, i=k]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, j=k]

    # Release memory we no longer need
    del left, right, left_gate, right_gate, x_h

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions so each GEMM is independent
    A = left_perm.view(B * H, N, N)          # (B*H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)     # (B*H, K=N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Batched GEMM with Triton (FP16 input, FP32 accumulation)
    # ------------------------------------------------------------------
    C = _batched_matmul_fp16(A, B_mat)       # (B*H, N, N), fp16

    # Restore original layout: (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (cast to FP32 first)
    # ------------------------------------------------------------------
    out = out.float()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 🔟 Apply the output gate (cast gate to float for stability)
    # ------------------------------------------------------------------
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to original `dim` (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    return out
```
