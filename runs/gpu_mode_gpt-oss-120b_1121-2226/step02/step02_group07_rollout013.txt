"""
Optimized Triton implementation of the outgoing TriMul operation.

The forward pass:
1. LayerNorm over the input channel dimension (float32) → fp16.
2. Fuse the five linear projections (left_proj, right_proj, left_gate, right_gate,
   out_gate) into a single matmul with a concatenated weight matrix (half‑precision).
3. Apply optional mask and sigmoid gates (half).
4. Compute the cubic update
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   for every hidden slice h using a batched half‑precision GEMM (A @ Bᵀ) with
   FP32 accumulation, implemented as a Triton kernel.
5. Second LayerNorm + output gate (float32) and final linear projection back to
   the original feature dimension.

The heavy cubic part is performed entirely in Triton, while the lighter
linear/LN parts use PyTorch’s highly‑optimized kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ah, stride_ai, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ for each hidden slice.
    A: (BH, M, K)   – half
    B: (BH, N, K)   – half (we interpret it as Bᵀ = (BH, K, N))
    C: (BH, M, N)   – half (FP32 accumulator)
    """
    pid_h = tl.program_id(axis=2)      # hidden slice (B * hidden_dim)
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M × K)
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)

        # Load Bᵀ tile (K × N)
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)

        # Compute outer product
        acc += tl.dot(a, b)

    # Store result as half
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] with dtype torch.float32.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.half()

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear layers into a single GEMM
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # (5*H, dim)

    # [B, N, N, 5*H] half
    fused = F.linear(x_h, fused_weight)

    B, N, _, _ = fused.shape
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B,N,N,5,H]

    left_proj = fused[..., 0, :]   # [B,N,N,H]
    right_proj = fused[..., 1, :]
    left_gate = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate = torch.sigmoid(fused[..., 4, :])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)   # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # --------------------------------------------------------------
    left = left_proj * left_gate          # [B,N,N,H] half
    right = right_proj * right_gate       # [B,N,N,H] half

    # --------------------------------------------------------------
    # 5️⃣ Reshape for batched GEMM (BH = B * H)
    # --------------------------------------------------------------
    # Bring hidden dimension to second axis for contiguous layout
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N] half
    right_flat = right_perm.view(BH, N, N) # [BH, N, N] half

    # --------------------------------------------------------------
    # 6️⃣ Triton batched GEMM (A @ Bᵀ)
    # --------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Dynamically choose tile sizes (helps for small N)
    if N <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                    # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),      # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (half)
    # --------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)  # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + out_gate
    # --------------------------------------------------------------
    out_fp32 = out_half.float()
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp32 = out_fp32 * out_gate.float()

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to dim (half → float32)
    # --------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out_fp32.half(), to_out_weight_h)  # half
    out_final = out_proj.float()

    return out_final

"""
Optimized Triton implementation of the outgoing TriMul operation.

The forward pass consists of:
1️⃣ Input LayerNorm (float32) → half‑precision.
2️⃣ Fuse the five linear layers (left_proj, right_proj, left_gate,
   right_gate, out_gate) into a single GEMM with a concatenated weight
   matrix (half).  The result is reshaped and split into projection
   tensors and gating tensors; the gates are passed through a sigmoid.
3️⃣ Optional pair‑wise mask is broadcast and applied to the projection
   tensors (still half).
4️⃣ Apply the element‑wise gates (half).
5️⃣ Compute the cubic update
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   for every hidden slice h using a batched half‑precision matrix
   multiply (A @ Bᵀ) with FP32 accumulation, implemented as a Triton
   kernel.
6️⃣ Second LayerNorm (float32) + output gate.
7️⃣ Final linear projection back to the original feature dimension.

All heavy linear and cubic work is performed in FP16, while the two
LayerNorms and the final projection use FP32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ   (A, B ∈ fp16, acc ∈ fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all equal to seq_len)
    stride_ah, stride_ai, stride_ak,   # A strides: (batch*H, i, k)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides: (batch*H, k, j)
    stride_ch, stride_ci, stride_cj,   # C strides: (batch*H, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ Bᵀ for every hidden slice."""
    pid_h = tl.program_id(axis=2)      # (batch * hidden) index
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile (M × K)
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)  # fp16 → fp32

        # Load Bᵀ tile (K × N)
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)  # fp16 → fp32

        # Inner product
        acc += tl.dot(a, b)   # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # Store result as fp16
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype torch.float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    x_h = x.half()                       # half‑precision for the heavy part

    # ------------------------------------------------------------------
    # 2️⃣ Fuse five linear layers into a single GEMM
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                  # (5*H, dim)  in half

    # Linear: (B,N,N,dim) @ (5*H, dim).T → (B,N,N,5*H)
    fused = F.linear(x_h, fused_weight)  # half
    B, N, _, _ = fused.shape

    # reshape to separate the five heads
    fused = fused.view(B, N, N, 5, hidden_dim)   # [B, N, N, 5, H]

    left_proj  = fused[..., 0, :]                # [B, N, N, H]
    right_proj = fused[..., 1, :]
    left_gate  = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)   # [B, N, N, 1]
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # ------------------------------------------------------------------
    left  = left_proj * left_gate      # [B, N, N, H]  half
    right = right_proj * right_gate   # [B, N, N, H]  half

    # ------------------------------------------------------------------
    # 5️⃣ Reshape for batched GEMM (BH = B·H)
    # ------------------------------------------------------------------
    # bring hidden dimension next to batch for a contiguous layout
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N]  half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]  half

    # ------------------------------------------------------------------
    # 6️⃣ Triton batched GEMM (A @ Bᵀ)
    # ------------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # choose tile sizes that fit the problem size (helps small N)
    if N <= 128:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32
    elif N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one program per (batch, hidden) slice
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),      # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # ------------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)   # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second LayerNorm + output gate (float32)
    # ------------------------------------------------------------------
    out_fp32 = out_half.float()
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_fp32 = out_fp32 * out_gate.float()   # apply output gating

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim` (half → float32)
    # ------------------------------------------------------------------
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out_fp32.half(), to_out_weight_h)  # half
    out_final = out_proj.float()                           # float32

    return out_final
