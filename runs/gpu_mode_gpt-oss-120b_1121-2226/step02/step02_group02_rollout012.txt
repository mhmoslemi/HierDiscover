@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_h = tl.program_id(1)
    pid_m = tl.program_id(2)  # row tile i
    pid_n = tl.program_id(3)  # col tile j

    # Compute offsets for this batch and hidden channel
    base_l = left_ptr + pid_b * stride_lb + pid_h * stride_lh
    base_r = right_ptr + pid_b * stride_rb + pid_h * stride_rh
    base_o = out_ptr + pid_b * stride_ob + pid_h * stride_oh

    # Compute tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-bounds rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        # left[i,k] located at base_l + offs_i[:,None]*stride_li + offs_k[None,:]*stride_lk
        a_ptrs = base_l + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        # load with mask
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile: need B tile of shape (BLOCK_N, BLOCK_K) as transposed

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Compute offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch/h
    base_l = left_ptr + b * stride_lb + h * stride_lh
    base_r = right_ptr + b * stride_rb + h * stride_rh
    base_o = out_ptr + b * stride_ob + h * stride_oh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K)
        a_ptrs = base_l + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile (K x N): note we need K as rows and N as columns
        b_ptrs = base_r + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Store result
    c_ptrs = base_o + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm
---------
1. Input LayerNorm (float32).
2. Cast to fp16 and fuse the two projection + gate linear layers:
      [proj , gate] = Linear(x)   → shape [..., 2*hidden]
   Split and apply sigmoid to the gate part.
3. Optional mask: a tiny Triton kernel converts the fp32 mask to fp16 and
   broadcasts it over the hidden dimension.
4. Element‑wise multiplication: left = proj_left * gate_left,
   right = proj_right * gate_right.
5. Core N³ reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   Implemented by a custom Triton kernel that works on the native
   (B,N,N,H) layout – no expensive permute / copy.
   The kernel accumulates in FP32 while reading FP16 inputs, thus
   exploiting H100 Tensor‑Cores.
6. Output LayerNorm (float32) → gating → final Linear (float32).
7. Return tensor of shape [B, N, N, dim] (float32).

Only part of the computation is required to be in Triton; the
projection / gating steps stay in PyTorch (cuBLAS) while the heavy
N³ reduction runs in a handcrafted Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Tiny kernel: fp32 → fp16 (used for the mask)
# ---------------------------------------------------------------------------
@triton.jit
def _mask_fp32_to_fp16_kernel(
    src_ptr,  # fp32 mask
    dst_ptr,  # fp16 mask
    numel,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel
    val_f32 = tl.load(src_ptr + offs, mask=mask, other=0.0, dtype=tl.float32)
    tl.store(dst_ptr + offs, tl.cast(val_f32, tl.float16), mask=mask)


# ---------------------------------------------------------------------------
# Core N³ reduction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
#   left/right : fp16 tensor of shape (B,N,N,H)
#   out        : fp32 tensor of shape (B,N,N,H)
# ---------------------------------------------------------------------------
@triton.jit
def _tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)   # flattened batch*hidden index
    pid_m = tl.program_id(1)    # tile over i (rows)
    pid_n = tl.program_id(2)    # tile over j (cols)

    # ---------------------------------------------------------------
    # Recover batch and hidden indices from the flattened id
    # ---------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ---------------------------------------------------------------
    # Offsets for the i‑ and j‑tiles
    # ---------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---------------------------------------------------------------
    # Base pointers for this (b, h) pair
    # ---------------------------------------------------------------
    base_l = left_ptr + b * stride_lb + h * stride_lh
    base_r = right_ptr + b * stride_rb + h * stride_rh
    base_o = out_ptr + b * stride_ob + h * stride_oh

    # ---------------------------------------------------------------
    # Accumulator (FP32)
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over the reduction axis k
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]  : shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_l + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # right[j,k] : we need (BLOCK_K, BLOCK_N) for tl.dot,
        #              i.e. load with k as the row dimension
        b_ptrs = base_r + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate the dot‑product
        acc += tl.dot(a, b)          # FP16·FP16 → FP32

    # ---------------------------------------------------------------
    # Store the result tile
    # ---------------------------------------------------------------
    c_ptrs = base_o + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ---------------------------------------------------------------------------
# Entrypoint required by the evaluation harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [bs, seq_len, seq_len, dim] (float32)
        - mask_tensor  : torch.Tensor [bs, seq_len, seq_len] (float32)
        - weights      : dict of model weights (float32 tensors)
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack and basic constants
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # Cast once to fp16 for all heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)                     # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # Fuse proj + gate linear layers (two concatenated weight matrices)
    # ------------------------------------------------------------------
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()                                      # (2*H, dim) fp16
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()                                      # (2*H, dim) fp16

    # Combined linear gives [proj , gate] in one GEMM
    left_combined = F.linear(x_h, left_combined_w)   # [B,N,N,2*H] fp16
    right_combined = F.linear(x_h, right_combined_w) # [B,N,N,2*H] fp16

    # Split back into projection and gate tensors
    left_proj = left_combined[..., :hidden]           # [B,N,N,H] fp16
    left_gate = torch.sigmoid(left_combined[..., hidden:])   # fp16

    right_proj = right_combined[..., :hidden]         # [B,N,N,H] fp16
    right_gate = torch.sigmoid(right_combined[..., hidden:]) # fp16

    # ------------------------------------------------------------------
    # Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # Convert fp32 mask → fp16 using a tiny Triton kernel
        mask_fp16 = torch.empty_like(mask, dtype=torch.float16, device=device)
        BLOCK_SZ = 1024
        grid = ((mask.numel() + BLOCK_SZ - 1) // BLOCK_SZ,)
        _mask_fp32_to_fp16_kernel[grid](
            mask.view(-1),
            mask_fp16.view(-1),
            mask.numel(),
            BLOCK_SIZE=BLOCK_SZ,
        )
        mask_broadcast = mask_fp16.unsqueeze(-1)  # [B,N,N,1]

        # Apply mask only to the *projected* tensors (as in the reference)
        left_proj = left_proj * mask_broadcast
        right_proj = right_proj * mask_broadcast
    # ------------------------------------------------------------------
    # Apply gating (still in fp16)
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # [B,N,N,H] fp16
    right = right_proj * right_gate       # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # Out‑gate (separate linear, still fp16)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # Core N³ operation – Triton kernel
    # ------------------------------------------------------------------
    # Allocate output buffer (FP32)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Strides in element units
    l_str = left.stride()   # (b, i, k, h)
    r_str = right.stride()
    o_str = out.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        bs * hidden,                               # flattened (b, h)
        (N + BLOCK_M - 1) // BLOCK_M,              # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,              # j‑tiles
    )
    _tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        hidden,
        # left strides
        l_str[0],
        l_str[1],
        l_str[2],
        l_str[3],
        # right strides
        r_str[0],
        r_str[1],   # j‑stride (= i‑stride)
        r_str[2],
        r_str[3],
        # out strides
        o_str[0],
        o_str[1],
        o_str[2],
        o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Output LayerNorm (float32) + gating + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B,N,N,H] fp32

    out = out * out_gate.to(torch.float32)   # element‑wise gate (fp32)

    # Final linear back to `dim` (float32)
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim] fp32

    return out
