"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

Algorithm
---------
1️⃣ Input LayerNorm (float32) → half‑precision.
2️⃣ **Fused linear**: concatenate the five projection / gate matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) into a single
   weight of shape (5·H, C) and apply one half‑precision GEMM.
   The result is reshaped → (B, N, N, 5, H) and split.
3️⃣ Optional mask is broadcasted on the hidden dimension and applied
   to ``left_proj`` and ``right_proj``.
4️⃣ Apply sigmoid gates (still half) and element‑wise gating.
5️⃣ Rearrange tensors so that each hidden slice becomes an independent batch:
   ``[B, H, N, N] → [B·H, N, N]``.
6️⃣ **Cubic update** – a batched matmul ``out = left @ rightᵀ`` for every
   hidden slice.  Implemented with a custom Triton kernel that:
   * loads tiles of A (M×K) and Bᵀ (K×N) in FP16,
   * accumulates in FP32 (to avoid overflow),
   * writes the result in FP16.
   Tile sizes (M,N,K) = (128,128,64) and a 3‑stage pipeline with 8 warps
   give good utilization on H100.
7️⃣ Reshape back to ``[B, N, N, H]`` (half).
8️⃣ Second LayerNorm (float32), multiply by the output gate, cast to half.
9️⃣ Final linear projection back to the model dimension C (half → float32).

Only the cubic update is written in Triton; all other steps use
high‑performance PyTorch kernels (LayerNorm, fused Linear, etc.).
The final tensor is returned as ``float32`` as required by the model.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ  (A,B half → C half, accumulate FP32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr,            # LEFT  [BH, M, K]  (half)
    b_ptr,            # RIGHT [BH, K, N]  (half) – we load it as Bᵀ
    c_ptr,            # OUT   [BH, M, N]  (half)
    M, N, K,          # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ
    stride_ch, stride_ci, stride_cj,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(axis=2)      # hidden‑batch index (BH)
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # ------------------------------------------------------------------
    # tile offsets & boundary masks
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑loop
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- A tile (M × K) -----------------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)   # FP16 → FP32

        # ---- Bᵀ tile (K × N) ----------------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)   # FP16 → FP32

        # ---- inner product -------------------------------------------
        acc += tl.dot(a, b)    # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ------------------------------------------------------------------
    # write output (FP16) – saves bandwidth
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # --------------------------------------------------------------
    # unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                          # [B, N, N, dim]  float32
    x_h = x.half()                             # half precision for heavy work

    # --------------------------------------------------------------
    # 2️⃣ Fused linear stage (5 × hidden_dim outputs)
    # --------------------------------------------------------------
    # concatenate weight matrices: (5·H, C)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (H, C)
            weights["right_proj.weight"],  # (H, C)
            weights["left_gate.weight"],   # (H, C)
            weights["right_gate.weight"],  # (H, C)
            weights["out_gate.weight"],    # (H, C)
        ],
        dim=0,
    ).to(torch.float16)

    # single GEMM: [B, N, N, 5·H] (half)
    fused = F.linear(x_h, fused_weight)        # half
    B, N, _, _ = fused.shape

    # reshape & split
    fused = fused.view(B, N, N, 5, hidden_dim)  # [B,N,N,5,H] half
    left_proj = fused[..., 0, :]                # [B,N,N,H]
    right_proj = fused[..., 1, :]               # [B,N,N,H]
    left_gate = torch.sigmoid(fused[..., 2, :])
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate = torch.sigmoid(fused[..., 4, :])  # [B,N,N,H]

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present) – broadcast on hidden dim
    # --------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)   # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 4️⃣ Apply gates (still half)
    # --------------------------------------------------------------
    left = left_proj * left_gate          # [B,N,N,H] half
    right = right_proj * right_gate       # [B,N,N,H] half

    # --------------------------------------------------------------
    # 5️⃣ Rearrange for batched GEMM (BH = B·H)
    # --------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B,H,N,N]

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)    # [BH, N, N] half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N] half

    # --------------------------------------------------------------
    # 6️⃣ Triton batched matmul A @ Bᵀ  →  out_flat [BH, N, N] (half)
    # --------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one program per (batch, hidden) slice
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (half)
    # --------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)   # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output gate
    # --------------------------------------------------------------
    out_fp32 = F.layer_norm(
        out_half.float(),
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                                            # [B,N,N,H] float32

    # cast back to half for the final linear and apply out_gate in half
    out_fp16 = out_fp32.half()
    out_fp16 = out_fp16 * out_gate.half()        # gating

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out_proj = F.linear(out_fp16, to_out_w)      # half → [B,N,N,dim] half
    out_final = out_proj.float()                 # back to float32

    return out_final
