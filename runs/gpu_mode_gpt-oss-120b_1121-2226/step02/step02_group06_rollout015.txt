    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

# Kernel for batched matmul without memory copy
@triton.jit
def _batched_matmul_no_copy_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ab, stride_am, stride_ak, stride_ah,
    stride_bb, stride_bk, stride_bn, stride_bh,
    stride_cb, stride_cm, stride_cn, stride_ch,
    H: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute tiling indices
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_m * num_n)      # combined (batch, hidden)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # unfold combined batch into (b, h)
    b = batch_idx // H
    h = batch_idx % H

    # base pointers for the current (b, h)
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_bb + h * stride_bh
    c_base = c_ptr + b * stride_cb + h * stride_ch

    # tile offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: rows=i, cols=k
        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        # B tile: rows=k, cols=j  (right matrix already transposed in memory)

def _batched_matmul_fp16_no_copy(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched FP16 matrix multiplication C = A @ B where
        A: (B, N, N, H)  with layout (batch, i, k, h)
        B: (B, N, N, H)  with layout (batch, j, k, h)  (treated as K×N)

        C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
    No memory copies or permutes are performed; the kernel works directly on
    the original strides.
    """

    B, N, N2, H = left.shape
    assert N == N2 and left.shape == right.shape

    BLOCK_N = 64
    BLOCK_K = 64

    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # projection & gate weights (cast to fp16 for the heavy work)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # second‑stage norm & output projection (keep in fp32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]   # fp32

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the input `dim` (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to fp16 for the projected tensors
    # ------------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers (no bias) into a single matmul.
    #    Result layout: [B, N, N, 5·hidden_dim]
    # ------------------------------------------------------------------
    # concatenate the weight matrices along the output‑feature dimension
    fused_weight = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0
    ).contiguous()                     # shape (5*hidden_dim, dim)

    # single linear projection
    proj = F.linear(x_h, fused_weight)  # [B, N, N, 5·hidden_dim]  fp16

    # split the fused tensor back into the five components
    left, right, left_gate, right_gate, out_gate = torch.split(proj, hidden_dim, dim=-1)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched N³ contraction C[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #     Implemented with a custom Triton kernel that works on the native
    #     (B,N,N,H) layout – no permute or view copies required.
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16_no_copy(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (convert to fp32 for stability)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by the output gate (broadcasted)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]  fp32

    return out

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # second‑stage norm & output projection (keep FP32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]          # FP32, bias=False

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm  (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.half()                       # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Fuse the five linear layers into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [left_proj_w, right_proj_w, left_gate_w, right_gate_w, out_gate_w],
        dim=0
    ).contiguous()                       # (5*hidden_dim, dim)

    proj = F.linear(x_h, fused_w)        # [B, N, N, 5·hidden_dim]  fp16
    left, right, left_gate, right_gate, out_gate = torch.split(
        proj, hidden_dim, dim=-1
    )

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5️⃣ Apply sigmoid gates (still fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Batched N³ contraction using the custom Triton kernel
    # ------------------------------------------------------------------
    out = _batched_matmul_fp16_no_copy(left, right)   # [B, N, N, hidden_dim] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm (convert to fp32 for stability)
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 8️⃣ Multiply by the output gate (broadcasted)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 9️⃣ Final projection back to `dim` (FP32)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)       # [B, N, N, dim]  fp32

    return out
```
