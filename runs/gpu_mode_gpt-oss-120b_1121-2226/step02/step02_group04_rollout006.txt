@triton.jit
def tri_mul_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ak,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          stride_batch,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)  # batch dim
    pid_m = tl.program_id(1)  # row block
    pid_n = tl.program_id(2)  # col block

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute pointer bases for this batch
    A_base = A_ptr + pid_b * stride_batch
    B_base = B_ptr + pid_b * stride_batch
    C_base = C_ptr + pid_b * stride_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M,K)
        a = tl.load(A_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # load B tile (K,N)
        b = tl.load(B_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product (fp16 -> fp32 accumulation)
        acc += tl.dot(a, b)

    # write output
    c = acc.to(tl.float16)
    tl.store(C_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
             c,
             mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – highly‑optimized Triton+PyTorch implementation.

Algorithm
---------
1. Layer‑norm the input (FP32) → cast to FP16.
2. Fuse the five linear projections (left‑proj, left‑gate, right‑proj,
   right‑gate, out‑gate) into a single GEMM (weight shape 5·H × D).
3. Apply sigmoid to the three gate tensors.
4. Fuse mask broadcasting, element‑wise masking and per‑head gates with a tiny
   Triton kernel (`_mask_gate_fuse_kernel`).  The kernel works on the flattened
   view of the tensors and writes the masked‑gated left/right tensors in FP16.
5. Reshape to (B·H, N, N) and compute the costly N³ contraction with a custom
   batched GEMM kernel (`_tri_mul_matmul_kernel`).  The kernel performs a
   blocked matrix multiplication for each (b,h) pair, accumulating in FP32
   (Tensor‑Core friendly) and writing back in FP16.
6. Post‑contraction LayerNorm (FP32), multiply by the output‑gate,
   and a final linear projection back to the original dimension (FP16→FP32).

All heavy work is done in the two Triton kernels; the remaining
operations are cheap PyTorch calls.  The final tensor is returned in
float32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,                 # left_proj  (fp16)
    right_ptr,                # right_proj (fp16)
    left_gate_ptr,            # left_gate  (fp16)
    right_gate_ptr,           # right_gate (fp16)
    mask_ptr,                 # mask (fp16) of size B·N·N
    out_left_ptr, out_right_ptr,  # fused results (fp16)
    total: tl.constexpr,          # total number of elements = B·N·N·H
    hidden: tl.constexpr,         # hidden dimension H
    BLOCK: tl.constexpr,          # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # load values
    l_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # each `hidden` consecutive entries share the same mask element
    mask_idx = offs // hidden
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # fuse mask & gates
    lo = l_val * l_gate * m_val
    ro = r_val * r_gate * m_val

    tl.store(out_left_ptr + offs, lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,          # pointers (fp16)
    M, N, K,                      # matrix sizes (all = seq_len)
    stride_am, stride_ak,         # strides for A (row, col)
    stride_bk, stride_bn,         # strides for B (row=K, col=N)
    stride_cm, stride_cn,         # strides for C (row, col)
    stride_batch,                 # stride to jump to next (b, h) slice
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched FP16 GEMM with FP32 accumulation.
    A : (M, K)   B : (K, N)   C : (M, N)
    Batch dimension is encoded in `stride_batch`.
    """
    pid_b = tl.program_id(0)                     # batch * hidden index
    pid_m = tl.program_id(1)                     # block row
    pid_n = tl.program_id(2)                     # block col

    # base pointers for this batch slice
    A_base = A_ptr + pid_b * stride_batch
    B_base = B_ptr + pid_b * stride_batch
    C_base = C_ptr + pid_b * stride_batch

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load tiles (A: MxK, B: KxN)
        a = tl.load(
            A_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            B_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # fp16 dot accumulation in fp32
        acc += tl.dot(a, b)

    # write back
    c = acc.to(tl.float16)
    tl.store(
        C_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (float32) or None
            - weights      : dict with model parameters (see code)
            - config       : dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # model weights (convert to fp16 where beneficial)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]            # (dim,)
    norm_bias   = weights["norm.bias"]              # (dim,)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    to_out_norm_w = weights["to_out_norm.weight"]    # (H,)
    to_out_norm_b = weights["to_out_norm.bias"]      # (H,)

    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (D, H)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32) → FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                                   # [B, N, N, D] float32
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. One fused linear for all projections & gates (5·H output features)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )                                   # (5*H, D)

    fused = F.linear(x, fused_weight)   # [B, N, N, 5*H] (FP16)

    B, N, _, _ = fused.shape
    # split fused tensor
    off = 0
    left_proj = fused[..., off : off + hidden_dim];            off += hidden_dim
    left_gate = torch.sigmoid(fused[..., off : off + hidden_dim]); off += hidden_dim
    right_proj = fused[..., off : off + hidden_dim];           off += hidden_dim
    right_gate = torch.sigmoid(fused[..., off : off + hidden_dim]); off += hidden_dim
    out_gate = torch.sigmoid(fused[..., off : off + hidden_dim]);   # (B,N,N,H) FP16

    # ------------------------------------------------------------------
    # 3. Fuse mask, left/right gates via Triton
    # ------------------------------------------------------------------
    if mask is None:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=input_tensor.device)
    else:
        mask_flat = mask.to(torch.float16).reshape(-1)  # (B·N·N,)

    # flatten for kernel
    left_proj_f  = left_proj.reshape(-1)
    right_proj_f = right_proj.reshape(-1)
    left_gate_f  = left_gate.reshape(-1)
    right_gate_f = right_gate.reshape(-1)

    left_out_f  = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elems = left_proj_f.shape[0]          # B·N·N·H
    BLOCK_SIZE = 1024

    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    _mask_gate_fuse_kernel[grid](
        left_proj_f,
        right_proj_f,
        left_gate_f,
        right_gate_f,
        mask_flat,
        left_out_f,
        right_out_f,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # reshape back to [B, N, N, H] (FP16)
    left  = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. N³ contraction with a custom Triton batched GEMM
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) – each slice is a standard GEMM (A @ Bᵀ)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                     # batch*head dimension
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _tri_mul_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,
        left_mat.stride(1), left_mat.stride(2),   # stride_am, stride_ak
        right_mat.stride(1), right_mat.stride(2), # stride_bk, stride_bn
        out_mat.stride(1), out_mat.stride(2),     # stride_cm, stride_cn
        left_mat.stride(0),                       # batch stride (same for A,B,C)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H] (FP16)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Post‑contraction LayerNorm (FP32) + output gate
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)   # broadcast over hidden dim

    # ------------------------------------------------------------------
    # 6. Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)                # (B, N, N, D) FP16
    out = out.to(torch.float32)

    return out
