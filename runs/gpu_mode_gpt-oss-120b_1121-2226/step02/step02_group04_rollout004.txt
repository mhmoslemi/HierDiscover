"""
TriMul (outgoing) – highly‑optimized Triton + PyTorch implementation
===================================================================

Algorithm
---------
1. **Layer‑norm (input)** – FP32 → cast to FP16.
2. **Fused projection** – The four linear maps (left/right projection,
   left/right gate) and the output gate are fused into a single matmul
   `W_fused ∈ ℝ^{5·H×D}` → produce `[B,N,N,5·H]`.  The three gate tensors are
   passed through a sigmoid.
3. **Mask & gate fusion** – A tiny Triton kernel multiplies the per‑position
   mask, left‑gate, right‑gate with the projected tensors, writing back the
   fused `left` and `right` tensors (all FP16).
4. **Batched N³ contraction** – The core “tri‑mul” is a batched matrix
   multiplication:
        out[b,i,j,h] = Σ_k left[b,i,k,h] · right[b,j,k,h] .
   We implement it as a custom Triton kernel that handles the full batch
   (`B·H` matrices) in one launch, performing the computation in FP16
   while accumulating in FP32 (Tensor‑Core friendly).  This replaces the
   slower `torch.bmm` call.
5. **Post‑contraction Layer‑norm, output gate** – FP16 → FP32 for the norm,
   element‑wise multiplication with the output gate, then back to FP16.
6. **Final projection** – `to_out` linear layer (FP16) and cast the result
   to float32 (the required output type).

All heavy arithmetic is kept in FP16/Tensor‑Core precision; only the two
LayerNorms and the final cast use FP32.  The implementation works for both
masked and un‑masked inputs and for all shapes listed in the test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1) Element‑wise kernel: mask * gate * projection (left/right)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, right_ptr,          # projected tensors (fp16)
    left_gate_ptr, right_gate_ptr,# gate tensors (fp16)
    mask_ptr,                     # mask (fp16) shape B·N·N
    out_left_ptr, out_right_ptr,  # destinations (fp16)
    total: tl.constexpr,          # B·N·N·H
    hidden: tl.constexpr,         # hidden dim H
    BLOCK: tl.constexpr,          # threads per program
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # load tensors (masked loads return 0.0 for out‑of‑bounds)
    l_val = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs, mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # each `hidden` consecutive entries share one mask element
    mask_idx = offs // hidden
    m_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # fused multiplication
    lo = l_val * l_gate * m_val
    ro = r_val * r_gate * m_val

    tl.store(out_left_ptr + offs, lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel (left @ rightᵀ) – computes the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched matrix multiplication C = A @ B  (A: (B,M,K), B: (B,K,N))."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the M/N tiles
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guard against out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to the beginning of the current batch
    a_batch_ptr = a_ptr + batch_id * stride_a_batch
    b_batch_ptr = b_ptr + batch_id * stride_b_batch
    c_batch_ptr = c_ptr + batch_id * stride_c_batch

    # accumulation buffer (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr \
               + (offs_m[:, None] * stride_a_m) \
               + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)   # FP16 load → auto‑casts to FP32 in dot

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch_ptr \
               + (offs_k[:, None] * stride_b_k) \
               + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # matrix product on the tile
        acc += tl.dot(a, b)

    # write back result in FP16
    c = acc.to(tl.float16)
    c_ptrs = c_batch_ptr \
           + (offs_m[:, None] * stride_c_m) \
           + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# 3) Entry point – forward pass of the outgoing TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of model parameters (see key list below)
            - config       : dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # weight tensors (convert to FP16 where appropriate)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]            # (dim,)
    norm_bias   = weights["norm.bias"]              # (dim,)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    to_out_norm_w = weights["to_out_norm.weight"]    # (H,)
    to_out_norm_b = weights["to_out_norm.bias"]      # (H,)

    to_out_w      = weights["to_out.weight"].to(torch.float16)      # (D, H)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32) → cast to FP16
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    x = x.to(torch.float16)          # [B,N,N,D]  FP16

    # ------------------------------------------------------------------
    # 2) Fuse all linear projections & gates into one matmul (5·H outputs)
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w,
         right_gate_w, out_gate_w],
        dim=0,
    )                                 # (5*H, D)

    fused = F.linear(x, fused_weight)   # [B,N,N,5*H]  FP16

    B, N, _, _ = fused.shape
    # split the fused tensor
    off = 0
    left_proj   = fused[..., off:off + hidden_dim]; off += hidden_dim
    left_gate   = torch.sigmoid(fused[..., off:off + hidden_dim]); off += hidden_dim
    right_proj  = fused[..., off:off + hidden_dim]; off += hidden_dim
    right_gate  = torch.sigmoid(fused[..., off:off + hidden_dim]); off += hidden_dim
    out_gate    = torch.sigmoid(fused[..., off:off + hidden_dim]);           # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Fuse mask, left/right gates via Triton kernel
    # ------------------------------------------------------------------
    if mask is None:
        mask_flat = torch.ones(B * N * N, dtype=torch.float16, device=device)
    else:
        mask_flat = mask.to(torch.float16).reshape(-1)   # [B·N·N]

    # flatten for the kernel (contiguous layout)
    left_proj_f   = left_proj.reshape(-1).contiguous()
    right_proj_f  = right_proj.reshape(-1).contiguous()
    left_gate_f   = left_gate.reshape(-1).contiguous()
    right_gate_f  = right_gate.reshape(-1).contiguous()

    left_out_f  = torch.empty_like(left_proj_f)
    right_out_f = torch.empty_like(right_proj_f)

    total_elems = left_proj_f.shape[0]                 # B·N·N·H
    BLOCK_SIZE = 1024                                 # threads per program
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    _mask_gate_fuse_kernel[grid](
        left_proj_f,
        right_proj_f,
        left_gate_f,
        right_gate_f,
        mask_flat,
        left_out_f,
        right_out_f,
        total=total_elems,
        hidden=hidden_dim,
        BLOCK=BLOCK_SIZE,
    )

    # reshape back to [B,N,N,H]
    left  = left_out_f.view(B, N, N, hidden_dim)
    right = right_out_f.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4) N³ contraction via a custom batched GEMM kernel (Tensor‑Core)
    # ------------------------------------------------------------------
    # re‑order to (batch, M, K) / (batch, K, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B,H,N,N] (already transposed)

    batch_dim = B * hidden_dim
    left_flat  = left_mat.view(batch_dim, N, N).contiguous()
    right_flat = right_mat.view(batch_dim, N, N).contiguous()
    out_flat   = torch.empty_like(left_flat)            # [B*H, N, N] FP16

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        batch_dim,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B,N,N,H]
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)          # [B,N,N,H]
        .contiguous()
    )                                 # FP16

    # ------------------------------------------------------------------
    # 5) Post‑contraction LayerNorm (FP32) + output gate
    # ------------------------------------------------------------------
    out = out.to(torch.float32)
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate.to(torch.float32)   # element‑wise scaling

    # ------------------------------------------------------------------
    # 6) Final linear projection back to original dimension D
    # ------------------------------------------------------------------
    out = out.to(torch.float16)
    out = F.linear(out, to_out_w)            # (D, H) weight
    out = out.to(torch.float32)

    return out
```

**Explanation of speed‑up points**

* **Fused linear projection** – one matrix multiplication replaces four separate
  `nn.Linear` calls, drastically reducing memory traffic.
* **Mask‑gate fusion kernel** – a single Triton kernel performs the mask
  broadcast and two gate multiplications while staying in FP16.
* **Custom batched GEMM** – the N³ contraction is executed by a dedicated
  Triton kernel that processes all `B·H` matrices in one launch, keeping the
  data in registers/shared memory and using Tensor‑Core `dot` (FP16→FP32).
  This removes the overhead of launching many small cuBLAS kernels and
  improves occupancy on H100.
* **Mixed‑precision workflow** – almost all heavy arithmetic runs in FP16,
  only the two LayerNorms (which need accurate statistics) run in FP32.
* **Minimal data reshaping** – tensors are kept contiguous and only a few
  cheap `permute/view` operations are performed.

The kernel respects the required API, works for masked/unmasked inputs, and
produces the exact same floating‑point results (up to the usual FP16 rounding)
as the reference PyTorch implementation.
