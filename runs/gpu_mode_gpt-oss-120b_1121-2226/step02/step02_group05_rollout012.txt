def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # Normalize
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)  # float32
    x_h = x.to(torch.float16)
    # Projections
    left_proj = F.linear(x_h, weights["left_proj.weight"].half())  # half
    right_proj = F.linear(x_h, weights["right_proj.weight"].half())
    # Gates
    left_gate = torch.sigmoid(F.linear(x_h, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_h, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_h, weights["out_gate.weight"].half()))
    # Apply mask
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h
    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate
    # Permute for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # reshape batch*hidden
    B, H, N, _ = left_perm.shape
    A = left_perm.view(B * H, N, N)  # (batch*hidden, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)  # (batch*hidden, K=N, N)
    # Compute batched matmul
    C = _batched_matmul_fp16(A, B_mat)  # (B*H,N,N) fp16
    # reshape back to [B,N,N,H]
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # Second layer norm
    out = out.float()
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # Apply output gate
    out = out * out_gate.float()
    # Final linear projection
    out = F.linear(out, weights["to_out.weight"])
    return out
```

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The forward pass consists of:
1️⃣ Layer‑norm over the last dimension.
2️⃣ FP16 linear projections (left/right) and gating (left/right/out).
   Optional binary mask is applied in FP16.
3️⃣ Fuse projection, mask and gate (simple element‑wise).
4️⃣ Compute the cubic contraction
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   as a batched GEMM (A @ Bᵀ) where A and B have shape (B·H, N, N).
   The heavy N³ work is performed by a custom Triton kernel that
   loads tiles in FP16, accumulates in FP32 and uses Tensor‑cores via
   ``tl.dot``.  Block sizes and the number of warps are tuned
   automatically for the problem size.
5️⃣ Second Layer‑norm over the hidden dimension, apply the output gate,
   and a final FP32 linear projection back to the original dimension.

All operations are fused as much as possible while keeping the
final result in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM kernel: C = A @ Bᵀ   (A: (B, M, K), B: (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # global memory pointers
    M, N, K,                            # matrix dimensions
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Matrix multiplication kernel for a batch of (M×K)·(K×N) products.
    Accumulates in FP32 and writes FP16 results.
    """
    pid_batch = tl.program_id(2)   # merged batch*hidden dimension
    pid_m = tl.program_id(0)       # tile row
    pid_n = tl.program_id(1)       # tile column

    # ------------------------------------------------------------------
    # Compute tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile of A and B
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_row
            + offs_k[None, :] * stride_a_col
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_row
            + offs_n[None, :] * stride_b_col
        )

        # Load tiles (with out‑of‑bounds masking)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Tensor‑core dot product
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result tile (convert back to FP16)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_row
        + offs_n[None, :] * stride_c_col
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched (B, M, K) @ (B, K, N) using the Triton kernel above.
    Arguments must be contiguous FP16 tensors.
    Returns FP16 tensor of shape (B, M, N).
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Heuristic block sizes based on matrix dimensions
    if max(M, N) <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
        num_warps = 4
    elif max(M, N) <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
        num_warps = 8
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 256, 256, 64
        num_warps = 8

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N

    # 3‑D launch grid: (tile_m, tile_n, batch*hidden)
    grid = (num_m, num_n, B)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),
        a.stride(1),
        a.stride(2),
        b.stride(0),
        b.stride(1),
        b.stride(2),
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,          # double‑buffering for latency hiding
    )
    return out


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
            - mask         : torch.Tensor of shape [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 2️⃣ Load weight tensors (cast to half where appropriate)
    # ------------------------------------------------------------------
    # LayerNorm 1
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear projections (no bias) – use FP16 for speed
    left_proj_w  = weights["left_proj.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()

    # Gating linear layers
    left_gate_w  = weights["left_gate.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # Output LayerNorm + linear
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3️⃣ First LayerNorm (float32) -> half precision for the heavy part
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )                                 # [B, N, N, dim]  float32
    x_h = x.half()                    # cast to fp16

    # ------------------------------------------------------------------
    # 4️⃣ Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)  # same shape

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask – broadcast over hidden dimension
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_h = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 6️⃣ Gating (sigmoid) – still in FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7️⃣ Prepare tensors for the N³ contraction
    #     left_perm:  (B, hidden, N, N)   – [i, k]
    #     right_perm: (B, hidden, N, N)   – [j, k]  (k axis swapped)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (j,k)

    # Release memory that is no longer needed
    del left, right, left_gate, right_gate, x_h, x

    B, H, N, _ = left_perm.shape

    # Merge batch and hidden dimensions for the kernel
    A = left_perm.view(B * H, N, N)          # (B·H, M=N, K=N)
    B_mat = right_perm.view(B * H, N, N)     # (B·H, K=N, N)

    # ------------------------------------------------------------------
    # 8️⃣ Batched matrix multiplication (FP16 → FP32 accumulation)
    # ------------------------------------------------------------------
    C = _batched_matmul_fp16(A, B_mat)      # (B·H, N, N) fp16

    # Restore original layout: [B, N, N, hidden_dim] (still fp16)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9️⃣ Second LayerNorm over hidden dimension (convert to FP32 first)
    # ------------------------------------------------------------------
    out = out.float()
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 🔟 Apply output gate (cast to FP32 for stability)
    # ------------------------------------------------------------------
    out = out * out_gate.float()

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Final linear projection back to `dim` (float32)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    return out
