"""
custom_kernel – Triton‑accelerated outgoing TriMul (AlphaFold3)

Algorithm (forward only)
------------------------
1. Layer‑norm over the input channel dimension `dim` (FP32).
2. Cast to FP16 and perform the four linear projections (left/right
   & their gates) in a *single* matmul per side by concatenating the
   projection‑ and gate‑weights:
        [proj; gate]  ∈ ℝ^{2·hidden_dim × dim}
   This halves the number of matmuls (2 instead of 4) and keeps the
   intermediate tensors in half‑precision.
3. Split the result, apply sigmoid to the gate part and (optionally)
   mask the projected part, then fuse the element‑wise product
   `proj * gate`.
4. Contraction (the N³ core):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h}
   is realised as a batched GEMM:
        A = left  → (B·H, N, N)
        B = rightᵀ → (B·H, N, N)
        C = A @ B
   A custom Triton kernel (`_batched_matmul_kernel`) performs this
   FP16×FP16 → FP32 multiplication with 64×64 tiles and Tensor‑cores.
5. Convert the result to FP32, apply a second LayerNorm over the hidden
   dimension, multiply by the output‑gate (computed from the
   normalized input), and finally linearly project back to `dim`.
6. The function returns a FP32 tensor of shape `[B, N, N, dim]`.

Only the N³ contraction is done in Triton; all other operations use
high‑performance PyTorch kernels (layernorm, linear, sigmoid, etc.).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ B
#   A: (batch, M, K)   FP16
#   B: (batch, K, N)   FP16
#   C: (batch, M, N)   FP16   (accumulated in FP32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,       # pointers
    M, N, K,                   # matrix sizes
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decompose pid into (batch, block_m, block_n)
    # ------------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_m * num_n)
    block_idx = pid % (num_m * num_n)
    block_m = block_idx // num_n
    block_n = block_idx % num_n

    # ------------------------------------------------------------------
    # Offsets for the current tile and masks for out‑of‑bounds elements
    # ------------------------------------------------------------------
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension, load tiles of A and B and accumulate
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A[batch, offs_m, offs_k]  (M×K tile)
        a_ptrs = (
            a_ptr
            + batch * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        # B[batch, offs_k, offs_n]  (K×N tile)
        b_ptrs = (
            b_ptr
            + batch * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot uses Tensor‑cores when possible (FP16×FP16→FP32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the tile back to C
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched half‑precision matrix multiplication (A @ B) using the Triton kernel
    defined above.

    Parameters
    ----------
    a : torch.Tensor
        Shape (B, M, K), dtype=torch.float16, contiguous.
    b : torch.Tensor
        Shape (B, K, N), dtype=torch.float16, contiguous.

    Returns
    -------
    torch.Tensor
        Shape (B, M, N), dtype=torch.float16, contiguous.
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match for batched matmul."

    out = torch.empty((B, M, N), dtype=torch.float16, device=a.device)

    # Tile sizes – 64 is a good trade‑off for H100 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_m = (M + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * num_m * num_n,)

    _batched_matmul_kernel[grid](
        a,
        b,
        out,
        M,
        N,
        K,
        a.stride(0),  # stride between batches
        a.stride(1),  # stride M (row)
        a.stride(2),  # stride K (col)
        b.stride(0),  # stride between batches
        b.stride(1),  # stride K (row)
        b.stride(2),  # stride N (col)
        out.stride(0),
        out.stride(1),
        out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # 8 warps => 256 threads, optimal for 64×64 tiles
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of torch.Tensor with the model parameters
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 0️⃣ weight shortcuts
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear & gate weights (FP16 for the heavy work)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]          # (dim, hidden_dim), stay fp32

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension `dim` (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 for all subsequent linear work
    # ------------------------------------------------------------------
    x_h = x.half()                     # [B, N, N, dim]  (FP16)

    # ------------------------------------------------------------------
    # 3️⃣ Fuse projection + gate for *left* and *right* in a single matmul
    # ------------------------------------------------------------------
    # concatenate proj and gate weights → shape (2*hidden_dim, dim)
    left_combined_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, dim)
    right_combined_w = torch.cat([right_proj_w, right_gate_w], dim=0)

    # combined linear – produces [B, N, N, 2*hidden_dim] (FP16)
    left_all = F.linear(x_h, left_combined_w)      # (B,N,N,2H)
    right_all = F.linear(x_h, right_combined_w)

    # split projection and gate part
    left_proj = left_all[..., :hidden_dim]                     # (B,N,N,H)
    left_gate_raw = left_all[..., hidden_dim:]                # (B,N,N,H)
    right_proj = right_all[..., :hidden_dim]
    right_gate_raw = right_all[..., hidden_dim:]

    # apply optional mask *before* gating (as in the reference)
    if mask is not None and mask.numel() > 0:
        mask_h = mask.unsqueeze(-1).to(left_proj.dtype)      # (B,N,N,1)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # sigmoid gates
    left_gate = torch.sigmoid(left_gate_raw)   # (B,N,N,H)  FP16
    right_gate = torch.sigmoid(right_gate_raw)

    # fuse projection & gate
    left = left_proj * left_gate               # (B,N,N,H)  FP16
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ Compute the N³ contraction via a batched GEMM
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for each side
    Bsz, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) -> (i,k)
    right_perm_t = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) -> (k,j)

    A = left_perm.view(Bsz * H, N, N)          # (batch, M=N, K=N)
    B_mat = right_perm_t.view(Bsz * H, N, N)   # (batch, K=N, N)

    # Batched FP16 GEMM (Tensor‑cores) – result still FP16
    C = _batched_matmul_fp16(A, B_mat)        # (B*H, N, N)  FP16

    # Restore the original 4‑D layout: (B, N, N, H)
    out_h = C.view(Bsz, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Cast back to FP32 for the second LayerNorm
    # ------------------------------------------------------------------
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # ------------------------------------------------------------------
    # 7️⃣ Output gate (computed from the normalized input `x_h`)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))   # (B,N,N,H) FP16
    out_gate_f32 = out_gate.to(torch.float32)

    out = out * out_gate_f32

    # ------------------------------------------------------------------
    # 8️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (B,N,N,dim)  FP32

    return out
```
