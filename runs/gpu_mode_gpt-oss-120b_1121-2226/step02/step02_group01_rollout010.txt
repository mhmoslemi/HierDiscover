"""
Triton‑accelerated outgoing TriMul (αFold‑3 style)

Algorithm
---------
1. Layer‑norm the 4‑D input (float32) → x.
2. Cast to fp16 and fuse the two linear projections *and* their
   corresponding gating linear layers into a single matmul per side:
        left_and_gate  = x @ [left_proj; left_gate]ᵀ
        right_and_gate = x @ [right_proj; right_gate]ᵀ
   (the two halves are split afterwards).  This halves the number of
   passes over the large (B·N·N·dim) tensor.
3. Optional mask is applied to the two projected tensors (still fp16).
4. Element‑wise sigmoid gates are applied and the gated tensors are
   multiplied together.
5. Core cubic update:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matmul  A (M×K) @ Bᵀ (K×N) with
   M=N=seq_len and a batch‑size B·hidden_dim.
   A is stored as (batch, M, K) and Bᵀ as (batch, K, N); the
   kernel tiles the product (default 128×128 tiles, K‑tile‑size 64)
   and accumulates in fp32 (Tensor‑Core friendly).
6. Reshape back to (B, N, N, hidden_dim) and apply a second
   LayerNorm.
7. Multiply by the output gate (sigmoid of a third linear) and finish
   with a linear projection back to the original feature dimension.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: batched matmul A (M×K) * Bᵀ (K×N) → C (M×N)
# --------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bk, stride_bn,   # B strides: batch, K, N
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_b = tl.program_id(2)          # batch (hidden‑channel)

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the M‑tile of A and the K‑tile of Bᵀ
    a_ptrs = (
        A_ptr
        + pid_b * stride_ah
        + offs_m[:, None] * stride_am
        + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    )
    b_ptrs = (
        B_ptr
        + pid_b * stride_bh
        + tl.arange(0, BLOCK_K)[:, None] * stride_bk
        + offs_n[None, :] * stride_bn
    )

    # Accumulator (fp32) – Tensor‑Core mixed‑precision accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K‑tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K

        # Masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (k_base + tl.arange(0, BLOCK_K)[None, :] < K)
        mask_b = (k_base + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # fp16

        # Tensor‑core dot, fp16·fp16 → fp32 accumulate
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Write out
    # ------------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + pid_b * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched matmul helper used by the TriMul forward pass.
    A : (batch, M, K)   fp16
    B : (batch, K, N)   fp16   (note the transposed layout)
    Returns C : (batch, M, N) fp32
    """
    assert A.is_cuda and B.is_cuda
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# --------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing version)
# --------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Entry‑point for the evaluation harness.
    data = (input_tensor, mask, weights_dict, config_dict)

    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – fp32

    # ------------------------------------------------------------------
    # 2) Cast to fp16 and fuse the projection+gate linear layers
    # ------------------------------------------------------------------
    x_h = x.half()  # [B, N, N, dim] – fp16

    # Concatenate the projection and gate weights for left and right side
    left_w_concat = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # (2*hidden, dim)
    right_w_concat = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # (2*hidden, dim)

    # Perform the two fused matmuls
    left_and_gate = F.linear(x_h, left_w_concat)   # [B, N, N, 2*hidden] (fp16)
    right_and_gate = F.linear(x_h, right_w_concat) # [B, N, N, 2*hidden] (fp16)

    # Split projection & gate, apply sigmoid to gates
    left = left_and_gate[..., :hidden_dim]          # [B, N, N, hidden]
    left_gate = left_and_gate[..., hidden_dim:].sigmoid()
    right = right_and_gate[..., :hidden_dim]
    right_gate = right_and_gate[..., hidden_dim:].sigmoid()

    # ------------------------------------------------------------------
    # 3) Optional masking (still fp16)
    # ------------------------------------------------------------------
    if not nomask:
        m = mask.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core cubic update – batched matmul via Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # B, seq_len, seq_len, hidden_dim

    # --------------------------------------------------------------
    #   Prepare tensors for the batched GEMM
    #   left  → (batch, M, K)   with M=N, K=N
    #   right → (batch, K, N)   i.e. the transposed layout required by the kernel
    # --------------------------------------------------------------
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (batch, M, K)
    right_mat_T = right.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)  # (batch, K, N)

    # Choose tile size – larger tiles for larger sequences
    if N <= 384:
        BLOCK_M, BLOCK_N = 64, 64
    else:
        BLOCK_M, BLOCK_N = 128, 128
    BLOCK_K = 64

    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat_T,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )  # (B*H, N, N) – fp32

    # Reshape back to [B, N, N, hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    # ------------------------------------------------------------------
    # 7) Output gate and final projection back to dim
    # ------------------------------------------------------------------
    out_gate = F.linear(x_h, weights["out_gate.weight"].half()).sigmoid()  # fp16
    out = out * out_gate.float()                                            # fp32

    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] – fp32
    return out
