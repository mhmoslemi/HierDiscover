"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The module receives a 4‑D tensor X ∈ [B, N, N, D] and an (optional) binary
mask M ∈ [B, N, N].  The computation follows the “outgoing” update from
AlphaFold‑3:

    1. Layer‑norm over the last channel (dim = D).
    2. Two bias‑free linear projections (left / right) from D → H.
    3. Three sigmoid gates (left_gate, right_gate, out_gate) from D → H.
    4. Optional mask is applied to the projected tensors.
    5. Element‑wise gating:
           left  = (left_proj * M)   * sigmoid(left_gate)
           right = (right_proj * M)  * sigmoid(right_gate)
    6. Core N³ operation (outgoing update):
           out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
       which is a batched matrix multiplication
           out = left @ rightᵀ   (per hidden channel).
       The heavy N³ work is performed by a custom Triton kernel that treats
       each hidden channel as an independent batch element, i.e. the
       tensors are reshaped to (B·H, N, N) and multiplied with
       fp16 inputs / fp32 accumulation.
    7. Layer‑norm over the hidden dimension, element‑wise out‑gate
       multiplication and a final linear projection H → D.

Speed‑ups relative to the naïve PyTorch implementation:
    • 2× fused linear layers (proj+gate) → fewer kernel launches.
    • 1× fused mask & gate multiplication (in‑place, half‑precision).
    • A tuned Triton batched‑matmul kernel (BLOCK_M=N_BLOCK=128,
      BLOCK_K=64, 8 warps, 4 stages) that uses Tensor‑Cores.
    • All heavy arithmetic is performed in fp16 → half the memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul A (M×K) @ Bᵀ (K×N)  →  C (M×N)
#   A : (batch, M, K)   fp16
#   B : (batch, K, N)   fp16   (here B is the transposed right tensor)
#   C : (batch, M, N)   fp32   (fp16·fp16 → fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes
    stride_ah, stride_am, stride_ak,   # A strides: batch, M, K
    stride_bh, stride_bk, stride_bn,   # B strides: batch, K, N
    stride_ch, stride_cm, stride_cn,   # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_b = tl.program_id(2)   # batch (B·H)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = A_ptr + pid_b * stride_ah
    b_batch = B_ptr + pid_b * stride_bh
    c_batch = C_ptr + pid_b * stride_ch

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        a_mask = (offs_m[:, None] < M) & (k_off + tl.arange(0, BLOCK_K)[None, :] < K)
        a_tile = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)   (note B is already transposed)
        b_ptrs = b_batch + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b_mask = (k_off + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # matrix multiply for this K‑tile
        acc += tl.dot(a_tile, b_tile)          # fp32 accumulation

    # write‑back
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def _batched_matmul_triton(
    A: torch.Tensor,
    B: torch.Tensor,
    *,
    BLOCK_M: int = 128,
    BLOCK_N: int = 128,
    BLOCK_K: int = 64,
) -> torch.Tensor:
    """
    Batched mat‑mul (A @ Bᵀ) using the Triton kernel above.
    A – (batch, M, K)  fp16
    B – (batch, K, N)  fp16   (already transposed)
    Returns C – (batch, M, N)  fp32
    """
    assert A.is_cuda and B.is_cuda, "inputs must be on CUDA"
    assert A.dtype == torch.float16 and B.dtype == torch.float16, "kernel expects fp16"

    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    stride_ah, stride_am, stride_ak = A.stride()
    stride_bh, stride_bk, stride_bn = B.stride()
    stride_ch, stride_cm, stride_cn = C.stride()

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=4,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Args:
        data: tuple (input_tensor, mask, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/int) or None
            - weights      : dict of model parameters (float32)
            - config       : dict with at least "dim" and "hidden_dim",
                             optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # 1️⃣ Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 2️⃣ Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D] – float32

    # ------------------------------------------------------------------
    # 3️⃣ Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_f16 = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 4️⃣ Fuse projection + gate linear layers (bias‑free)
    # ------------------------------------------------------------------
    # left_proj + left_gate
    left_comb_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)       # (2*H, D)
    left_comb = F.linear(x_f16, left_comb_w)  # [B, N, N, 2*H]  fp16
    left_proj, left_gate_pre = torch.split(left_comb, hidden_dim, dim=-1)
    left_gate = torch.sigmoid(left_gate_pre)   # fp16

    # right_proj + right_gate
    right_comb_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)       # (2*H, D)
    right_comb = F.linear(x_f16, right_comb_w)  # [B, N, N, 2*H]  fp16
    right_proj, right_gate_pre = torch.split(right_comb, hidden_dim, dim=-1)
    right_gate = torch.sigmoid(right_gate_pre)  # fp16

    # ------------------------------------------------------------------
    # 5️⃣ Optional mask (broadcasted to the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left_proj = left_proj * m
        right_proj = right_proj * m

    # ------------------------------------------------------------------
    # 6️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate    # fp16
    right = right_proj * right_gate  # fp16

    # ------------------------------------------------------------------
    # 7️⃣ Core N³ batched mat‑mul (outgoing update)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, _, hidden_dim

    # reshape to (B·H, N, N) – each hidden channel becomes an independent batch
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # fp16, (M= N, K = N)
    # right needs to be transposed on the last two dims so that the kernel sees (K, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)  # fp16
    right_mat_t = right_mat.permute(0, 2, 1).contiguous()                # (B·H, K=N, N)

    # Batched mat‑mul → fp32 result
    out_mat = _batched_matmul_triton(
        left_mat,
        right_mat_t,
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=64,
    )   # [B·H, N, N]  fp32

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # fp32

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    # out_gate (σ(Linear(x))) – still half‑precision
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_f16, out_gate_w))   # [B, N, N, H] fp16

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # fp32

    out = out * out_gate                                 # fp32 (out_gate up‑cast)

    # final projection back to original dimension D
    out = F.linear(out, weights["to_out.weight"])        # fp32

    return out
```
