1️⃣ **Layer‑Norm** on the input (float32 → half).  
2️⃣ **Fused linear projection** – the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) are
   concatenated into a single matrix of shape (5·H, C) and multiplied
   with the normalized input in one GEMM (FP16). The result is reshaped
   into the three projection tensors and three gating tensors; the
   gates are passed through a sigmoid (half‑precision).

3️⃣ **Optional mask** (broadcasted over the hidden dimension) is applied
   to the left and right projected tensors.

4️⃣ **Cubic update** – for each hidden slice `h` we compute
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.  
   This is a batched matrix multiplication `A @ Bᵀ` where
   `A = left.reshape(B·H, N, N)` and `B = right.reshape(B·H, N, N)`.
   A custom Triton kernel performs the batched GEMM in FP16 with FP32
   accumulation, using a 128×128×64 tile (tuned for H100 tensor‑cores)
   and double‑buffering (`num_stages=3`).

5️⃣ **Second Layer‑Norm + output‑gate** (float32).  

6️⃣ **Output linear projection** back to the original channel dimension
   (half → float32).

The implementation keeps the heavy work in half‑precision while
maintaining float‑32 precision for the two LayerNorms, matching the
reference PyTorch behaviour.  Only the cubic update is written as a
custom Triton kernel, as required.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched A @ Bᵀ  (A,B : fp16 → C : fp16, acc in fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # problem sizes (all == seq_len)
    stride_ah, stride_ai, stride_ak,   # strides for A (BH, i, k)
    stride_bh, stride_bk, stride_bn,   # strides for Bᵀ (BH, k, j)
    stride_ch, stride_ci, stride_cj,   # strides for C (BH, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matrix‑multiply C = A @ Bᵀ for each hidden slice."""
    pid_h = tl.program_id(axis=2)      # combined batch × hidden index
    pid_m = tl.program_id(axis=0)      # i‑tile
    pid_n = tl.program_id(axis=1)      # j‑tile

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ---- Load tile of A (M × K) ---------------------------------
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)   # fp16 → fp32 for accumulation

        # ---- Load tile of Bᵀ (K × N) --------------------------------
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)   # fp16 → fp32

        # ---- Inner product -----------------------------------------
        acc += tl.dot(a, b)    # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

    # ---- Write result (fp16) ---------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : mapping from layer names to torch.Parameters
        - config_dict  : must contain keys "dim", "hidden_dim",
                         optional "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32) → half
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # [B, N, N, dim]  float32
    x_h = x.half()

    # --------------------------------------------------------------
    # 2️⃣ Fused linear stage (5·hidden_dim outputs, half‑precision)
    # --------------------------------------------------------------
    # Concatenate the five weight matrices once (shape (5·H, dim))
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)            # [5·H, dim]  half

    # Single GEMM -> shape [B, N, N, 5·H]
    fused = F.linear(x_h, fused_weight)   # half
    B, N, _, _ = fused.shape

    # Split fused result
    fused = fused.view(B, N, N, 5, hidden_dim)    # [B, N, N, 5, H] (half)
    left_proj  = fused[..., 0, :]                 # [B, N, N, H]
    right_proj = fused[..., 1, :]
    left_gate  = torch.sigmoid(fused[..., 2, :])  # half sigmoid
    right_gate = torch.sigmoid(fused[..., 3, :])
    out_gate   = torch.sigmoid(fused[..., 4, :])  # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # mask: [B, N, N] → [B, N, N, 1] → half
        mask = mask_tensor.unsqueeze(-1).to(x_h.dtype)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # --------------------------------------------------------------
    # 4️⃣ Apply gating (still half)
    # --------------------------------------------------------------
    left  = left_proj * left_gate   # [B, N, N, H] half
    right = right_proj * right_gate # [B, N, N, H] half

    # --------------------------------------------------------------
    # 5️⃣ Rearrange for batched GEMM (BH = B·H)
    # --------------------------------------------------------------
    # bring hidden dim next to batch and make it contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)   # [BH, N, N]   half
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]   half

    # --------------------------------------------------------------
    # 6️⃣ Triton GEMM kernel (A @ Bᵀ) → half output
    # --------------------------------------------------------------
    out_flat = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # Tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # one launch per (batch, hidden) slice
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                                   # M, N, K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # --------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] (still half)
    # --------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out_half = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 8️⃣ Second LayerNorm (float32) + output gate
    # --------------------------------------------------------------
    out_fp32 = out_half.float()   # cast for the LayerNorm
    out_fp32 = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )                               # [B, N, N, H] float32
    out_fp32 = out_fp32 * out_gate.float()   # apply output gate

    # --------------------------------------------------------------
    # 9️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w_h = weights["to_out.weight"].to(torch.float16)
    out_proj_half = F.linear(out_fp32.half(), to_out_w_h)   # [B,N,N,dim] half
    out_final = out_proj_half.float()                       # cast to float32

    return out_final
```
