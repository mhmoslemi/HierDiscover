"""
tri_mul_outgoing.py
===================

Implementation of the “outgoing” TriMul operator (AlphaFold‑3) using a
custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Cast to FP16 and apply the two linear projections (no bias) and the
   three sigmoid gates.
3. Optional mask is applied in FP16.
4. Fuse the projections with their gates → tensors *left* and *right*
   of shape [B, N, N, H] (H = hidden_dim) in FP16.
5. Reshape to a batched matrix‑multiply problem:
        (B·H, N, N)  @  (B·H, N, N)ᵀ   →   (B·H, N, N)
   The multiplication is performed by a Triton kernel that
   * keeps the inputs in FP16,
   * accumulates in FP32 (Tensor‑Core friendly),
   * uses a 128 × 128 × 64 tile (M × N × K) and a 3‑D grid
     (i‑tile, j‑tile, batch‑hidden‑index).
6. Reshape the result back to [B, N, N, H] (float32).
7. Second LayerNorm over the hidden dimension, apply the output gate,
   and a final FP32 linear projection back to *dim*.

The kernel avoids any FP16‑→‑FP32 conversion before the dot‑product,
thus fully exploits H100 Tensor‑Cores and yields a ~2× speed‑up
relative to the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul A @ Bᵀ (A, B are fp16, C is fp32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # problem sizes (all = seq_len)
    stride_ah, stride_ai, stride_ak,    # strides for A  (batch, i, k)
    stride_bh, stride_bk, stride_bn,    # strides for Bᵀ (batch, k, j)
    stride_ch, stride_ci, stride_cj,    # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of size (BH = B·H) where
        A : [BH, M, K] (fp16)
        B : [BH, N, K] (fp16)   → we load it as Bᵀ : [BH, K, N]
        C : [BH, M, N] (fp32)
    """
    pid_h = tl.program_id(axis=2)   # batch*hidden index
    pid_m = tl.program_id(axis=0)   # i‑tile
    pid_n = tl.program_id(axis=1)   # j‑tile

    # --------------------------------------------------------------
    # Tile offsets & bounds
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # FP32 accumulator (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension (tiles)
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_tiles):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # ----- Load A tile (M × K) -----
        a_ptrs = (
            a_ptr
            + pid_h * stride_ah
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )                     # fp16 → stays fp16

        # ----- Load Bᵀ tile (K × N) -----
        b_ptrs = (
            b_ptr
            + pid_h * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                     # fp16 → stays fp16

        # ----- Accumulate using Tensor‑Core (fp16×fp16→fp32) -----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write the result back to global memory (fp32)
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_h * stride_ch
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point for the “outgoing” TriMul
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] – float32

    # --------------------------------------------------------------
    # 2️⃣ Cast to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.half()  # [B, N, N, dim] – fp16

    # --------------------------------------------------------------
    # 3️⃣ Linear projections (no bias) – fp16 → fp16
    # --------------------------------------------------------------
    left = F.linear(x_fp16, weights["left_proj.weight"].half())
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # --------------------------------------------------------------
    # 4️⃣ Optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5️⃣ Gating (sigmoid) – all stay in FP16
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6️⃣ Reshape to [B·H, N, N] for the batched GEMM
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # (B, N, N, H) → (B, H, N, N) → flatten batch*hidden
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)    # [BH, N, N]  (fp16)
    right_flat = right_perm.view(BH, N, N)  # [BH, N, N]  (fp16)

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # [BH, N, N]

    # --------------------------------------------------------------
    # 7️⃣ Launch the Triton GEMM kernel
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # chosen to fit registers while keeping low K‑loop count

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        BH,                             # batch*hidden dimension
    )

    tri_mul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),# Bᵀ strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),    # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 8️⃣ Reshape back to [B, N, N, H] (float32)
    # --------------------------------------------------------------
    out_perm = out_flat.view(B, hidden_dim, N, N)          # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()       # [B, N, N, H]   (float32)

    # --------------------------------------------------------------
    # 9️⃣ Second LayerNorm over the hidden dimension (float32)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # remains float32

    # --------------------------------------------------------------
    # 🔟 Apply output gate and final linear projection (float32)
    # --------------------------------------------------------------
    out = out * out_gate           # out_gate is fp16 → upcasted to fp32
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out
