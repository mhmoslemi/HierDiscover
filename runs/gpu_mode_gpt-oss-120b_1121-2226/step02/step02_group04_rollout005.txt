import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_inplace_kernel(
    left_ptr,
    right_ptr,
    left_gate_ptr,
    right_gate_ptr,
    mask_ptr,
    total: tl.constexpr,
    hidden: tl.constexpr,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # load
    l = tl.load(left_ptr + offs, mask=mask, other=0.0)
    r = tl.load(right_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)

    # mask per hidden group
    mask_idx = offs // hidden
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    lo = l * lg * m
    ro = r * rg * m

    tl.store(left_ptr + offs, lo, mask=mask)
    tl.store(right_ptr + offs, ro, mask=mask)

"""
Optimized forward pass for the *outgoing* TriMul operator.

Algorithm
---------
1. **Input LayerNorm** – performed in FP32 for numerical stability, then
   the result is cast to FP16.
2. **Fused linear projection** – a single `F.linear` computes the five
   linear maps (left_proj, left_gate, right_proj, right_gate, out_gate)
   in one matrix‑multiply.
3. **Sigmoid gates** – applied to the three gate tensors.
4. **Mask & gate fusion** – a tiny Triton kernel updates the projected
   tensors *in‑place*:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
   The kernel works on a flattened view and avoids any extra
   allocation.
5. **N³ contraction** – the costly sum‑over‑k is expressed as an
   `torch.einsum('b i k h, b j k h -> b i j h')`.  This avoids the
   explicit `permute/contiguous` copies required for a batched gemm and
   lets the PyTorch dispatcher select the best cuBLAS / CUTLASS kernel.
6. **Post‑contraction LayerNorm + output gate** – again in FP32.
7. **Final linear** – maps from the hidden dimension back to the original
   feature dimension in FP16, then cast to FP32 for the returned tensor.

All heavy element‑wise work (mask‑gate fusion) is off‑loaded to a custom
Triton kernel, while the compute‑intensive contraction re‑uses the
highly‑optimized kernels already present in the PyTorch CUDA stack.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_inplace_kernel(
    left_ptr, right_ptr,           # projected tensors (fp16) – flattened
    left_gate_ptr, right_gate_ptr, # gate tensors (fp16) – flattened
    mask_ptr,                      # mask tensor (fp16) of size B·N·N
    total: tl.constexpr,           # total number of elements = B·N·N·H
    hidden: tl.constexpr,          # hidden dimension H
    BLOCK: tl.constexpr,           # threads per program
):
    """
    In‑place fusion of mask and gates:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    The mask is broadcast over the hidden dimension, i.e. every `hidden`
    consecutive elements share the same mask value.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total

    # load the four tensors (scalar per thread)
    l_val = tl.load(left_ptr + offs,            mask=mask, other=0.0)
    r_val = tl.load(right_ptr + offs,           mask=mask, other=0.0)
    l_gate = tl.load(left_gate_ptr + offs,      mask=mask, other=0.0)
    r_gate = tl.load(right_gate_ptr + offs,     mask=mask, other=0.0)

    # each hidden‑dim vector shares one mask entry
    mask_idx = offs // hidden
    m_val = tl.load(mask_ptr + mask_idx,        mask=mask, other=1.0)

    # fuse
    lo = l_val * l_gate * m_val
    ro = r_val * r_gate * m_val

    # write back in‑place
    tl.store(left_ptr + offs,  lo, mask=mask)
    tl.store(right_ptr + offs, ro, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, D] (float32)
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict with model parameters (FP32 tensors)
            - config       : dict with keys "dim", "hidden_dim", optional "eps"

    Returns:
        torch.Tensor [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    device = input_tensor.device

    # --------------------------------------------------------------
    # cast weights to the precision we will use (FP16 for matmuls)
    # --------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16)          # (D,)
    norm_b = weights["norm.bias"].to(torch.float16)            # (D,)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, D)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16) # (H, D)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, D)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16) # (H, D)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, D)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)   # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)     # (H,)
    to_out_w      = weights["to_out.weight"].to(torch.float16)       # (D, H)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (FP32) → FP16
    # --------------------------------------------------------------
    # keep the normalization in FP32 for stability, then cast
    x = F.layer_norm(
        input_tensor, (dim,),
        weight=norm_w.to(torch.float32),
        bias=norm_b.to(torch.float32),
        eps=eps,
    ).to(torch.float16)                               # [B, N, N, D] fp16

    # --------------------------------------------------------------
    # 2. One fused linear for all projections & gates (5·H output features)
    # --------------------------------------------------------------
    fused_weight = torch.cat(
        [left_proj_w, left_gate_w, right_proj_w, right_gate_w, out_gate_w],
        dim=0,
    )   # shape (5*H, D)

    fused = F.linear(x, fused_weight)                 # [B, N, N, 5*H] fp16

    B, N, _, _ = fused.shape
    H = hidden_dim

    # --------------------------------------------------------------
    # 3. Split fused tensor into the five components
    # --------------------------------------------------------------
    left_proj, left_gate, right_proj, right_gate, out_gate = torch.split(
        fused, H, dim=-1
    )   # each: [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 4. Apply sigmoid to the three gate tensors
    # --------------------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # --------------------------------------------------------------
    # 5. Fuse mask, left/right gates via an in‑place Triton kernel
    # --------------------------------------------------------------
    if mask is None:
        # no mask → simply apply the gates
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # flatten everything for the kernel
        left_f = left_proj.reshape(-1)
        right_f = right_proj.reshape(-1)
        left_gate_f = left_gate.reshape(-1)
        right_gate_f = right_gate.reshape(-1)

        mask_f = mask.to(torch.float16).reshape(-1)   # [B·N·N]

        total_elems = left_f.shape[0]                # B·N·N·H
        BLOCK_SIZE = 1024
        grid = (triton.cdiv(total_elems, BLOCK_SIZE),)

        _mask_gate_inplace_kernel[grid](
            left_f,
            right_f,
            left_gate_f,
            right_gate_f,
            mask_f,
            total=total_elems,
            hidden=hidden_dim,
            BLOCK=BLOCK_SIZE,
        )

        # reshape back to the original 4‑D layout
        left = left_f.view(B, N, N, H)
        right = right_f.view(B, N, N, H)

    # --------------------------------------------------------------
    # 6. N³ contraction : out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with torch.einsum to avoid extra permutes / copies.
    # --------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # --------------------------------------------------------------
    # 7. Post‑contraction LayerNorm (FP32) + output gate
    # --------------------------------------------------------------
    out = F.layer_norm(
        out.to(torch.float32),
        (hidden_dim,),
        weight=to_out_norm_w.to(torch.float32),
        bias=to_out_norm_b.to(torch.float32),
        eps=eps,
    )
    out = out * out_gate.to(out.dtype)   # broadcast over H

    # --------------------------------------------------------------
    # 8. Final linear projection back to original dim (FP16)
    # --------------------------------------------------------------
    out = F.linear(out.to(torch.float16), to_out_w)   # [B, N, N, D] fp16

    # --------------------------------------------------------------
    # 9. Cast to float32 for the final output
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
