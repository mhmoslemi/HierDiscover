===================================================

Algorithm
---------

   Triton GEMM kernel (`trimul_matmul_batch_kernel`) that works on
   half‑precision tensors (FP16) and accumulates in FP32.
6. **Output LayerNorm → out_gate → linear projection** back to `dim`.

Only the mask‑gate fusion and the heavy cubic contraction are performed
with Triton; all other operations reuse PyTorch’s highly‑optimized kernels.
The final result is returned in `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1) Mask + gate fusion kernel
# ----------------------------------------------------------------------
@triton.jit
def fused_mask_gate_kernel(
    left_ptr,        # [B,N,N,H] – in‑place output
    left_gate_ptr,   # [B,N,N,H]
    right_ptr,       # [B,N,N,H] – in‑place output
    right_gate_ptr,  # [B,N,N,H]
    mask_ptr,        # [B,N,N] (0/1)
    B, N, H,         # problem sizes
    # strides for left
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    # strides for mask (no hidden dim)
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK: tl.constexpr,
):
    """Fuse ``left = left * left_gate * mask`` and ``right = right * right_gate * mask``.
    The operation is performed in‑place on ``left`` and ``right``.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index inside the tensor
    total = B * N * N * H
    valid = offs < total

    # Decode linear offset → (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # Bounds check (should be true for all valid offsets)
    valid = valid & (b < B) & (i < N) & (j < N) & (h < H)

    # Pointers
    left_off   = left_ptr   + b * stride_l_b   + i * stride_l_i   + j * stride_l_j   + h * stride_l_h
    leftg_off  = left_gate_ptr + b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    right_off  = right_ptr  + b * stride_r_b   + i * stride_r_i   + j * stride_r_j   + h * stride_r_h
    rightg_off = right_gate_ptr + b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    mask_off   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + j * stride_m_j

    # Load
    left      = tl.load(left_off,   mask=valid, other=0.0)
    left_gate = tl.load(leftg_off,  mask=valid, other=0.0)
    right     = tl.load(right_off,  mask=valid, other=0.0)
    right_gate= tl.load(rightg_off, mask=valid, other=0.0)
    mask_val  = tl.load(mask_off,   mask=valid, other=1.0)   # default = 1 (no masking)

    # Fuse mask + gate
    left  = left  * left_gate  * mask_val
    right = right * right_gate * mask_val

    # Store back in‑place
    tl.store(left_off,  left,  mask=valid)
    tl.store(right_off, right, mask=valid)


# ----------------------------------------------------------------------
# 2) Batched GEMM kernel for the cubic contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_matmul_batch_kernel(
    a_ptr,           # left  : [B*H, M, K]  (FP16)
    b_ptr,           # right : [B*H, K, N]  (FP16)
    c_ptr,           # out   : [B*H, M, N]  (FP16)
    B_H, M, N, K,    # problem sizes
    # strides for a (batch, m, k)
    stride_a_batch, stride_a_m, stride_a_k,
    # strides for b (batch, k, n)
    stride_b_batch, stride_b_k, stride_b_n,
    # strides for c (batch, m, n)
    stride_c_batch, stride_c_m, stride_c_n,
    # tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched half‑precision GEMM (FP16 inputs, FP32 accumulation)."""
    pid_batch = tl.program_id(0)        # batch * hidden index
    pid_m = tl.program_id(1)            # tile over M (i axis)
    pid_n = tl.program_id(2)            # tile over N (j axis)

    # Compute start offsets of the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for the current tile (without k offset yet)
    a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m
    b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_n[None, :] * stride_b_n

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for load boundaries
        a_mask = (offs_m < M)[:, None] & (cur_k < K)[None, :]
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs + cur_k * stride_a_k, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs + cur_k[:, None] * stride_b_k, mask=b_mask, other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Convert back to FP16
    c = acc.to(tl.float16)

    # Store the result
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, c, mask=c_mask)


# ----------------------------------------------------------------------
# 3) Entry point: custom Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)   # True → no mask required

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑less) → left / right
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Fuse mask + gate (Triton) or apply gates only
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask dtype matches projection dtype (float32 before casting)
        mask_fp = mask_tensor.to(x.dtype)

        BLOCK = 256
        total = left.numel()                     # B * N * N * H

        # Strides for all involved tensors (int values)
        s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
        s_lg_b, s_lg_i, s_lg_j, s_lg_h = left_gate.stride()
        s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
        s_rg_b, s_rg_i, s_rg_j, s_rg_h = right_gate.stride()
        s_m_b, s_m_i, s_m_j = mask_fp.stride()

        grid = (triton.cdiv(total, BLOCK),)     # 1‑D grid

        fused_mask_gate_kernel[grid](
            left_ptr=left,
            left_gate_ptr=left_gate,
            right_ptr=right,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp,
            B=input_tensor.shape[0],
            N=input_tensor.shape[1],
            H=hidden_dim,
            stride_l_b=s_l_b, stride_l_i=s_l_i, stride_l_j=s_l_j, stride_l_h=s_l_h,
            stride_lg_b=s_lg_b, stride_lg_i=s_lg_i, stride_lg_j=s_lg_j, stride_lg_h=s_lg_h,
            stride_r_b=s_r_b, stride_r_i=s_r_i, stride_r_j=s_r_j, stride_r_h=s_r_h,
            stride_rg_b=s_rg_b, stride_rg_i=s_rg_i, stride_rg_j=s_rg_j, stride_rg_h=s_rg_h,
            stride_m_b=s_m_b, stride_m_i=s_m_i, stride_m_j=s_m_j,
            BLOCK=BLOCK,
        )
        # Gates are already fused; free the intermediate tensors
        del left_gate, right_gate
    else:
        # No mask – simply apply the gates
        left  = left * left_gate
        right = right * right_gate
        del left_gate, right_gate

    # ------------------------------------------------------------------
    # 5. Core cubic contraction using a batched FP16 GEMM (Triton)
    # ------------------------------------------------------------------
    # Cast to half‑precision before the heavy GEMM
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape               # B, N, N, H   (H == hidden_dim)

    # Rearrange tensors so that each hidden channel becomes a separate batch element
    # left_h: (B, H, N, N)   where the last two dims are (i, k)
    left_batch  = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # right_h: (B, H, N, N) where the last two dims are (k, j)
    right_batch = right_h.permute(0, 3, 2, 1).contiguous().view(B * H, N, N)

    # Allocate output tensor for the batched GEMM (FP16)
    out_batch = torch.empty_like(left_batch, dtype=torch.float16)

    # Tile sizes – these values work well on H100 Tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides for the three tensors (batch‑major layout)
    stride_a_batch, stride_a_m, stride_a_k = left_batch.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_batch.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_batch.stride()

    grid = (
        B * H,                               # batch*hidden dimension
        triton.cdiv(N, BLOCK_M),            # tiles over M (i)
        triton.cdiv(N, BLOCK_N),            # tiles over N (j)
    )

    trimul_matmul_batch_kernel[grid](
        a_ptr=left_batch,
        b_ptr=right_batch,
        c_ptr=out_batch,
        B_H=B * H,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_batch.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to FP32 for the remaining ops

    # ------------------------------------------------------------------
    # 6. Output LayerNorm → out_gate → final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
