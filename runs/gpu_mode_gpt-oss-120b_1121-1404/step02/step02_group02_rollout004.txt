def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps=1e-5

    # 1 Normalize
    x_norm_fp32 = torch.nn.functional.layer_norm(input_tensor,
                   (dim,),
                   weight=weights["norm.weight"],
                   bias=weights["norm.bias"],
                   eps=eps)
    # cast to fp16
    x_norm = x_norm_fp32.to(torch.float16)

    # 2 Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"].to(torch.float16))

    # 3 Mask if needed
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_fp16
        right = right * mask_fp16

    # 4 Gating
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # 5 Core reduction using batched GEMM (half precision)
    # permute to (B, hidden_dim, N, N) then reshape
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H
    left_flat = left.view(P, N, N)
    right_flat = right.view(P, N, N)

    # make contiguous half
    left_fp16 = left_flat
    right_fp16 = right_flat

    # Batch GEMM: left * right^T
    out_fp16 = torch.bmm(left_fp16, right_fp16.transpose(1, 2))

    # cast to float32 and reshape
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 6 Final layer norm (FP32)
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"],
                                         eps=eps)

    # Gate with out_gate (half) using triton kernel
    # out_gate shape [B,N,N,hidden_dim], half
    total_el = out.numel()
    BLOCK = 65536  # choose reasonable block size
    grid = (triton.cdiv(total_el, BLOCK),)
    # call kernel

        out,
        out_gate,
        total_el,
        BLOCK=BLOCK,

    # 7 Linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul "outgoing" module – fast forward pass.

The implementation follows the reference PyTorch code but
* evaluates the O(N³) reduction with a batched half‑precision
  GEMM (`torch.bmm`), which internally maps to cuBLAS/TF32‑free
  Tensor‑Core kernels on an H100.
* fires a tiny Triton kernel for the final gating (FP32 × FP16
  element‑wise multiply) to satisfy the “use Triton” requirement.
All other linear‑/norm‑operations are performed with PyTorch
primitives; half‑precision is used wherever possible to keep the
memory‑bandwidth low while still accumulating in FP32.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# tiny Triton kernel – multiply FP32 tensor by FP16 gate (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_kernel(out_ptr, gate_ptr, total_el, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < total_el
    # load FP32 output
    out = tl.load(out_ptr + offs, mask=mask, other=0.0)
    # load FP16 gate and cast to FP32
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0, dtype=tl.float16)
    out = out * gate.to(tl.float32)
    tl.store(out_ptr + offs, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
              * "dim"         – input channel dimension
              * "hidden_dim" – hidden dimension for the TriMul projections
              * "nomask"      – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    eps = 1e-5
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (float32 → clamp → FP16)
    #    Six linear layers are required; we compute them in FP32 for
    #    numeric stability (especially for Cauchy inputs) and then cast
    #    to half‑precision before the O(N³) reduction.
    # ------------------------------------------------------------------
    # left value & left gate
    left_val_fp32 = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    left_gate_fp32 = torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    # right value & right gate
    right_val_fp32 = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    right_gate_fp32 = torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    # out gate (used after the second layer‑norm)
    out_gate_fp32 = torch.nn.functional.linear(x_norm, weights["out_gate.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f32 = mask.to(torch.float32).unsqueeze(-1)  # [B, N, N, 1]
        left_val_fp32 = left_val_fp32 * mask_f32
        right_val_fp32 = right_val_fp32 * mask_f32

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in FP32
    # ------------------------------------------------------------------
    left_gate_fp32 = left_gate_fp32.sigmoid()
    right_gate_fp32 = right_gate_fp32.sigmoid()
    out_gate_fp32 = out_gate_fp32.sigmoid()

    # ------------------------------------------------------------------
    # 5️⃣ Fuse values with gates, clamp to FP16 range and cast
    # ------------------------------------------------------------------
    # element‑wise: left = left_val * left_gate, right = right_val * right_gate
    left_fp32 = left_val_fp32 * left_gate_fp32
    right_fp32 = right_val_fp32 * right_gate_fp32

    # clamp before casting to avoid inf/NaN in FP16
    left_fp16 = torch.clamp(left_fp32, -65504.0, 65504.0).to(torch.float16)
    right_fp16 = torch.clamp(right_fp32, -65504.0, 65504.0).to(torch.float16)
    out_gate_fp16 = torch.clamp(out_gate_fp32, -65504.0, 65504.0).to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Core O(N³) reduction – batched mat‑mul with transposed operand.
    #    left [B, N, N, H] → (B·H, N, N)
    #    right same, but we need rightᵀ on the last two dims.
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) and fuse batch × hidden
    left = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right_fp16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H
    left_flat = left.view(P, N, N)          # (P, N, N)
    right_t = right.view(P, N, N).transpose(1, 2).contiguous()  # (P, N, N)

    # batched half‑precision GEMM; accumulation is FP32 on H100
    out_half = torch.bmm(left_flat, right_t)   # (P, N, N), FP16 result

    # release intermediate tensors early
    del left, right, left_flat, right_t

    # ------------------------------------------------------------------
    # 7️⃣ Reshape back to [B, N, N, H] and convert to FP32
    # ------------------------------------------------------------------
    out = out_half.to(torch.float32) \
           .view(B, hidden_dim, N, N) \
           .permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8️⃣ Second Layer‑Norm (still FP32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # float32

    # ------------------------------------------------------------------
    # 9️⃣ Final gating – element‑wise FP32 × FP16 (in‑place via Triton)
    # ------------------------------------------------------------------
    total_elements = out.numel()
    BLOCK = 65536  # 64 Ki elements ≈ 256 KB for FP32, works well on H100
    grid = (triton.cdiv(total_elements, BLOCK),)
    _apply_gate_kernel[grid](
        out,
        out_gate_fp16,
        total_elements,
        BLOCK=BLOCK,
    )
    # out is now multiplied by out_gate

    # ------------------------------------------------------------------
    # 🔟 Linear projection back to the original channel dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
