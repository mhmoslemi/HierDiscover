import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
Outgoing TriMul (AlphaFold‑3) – fast forward implementation

Algorithm
---------
1. Layer‑norm the input tensor.
2. Two linear projections (left / right) and three gating projections
   (left_gate, right_gate, out_gate).  All linear layers are bias‑free.
3. If a mask is supplied we fuse the mask with the two element‑wise
   gates using a tiny Triton kernel that computes
        out = proj * gate * mask
   in a single pass (half‑precision).
   When no mask is present the fusion step is skipped.
4. The cubic reduction
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   is a set of independent matrix‑multiplications
   (one for each hidden‑dim slice).  We reshape the tensors so that
   the whole batch of (B·hidden_dim) matrix‑multiplies can be performed
   by a single call to `torch.bmm`, which uses cuBLAS’s strided‑batched
   GEMM and fully exploits Tensor‑Cores on an H100.
5. The result (still in fp16) is cast to fp32, Layer‑normed,
   multiplied by the output gate and finally projected back to the
   original channel dimension.

The kernel only handles the cheap element‑wise fusion; the heavy
N³ computation is delegated to cuBLAS where it is optimal.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise fusion of projection, gate and (optional) mask
# ----------------------------------------------------------------------
@triton.jit
def fuse_mul_kernel(
    proj_ptr,            # projected tensor   (fp16)
    gate_ptr,            # gate tensor        (fp16)
    mask_ptr,            # mask tensor        (fp16) – can be all‑ones
    out_ptr,             # result tensor      (fp16)
    total_elems,         # number of elements (int64)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    a = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    b = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # default 1 → no effect

    tl.store(out_ptr + offs, a * b * m, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – flatten to (B·N·N, dim)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (B·N·N, dim)

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = F.linear(x_flat, left_proj_weight).view(bs, N, N, hidden_dim)
    right_proj = F.linear(x_flat, right_proj_weight).view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid(linear))
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight)).view(bs, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Cast projections & gates to fp16 (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    left_proj_fp16 = left_proj.to(torch.float16)
    right_proj_fp16 = right_proj.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (bs, N, N) → (bs, N, N, hidden_dim) broadcasted
        mask_exp = mask.to(x.dtype).unsqueeze(-1)                 # (bs, N, N, 1)
        mask_exp = mask_exp.expand(-1, -1, -1, hidden_dim).contiguous()
        mask_fp16 = mask_exp.to(torch.float16)

        total = left_proj_fp16.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        left = torch.empty_like(left_proj_fp16)
        right = torch.empty_like(right_proj_fp16)

        fuse_mul_kernel[grid](
            left_proj_fp16,
            left_gate_fp16,
            mask_fp16,
            left,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        fuse_mul_kernel[grid](
            right_proj_fp16,
            right_gate_fp16,
            mask_fp16,
            right,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just element‑wise multiplication
        left = left_proj_fp16 * left_gate_fp16
        right = right_proj_fp16 * right_gate_fp16

    # ------------------------------------------------------------------
    # 6. Batched GEMM (left @ rightᵀ) – all in fp16
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (j,k) → (k,j)

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, N, N)                # (BH, M, K)
    right_flat = right_perm.view(BH, N, N)              # (BH, K, N) – already transposed

    # torch.bmm uses cuBLAS and fully leverages Tensor Cores on H100
    out_flat_fp16 = torch.bmm(left_flat, right_flat)    # (BH, N, N)

    # ------------------------------------------------------------------
    # 7. Restore shape (B, N, N, H) and convert to fp32
    # ------------------------------------------------------------------
    out = out_flat_fp16.view(bs, hidden_dim, N, N)      # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).to(torch.float32)    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9. Apply the output gate (still in fp32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out.view(-1, hidden_dim), to_out_weight).view(bs, N, N, dim)

    return out
