"""
Fast outgoing TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1. Layer‑norm the 4‑D pairwise tensor `x` (B,N,N,D) in fp16.
2. Fuse the five linear projections (left_proj, right_proj,
   left_gate, right_gate, out_gate) into a **single** matrix‑multiply:
       Y = X @ Wᵀ ,   X : (B·N·N, D) ,   W : (5·H, D)
   where H = hidden_dim.  This reads `x` only once instead of five
   times, dramatically cutting memory traffic.
3. Split Y into the five tensors, reshape to (B,N,N,H) and apply
   sigmoid to the three gate tensors.
4. Optional pairwise mask (broadcast over H) is applied to the two
   projection tensors.
5. Element‑wise gate the projections:  L = left_proj * left_gate,
   R = right_proj * right_gate.
6. Compute the N³ reduction
       out[b,h,i,j] = Σₖ L[b,h,i,k] * R[b,h,k,j]
   with a custom Triton kernel that performs a batched GEMM over
   the hidden dimension.  The kernel works in fp16 and accumulates
   in fp32.
7. Post‑process:  layer‑norm over the hidden dimension, multiply by
   `out_gate`, and a final linear projection back to the original
   dimension D.
All heavy arithmetic stays in fp16 (Tensor‑Core friendly) while the
final result is returned as fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the outgoing TriMul reduction.
# ----------------------------------------------------------------------
@triton.jit
def outgoing_trimul_kernel(
    left_ptr,                     # fp16  (B,H,N,N)   i‑k
    right_ptr,                    # fp16  (B,H,N,N)   k‑j
    out_ptr,                      # fp16  (B,H,N,N)   i‑j
    seq_len,                      # int N
    # left strides (in elements)
    l_str_batch, l_str_head, l_str_i, l_str_k,
    # right strides
    r_str_batch, r_str_head, r_str_k, r_str_j,
    # out strides
    o_str_batch, o_str_head, o_str_i, o_str_j,
    hidden_dim,                   # int H
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """Batched GEMM implementing
       out[b,h,i,j] = Σₖ left[b,h,i,k] * right[b,h,k,j]   (fp16→fp32)."""
    # grid: (batch*head, block_i, block_j)
    bh_id   = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # split combined id into batch and head
    b_id = bh_id // hidden_dim
    h_id = bh_id % hidden_dim

    # base offsets for the three tensors
    left_base  = b_id * l_str_batch + h_id * l_str_head
    right_base = b_id * r_str_batch + h_id * r_str_head
    out_base   = b_id * o_str_batch + h_id * o_str_head

    # tile offsets within the (i,j) plane
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # left tile (i,k)
        a_offset = (left_base
                    + offs_i[:, None] * l_str_i
                    + offs_k[None, :] * l_str_k)
        mask_a = (offs_i[:, None] < seq_len) & (offs_k[None, :] < seq_len)

        # right tile (k,j)
        b_offset = (right_base
                    + offs_k[:, None] * r_str_k
                    + offs_j[None, :] * r_str_j)
        mask_b = (offs_k[:, None] < seq_len) & (offs_j[None, :] < seq_len)

        a = tl.load(left_ptr + a_offset, mask=mask_a, other=0.0)
        b = tl.load(right_ptr + b_offset, mask=mask_b, other=0.0)

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # write‑back
    c_offset = (out_base
                + offs_i[:, None] * o_str_i
                + offs_j[None, :] * o_str_j)
    mask_c = (offs_i[:, None] < seq_len) & (offs_j[None, :] < seq_len)
    tl.store(out_ptr + c_offset, acc.to(tl.float16), mask=mask_c)

    # ------------------------------------------------------------------
    # 2) Fuse the five linear projections into one matmul
    # ------------------------------------------------------------------
    x = x.to(torch.float16)                         # (B,N,N,D) -> fp16
    x_flat = x.reshape(-1, dim)                     # (B*N*N, D)

    # fused projection + gates  (shape: (B*N*N, 5*H))
    fused_out = F.linear(x_flat, fused_weight)     # fp16 matmul via cuBLAS
    # split the fused result back to the five tensors
    splits = torch.split(fused_out, hidden_dim, dim=1)
    left_proj, right_proj, left_gate_pre, right_gate_pre, out_gate_pre = splits

    # reshape to (B,N,N,H)
    left_proj   = left_proj.reshape(bs, N, N, hidden_dim)
    right_proj  = right_proj.reshape(bs, N, N, hidden_dim)
    left_gate   = torch.sigmoid(left_gate_pre).reshape(bs, N, N, hidden_dim)
    right_gate  = torch.sigmoid(right_gate_pre).reshape(bs, N, N, hidden_dim)
    out_gate    = torch.sigmoid(out_gate_pre).reshape(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (B,N,N) -> (B,N,N,1) -> fp16
        mask = mask.to(dtype=left_proj.dtype, device=device).unsqueeze(-1)
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4) Apply element‑wise gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate          # (B,N,N,H)
    right = right_proj * right_gate        # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM (i‑k)·(k‑j)
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N) for left  (i‑k)
    left_perm = left.permute(0, 3, 1, 2)    # not contiguous
    # (B,N,N,H) -> (B,H,N,N) where axes are (k‑j)
    right_perm = right.permute(0, 3, 2, 1)  # not contiguous

    out_perm = torch.empty(
        (bs, hidden_dim, N, N), dtype=torch.float16, device=device
    )

    # ------------------------------------------------------------------
    # 6) Launch Triton kernel
    # ------------------------------------------------------------------
    # strides (in *elements*) for the three tensors
    l_str = left_perm.stride()   # (batch, head, i, k)
    r_str = right_perm.stride()  # (batch, head, k, j)
    o_str = out_perm.stride()    # (batch, head, i, j)

    # Tunable block sizes – work well on H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    num_warps = 8

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (bs * hidden_dim, num_blocks_m, num_blocks_n)

    outgoing_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,                                 # seq_len
        # left strides
        l_str[0],
        l_str[1],
        l_str[2],
        l_str[3],
        # right strides
        r_str[0],
        r_str[1],
        r_str[2],
        r_str[3],
        # out strides
        o_str[0],
        o_str[1],
        o_str[2],
        o_str[3],
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ------------------------------------------------------------------
    # 7) Reshape back to (B,N,N,H)  – result is still fp16
    # ------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out.view(-1, hidden_dim), to_out_w).view(bs, N, N, dim)

    # ------------------------------------------------------------------
    # 11) Cast to float32 for the final output
    # ------------------------------------------------------------------
    return out.to(torch.float32)
