"""
fast_outgoing_trimul_triton.py
--------------------------------
Triton‑accelerated outgoing TriMul (AlphaFold‑3) forward pass.

Key ideas
---------
1. Fuse all per‑position linear projections (left/right and the four
   gating projections) into a **single** matrix‑multiplication by stacking
   the weight matrices along the output dimension.  This removes three
   extra calls to `torch.nn.functional.linear` and cuts the amount of
   memory traffic by a factor of ~2.

2. The N³ reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,k,j,h]
   is performed by one custom Triton kernel that does a batched GEMM
   over the `(batch × hidden_dim)` dimension.  The kernel works on the
   non‑contiguous `(B,H,N,N)` layout produced by `permute`, thereby
   avoiding the costly copy that a naïve `torch.bmm` would need.

3. Optional masking and the three sigmoids are kept in FP16; the
   GEMM accumulates in FP32 for numerical stability.  The surrounding
   LayerNorms and the final linear projection are left to PyTorch
   (highly optimised cuDNN/cuBLAS kernels).

The implementation satisfies the requirement of “using at least part of
the operations in a Triton kernel” while providing a sizeable speed‑up
over the reference version.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def outgoing_trimul_kernel(
    left_ptr,            # fp16  (B, H, N, N) -- i‑k matrix
    right_ptr,           # fp16  (B, H, N, N) -- k‑j matrix
    out_ptr,             # fp16  (B, H, N, N) -- i‑j result
    seq_len,             # int   = N
    # left strides
    l_str_b, l_str_h, l_str_i, l_str_k,
    # right strides
    r_str_b, r_str_h, r_str_k, r_str_j,
    # output strides
    o_str_b, o_str_h, o_str_i, o_str_j,
    hidden_dim,          # int = H
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """Batched matrix‑multiply for the outgoing TriMul reduction.

    Computes for each (batch, head) pair:
        out[i, j] = Σ_k left[i, k] * right[k, j]
    """
    # -------------------------------------------------
    # Identify the current (batch, head) pair
    # -------------------------------------------------
    bh_id   = tl.program_id(0)                # 0 … B*H‑1
    block_i = tl.program_id(1)                # row tile index
    block_j = tl.program_id(2)                # col tile index

    b_id = bh_id // hidden_dim                # batch index
    h_id = bh_id % hidden_dim                 # head  index

    # Base pointer offsets for the three tensors (in elements)
    left_base  = b_id * l_str_b + h_id * l_str_h
    right_base = b_id * r_str_b + h_id * r_str_h
    out_base   = b_id * o_str_b + h_id * o_str_h

    # -------------------------------------------------
    # Tile offsets in the i, j dimensions
    # -------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator (FP32 for stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # Loop over the reduction dimension k
    # -------------------------------------------------
    num_k = tl.cdiv(seq_len, BLOCK_K)
    for k in range(num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # left tile: (i, k)
        a_ptr = left_ptr + (
            left_base
            + offs_i[:, None] * l_str_i
            + offs_k[None, :] * l_str_k
        )
        mask_a = (offs_i[:, None] < seq_len) & (offs_k[None, :] < seq_len)

        # right tile: (k, j)
        b_ptr = right_ptr + (
            right_base
            + offs_k[:, None] * r_str_k
            + offs_j[None, :] * r_str_j
        )
        mask_b = (offs_k[:, None] < seq_len) & (offs_j[None, :] < seq_len)

        a = tl.load(a_ptr, mask=mask_a, other=0.0)
        b = tl.load(b_ptr, mask=mask_b, other=0.0)

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # -------------------------------------------------
    # Write the result back
    # -------------------------------------------------
    c_ptr = out_ptr + (
        out_base
        + offs_i[:, None] * o_str_i
        + offs_j[None, :] * o_str_j
    )
    mask_c = (offs_i[:, None] < seq_len) & (offs_j[None, :] < seq_len)
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using Triton.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # -------------------------------------------------
    # 1) LayerNorm over the feature dimension (dim)
    # -------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # 2) Fuse all per‑position linear layers into one matmul
    # -------------------------------------------------
    # Cast to fp16 for the heavy GEMM work
    x = x.to(torch.float16)
    x_flat = x.view(-1, dim)                         # (B·N·N, dim)

    # Stack the five weight matrices (each is (hidden_dim, dim))
    combined_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,                                         # stack rows (out_features)
    ).to(torch.float16)

    # Single fused linear projection
    proj = F.linear(x_flat, combined_w)               # (B·N·N, 5·hidden_dim)

    # Slice the result into the individual tensors
    hd = hidden_dim
    left      = proj[:, 0 * hd : 1 * hd]
    right     = proj[:, 1 * hd : 2 * hd]
    left_gate = proj[:, 2 * hd : 3 * hd]
    right_gate = proj[:, 3 * hd : 4 * hd]
    out_gate  = proj[:, 4 * hd : 5 * hd]

    # Restore the original 4‑D layout (B, N, N, H)
    left      = left.view(bs, N, N, hd)
    right     = right.view(bs, N, N, hd)
    left_gate = left_gate.view(bs, N, N, hd)
    right_gate = right_gate.view(bs, N, N, hd)
    out_gate  = out_gate.view(bs, N, N, hd)

    # -------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # -------------------------------------------------
    if not nomask:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # -------------------------------------------------
    # 4) Apply the three sigmoids (gates) and element‑wise gate left/right
    # -------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 5) Batched GEMM (the N³ reduction) via Triton
    # -------------------------------------------------
    # Permute to (B, H, N, N); keep non‑contiguous to avoid a full copy.
    left_perm  = left.permute(0, 3, 1, 2)   # (B, H, N, N) – i‑k
    right_perm = right.permute(0, 3, 2, 1)  # (B, H, N, N) – k‑j

    # Output buffer in fp16 (B, H, N, N)
    out_perm = torch.empty(
        (bs, hidden_dim, N, N), dtype=torch.float16, device=device
    )

    # Strides in *elements* (not bytes)
    l_str = left_perm.stride()
    r_str = right_perm.stride()
    o_str = out_perm.stride()

    # -------------------------------------------------
    # 6) Triton launch configuration
    # -------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64          # 64 gives a good balance between register usage and K‑loop count
    num_warps = 8

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (bs * hidden_dim, num_blocks_m, num_blocks_n)

    outgoing_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,                       # seq_len
        # left strides
        l_str[0], l_str[1], l_str[2], l_str[3],
        # right strides
        r_str[0], r_str[1], r_str[2], r_str[3],
        # out strides
        o_str[0], o_str[1], o_str[2], o_str[3],
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # -------------------------------------------------
    # 7) Restore the original layout (B, N, N, H) and promote to fp32
    # -------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)   # (B, N, N, H)

    # -------------------------------------------------
    # 8) Output LayerNorm over the hidden dimension
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------
    # 9) Apply the output gate (broadcasted over dim)
    # -------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------
    # 10) Final projection back to `dim`
    # -------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.view(-1, hidden_dim), to_out_w).view(bs, N, N, dim)

    # Return the result in float32 (matching the reference implementation)
    return out.to(torch.float32)
