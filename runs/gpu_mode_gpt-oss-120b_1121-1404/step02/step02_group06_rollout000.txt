"""
Outgoing TriMul (AlphaFold‑3) – fast fused implementation

Algorithm
---------
1. Layer‑norm the input (FP32).
2. Cast the normalised tensor to fp16 and apply **one** large linear layer that
   produces all five required tensors in a single GEMM:
   * left projection
   * right projection
   * left gate   (sigmoid)
   * right gate  (sigmoid)
   * output gate (sigmoid)
3. Fuse the optional mask with the two gated projections using a tiny Triton
   kernel that computes   out = proj * gate * mask   in fp16.
4. Perform the cubic reduction (einsum) as a batched fp16 GEMM:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   which is exactly a batched matrix multiply L_d @ R_dᵀ for every hidden‑dim d.
5. Cast the result to FP32, apply a second Layer‑Norm (over ``hidden_dim``),
   multiply by the output‑gate and finally project back to ``dim`` with a
   fp16 linear layer.
6. Return the result in FP32.

Only the element‑wise mask‑gate‑proj fusion is written in Triton; the heavy
matrix multiplies are delegated to cuBLAS (torch.bmm), which on an H100
leverages Tensor‑Cores and is already optimal.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: proj * gate * mask   (all fp16)
# ----------------------------------------------------------------------
@triton.jit
def fuse_mul_kernel(
    proj_ptr,          # projected tensor   (fp16)
    gate_ptr,          # gate tensor        (fp16)
    mask_ptr,          # mask tensor        (fp16) – broadcasted
    out_ptr,           # output tensor      (fp16)
    total_elems,       # total number of elements (int64)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    # guard against out‑of‑bounds
    mask = offs < total_elems

    a = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    b = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # default 1 → no effect

    tl.store(out_ptr + offs, a * b * m, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and compute *all* five linear maps in one GEMM
    # ------------------------------------------------------------------
    #   combined_weight = [left_proj; right_proj; left_gate; right_gate; out_gate]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    combined_weight_fp16 = torch.cat(
        [
            left_proj_w,
            right_proj_w,
            left_gate_w,
            right_gate_w,
            out_gate_w,
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)

    # flatten and cast the normalised tensor
    x_flat_fp16 = x.view(-1, dim).to(torch.float16, non_blocking=True)

    # single large linear projection
    proj_all_fp16 = F.linear(x_flat_fp16, combined_weight_fp16)  # (B·N·N, 5·H)

    # reshape to (bs, N, N, 5, hidden_dim)
    proj_all_fp16 = proj_all_fp16.view(bs, N, N, 5, hidden_dim)

    left_proj_fp16 = proj_all_fp16[..., 0, :]
    right_proj_fp16 = proj_all_fp16[..., 1, :]
    left_gate_fp16 = torch.sigmoid(proj_all_fp16[..., 2, :])
    right_gate_fp16 = torch.sigmoid(proj_all_fp16[..., 3, :])
    out_gate_fp16 = torch.sigmoid(proj_all_fp16[..., 4, :])

    # ------------------------------------------------------------------
    # 3. Fuse mask + gate + projection (Triton) or simple mul if no mask
    # ------------------------------------------------------------------
    if not nomask:
        # expand mask to hidden_dim and make it contiguous fp16
        mask_fp16 = mask.to(torch.float16, non_blocking=True)
        mask_fp16 = mask_fp16.unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()

        total = left_proj_fp16.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )

        left = torch.empty_like(left_proj_fp16)
        right = torch.empty_like(right_proj_fp16)

        # left = left_proj * left_gate * mask
        fuse_mul_kernel[grid](
            left_proj_fp16,
            left_gate_fp16,
            mask_fp16,
            left,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        # right = right_proj * right_gate * mask
        fuse_mul_kernel[grid](
            right_proj_fp16,
            right_gate_fp16,
            mask_fp16,
            right,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        left = left_proj_fp16 * left_gate_fp16
        right = right_proj_fp16 * right_gate_fp16

    # ------------------------------------------------------------------
    # 4. Batched GEMM: left @ rightᵀ   (all fp16)
    # ------------------------------------------------------------------
    # change layout to (B, H, N, N) – left: (i,k), rightᵀ: (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) == Rᵀ

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)    # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)  # (B·H, N, N)

    # cuBLAS batched GEMM – uses Tensor Cores on H100
    out_mat_fp16 = torch.bmm(left_mat, right_mat)   # (B·H, N, N)

    # reshape back to (B, N, N, H) and cast to FP32 for the next layer‑norm
    out = out_mat_fp16.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32, non_blocking=True)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm + output‑gate
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # out_gate was computed in fp16 – cast to fp32 for the element‑wise product
    out_gate_fp32 = out_gate_fp16.to(torch.float32, non_blocking=True)
    out = out * out_gate_fp32

    # ------------------------------------------------------------------
    # 6. Final projection back to `dim` (fp16 GEMM + cast)
    # ------------------------------------------------------------------
    to_out_weight_fp16 = weights["to_out.weight"].to(torch.float16, non_blocking=True)

    out_fp16 = out.to(torch.float16, non_blocking=True)
    out_proj_fp16 = F.linear(out_fp16.view(-1, hidden_dim), to_out_weight_fp16)
    out_proj = out_proj_fp16.view(bs, N, N, dim).to(torch.float32, non_blocking=True)

    return out_proj
