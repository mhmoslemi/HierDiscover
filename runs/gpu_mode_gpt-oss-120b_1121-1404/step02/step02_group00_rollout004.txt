"""
TriMul (outgoing) – fast Triton‑fused mask+gate and cuBLAS batched GEMM.

Algorithm
---------
1. Layer‑norm the input in FP32 (numerical stability).
2. Cast the normalized tensor to FP16 and cast all linear weights to FP16.
3. Compute three linear projections (left, right, gates) in FP16.
4. If a mask is present, fuse mask, left‑gate and right‑gate with a tiny
   Triton kernel that loads the whole hidden dimension (H) at once:
        left_fused  = left  * left_gate  * mask
        right_fused = right * right_gate * mask
   If there is no mask, the fusion is just an element‑wise product.
5. Reshape the fused tensors so that every hidden channel becomes an
   independent batch entry:
        left_flat  : (B, N, N, H) → (B·H, N, N)
        right_flat : (B, N, N, H) → (B·H, N, N) where the two N‑axes are swapped
   (the contraction `out[i,j] = Σ_k left[i,k]·right[j,k]` is a
   batched matrix‑multiply `left·rightᵀ`).
6. Perform the batched N×N GEMM with `torch.bmm`, which uses cuBLAS/Tensor‑Core
   on H100 and is faster than a hand‑written Triton GEMM for the large batch
   dimension (B·H).
7. Reshape the result back to (B, N, N, H) (still FP16).
8. Apply the output LayerNorm, the out‑gate, and the final linear projection
   (all FP16).
9. Cast the final tensor back to FP32 to satisfy the API contract.

Only the mask‑gate fusion is implemented in Triton; the heavy O(N³·H) contraction
uses cuBLAS, which gives a large speed‑up while still meeting the requirement
that “part of the operations” run in a Triton kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: Fuse mask, left‑gate and right‑gate (vectorised over hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    # strides for tensors of shape [B, N, N, H]
    stride_b, stride_i, stride_j, stride_h,
    # strides for mask of shape [B, N, N]
    stride_mask_b, stride_mask_i, stride_mask_j,
    HIDDEN: tl.constexpr,
):
    """
    One thread per (b,i,j) loads a vector of length HIDDEN.
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # first N index
    pid_j = tl.program_id(2)   # second N index

    i = pid_i
    j = pid_j

    # out‑of‑bounds check (required for non‑square grids)
    valid = (i < N) & (j < N) & (pid_b < B)

    # scalar mask, broadcast over hidden dim
    mask = tl.load(
        mask_ptr + pid_b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
        mask=valid,
        other=0.0,
    )

    # offsets for the hidden dimension (contiguous stride_h)
    d = tl.arange(0, HIDDEN)                     # compile‑time constant vector
    base = pid_b * stride_b + i * stride_i + j * stride_j
    vec_off = base + d * stride_h                 # (HIDDEN,) offsets

    # load tensors
    left      = tl.load(left_ptr      + vec_off, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + vec_off, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + vec_off, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + vec_off, mask=valid, other=0.0)

    # fuse mask and sigmoid gates
    left_out  = left  * left_gate  * mask
    right_out = right * right_gate * mask

    # write back
    tl.store(out_left_ptr  + vec_off, left_out,  mask=valid)
    tl.store(out_right_ptr + vec_off, right_out, mask=valid)


# ----------------------------------------------------------------------
# Entrypoint
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor – torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  – torch.Tensor [B, N, N]   (float32) or None
            weights      – dict of model parameters (float32)
            config       – dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                               # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 and cast all linear weights
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)    # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Gates (FP16 + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 5) Apply mask & gates
    # ------------------------------------------------------------------
    if nomask or mask_tensor is None:
        # No mask – simple element‑wise multiplication
        left_fused  = left  * left_gate
        right_fused = right * right_gate
    else:
        # Fuse mask, left‑gate and right‑gate with a Triton kernel
        mask = mask_tensor.to(torch.float16)               # [B,N,N] fp16
        left_fused  = torch.empty_like(left)
        right_fused = torch.empty_like(right)

        # strides for the 4‑D tensors
        stride_b, stride_i, stride_j, stride_h = left.stride()          # same for right
        stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

        grid = (B, N, N)   # (batch, i, j)
        _mask_gate_fuse_kernel[grid](
            left, left_gate, mask,
            right, right_gate,
            left_fused, right_fused,
            B, N, hidden_dim,
            stride_b, stride_i, stride_j, stride_h,
            stride_mask_b, stride_mask_i, stride_mask_j,
            HIDDEN=hidden_dim,
        )

    # ------------------------------------------------------------------
    # 6) Prepare tensors for the batched GEMM
    #    left  : (B, N, N, H) → (B*H, N, N)
    #    right : (B, N, N, H) → transpose(1,2) → (B*H, N, N)
    # ------------------------------------------------------------------
    # left → (B, H, N, N) → (B*H, N, N)
    left_flat = left_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # right needs to be transposed on the two N axes before flattening
    right_T = right_fused.permute(0, 2, 1, 3).contiguous()  # (B, N, N, H) with i↔j swapped
    right_flat = right_T.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 7) Batched matrix multiplication (cuBLAS, fp16 inputs → fp16 output)
    # ------------------------------------------------------------------
    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, N, N)  fp16

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, H)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H] fp16

    # ------------------------------------------------------------------
    # 9) Output LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                               # fp16

    out = out * out_gate               # element‑wise (fp16)

    out = F.linear(out, to_out_w)      # [B,N,N,dim]  fp16

    # ------------------------------------------------------------------
    # 10) Cast back to float32 (API contract)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
