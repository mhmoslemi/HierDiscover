"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
========================================================

The only cubic work in the module is the batched matrix multiplication
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]               (1)
for every hidden channel `h`.  By reshaping the tensors to
    (B·H, N, N) we obtain a batch of H·B independent GEMMs.
We compute (1) with a custom Triton kernel that evaluates
    C = A @ Bᵀ
where `A` holds the *left* tensor and `B` holds the *right* tensor.
`B` is accessed as its transpose simply by swapping the K/N strides,
hence no temporary copy is required.

All other operations (LayerNorm, linear projections, gating) are done
with ordinary PyTorch primitives – they are negligible compared with the
N³ work.  The kernel works for any mask (or none) and for all shapes
listed in the statement, and runs fully in FP16 with a FP32 accumulator.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # tensor pointers
    batch, M, N, K,                          # dimensions
    stride_ah, stride_am, stride_ak,         # A strides (batch, row, col)
    stride_bh, stride_bk, stride_bn,         # B strides (batch, col, row) – note swapped K/N
    stride_ch, stride_cm, stride_cn,         # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute   C = A @ Bᵀ   for a batch of matrices.

    A : [batch, M, K]    (fp16)
    B : [batch, K, N]    (fp16)   – accessed as its transpose
    C : [batch, M, N]    (fp32 accumulator)
    """

    pid_batch = tl.program_id(0)   # which (b·h) slice
    pid_m     = tl.program_id(1)   # row tile
    pid_n     = tl.program_id(2)   # col tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the beginning of the current tile
    a_tile_ptr = a_ptr + pid_batch * stride_ah \
                       + offs_m[:, None] * stride_am \
                       + offs_k[None, :] * stride_ak
    b_tile_ptr = b_ptr + pid_batch * stride_bh \
                       + offs_k[:, None] * stride_bk \
                       + offs_n[None, :] * stride_bn

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles (ceil)
    num_k = tl.cdiv(K, BLOCK_K)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for k_iter in range(num_k):
        # Global K offset for the current tile (used only for masking)
        k_offset = k_iter * BLOCK_K

        # Build masks for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & ((k_offset + offs_k) < K)[None, :]
        b_mask = ((k_offset + offs_k) < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_tile_ptr, mask=a_mask, other=0.0)   # (BLOCK_M, BLOCK_K) fp16
        b = tl.load(b_tile_ptr, mask=b_mask, other=0.0)   # (BLOCK_K, BLOCK_N) fp16

        acc += tl.dot(a, b)   # FP32 accumulation

        # Advance pointers for the next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store the result (with bounds check)
    # ------------------------------------------------------------------
    c_tile_ptr = c_ptr + pid_batch * stride_ch \
                         + offs_m[:, None] * stride_cm \
                         + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_tile_ptr, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for
    the cubic matrix‑multiplication step.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim] (float32)
        - mask         : torch.Tensor  [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensor containing all model params
        - config       : dict with keys ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1) Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden_dim = cfg["hidden_dim"]

    # ------------------------------------------------------------------
    # 2) Load weights (all are 2‑D tensors, no bias for the first four
    #    Linear layers)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 3) Input LayerNorm (dim → dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 4) Linear projections to hidden dimension (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B, N, N, H]
    right = F.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6) Gating (sigmoid) + element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B, N, N, H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Cast to fp16 for the heavy GEMM
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 8) Reshape to (B·H, N, N) – each hidden slice becomes an
    #    independent matrix multiplication.
    #    Right is kept in “row‑major” order; the kernel treats it as
    #    transposed by swapping its K/N strides.
    # ------------------------------------------------------------------
    left_mat  = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Output buffer (FP32 accumulator)
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 9) Choose block sizes.  The values below give good occupancy on an
    #    H100 for the whole range of N we have to support.
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    elif N <= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64
    else:   # N up to 1024
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 64

    # Strides (in element units, not bytes)
    stride_ah = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    # For B we swap the K / N strides so that the kernel sees Bᵀ
    stride_bh = right_mat.stride(0)
    stride_bk = right_mat.stride(2)   # original column stride → K stride
    stride_bn = right_mat.stride(1)   # original row    stride → N stride

    stride_ch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # ------------------------------------------------------------------
    # 10) Launch the Triton kernel
    # ------------------------------------------------------------------
    grid = (
        B * hidden_dim,                       # number of matrix pairs
        (N + BLOCK_M - 1) // BLOCK_M,         # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,         # tiles along N (j)
    )
    _batched_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        B * hidden_dim, N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 11) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 12) Output LayerNorm, gating and final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)      # [B, N, N, dim]

    return out
