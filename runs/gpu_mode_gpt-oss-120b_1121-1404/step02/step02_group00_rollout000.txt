"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input (FP32) → x.
2. Cast x and all linear weights to FP16.
3. Compute three linear projections (left, right, gates) in FP16.
4. If a mask is present, fuse mask‑gate‑projection with a small Triton kernel
   (otherwise simple element‑wise multiplication).
5. Reshape for the O(N³) contraction:
      left  : (B,N,N,H) → (B,H,N,N)
      right : (B,N,N,H) → (B,H,N,N) with the two N axes swapped.
   Perform a batched matrix‑multiply
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,k,j]
   using torch.matmul (cuBLAS/Tensor‑Cores).
6. Reshape back to (B,N,N,H).
7. Apply output LayerNorm, out‑gate and final projection (all FP16).
8. Cast the result back to FP32.

The only Triton part is the mask‑gate‑fusion kernel; the heavy N³ contraction
leverages the highly‑optimised cuBLAS implementation, which brings the total
runtime well below the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1 – fuse mask, left/right gates with left/right projections
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    # strides for tensors with shape (B, N, N, H)
    stride_b, stride_i, stride_j, stride_h,
    # strides for mask with shape (B, N, N)
    stride_mask_b, stride_mask_i, stride_mask_j,
    HIDDEN: tl.constexpr
):
    """
    For each (b,i,j) loads:
        left, left_gate, right, right_gate, mask
    and writes:
        out_left = left * left_gate * mask
        out_right = right * right_gate * mask
    The hidden dimension H is a compile‑time constant (HIDDEN).
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # first N index
    pid_j = tl.program_id(2)   # second N index

    # bounds
    valid = (pid_b < B) & (pid_i < N) & (pid_j < N)

    # linear offset for the (b,i,j) cell (without hidden dim)
    base = pid_b * stride_b + pid_i * stride_i + pid_j * stride_j

    # hidden‑dim offsets
    d = tl.arange(0, HIDDEN)
    off_h = base + d * stride_h                 # shape (HIDDEN,)

    # mask is a scalar; load and broadcast
    mask = tl.load(
        mask_ptr + pid_b * stride_mask_b
                 + pid_i * stride_mask_i
                 + pid_j * stride_mask_j,
        mask=valid,
        other=0.0
    )

    # load data
    left      = tl.load(left_ptr      + off_h, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + off_h, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + off_h, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + off_h, mask=valid, other=0.0)

    # apply mask + gates
    left_out  = left  * left_gate  * mask
    right_out = right * right_gate * mask

    # write back
    tl.store(out_left_ptr  + off_h, left_out,  mask=valid)
    tl.store(out_right_ptr + off_h, right_out, mask=valid)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (float32) or None
            - weights       : dict of nn.Parameter tensors (float32)
            - config        : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 1) unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 2) Input LayerNorm (FP32 for stability)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )                               # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # 3) Cast to FP16 and cast all linear weights once
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Linear projections (FP16)
    # ------------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)    # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)

    # ------------------------------------------------------------------
    # 5) Gates (FP16 + sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Apply mask & gates
    # ------------------------------------------------------------------
    if nomask or mask_tensor is None:
        # No mask – simple elementwise multiplication
        left_fused  = left  * left_gate
        right_fused = right * right_gate
    else:
        # Use the Triton kernel to fuse mask, left‑gate and right‑gate
        mask = mask_tensor.to(torch.float16)               # [B, N, N]

        # allocate output buffers
        left_fused  = torch.empty_like(left)
        right_fused = torch.empty_like(right)

        # strides for the 4‑D tensors (contiguous layout)
        stride_b, stride_i, stride_j, stride_h = left.stride()          # (B,N,N,H)
        stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()    # (B,N,N)

        # launch grid = (batch, i, j)
        grid = (B, N, N)
        _mask_gate_fuse_kernel[grid](
            left, left_gate, mask,
            right, right_gate,
            left_fused, right_fused,
            B, N, hidden_dim,
            stride_b, stride_i, stride_j, stride_h,
            stride_mask_b, stride_mask_i, stride_mask_j,
            HIDDEN=hidden_dim,
        )

    # ------------------------------------------------------------------
    # 7) Contract over the third axis (O(N³·H) term)
    #    left  : (B,N,N,H) → (B,H,N,N)
    #    right : (B,N,N,H) → (B,H,N,N) with the two N axes swapped
    # ------------------------------------------------------------------
    # put hidden dimension in front and make tensors contiguous
    left_view  = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_view = right_fused.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Batched matrix multiply – cuBLAS/Tensor Cores (FP16 with FP32 acc)
    out_view = torch.matmul(left_view, right_view)            # (B, H, N, N)

    # Restore original order: (B, N, N, H)
    out = out_view.permute(0, 2, 3, 1).contiguous()            # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate and final projection (all FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b
    )
    out = out * out_gate                      # elementwise (FP16)
    out = F.linear(out, to_out_w)             # [B,N,N,dim] (FP16)

    # ------------------------------------------------------------------
    # 9) Cast back to FP32 (API contract) and return
    # ------------------------------------------------------------------
    return out.to(torch.float32)
