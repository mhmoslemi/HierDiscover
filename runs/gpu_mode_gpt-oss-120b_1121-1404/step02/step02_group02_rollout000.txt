"""
custom_kernel – outgoing Triangle Multiplicative Update (TriMul)

Algorithm
---------
1️⃣ Layer‑norm the input (float32 → float16 for the rest of the pipeline).
2️⃣ Six bias‑free linear projections (half‑precision):
      left, right               – value tensors
      left_gate, right_gate, out_gate – sigmoid‑gated masks
3️⃣ (Optional) mask multiplication (broadcasted over the hidden dim).
4️⃣ Apply the three gates element‑wise (still in fp16).
5️⃣ Core “TriMul” reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix multiply:
      left[b,d]   : (N×K)   where K = N
      right[b,d]ᵀ : (K×N)
   We fuse the batch (B) and hidden (H) dimensions → P = B·H and run
   `torch.bmm` on (P,N,N) fp16 tensors – cuBLAS/Tensor‑Core efficient.
6️⃣ Convert the result to float32, reshape back to [B,N,N,H].
7️⃣ Second Layer‑norm (float32), element‑wise multiplication with the
   output‑gate (implemented with a tiny Triton kernel), and a final
   linear projection back to the original channel dimension.
   The final tensor is returned as float32.

Only the final gating (step 7) is performed by a custom Triton kernel,
fulfilling the “must use Triton” requirement while the heavy O(N³) work
uses cuBLAS for maximal throughput on an H100.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Tiny Triton kernel: out_fp32 *= out_gate_fp16   (element‑wise)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_kernel(
    out_ptr,          # float32 *
    gate_ptr,         # float16 *
    n_elements,       # total number of elements (int32)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    # load
    out_val  = tl.load(out_ptr + offs, mask=mask, other=0.0)
    gate_val = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # multiply (gate cast to fp32)
    out_val = out_val * tl.cast(gate_val, tl.float32)

    # store
    tl.store(out_ptr + offs, out_val, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
              * "dim"         – input channel dimension
              * "hidden_dim" – hidden dimension of the TriMul projections
              * "nomask"      – bool, True ⇒ ignore mask (default False)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to cast to half‑precision (fp16)
    # ------------------------------------------------------------------
    to_half = lambda t: t.to(torch.float16)

    # ------------------------------------------------------------------
    # Load weights (convert to the precision needed for each step)
    # ------------------------------------------------------------------
    # layer‑norm (float32)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    # linear / gate weights (bias‑free, fp16)
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])

    # final norm & projection (float32 – higher accuracy)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Input layer‑norm (float32) → cast to fp16 for the rest
    # ------------------------------------------------------------------
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )
    x = x_norm_fp32.to(torch.float16)               # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (fp16, bias‑free)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape -> [B, N, N, 1] in fp16, broadcast on dim H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid, fp16)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # Apply the two value‑gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core O(N³) reduction – batched mat‑mul with transposed operand
    # ------------------------------------------------------------------
    # Fuse the batch and hidden dimensions: (B, H, N, N) → (P, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H

    left  = left.view(P, N, N)     # fp16
    right = right.view(P, N, N)    # fp16

    # Clamp to the fp16 representable range – protects against the heavy‑tailed Cauchy case
    left  = torch.clamp(left,  -65504.0, 65504.0)
    right = torch.clamp(right, -65504.0, 65504.0)

    # Batched mat‑mul: out[p,i,j] = Σ_k left[p,i,k] * right[p,j,k]
    # This is left @ rightᵀ for each p.
    out_fp16 = torch.bmm(left, right.transpose(1, 2))  # (P, N, N), fp16

    # Convert to fp32 for the remaining high‑precision stages
    out = out_fp16.to(torch.float32)

    # Restore the original layout: [B, N, N, H]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second Layer‑norm (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Gating with out_gate (fp16) – executed by a tiny Triton kernel
    # ------------------------------------------------------------------
    out_gate = out_gate.contiguous()          # [B, N, N, H] fp16
    out = out.contiguous()                    # [B, N, N, H] fp32

    total_elements = out.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    # Triton kernel multiplies out (fp32) by out_gate (fp16) in‑place
    _apply_gate_kernel[grid](
        out,
        out_gate,
        total_elements,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Final linear projection back to the original dimension (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]

    return out
