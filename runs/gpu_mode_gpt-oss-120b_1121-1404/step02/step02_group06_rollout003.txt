def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1. Input LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # flatten
    x_flat = x.reshape(-1, dim)  # (B*N*N, dim)

    # 2. Combined projection and gates via single matmul
    # Concatenate weights: left_proj, left_gate, right_proj, right_gate
    w_left_proj = weights["left_proj.weight"]          # (hidden_dim, dim)
    w_left_gate = weights["left_gate.weight"]          # (hidden_dim, dim)
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    weight_all = torch.cat([w_left_proj, w_left_gate, w_right_proj, w_right_gate], dim=0)  # (4*hidden_dim, dim)
    # Optionally cast to half for faster compute
    weight_all_fp16 = weight_all.to(torch.float16)

    # Compute combined projection
    # Use torch.nn.functional.linear with half input
    x_flat_fp16 = x_flat.to(torch.float16)
    proj_all_fp16 = torch.nn.functional.linear(x_flat_fp16, weight_all_fp16)  # (M, 4*hidden_dim)
    # split
    split = torch.split(proj_all_fp16, hidden_dim, dim=1)
    left_proj_fp16, left_gate_fp16, right_proj_fp16, right_gate_fp16 = split

    # 3. Out gate
    out_gate_weight = weights["out_gate.weight"]
    out_gate_weight_fp16 = out_gate_weight.to(torch.float16)
    out_gate_fp16 = torch.nn.functional.linear(x_flat_fp16, out_gate_weight_fp16)
    out_gate = torch.sigmoid(out_gate_fp16).to(torch.float32)  # back to fp32 for later

    # 4. Apply sigmoid to gates (left_gate, right_gate)
    left_gate_fp16 = torch.sigmoid(left_gate_fp16)
    right_gate_fp16 = torch.sigmoid(right_gate_fp16)

    # 5. Optional mask fusion using Triton
    if not nomask:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1).expand(-1, -1, -1, hidden_dim).contiguous()
        total = left_proj_fp16.numel()
        BLOCK = 1024
        grid = ( (total + BLOCK - 1) // BLOCK, )
        left = torch.empty_like(left_proj_fp16)
        right = torch.empty_like(right_proj_fp16)
        fuse_mul_kernel[grid](
            left_proj_fp16,
            left_gate_fp16,
            mask_fp16,
            left,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        fuse_mul_kernel[grid](
            right_proj_fp16,
            right_gate_fp16,
            mask_fp16,
            right,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        left = left_proj_fp16 * left_gate_fp16
        right = right_proj_fp16 * right_gate_fp16

    # reshape to (B,N,N,hidden_dim)
    left = left.view(bs, N, N, hidden_dim)
    right = right.view(bs, N, N, hidden_dim)

    # 6. Batched matmul
    left_perm = left.transpose(1, 3)   # (B, H, N, N)
    right_perm = right.transpose(1, 3) # (B, H, N, N)
    # Compute out_perm = left_perm @ right_perm^T  (batched)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, N, N)
    # Transpose back to (B, N, N, H)
    out = out_perm.transpose(1, 3)  # (B, N, N, H)

    # 7. Output layernorm (in float32)
    out = out.to(torch.float32)
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # 8. Apply out_gate (out_gate shape (bs, N, N, hidden_dim))
    out_gate = out_gate.view(bs, N, N, hidden_dim)  # reshape
    out = out * out_gate

    # 9. Final projection
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out.view(-1, hidden_dim), to_out_weight).view(bs, N, N, dim)

    return out

"""
Optimized outgoing TriMul implementation.

Key ideas
---------
* Fuse the four linear layers (left/right projections + their gates) into a
  single half‑precision GEMM.  This reads the large input tensor only once
  and reduces kernel launch overhead.
* Apply the sigmoid gates in FP16 and fuse the optional mask * gate *
  multiplication with a tiny Triton kernel (no extra temporaries).
* Perform the cubic reduction with one batched `torch.matmul` on the
  (B·hidden_dim) batch dimension – no costly reshapes or copies.
* All heavy arithmetic is in float16 to exploit Tensor‑Cores on an H100;
  the final output is converted back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fuse_mul_kernel(
    proj_ptr,        # projected tensor (fp16)
    gate_ptr,        # gate tensor (fp16)
    mask_ptr,        # mask tensor (fp16) – element‑wise 0/1
    out_ptr,         # destination (fp16)
    total_elems,     # number of elements (int64)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    a = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    b = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # 1 → neutral when mask not used

    tl.store(out_ptr + offs, a * b * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Combine four linear maps (left/right proj + gates) into one GEMM
    # ------------------------------------------------------------------
    # flatten to (B·N·N, dim)
    x_flat = x.reshape(-1, dim)                     # (M, dim),  M = bs * N * N

    # concatenate the four weight matrices (out_features stacked)
    w_left_proj = weights["left_proj.weight"]       # (hidden_dim, dim)
    w_left_gate = weights["left_gate.weight"]       # (hidden_dim, dim)
    w_right_proj = weights["right_proj.weight"]     # (hidden_dim, dim)
    w_right_gate = weights["right_gate.weight"]     # (hidden_dim, dim)

    # shape -> (4*hidden_dim, dim)  – half precision for speed
    weight_all_fp16 = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate], dim=0
    ).to(torch.float16)

    # half‑precision input
    x_flat_fp16 = x_flat.to(torch.float16)

    # one big linear: (M, 4*hidden_dim)
    proj_all_fp16 = F.linear(x_flat_fp16, weight_all_fp16)   # bias is None

    # split into the four components
    left_proj_fp16, left_gate_fp16, right_proj_fp16, right_gate_fp16 = torch.split(
        proj_all_fp16, hidden_dim, dim=1
    )

    # ------------------------------------------------------------------
    # 3. Output gate (separate linear)
    # ------------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"]         # (hidden_dim, dim)
    out_gate_fp16 = F.linear(x_flat_fp16, w_out_gate.to(torch.float16))
    out_gate = torch.sigmoid(out_gate_fp16).to(torch.float32)   # keep fp32 for later mul

    # ------------------------------------------------------------------
    # 4. Sigmoid on the gate tensors (still fp16)
    # ------------------------------------------------------------------
    left_gate_fp16 = torch.sigmoid(left_gate_fp16)
    right_gate_fp16 = torch.sigmoid(right_gate_fp16)

    # ------------------------------------------------------------------
    # 5. Optional mask + element‑wise fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (bs, N, N) → (bs, N, N, hidden_dim) broadcasted
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        mask_fp16 = mask_fp16.expand(-1, -1, -1, hidden_dim).contiguous()

        total = left_proj_fp16.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        left = torch.empty_like(left_proj_fp16)
        right = torch.empty_like(right_proj_fp16)

        fuse_mul_kernel[grid](
            left_proj_fp16,
            left_gate_fp16,
            mask_fp16,
            left,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        fuse_mul_kernel[grid](
            right_proj_fp16,
            right_gate_fp16,
            mask_fp16,
            right,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        left = left_proj_fp16 * left_gate_fp16
        right = right_proj_fp16 * right_gate_fp16

    # ------------------------------------------------------------------
    # 6. Reshape to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    left = left.view(bs, N, N, hidden_dim)
    right = right.view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7. Batched GEMM: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    Implemented as a single torch.matmul over the (B·hidden_dim) batch.
    # ------------------------------------------------------------------
    # bring hidden dimension to the batch axis: (B, H, N, N)
    left_perm = left.transpose(1, 3)    # (B, H, N, N)
    right_perm = right.transpose(1, 3)  # (B, H, N, N)

    # matmul with transpose on the last two dims of right
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))   # (B, H, N, N)

    # back to (B, N, N, H)
    out = out_perm.transpose(1, 3)     # (B, N, N, H)
    out = out.to(torch.float32)        # layer‑norm works in FP32

    # ------------------------------------------------------------------
    # 8. Output LayerNorm
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9. Apply the output gate (still in FP32)
    # ------------------------------------------------------------------
    out_gate = out_gate.view(bs, N, N, hidden_dim)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out.view(-1, hidden_dim), weights["to_out.weight"]).view(
        bs, N, N, dim
    )

    return out
