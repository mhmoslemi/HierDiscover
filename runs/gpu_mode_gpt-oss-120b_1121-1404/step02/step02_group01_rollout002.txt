"""
TriMul (outgoing) – Triton‑accelerated implementation.

The forward pass consists of:
1️⃣ LayerNorm on the input (float32).
2️⃣ Half‑precision linear projections (left/right) and three sigmoid gates.
3️⃣ Optional element‑wise mask on the projected tensors.
4️⃣ **Fused N³ outer‑product** for every (batch, hidden) pair:
       out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
   This is implemented as a custom Triton GEMM (A @ Bᵀ) where each (b,h)
   page is treated independently.  The kernel works on the original
   `[B, N, N, H]` layout – no costly `permute/contiguous` copy is required.
   The computation uses FP16 inputs and FP32 accumulation (Tensor‑cores).
5️⃣ LayerNorm over the hidden dimension, output‑gate multiplication and
   final linear projection back to the original `dim`.

All heavy arithmetic runs in half precision; only the two LayerNorms and the
final linear layer use FP32.  The implementation avoids the O(B·H·N²) copy
present in naïve versions, making the kernel memory‑bound and fast enough
to meet the 1 ms target on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    a_ptr,                     # const half*   (left)
    b_ptr,                     # const half*   (right, accessed as transposed)
    c_ptr,                     # float*        (output)
    B, H, N,                   # runtime scalars
    stride_az, stride_ah, stride_ai, stride_ak,   # A strides (page, hidden, row, col)
    stride_bz, stride_bh, stride_bk, stride_bn,   # B strides (page, hidden, row (=k), col (=j))
    stride_cz, stride_ci, stride_cj,               # C strides (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a block of C = A @ Bᵀ for a single (batch, hidden) page.

    A : (B, N, N, H)   – row‑major, accessed as A[b,i,k,h]
    B : (B, N, N, H)   – row‑major, accessed as B[b,j,k,h] → Bᵀ uses k as row, j as col
    C : (B·H, N, N)    – row‑major, FP32 accumulator
    """
    pid_z = tl.program_id(0)      # which (batch, hidden) page  (0 … B*H-1)
    pid_m = tl.program_id(1)      # block row
    pid_n = tl.program_id(2)      # block col

    # ------------------------------------------------------------------
    # decode batch & hidden indices from the linear page id
    b_idx = pid_z // H
    h_idx = pid_z % H

    # base pointers for this page
    a_base = a_ptr + b_idx * stride_az + h_idx * stride_ah
    b_base = b_ptr + b_idx * stride_bz + h_idx * stride_bh
    c_base = c_ptr + pid_z * stride_cz

    # ------------------------------------------------------------------
    # tile offsets inside the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    cur_k = 0
    while cur_k < N:
        k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = k_offs < N

        # ----- load a tile of A (row‑major, shape BLOCK_M×BLOCK_K) -----
        a_ptrs = a_base + \
                 offs_m[:, None] * stride_ai + \
                 k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                # half → fp16

        # ----- load a tile of Bᵀ (shape BLOCK_K×BLOCK_N) -----
        b_ptrs = b_base + \
                 k_offs[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                # half

        # ----- FP16×FP16 → FP32 dot product -----
        acc += tl.dot(a, b)

        cur_k += BLOCK_K

    # ------------------------------------------------------------------
    # write back the computed block
    c_ptrs = c_base + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

    # ------------------------------------------------------------------ FP32 LayerNorm on the input
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------ cast to half for the heavy Linear ops
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------ Pre‑cast the heavy weights to fp16 once
    w_fp16 = {
        "left_proj.weight":   weights["left_proj.weight"].to(torch.float16),
        "right_proj.weight":  weights["right_proj.weight"].to(torch.float16),
        "left_gate.weight":   weights["left_gate.weight"].to(torch.float16),
        "right_gate.weight":  weights["right_gate.weight"].to(torch.float16),
        "out_gate.weight":    weights["out_gate.weight"].to(torch.float16),
    }

    # ------------------------------------------------------------------ Linear projections (no bias)
    left = F.linear(x_norm_h, w_fp16["left_proj.weight"])   # [B, N, N, hidden] (fp16)
    right = F.linear(x_norm_h, w_fp16["right_proj.weight"])  # same shape

    # ------------------------------------------------------------------ Apply optional mask (half precision)
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)       # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------ Gates (half→half after sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["out_gate.weight"]))

    # ------------------------------------------------------------------ Apply the two channel‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ Prepare strides for the Triton kernel (all in *elements*)
    # layout of left/right: (B, N, N, H)  – contiguous in H
    stride_az = left.stride(0)   # B‑stride  = N * N * hidden
    stride_ah = left.stride(3)   # hidden‑stride = 1
    stride_ai = left.stride(1)   # row‑stride = N * hidden
    stride_ak = left.stride(2)   # col‑stride = hidden

    stride_bz = right.stride(0)
    stride_bh = right.stride(3)
    stride_bk = right.stride(2)   # k‑dim stride  (row of Bᵀ)
    stride_bn = right.stride(1)   # n‑dim stride  (col of Bᵀ)

    # ------------------------------------------------------------------ Allocate output buffer (float32, contiguous = (B*H, N, N))
    out_mat = torch.empty((B * hidden, N, N), dtype=torch.float32, device=x.device)

    stride_cz = out_mat.stride(0)   # page stride = N * N
    stride_ci = out_mat.stride(1)   # row stride = N
    stride_cj = out_mat.stride(2)   # col stride = 1

    # ------------------------------------------------------------------ Tile size selection
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    grid = (B * hidden,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # ------------------------------------------------------------------ Launch the Triton kernel
    _trimul_kernel[grid](
        left,
        right,
        out_mat,
        B,
        hidden,
        N,
        stride_az, stride_ah, stride_ai, stride_ak,
        stride_bz, stride_bh, stride_bk, stride_bn,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps give good occupancy on H100
        num_stages=2,         # double‑buffered loads
    )

    # ------------------------------------------------------------------ Reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)   # float32

    # ------------------------------------------------------------------ LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------ Apply output gate (cast to FP32 for safety)
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------ Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
```
