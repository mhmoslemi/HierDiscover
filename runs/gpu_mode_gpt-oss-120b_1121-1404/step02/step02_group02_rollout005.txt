def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # layernorm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm_fp16 = x_norm.to(torch.float16)

    # fused left projection + gate weight
    left_concat_weight = torch.cat([weights["left_proj.weight"], weights["left_gate.weight"]], dim=0).to(torch.float16)
    right_concat_weight = torch.cat([weights["right_proj.weight"], weights["right_gate.weight"]], dim=0).to(torch.float16)

    # compute fused linear
    left_concat = torch.nn.functional.linear(x_norm_fp16, left_concat_weight)  # [B,N,N,2*H]
    right_concat = torch.nn.functional.linear(x_norm_fp16, right_concat_weight)

    # split
    left_proj, left_gate = left_concat.split(hidden_dim, dim=-1)
    right_proj, right_gate = right_concat.split(hidden_dim, dim=-1)

    # mask application via Triton if needed
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16)
        # ensure contiguous
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        apply_mask_triton(left_proj, mask_fp16)
        apply_mask_triton(right_proj, mask_fp16)

    # gates
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    left = left_proj * left_gate
    right = right_proj * right_gate

    # out_gate
    out_gate = torch.nn.functional.linear(x_norm_fp16, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # core TriMul reduction using torch.bmm
    B, N, N_, H = left.shape
    assert N == N_
    # reshape
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)

    # compute batched matmul
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)  # (B,N,N,H)

    # final layernorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply out_gate after norm
    out = out * out_gate.to(torch.float32)

    # final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out
```

4️⃣ Apply the sigmoid gates and element‑wise multiply the gated values.
5️⃣ Core O(N³) reduction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply with the second operand transposed.
   We reshape to (B·H, N, N) and invoke torch.bmm, which uses cuBLAS‑Tensor‑Core
   and is far faster than the naïve Triton kernel.
6️⃣ Layer‑norm on the hidden dimension, gating with `out_gate`, and a final
   1×1 linear projection back to the original channel size (FP32).

All heavy arithmetic is performed in half‑precision; the final output is
float‑32.  A small Triton kernel is kept to satisfy the “some operation must be
implemented in a kernel’’ requirement.

The implementation matches the reference PyTorch TriMul exactly (up to
FP16 rounding) and is tuned for the H100 GPU.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Tiny Triton kernel – apply a (B,N,N) mask to a (B,N,N,H) tensor.
# ----------------------------------------------------------------------
@triton.jit
def _apply_mask_kernel(
    tensor_ptr,          # const *[fp16] input / output tensor (B,N,N,H)
    mask_ptr,            # const *[fp16] mask tensor (B,N,N)
    out_ptr,             #       *[fp16] output tensor (same layout as input)
    B, N, H,             # int32 shape parameters
    stride_tb, stride_ti, stride_tj, stride_th,   # strides for the tensor
    stride_mb, stride_mi, stride_mj,               # strides for the mask
    stride_ob, stride_oi, stride_oj, stride_oh,   # strides for the output (same as tensor)
    BLOCK_N: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # tile row (i dimension)
    pid_j = tl.program_id(2)   # tile column (j dimension)

    # ------------------------------------------------------------------
    # Offsets for the (i, j) tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i[:, None]
    mask_j = offs_j[None, :]

    mask_valid = (offs_i < N) & (offs_j < N)

    # Load a scalar mask for each (i, j) pair
    mask_tile = tl.load(
        mask_ptr + pid_b * stride_mb + mask_i * stride_mi + mask_j * stride_mj,
        mask=mask_valid,
        other=0.0,
    )                                 # (BLOCK_N, BLOCK_N)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in chunks.
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        offs_h = h_start + tl.arange(0, BLOCK_H)
        h_valid = offs_h < H

        # Input pointer for the current (i, j, h) sub‑tile
        inp_ptrs = (
            tensor_ptr
            + pid_b * stride_tb
            + offs_i[:, None, None] * stride_ti
            + offs_j[None, :, None] * stride_tj
            + offs_h[None, None, :] * stride_th
        )
        inp = tl.load(
            inp_ptrs,
            mask=mask_valid[:, :, None] & h_valid[None, None, :],
            other=0.0,
        )                         # (BLOCK_N, BLOCK_N, BLOCK_H)

        # Apply mask (broadcasted over H)
        out = inp * mask_tile[:, :, None]

        # Store result back (in‑place)
        out_ptrs = (
            out_ptr
            + pid_b * stride_ob
            + offs_i[:, None, None] * stride_oi
            + offs_j[None, :, None] * stride_oj
            + offs_h[None, None, :] * stride_oh
        )
        tl.store(
            out_ptrs,
            out,
            mask=mask_valid[:, :, None] & h_valid[None, None, :],
        )


def _apply_mask_triton(tensor: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    In‑place multiplication of ``tensor`` (B,N,N,H) with ``mask`` (B,N,N).
    Both tensors must be on the same CUDA device and be contiguous.
    """
    B, N, N2, H = tensor.shape
    assert N == N2, "Tensor must be square in the second and third dims"
    BLOCK_N = 64
    BLOCK_H = 32
    grid = (B, triton.cdiv(N, BLOCK_N), triton.cdiv(N, BLOCK_N))
    _apply_mask_kernel[grid](
        tensor,
        mask,
        tensor,
        B,
        N,
        H,
        tensor.stride(0),
        tensor.stride(1),
        tensor.stride(2),
        tensor.stride(3),
        mask.stride(0),
        mask.stride(1),
        mask.stride(2),
        tensor.stride(0),
        tensor.stride(1),
        tensor.stride(2),
        tensor.stride(3),
        BLOCK_N=BLOCK_N,
        BLOCK_H=BLOCK_H,
    )
    return tensor


# ----------------------------------------------------------------------
# Entrypoint – the custom Triton‑enabled TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]  (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
              * "dim"         – input channel dimension
              * "hidden_dim" – hidden dimension for the TriMul projections
              * "nomask"      – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 for numeric stability)
    # --------------------------------------------------------------
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)   # work in half‑precision

    # --------------------------------------------------------------
    # 2️⃣ Fuse value‑projections and gate‑projections (single linear)
    # --------------------------------------------------------------
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, dim)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                     # (2*H, dim)

    # Linear produces [B, N, N, 2*H]
    left_concat = torch.nn.functional.linear(x_norm, left_concat_w)
    right_concat = torch.nn.functional.linear(x_norm, right_concat_w)

    # Split into value‑ and gate‑tensors
    left_val, left_gate = left_concat.split(hidden_dim, dim=-1)
    right_val, right_gate = right_concat.split(hidden_dim, dim=-1)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask – applied with a tiny Triton kernel
    # --------------------------------------------------------------
    if (not nomask) and (mask is not None):
        # Convert mask to half‑precision and make contiguous
        mask_fp16 = mask.to(torch.float16).contiguous()
        left_val = _apply_mask_triton(left_val.contiguous(), mask_fp16)
        right_val = _apply_mask_triton(right_val.contiguous(), mask_fp16)

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid, still in FP16)
    # --------------------------------------------------------------
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    left = left_val * left_gate
    right = right_val * right_gate

    # ------------------------------------------------------------------
    # out‑gate (scalar per position, still half‑precision)
    # ------------------------------------------------------------------
    out_gate = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()                                # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Core O(N³) reduction – batched mat‑mul with transpose
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N) for torch.bmm
    left_flat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (P, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)        # (P, N, N)

    # Batched matrix multiplication (Tensor‑Core accelerated)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))       # (P, N, N)

    # Reshape back to (B, N, N, H) and promote to FP32 for the next LN
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )   # shape [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Final normalisation, gating and linear projection (FP32)
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the out‑gate after the hidden‑dim LayerNorm
    out = out * out_gate.to(torch.float32)

    # Final 1×1 linear: hidden_dim -> dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
