@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          batch_stride_a, batch_stride_b, batch_stride_c,
                          stride_am, stride_ak,
                          stride_bk, stride_bn,
                          stride_cm, stride_cn,
                          M, N, K,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr,
                          num_stages: tl.constexpr,
                          num_warps: tl.constexpr):
   batch_id = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute base pointers for this batch
   a_batch_ptr = a_ptr + batch_id * batch_stride_a
   b_batch_ptr = b_ptr + batch_id * batch_stride_b
   c_batch_ptr = c_ptr + batch_id * batch_stride_c

   # compute tile offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # maybe check bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over K dimension
   for k in range(0, tl.cdiv(K, BLOCK_K)):
       # compute offsets within K block
       offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

       # load A tile: shape (BLOCK_M, BLOCK_K)
       a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
       a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None,:] < K), other=0.0).to(tl.float32)

       # load B tile: shape (BLOCK_K, BLOCK_N) (since B is KxN)
       b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)
       b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)

       # compute dot product
       acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

   # write back
   c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
   tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def trilmul_kernel(
    a_proj_ptr, a_gate_ptr,
    b_proj_ptr, b_gate_ptr,
    mask_ptr,
    c_ptr,
    batch_stride_a_proj, batch_stride_a_gate,
    batch_stride_b_proj, batch_stride_b_gate,
    batch_stride_c,
    stride_a_m, stride_a_k,
    stride_a_gate_m, stride_a_gate_k,
    stride_b_k, stride_b_n,
    stride_b_gate_k, stride_b_gate_n,
    mask_stride_batch, mask_stride_i, mask_stride_j,
    M, N, K, hidden_dim,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr,
    num_warps: tl.constexpr,
):
    pid = tl.program_id(0)  # batch* hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch id and orig batch index
    batch_id = pid
    orig_batch = batch_id // hidden_dim
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_orig_batch_offset = orig_batch * mask_stride_batch

    # ensure within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load left projection and gate and mask for A tile
        a_proj_ptrs = a_proj_ptr + (batch_id * batch_stride_a_proj + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k)
        a_gate_ptrs = a_gate_ptr + (batch_id * batch_stride_a_gate + offs_m[:, None] * stride_a_gate_m + offs_k[None, :] * stride_a_gate_k)
        mask_a_ptrs = mask_ptr + (mask_orig_batch_offset + offs_m[:, None] * mask_stride_i + offs_k[None, :] * mask_stride_j)

        a_proj = tl.load(a_proj_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        a_gate = tl.load(a_gate_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        a_mask = tl.load(mask_a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=1.0)

        # compute a tile as fp32
        a_val = (a_proj * a_gate * a_mask).to(tl.float32)

        # Load right projection, gate, mask for B tile (k, n)
        b_proj_ptrs = b_proj_ptr + (batch_id * batch_stride_b_proj + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n)
        b_gate_ptrs = b_gate_ptr + (batch_id * batch_stride_b_gate + offs_k[:, None] * stride_b_gate_k + offs_n[None, :] * stride_b_gate_n)
        mask_b_ptrs = mask_ptr + (mask_orig_batch_offset + offs_k[:, None] * mask_stride_i + offs_n[None, :] * mask_stride_j)

        b_proj = tl.load(b_proj_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        b_gate = tl.load(b_gate_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        b_mask = tl.load(mask_b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=1.0)

        b_val = (b_proj * b_gate * b_mask).to(tl.float32)

        # dot product
        acc += tl.dot(a_val, b_val)

    # write output
    c_ptrs = c_ptr + (batch_id * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    # Cast back to fp16 for storage
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
Outgoing TriMul (AlphaFold‑3) – fused Triton kernel

Algorithm
---------
1. Layer‑norm the original tensor (float32).
2. Cast to fp16 and compute four linear projections (left/right &
   left/right‑gate) – all weights are cast to fp16.
3. Fuse the element‑wise multiplication with the optional mask *inside*
   the batched matrix‑multiply kernel.  The kernel computes for every
   hidden slice `h` and batch `b`

        out[b,h,i,j] = Σ_k (Lproj[b,h,i,k] * Lgate[b,h,i,k] * mask[b,i,k]) *
                       (Rproj[b,h,k,j] * Rgate[b,h,k,j] * mask[b,k,j])

   This eliminates the separate element‑wise fusion pass and saves a
   huge amount of memory traffic.
4. The kernel works on a (B·H) batch of (N×N) matrices, using
   a classic tiled GEMM (BLOCK_M×BLOCK_N tiles, BLOCK_K inner
   reduction) with fp16 loads and fp32 accumulation.
5. The result (fp16) is cast back to fp32, reshaped to [bs, N, N, H],
   and passed through a second LayerNorm.
6. Apply the output gate (computed in step 2) and a final
   linear projection back to `dim`.
7. Return a float32 tensor of shape [bs, N, N, dim].

The heavy cubic O(N³·H·B) work is now performed by a single fused
Triton kernel that is far more bandwidth‑efficient than the
straight‑forward PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM with fused projection, gate and mask
# ----------------------------------------------------------------------
@triton.jit
def trilmul_gemm_kernel(
    # pointers
    a_proj_ptr, a_gate_ptr,  # left projection & gate (fp16)
    b_proj_ptr, b_gate_ptr,  # right projection & gate (fp16)
    mask_ptr,                # mask (fp16) – shape (B,N,N)
    c_ptr,                   # output (fp16)

    # batch strides (in elements)
    batch_stride_a, batch_stride_b, batch_stride_c,

    # strides inside each [M,K] / [K,N] matrix (row‑major)
    stride_a_m, stride_a_k,   # left (M=row i, K=col k)
    stride_b_k, stride_b_n,   # right (K=row k, N=col j)

    # mask strides (batch, i, j)
    mask_stride_batch, mask_stride_i, mask_stride_j,

    # output strides (row‑major)
    stride_c_m, stride_c_n,

    # problem size
    M, N, K, hidden_dim,

    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_stages: tl.constexpr,
    num_warps: tl.constexpr,
):
    # --------------------------------------------------------------
    # program ids
    # --------------------------------------------------------------
    batch_id = tl.program_id(0)      # (B·H) – each hidden slice is a batch
    pid_m = tl.program_id(1)        # tile row
    pid_n = tl.program_id(2)        # tile col

    # original batch index (mask is not expanded over H)
    orig_batch = batch_id // hidden_dim

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # block‑level bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    # accumulator in fp32
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # K‑loop
    # -----------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ------- left side (Lproj * Lgate * mask) ------------------
        a_proj_ptrs = a_proj_ptr + (batch_id * batch_stride_a
                                    + offs_m[:, None] * stride_a_m
                                    + offs_k[None, :] * stride_a_k)
        a_gate_ptrs = a_gate_ptr + (batch_id * batch_stride_a
                                    + offs_m[:, None] * stride_a_m
                                    + offs_k[None, :] * stride_a_k)
        mask_a_ptrs = mask_ptr + (orig_batch * mask_stride_batch
                                  + offs_m[:, None] * mask_stride_i
                                  + offs_k[None, :] * mask_stride_j)

        a_proj = tl.load(a_proj_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0).to(tl.float32)
        a_gate = tl.load(a_gate_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=0.0).to(tl.float32)
        a_mask = tl.load(mask_a_ptrs,
                         mask=mask_m[:, None] & mask_k[None, :],
                         other=1.0).to(tl.float32)
        a_val = a_proj * a_gate * a_mask   # (BLOCK_M, BLOCK_K) fp32

        # ------- right side (Rproj * Rgate * mask) -----------------
        b_proj_ptrs = b_proj_ptr + (batch_id * batch_stride_b
                                    + offs_k[:, None] * stride_b_k
                                    + offs_n[None, :] * stride_b_n)
        b_gate_ptrs = b_gate_ptr + (batch_id * batch_stride_b
                                    + offs_k[:, None] * stride_b_k
                                    + offs_n[None, :] * stride_b_n)
        mask_b_ptrs = mask_ptr + (orig_batch * mask_stride_batch
                                  + offs_k[:, None] * mask_stride_i
                                  + offs_n[None, :] * mask_stride_j)

        b_proj = tl.load(b_proj_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)
        b_gate = tl.load(b_gate_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0).to(tl.float32)
        b_mask = tl.load(mask_b_ptrs,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=1.0).to(tl.float32)
        b_val = b_proj * b_gate * b_mask   # (BLOCK_K, BLOCK_N) fp32

        # ------- accumulate -------------------------------------------------
        acc += tl.dot(a_val, b_val)       # fp32 accumulation

    # -----------------------------------------------------------------
    # write back (cast to fp16)
    # -----------------------------------------------------------------
    c_ptrs = c_ptr + (batch_id * batch_stride_c
                      + offs_m[:, None] * stride_c_m
                      + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3).
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    eps = 1e-5

    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and flatten for linear layers
    # ------------------------------------------------------------------
    x_fp16 = x.to(torch.float16)
    x_flat = x_fp16.view(-1, dim)                         # (B·N·N, dim)

    # ------------------------------------------------------------------
    # 3. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_proj = F.linear(x_flat, left_proj_w).view(bs, N, N, hidden_dim)
    right_proj = F.linear(x_flat, right_proj_w).view(bs, N, N, hidden_dim)

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(bs, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(bs, N, N, hidden_dim).to(torch.float32)

    # ------------------------------------------------------------------
    # 4. Prepare tensors for the fused kernel
    # ------------------------------------------------------------------
    # (B, H, N, N) layout, then collapse batch & head dimensions
    left_proj = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    left_gate  = left_gate.permute(0, 3, 1, 2).contiguous()
    right_proj = right_proj.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)  (k,j)
    right_gate = right_gate.permute(0, 3, 2, 1).contiguous()

    BH = bs * hidden_dim
    left_proj_f = left_proj.view(BH, N, N)
    left_gate_f  = left_gate.view(BH,  N, N)
    right_proj_f = right_proj.view(BH, N, N)
    right_gate_f = right_gate.view(BH, N, N)

    # ------------------------------------------------------------------
    # 5. Mask handling
    # ------------------------------------------------------------------
    if nomask:
        mask_fp16 = torch.ones((bs, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Allocate output buffer (fp16)
    # ------------------------------------------------------------------
    out_fp16 = torch.empty((BH, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7. Compute strides (in elements)
    # ------------------------------------------------------------------
    batch_stride_a = left_proj_f.stride(0)
    stride_a_m = left_proj_f.stride(1)
    stride_a_k = left_proj_f.stride(2)

    batch_stride_b = right_proj_f.stride(0)
    stride_b_k = right_proj_f.stride(1)
    stride_b_n = right_proj_f.stride(2)

    batch_stride_c = out_fp16.stride(0)
    stride_c_m = out_fp16.stride(1)
    stride_c_n = out_fp16.stride(2)

    mask_stride_batch = mask_fp16.stride(0)
    mask_stride_i = mask_fp16.stride(1)
    mask_stride_j = mask_fp16.stride(2)

    # ------------------------------------------------------------------
    # 8. Launch the fused kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trilmul_gemm_kernel[grid](
        a_proj_f.data_ptr(),
        a_gate_f.data_ptr(),
        b_proj_f.data_ptr(),
        b_gate_f.data_ptr(),
        mask_fp16.data_ptr(),
        out_fp16.data_ptr(),
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        stride_a_m,
        stride_a_k,
        stride_b_k,
        stride_b_n,
        mask_stride_batch,
        mask_stride_i,
        mask_stride_j,
        stride_c_m,
        stride_c_n,
        N,          # M
        N,          # N
        N,          # K
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_stages=3,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to [bs, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = out_fp16.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 10. Output LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # ------------------------------------------------------------------
    # 11. Apply output gate (still float32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12. Final linear back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    out = F.linear(out.view(-1, hidden_dim), to_out_w).view(bs, N, N, dim)

    return out
