"""
TriMul (outgoing) – high‑performance Triton implementation.

The forward pass consists of:
1️⃣ Layer‑Norm on the input (FP32).
2️⃣ Half‑precision linear projections and sigmoid gates
   (weights are cast to fp16 so the bulk of the work runs on Tensor‑cores).
3️⃣ Optional mask is applied in fp16.
4️⃣ A fused N³ outer‑product for each hidden channel:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is performed by a custom Triton GEMM kernel that computes
   many independent (batch, hidden) pages as A @ Bᵀ with
   fp16 inputs and fp32 accumulation.
5️⃣ Layer‑Norm on the hidden dimension, output‑gate (sigmoid) and a final
   fp32 linear projection back to the original feature dimension.

All heavy arithmetic is in half‑precision; the only fp32 work is the
two LayerNorms and the final linear layer.  The Triton kernel is the
only non‑trivial part of the implementation, satisfying the “must‑use‑kernel”
requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: compute C = A @ Bᵀ for many independent pages.
# A, B : (P, N, N)  fp16
# C    : (P, N, N)  fp32
# P = batch * hidden
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    a_ptr,                     # const half*   (left)
    b_ptr,                     # const half*   (right, accessed as transposed)
    c_ptr,                     # float*        (output)
    P,                         # int32   total pages = B * hidden
    N,                         # int32   sequence length
    stride_az, stride_ai, stride_ak,   # strides for A (page, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (page, col‑of‑orig, row‑of‑orig)
    stride_cz, stride_ci, stride_cj,   # strides for C (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a block of C = A @ Bᵀ.
    A : (P, N, N) row‑major, dtype half
    B : (P, N, N) row‑major, accessed as Bᵀ
    C : (P, N, N) row‑major, dtype float32
    """
    pid_z = tl.program_id(0)       # page index (batch * hidden)
    pid_m = tl.program_id(1)       # block‑row
    pid_n = tl.program_id(2)       # block‑col

    # ----------------------------------- block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the beginning of the current page
    a_page = a_ptr + pid_z * stride_az
    b_page = b_ptr + pid_z * stride_bz
    c_page = c_ptr + pid_z * stride_cz

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------- reduction over K dimension
    cur_k = 0
    while cur_k < N:
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_offs < N

        # ----- load a tile from A (row‑major) -----
        a_ptrs = a_page + \
                 offs_m[:, None] * stride_ai + \
                 cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # half → fp16

        # ----- load a tile from Bᵀ.
        #   We treat the original B as row‑major and read it with
        #   swapped strides so that we get Bᵀ[k, n] = B[n, k].
        b_ptrs = b_page + \
                 cur_k_offs[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # half

        # ----- multiply‑accumulate (fp16 × fp16 → fp32) -----
        acc += tl.dot(a, b)

        cur_k += BLOCK_K

    # ----------------------------------- write back
    c_ptrs = c_page + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------ unpack
    x, mask, weights, config = data
    # x : [B, N, N, dim]  (float32)
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------ half‑precision weight copies (for the heavy ops)
    w_fp16 = {
        "left_proj.weight":   weights["left_proj.weight"].to(torch.float16),
        "right_proj.weight":  weights["right_proj.weight"].to(torch.float16),
        "left_gate.weight":   weights["left_gate.weight"].to(torch.float16),
        "right_gate.weight":  weights["right_gate.weight"].to(torch.float16),
        "out_gate.weight":    weights["out_gate.weight"].to(torch.float16),
    }

    # ------------------------------------------------------------------ 1️⃣ Input LayerNorm (FP32)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------ cast to fp16 for the remaining heavy ops
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------------------------------------------ 2️⃣ Linear projections (half‑precision, bias‑free)
    left = F.linear(x_norm_h, w_fp16["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x_norm_h, w_fp16["right_proj.weight"])  # [B, N, N, hidden]

    # ------------------------------------------------------------------ optional mask (applied in fp16)
    if not nomask and mask is not None:
        # mask shape [B, N, N] → [B, N, N, 1]
        mask_h = mask.unsqueeze(-1).to(torch.float16)
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------ 3️⃣ Gates (sigmoid, still fp16)
    left_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["right_gate.weight"])) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["out_gate.weight"]))     # [B,N,N,hidden]

    # ------------------------------------------------------------------ Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ 4️⃣ Prepare tensors for the Triton GEMM
    #   Bring the hidden dimension to the second axis so that we can fuse
    #   batch and hidden as a single page dimension.
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    pages = B * hidden
    left_mat = left.view(pages, N, N)   # fp16
    right_mat = right.view(pages, N, N) # fp16

    # allocate output buffer in fp32 (direct accumulator type of the kernel)
    out_mat = torch.empty((pages, N, N), dtype=torch.float32, device=x.device)

    # ------------------------------------------------------------------ strides in elements (not bytes)
    stride_az = left_mat.stride(0)   # page stride = N*N
    stride_ai = left_mat.stride(1)   # row stride = N
    stride_ak = left_mat.stride(2)   # col stride = 1

    # B is accessed as transposed: we need column‑stride and row‑stride swapped
    stride_bz = right_mat.stride(0)   # page stride
    stride_bk = right_mat.stride(2)   # column (=1) – becomes the k‑dim stride
    stride_bn = right_mat.stride(1)   # row (=N)   – becomes the n‑dim stride

    stride_cz = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # ------------------------------------------------------------------ tile size selection (occupancy‑aware)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 16
        num_warps = 4
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32
        num_warps = 8

    grid = (pages,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # ------------------------------------------------------------------ launch the Triton kernel
    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        pages,
        N,
        stride_az, stride_ai, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=2,
    )

    # ------------------------------------------------------------------ reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)   # fp32

    # ------------------------------------------------------------------ 5️⃣ Hidden‑dim LayerNorm (FP32)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------ Apply output gate (cast to FP32 for safety)
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------ Final linear projection back to original dim (FP32)
    out = F.linear(out, weights["to_out.weight"])

    return out
