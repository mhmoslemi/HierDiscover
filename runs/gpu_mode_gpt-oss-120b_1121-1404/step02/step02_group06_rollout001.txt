    left_concat = F.linear(x_flat, left_concat_w)   # (B*N*N, 2*H)

5. Output LayerNorm, gating and final linear projection are done
   in fp16 where possible, with a final cast to fp32.
The implementation avoids any Python‑level loops, reduces the number
of large temporary tensors and keeps the entire compute path in
half‑precision except the two LayerNorms.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernels:  (proj * sigmoid(gate) * optional mask)
# ---------------------------------------------------------------
@triton.jit
def _fuse_no_mask_kernel(
    proj_ptr, gate_ptr, out_ptr,
    total_elems,                        # int64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * gate
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def _fuse_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    hidden_dim,                         # int64
    total_elems,                        # int64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # sigmoid
    gate = 1.0 / (1.0 + tl.exp(-gate))

    # mask is (B,N,N) → broadcast over hidden_dim.
    # Compute index of the mask element: floor(off / hidden_dim)
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)


# ---------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 and flatten for linear layers
    # ------------------------------------------------------------------
    x_fp16 = x.half()
    x_flat = x_fp16.view(-1, dim)                     # (B·N·N, dim)

    # ------------------------------------------------------------------
    # 3. Concatenate proj & gate weights for left and right sides
    # ------------------------------------------------------------------
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16, device=device)

    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16, device=device)

    # ------------------------------------------------------------------
    # 4. Linear passes (fp16)
    # ------------------------------------------------------------------
    # (B·N·N, 2·hidden)
    left_concat = F.linear(x_flat, left_concat_w)
    right_concat = F.linear(x_flat, right_concat_w)

    H = hidden_dim
    left_proj = left_concat[:, :H].view(bs, N, N, H)
    left_gate_pre = left_concat[:, H:].view(bs, N, N, H)

    right_proj = right_concat[:, :H].view(bs, N, N, H)
    right_gate_pre = right_concat[:, H:].view(bs, N, N, H)

    # ------------------------------------------------------------------
    # 5. Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"].to(torch.float16, device=device)
    out_gate_pre = F.linear(x_flat, out_gate_w).view(bs, N, N, H)
    out_gate = torch.sigmoid(out_gate_pre)                     # fp16

    # ------------------------------------------------------------------
    # 6. Fuse proj, gate and (optional) mask with Triton
    # ------------------------------------------------------------------
    total = left_proj.numel()               # B·N·N·H
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    left_out = torch.empty_like(left_proj)   # fp16
    right_out = torch.empty_like(right_proj) # fp16

    if not nomask:
        # mask is (B,N,N) → flatten, keep fp16
        mask_fp16 = mask.to(torch.float16, device=device).contiguous().view(-1)
        _fuse_mask_kernel[grid](
            left_proj.reshape(-1),
            left_gate_pre.reshape(-1),
            mask_fp16,
            left_out.reshape(-1),
            hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        _fuse_mask_kernel[grid](
            right_proj.reshape(-1),
            right_gate_pre.reshape(-1),
            mask_fp16,
            right_out.reshape(-1),
            hidden_dim,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        _fuse_no_mask_kernel[grid](
            left_proj.reshape(-1),
            left_gate_pre.reshape(-1),
            left_out.reshape(-1),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        _fuse_no_mask_kernel[grid](
            right_proj.reshape(-1),
            right_gate_pre.reshape(-1),
            right_out.reshape(-1),
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )

    # ------------------------------------------------------------------
    # 7. Batched GEMM: Σₖ left[i,k,d]·right[j,k,d]
    # ------------------------------------------------------------------
    # (B, N, N, H) → (B, H, N, N)   and   right transposed for GEMM
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_out.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    BH = bs * H
    left_mat = left_perm.view(BH, N, N)    # (B·H, N, N)
    right_mat = right_perm.view(BH, N, N)  # (B·H, N, N)

    # All fp16 → use Tensor‑Core accelerated cuBLAS
    out_mat_fp16 = torch.bmm(left_mat, right_mat)           # (B·H, N, N)

    # Restore (B, N, N, H)
    out_fp16 = (
        out_mat_fp16.view(bs, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm (float32 for numerical safety)
    # ------------------------------------------------------------------
    out_fp32 = out_fp16.to(torch.float32)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out_fp32,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # (B, N, N, H) float32

    # ------------------------------------------------------------------
    # 9. Apply the output gate (still fp16 → cast to fp32)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 10. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)
    out_norm_fp16 = out_norm.half()
    out_flat = out_norm_fp16.view(-1, H)                     # (B·N·N, H)
    out_proj_fp16 = F.linear(out_flat, to_out_w)             # (B·N·N, dim)
    out = out_proj_fp16.view(bs, N, N, dim).to(torch.float32)

    return out
```
