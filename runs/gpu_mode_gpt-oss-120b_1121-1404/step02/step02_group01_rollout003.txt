@triton.jit
def trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_az, stride_ai, stride_ak, stride_ad,
    stride_bz, stride_bk, stride_bn, stride_bd,
    stride_cz, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden indices
    hid = pid_z % H
    batch = pid_z // H

    # Compute page start pointers (including hidden offset)
    a_page = a_ptr + batch * stride_az + hid * stride_ad
    b_page = b_ptr + batch * stride_bz + hid * stride_bd
    c_page = c_ptr + batch * stride_cz + hid * stride_cd

    # Offsets within block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    cur_k = 0
    while cur_k < N:
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_offs < N

        # load A tile (i,k)
        a_ptrs = a_page + \
                 offs_m[:, None] * stride_ai + \
                 cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)   # fp16

        # load B tile as transposed (k,j)
        b_ptrs = b_page + \
                 cur_k_offs[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)   # fp16

        acc += tl.dot(a, b)

        cur_k += BLOCK_K

    # write result
    c_ptrs = c_page + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) – fast Triton implementation.

The forward pass consists of:
1️⃣ Layer‑Norm on the input (FP32).
2️⃣ Two half‑precision linear projections (left/right) and three half‑precision
   gated linear projections (left_gate, right_gate, out_gate).  All heavy
   arithmetic stays in FP16.
3️⃣ A fused batched GEMM that computes, for every batch‑hidden pair,
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   i.e. a collection of A @ Bᵀ operations.  The kernel works on the
   original layout (B,N,N,hidden) and therefore avoids the costly
   `permute(...).contiguous()` copy used in a naïve implementation.
   It receives the hidden index via a modulo operation, so the grid
   dimension is (B*hidden, ceil(N/TileM), ceil(N/TileN)).
4️⃣ Layer‑Norm + output‑gate on the hidden dimension (FP32).
5️⃣ Final linear projection back to the original feature dim (FP32).

All tensor‑core work is performed in FP16 with FP32 accumulation,
making the kernel bandwidth‑friendly on H100.  The only non‑Triton
operations are cheap LayerNorms, sigmoids and the final linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    a_ptr,                # const half*   – left tensor   (B,N,N,hidden)
    b_ptr,                # const half*   – right tensor  (B,N,N,hidden)
    c_ptr,                # float*        – output tensor (B,N,N,hidden)
    B, N, H,              # runtime scalar arguments
    stride_az, stride_ai, stride_ak, stride_ad,   # A strides: batch, i, k, hidden
    stride_bz, stride_bk, stride_bn, stride_bd,   # B strides: batch, k, j, hidden
    stride_cz, stride_ci, stride_cj, stride_cd,   # C strides: batch, i, j, hidden
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ   (batched over B*H)"""
    pid_z = tl.program_id(0)          # combined batch‑hidden index
    pid_m = tl.program_id(1)          # block‑row
    pid_n = tl.program_id(2)          # block‑col

    # decode hidden and batch indices
    hid   = pid_z % H                 # hidden index
    batch = pid_z // H                # batch index

    # page‑base pointers (one (batch, hidden) page)
    a_page = a_ptr + batch * stride_az + hid * stride_ad
    b_page = b_ptr + batch * stride_bz + hid * stride_bd
    c_page = c_ptr + batch * stride_cz + hid * stride_cd

    # block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    cur_k = 0
    while cur_k < N:
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_offs < N

        # ---------- load a tile (i,k) ----------
        a_ptrs = a_page + \
                 offs_m[:, None] * stride_ai + \
                 cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # FP16 -> auto‑cast to fp16

        # ---------- load b tile as Bᵀ (k,j) ----------
        b_ptrs = b_page + \
                 cur_k_offs[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # FP16

        # ---------- multiply‑accumulate ----------
        acc += tl.dot(a, b)               # fp16×fp16 → fp32

        cur_k += BLOCK_K

    # ---------- write back ----------
    c_ptrs = c_page + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------ unpack
    x, mask, weights, cfg = data
    # x: [B, N, N, dim] (float32)
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]                     # hidden dimension
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------ LayerNorm on the input (FP32)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                                        # [B, N, N, dim]  (fp32)

    # ------------------------------------------------------------------ cast to half for heavy ops
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------ pre‑cast weights (half‑precision, bias‑free)
    w16 = {
        "left_proj.weight":   weights["left_proj.weight"].to(torch.float16),
        "right_proj.weight":  weights["right_proj.weight"].to(torch.float16),
        "left_gate.weight":   weights["left_gate.weight"].to(torch.float16),
        "right_gate.weight":  weights["right_gate.weight"].to(torch.float16),
        "out_gate.weight":    weights["out_gate.weight"].to(torch.float16),
    }

    # ------------------------------------------------------------------ linear projections (half)
    left  = F.linear(x_h, w16["left_proj.weight"])   # [B, N, N, H]  (fp16)
    right = F.linear(x_h, w16["right_proj.weight"]) # [B, N, N, H]  (fp16)

    # ------------------------------------------------------------------ optional mask
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------ gates (half → half after sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_h, w16["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_h, w16["right_gate.weight"])) # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x_h, w16["out_gate.weight"]))   # [B, N, N, H]

    # ------------------------------------------------------------------ apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ allocate output buffer (float32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=x.device)

    # --------------------------------------------------------------- strides (in elements)
    # left/right layout: (B, N, N, H) – row‑major
    stride_az = left.stride(0)   # B‑stride = N*N*H
    stride_ai = left.stride(1)   # i‑stride = N*H
    stride_ak = left.stride(2)   # k‑stride = H
    stride_ad = left.stride(3)   # hidden‑stride = 1

    stride_bz = right.stride(0)
    stride_bk = right.stride(2)  # k‑stride = H
    stride_bn = right.stride(1)  # j‑stride = N*H
    stride_bd = right.stride(3)  # hidden‑stride = 1

    stride_cz = out.stride(0)
    stride_ci = out.stride(1)    # i‑stride = N*H
    stride_cj = out.stride(2)    # j‑stride = H
    stride_cd = out.stride(3)    # hidden‑stride = 1

    # --------------------------------------------------------------- tile sizes
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 16
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    # --------------------------------------------------------------- launch
    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_az, stride_ai, stride_ak, stride_ad,
        stride_bz, stride_bk, stride_bn, stride_bd,
        stride_cz, stride_ci, stride_cj, stride_cd,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------ LayerNorm over hidden dim (FP32)
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------ apply output gate
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------ final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
