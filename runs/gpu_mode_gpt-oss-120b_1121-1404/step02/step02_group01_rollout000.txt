"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The operation consists of:
1️⃣ Layer‑norm on the input tensor (FP32).
2️⃣ Two half‑precision linear projections (left/right) and three half‑precision
   gating linear layers.  A (optional) binary mask is applied before the gates.
3️⃣ A fused N³ “outer‑product” for every (batch, hidden) page:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is implemented as a batched GEMM (A @ Bᵀ) in a custom Triton kernel.
   The kernel works on FP16 inputs and accumulates in FP32.
4️⃣ Layer‑norm over the hidden dimension, an output gate, and a final
   FP32 linear projection back to the original feature dimension.

All heavy arithmetic runs in FP16 (Tensor‑core friendly).  Only the two
LayerNorms and the final linear are kept in FP32.  The kernel adapts its
tile size to the sequence length to keep occupancy high on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    a_ptr,                     # const half*   (left)
    b_ptr,                     # const half*   (right, accessed transposed)
    c_ptr,                     # float*        (output)
    batch_hidden, N,           # int32 scalars
    stride_az, stride_ai, stride_ak,   # strides for A (page, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (page, k‑dim, n‑dim)
    stride_cz, stride_ci, stride_cj,   # strides for C (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a block of C = A @ Bᵀ.
    A : (batch_hidden, N, N) – row‑major, dtype half
    B : (batch_hidden, N, N) – row‑major, accessed as Bᵀ
    C : (batch_hidden, N, N) – row‑major, dtype float32
    """
    pid_z = tl.program_id(0)          # which (batch, hidden) page
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block column

    # ------------------- block coordinates -------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # page base pointers
    a_page = a_ptr + pid_z * stride_az
    b_page = b_ptr + pid_z * stride_bz
    c_page = c_ptr + pid_z * stride_cz

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------- reduction over K --------------------
    cur_k = 0
    while cur_k < N:
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_offs < N

        # A tile (row‑major)
        a_ptrs = a_page + \
                 offs_m[:, None] * stride_ai + \
                 cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # half → FP16

        # B tile accessed as Bᵀ
        b_ptrs = b_page + \
                 cur_k_offs[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # half

        # FP16×FP16 → FP32 accumulation
        acc += tl.dot(a, b)

        cur_k += BLOCK_K

    # ------------------- write result ------------------------
    c_ptrs = c_page + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]  (float32 or bool, optional)
        - weights_dict : dict of pre‑loaded torch.Tensor parameters
        - config_dict  : dict containing:
              * dim         – input feature dimension
              * hidden_dim  – hidden dimension of the TriMul block
              * nomask      – True if mask should be ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------- unpack
    x, mask, weights, cfg = data
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------- fp16 weights
    # heavy linear layers: left/right proj + four gates
    w_fp16 = {
        "left_proj":   weights["left_proj.weight"].to(torch.float16),
        "right_proj":  weights["right_proj.weight"].to(torch.float16),
        "left_gate":   weights["left_gate.weight"].to(torch.float16),
        "right_gate":  weights["right_gate.weight"].to(torch.float16),
        "out_gate":    weights["out_gate.weight"].to(torch.float16),
    }

    # ---------------------------------------------------------- LayerNorm (FP32)
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim] float32

    # ---------------------------------------------------------- cast to half
    x_norm_h = x_norm.to(torch.float16)

    # ------------------------------ linear projections (half‑precision, no bias)
    left  = F.linear(x_norm_h, w_fp16["left_proj"])   # [B,N,N,hidden]  fp16
    right = F.linear(x_norm_h, w_fp16["right_proj"]) # [B,N,N,hidden]  fp16

    # ---------------------------------------------------------- optional mask
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ----------------------------------------------------------- gates (half → half)
    left_gate  = torch.sigmoid(F.linear(x_norm_h, w_fp16["left_gate"]))
    right_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["right_gate"]))
    out_gate   = torch.sigmoid(F.linear(x_norm_h, w_fp16["out_gate"]))

    # ----------------------------------------------------------- apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------ reshape for GEMM
    # we need (B*hidden, N, N) layout for the Triton kernel
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    BxH = B * hidden
    left_mat  = left.view(BxH, N, N)   # half
    right_mat = right.view(BxH, N, N)  # half

    # --------------------------------------------------- output buffer (FP32)
    out_mat = torch.empty((BxH, N, N), dtype=torch.float32, device=x.device)

    # ----------------------------------------------------------- strides (elements)
    stride_az = left_mat.stride(0)   # page stride = N * N
    stride_ai = left_mat.stride(1)   # row stride = N
    stride_ak = left_mat.stride(2)   # col stride = 1

    stride_bz = right_mat.stride(0)
    stride_bk = right_mat.stride(2)  # =1  (k‑dim stride, col)
    stride_bn = right_mat.stride(1)  # =N  (n‑dim stride, row)

    stride_cz = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # -------------------------------------------------------- tile selection
    # Pick tile sizes that give good occupancy on H100.
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 16
    elif N <= 512:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 32
    elif N <= 768:
        BLOCK_M = BLOCK_N = 96
        BLOCK_K = 32
    else:  # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 32

    grid = (BxH,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # ----------------------------------------------------------- launch
    _trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BxH,
        N,
        stride_az, stride_ai, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps works well on SM90
        num_stages=2,         # double‑buffered loads
    )

    # ---------------------------------------------------------- reshape back
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden] FP32

    # ------------------------------------------------------ to_out_norm (FP32)
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -------------------------------------------------------- output gate
    out = out * out_gate.to(torch.float32)   # broadcast over hidden dim

    # ------------------------------------------------------- final linear (FP32)
    out = F.linear(out, weights["to_out.weight"])

    return out
