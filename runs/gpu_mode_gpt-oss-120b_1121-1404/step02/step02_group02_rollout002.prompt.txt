import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul “outgoing” kernel (AlphaFold3) – fast half‑precision implementation.

The forward pass consists of:
1️⃣ Layer‑norm on the input (FP32 for numeric stability).
2️⃣ Six linear projections (norm → left/right values & three gates).  All
   projections are performed in FP16 to exploit Tensor‑Cores.
3️⃣ Optional mask application.
4️⃣ Element‑wise gating (sigmoid) – still in FP16.
5️⃣ Core O(N³) reduction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   This is a batched mat‑mul with a transposed operand.
   We fuse the batch and hidden dimensions (P = B·H) and run a
   custom Triton kernel that computes:
        out[p,i,j] = Σ_k A[p,i,k] * B[p,j,k]   (A,B are FP16, acc. FP32)
   Tile sizes (128×128×64) are chosen to fully occupy the H100
   Tensor‑Core pipeline.
6️⃣ Final Layer‑norm (FP32), gating and linear projection back to the
   original channel dimension (FP32).

All heavy arithmetic is performed in half‑precision; the final result
is returned as float32.  The implementation fits the required
`custom_kernel(data)` signature and runs on Triton 3.3.1 / H100.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul with the second operand transposed.
#   out[p, i, j] = Σ_k A[p, i, k] * B[p, j, k]
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr,                      # const *[fp16]  left  tensor   (P, M, K)
    b_ptr,                      # const *[fp16]  right tensor   (P, N, K)
    c_ptr,                      #       *[fp32]  output tensor  (P, M, N)
    batch, M, N, K,             # int32 shape parameters
    stride_apb, stride_amn, stride_ak,   # int64 strides for A
    stride_bpb, stride_bmn, stride_bk,   # int64 strides for B
    stride_cpb, stride_cmn, stride_cnk,   # int64 strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)               # combined batch × hidden index
    pid_m = tl.program_id(0)               # tile row index
    pid_n = tl.program_id(1)               # tile column index

    # --------------------------------------------------------------
    # offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator (FP32) – Tensor‑Core friendly
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑dimension loop
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A : (BLOCK_M, BLOCK_K)
        a_ptrs = (a_ptr
                  + pid_b * stride_apb
                  + offs_m[:, None] * stride_amn
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # B : (BLOCK_K, BLOCK_N) – note the transposed view
        b_ptrs = (b_ptr
                  + pid_b * stride_bpb
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bmn)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # Dot product – accumulation in FP32
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    c_ptrs = (c_ptr
              + pid_b * stride_cpb
              + offs_m[:, None] * stride_cmn
              + offs_n[None, :] * stride_cnk)
    tl.store(c_ptrs, acc,
             mask=(mask_m[:, None] & mask_n[None, :]))

# Entrypoint used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]  (bool/float, may be None)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
              * "dim"         – input channel dimension
              * "hidden_dim" – hidden dimension for the TriMul projections
              * "nomask"      – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0️⃣ Cast everything we will compute in half‑precision
    # ------------------------------------------------------------------
    def to_half(t): return t.to(torch.float16)
    # layer‑norm params
    norm_w = to_half(weights["norm.weight"])
    norm_b = to_half(weights["norm.bias"])
    # projection / gate weights
    left_proj_w   = to_half(weights["left_proj.weight"])
    right_proj_w  = to_half(weights["right_proj.weight"])
    left_gate_w   = to_half(weights["left_gate.weight"])
    right_gate_w  = to_half(weights["right_gate.weight"])
    out_gate_w    = to_half(weights["out_gate.weight"])
    # final norm & projection (keep FP32 for higher accuracy)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Input layer‑norm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)   # half‑precision tensor

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP16, no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_norm, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)       # [B, N, N, 1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear transforms, still FP16)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    # apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul reduction – batched mat‑mul with transposed operand
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] → fuse B & H
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H                       # combined batch × hidden dimension

    # flatten to (P, N, N)
    left_flat  = left.view(P, N, N)
    right_flat = right.view(P, N, N)

    # clamp before FP16 conversion to avoid overflow on extreme Cauchy samples
    left_fp16  = torch.clamp(left_flat,  -65504.0, 65504.0).to(torch.float16)
    right_fp16 = torch.clamp(right_flat, -65504.0, 65504.0).to(torch.float16)

    out_fp32 = torch.empty((P, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Triton launch configuration – tuned for H100
    # --------------------------------------------------------------
    BLOCK_M = 128   # rows per tile
    BLOCK_N = 128   # cols per tile
    BLOCK_K = 64    # reduction step size

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks along M
        triton.cdiv(N, BLOCK_N),   # blocks along N
        P,                         # batched dimension (B·H)
    )

    _batched_matmul_transpose_kernel[grid](
        # pointers
        left_fp16,
        right_fp16,
        out_fp32,
        # shape parameters
        P, N, N, N,
        # strides (in elements)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2),
        # compile‑time constants
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP32

    # ------------------------------------------------------------------
    # 6️⃣ Final normalisation, gating & linear projection (FP32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    # out_gate is still FP16 – cast to FP32 for the final multiply
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_w)   # → [B, N, N, dim] (FP32)

    return out
