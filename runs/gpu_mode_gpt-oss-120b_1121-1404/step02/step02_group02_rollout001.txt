"""
Outgoing Triangle Multiplicative Update (TriMul) – fast Triton implementation.

The heavy part of TriMul is the O(N³) reduction:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
This is exactly a batched matrix multiplication
    (B·H, N, N) × (B·H, N, N)ᵀ
for which we provide a custom Triton kernel.

Key optimisations
-----------------
* No permute‑or‑contiguous copies – the kernel works directly on the
  original [B, N, N, H] layout.  The hidden dimension is fused into the
  batch dimension inside the kernel (`pid_hb = program_id(2)`).
* All heavy arithmetic is performed in FP16 with FP32 accumulation,
  fully exploiting H100 Tensor‑Cores.
* Small, Tensor‑Core‑friendly tile sizes (64×64×64) minimise shared‑memory
  traffic and keep the launch occupancy high.
* Masking and gating are done in regular PyTorch (they are cheap compared
  with the O(N³) work) and the result is cast to FP16 before the kernel.

The final LayerNorm, output gating and projection are kept in FP32 for
numerical stability.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul with the second operand transposed.
#   out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    a_ptr,  # const *[fp16]  left  tensor  (B, N, N, H)
    b_ptr,  # const *[fp16]  right tensor  (B, N, N, H)
    c_ptr,  #       *[fp32]  output tensor (B, N, N, H)

    B, N, H,               # static shape parameters
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,   # strides for A
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,   # strides for B
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,   # strides for C

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # Program IDs
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)                # tile row index (i)
    pid_n = tl.program_id(1)                # tile col index (j)
    pid_hb = tl.program_id(2)               # fused batch * hidden index

    # --------------------------------------------------------------
    # Decompose fused index into batch and hidden channel
    # --------------------------------------------------------------
    hid = pid_hb % H
    batch = pid_hb // H

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch * stride_a_batch + hid * stride_a_h
    b_base = b_ptr + batch * stride_b_batch + hid * stride_b_h
    c_base = c_ptr + batch * stride_c_batch + hid * stride_c_h

    # --------------------------------------------------------------
    # Accumulator in FP32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_base \
                 + offs_i[:, None] * stride_a_i \
                 + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)

        # B tile (transposed view): (BLOCK_K, BLOCK_N)
        # We need B[j, k] -> (k, j) layout for tl.dot
        b_ptrs = b_base \
                 + offs_k[:, None] * stride_b_k \
                 + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)

        # Tensor‑core friendly dot (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result back to C (float32)
    # --------------------------------------------------------------
    c_ptrs = c_base \
             + offs_i[:, None] * stride_c_i \
             + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs, acc,
             mask=(mask_i[:, None] & mask_j[None, :]))


# ----------------------------------------------------------------------
# Entrypoint used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N]       (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
              * "dim"         – input channel dimension (C)
              * "hidden_dim" – hidden dimension (H)
              * "nomask"      – bool, True ⇒ ignore mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to cast to half‑precision (FP16)
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(torch.float16)

    # ------------------------------------------------------------------
    # Load model weights (some kept in FP16 for speed)
    # ------------------------------------------------------------------
    # LayerNorm before the projections
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Projection & gate weights (no bias)
    left_proj_w  = to_half(weights["left_proj.weight"])
    right_proj_w = to_half(weights["right_proj.weight"])
    left_gate_w  = to_half(weights["left_gate.weight"])
    right_gate_w = to_half(weights["right_gate.weight"])
    out_gate_w   = to_half(weights["out_gate.weight"])

    # Final norm & projection (keep FP32 for stability)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm_fp32 = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)   # now FP16

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (FP16, bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x_norm, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid, still FP16)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_norm, left_gate_w).sigmoid()
    right_gate = F.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = F.linear(x_norm, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul reduction – batched mat‑mul with transposed operand.
    #    We keep everything in FP16 inside the kernel, accumulate in FP32.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Clamp extreme values before FP16 conversion (Cauchy tails)
    left_fp16  = torch.clamp(left,  -65504.0, 65504.0).to(torch.float16)
    right_fp16 = torch.clamp(right, -65504.0, 65504.0).to(torch.float16)

    # Output buffer (float32) – shape [B, N, N, H]
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # Triton launch configuration – tile sizes tuned for H100.
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # tiles along i (rows)
        triton.cdiv(N, BLOCK_N),          # tiles along j (cols)
        B * hidden_dim,                   # fused batch * hidden dimension
    )

    _tri_mul_kernel[grid](
        # pointers
        left_fp16,
        right_fp16,
        out_fp32,
        # static shape parameters
        B, N, hidden_dim,
        # strides for A (left)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        # strides for B (right)
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        # strides for C (output)
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2), out_fp32.stride(3),
        # compile‑time constants
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final LayerNorm, gating and projection back to original dim
    # ------------------------------------------------------------------
    # out_fp32 currently has shape [B, N, N, H]
    out = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    # Apply the output gate (convert to FP32 for the multiplication)
    out = out * out_gate.to(torch.float32)

    # Linear projection to the original channel dimension (FP32)
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    return out
