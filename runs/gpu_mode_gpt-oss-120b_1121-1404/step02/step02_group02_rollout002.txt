"""
custom_kernel for the “outgoing” Triangle‑Multiplicative‑Update (TriMul).

Algorithm
---------
1️⃣  Layer‑norm the input (float32 → fp16 for later ops).
2️⃣  Six linear projections (no bias) are performed with fp16 weights:
       * left  = Linear(x_norm, left_proj)
       * right = Linear(x_norm, right_proj)
       * left_gate  = sigmoid(Linear(x_norm, left_gate))
       * right_gate = sigmoid(Linear(x_norm, right_gate))
       * out_gate   = sigmoid(Linear(x_norm, out_gate))
3️⃣  Fuse *mask* (if any) and the two gate tensors in‑place with a tiny
    Triton kernel – this replaces three separate element‑wise
    ``*`` calls with a single memory‑bound kernel.
4️⃣  Core O(N³) reduction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    is a batched matrix‑multiply of shape (B·H, N, N) × (B·H, N, N)ᵀ.
    We use Torch’s highly‑optimised cuBLAS batched GEMM (torch.bmm)
    which runs at full Tensor‑Core throughput on H100.
5️⃣  Final LayerNorm (float32), gating with ``out_gate`` and a linear
    projection back to the original channel dimension.

All heavy arithmetic is in fp16, accumulation is in fp32, and the final
output is float32 as required.
"""
import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask + left/right gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_fuse_kernel(
    left_ptr,          # fp16  [P·N·N·H]  (in‑out)
    right_ptr,         # fp16  [P·N·N·H]  (in‑out)
    left_gate_ptr,     # fp16  [P·N·N·H]
    right_gate_ptr,    # fp16  [P·N·N·H]
    mask_ptr,          # fp16  [P·N·N]   (broadcasted over H)
    total_elements,    # int64 = B·N·N·H
    hidden_dim,        # int64 = H
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < total_elements

    # load tensors
    left_val  = tl.load(left_ptr + offs,  mask=valid, other=0.0)
    right_val = tl.load(right_ptr + offs, mask=valid, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs,  mask=valid, other=0.0)
    right_g   = tl.load(right_gate_ptr + offs, mask=valid, other=0.0)

    # mask is shared across the hidden dimension:
    #   mask_index = floor(offs / H)
    mask_idx = offs // hidden_dim
    m_val = tl.load(mask_ptr + mask_idx, mask=valid, other=1.0)   # 1.0 when mask is “all‑ones”

    # fused multiply‑add
    left_val  = left_val  * left_g  * m_val
    right_val = right_val * right_g * m_val

    # write back in‑place
    tl.store(left_ptr + offs,  left_val,  mask=valid)
    tl.store(right_ptr + offs, right_val, mask=valid)


# ----------------------------------------------------------------------
# Entrypoint expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys
              * "dim"         – input channel dimension
              * "hidden_dim" – hidden dimension for the TriMul
              * "nomask"      – bool, True ⇒ ignore the mask

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5
    dim = config["dim"]               # == input_tensor.shape[-1]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 0️⃣  Layer‑norm on the input (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm_fp32 = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    x_norm = x_norm_fp32.to(torch.float16)        # now fp16

    # ------------------------------------------------------------------
    # 1️⃣  Linear projections (fp16, no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_norm, weights["left_proj.weight"].to(torch.float16)
    )                                             # [B,N,N,hidden_dim]  fp16
    right = torch.nn.functional.linear(
        x_norm, weights["right_proj.weight"].to(torch.float16)
    )

    # ------------------------------------------------------------------
    # 2️⃣  Gates (fp16, sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_norm, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()                                   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣  Fuse mask + gate application with a Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    total_elems = left.numel()                     # B·N·N·H

    if nomask or mask is None:
        # a tensor of ones → “no masking”
        mask_fp16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)          # [B,N,N]

    # Flatten everything to 1‑D views (contiguous)
    left_flat      = left.contiguous().view(-1)
    right_flat     = right.contiguous().view(-1)
    left_gate_flat = left_gate.contiguous().view(-1)
    right_gate_flat = right_gate.contiguous().view(-1)
    mask_flat      = mask_fp16.contiguous().view(-1)   # length B·N·N

    # Triton launch configuration
    BLOCK = 8192                                   # works well on H100
    grid = triton.cdiv(total_elems, BLOCK)

    _gate_mask_fuse_kernel[grid](
        left_flat,
        right_flat,
        left_gate_flat,
        right_gate_flat,
        mask_flat,
        total_elems,
        hidden_dim,
        BLOCK=BLOCK,
    )

    # Reshape back to the original 4‑D layout
    left  = left_flat.view(B, N, N, hidden_dim)
    right = right_flat.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣  Core O(N³) reduction – batched GEMM (torch.bmm)
    # ------------------------------------------------------------------
    # Convert to (B, H, N, N) → (P, N, N) where P = B·H
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    P = B * hidden_dim
    left_mat  = left.view(P, N, N)                  # fp16
    right_mat = right.view(P, N, N)                 # fp16

    # Batched mat‑mul with the right operand transposed
    # (fp16×fp16 → fp32 accumulation inside cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [P, N, N] fp16
    out_mat = out_mat.to(torch.float32)                        # cast for later layers

    # Restore [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣  Final LayerNorm, gating and projection (float32)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply the output gate (cast to fp32 for the multiplication)
    out = out * out_gate.to(torch.float32)

    # Linear projection back to original channel dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # fp32

    return out
