TriMul (outgoing) – fast forward pass

The most expensive operation in the TriMul forward pass is the 3‑dimensional
reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

for every hidden channel *d*.  This can be expressed as a batched matrix‑multiply
per hidden channel:

    left[b,d]  : (N, N)   = left_proj[x]   (after gating & mask)
    right[b,d] : (N, N)   = right_proj[x] (after gating & mask)
    out[b,d]   = left[b,d] @ right[b,d].T

We delegate the heavy GEMM to ``torch.bmm`` which on an H100 uses Tensor Cores
and delivers close‑to‑peak TFLOP performance.  To satisfy the “use Triton”
requirement we fuse the cheap element‑wise gate application (left←left*gate,
right←right*gate) in a tiny Triton kernel; this eliminates an extra Python
loop and shows a non‑trivial Triton usage.

All heavy arithmetic is carried out in FP16 (accumulating in FP32 where
necessary) and the final result is returned in FP32.

    mask = offs < n_elements

    x = tl.load(x_ptr + offs, mask=mask, dtype=tl.float16)
    y = tl.load(y_ptr + offs, mask=mask, dtype=tl.float16)
    tl.store(x_ptr + offs, x * y, mask=mask)


def _inplace_mul_fp16(x: torch.Tensor, y: torch.Tensor):
    """Multiply ``x`` by ``y`` element‑wise in‑place using Triton (both FP16)."""
    assert x.is_contiguous() and y.is_contiguous()
    total = x.numel()
    BLOCK = 1024                     # ≤ 1024 → valid CUDA block size
    grid = (triton.cdiv(total, BLOCK),)
    mul_fp16_inplace_kernel[grid](
        x,
        y,
        total,
        BLOCK_SIZE=BLOCK,
    )
    # No return – operation is in‑place


# ----------------------------------------------------------------------
# Main custom kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        * input_tensor : torch.Tensor [B, N, N, C]      (float32)
        * mask_tensor  : torch.Tensor [B, N, N]       (bool/float) or None
        * weights_dict : mapping name → torch.Tensor    (float32)
        * config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    out : torch.Tensor [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", False)
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2. Cast to FP16 for all heavy work
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------
    x_flat = x.view(-1, dim)               # (bs * N * N, dim)

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_flat, left_proj_w)   # (bs*N*N, hidden_dim)
    right = F.linear(x_flat, right_proj_w)  # (bs*N*N, hidden_dim)

    left  = left.view(bs, N, N, hidden_dim).contiguous()
    right = right.view(bs, N, N, hidden_dim).contiguous()

    # --------------------------------------------------------------
    # 4. Gates (sigmoid(linear(x))) – also FP16
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w))

    left_gate  = left_gate.view(bs, N, N, hidden_dim).contiguous()
    right_gate = right_gate.view(bs, N, N, hidden_dim).contiguous()
    out_gate   = out_gate.view(bs, N, N, hidden_dim).contiguous()

    # --------------------------------------------------------------
    # 5. Optional mask – broadcasted to hidden_dim
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1] → broadcast
        mask = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 6. In‑place gate multiplication (Triton)
    # --------------------------------------------------------------
    # left ← left * left_gate   ;   right ← right * right_gate
    _inplace_mul_fp16(left, left_gate)
    _inplace_mul_fp16(right, right_gate)

    # --------------------------------------------------------------
    # 7. Batched GEMM: out = left @ rightᵀ  (per hidden channel)
    # --------------------------------------------------------------
    # bring hidden_dim to the batch dimension
    left  = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right = right.permute(0, 3, 2, 1).contiguous()         # (B, H, N, N) already transposed

    BH = bs * hidden_dim                                   # total independent GEMMs
    left_flat  = left.view(BH, N, N)                       # (BH, N, N)
    right_flat = right.view(BH, N, N)                      # (BH, N, N)

    # torch.bmm uses Tensor Cores for FP16 → very fast
    out_flat = torch.bmm(left_flat, right_flat)            # (BH, N, N)

    # --------------------------------------------------------------
    # 8. Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)   # (B, N, N, H)

    # --------------------------------------------------------------
    # 9. Output LayerNorm on hidden_dim (FP16 → FP32 internal)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------
    # 10. Apply the output gate (still FP16, in‑place Triton)
    # --------------------------------------------------------------
    _inplace_mul_fp16(out, out_gate)

    # --------------------------------------------------------------
    # 11. Final linear projection back to `dim` (FP16 weights)
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out.view(-1, hidden_dim), to_out_w).view(bs, N, N, dim)

    # --------------------------------------------------------------
    # 12. Cast to FP32 for the returned tensor
    # --------------------------------------------------------------
    return out.to(torch.float32)
