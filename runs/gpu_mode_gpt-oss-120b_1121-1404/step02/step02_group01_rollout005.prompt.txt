import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing) – a highly‑optimized Triton implementation.

The forward pass consists of:
1. LayerNorm on the input (float32).
2. Half‑precision linear projections and sigmoid gates
   (weights are cast to float16 so that cuBLAS uses Tensor‑cores).
3. A fused N³ “outer‑product” for every hidden channel:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is performed by a custom Triton GEMM kernel that
   treats each (batch, hidden) pair as an independent page and
   computes A @ Bᵀ with FP16 inputs and FP32 accumulation.
   The kernel uses 8 warps, double‑buffered loads (num_stages=2)
   and 128×128×32 tiles (or 64×64×16 for very small N) to
   fully exploit H100 Tensor‑cores.
4. LayerNorm on the hidden dimension, an output‑gate (sigmoid)
   and a final linear projection back to the original dim.

All heavy arithmetic runs in half precision; the only FP32
operations are the two LayerNorms and the final linear layer.
The kernel returns FP32 values directly, avoiding an extra
half‑to‑float conversion.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    a_ptr,                     # const half*   (left)
    b_ptr,                     # const half*   (right, accessed as transposed)
    c_ptr,                     # float*        (output)
    batch_hidden, N,           # int32 scalars
    stride_az, stride_ai, stride_ak,   # strides for A (page, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (page, k‑dim, n‑dim)
    stride_cz, stride_ci, stride_cj,   # strides for C (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a block of C = A @ Bᵀ.
    A : (batch_hidden, N, N) – row‑major, dtype half
    B : (batch_hidden, N, N) – row‑major, accessed as Bᵀ
    C : (batch_hidden, N, N) – row‑major, dtype float32
    """
    pid_z = tl.program_id(0)       # which (batch, hidden) page
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # --------------------------- block offsets ---------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to the beginning of the current page
    a_page = a_ptr + pid_z * stride_az
    b_page = b_ptr + pid_z * stride_bz
    c_page = c_ptr + pid_z * stride_cz

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------- reduction over K -----------------------
    cur_k = 0
    while cur_k < N:
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_offs < N

        # ----- load A tile (row‑major) -----
        a_ptrs = a_page + \
                 offs_m[:, None] * stride_ai + \
                 cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # half -> automatically FP16

        # ----- load B tile (as Bᵀ) -----
        b_ptrs = b_page + \
                 cur_k_offs[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # half

        # ----- multiply‑accumulate (FP16×FP16→FP32) -----
        acc += tl.dot(a, b)

        cur_k += BLOCK_K

    # --------------------------- write back ----------------------------
    c_ptrs = c_page + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------ unpack
    x, mask, weights, config = data
    # x : [B, N, N, dim]  (float32)
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------ half‑precision weights for the expensive ops
    # keep LayerNorm and the final Linear in FP32 (they are cheap)
    w_fp16 = {
        "left_proj.weight":   weights["left_proj.weight"].to(torch.float16),
        "right_proj.weight":  weights["right_proj.weight"].to(torch.float16),
        "left_gate.weight":   weights["left_gate.weight"].to(torch.float16),
        "right_gate.weight":  weights["right_gate.weight"].to(torch.float16),
        "out_gate.weight":    weights["out_gate.weight"].to(torch.float16),
    }

    # ------------------------------------------------------------------ LayerNorm on the input (FP32)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------ cast to half for the heavy linear layers
    x_norm_h = x_norm.to(torch.float16)   # [B, N, N, dim]  (half)

    # ------------------------------------------------------------------ Linear projections (half precision, no bias)
    left = F.linear(x_norm_h, w_fp16["left_proj.weight"])    # [B, N, N, hidden]
    right = F.linear(x_norm_h, w_fp16["right_proj.weight"])  # [B, N, N, hidden]

    # ------------------------------------------------------------------ optional mask (half)
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------ Gates (half → half after sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["out_gate.weight"]))

    # ------------------------------------------------------------------ Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ Rearrange to (B, hidden, N, N) – contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    # collapse batch & hidden into a single page dimension
    BxH = B * hidden
    left_mat = left.view(BxH, N, N)    # half
    right_mat = right.view(BxH, N, N)  # half

    # ------------------------------------------------------------------ Allocate output buffer (FP32 for direct accumulation)
    out_mat = torch.empty((BxH, N, N), dtype=torch.float32, device=x.device)

    # --------------------------------------------------------------- strides (in elements)
    stride_az = left_mat.stride(0)   # page stride = N*N
    stride_ai = left_mat.stride(1)   # row stride = N
    stride_ak = left_mat.stride(2)   # col stride = 1

    stride_bz = right_mat.stride(0)  # page stride
    stride_bk = right_mat.stride(2)  # =1  (k‑dim stride)
    stride_bn = right_mat.stride(1)  # =N  (n‑dim / output‑col stride)

    stride_cz = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # -------------------------------------------------------------- block sizes
    # Choose tile size depending on N to keep occupancy high
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 16
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    grid = (BxH,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # --------------------------------------------------------------- launch kernel
    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BxH,
        N,
        stride_az, stride_ai, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 performs best with 8‑warp kernels
        num_stages=2,         # double‑buffered loads
    )

    # ------------------------------------------------------------------ reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)   # float32

    # ------------------------------------------------------------------ LayerNorm over hidden dimension (FP32)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------ Apply output gate (cast to FP32 for safety)
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------ Final linear projection back to dim (FP32)
    out = F.linear(out, weights["to_out.weight"])

    return out
