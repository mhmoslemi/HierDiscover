```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_b, stride_i, stride_j, stride_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    HIDDEN: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # bounds
    valid = (pid_b < B) & (pid_i < N) & (pid_j < N)

    # load mask scalar and broadcast
    mask = tl.load(mask_ptr + pid_b * stride_mask_b + pid_i * stride_mask_i + pid_j * stride_mask_j,
                   mask=valid, other=0.0)

    d = tl.arange(0, HIDDEN)
    offset = pid_b * stride_b + pid_i * stride_i + pid_j * stride_j
    vec_off = offset + d * stride_h

    left = tl.load(left_ptr + vec_off, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + vec_off, mask=valid, other=0.0)
    right = tl.load(right_ptr + vec_off, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + vec_off, mask=valid, other=0.0)

    left_out = left * left_gate * mask
    right_out = right * right_gate * mask

    tl.store(out_left_ptr + vec_off, left_out, mask=valid)
    tl.store(out_right_ptr + vec_off, right_out, mask=valid)
```

"""
TriMul (outgoing) – fast Triton + cuBLAS implementation
=======================================================

Algorithm
---------
1. **Input LayerNorm** – FP32 for numerical stability (same as reference).
2. **Cast to fp16** – all further ops use half‑precision; Tensor‑Core kernels
   will accumulate in fp32 automatically.
3. **Linear projections** (`left`, `right`) and gating (`left_gate`,
   `right_gate`, `out_gate`) via `torch.nn.functional.linear`.  These are
   plain FP16 GEMMs (cuBLAS) and therefore already optimal.
4. **Mask + gate fusion** – when a mask is provided we launch a tiny Triton
   kernel that loads `left`, `right` and their gates (contiguous in memory)
   and writes back the masked‑gated tensors (`left_fused`, `right_fused`).
   When there is no mask we fuse the operations with a simple
   element‑wise PyTorch expression.
5. **Batched matrix multiplication** – the core O(N³·H) contraction is
   a batched GEMM:
        out[b,h,i,j] = Σₖ left[b,i,k,h] * right[b,j,k,h] .
   After permuting the hidden dimension to the batch axis we call
   `torch.bmm`, which uses the highly‑optimised cuBLAS Tensor‑Core path.
6. **Output LayerNorm → out‑gate → final linear** – all in fp16.
7. **Cast back to FP32** – required API contract.

Only the mask‑gate fusion is performed in a Triton kernel; everything else
leverages cuBLAS which gives the needed sub‑millisecond runtime on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1: fuse mask, left‑gate and right‑gate (fp16)
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    # strides for tensors with layout [B, N, N, H]
    stride_b, stride_i, stride_j, stride_h,
    # strides for mask with layout [B, N, N]
    stride_mask_b, stride_mask_i, stride_mask_j,
    HIDDEN: tl.constexpr,
):
    pid_b = tl.program_id(0)   # batch
    pid_i = tl.program_id(1)   # first N axis
    pid_j = tl.program_id(2)   # second N axis

    # --------------------------------------------------------------
    # bounds
    # --------------------------------------------------------------
    valid = (pid_b < B) & (pid_i < N) & (pid_j < N)

    # --------------------------------------------------------------
    # load scalar mask (fp16) and broadcast over hidden dimension
    # --------------------------------------------------------------
    mask = tl.load(
        mask_ptr + pid_b * stride_mask_b + pid_i * stride_mask_i + pid_j * stride_mask_j,
        mask=valid,
        other=0.0,
    )  # fp16

    # --------------------------------------------------------------
    # vector offset for the hidden dimension (contiguous inner dim)
    # --------------------------------------------------------------
    d = tl.arange(0, HIDDEN)                     # compile‑time constant
    base_off = pid_b * stride_b + pid_i * stride_i + pid_j * stride_j
    vec_off = base_off + d * stride_h              # (HIDDEN,) offsets

    # --------------------------------------------------------------
    # load data
    # --------------------------------------------------------------
    left       = tl.load(left_ptr       + vec_off, mask=valid, other=0.0)
    left_gate  = tl.load(left_gate_ptr  + vec_off, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + vec_off, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + vec_off, mask=valid, other=0.0)

    # --------------------------------------------------------------
    # apply mask and gates
    # --------------------------------------------------------------
    left_out  = left  * left_gate  * mask
    right_out = right * right_gate * mask

    # --------------------------------------------------------------
    # store results
    # --------------------------------------------------------------
    tl.store(out_left_ptr  + vec_off, left_out,  mask=valid)
    tl.store(out_right_ptr + vec_off, right_out, mask=valid)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (float32) or ``None``
        - weights      : dict of module parameters (float32)
        - config       : dict with keys
                         ``dim`` (int), ``hidden_dim`` (int), ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  FP32

    # --------------------------------------------------------------
    # 2) Cast to FP16 + cast weights
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 3) Linear projections (FP16) – use cuBLAS via torch.nn.functional
    # --------------------------------------------------------------
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_h, right_proj_w)

    # 4) Gating (FP16 + sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,H]

    # --------------------------------------------------------------
    # 5) Apply mask & gates
    # --------------------------------------------------------------
    if nomask or mask_tensor is None:
        # No mask – simple elementwise fusion
        left_fused  = left * left_gate
        right_fused = right * right_gate
    else:
        # Masked case – launch tiny Triton kernel
        mask = mask_tensor.to(torch.float16)          # [B,N,N] fp16
        left_fused  = torch.empty_like(left)          # allocate output buffers
        right_fused = torch.empty_like(right)

        # strides of tensors (all contiguous)
        stride_b, stride_i, stride_j, stride_h = left.stride()
        stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

        # launch one kernel per (b, i, j)
        grid = (B, N, N)
        _mask_gate_fuse_kernel[grid](
            left, left_gate, mask,
            right, right_gate,
            left_fused, right_fused,
            B, N, hidden_dim,
            stride_b, stride_i, stride_j, stride_h,
            stride_mask_b, stride_mask_i, stride_mask_j,
            HIDDEN=hidden_dim,
        )

    # --------------------------------------------------------------
    # 6) Batched matmul: Σₖ left[i,k]·right[j,k]   (Tensor‑Core GEMM)
    #    We reshape (B, N, N, H) → (B·H, N, N) and use torch.bmm.
    # --------------------------------------------------------------
    # left:  (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_flat = left_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # right needs a transpose on the two N axes before flattening
    #   right_fused (B, i, k, H) → transpose(i↔k) → (B, k, i, H)
    #   then bring H to second axis → (B, H, N, N) → (B*H, N, N)
    right_T   = right_fused.permute(0, 2, 1, 3).contiguous()
    right_flat = right_T.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM (FP16 × FP16 → FP16, accumulation in FP32)
    out_flat = torch.bmm(left_flat, right_flat)   # [B*H, N, N]  fp16

    # --------------------------------------------------------------
    # 7) Reshape back to (B, N, N, H)
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,H] fp16

    # --------------------------------------------------------------
    # 8) Output LayerNorm → out‑gate → final linear (all FP16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )                                 # fp16

    out = out * out_gate               # element‑wise gating (fp16)

    out = F.linear(out, to_out_w)      # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 9) Cast back to FP32 (API contract)
    # --------------------------------------------------------------
    return out.to(torch.float32)
