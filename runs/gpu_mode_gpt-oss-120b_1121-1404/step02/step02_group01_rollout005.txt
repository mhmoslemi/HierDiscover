    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Strides for out and gate: assuming contiguous layout (B,N,N,hidden)
    stride_ob = tl.load(stride_ob_ptr) ? Not needed: we can compute based on tensor strides passed as scalars.

    # out stride: (N,N,hidden)
    # Actually we can pass strides: stride_b, stride_i, stride_j, stride_h

    # Need to read sum and sum_sq
    sum_val = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for h in range(HIDDEN):
        ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj + h * stride_oh
        val = tl.load(ptr, mask=mask_m[:, None] & mask_n[None, :], other=0.0)
        sum_val += val

    mean = sum_val / HIDDEN

    inv_std = 1.0 / tl.sqrt(var + EPS)

    # loop again to apply norm weight, bias, gate
    for h in range(HIDDEN):
        # load val
        ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None,:] * stride_oj + h * stride_oh

```
@triton.jit
def norm_gate_kernel(
    out_ptr,         # float* output & input (in-place)
    gate_ptr,        # half*
    norm_weight_ptr, # float* (hidden)
    norm_bias_ptr,   # float* (hidden)
    B, N, HIDDEN,
    stride_ob, stride_oi, stride_oj, stride_oh,
    stride_gb, stride_gi, stride_gj, stride_gh,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HIDDEN: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulators for mean and variance
    sum_val = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # First pass: compute sum and sum of squares across hidden
    for h in range(HIDDEN):
        ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj + h * stride_oh
        val = tl.load(ptr, mask=mask_m[:, None] & mask_n[None, :], other=0.0)
        sum_val += val
        sum_sq += val * val

    mean = sum_val / HIDDEN
    var = sum_sq / HIDDEN - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Second pass: apply norm weight, bias, gate and store back
    for h in range(HIDDEN):
        # load raw value
        ptr = out_ptr + pid_b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj + h * stride_oh
        val = tl.load(ptr, mask=mask_m[:, None] & mask_n[None, :], other=0.0)

        # normalize
        normed = (val - mean) * inv_std

        # apply layernorm scale and shift
        w = tl.load(norm_weight_ptr + h)
        b = tl.load(norm_bias_ptr + h)
        normed = normed * w + b

        # load gate
        g_ptr = gate_ptr + pid_b * stride_gb + offs_m[:, None] * stride_gi + offs_n[None, :] * stride_gj + h * stride_gh
        gate_val = tl.load(g_ptr, mask=mask_m[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # apply gate
        out_val = normed * gate_val

        # store back
        tl.store(ptr, out_val, mask=mask_m[:, None] & mask_n[None, :])
```

"""
Triton‑accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1.  Layer‑Norm the input tensor (FP32).
2.  Cast to FP16 and apply the four linear projections (no bias):
        left  = x·W_left   , right = x·W_right
        left_gate , right_gate , out_gate = sigmoid(x·W_gate)
    Optionally mask the two projected tensors.
3.  Fuse the hidden dimension into a batch dimension and perform a
    batched GEMM (left @ rightᵀ) with torch.bmm which uses cuBLAS/
    Tensor‑Cores.  Result: out_hidden  ∈ ℝ^{B×N×N×H}.
4.  Apply the output LayerNorm, the output gate and the scale‑bias of
    the LayerNorm in a single Triton kernel (in‑place on out_hidden).
5.  Final linear projection back to the original channel dimension
    (FP32).

Only the heavy outer‑product (step 3) is performed by cuBLAS; the
LayerNorm + gate fusion (step 4) is written in Triton to avoid three
separate memory passes.  All arithmetic except the final LayerNorm and
linear projection is in half precision, matching the reference
implementation while achieving much higher throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def norm_gate_kernel(
    out_ptr,                     # float*  (in‑place hidden tensor)
    gate_ptr,                    # half*   (output gate)
    norm_w_ptr,                  # float*  (LayerNorm scale, shape [H])
    norm_b_ptr,                  # float*  (LayerNorm bias, shape [H])
    B, N, HIDDEN,                # compile‑time constants
    stride_ob, stride_oi, stride_oj, stride_oh,   # strides of out_ptr (B,N,N,H)
    stride_gb, stride_gi, stride_gj, stride_gh,   # strides of gate_ptr
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    HIDDEN: tl.constexpr,
    EPS: tl.constexpr = 1e-5,
):
    """
    For each tile (BLOCK_M×BLOCK_N) of positions (i,j) computes:
        μ  = mean_h out[b,i,j,h]
        σ² = var_h  out[b,i,j,h]
        out = ((out-μ)/√(σ²+ε) * norm_w + norm_b) * gate
    The operation is performed in‑place on ``out_ptr``.
    """
    pid_b = tl.program_id(0)          # batch index
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile column

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]

    # ---------- first pass: accumulate sum and sum of squares ----------
    sum_val = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for h in range(HIDDEN):
        # address of out[b,i,j,h]
        out_addr = (
            out_ptr
            + pid_b * stride_ob
            + offs_m[:, None] * stride_oi
            + offs_n[None, :] * stride_oj
            + h * stride_oh
        )
        val = tl.load(out_addr, mask=mask, other=0.0)          # float32
        sum_val += val
        sum_sq  += val * val

    mean = sum_val / HIDDEN
    var  = sum_sq / HIDDEN - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ---------- second pass: normalize, apply scale/bias and gate ----------
    for h in range(HIDDEN):
        # load the raw hidden value
        out_addr = (
            out_ptr
            + pid_b * stride_ob
            + offs_m[:, None] * stride_oi
            + offs_n[None, :] * stride_oj
            + h * stride_oh
        )
        val = tl.load(out_addr, mask=mask, other=0.0)          # float32

        # normalize
        normed = (val - mean) * inv_std

        # LayerNorm weight & bias (scalar per hidden channel)
        w = tl.load(norm_w_ptr + h)      # float32
        b = tl.load(norm_b_ptr + h)      # float32
        normed = normed * w + b

        # gate (half → float32)
        gate_addr = (
            gate_ptr
            + pid_b * stride_gb
            + offs_m[:, None] * stride_gi
            + offs_n[None, :] * stride_gj
            + h * stride_gh
        )
        gate_val = tl.load(gate_addr, mask=mask, other=0.0).to(tl.float32)

        out_val = normed * gate_val

        # store back in‑place
        tl.store(out_addr, out_val, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------ unpack
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------ half‑precision heavy weights
    w_fp16 = {
        "left_proj.weight":   weights["left_proj.weight"].to(torch.float16),
        "right_proj.weight":  weights["right_proj.weight"].to(torch.float16),
        "left_gate.weight":   weights["left_gate.weight"].to(torch.float16),
        "right_gate.weight":  weights["right_gate.weight"].to(torch.float16),
        "out_gate.weight":    weights["out_gate.weight"].to(torch.float16),
    }

    # ------------------------------------------------------------------ Input LayerNorm (FP32)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------ Cast to FP16 for the heavy linear layers
    x_norm_h = x_norm.to(torch.float16)   # [B, N, N, dim]  (FP16)

    # ------------------------------------------------------------------ Projections (no bias, FP16)
    left  = F.linear(x_norm_h, w_fp16["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x_norm_h, w_fp16["right_proj.weight"])  # [B, N, N, hidden]

    # ------------------------------------------------------------------ Optional mask (FP16)
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)      # [B, N, N, 1]
        left  = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------ Gates (FP16 → sigmoid → FP16)
    left_gate  = torch.sigmoid(F.linear(x_norm_h, w_fp16["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm_h, w_fp16["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm_h, w_fp16["out_gate.weight"]))

    # ------------------------------------------------------------------ Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # reshape to (B*hidden, N, N) and use torch.bmm (cuBLAS Tensor‑Cores)
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    B, hidden, N, _ = left.shape
    left_mat  = left.view(B * hidden, N, N)          # FP16
    right_mat = right.view(B * hidden, N, N)         # FP16

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2)).to(torch.float32)  # [B*hidden, N, N]

    # ------------------------------------------------------------------ Reshape to [B, N, N, hidden] (contiguous)
    out_hidden = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # FP32

    # ------------------------------------------------------------------ Fuse output LayerNorm + out_gate using Triton
    # Tensor shapes:
    #   out_hidden : [B, N, N, hidden]  (float32, contiguous)
    #   out_gate   : [B, N, N, hidden]  (float16, contiguous)
    out_gate = out_gate.contiguous()  # ensure stride layout matches out_hidden

    BLOCK_M, BLOCK_N = 32, 32                # tile size (32×32 → 1024 threads per block)
    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    norm_gate_kernel[grid](
        out_hidden,
        out_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        B, N, hidden,
        out_hidden.stride(0), out_hidden.stride(1), out_hidden.stride(2), out_hidden.stride(3),
        out_gate.stride(0),   out_gate.stride(1),   out_gate.stride(2),   out_gate.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        HIDDEN=hidden,
        EPS=1e-5,
        num_warps=4,          # 4‑warp kernel gives good occupancy for this tile size
        num_stages=2,         # double‑buffered loads
    )

    # ------------------------------------------------------------------ Final linear back to the original dim (FP32)
    out = F.linear(out_hidden, weights["to_out.weight"])   # [B, N, N, dim]

    return out
