import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
fast_trimul.py
----------------
Triton‑accelerated **outgoing** TriMul (AlphaFold‑3) implementation.

The core of the operator is the N³ reduction

    out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, k, j]

where `left`/`right` are the linearly‑projected and gated pairwise
features.  For every head (hidden‑dim) we therefore need a
batch‑wise matrix‑multiply `left @ rightᵀ`.  The implementation below
fuses the GEMM across all heads with a single Triton kernel, *without*
materialising the transposed, contiguous `(B·H, N, N)` tensors that the
reference code creates.  By using strided loads we avoid a
≈ 0.5 GiB copy for the largest test case (B=2, N=1024, H=384) and
reduce kernel launch overhead.  The GEMM works in fp16 (fast tensor‑core
arithmetic) while accumulating in fp32 for numerical stability.

The surrounding Light‑weigth ops (LayerNorm, linear projections,
sigmoids, masking) are performed with PyTorch‑CUDA which is already
highly optimised.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def outgoing_trimul_kernel(
    left_ptr,                     # fp16, shape (B, H, N, N)  (i‑k)
    right_ptr,                    # fp16, shape (B, H, N, N)  (k‑j)
    out_ptr,                      # fp16, shape (B, H, N, N)  (i‑j)
    seq_len,                      # int == N
    # left strides
    l_str_batch, l_str_head, l_str_i, l_str_k,
    # right strides
    r_str_batch, r_str_head, r_str_k, r_str_j,
    # out strides
    o_str_batch, o_str_head, o_str_i, o_str_j,
    hidden_dim,                   # int = H
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    # -------------------------------------------------------------
    # grid layout: (bh_id, block_i, block_j)
    # -------------------------------------------------------------
    bh_id   = tl.program_id(0)          # 0 … B*H-1
    block_i = tl.program_id(1)          # row tile
    block_j = tl.program_id(2)          # column tile

    # split bh_id into batch and head indices
    b_id = bh_id // hidden_dim
    h_id = bh_id % hidden_dim

    # base offsets for the three tensors (in elements)
    left_base  = b_id * l_str_batch + h_id * l_str_head
    right_base = b_id * r_str_batch + h_id * r_str_head
    out_base   = b_id * o_str_batch + h_id * o_str_head

    # -------------------------------------------------------------
    # tile offsets (i‑dimension, j‑dimension, k‑dimension)
    # -------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 (tensor‑core dot works in FP32 precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # K‑dimension loop
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # ----- left tile (i, k) -------------------------------------------------
        a_offset = (left_base
                    + offs_i[:, None] * l_str_i
                    + offs_k[None, :] * l_str_k)
        mask_a = (offs_i[:, None] < seq_len) & (offs_k[None, :] < seq_len)

        # ----- right tile (k, j) ------------------------------------------------
        b_offset = (right_base
                    + offs_k[:, None] * r_str_k
                    + offs_j[None, :] * r_str_j)
        mask_b = (offs_k[:, None] < seq_len) & (offs_j[None, :] < seq_len)

        a = tl.load(left_ptr + a_offset, mask=mask_a, other=0.0)
        b = tl.load(right_ptr + b_offset, mask=mask_b, other=0.0)

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # write back
    # -------------------------------------------------------------
    c_offset = (out_base
                + offs_i[:, None] * o_str_i
                + offs_j[None, :] * o_str_j)
    mask_c = (offs_i[:, None] < seq_len) & (offs_j[None, :] < seq_len)
    tl.store(out_ptr + c_offset, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using Triton.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Cast to half and flatten for linear layers
    # ------------------------------------------------------------------
    x = x.to(torch.float16)
    x_flat = x.view(-1, dim)                      # (bs*N*N, dim)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias) – fp16
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left  = F.linear(x_flat, left_proj_w).view(bs, N, N, hidden_dim)   # (B, N, N, H)
    right = F.linear(x_flat, right_proj_w).view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid(linear(x))) – fp16
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(bs, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 6) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7) Prepare tensors for the batched GEMM.
    #    We keep the tensors **non‑contiguous** to avoid the costly copy.
    # ------------------------------------------------------------------
    # left : (B, N, N, H) -> (B, H, N, N)  (i‑k)
    left_perm = left.permute(0, 3, 1, 2)          # (B, H, N, N)  not contiguous
    # right: (B, N, N, H) -> (B, H, N, N) where axes are (k, j)
    right_perm = right.permute(0, 3, 2, 1)       # (B, H, N, N)  not contiguous

    # output buffer (B, H, N, N) – fp16
    out_perm = torch.empty((bs, hidden_dim, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 8) Strides (in elements) – required for the Triton kernel
    # ------------------------------------------------------------------
    l_str = left_perm.stride()    # (batch, head, i, k)
    r_str = right_perm.stride()   # (batch, head, k, j)
    o_str = out_perm.stride()     # (batch, head, i, j)

    # ------------------------------------------------------------------
    # 9) Triton launch configuration
    # ------------------------------------------------------------------
    # Block sizes tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    num_warps = 8

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    grid = (bs * hidden_dim, num_blocks_m, num_blocks_n)

    outgoing_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,                                 # seq_len
        # left strides
        l_str[0], l_str[1], l_str[2], l_str[3],
        # right strides
        r_str[0], r_str[1], r_str[2], r_str[3],
        # out strides
        o_str[0], o_str[1], o_str[2], o_str[3],
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ------------------------------------------------------------------
    # 10) Bring the GEMM result back to the original layout
    # ------------------------------------------------------------------
    # out_perm: (B, H, N, N) -> (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 11) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 12) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 13) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out.view(-1, hidden_dim), weights["to_out.weight"]).view(bs, N, N, dim)

    return out
