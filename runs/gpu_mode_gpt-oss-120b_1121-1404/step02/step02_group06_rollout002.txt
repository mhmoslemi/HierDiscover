def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # layernorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # flatten and cast to fp16
    x_flat = x.view(-1, dim).to(torch.float16)

    # concatenate left_proj and left_gate weight
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    left_gate_weight = weights["left_gate.weight"]  # shape (hidden_dim, dim)

    left_concat_weight = torch.cat([left_proj_weight, left_gate_weight], dim=0).to(torch.float16) # (2*hidden_dim, dim)

    # same for right
    right_proj_weight = weights["right_proj.weight"]
    right_gate_weight = weights["right_gate.weight"]
    right_concat_weight = torch.cat([right_proj_weight, right_gate_weight], dim=0).to(torch.float16)

    # linear
    left_concat = torch.nn.functional.linear(x_flat, left_concat_weight)  # shape (M, 2*hidden_dim)
    right_concat = torch.nn.functional.linear(x_flat, right_concat_weight)  # shape (M, 2*hidden_dim)

    # split
    left_proj_fp16, left_gate_fp16 = left_concat.split([hidden_dim, hidden_dim], dim=1)
    right_proj_fp16, right_gate_fp16 = right_concat.split([hidden_dim, hidden_dim], dim=1)

    # gate sigmoid
    left_gate_fp16 = torch.sigmoid(left_gate_fp16)
    right_gate_fp16 = torch.sigmoid(right_gate_fp16)

    # mask broadcast
    if config.get("nomask", False):
        left = left_proj_fp16 * left_gate_fp16
        right = right_proj_fp16 * right_gate_fp16
    else:
        # mask shape (bs,N,N) and broadcast across hidden_dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)  # shape (bs,N,N,1)
        # we need to broadcast over hidden_dim; to avoid expansion we can compute masked multiplication using Triton kernel.
        # We'll compute left = left_proj * left_gate * mask
        total = left_proj_fp16.numel()  # bs*N*N*hidden
        BLOCK = 131072  # maybe bigger
        grid = ( (total + BLOCK - 1) // BLOCK, )
        left = torch.empty_like(left_proj_fp16)
        right = torch.empty_like(right_proj_fp16)
        # We need a kernel that reads mask element per hidden block
        # Access mask like mask_ptr + (offset // hidden_dim)
        @triton.jit
        def fuse_mask_mul(inp_proj_ptr, inp_gate_ptr, mask_ptr, out_ptr, total_elems, hidden, BLOCK_SIZE: tl.constexpr):
            pid = tl.program_id(0)
            offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
            mask = offset < total_elems
            proj = tl.load(inp_proj_ptr + offset, mask=mask, other=0.0)
            gate = tl.load(inp_gate_ptr + offset, mask=mask, other=0.0)
            # compute mask index
            mask_idx = offset // hidden
            m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)
            tl.store(out_ptr + offset, proj * gate * m, mask=mask)

        fuse_mask_mul[grid](
            left_proj_fp16, left_gate_fp16, mask_fp16, left,
            total, hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        fuse_mask_mul[grid](
            right_proj_fp16, right_gate_fp16, mask_fp16, right,
            total, hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    # now left/right shape (bs,N,N,hidden) fp16

    # Now perform batched GEMM
    # left_perm = left.permute(0,3,1,2) -> (bs, hidden, N, N)
    # right_perm = right.permute(0,3,2,1) -> (bs, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    out_fp16 = torch.bmm(left_flat, right_flat)   # (BH,N,N)

    # reshape back
    out = out_fp16.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)  # (bs,N,N,hidden)

    # to_out_norm layer norm on hidden dim
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # out_gate
    out_gate_weight = weights["out_gate.weight"]
    # Compute out_gate using linear + sigmoid (we haven't computed it yet)
    # out_gate_linear = F.linear(x_flat, out_gate_weight) -> shape (M, hidden_dim)
    out_gate_linear = torch.nn.functional.linear(x_flat, out_gate_weight.to(torch.float16))
    out_gate = torch.sigmoid(out_gate_linear).view(bs, N, N, hidden_dim).to(torch.float32)  # convert to float32 for multiplication later (since out is float32)
    out = out * out_gate

    # final projection
    to_out_weight = weights["to_out.weight"]
    # to_out_weight shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out.view(-1, hidden_dim), to_out_weight).view(bs, N, N, dim)

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def masked_fuse_mul_kernel(
    proj_ptr,  # fp16
    gate_ptr,  # fp16
    mask_ptr,  # fp16 (flattened)
    out_ptr,   # fp16
    total_elems,  # i64
    hidden_dim,   # i64
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # broadcast mask over hidden dimension
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)

def custom_kernel(data):
    """
    Outgoing TriMul forward (AlphaFold‑3) – mixed‑precision implementation.
    1. Layer‑norm the input.
    2. Fuse the two linear projections (proj + gate) per side by concatenating
       the weight matrices, reducing the number of GEMMs.
    3. Apply sigmoid to the gate, then fuse mask, gate and projection with a
       small Triton kernel (mask broadcast across the hidden dimension).
    4. Perform the cubic reduction with a single strided‑batched GEMM
       (`torch.bmm`) on fp16 tensors – this is the dominant cost.
    5. Layer‑norm, output‑gate and final linear projection back to `dim`.
    All heavy arithmetic runs in fp16; only the two layer‑norms, the output
    gate and the final projection use fp32 for numerical stability.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (fp32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # (bs, N, N, dim), fp32

    # ------------------------------------------------------------------
    # 2. Flatten and cast to fp16 for the linear layers
    # ------------------------------------------------------------------
    x_flat_fp16 = x.reshape(-1, dim).to(torch.float16)  # (M, dim) where M=bs*N*N

    # ------------------------------------------------------------------
    # 3. Concatenate proj & gate weights (2*hidden_dim, dim) and run GEMM
    # ------------------------------------------------------------------
    # left side
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    left_concat = torch.nn.functional.linear(x_flat_fp16, left_concat_w)  # (M, 2*H)
    left_proj_fp16, left_gate_fp16 = left_concat.split([hidden_dim, hidden_dim], dim=1)

    # right side
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_concat = torch.nn.functional.linear(x_flat_fp16, right_concat_w)
    right_proj_fp16, right_gate_fp16 = right_concat.split([hidden_dim, hidden_dim], dim=1)

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_fp16 = torch.sigmoid(left_gate_fp16)
    right_gate_fp16 = torch.sigmoid(right_gate_fp16)

    # ------------------------------------------------------------------
    # 5. Fuse mask (if present) and gate * projection with Triton
    # ------------------------------------------------------------------
    total = left_proj_fp16.numel()  # = bs * N * N * hidden_dim
    if not nomask:
        # flatten mask to (bs*N*N,) and cast to fp16
        mask_flat_fp16 = mask.to(torch.float16).reshape(-1)  # (M,)
        BLOCK = 131072  # tuneable
        grid = ((total + BLOCK - 1) // BLOCK,)

        left = torch.empty_like(left_proj_fp16)
        right = torch.empty_like(right_proj_fp16)

        masked_fuse_mul_kernel[grid](
            left_proj_fp16,
            left_gate_fp16,
            mask_flat_fp16,
            left,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        masked_fuse_mul_kernel[grid](
            right_proj_fp16,
            right_gate_fp16,
            mask_flat_fp16,
            right,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        left = left_proj_fp16 * left_gate_fp16
        right = right_proj_fp16 * right_gate_fp16

    # reshape back to (bs, N, N, hidden_dim)
    left = left.view(bs, N, N, hidden_dim)
    right = right.view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Cubic reduction (batched GEMM) – fp16
    # ------------------------------------------------------------------
    # Permute to (bs, hidden, N, N) and view as (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose inner dim k<->j

    BH = bs * hidden_dim
    left_mat = left_perm.view(BH, N, N)          # (BH, N, K) where K=N
    right_mat = right_perm.view(BH, N, N)        # (BH, K, N)

    out_fp16 = torch.bmm(left_mat, right_mat)   # (BH, N, N)

    # Restore shape (bs, N, N, hidden_dim) and cast to fp32
    out = out_fp16.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (fp32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )

    # ------------------------------------------------------------------
    # 8. Output gate (sigmoid) – computed from the original normalized x
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"]
    out_gate_fp16 = torch.nn.functional.linear(x_flat_fp16, out_gate_w.to(torch.float16))
    out_gate = torch.sigmoid(out_gate_fp16).view(bs, N, N, hidden_dim).to(torch.float32)

    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out.view(-1, hidden_dim), to_out_w).view(bs, N, N, dim)

    return out

"""
Outgoing TriMul (AlphaFold‑3) – mixed‑precision Triton‑accelerated implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input (FP32).
2. Fuse the two linear layers per side (proj + gate) by concatenating their
   weight matrices – this reduces the number of GEMMs from four to two.
3. Apply the sigmoid to the gate tensors.
4. If a mask is present, fuse ``proj * gate * mask`` with a tiny Triton
   kernel that broadcasts the 2‑D mask over the hidden dimension without
   materialising a full 4‑D mask.
5. Perform the cubic reduction
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   as a strided‑batched GEMM (`torch.bmm`) on FP16 tensors.
6. Layer‑norm over the hidden dimension (FP32), multiply by the output gate,
   and finally project back to the original channel dimension.
All heavy arithmetic runs in FP16; the two layer‑norms, the output gate and
the final projection stay in FP32 for numerical stability.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: mask‑aware element‑wise multiplication
# ----------------------------------------------------------------------
@triton.jit
def masked_fuse_mul_kernel(
    proj_ptr,         # fp16,   projection tensor
    gate_ptr,         # fp16,   sigmoid‑ed gate tensor
    mask_ptr,         # fp16,   flattened mask (size = B·N·N)
    out_ptr,          # fp16,   result tensor
    total_elems,      # i64,    number of elements = B·N·N·H
    hidden_dim,       # i64,    H
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)

    mask = offs < total_elems

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # mask is broadcast over the hidden dimension: each H consecutive
    # elements share the same mask entry.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    tl.store(out_ptr + offs, proj * gate * m, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_w, bias=norm_b, eps=eps
    )  # (bs, N, N, dim), FP32

    # ------------------------------------------------------------------
    # 2. Flatten & cast to FP16 for the linear layers
    # ------------------------------------------------------------------
    x_flat_fp16 = x.reshape(-1, dim).to(torch.float16)   # (M, dim)  M = bs*N*N

    # ------------------------------------------------------------------
    # 3. Concatenated weight matrices (proj + gate) → one GEMM per side
    # ------------------------------------------------------------------
    # left side
    left_proj_w = weights["left_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    left_concat_w = torch.cat([left_proj_w, left_gate_w], dim=0).to(torch.float16)
    left_concat = torch.nn.functional.linear(x_flat_fp16, left_concat_w)
    left_proj, left_gate = left_concat.split([hidden_dim, hidden_dim], dim=1)

    # right side
    right_proj_w = weights["right_proj.weight"]
    right_gate_w = weights["right_gate.weight"]
    right_concat_w = torch.cat([right_proj_w, right_gate_w], dim=0).to(torch.float16)
    right_concat = torch.nn.functional.linear(x_flat_fp16, right_concat_w)
    right_proj, right_gate = right_concat.split([hidden_dim, hidden_dim], dim=1)

    # ------------------------------------------------------------------
    # 4. Apply sigmoid to the gate tensors (still FP16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)

    # ------------------------------------------------------------------
    # 5. Fuse mask (if any) with projection and gate using Triton
    # ------------------------------------------------------------------
    total = left_proj.numel()                     # = bs * N * N * hidden_dim
    if not nomask:
        # flatten mask to (bs·N·N,) and cast to FP16
        mask_flat_fp16 = mask.to(torch.float16).reshape(-1)  # (M,)
        BLOCK = 131072                               # tune‑able
        grid = ((total + BLOCK - 1) // BLOCK,)

        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        masked_fuse_mul_kernel[grid](
            left_proj,
            left_gate,
            mask_flat_fp16,
            left,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        masked_fuse_mul_kernel[grid](
            right_proj,
            right_gate,
            mask_flat_fp16,
            right,
            total,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # No mask → simple element‑wise multiplication
        left = left_proj * left_gate
        right = right_proj * right_gate

    # restore 4‑D shape (bs, N, N, hidden_dim)
    left = left.view(bs, N, N, hidden_dim)
    right = right.view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Cubic reduction via a single strided‑batched GEMM (FP16)
    # ------------------------------------------------------------------
    # (b, i, k, d) → (b, d, i, k)   and   (b, j, k, d) → (b, d, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (bs, d, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (bs, d, N, N)

    BH = bs * hidden_dim                                 # batch for bmm
    left_mat = left_perm.view(BH, N, N)                  # (BH, N, K) with K=N
    right_mat = right_perm.view(BH, N, N)                # (BH, K, N)

    out_fp16 = torch.bmm(left_mat, right_mat)            # (BH, N, N)

    # Back to (bs, N, N, hidden_dim) and convert to FP32
    out = (
        out_fp16.view(bs, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .to(torch.float32)
    )  # (bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Output gate (sigmoid) – computed from the normalized input
    # ------------------------------------------------------------------
    out_gate_w = weights["out_gate.weight"]
    out_gate_fp16 = torch.nn.functional.linear(
        x_flat_fp16, out_gate_w.to(torch.float16)
    )
    out_gate = torch.sigmoid(out_gate_fp16).view(bs, N, N, hidden_dim).to(torch.float32)

    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final projection back to the original channel dimension
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(
        out.view(-1, hidden_dim), to_out_w
    ).view(bs, N, N, dim)

    return out
