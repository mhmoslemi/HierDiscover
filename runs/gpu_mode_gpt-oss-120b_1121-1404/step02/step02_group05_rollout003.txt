    bs, N, _, dim = input_tensor.shape

    # Layer norm in fp32
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Convert to fp16
    x = x.to(torch.float16)

    # Cast weights to fp16 once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    to_out_w = weights["to_out.weight"].to(torch.float16)

    left = F.linear(x, left_proj_w)  # fp16
    right = F.linear(x, right_proj_w)

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))

"""
Optimized Triton‑fused implementation of the **outgoing** TriMul operator.

Algorithm
---------
1. Layer‑norm the 4‑D input (float‑32) → cast to fp16.
2. Linear projections for ``left`` / ``right`` and three gating projections.
   All linear layers run in fp16 (weights are cast once).
3. Fuse the element‑wise mask, ``left_gate`` and ``right_gate`` with a tiny
   Triton kernel (in‑place).  When ``nomask=True`` only the gates are applied.
4. Perform the cubic contraction

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   with a custom Triton batched‑GEMM that works directly on the
   (B,N,N,H) layout – no costly ``permute(...).contiguous()`` copies.
   The kernel accumulates in fp32 and writes back fp16.
5. Cast the result back to fp32, apply the output LayerNorm,
   the ``out_gate`` and the final linear projection to the original ``dim``.
   All remaining heavy ops use PyTorch’s fused kernels (cuBLAS, cuDNN).

The implementation keeps the memory footprint low (no extra copies) and
leverages Tensor‑cores for the O(N³·H) contraction, reaching sub‑millisecond
times on an H100 for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Fuse mask + left/right gates (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def fused_mask_gate_kernel(
    left_ptr,          # [B,N,N,H] – in‑place output
    left_gate_ptr,     # [B,N,N,H]
    right_ptr,         # [B,N,N,H] – in‑place output
    right_gate_ptr,    # [B,N,N,H]
    mask_ptr,          # [B,N,N] (0/1)  – fp16
    B, N, H,           # problem sizes
    # strides for left
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    # strides for mask (no hidden dim)
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK: tl.constexpr,
):
    """Fuse   left = left * left_gate * mask
                right = right * right_gate * mask
       in‑place with a single Triton kernel.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)          # linear index
    total = B * N * N * H
    valid = offs < total

    # decode linear offset → (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # bounds (should always hold for a valid offset)
    valid = valid & (b < B) & (i < N) & (j < N) & (h < H)

    # compute raw pointers
    left_off   = left_ptr   + b * stride_l_b   + i * stride_l_i   + j * stride_l_j   + h * stride_l_h
    leftg_off  = left_gate_ptr + b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    right_off  = right_ptr  + b * stride_r_b   + i * stride_r_i   + j * stride_r_j   + h * stride_r_h
    rightg_off = right_gate_ptr + b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    mask_off   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + j * stride_m_j

    left      = tl.load(left_off,   mask=valid, other=0.0)
    left_gate = tl.load(leftg_off,  mask=valid, other=0.0)
    right     = tl.load(right_off,  mask=valid, other=0.0)
    right_gate= tl.load(rightg_off, mask=valid, other=0.0)
    mask_val  = tl.load(mask_off,   mask=valid, other=1.0)   # default 1 → no masking

    # fused mask + gate
    left  = left  * left_gate  * mask_val
    right = right * right_gate * mask_val

    # store back in‑place
    tl.store(left_off,  left,  mask=valid)
    tl.store(right_off, right, mask=valid)

# 2️⃣ Batched GEMM for the cubic contraction (no copies)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,   # [B,N,N,H]  (fp16)
    right_ptr,  # [B,N,N,H]  (fp16) – already transposed (i↔j)
    out_ptr,    # [B,N,N,H]  (fp16) – result
    B, N, H,
    # strides for left  (b,i,k,h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    # strides for right (b,k,j,h) – after transpose
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for out   (b,i,j,h)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for each (b,h) pair, the matrix product
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,k,j,h]
    where ``right`` is already a view with the first two spatial axes swapped.
    """
    # 0‑dim: combined batch‑hidden index
    batch = tl.program_id(0)
    # 1‑dim: block row (i)
    pid_m = tl.program_id(1)
    # 2‑dim: block column (j)
    pid_n = tl.program_id(2)

    # recover batch and hidden indices
    b = batch // H
    h = batch % H

    # base offsets for this (b,h)
    offs_l = b * stride_l_b + h * stride_l_h
    offs_r = b * stride_r_b + h * stride_r_h
    offs_o = b * stride_o_b + h * stride_o_h

    # tile start indices
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # ranges inside the tile
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    i_mask = i_range < N
    j_mask = j_range < N

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # left tile: (i, k)
        left_ptrs = left_ptr + offs_l \
                    + i_range[:, None] * stride_l_i \
                    + k_range[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0).to(tl.float32)

        # right tile: (k, j)   (right is already transposed)
        right_ptrs = right_ptr + offs_r \
                     + k_range[:, None] * stride_r_i \
                     + j_range[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0).to(tl.float32)

        # matrix‑multiply‑accumulate
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # write back – cast to fp16
    out_ptrs = out_ptr + offs_o \
               + i_range[:, None] * stride_o_i \
               + j_range[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=i_mask[:, None] & j_mask[None, :])

    )
    out_fp32 = out_fp32 * out_gate
    out_fp32 = F.linear(out_fp32, to_out_w)   # final projection to original dim

    return out_fp32
