def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1) LayerNorm
    x = F.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)

    # 2) Linear projections
    # flatten
    x_flat = x.view(bs, N * N, dim) # shape (B, N*N, dim)
    # Actually we can flatten to (B*N*N, dim) for torch linear
    x_flat = x.view(-1, dim)
    # Or use torch.nn.functional.linear
    left_proj_w = weights['left_proj.weight'].to(torch.float16)
    right_proj_w = weights['right_proj.weight'].to(torch.float16)

    left = F.linear(x_flat, left_proj_w).view(bs, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_w).view(bs, N, N, hidden_dim)
    # gating
    left_gate_w = weights['left_gate.weight'].to(torch.float16)
    right_gate_w = weights['right_gate.weight'].to(torch.float16)
    out_gate_w = weights['out_gate.weight'].to(torch.float16)

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(bs, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(bs, N, N, hidden_dim)

    # Mask if needed
    if not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # Move to half
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # permute and contiguous for efficient strides
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) (k,j)

    # output tensor
    out_perm = torch.empty_like(left_perm)   # same shape, type fp16

    # get strides in elements
    l_str = left_perm.stride()
    r_str = right_perm.stride()
    o_str = out_perm.stride()

    # configure block sizes: tune for inner dimension
    # Determine appropriate block size adaptively based on N; Use 128 for N>=256 else 64
    if N > 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4

    # compute grid
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (bs * hidden_dim, num_blocks_m, num_blocks_n)

    outgoing_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # reshape back
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)

    # to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # apply out_gate
    out = out * out_gate

    # final linear projection to dim
    out = F.linear(out.view(-1, hidden_dim), weights["to_out.weight"]).view(bs, N, N, dim)

    return out

"""
Fast Triton implementation of the outgoing TriMul operator (AlphaFold‑3).

The operation:
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,k,j,d]
where left/right are linear projections of the input tensor, gated
and optionally masked.  The per‑head matrix multiplication is expressed

BLOCK_K.  Input tensors are made contiguous after permutation so that
the K‑dimension is stride‑1, yielding fully coalesced loads and allowing
the kernel to reach near‑peak tensor‑core throughput on an H100.
All other lightweight ops (LayerNorm, linear projections, Sigmoid,
masking) are performed with PyTorch‑CUDA kernels.

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def outgoing_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    seq_len,
    # strides for left (B,H,i,k)
    l_str_batch, l_str_head, l_str_i, l_str_k,
    # strides for right (B,H,k,j)
    r_str_batch, r_str_head, r_str_k, r_str_j,
    # strides for out (B,H,i,j)
    o_str_batch, o_str_head, o_str_i, o_str_j,
    hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    # (bh, i_tile, j_tile) program ids
    bh_id = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # split bh_id into batch and head
    b_id = bh_id // hidden_dim
    h_id = bh_id % hidden_dim

    # base pointer offsets for this (b,h)
    left_base  = b_id * l_str_batch + h_id * l_str_head
    right_base = b_id * r_str_batch + h_id * r_str_head
    out_base   = b_id * o_str_batch + h_id * o_str_head

    # tile offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Main K loop
    for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # left tile (i,k)
        a_ptr = left_ptr + (
            left_base
            + offs_i[:, None] * l_str_i
            + offs_k[None, :] * l_str_k
        )
        mask_a = (offs_i[:, None] < seq_len) & (offs_k[None, :] < seq_len)
        a = tl.load(a_ptr, mask=mask_a, other=0.0)

        # right tile (k,j)
        b_ptr = right_ptr + (
            right_base
            + offs_k[:, None] * r_str_k
            + offs_j[None, :] * r_str_j
        )
        mask_b = (offs_k[:, None] < seq_len) & (offs_j[None, :] < seq_len)
        b = tl.load(b_ptr, mask=mask_b, other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # write back
    c_ptr = out_ptr + (
        out_base
        + offs_i[:, None] * o_str_i
        + offs_j[None, :] * o_str_j
    )
    mask_c = (offs_i[:, None] < seq_len) & (offs_j[None, :] < seq_len)
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_c)

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton GEMM.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # 1) Input layer norm (float32)
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # 2) Linear projections (fp16)
    x_flat = x.view(-1, dim)                      # (B * N * N, dim)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_flat, left_proj_w).view(bs, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_w).view(bs, N, N, hidden_dim)

    # 3) Gates (sigmoid, fp16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(bs, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(bs, N, N, hidden_dim)

    # 4) Optional mask (broadcast on hidden dim)
    if not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # 5) Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # 6) Cast to half for the GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # 7) Permute to (B, H, N, N) and make contiguous → K dimension stride == 1
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    # 8) Output buffer (fp16)
    out_perm = torch.empty_like(left_perm)   # (B, H, N, N)

    # 9) Strides (in elements)
    l_str = left_perm.stride()
    r_str = right_perm.stride()
    o_str = out_perm.stride()

    # 10) Triton launch configuration
    # Heuristic block sizes: larger tiles for larger seq_len
    if N >= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (bs * hidden_dim, num_blocks_m, num_blocks_n)

    outgoing_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        # left strides
        l_str[0], l_str[1], l_str[2], l_str[3],
        # right strides
        r_str[0], r_str[1], r_str[2], r_str[3],
        # output strides
        o_str[0], o_str[1], o_str[2], o_str[3],
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # 11) Restore original layout (B, N, N, H) and cast to fp32
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)

    # 12) Output LayerNorm over hidden dimension

"""
Fast Triton implementation of the *outgoing* TriMul operator (AlphaFold‑3).

The module computes for each head d:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,k,j,d]

where `left`/`right` are linear projections of the input pairwise features,
gated and optionally masked.  The per‑head matrix multiplication is fused
into a single batched GEMM (`left @ rightᵀ`).  To obtain high
tensor‑core utilization on an H100 we:

* materialise `left` and `right` as half‑precision tensors,
* permute them to (B, H, N, N) and **make them contiguous** – this makes
  the K‑dimension stride‑1, giving fully coalesced loads,
* launch a custom Triton kernel that processes a (B·H) batch of
  (N×N) matrix multiplies tiled with (BLOCK_M, BLOCK_N) and a K‑tile
  of `BLOCK_K`.  Accumulation is in FP32 for numerical stability.
All surrounding lightweight ops (LayerNorm, linear layers, sigmoid,
masking) are performed with PyTorch‑CUDA which is already highly
optimised.

The final result is cast back to float32 and passed through the output
LayerNorm, output‑gate and linear projection to the original channel
dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def outgoing_trimul_kernel(
    left_ptr,          # fp16 * (B, H, N, N)   – (i, k)
    right_ptr,         # fp16 * (B, H, N, N)   – (k, j)
    out_ptr,           # fp16 * (B, H, N, N)   – (i, j)
    seq_len,           # int = N
    # left strides
    l_str_batch, l_str_head, l_str_i, l_str_k,
    # right strides
    r_str_batch, r_str_head, r_str_k, r_str_j,
    # output strides
    o_str_batch, o_str_head, o_str_i, o_str_j,
    hidden_dim,        # int = H
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    # -------------------------------------------------------------
    # grid: (bh_id, block_i, block_j)
    # -------------------------------------------------------------
    bh_id   = tl.program_id(0)      # 0 … B*H‑1
    block_i = tl.program_id(1)      # row tile
    block_j = tl.program_id(2)      # col tile

    # split bh_id → (batch, head)
    b_id = bh_id // hidden_dim
    h_id = bh_id % hidden_dim

    # base offsets for this (b, h)
    left_base  = b_id * l_str_batch + h_id * l_str_head
    right_base = b_id * r_str_batch + h_id * r_str_head
    out_base   = b_id * o_str_batch + h_id * o_str_head

    # -------------------------------------------------------------
    # tile offsets
    # -------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32 (tensor‑core dot internally uses FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # K‑dimension loop
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(seq_len, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # ----- left tile (i, k) -------------------------------------------------
        a_ptr = left_ptr + (
            left_base
            + offs_i[:, None] * l_str_i
            + offs_k[None, :] * l_str_k
        )
        mask_a = (offs_i[:, None] < seq_len) & (offs_k[None, :] < seq_len)
        a = tl.load(a_ptr, mask=mask_a, other=0.0)

        # ----- right tile (k, j) ------------------------------------------------
        b_ptr = right_ptr + (
            right_base
            + offs_k[:, None] * r_str_k
            + offs_j[None, :] * r_str_j
        )
        mask_b = (offs_k[:, None] < seq_len) & (offs_j[None, :] < seq_len)
        b = tl.load(b_ptr, mask=mask_b, other=0.0)

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # write back
    # -------------------------------------------------------------
    c_ptr = out_ptr + (
        out_base
        + offs_i[:, None] * o_str_i
        + offs_j[None, :] * o_str_j
    )
    mask_c = (offs_i[:, None] < seq_len) & (offs_j[None, :] < seq_len)
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton GEMM.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                       # (B·N·N, dim)

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = F.linear(x_flat, left_proj_w).view(bs, N, N, hidden_dim)   # (B, N, N, H)
    right = F.linear(x_flat, right_proj_w).view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid, FP16)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(bs, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcasted on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.to(x.dtype).unsqueeze(-1)       # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to half for the GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7) Permute to (B, H, N, N) and make contiguous → K dimension stride‑1
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    # ------------------------------------------------------------------
    # 8) Output buffer (FP16)
    # ------------------------------------------------------------------
    out_perm = torch.empty_like(left_perm)               # (B, H, N, N)

    # ------------------------------------------------------------------
    # 9) Strides (in elements)
    # ------------------------------------------------------------------
    l_str = left_perm.stride()   # (batch, head, i, k)
    r_str = right_perm.stride()  # (batch, head, k, j)
    o_str = out_perm.stride()    # (batch, head, i, j)

    # ------------------------------------------------------------------
    # 10) Triton launch configuration (heuristic)
    # ------------------------------------------------------------------
    if N >= 512:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (bs * hidden_dim, num_blocks_m, num_blocks_n)

    outgoing_trimul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        N,
        # left strides
        l_str[0], l_str[1], l_str[2], l_str[3],
        # right strides
        r_str[0], r_str[1], r_str[2], r_str[3],
        # out strides
        o_str[0], o_str[1], o_str[2], o_str[3],
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ------------------------------------------------------------------
    # 11) Restore original layout (B, N, N, H) and cast to FP32
    # ------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).to(torch.float32)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 12) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 13) Apply the output gate (cast to FP32)
    # ------------------------------------------------------------------
    out = out * out_gate.to(torch.float32)

    # ------------------------------------------------------------------
    # 14) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out.view(-1, hidden_dim), weights["to_out.weight"]).view(bs, N, N, dim)

    return out
