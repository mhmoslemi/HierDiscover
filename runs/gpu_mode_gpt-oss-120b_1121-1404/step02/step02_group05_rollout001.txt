@triton.jit
def mask_mul_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK: tl.constexpr
):
    # This kernel multiplies left and right by mask.
    # mask shape: [B,N,N] ; left,right shape: [B,N,N,H]
    start = tl.program_id(0) * BLOCK
    offs = start + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = offs < total

    # decode indices
    # b = offs // (N * N * H)
    # rem = offs % (N * N * H)
    # i = rem // (N * H)
    # rem2 = rem % (N * H)
    # j = rem2 // H
    # h = rem2 % H

    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute pointers
    left_off = left_ptr + b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    right_off = right_ptr + b * stride_r_b + i * stride_r_i + j * stride_r_j + h * stride_r_h
    mask_off = mask_ptr + b * stride_m_b + i * stride_m_i + j * stride_m_j

    # load
    left_val = tl.load(left_off, mask=mask, other=0.0)
    right_val = tl.load(right_off, mask=mask, other=0.0)
    mask_val = tl.load(mask_off, mask=mask, other=1.0)  # default 1 (no mask)

    # multiply
    left_val = left_val * mask_val
    right_val = right_val * mask_val

    tl.store(left_off, left_val, mask=mask)
    tl.store(right_off, right_val, mask=mask)

"""
Optimized “outgoing” TriMul forward pass.

Algorithm
---------
1️⃣ Layer‑norm on the last dimension.
2️⃣ Fuse the two linear projections (proj + gate) for the left and right
   branches by concatenating the weights → a single matmul per branch.
   The gate part is obtained with a sigmoid and multiplied in‑place.
3️⃣ Compute the output‑gate (a separate linear → sigmoid).
4️⃣ If a mask is present, multiply both left/right tensors by the mask
   with a tiny Triton kernel (mask * left/right, in‑place).
5️⃣ Core cubic contraction:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
   This is a batched half‑precision GEMM for each hidden channel.
   We cast to fp16, permute to (B·H,N,N), call `torch.bmm` and
   reshape back to (B,N,N,H).
6️⃣ Layer‑norm on the hidden dimension, multiply by the output‑gate,
   and a final linear projection back to the original `dim`.

Only the mask‑multiplication step is written in Triton; all heavy
linear/algebraic work uses PyTorch/cuBLAS which already exploits the
H100 Tensor‑cores.  The implementation keeps memory traffic low by
reading the input only once for each projection‑gate pair and by
working in fp16 for the O(N³) contraction.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_mul_kernel(
    left_ptr,      # float32 *  (B,N,N,H) – in‑place
    right_ptr,     # float32 *  (B,N,N,H) – in‑place
    mask_ptr,      # float32 *  (B,N,N)
    B, N, H,       # problem sizes
    # strides for left  (b,i,j,h)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for right (b,i,j,h)
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for mask  (b,i,j)
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK: tl.constexpr,
):
    """
    Multiply `left` and `right` element‑wise by a broadcast mask.
    Operates in‑place.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask_valid = offs < total

    # decode linear index → (b,i,j,h)
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute raw pointers
    left_off  = left_ptr  + b * stride_l_b  + i * stride_l_i  + j * stride_l_j  + h * stride_l_h
    right_off = right_ptr + b * stride_r_b + i * stride_r_i + j * stride_r_j + h * stride_r_h
    mask_off  = mask_ptr  + b * stride_m_b + i * stride_m_i + j * stride_m_j

    # load (masked loads return 0/1 for OOB)
    left_val  = tl.load(left_off,  mask=mask_valid, other=0.0)
    right_val = tl.load(right_off, mask=mask_valid, other=0.0)
    mask_val  = tl.load(mask_off, mask=mask_valid, other=1.0)   # default 1 → no masking

    # fused mask
    left_val  = left_val * mask_val
    right_val = right_val * mask_val

    # store back
    tl.store(left_off,  left_val,  mask=mask_valid)
    tl.store(right_off, right_val, mask=mask_valid)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the TriMul output.
    """
    # ------------------------------------------------------------------
    # 0. Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (last dim = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Fused projection + gate for the left branch
    # ------------------------------------------------------------------
    #   [hidden_dim*2, dim] weight = [proj; gate]
    left_fused_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    )
    left_fused = F.linear(x, left_fused_w)                     # [B,N,N,2*H]
    left_proj = left_fused[..., :hidden_dim]                  # [B,N,N,H]
    left_gate = torch.sigmoid(left_fused[..., hidden_dim:])   # [B,N,N,H]
    left = left_proj * left_gate                               # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Fused projection + gate for the right branch
    # ------------------------------------------------------------------
    right_fused_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    )
    right_fused = F.linear(x, right_fused_w)                    # [B,N,N,2*H]
    right_proj = right_fused[..., :hidden_dim]
    right_gate = torch.sigmoid(right_fused[..., hidden_dim:])
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Output gate (separate linear + sigmoid)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5. Apply mask (if any) – done with a tiny Triton kernel
    # ------------------------------------------------------------------
    if not nomask:
        # cast mask to float32 (same dtype as left/right)
        mask_fp = mask_tensor.to(left.dtype)
        total = left.numel()
        BLOCK = 256
        grid = (triton.cdiv(total, BLOCK),)

        # strides
        sl = left.stride()
        sr = right.stride()
        sm = mask_fp.stride()

        mask_mul_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask_fp,
            B=B,
            N=N,
            H=hidden_dim,
            stride_l_b=sl[0],
            stride_l_i=sl[1],
            stride_l_j=sl[2],
            stride_l_h=sl[3],
            stride_r_b=sr[0],
            stride_r_i=sr[1],
            stride_r_j=sr[2],
            stride_r_h=sr[3],
            stride_m_b=sm[0],
            stride_m_i=sm[1],
            stride_m_j=sm[2],
            BLOCK=BLOCK,
        )
        del mask_fp  # free asap

    # ------------------------------------------------------------------
    # 6. Core cubic contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched fp16 GEMM (tensor‑cores)
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)    # [B,N,N,H]  fp16
    right_h = right.to(torch.float16)

    # free the fp32 tensors early
    del left, right

    # reshape for batched matrix multiplication
    # left_h: (B,N,N,H) → (B*H, N, N)
    left_batch = left_h.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right_h: (B,N,N,H) → (B*H, N, N)  (note the transpose on the last two axes)
    right_batch = right_h.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # batched GEMM (fp16)
    out_batch = torch.bmm(left_batch, right_batch.transpose(1, 2))   # (B*H,N,N)

    # reshape back to (B,N,N,H)
    out = out_batch.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
