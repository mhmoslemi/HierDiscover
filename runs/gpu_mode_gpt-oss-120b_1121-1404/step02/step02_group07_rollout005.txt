"""
Outgoing TriMul (AlphaFold3) – forward only.

The heavy part of the module is the three‑dimensional tensor contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

which can be seen as a batch of independent matrix multiplications
(A @ Bᵀ) where A and B have shape (N, N) and the batch dimension is
B·hidden_dim.  The implementation below

* fuses the five linear layers (projections + gates + out‑gate) into a
  single matrix‑multiply to minimise reading the large input tensor.
* applies the optional causal mask only to the projected tensors.
* runs the cubic A@Bᵀ using a custom Triton kernel that works on fp16
  data with an fp32 accumulator.
* keeps the rest of the module (LayerNorm, sigmoid gates, final
  projection) in PyTorch – they are negligible compared to the N³ work.

The kernel tiles the GEMM with configurable block sizes
(BLOCK_M, BLOCK_N, BLOCK_K) and uses `tl.dot` (Tensor‑Core‑enabled on
H100) for maximal throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, M, N, K,                     # dimensions
    stride_ah, stride_am, stride_ak,    # A strides (batch, row i, col k)
    stride_bh, stride_bk, stride_bn,    # B strides (batch, k, col j) – B is accessed transposed
    stride_ch, stride_cm, stride_cn,    # C strides (batch, row i, col j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # (batch*hidden) index and tile indices
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ------------------------------------------------------------------
    #  Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    #  Base pointers for this tile
    a_ptrs = a_ptr + pid_batch * stride_ah \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * stride_bh \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    #  Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #  Number of K‑blocks (ceil division)
    num_k_blocks = tl.cdiv(K, BLOCK_K)

    for cur_k in range(num_k_blocks):
        k_offset = cur_k * BLOCK_K

        #  Masks for loads that go out of bounds
        a_mask = (offs_m < M)[:, None] & ((k_offset + offs_k) < K)[None, :]
        b_mask = ((k_offset + offs_k) < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        acc += tl.dot(a, b)   # fp32 accumulation

        #  Advance to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    #  Store the result (with bounds check)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid_batch * stride_ch \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for
    the cubic matrix‑multiplication part.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (torch.Tensors)
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    #  Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    #  Load model weights (all are float32, bias‑free Linear)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    #  1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    #  2) Fuse the five Linear layers into a single matmul:
    #
    #      combined = x @ [ left_proj ; left_gate ; right_proj ;
    #                       right_gate ; out_gate ]ᵀ
    #
    #  The concatenated weight has shape (5*hidden_dim, dim).
    # ------------------------------------------------------------------
    fused_weight = torch.cat(
        [
            left_proj_weight,
            left_gate_weight,
            right_proj_weight,
            right_gate_weight,
            out_gate_weight,
        ],
        dim=0,
    )  # (5*hidden_dim, dim)

    combined = F.linear(x, fused_weight)  # [B, N, N, 5*hidden_dim]

    # ------------------------------------------------------------------
    #  3) Slice the combined tensor back into its components
    # ------------------------------------------------------------------
    #   0 … hidden_dim                     → left_proj
    #   hidden_dim … 2*hidden_dim         → left_gate
    #   2*hidden_dim … 3*hidden_dim       → right_proj
    #   3*hidden_dim … 4*hidden_dim       → right_gate
    #   4*hidden_dim … 5*hidden_dim       → out_gate
    # ------------------------------------------------------------------
    h0 = hidden_dim
    h1 = 2 * hidden_dim
    h2 = 3 * hidden_dim
    h3 = 4 * hidden_dim
    h4 = 5 * hidden_dim

    left      = combined[..., :h0]          # [B,N,N,hidden_dim]
    left_gate = torch.sigmoid(combined[..., h0:h1])
    right     = combined[..., h1:h2]
    right_gate = torch.sigmoid(combined[..., h2:h3])
    out_gate  = torch.sigmoid(combined[..., h3:h4])

    # ------------------------------------------------------------------
    #  4) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    #  5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #  6) Cast to FP16 for the heavy GEMM
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    #  7) Prepare tensors for the batched GEMM:
    #      left_f16  -> (B*hidden_dim, N, N)   (A)
    #      right_f16 -> (B*hidden_dim, N, N)   (B, accessed as Bᵀ)
    # ------------------------------------------------------------------
    left_mat  = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    #  8) Choose tile sizes – tuned for the H100
    # ------------------------------------------------------------------
    if N <= 256:
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32
    else:
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 64

    # ------------------------------------------------------------------
    #  9) Compute strides (in element counts, not bytes)
    # ------------------------------------------------------------------
    stride_ah = left_mat.stride(0)   # batch stride
    stride_am = left_mat.stride(1)   # row (i) stride
    stride_ak = left_mat.stride(2)   # col (k) stride

    # For B we intentionally swap K/N strides to avoid a transpose copy
    stride_bh = right_mat.stride(0)   # batch stride
    stride_bk = right_mat.stride(2)   # stride for the K‑dimension (original col)
    stride_bn = right_mat.stride(1)   # stride for the N‑dimension (original row)

    stride_ch = out_mat.stride(0)     # batch stride
    stride_cm = out_mat.stride(1)     # row stride
    stride_cn = out_mat.stride(2)     # col stride

    # ------------------------------------------------------------------
    # 10) Launch the Triton kernel (batched GEMM)
    # ------------------------------------------------------------------
    grid = (
        B * hidden_dim,                     # batch * hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles along N (j)
    )
    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * hidden_dim,
        N,
        N,
        N,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # (kernel launches are asynchronous; synchronize only if needed)
    # torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 11) Reshape back to the original 4‑D layout
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 12) Output LayerNorm, gating, and final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)          # [B, N, N, dim] (float32)

    return out
