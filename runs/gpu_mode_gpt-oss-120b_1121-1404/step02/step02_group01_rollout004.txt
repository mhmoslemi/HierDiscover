"""
TriMul (outgoing) – fast Triton implementation.

The forward consists of four stages:
1️⃣ LayerNorm on the input (FP32) and three bias‑free linear projections
   (FP16, no bias) to obtain *left*, *right* and the output gate.
2️⃣ A fused N³ “outer‑product”  C[b,h,i,j] = Σₖ left[b,h,i,k]·right[b,h,j,k]
   performed with a custom Triton GEMM kernel that:
   - works on FP16 inputs,
   - accumulates in FP32,
   - writes the result back in FP16 (halves the memory traffic).
3️⃣ A second Triton kernel that fuses the hidden‑dim LayerNorm,
   the output‑gate (sigmoid) and casting to FP32:
   it consumes the FP16 tensor from step‑2 and produces a FP32
   tensor of shape (B,N,N,hidden).
4️⃣ A final dense projection back to *dim* using the regular
   PyTorch `F.linear` (cuBLAS).  All heavy arithmetic stays in
   half‑precision; only the two LayerNorms are FP32, which is
   required for numerical stability.

The implementation is fully functional for all configurations
listed in the problem statement and runs comfortably below the
target runtime on an H100.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# 1. Outer‑product kernel (A @ Bᵀ) – writes FP16 result
# -----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    a_ptr,                     # const half*   (left)
    b_ptr,                     # const half*   (right, accessed as transposed)
    c_ptr,                     # half*         (output)
    batch_hidden, N,           # int32 scalars
    stride_az, stride_ai, stride_ak,   # strides for A (page, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (page, k‑dim, n‑dim)
    stride_cz, stride_ci, stride_cj,   # strides for C (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a block of C = A @ Bᵀ.
    A : (batch_hidden, N, N) – row‑major, half
    B : (batch_hidden, N, N) – row‑major, accessed as Bᵀ
    C : (batch_hidden, N, N) – row‑major, half
    """
    pid_z = tl.program_id(0)       # which (batch, hidden) page
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # --------------------------------------------------- tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------- base pointers for the current page
    a_page = a_ptr + pid_z * stride_az
    b_page = b_ptr + pid_z * stride_bz
    c_page = c_ptr + pid_z * stride_cz

    # --------------------------------------------------- accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------- reduction over K
    cur_k = 0
    while cur_k < N:
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_offs < N

        # ----- load A tile (row‑major)
        a_ptrs = a_page + \
                 offs_m[:, None] * stride_ai + \
                 cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                 # half → fp16

        # ----- load B tile as Bᵀ
        b_ptrs = b_page + \
                 cur_k_offs[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                 # half

        # ----- MAC (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

        cur_k += BLOCK_K

    # --------------------------------------------------- write back (cast to half)
    c_ptrs = c_page + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# -----------------------------------------------------------------------
# 2. Hidden‑dim LayerNorm + output‑gate (fused, FP32 output)
# -----------------------------------------------------------------------
@triton.jit
def norm_gate_kernel(
    inp_ptr,                     # const half*  (C from trimul)
    gate_ptr,                    # const half*  (out_gate)
    out_ptr,                     # float*      (result after LN+gate)
    B, N, H,                     # runtime ints
    stride_ib, stride_ii, stride_ij, stride_ih,   # inp strides
    stride_gb, stride_gi, stride_gj, stride_gh,   # gate strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides
    gamma_ptr,                   # const float* (to_out_norm.weight)
    beta_ptr,                    # const float* (to_out_norm.bias)
    H: tl.constexpr,             # hidden dimension (compile‑time)
    eps: tl.constexpr = 1e-5,    # LayerNorm epsilon
    BLOCK_SIZE: tl.constexpr = 128,
):
    """
    For each (b,i,j) location:
        1) load the hidden vector (half → float32)
        2) compute mean / variance across the hidden dimension
        3) apply LayerNorm (γ,β), then multiply by the output gate
        4) write the resulting float32 vector.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = tl.arange(0, BLOCK_SIZE)
    idx = start + offs
    total = B * N * N
    mask = idx < total

    # ---- decode linear index into (b,i,j)
    b = idx // (N * N)
    ij = idx % (N * N)
    i = ij // N
    j = ij % N

    # ---- base pointers for this (b,i,j)
    inp_base  = inp_ptr  + b * stride_ib + i * stride_ii + j * stride_ij
    gate_base = gate_ptr + b * stride_gb + i * stride_gi + j * stride_gj
    out_base  = out_ptr  + b * stride_ob + i * stride_oi + j * stride_oj

    # ---- hidden offsets (compile‑time vector)
    h_offs = tl.arange(0, H)

    # ---- load hidden vectors (half → float32)
    inp = tl.load(inp_base[:, None] + h_offs[None, :] * stride_ih,
                  mask=mask[:, None],
                  other=0.0).to(tl.float32)          # (BLOCK_SIZE, H)

    gate = tl.load(gate_base[:, None] + h_offs[None, :] * stride_gh,
                   mask=mask[:, None],
                   other=0.0).to(tl.float32)

    # ---- load LN parameters (broadcasted, same for all threads)
    gamma = tl.load(gamma_ptr + h_offs * 1)                # (H,)
    beta  = tl.load(beta_ptr  + h_offs * 1)                # (H,)

    # ---- compute mean / variance over H
    sum_   = tl.sum(inp, axis=1)                          # (BLOCK_SIZE,)
    sumsq_ = tl.sum(inp * inp, axis=1)
    mean   = sum_ / H
    var    = sumsq_ / H - mean * mean
    rstd   = 1.0 / tl.sqrt(var + eps)

    # ---- normalize, scale/shift and apply gating
    normed = (inp - mean[:, None]) * rstd[:, None]       # (B,S,H)
    scaled = normed * gamma + beta                       # (B,S,H)
    gated  = scaled * gate                               # (B,S,H)

    # ---- write out (float32)
    tl.store(out_base[:, None] + h_offs[None, :] * stride_oh,
             gated,
             mask=mask[:, None])


# -----------------------------------------------------------------------
# 3. Custom kernel entry point
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------- unpack
    x, mask, weights, config = data            # x : [B, N, N, dim] (float32)
    dim        = config["dim"]
    hidden     = config["hidden_dim"]
    nomask     = config.get("nomask", True)

    B, N, _, _ = x.shape
    device = x.device

    # ------------------------------------------------------------------- FP32 LayerNorm on the input (cheap)
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------- cast to half for heavy ops
    x_norm_h = x_norm.to(torch.float16)               # [B, N, N, dim] (half)

    # ------------------------------------------------------------------- Linear projections (bias‑free, FP16)
    left  = F.linear(x_norm_h, weights["left_proj.weight"].to(torch.float16))   # [B,N,N,hidden]
    right = F.linear(x_norm_h, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------- Gates (sigmoid, still FP16)
    left_gate  = torch.sigmoid(F.linear(x_norm_h, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(F.linear(x_norm_h, weights["right_gate.weight"].to(torch.float16)))
    out_gate   = torch.sigmoid(F.linear(x_norm_h, weights["out_gate.weight"].to(torch.float16)))

    # ------------------------------------------------------------------- optional mask (half)
    if not nomask and mask is not None:
        mask_h = mask.unsqueeze(-1).to(torch.float16)   # [B,N,N,1]
        left  = left  * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------- apply left/right gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- re‑order to (B, hidden, N, N) – contiguous
    left  = left.permute(0, 3, 1, 2).contiguous()    # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # ------------------------------------------------------------------- collapse batch*hidden into a single “page” dimension
    BxH = B * hidden
    left_mat  = left.view(BxH, N, N)   # half
    right_mat = right.view(BxH, N, N)   # half

    # ------------------------------------------------------------------- allocate output of outer‑product (FP16 to cut bandwidth)
    out_mat_half = torch.empty((BxH, N, N), dtype=torch.float16, device=device)

    # ------------------------------------------------------------------- strides (in elements)
    stride_az = left_mat.stride(0)
    stride_ai = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bz = right_mat.stride(0)
    stride_bk = right_mat.stride(2)     # column stride (k‑dim) = 1
    stride_bn = right_mat.stride(1)     # row stride (n‑dim) = N

    stride_cz = out_mat_half.stride(0)
    stride_ci = out_mat_half.stride(1)
    stride_cj = out_mat_half.stride(2)

    # ------------------------------------------------------------------- tile size selection (tuned for H100)
    if N <= 256:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 16
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 128, 128, 32

    grid = (BxH,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # ------------------------------------------------------------------- launch outer‑product kernel
    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat_half,
        BxH,
        N,
        stride_az, stride_ai, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------- reshape to (B, N, N, hidden) – hidden as innermost dim
    out_hidden_half = out_mat_half.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- fuse LayerNorm (over hidden) + output‑gate
    out_hidden = torch.empty_like(out_hidden_half, dtype=torch.float32)

    # strides for the tensors (all already in elements)
    in_stride_b = out_hidden_half.stride(0)
    in_stride_i = out_hidden_half.stride(1)
    in_stride_j = out_hidden_half.stride(2)
    in_stride_h = out_hidden_half.stride(3)

    gate_stride_b = out_gate.stride(0)
    gate_stride_i = out_gate.stride(1)
    gate_stride_j = out_gate.stride(2)
    gate_stride_h = out_gate.stride(3)

    out_stride_b = out_hidden.stride(0)
    out_stride_i = out_hidden.stride(1)
    out_stride_j = out_hidden.stride(2)
    out_stride_h = out_hidden.stride(3)

    total_locs = B * N * N
    BLOCK_SIZE = 256                     # threads per block (≤1024)

    grid_norm = (triton.cdiv(total_locs, BLOCK_SIZE),)

    norm_gate_kernel[grid_norm](
        out_hidden_half,                 # inp_ptr
        out_gate,                        # gate_ptr
        out_hidden,                      # out_ptr
        B, N, hidden,
        in_stride_b, in_stride_i, in_stride_j, in_stride_h,
        gate_stride_b, gate_stride_i, gate_stride_j, gate_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        weights["to_out_norm.weight"].contiguous(),
        weights["to_out_norm.bias"].contiguous(),
        H=hidden,
        eps=1e-5,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=8,
    )

    # ------------------------------------------------------------------- final dense projection (cuBLAS)
    # out_hidden : [B, N, N, hidden]  (float32)
    # to_out.weight : [dim, hidden]   (float32)
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
