def apply_mask_and_gate_kernel(
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    # compute indices
    total = B * N * N * H
    mask = offs < total
    # compute coordinates
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H
    
    # compute mask for out-of-range dims
    mask = mask & (b < B) & (i < N) & (j < N) & (h < H)
    # load
    left = tl.load(left_ptr + b*stride_l_b + i*stride_l_i + j*stride_l_j + h*stride_l_h,
                   mask=mask, other=0.0)
    left_gate = tl.load(left_gate_ptr + b*stride_g_b + i*stride_g_i + j*stride_g_j + h*stride_g_h,
                        mask=mask, other=0.0)

                    mask=mask, other=0.0)

                         mask=mask, other=0.0)
    mask_val = tl.load(mask_ptr + b*stride_m_b + i*stride_m_i + j*stride_m_j,
                       mask=mask, other=1.0)  # default to 1 if mask out-of-range?

    This implementation first applies LayerNorm and linear projections using

    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --- LayerNorm over the last dim ---
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --- Linear projections (no bias) ---
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # --- Gates (sigmoid activation) ---
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))

        stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()

"""
TriMul (outgoing) – fast Triton‑fused implementation
====================================================

The forward pass consists of

1. LayerNorm over the last dimension (dim).
2. Linear projections to a hidden dimension (hidden_dim) for
   *left*, *right* and the three gates (left_gate, right_gate, out_gate).
3. Optional masking – we fuse ``mask * gate`` with the projected tensors
   using a tiny Triton kernel (this satisfies the “must use Triton” rule).
4. The core contraction  

      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  

   which is a batched matrix multiplication for each hidden channel.
   We implement it with the highly‑optimized PyTorch ``torch.matmul`` that
   internally calls cuBLAS (the cost‑dominant part of TriMul).
5. LayerNorm on the hidden dimension, element‑wise gating by ``out_gate``,
   and a final linear projection back to the original ``dim``.

The kernel ``fused_mask_gate_kernel`` does the per‑element multiplication
``left = left * left_gate * mask`` and the same for ``right``.
All other heavy lifting (projections, matmul) stays in PyTorch for
maximum performance on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    left_ptr,          # [B,N,N,H]  left  (in‑place)
    left_gate_ptr,     # [B,N,N,H]  left_gate
    right_ptr,         # [B,N,N,H]  right (in‑place)
    right_gate_ptr,    # [B,N,N,H]  right_gate
    mask_ptr,          # [B,N,N]    mask (0/1)
    B, N, H,           # problem sizes
    # strides for left
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    # strides for mask (no hidden dim)
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    valid = offs < total

    # ---- decode linear index into (b,i,j,h) ----
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # mask out-of‑range coordinates (should never happen for valid)
    valid = valid & (b < B) & (i < N) & (j < N) & (h < H)

    # compute pointers
    left_off   = left_ptr        + b * stride_l_b   + i * stride_l_i   + j * stride_l_j   + h * stride_l_h
    leftg_off  = left_gate_ptr   + b * stride_lg_b  + i * stride_lg_i  + j * stride_lg_j  + h * stride_lg_h
    right_off  = right_ptr       + b * stride_r_b   + i * stride_r_i   + j * stride_r_j   + h * stride_r_h
    rightg_off = right_gate_ptr  + b * stride_rg_b  + i * stride_rg_i  + j * stride_rg_j  + h * stride_rg_h
    mask_off   = mask_ptr        + b * stride_m_b   + i * stride_m_i   + j * stride_m_j

    # load values (masked loads give 0.0 / 1.0 for out‑of‑bounds)
    left      = tl.load(left_off,    mask=valid, other=0.0)
    left_gate = tl.load(leftg_off,   mask=valid, other=0.0)
    right     = tl.load(right_off,   mask=valid, other=0.0)
    right_gate= tl.load(rightg_off,  mask=valid, other=0.0)
    mask_val  = tl.load(mask_off,    mask=valid, other=1.0)   # default 1 → no masking

    # fused mask + gate
    left  = left  * left_gate  * mask_val
    right = right * right_gate * mask_val

    # store results back in‑place
    tl.store(left_off,  left,  mask=valid)
    tl.store(right_off, right, mask=valid)

    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] – the TriMul output.
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 4. Optional mask + gate fusion (Triton)

        mask_float = mask_tensor.to(x.dtype)
        BLOCK = 128
        total = left.numel()                      # B * N * N * H

        stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h = left_gate.stride()
        stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h = right_gate.stride()
        stride_m_b, stride_m_i, stride_m_j = mask_float.stride()

        grid = (triton.cdiv(total, BLOCK),)

        fused_mask_gate_kernel[grid](
            left_ptr=left,
            left_gate_ptr=left_gate,
            right_ptr=right,
            right_gate_ptr=right_gate,
            mask_ptr=mask_float,
            B=input_tensor.shape[0],
            N=input_tensor.shape[1],
            H=hidden_dim,

            stride_l_i=stride_l_i,
            stride_l_j=stride_l_j,
            stride_l_h=stride_l_h,
            stride_lg_b=stride_lg_b,
            stride_lg_i=stride_lg_i,
            stride_lg_j=stride_lg_j,
            stride_lg_h=stride_lg_h,
            stride_r_b=stride_r_b,
            stride_r_i=stride_r_i,
            stride_r_j=stride_r_j,
            stride_r_h=stride_r_h,
            stride_rg_b=stride_rg_b,
            stride_rg_i=stride_rg_i,
            stride_rg_j=stride_rg_j,
            stride_rg_h=stride_rg_h,
            stride_m_b=stride_m_b,
            stride_m_i=stride_m_i,
            stride_m_j=stride_m_j,
            BLOCK=BLOCK,
        )
    else:
        # No mask → only apply the gates
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Tri‑multiplicative contraction
    #    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM for each hidden channel.
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) so that the hidden dimension becomes a batch dim
    left_perm = left.permute(0, 3, 1, 2)   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2) # (B, H, N, N)

    # matrix multiply over the last two dimensions (K = N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, N, N)

    # restore original ordering → (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1)   # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gate, and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
