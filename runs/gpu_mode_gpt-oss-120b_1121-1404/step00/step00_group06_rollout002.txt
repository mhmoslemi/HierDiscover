def trimul_matmul_kernel(
    left_ptr, right_ptr,
    out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0) # row block
    pid_n = tl.program_id(1) # col block
    pid_z = tl.program_id(2) # batch*hidden idx

    batch_id = pid_z // D
    d = pid_z % D

    # compute start indices for i and j
    offs_i = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute mask for in-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)
    batch_id = pid_z // D
    d = pid_z % D

    # compute block offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # ranges may be out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M x K)
        left_ptrs = left_ptr + batch_id * stride_left_b \
                    + offs_i[:, None] * stride_left_i \
                    + offs_k[None, :] * stride_left_k \
                    + d * stride_left_d
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile (K x N) i.e., transposed right
        # Here we load right as (K, N) with rows = k, cols = j
        right_ptrs = right_ptr + batch_id * stride_right_b \
                    + offs_k[:, None] * stride_right_k \
                    + offs_j[None, :] * stride_right_i \
                    + d * stride_right_d
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Dot product accumulate
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + batch_id * stride_out_b \
               + offs_i[:, None] * stride_out_i \
               + offs_j[None, :] * stride_out_j \
               + d * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

(LayerNorms, linear projections, gating, final linear) are performed
with PyTorch.

The implementation follows the reference PyTorch code:
1. Layer‑norm the 4‑D input.
2. Linear projections → left / right.
3. Optional mask and sigmoid gates.
4. Triton kernel computes the N‑wise contraction.
5. Layer‑norm, output‑gate and final linear projection.
"""
import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – one (batch, hidden) pair per program instance
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # sizes
    B, N, D,
    # strides – contiguous (B,N,N,D) tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)               # tile row (i)
    pid_j = tl.program_id(1)               # tile col (j)
    pid_z = tl.program_id(2)               # flattened (batch, hidden)

    # recover batch index and hidden channel
    batch_id = pid_z // D
    d = pid_z % D

    # ------------------------------------------------------------------
    # tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_i_in_bounds = offs_i < N
    offs_j_in_bounds = offs_j < N

    # accumulator in fp16 (can also be fp32 for higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # loop over the reduction dimension k (seq_len)
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        offs_k_in_bounds = offs_k < N

        # --------------------------------------------------------------
        # load left tile  (M × K)  – left[b, i, k, d]
        # --------------------------------------------------------------
        left_offset = (
            batch_id * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + d * stride_left_d
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=offs_i_in_bounds[:, None] & offs_k_in_bounds[None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # load right tile (K × N) – right_T[b, k, j, d]  (i.e. rightᵀ)
        # --------------------------------------------------------------
        right_offset = (
            batch_id * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_i
            + d * stride_right_d
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=offs_k_in_bounds[:, None] & offs_j_in_bounds[None, :],
            other=0.0,
        )

        # --------------------------------------------------------------
        # accumulate the partial matrix product
        # --------------------------------------------------------------
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # write the result tile
    # ------------------------------------------------------------------
    out_offset = (
        batch_id * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + d * stride_out_d
    )
    tl.store(
        out_ptr + out_offset,
        acc,
        mask=offs_i_in_bounds[:, None] & offs_j_in_bounds[None, :],
    )


# ----------------------------------------------------------------------
# Public entry point – called by the external harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N]  (bool / float, may be all‑ones)
        - weights      : dict of torch.Tensor containing all linear / layer‑norm params
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    (inp, mask, weights, config) = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, C = inp.shape
    device = inp.device
    dtype = inp.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        inp, (C,), weight=norm_w, bias=norm_b
    )  # [B,N,N,C]

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]      # (hidden_dim, C)
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)     # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on last dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    mask_f = mask.to(dtype).unsqueeze(-1)       # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – they remain in float32
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast projection tensors to fp16 (reduce memory traffic)
    # ------------------------------------------------------------------
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 6) Triton kernel – N×N contraction for each hidden channel
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # strides for the contiguous tensors (B,N,N,D)
    s_left = left.stride()
    s_right = right.stride()
    s_out = out.stride()

    # tile sizes – empirically good for H100, fp16
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _trimul_matmul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Back to float32 for the remaining operations
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )
    out = out * out_gate                             # broadcast over hidden dim
    to_out_w = weights["to_out.weight"]               # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)   # final projection

    return out
