@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets
    i_base = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k_base = tl.arange(0, BLOCK_K)  # used inside loop
    d_base = tl.arange(0, BLOCK_H)

    # Mask out-of-range
    mask_i = i_base < N
    mask_j = j_base < N

    # Initialize accumulation buffer
    acc = tl.zeros((BLOCK_N, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over k tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k_offset = kk * BLOCK_K + k_base
        mask_k = k_offset < N

        # For each hidden tile
        for h_off in range(tl.cdiv(H, BLOCK_H)):
            d_offset = h_off * BLOCK_H + d_base
            mask_d = d_offset < H

            # Compute pointer offsets
            left_ptrs = left_ptr + \
                pid_b * stride_left_b + \
                i_base[:, None, None] * stride_left_i + \
                k_offset[None, :, None] * stride_left_k + \
                d_offset[None, None, :] * stride_left_h

            right_ptrs = right_ptr + \
                pid_b * stride_right_b + \
                j_base[:, None, None] * stride_right_j + \
                k_offset[None, :, None] * stride_right_k + \
                d_offset[None, None, :] * stride_right_h

            # Load blocks
            left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)
            right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

            # Compute contribution for this k tile:
            # left shape (BLOCK_N, BLOCK_K, BLOCK_H)
            # right shape (BLOCK_N, BLOCK_K, BLOCK_H)
            # We need sum_k left[i,k,d] * right[j,k,d] -> shape (BLOCK_N, BLOCK_N, BLOCK_H)
            # Compute using broadcasting
            # Expand left across j, right across i
            # left_exp: (i,k,1,d), right_exp: (1,k,j,d)
            left_exp = left[:, :, None, :]   # shape (i,k,1,d)
            right_exp = right[None, :, :, :] # shape (1,k,j,d)
            prod = left_exp * right_exp      # shape (i,k,j,d)
            # sum over k dimension (axis=1)
            acc += tl.sum(prod, axis=1)      # shape (i,j,d)
    # Write back out
    out_ptrs = out_ptr + \
        pid_b * stride_out_b + \
        i_base[:, None, None] * stride_out_i + \
        j_base[None, :, None] * stride_out_j + \
        d_base[None, None, :] * stride_out_h

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])
```

"""
custom_kernel
==============

Implements the *outgoing* Triangle Multiplicative Update (TriMul) from AlphaFold‑3.
The heavy N³ reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is performed by a hand‑written Triton kernel.  All other
operations (LayerNorms, Linear projections, sigmoid gates and the final
linear projection) are expressed with PyTorch functions for simplicity.
The kernel works on FP16 tensors (the expensive reduction) while the
rest of the network stays in FP32, and the final result is returned in
float32.

The kernel processes a 4‑D tensor ``[B, N, N, H]`` (B≤2, N up to 1024,
H = hidden_dim, typically 128) using a 3‑D launch grid ``(B, ceil(N/BLOCK_N),
ceil(N/BLOCK_N))``.  Each program instance computes a ``BLOCK_N×BLOCK_N×BLOCK_H``
tile of the output by iterating over the K dimension in tiles of size
``BLOCK_K``.  The inner work‑load is

    acc[i,j,d] += Σ_k left[i,k,d] * right[j,k,d]

implemented with a broadcasted element‑wise product followed by a
reduction over the K‑axis (`tl.sum`).  The accumulator is FP32 for
numerical stability.  Edge handling (non‑multiple dimensions and masking)
is done with Triton masks.

The overall workflow:

    x      = LayerNorm(x)
    left   = Linear(x, left_proj)   * mask * sigmoid(Linear(x, left_gate))
    right  = Linear(x, right_proj)  * mask * sigmoid(Linear(x, right_gate))
    out    = tri_mul(left, right)                      # Triton kernel
    out    = LayerNorm(out) * sigmoid(Linear(x, out_gate))
    output = Linear(out, to_out)                       # final projection
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: triangular multiplicative reduction
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # problem size
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # ------------------------------------------------------
    #  program identifiers
    # ------------------------------------------------------
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # tile over the first N axis
    pid_j = tl.program_id(2)   # tile over the second N axis

    # ------------------------------------------------------
    #  tile offsets
    # ------------------------------------------------------
    i = pid_i * BLOCK_N + tl.arange(0, BLOCK_N)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)
    d = tl.arange(0, BLOCK_H)

    mask_i = i < N
    mask_j = j < N

    # accumulator for a (BLOCK_N, BLOCK_N, BLOCK_H) tile in FP32
    acc = tl.zeros((BLOCK_N, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # number of tiles along the reduction (k) dimension
    num_k = tl.cdiv(N, BLOCK_K)
    # number of tiles along the hidden dimension
    num_h = tl.cdiv(H, BLOCK_H)

    # ------------------------------------------------------
    #  outer loops over K and hidden tiles
    # ------------------------------------------------------
    for kk in range(num_k):
        k_off = kk * BLOCK_K + k               # absolute k indices
        mask_k = k_off < N

        for hh in range(num_h):
            d_off = hh * BLOCK_H + d           # absolute hidden indices
            mask_d = d_off < H

            # --------------------------------------------------
            #  pointer arithmetic for the current tile
            # --------------------------------------------------
            left_ptrs = left_ptr + \
                pid_b * stride_l_b + \
                i[:, None, None] * stride_l_i + \
                k_off[None, :, None] * stride_l_k + \
                d_off[None, None, :] * stride_l_h

            right_ptrs = right_ptr + \
                pid_b * stride_r_b + \
                j[:, None, None] * stride_r_j + \
                k_off[None, :, None] * stride_r_k + \
                d_off[None, None, :] * stride_r_h

            # --------------------------------------------------
            #  load with masks (out‑of‑bounds elements are zero)
            # --------------------------------------------------
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )
            # --------------------------------------------------
            #  broadcast to (i,k,j,d) and reduce over k
            # --------------------------------------------------
            # left  : (i, k, 1, d)
            # right : (1, k, j, d)
            left_exp  = left[:, :, None, :]      # (i, k, 1, d)
            right_exp = right[None, :, :, :]    # (1, k, j, d)
            prod = left_exp * right_exp          # (i, k, j, d)
            acc += tl.sum(prod, axis=1)          # sum over k -> (i, j, d)

    # ------------------------------------------------------
    #  write the tile back to memory (masked)
    # ------------------------------------------------------
    out_ptrs = out_ptr + \
        pid_b * stride_o_b + \
        i[:, None, None] * stride_o_i + \
        j[None, :, None] * stride_o_j + \
        d[None, None, :] * stride_o_h

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # optional mask (if config specifies a mask)
    if not config.get("nomask", False) and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)           # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # apply gating
    left = left * left_gate
    right = right * right_gate

    ##################################################################
    # 3. Triangular multiplicative reduction (Triton)
    ##################################################################
    # Cast to FP16 for the heavy reduction
    left_f16 = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # output buffer (zero‑initialized, FP32 as required by later layers)
    out = torch.zeros(bs, N, N, hidden_dim, dtype=torch.float32, device=device)

    # strides (in elements) – Triton expects them as ints
    ls = left_f16.stride()
    rs = right_f16.stride()
    os = out.stride()

    # block sizes – tuned for H100 FP16 tensor‑core occupancy
    BLOCK_N = 16   # tile over i / j
    BLOCK_K = 16   # reduction tile
    BLOCK_H = 16   # hidden‑dim tile

    grid = (bs, (N + BLOCK_N - 1) // BLOCK_N, (N + BLOCK_N - 1) // BLOCK_N)
    tri_mul_kernel[grid](
        left_f16,
        right_f16,
        out,
        bs,
        N,
        hidden_dim,
        ls[0], ls[1], ls[2], ls[3],
        rs[0], rs[1], rs[2], rs[3],
        os[0], os[1], os[2], os[3],
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,               # enough warps to hide latency
    )

    ##################################################################
    # 4. Final LayerNorm, gating, and linear projection back to dim
    ##################################################################
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return output
```
