    hidden_dim = config["hidden_dim"]

    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if not nomask i.e., mask given. mask shape [bs,N,N]
    if mask is not None:

        right = right * mask

    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

@triton.jit
def trimul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                           B, N, D,
                           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_bd = tl.program_id(axis=2)

    # compute batch and hidden index
    b = pid_bd // D
    d = pid_bd % D

    # convert to int64 for offset calculations
    N_i = tl.int64(N)
    D_i = tl.int64(D)
    b_i = tl.int64(b)
    d_i = tl.int64(d)

    # i and j coordinates of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N_i
    mask_j = offs_j < N_i

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N_i

        # left: shape (BLOCK_M, BLOCK_K)
        # offset = ((b * N + i) * N + k) * D + d
        left_offset = ((b_i * N_i + offs_i[:, None]) * N_i + cur_k[None, :]) * D_i + d_i
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right: shape (BLOCK_K, BLOCK_N) (k as row, j as column)
        # offset = ((b * N + j) * N + k) * D + d
        right_offset = ((b_i * N_i + offs_j[None, :]) * N_i + cur_k[:, None]) * D_i + d_i
        right = tl.load(right_ptr + right_offset,
                        mask=mask_j[None, :] & mask_k[:, None],
                        other=0.0)

        # multiply-accumulate
        acc += tl.dot(left, right)

    # store result
    out_offset = ((b_i * N_i + offs_i[:, None]) * N_i + offs_j[None, :]) * D_i + d_i
    tl.store(out_ptr + out_offset,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def trimul_outgoing_kernel(left_ptr, right_ptr, out_ptr,
                           B, N, D,
                           BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d].
    Each program instance processes a tile of (i,j) for a given batch b and hidden index d.
    """
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_bd = tl.program_id(axis=2)

    # decode batch and hidden dimension index
    b = pid_bd // D
    d = pid_bd % D

    # Cast to 64‑bit for safe address arithmetic
    N_i = tl.int64(N)
    D_i = tl.int64(D)
    b_i = tl.int64(b)
    d_i = tl.int64(d)

    # i‑ and j‑coordinates of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N_i
    mask_j = offs_j < N_i

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N_i

        # left[b,i,k,d]  -> shape (BLOCK_M, BLOCK_K)
        left_off = ((b_i * N_i + offs_i[:, None]) * N_i + cur_k[None, :]) * D_i + d_i
        left = tl.load(left_ptr + left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b,j,k,d] but we need (k,j) for the dot: shape (BLOCK_K, BLOCK_N)
        right_off = ((b_i * N_i + offs_j[None, :]) * N_i + cur_k[:, None]) * D_i + d_i
        right = tl.load(right_ptr + right_off,
                        mask=mask_j[None, :] & mask_k[:, None],
                        other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(left, right)

    # -----------------------------------------------------------------
    # Write the tile back to global memory
    # -----------------------------------------------------------------
    out_off = ((b_i * N_i + offs_i[:, None]) * N_i + offs_j[None, :]) * D_i + d_i
    tl.store(out_ptr + out_off,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward pass implemented with Triton.

The heavy N³ reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed by a custom Triton kernel (trimul_outgoing_kernel).  All
layer‑norms, linear projections and gating are handled with regular
PyTorch for simplicity.  The kernel processes a tile of the (i,j)
plane for a fixed batch index *b* and hidden‑dim index *d*.  Each
program (block) loads a (BLOCK_M, BLOCK_K) tile of `left` and a
(BLOCK_K, BLOCK_N) tile of `right`, accumulates the dot product over
k and writes the result to `out`.  Masks are applied in the PyTorch
pre‑processing stage, so the kernel only needs to handle the pure
tensor contraction.

The implementation works for the full range of test configurations
(batches ≤ 2, seq_len up to 1024, hidden_dim up to 384) on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,   # const float*  [B, N, N, D]
    right_ptr,  # const float*  [B, N, N, D]
    out_ptr,    # float*        [B, N, N, D]
    B, N, D,    # int32 scalars
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    where the reduction runs over the second dimension (k) of the
    left/right tensors.  One program instance handles a tile of size
    (BLOCK_M, BLOCK_N) for a specific (b, d) pair.
    """
    # -----------------------------------------------------------------
    # 1) Decode the 2‑D tile and the (b, d) pair from program ids.
    # -----------------------------------------------------------------
    pid_m = tl.program_id(axis=0)      # tile row   (i)
    pid_n = tl.program_id(axis=1)      # tile col   (j)
    pid_bd = tl.program_id(axis=2)     # combined batch & hidden dim

    b = pid_bd // D                     # batch index
    d = pid_bd % D                      # hidden‑dim index

    # 64‑bit arithmetic to avoid overflow for large N, D, B
    N_i = tl.int64(N)
    D_i = tl.int64(D)
    b_i = tl.int64(b)
    d_i = tl.int64(d)

    # -----------------------------------------------------------------
    # 2) Compute global i/j coordinates for this tile.
    # -----------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = offs_i < N_i
    mask_j = offs_j < N_i

    # -----------------------------------------------------------------
    # 3) Accumulate over the reduction dimension k.
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)               # [BLOCK_K]
        mask_k = cur_k < N_i

        # left[b, i, k, d]  -> shape (BLOCK_M, BLOCK_K)
        left_off = ((b_i * N_i + offs_i[:, None]) * N_i + cur_k[None, :]) * D_i + d_i
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right[b, j, k, d] but we need (k, j) for the dot product:
        # shape (BLOCK_K, BLOCK_N)
        right_off = ((b_i * N_i + offs_j[None, :]) * N_i + cur_k[:, None]) * D_i + d_i
        right = tl.load(
            right_ptr + right_off,
            mask=mask_j[None, :] & mask_k[:, None],
            other=0.0,
        )

        # Multiply‑accumulate over k
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # -----------------------------------------------------------------
    # 4) Store the computed tile back to global memory.
    # -----------------------------------------------------------------
    out_off = ((b_i * N_i + offs_i[:, None]) * N_i + offs_j[None, :]) * D_i + d_i
    tl.store(
        out_ptr + out_off,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )
