                       stride_cob, stride_cik, stride_cjk, stride_cd,

   d = pid_group % HIDDEN

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The module implements the forward computation from the AlphaFold3 paper:

1. Layer‑norm on the input tensor (last dim = dim).
2. Linear projections `left_proj` and `right_proj` (dim → hidden_dim).
3. Optional pair‑wise mask multiplication.
4. Gated projections: element‑wise sigmoid gates applied to the projected tensors.
5. Core multiplicative update  
   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]    
   which is exactly a batched matrix multiplication:
   out[b, :, :, h] = left[b, :, :, h] @ right[b, :, :, h].T
   This O(N³) contraction is performed by a custom Triton kernel that
   processes (batch, hidden) groups in parallel and uses FP16 arithmetic
   with FP32 accumulation.
6. Layer‑norm on the hidden dimension, gating with `out_gate`,
   and a final linear projection back to `dim`.

Only the O(N³) step is written in Triton; all other ops use
torch.nn.functional for simplicity and correctness.
"""

import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trmul_matmul_kernel(
    left_ptr,                # const half *
    right_ptr,               # const half *
    out_ptr,                 # half *
    BATCH, HIDDEN,           # i32
    M, N, K,                 # i32  (M=N=seq_len, K=seq_len)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ right^T for each (batch, hidden) pair.
    left  : [B, M, K, H]
    right : [B, N, K, H]   (note the second seq dim is the column index)
    out   : [B, M, N, H]
    """

    pid_group = tl.program_id(2)                     # 0 .. B*H-1
    b = pid_group // HIDDEN
    h = pid_group % HIDDEN

    pid_n = tl.program_id(1)                         # block col (j)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    offs_k = tl.arange(0, BLOCK_K)                     # [BLOCK_K]

    mask_n = offs_n < N

    a_ptrs = (
        left_ptr
        + b * stride_left_b
        + h * stride_left_h
        + offs_m[:, None] * stride_left_i
        + offs_k[None, :] * stride_left_k
    )

    b_ptrs = (
        right_ptr
        + b * stride_right_b
        + h * stride_right_h
        + offs_k[:, None] * stride_right_k
        + offs_n[None, :] * stride_right_i
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # Loop over K dimension (seq_len)
    # -------------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k_tiles):
        k_offset = k_iter * BLOCK_K
        # masks for the current k‑slice
        mask_k = (k_offset + offs_k) < K
        mask_a = mask_m[:, None] & mask_k[None, :]          # [M, K]
        mask_b = mask_k[:, None] & mask_n[None, :]          # [K, N]

        # fp16 loads, cast to fp32 for accumulation
        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

        # GEMM update
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # ----- write result -------------------------------------------------------
    c_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_m[:, None] * stride_out_i
        + offs_n[None, :] * stride_out_j
    )
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        [bs, seq_len, seq_len, dim] output tensor (float32)
    """
    # -------------------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    # Expected shapes
    # input_tensor : [B, N, N, dim]
    # mask         : [B, N, N]   (may be ignored if `nomask` is True)

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------------------------
    # 1. Input layer‑norm (dim -> dim)
    # -------------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim)
    # -------------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # 3. Optional mask
    # -------------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        # mask is [B,N,N]; broadcast over hidden dim
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------------
    # 4. Gated projections
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Core TriMul contraction via Triton (fp16 for speed)
    # -------------------------------------------------------------------------
    # Cast to fp16 before the kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Allocate output tensor (fp16) : [B,N,N,hidden_dim]
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left.device)

    # Strides (int64 values)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp16.stride()

    # Tile sizes – tuned for H100, FP16 matmul
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Launch grid: (M‑tiles, N‑tiles, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _trmul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        hidden_dim,
        N,
        N,
        N,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # -------------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim layer‑norm, gating, final projection
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of the Triton kernel**

* The kernel computes a batched matrix multiplication for each `(batch, hidden)` pair.
* `left` is interpreted as an `M × K` matrix, `right` as a `K × N` matrix (the latter is read transposed so that the contraction matches the required einsum).
* Tiling (`BLOCK_M × BLOCK_N × BLOCK_K`) enables high occupancy on the H100.
* Inputs are in FP16 for memory‑bandwidth efficiency; the accumulation is performed in FP32 to retain numerical stability.
* Masks guard against out‑of‑bounds loads for the final incomplete tiles.

All remaining linear layers, LayerNorms, and sigmoid gates are performed with PyTorch’s highly‑optimized kernels, keeping the implementation correct, flexible, and fast across the required test configurations.
