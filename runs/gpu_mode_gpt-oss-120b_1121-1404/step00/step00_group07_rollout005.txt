@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_lk, stride_li, stride_lh,
    stride_rb, stride_ri, stride_rj, stride_rh,
    stride_ob, stride_ok, stride_oj, stride_oh,
    BLOCK_K: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_I: tl.constexpr
):
    bh_idx = tl.program_id(0)  # combined batch and head dimension
    tile_k = tl.program_id(1)  # output row tile (k)
    tile_j = tl.program_id(2)  # output col tile (j)

    # compute batch and head
    b = bh_idx // H
    h = bh_idx % H

    # Compute offsets
    offs_k = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
    offs_j = tile_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # Bound checking
    mask_k = offs_k < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_K, BLOCK_J), dtype=tl.float32)

    # Loop over i dimension
    for start_i in range(0, tl.cdiv(N, BLOCK_I)):
        offs_i = start_i * BLOCK_I + tl.arange(0, BLOCK_I)
        mask_i = offs_i < N

        # Compute pointers
        # left shape: (b, k, i, h)
        left_ptrs = left_ptr + b * stride_lb + h * stride_lh + offs_k[:, None] * stride_lk + offs_i[None, :] * stride_li
        # right shape: (b, i, j, h)
        right_ptrs = right_ptr + b * stride_rb + h * stride_rh + offs_i[:, None] * stride_ri + offs_j[None, :] * stride_rj

        # Load
        left = tl.load(left_ptrs, mask=mask_k[:, None] & mask_i[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_i[:, None] & mask_j[None, :], other=0.0)

        # Cast to float32 for accumulation
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        # Perform dot

        right_ptr,  # *fp16
        out_ptr,    # *fp32

        stride_rb, stride_ri, stride_rj, stride_rh,
        stride_ob, stride_ok, stride_oj, stride_oh,
        BLOCK_K: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_I: tl.constexpr
    ):

        tile_k = tl.program_id(1)  # output row tile (k dimension)
        tile_j = tl.program_id(2)  # output col tile (j dimension)

        # decode batch and head
        b = bh // H
        h = bh % H

        # compute offsets
        offs_k = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
        offs_j = tile_j * BLOCK_J + tl.arange(0, BLOCK_J)
        # Load mask for output dims
        kmask = offs_k < N
        jmask = offs_j < N

        acc = tl.zeros((BLOCK_K, BLOCK_J), dtype=tl.float32)

        # loop over i (reduction dimension)
        for start_i in range(0, tl.cdiv(N, BLOCK_I)):
            offs_i = start_i * BLOCK_I + tl.arange(0, BLOCK_I)
            imask = offs_i < N

            # pointers
            # left_T has shape (b, k, i, h)
            left_ptrs = left_ptr + b * stride_lb + h * stride_lh + offs_k[:, None] * stride_lk + offs_i[None, :] * stride_li
            # right has shape (b, i, j, h)
            right_ptrs = right_ptr + b * stride_rb + h * stride_rh + offs_i[:, None] * stride_ri + offs_j[None, :] * stride_rj

            # load
            left = tl.load(left_ptrs, mask=kmask[:, None] & imask[None, :], other=0.0)
            right = tl.load(right_ptrs, mask=imask[:, None] & jmask[None, :], other=0.0)

            # cast to float32 for accumulation
            left_f = left.to(tl.float32)
            right_f = right.to(tl.float32)

            acc += tl.dot(left_f, right_f)  # (BLOCK_K, BLOCK_I) @ (BLOCK_I, BLOCK_J)

        # store result
        out_ptrs = out_ptr + b * stride_ob + h * stride_oh + offs_k[:, None] * stride_ok + offs_j[None, :] * stride_oj
        tl.store(out_ptrs, acc, mask=kmask[:, None] & jmask[None, :])

    # Choose block sizes heuristically
    BLOCK_K = 64
    BLOCK_J = 64
    BLOCK_I = 32

    grid = (B * hidden_dim,
            (N + BLOCK_K - 1) // BLOCK_K,
            (N + BLOCK_J - 1) // BLOCK_J)

    tri_mul_outgoing_kernel[grid](
        left_T_fp16,
        right_fp16,
        out_fp32,
        B, N, hidden_dim,
        lb, lk, li, lh,
        rb, ri, rj, rh,
        ob, ok, oj, oh,
        BLOCK_K=BLOCK_K, BLOCK_J=BLOCK_J, BLOCK_I=BLOCK_I
    )
    # ensure kernel finished
    torch.cuda.synchronize()

    # out now shape [B,N,N,hidden] corresponding to (k,j) as required for outgoing
    out = out_fp32

    # Apply second LayerNorm and gating
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # weight shape (dim, hidden_dim)

    # For consistency return float32
    return out

"""
custom_kernel: Outgoing triangular multiplicative update (TriMul).

The forward pass computes
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,k,j,d]           (1)

where
    left  = (x • W_left)   * mask * sigmoid(x • W_left_gate)
    right = (x • W_right)  * mask * sigmoid(x • W_right_gate)

x is first layer‑normed (over the last dimension). After the
multiplication (1) a second layer‑norm and an output gate are applied
and finally a linear projection back to the original dimension.

The heavy O(N³·C) part (1) is implemented in a fused Triton kernel.
All other cheap ops (layer‑norms, linear projections, sigmoid gates,
masking) are done with PyTorch.
The kernel works on a batch*head (hidden) dimension, i.e. for each
pair (batch, hidden) it performs a dense matrix‑multiply:
    L : (N, N)  (i‑k)   ×   R : (N, N)  (k‑j)   →   O : (N, N)  (i‑j)

Mixed‑precision is used: the inputs to the kernel are fp16, the
accumulator is fp32, and the result is stored as fp32 before the
subsequent layer‑norms.
"""
import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,    # fp16  * [B, N, N, H]  (i,k)
    right_ptr,   # fp16  * [B, N, N, H]  (k,j)
    out_ptr,     # fp32  * [B, N, N, H]  (i,j)
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rk, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program_id(0) : combined batch * head index
    # program_id(1) : output tile row (i)   index
    # program_id(2) : output tile column (j) index
    bh_idx = tl.program_id(0)
    tile_i = tl.program_id(1)
    tile_j = tl.program_id(2)

    # decode batch and head
    b = bh_idx // H
    h = bh_idx % H

    # offsets within the tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block  (i,k)   shape [BLOCK_M, BLOCK_K]
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        # right block (k,j)   shape [BLOCK_K, BLOCK_N]
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
        )

        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)          # fp16
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)          # fp16

        # promote to fp32 for the dot product
        left_f = left.to(tl.float32)
        right_f = right.to(tl.float32)

        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(left_f, right_f)

    # store the result (fp32)
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of model weight tensors
            - config       : dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Config / device
    # ------------------------------------------------------------------
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Weights (all are stored as float32)
    # ------------------------------------------------------------------
    # LayerNorm on the input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear proj + gates (no bias)
    left_proj_weight   = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_weight  = weights["right_proj.weight"]   # (hidden_dim, dim)
    left_gate_weight   = weights["left_gate.weight"]    # (hidden_dim, dim)
    right_gate_weight  = weights["right_gate.weight"]   # (hidden_dim, dim)
    out_gate_weight    = weights["out_gate.weight"]     # (hidden_dim, dim)

    # Output processing
    to_out_norm_weight = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]     # (hidden_dim,)
    to_out_weight      = weights["to_out.weight"]        # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Optional masking
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – computed in float32
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate    # [B,N,N,hidden]
    right = right_proj * right_gate  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6) Core O(N³) multiplication (outgoing TriMul) – Triton kernel
    #    out[i,j] = Σ_k left[i,k] * right[k,j]
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (reduces bandwidth)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Output will be stored as fp32 because the following LayerNorm
    # expects a float32 tensor.
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides (in elements)
    stride_lb, stride_li, stride_lk, stride_lh = left_fp16.stride()
    stride_rb, stride_rk, stride_rj, stride_rh = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_fp32.stride()

    # Tile sizes – chosen to give good occupancy on H100.
    BLOCK_M = 64   # rows (i)
    BLOCK_N = 64   # cols (j)
    BLOCK_K = 32   # reduction (k)

    grid = (
        B * hidden_dim,                         # batch * head dimension
        (N + BLOCK_M - 1) // BLOCK_M,           # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,           # tiles over j
    )

    tri_mul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rk, stride_rj, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 8) Output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate                           # broadcast over hidden
    out = torch.nn.functional.linear(out, to_out_weight)   # (dim, hidden)

    return out
```
