"""
TriMul forward pass with Triton-accelerated contraction.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×D}.
2. Project x to a hidden dimension H with two linear layers (left/right).
3. Apply the pairwise mask (if provided) and element‑wise sigmoid gates.
4. Compute the heavy pairwise contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batch of H independent matrix multiplications Lᵇʰ @ (Rᵇʰ)ᵀ.
   This step is implemented in a Triton kernel that tiles the (i,j) plane
   and iterates over k.  Each program instance processes a (BLOCK_M×BLOCK_N)
   tile for a single (batch, hidden) pair.
5. Layer‑norm the contraction output, multiply by an output‑gate,
   and project back to the original dimension D with a final linear layer.

The kernel uses float16 for the large tensors (left/right) and accumulates
in float32 for numerical stability.  The rest of the operations are performed
with PyTorch (they are cheap compared to the O(B·H·N³) contraction).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: computes out = left @ rightᵀ for each (batch, hidden) pair.
# left : [B, N, N, H]  (i, k)
# right: [B, N, N, H]  (j, k)
# out  : [B, N, N, H]  (i, j)
# ----------------------------------------------------------------------
@triton.jit
def trimul_contract_kernel(
    left_ptr,               # *fp16
    right_ptr,              # *fp16
    out_ptr,                # *fp32
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)               # i‑tile
    pid_j = tl.program_id(1)               # j‑tile
    pid_task = tl.program_id(2)            # combined (batch, hidden)

    batch = pid_task // H
    hidden = pid_task % H

    # tile start offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load left tile (M x K) ----------
        # left[b, i, k, h]
        left_offset = (
            batch * stride_l_batch
            + hidden * stride_l_h
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)          # (BLOCK_M, BLOCK_K)

        # -------- load right tile (K x N) ----------
        # right[b, j, k, h]   we need (K, N) layout for tl.dot
        right_offset = (
            batch * stride_r_batch
            + hidden * stride_r_h
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        ).to(tl.float32)          # (BLOCK_K, BLOCK_N)

        # -------- matrix multiplication over K ----------
        #   (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(left, right)

    # -------- store result ----------
    out_offset = (
        batch * stride_o_batch
        + hidden * stride_o_h
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
    )
    tl.store(
        out_ptr + out_offset,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul (‘outgoing’) operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, D] (float32)
        - mask          : torch.Tensor[B, N, N]   (bool or float)
        - weights       : dict of model parameters (torch.Tensor)
        - config        : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor[B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    eps = 1e-6

    # ------------------------------------------------------------------
    # Helper: LayerNorm with provided weight/bias
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias):
        return F.layer_norm(x, (x.shape[-1],), weight=weight, bias=bias, eps=eps)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]          # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = layer_norm(input_tensor, norm_weight, norm_bias)  # [B,N,N,D]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape after projection: [B,N,N,hidden_dim]
    left = F.linear(x, left_proj_weight)      # (hidden_dim, dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Apply mask (if present) – mask broadcasted over hidden dimension
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        # No masking – treat as all‑ones
        mask_f = torch.ones_like(x[..., 0], dtype=x.dtype, device=device)  # [B,N,N]
    else:
        mask_f = mask.to(x.dtype)  # ensure float
    mask_f = mask_f.unsqueeze(-1)          # [B,N,N,1]
    left = left * mask_f
    right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) – computed from the *unmasked* x
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 for the heavy contraction
    # ------------------------------------------------------------------
    left_fp16 = left.contiguous().to(torch.float16)
    right_fp16 = right.contiguous().to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Allocate output buffer for the contraction (float32)
    # ------------------------------------------------------------------
    out_contract = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7. Triton kernel launch
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Strides are given in element units (not bytes)
    l_strides = left_fp16.stride()     # (batch, i, k, h)
    r_strides = right_fp16.stride()
    o_strides = out_contract.stride()

    trimul_contract_kernel[grid](
        left_fp16,
        right_fp16,
        out_contract,
        B, N, hidden_dim,
        l_strides[0], l_strides[1], l_strides[2], l_strides[3],
        r_strides[0], r_strides[1], r_strides[2], r_strides[3],
        o_strides[0], o_strides[1], o_strides[2], o_strides[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 8. Post‑contraction processing (LayerNorm, gate, final projection)
    # ------------------------------------------------------------------
    # a) Layer‑norm over hidden dimension
    out_norm = layer_norm(out_contract, to_out_norm_weight, to_out_norm_bias)

    # b) Multiply by the output gate
    out_gated = out_norm * out_gate

    # c) Final linear projection back to dim
    out = F.linear(out_gated, to_out_weight)  # (dim, hidden_dim) weight

    return out
