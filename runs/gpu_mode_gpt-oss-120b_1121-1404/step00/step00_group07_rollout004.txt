def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    # total number of blocks: B * ceil(H/BLOCK_H) * ceil(N/BLOCK_M) * ceil(N/BLOCK_N)
    num_pid_h = tl.cdiv(H, BLOCK_H)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    total_blocks = B * num_pid_h * num_pid_m * num_pid_n
    # guard: ignore threads beyond total_blocks
    if pid >= total_blocks:
        return
    pid_tmp = pid
    b = pid_tmp // (num_pid_h * num_pid_m * num_pid_n)
    pid_tmp = pid_tmp % (num_pid_h * num_pid_m * num_pid_n)
    pid_h = pid_tmp // (num_pid_m * num_pid_n)
    pid_tmp = pid_tmp % (num_pid_m * num_pid_n)
    pid_m = pid_tmp // num_pid_n
    pid_n = pid_tmp % num_pid_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # j indices
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)  # hidden indices

    # mask out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_h = offs_h < H

    # pointers base offset per batch
    # compute base address:
    # left_ptr + b*stride_left_b
    left_base = left_ptr + b * stride_left_b
    right_base = right_ptr + b * stride_right_b
    out_base = out_ptr + b * stride_out_b

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # reduction loop over K dimension
    num_pid_k = tl.cdiv(N, BLOCK_K)
    for pk in range(num_pid_k):
        k_offset = pk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # left tile: shape (M, K, H)
        left_ptrs = left_base \
            + (offs_m[:, None, None] * stride_left_i) \
            + (k_offset[None, :, None] * stride_left_k) \
            + (offs_h[None, None, :] * stride_left_h)
        left = tl.load(left_ptrs, mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

        # right tile: (N, K, H) but we need (K, N, H) for dot. So load as (K, N, H)
        # We'll load right such that rows = K, cols = N.
        right_ptrs = right_base \
            + (k_offset[:, None, None] * stride_right_k) \
            + (offs_n[None, :, None] * stride_right_j) \
            + (offs_h[None, None, :] * stride_right_h)
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :], other=0.0)

        # compute dot product per hidden channel
        # left: (M, K), right: (K, N)
        # Expand hidden as vector dimension: we can use broadcasting
        # Use tl.dot on slices per hidden channel
        # We'll loop over h in BLOCK_H dimension manually (or use vectorized dot with broadcasting?). Let's loop.
        for dh in range(0, BLOCK_H):
            # guard hidden index within mask_h
            if not mask_h[dh]:
                continue
            # left slice MxK, right slice KxN
            left_slice = left[:, :, dh]  # shape (M, K)
            right_slice = right[:, :, dh]  # shape (K, N)
            # left_slice is (M, K), right_slice is (K, N)
            # Use tl.dot
            acc[:, :, dh] += tl.dot(left_slice, right_slice)
        # end hidden loop

    # store results
    out_ptrs = out_base \
        + (offs_m[:, None, None] * stride_out_i) \
        + (offs_n[None, :, None] * stride_out_j) \
        + (offs_h[None, None, :] * stride_out_h)

    tl.store(out_ptrs, acc.to(tl.float16) if tl.constexpr(False) else acc, mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :])

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device
    dtype = x.dtype  # assume float32
    eps = 1e-5

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # also assume biases for linear are none per spec

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,H
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if not nomask
    if mask is not None:
        # mask shape [B, N, N]
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel (optional)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Prepare output tensor (half)
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Launch Triton kernel
    B, N, _, H = left_h.shape

    # Strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_h.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_h.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out_h.stride()

    # Choose block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * num_pid_m * num_pid_n

    grid = (total_programs,)

    trimul_outgoing_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_k, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

    # Convert out back to float32
    out = out_h.to(torch.float32)

    # Apply to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out = out * out_gate

    # Linear to final dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)

    # Compute block indices
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # total programs per batch = H * num_pid_m * num_pid_n
    total_per_batch = H * num_pid_m * num_pid_n
    b = pid // total_per_batch
    pid_rem = pid % total_per_batch
    h = pid_rem // (num_pid_m * num_pid_n)
    pid_mn = pid_rem % (num_pid_m * num_pid_n)
    pid_m = pid_mn // num_pid_n
    pid_n = pid_mn % num_pid_n

    # Offsets for current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this batch & hidden channel
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_pid_k = tl.cdiv(N, BLOCK_K)
    for pk in range(num_pid_k):
        offs_k = pk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: (M, K)
        left_ptrs = left_base \
            + offs_m[:, None] * stride_li \
            + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],

        # right tile: (K, N) – we need the k dimension as rows, j as columns
        right_ptrs = right_base \
            + offs_k[:, None] * stride_rk \
            + offs_n[None, :] * stride_rj
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        + offs_n[None, :] * stride_oj

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    The kernel computes the einsum
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for each batch b and hidden channel h using a tiled
    matrix‑multiply implementation.  All other layers (LayerNorm,
    linear projections, sigmoid gates) are performed with PyTorch
    primitives.
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel: pairwise multiplication over the third axis
    # ------------------------------------------------------------------
    # Cast to half for the kernel – this reduces bandwidth while the
    # accumulator is kept in FP32 inside the kernel.
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape  # H == hidden_dim
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # strides (in elements)
    stride_lb, stride_li, stride_lk, stride_lh = left_h.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_h.stride()

    # tiling parameters (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * num_pid_m * num_pid_n

    grid = (total_programs,)

    trimul_outgoing_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # enough warps for 64×64 tiles on H100
    )

    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output‑norm, outer gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy part of the module is the 4‑D einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is an N×N matrix multiply for every hidden channel d.
We implement this reduction with a tiled GEMM kernel written in Triton.
All other operations (LayerNorm, linear projections, sigmoid gates,
outer‑gate, final projection) are performed with PyTorch functions.
The kernel works on half‑precision tensors for higher memory bandwidth
while accumulating in FP32 to preserve numerical quality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,    # left strides: (b,i,k,d)
    stride_rb, stride_rj, stride_rk, stride_rh,    # right strides: (b,j,k,d)
    stride_ob, stride_oi, stride_oj, stride_oh,    # out strides:   (b,i,j,d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Each program instance processes one tile (BLOCK_M×BLOCK_N) for a
    specific batch element and hidden channel.
    """
    pid = tl.program_id(0)

    # -------------------------------------------------------------
    # Grid decomposition: (b, h, i_tile, j_tile)
    # -------------------------------------------------------------
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    # total programs that belong to a single batch element
    programs_per_batch = H * num_pid_m * num_pid_n

    b = pid // programs_per_batch
    pid_rem = pid % programs_per_batch
    h = pid_rem // (num_pid_m * num_pid_n)
    pid_mn = pid_rem % (num_pid_m * num_pid_n)
    pid_m = pid_mn // num_pid_n
    pid_n = pid_mn % num_pid_n

    # -------------------------------------------------------------
    # Tile offsets
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for the selected batch & hidden channel
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # K‑reduction
    # -------------------------------------------------------------
    num_pid_k = tl.cdiv(N, BLOCK_K)
    for pk in range(num_pid_k):
        offs_k = pk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = left_base \
            + offs_m[:, None] * stride_li \
            + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        right_ptrs = right_base \
            + offs_k[:, None] * stride_rk \
            + offs_n[None, :] * stride_rj
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0)

        acc += tl.dot(left, right)      # (M, N) += (M, K)·(K, N)

    # -------------------------------------------------------------
    out_ptrs = out_base \
        + offs_m[:, None] * stride_oi \
        + offs_n[None, :] * stride_oj
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, dim] (float32)
            - mask : torch.Tensor [B, N, N] (optional; ignored if config["nomask"])
            - weights: dict of model parameters (torch.Tensors)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the feature dimension)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton kernel for the N³ reduction
    # ------------------------------------------------------------------
    # work in half precision for the two large tensors
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, H = left_h.shape               # H == hidden_dim

    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # strides expressed in element counts (as required by Triton)
    stride_lb, stride_li, stride_lk, stride_lh = left_h.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_h.stride()

    # tiling parameters (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * num_pid_m * num_pid_n
    grid = (total_programs,)

    trimul_outgoing_kernel[grid](
        left_h, right_h, out_h,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # enough parallelism for 64×64 tiles on H100
    )

    out = out_h.to(torch.float32)          # back to FP32 for the rest

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, outer gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                   # element‑wise outer gate
    out = F.linear(out, weights["to_out.weight"])  # project back to dim

    return out
