@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_mat = tl.program_id(axis=2)  # enumerates B*H

    # Compute batch and hidden index:
    b = pid_mat // H
    h = pid_mat % H

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Bounds checking:
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Pointers base for this mat:
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base = out_ptr + b * stride_o_b + h * stride_o_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        k_offset = k * BLOCK_K
        cur_k = k_offset + offs_k
        # mask for k_out_of_range
        mask_k = cur_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_m[:, None] * stride_l_i) + (cur_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + (cur_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_i)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Dot product accumulation
        acc += tl.dot(left, right)

    # Apply mask for out-of-bound rows/cols
    out_ptrs = out_base + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    B,N,N,dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute batched matrix-multiplication:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    where left and right have shape (B,N,N,H).
    This kernel tiles the i and j dimensions with BLOCK_M and BLOCK_N,
    and the reduction k dimension with BLOCK_K.
    It launches a 3‑D grid (M_tiles, N_tiles, B*H) where the third axis
    indexes the (batch, hidden) pair.
    """
    pid_m = tl.program_id(axis=0)   # tile along i (rows)
    pid_n = tl.program_id(axis=1)   # tile along j (cols)
    pid_mat = tl.program_id(axis=2) # combined batch/hidden index

    # Decode batch and hidden dimension from pid_mat
    b = pid_mat // H
    h = pid_mat % H

    # Compute absolute indices for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        cur_k = k_base + offs_k
        mask_k = cur_k < N

        # Load left block (M x K)
        # left[i, k, h]
        left_ptrs = left_base + (offs_m[:, None] * stride_l_i) + (cur_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right block transposed (K x N)
        # We need right[j, k, h] but we treat it as (k, j) for the dot.
        right_ptrs = right_base + (cur_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_i)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_base + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """Entry point implementing the forward pass of the TriMul module.

    The heavy pairwise multiplicative reduction is executed by a custom
    Triton kernel (``tri_mul_kernel``). All other linear transforms,
    LayerNorms and sigmoid gates are performed with PyTorch for
    clarity and to keep the Triton kernel focused on the O(N³) work.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    # Shapes
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ---------- LayerNorm on the input ----------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---------- Linear projections ----------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ---------- Optional mask ----------
    if not config.get("nomask", False):
        # mask shape: (B,N,N) -> (B,N,N,1) for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    left_gate  = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Core TriMul (batched matmul) ----------
    # Allocate output tensor for the intermediate hidden representation
    out = torch.empty_like(left)   # shape (B,N,N,hidden_dim)

    # Strides needed by the Triton kernel
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tiling parameters – these work well for the target N ∈ {128…1024}
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j
        B * hidden_dim                  # one program per (batch, hidden)
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ---------- Output LayerNorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)

    # Apply output gate
    out_norm = out_norm * out_gate

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]
    final = F.linear(out_norm, to_out_weight)   # (B,N,N,dim)

    return final

# Triton + PyTorch implementation of the “outgoing” TriMul
# operator (forward‑only). The expensive O(N³) reduction
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# is carried out by a custom Triton kernel; all other linear
# layers, LayerNorms and sigmoid gates are handled by PyTorch.

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute one (batch, hidden) slice of the TriMul reduction:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    left/right have shape (B, N, N, H).  The kernel tiles the
    i‑dimension with BLOCK_M, the j‑dimension with BLOCK_N and
    reduces over k with BLOCK_K.  A 3‑D grid (M‑tiles, N‑tiles,
    B*H) is launched, where the third axis indexes the (batch,
    hidden) pair.
    """
    pid_m   = tl.program_id(axis=0)   # tile along i
    pid_n   = tl.program_id(axis=1)   # tile along j
    pid_mat = tl.program_id(axis=2)   # combined batch/hidden index

    # Decode batch (b) and hidden (h) from the third dimension
    b = pid_mat // H
    h = pid_mat %  H

    # Absolute indices for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑indices
    offs_k = tl.arange(0, BLOCK_K)                    # k‑indices

    # Bounds masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator in FP32 (more accurate & safe)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k in chunks of BLOCK_K
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        cur_k = k_base + offs_k
        mask_k = cur_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        # left[b, i, k, h]
        left_ptrs = left_base + (offs_m[:, None] * stride_l_i) + (cur_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        # We need right[b, j, k, h] but arrange it as (k, j) for the dot.
        right_ptrs = right_base + (cur_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_i)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate dot product over the reduction axis k
        acc += tl.dot(left, right)

    # Store the computed tile
    out_ptrs = out_base + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def custom_kernel(data):
    """
    Forward pass of the TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)                     # (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (if a mask is present)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask: (B,N,N) → (B,N,N,1) for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left  = left  * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul: batched GEMM via Triton
    # ------------------------------------------------------------------
    # Allocate output tensor for the (B,N,N,hidden_dim) intermediate result
    out = torch.empty_like(left)  # same shape and dtype (float32)

    # Strides needed by the kernel (C‑contiguous layout)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – empirically good for N up to 1024 on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # 3‑D grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along j
        B * hidden_dim                  # one program per (batch, hidden)
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm + out‑gate
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out,
                            (hidden_dim,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b,
                            eps=eps)
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    final = F.linear(out_norm, to_out_w)  # (B,N,N,dim)

    return final
