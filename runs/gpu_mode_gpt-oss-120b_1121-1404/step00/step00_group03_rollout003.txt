def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr, EPS: tl.constexpr=1e-5
):
    pid_b = tl.program_id(2)  # batch dimension
    pid_i = tl.program_id(0)  # i tile
    pid_j = tl.program_id(1)  # j tile

    # Compute offsets
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k_offsets = tl.arange(0, BLOCK_K)  # inside loop

    d_offsets = tl.arange(0, BLOCK_D)

    # Mask for i,j within N
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K + k_offsets
        k_mask = k_offset < N

        # Load left block
        left_ptrs = left_ptr + pid_b * stride_left_b + i_offset[:, None, None] * stride_left_i + k_offset[None, :, None] * stride_left_k + d_offsets[None, None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=(i_mask[:, None, None] & k_mask[None, :, None] & (d_offsets[None, None, :] < D)), other=0.0)

        # Load right block

        right = tl.load(right_ptrs, mask=(j_mask[:, None, None] & k_mask[None, :, None] & (d_offsets[None, None, :] < D)), other=0.0)

        # Compute dot product along k for each d
        for d in range(BLOCK_D):

            right_d = right[:, :, d]      # shape [BLOCK_J, BLOCK_K]

            acc[:, :, d] += tl.dot(left_d, tl.trans(right_d))

    out_ptrs = out_ptr + pid_b * stride_out_b + i_offset[:, None, None] * stride_out_i + j_offset[None, :, None] * stride_out_j + d_offsets[None, None, :] * stride_out_d

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor (shape B×N×N×C) using the provided
   `norm.weight` and `norm.bias`.
2. Project the normalized tensor to a hidden dimension `H` with two linear
   layers (`left_proj` and `right_proj`), then apply a per‑channel mask (if
   present) and multiplicative gates (`left_gate`, `right_gate`).
3. The core operation computes, for every batch `b`, pair indices `i`, `j`
   and hidden channel `h`:

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   This is an Einstein‑summation (`... i k d, ... j k d -> ... i j d`).
   It is realised with a custom Triton kernel that tiles the `i` and `j`
   dimensions and reduces over `k`.  The hidden dimension `H` is processed
   in small blocks (`BLOCK_D`) to keep register pressure low.
4. Apply a second LayerNorm (`to_out_norm`) over the hidden dimension,
   multiply by the output gate (`out_gate`) and finally project back to the
   original dimension with `to_out`.
5. Return the result tensor of shape B×N×N×C.

The kernel runs in FP32 (accumulation) but accepts any input dtype.
Only the heavy reduction (step 3) is executed on the GPU via Triton;
all remaining operations are performed with standard PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,           # pointers
    B, N, H,                                 # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time constants
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]."""
    pid_b = tl.program_id(2)                 # batch dimension
    pid_i = tl.program_id(0)                 # tile i
    pid_j = tl.program_id(1)                 # tile j

    # Linear offsets for the i/j tiles
    i_idx = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_idx = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_idx < N
    j_mask = j_idx < N

    # Hidden‑channel offsets
    h_idx = tl.arange(0, BLOCK_H)
    h_mask = h_idx < H

    # Accumulator for the output tile (i,j,h)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for cur_k in range(num_k):
        k_idx = cur_k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_idx < N

        # ------------------- load left[b,i,k,h] -------------------
        off_i = i_idx[:, None, None] * stride_l_i
        off_k = k_idx[None, :, None] * stride_l_k
        off_h = h_idx[None, None, :] * stride_l_h
        left_ptrs = left_ptr + pid_b * stride_l_b + off_i + off_k + off_h
        left = tl.load(
            left_ptrs,
            mask=(i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]),
            other=0.0,
        )  # shape [BI, BK, BH]

        # ------------------- load right[b,j,k,h] -------------------
        off_j = j_idx[:, None, None] * stride_r_j
        off_k_r = k_idx[None, :, None] * stride_r_k
        off_h_r = h_idx[None, None, :] * stride_r_h
        right_ptrs = right_ptr + pid_b * stride_r_b + off_j + off_k_r + off_h_r
        right = tl.load(
            right_ptrs,
            mask=(j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]),
            other=0.0,
        )  # shape [BJ, BK, BH]

        # ------------------- accumulate -------------------------------------------------
        # For each hidden channel block we perform a small GEMM:
        #   left[:, :, h] (BI x BK)  @  right[:, :, h].T (BK x BJ)  ->  (BI x BJ)
        for h in range(BLOCK_H):
            left_h = left[:, :, h]          # (BI, BK)
            right_h = right[:, :, h]        # (BJ, BK)
            # dot expects (M,K) @ (K,N)
            acc[:, :, h] += tl.dot(left_h, tl.trans(right_h))

    # ------------------- store result ----------------------------------------------------
    off_i_o = i_idx[:, None, None] * stride_o_i
    off_j_o = j_idx[None, :, None] * stride_o_j
    off_h_o = h_idx[None, None, :] * stride_o_h
    out_ptrs = out_ptr + pid_b * stride_o_b + off_i_o + off_j_o + off_h_o
    tl.store(
        out_ptrs,
        acc,
        mask=(i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]),
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        torch.Tensor of shape [bs, seqlen, seqlen, dim]
    """
    # -------------------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -------------------------------------------------------------------------
    # Helper: LayerNorm on the input
    # -------------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # -------------------------------------------------------------------------
    # Linear projections (no bias) and gates
    # -------------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])       # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # mask handling
    use_mask = not config.get("nomask", True)
    if use_mask:
        # mask shape: [B,N,N] -> broadcast to hidden dim
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # gates (sigmoid activation)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # Core TriMul reduction via Triton
    # -------------------------------------------------------------------------
    # Allocate output tensor
    out = torch.empty_like(left)  # same dtype, same device, shape [B,N,N,hidden]

    # Determine tiling parameters (tuned for H100, FP32)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 8    # hidden‑channel block size

    # Strides (in element counts)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Grid : (i_tiles, j_tiles, batch)
    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        bs,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=bs,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=2,
    )
    # Ensure kernel completion before proceeding
    torch.cuda.synchronize()

    # -------------------------------------------------------------------------
    # Post‑processing: second LayerNorm, output gate, final projection
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
