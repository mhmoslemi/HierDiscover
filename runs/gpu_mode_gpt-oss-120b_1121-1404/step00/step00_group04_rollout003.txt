@triton.jit
def tr_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_k, stride_right_j,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 1e-5
):
    pid_m = tl.program_id(0)  # block row i
    pid_n = tl.program_id(1)  # block col j
    pid_b = tl.program_id(2)  # batch (B dimension)

    # compute block start indices
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # compute offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointer offsets for left (b,i,k) and right (b,k,j)
        left_ptrs = left_ptr + pid_b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        right_ptrs = right_ptr + pid_b * stride_right_b + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j

        # load with guard
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot and accumulate
        acc += tl.dot(left, right)

        # advance k offsets? Not needed as we recompute each iteration.

    # apply mask to output
    out_ptrs = out_ptr + pid_b * stride_out_b + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # load weights
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if any
    if mask is not None:

        right = right * mask_exp

    B, N,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_k, stride_r_j,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

```python
import torch
import triton
import triton.language as tl

@triton.jit
def tr_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_k, stride_r_j,
    stride_o_b, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication kernel for the TriMul operator.
    Computes:
        out[b, i, j] = sum_k left[b, i, k] * right[b, k, j]
    for each batch (b) which corresponds to original (batch, hidden_dim) pair.
    """
    pid_m = tl.program_id(0)   # block row index (i)
    pid_n = tl.program_id(1)   # block column index (j)
    pid_b = tl.program_id(2)   # batch index (flattened bs*hidden_dim)

    # row and column offsets for the current block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension k in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block: (M, K) -> left[b, i, k]
        left_ptrs = left_ptr + pid_b * stride_l_b + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block: (K, N) -> right[b, k, j]
        right_ptrs = right_ptr + pid_b * stride_r_b + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # store the result
    out_ptrs = out_ptr + pid_b * stride_o_b + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.
    Performs LayerNorm → linear projections + gating → masked outer‑product
    reduction (implemented in a fused Triton kernel) → second LayerNorm →
    gating → final linear projection.
    """
    input_tensor, mask, weights, config = data
    # ----------------------------------------------------------------------
    # config
    # ----------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ----------------------------------------------------------------------
    # unpack weights
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm
    # ----------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [bs, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----------------------------------------------------------------------
    # 3. Optional mask
    # ----------------------------------------------------------------------
    nomask = config.get("nomask", False)
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype=dtype)       # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------------
    # 4. Gating
    # ----------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Batched matrix multiplication via Triton
    #    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ----------------------------------------------------------------------
    # Permute to bring the channel dimension into the batch dimension
    # left : (bs, N, N, hidden) -> (bs, hidden, N, N)  (i,k)
    # right: (bs, N, N, hidden) -> (bs, hidden, N, N)  (k,j) after swapping the last two axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    bs, hd, N, _ = left_perm.shape
    # flatten batch & channel dimension -> B = bs * hidden_dim
    left_flat = left_perm.view(bs * hd, N, N)
    right_flat = right_perm.view(bs * hd, N, N)

    out_flat = torch.empty_like(left_flat)

    # strides in number of elements (Torch returns strides in elements)
    stride_l_b, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_b, stride_r_k, stride_r_j = right_flat.stride()
    stride_o_b, stride_o_i, stride_o_j = out_flat.stride()

    # block sizes – tuned for H100 (fp32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hd,
    )

    tr_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        bs * hd,
        N,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_k,
        stride_r_j,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ----------------------------------------------------------------------
    # 6. Reshape back to original layout
    # ----------------------------------------------------------------------
    out = out_flat.view(bs, hd, N, N).permute(0, 2, 3, 1).contiguous()   # [bs, N, N, hidden]

    # ----------------------------------------------------------------------
    # 7. Second LayerNorm
    # ----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ----------------------------------------------------------------------
    # 8. Output gate and final projection
    # ----------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # [bs, N, N, dim]

    return out

import torch
import triton
import triton.language as tl


@triton.jit
def tr_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_l_b, stride_l_i, stride_l_k,
    stride_r_b, stride_r_k, stride_r_j,
    stride_o_b, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton kernel that performs the core TriMul reduction:

        out[b, i, j] = Σ_k left[b, i, k] * right[b, k, j]

    ``left`` and ``right`` are the already‑projected and gated tensors,
    flattened so that the original batch*hidden_dim dimension is the first
    axis (B).  The kernel processes a (BLOCK_M, BLOCK_N) tile of the output
    for each (i, j) block and iterates over the reduction dimension k in
    BLOCK_K sized chunks.
    """
    pid_m = tl.program_id(0)          # block row   (i)
    pid_n = tl.program_id(1)          # block col   (j)
    pid_b = tl.program_id(2)          # flattened batch (batch * hidden_dim)

    # Offsets for the block we are responsible for
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k]   → (M, K)
        left_ptrs = left_ptr + pid_b * stride_l_b \
                    + offs_i[:, None] * stride_l_i \
                    + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b, k, j]  → (K, N)
        right_ptrs = right_ptr + pid_b * stride_r_b \
                     + offs_k[:, None] * stride_r_k \
                     + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate the matrix product for this tile
        acc += tl.dot(left, right)    # (M,K) @ (K,N) → (M,N)

    # Store the computed tile
    out_ptrs = out_ptr + pid_b * stride_o_b \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator, heavily fused with a
    custom Triton kernel for the N‑by‑N reduction.

    Steps:
    1. Layer‑norm the input.
    2. Linear projections (left/right) and gating linear layers.
    3. Optional mask application.
    4. Element‑wise gating of the projected tensors.
    5. Triton‑accelerated batched matrix multiplication:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
    6. Second Layer‑norm + output gate.
    7. Final linear projection to the original dimensionality.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # input & output channel dimension
    hidden_dim = config["hidden_dim"] # inner channel dimension
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # weight tensors (all assumed to be on the same device as the input)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    dtype = input_tensor.dtype
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. First LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [bs, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Apply optional mask
    # ------------------------------------------------------------------
    if (not nomask) and (mask is not None):
        mask = mask.to(dtype=dtype)               # [bs, N, N]
        mask = mask.unsqueeze(-1)                  # [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton‑fused batched matrix multiplication
    #    Permute to bring the hidden dimension into the batch dimension.
    # ------------------------------------------------------------------
    # left: (bs, N, N, hidden) → (bs, hidden, N, N)  (i,k)
    # right: (bs, N, N, hidden) → (bs, hidden, N, N) (k,j) after swapping axes
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # [bs, hidden, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # [bs, hidden, N, N]

    bs, hd, N, _ = left_perm.shape
    # Fuse batch & hidden dim: B = bs * hidden_dim
    left_flat = left_perm.view(bs * hd, N, N)
    right_flat = right_perm.view(bs * hd, N, N)

    out_flat = torch.empty_like(left_flat)                # [B, N, N]

    # Strides are expressed in element counts (not bytes)
    stride_l_b, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_b, stride_r_k, stride_r_j = right_flat.stride()
    stride_o_b, stride_o_i, stride_o_j = out_flat.stride()

    # Tuned block sizes for the H100 (fp32 precision)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along i
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along j
        bs * hd,                        # fused batch*hidden_dim
    )

    tr_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        bs * hd,
        N,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_k,
        stride_r_j,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to [bs, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hd, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Second LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                     # gating
    out = torch.nn.functional.linear(out, to_out_weight)  # [bs, N, N, dim]

    return out
