"""
TriMul forward (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (`x`) on the last dimension.
2. Linear projections (no bias) produce `left` and `right` tensors of shape
   [B, N, N, H] where H = hidden_dim.
3. (Optional) apply a pair‑wise mask (broadcasted on the channel dim).
4. Compute three sigmoid gates (left‑gate, right‑gate, out‑gate) and
   multiply the gates into `left` and `right`.
   All tensors are cast to float16 before the heavy O(N³·H) step to cut
   memory traffic.
5. The core operation is
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched matrix multiplication:
        out_h = L_h @ R_hᵀ   (L_h,R_h ∈ ℝ^{N×N})
   A Triton kernel (`batched_matmul_kernel`) performs this multiplication
   for all batch‑hidden slices in parallel, using tiling (BLOCK_M/N/K) and
   fp16 arithmetic with fp32 accumulation.
6. Reshape the result back to [B, N, N, H] (float32).
7. Layer‑norm over H, multiply by `out_gate`, and apply the final linear
   projection back to `dim` channels.

The Triton kernel handles the O(N³·H) work, while the remaining
operations (norms, linear layers, gates) are performed with performant
PyTorch primitives.

The implementation supports:
* Variable batch size (1 or 2)
* Sequence lengths up to 1024
* Hidden dimensions 128 / 384 / 768
* Optional mask (`nomask` flag)
* Both normal and heavy‑tailed input distributions
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A, B : [batch, N, K] , K = N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)          # index over (B * hidden_dim)
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block column

    # ------------------ offsets & masks ------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------ reduction over K ------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of A (M × K)
        a_ptrs = (A_ptr + pid_batch * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load a tile of Bᵀ (K × N) – note the swapped strides
        b_ptrs = (B_ptr + pid_batch * stride_b_batch
                  + offs_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Dot product and accumulate (fp16 × fp16 → fp32)
        acc += tl.dot(a, b)

    # ------------------ store result ------------------
    c_ptrs = (C_ptr + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments -------------------------------------------------
    x, mask, weights, config = data
    # Shapes / config --------------------------------------------------
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of left/right
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Cast to fp16 for the massive O(N³·H) computation
    left = (left * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)
    out_gate = out_gate  # stay in float32 for later use

    # -----------------------------------------------------------------
    # 5. Core TriMul: batched matmul using Triton
    # -----------------------------------------------------------------
    # Reshape to (B*hidden, N, N) so each (b,h) pair is an independent batch
    left_view = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_view = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # Allocate output buffer (fp32 accumulation)
    out_view = torch.empty_like(left_view, dtype=torch.float32)

    # Strides (in elements, not bytes)
    stride_a_batch = left_view.stride(0)
    stride_a_m = left_view.stride(1)
    stride_a_k = left_view.stride(2)

    stride_b_batch = right_view.stride(0)
    # For B we need stride over K (the reduction dim) and N (output col dim)
    stride_b_k = right_view.stride(2)   # original innermost dim (K)
    stride_b_n = right_view.stride(1)   # the other N dim

    stride_c_batch = out_view.stride(0)
    stride_c_m = out_view.stride(1)
    stride_c_n = out_view.stride(2)

    # Tiling sizes – chosen to fit H100 shared memory comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    batched_matmul_kernel[grid](
        left_view,
        right_view,
        out_view,
        BATCH=B * hidden_dim,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # typical for fp16 matmul on H100
    )

    # Reshape back to [B, N, N, hidden_dim] (float32)
    out = (
        out_view.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape [B, N, N, hidden]

    # -----------------------------------------------------------------
    # 6. Output layer norm + out_gate
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
