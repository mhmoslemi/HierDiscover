def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_j, stride_l_h,
    stride_r_batch, stride_r_i, stride_r_j, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)  # batch * hidden

    batch_idx = pid_hb // H
    h_idx = pid_hb % H

    # Compute block start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    j_mask = j0 + tl.arange(0, BLOCK_N) < N

    offs_j = j0 + tl.arange(0, BLOCK_N)[None, :]  # shape (1,N)

def custom_kernel(data):
    """
    Triton implementation of the outgoing triangle multiplicative update (TriMul).
    This function performs LayerNorm, linear projections, gating, and the core
    quadratic update using a fused Triton kernel.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Ensure contiguous
    x = input_tensor.contiguous()
    B, N, _, C = x.shape
    assert C == dim

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)      # shape [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if mask is None:
        mask_tensor = torch.ones(B, N, N, dtype=x.dtype, device=x.device)
    else:
        mask_tensor = mask
    if not nomask:
        mask_tensor = mask_tensor.to(x.dtype)
        mask_tensor = mask_tensor.unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask_tensor
        right = right * mask_tensor
    # else if nomask True we skip mask multiplication (already left,right unchanged)

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for faster Triton GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Core cubic update using Triton kernel
    out = trimul_outgoing_kernel_launcher(left, right)   # returns fp32 tensor

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out
```

```
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    # decode batch and hidden index
    batch_idx = pid_hb // H
    hidden_idx = pid_hb % H

    # compute block offsets
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # range of indices
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for i and j within bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(num_k_tiles):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # compute pointers for left: (batch, i, k, h)
        left_ptrs = left_ptr + batch_idx * stride_l_b \
                    + offs_i[:, None, None] * stride_l_i \
                    + offs_k[None, :, None] * stride_l_j \
                    + hidden_idx * stride_l_h

        # compute pointers for right: (batch, j, k, h)
        right_ptrs = right_ptr + batch_idx * stride_r_b \
                     + offs_j[None, :, None] * stride_r_i \
                     + offs_k[None, :, :] * stride_r_j \
                     + hidden_idx * stride_r_h

        # load with mask
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)
        right = tl.load(right_ptrs,
                        mask=mask_j[None, :] & mask_k[:, None],
                        other=0.0,
                        dtype=tl.float16)

        # compute dot product: (M,K) x (K,N) -> (M,N)
        # tl.dot expects shape (M,K) and (K,N)
        acc += tl.dot(left, right).to(tl.float32)

    # store results
    out_ptrs = out_ptr + batch_idx * stride_o_b \
               + offs_i[:, None] * stride_o_i \
               + offs_j[None, :] * stride_o_j \
               + hidden_idx * stride_o_h

    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel
==============

Triton‑accelerated implementation of the *outgoing* Triangle Multiplicative Update
(TriMul) used in AlphaFold‑3 and related protein‑structure models.

The forward pass performs:

1. LayerNorm over the last channel dimension (dim).
2. Two linear projections (left / right) without bias → [B,N,N,hidden_dim].
3. Optional pairwise mask applied to the projections.
4. Gated linear layers (sigmoid) for left, right and output.
5. Core cubic update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for every hidden channel `h`.
   The heavy O(N³·H) work is fused into a Triton kernel that processes one
   (batch, hidden) pair per program and tiles the i/j dimensions.
6. LayerNorm over the hidden dimension, gating by `out_gate`,
   and a final linear projection back to `dim`.

The Triton kernel (`trimul_outgoing_kernel`) operates on `float16` inputs
and accumulates in `float32` for numerical stability.  The surrounding
operations are performed with PyTorch for simplicity.

The implementation supports variable batch size (1‑2), sequence length
(128‑1024), and hidden dimensions (≤384).  It works with or without a
mask (the `nomask` flag in `config`).  The final output tensor is
`float32` with shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: one program per (batch, hidden) pair, tiled i/j.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # ptr to left [B,N,N,H]   (fp16)
    right_ptr,           # ptr to right [B,N,N,H]  (fp16)
    out_ptr,             # ptr to out  [B,N,N,H]   (fp32)
    B, N, H,
    # strides (in elements)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 3‑D grid:
    #   program_id(0) → i‑tile
    #   program_id(1) → j‑tile
    #   program_id(2) → (batch * hidden) linear index
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hb = tl.program_id(2)

    # decode batch and hidden channel from linear index
    batch_idx = pid_hb // H
    hidden_idx = pid_hb % H

    # ------------------------------------------------------------------
    # Tile start indices
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # index vectors within the tile (may exceed N – mask later)
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)

    # predicates for out‑of‑bounds i / j
    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in chunks of BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # -------------------------
        # Load left tile  [M, K]
        #   left[b, i, k, h]
        # -------------------------
        left_ptrs = (
            left_ptr
            + batch_idx * stride_l_b
            + i_idx[:, None] * stride_l_i
            + k_idx[None, :] * stride_l_j
            + hidden_idx * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # -------------------------
        # Load right tile [K, N]
        #   right[b, j, k, h]
        # -------------------------
        right_ptrs = (
            right_ptr
            + batch_idx * stride_r_b
            + j_idx[None, :] * stride_r_i
            + k_idx[:, None] * stride_r_j
            + hidden_idx * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[None, :] & mask_k[:, None],
            other=0.0,
            dtype=tl.float16,
        )

        # -------------------------
        # FP16 matmul, accumulate in FP32
        # -------------------------
        # tl.dot returns FP16 → cast to FP32 before accumulation
        acc += tl.cast(tl.dot(left, right), tl.float32)

    # ------------------------------------------------------------------
    # Store the result tile (only where i and j are in‑range)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_idx * stride_o_b
        + i_idx[:, None] * stride_o_i
        + j_idx[None, :] * stride_o_j
        + hidden_idx * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel.
# ----------------------------------------------------------------------
def _trimul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right : [B, N, N, H]  (fp16)
    returns out   : [B, N, N, H] (fp32)
    """
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), device=left.device, dtype=torch.float32)

    # strides in *elements* (not bytes)
    lsb, lsi, lsj, lsh = left.stride()
    rsb, rsi, rsj, rsh = right.stride()
    osb, osi, osj, osh = out.stride()

    # Tile sizes – chosen to hit the H100 sweet spot
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,                     # one program per (batch, hidden) channel
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        lsb,
        lsi,
        lsj,
        lsh,
        rsb,
        rsi,
        rsj,
        rsh,
        osb,
        osi,
        osj,
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the problem statement.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask         : torch.Tensor [B, N, N]   (bool / uint8 / float)
            - weights      : dict of torch.Tensors containing all model weights
            - config       : dict with keys 'dim', 'hidden_dim', 'nomask' (bool)

    Returns:
        out : torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]    # [H, dim]
    right_proj_weight = weights["right_proj.weight"]  # [H, dim]

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B,N,N] -> [B,N,N,1] for broadcasting
        mask_broadcast = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core cubic update via Triton
    #    Cast to fp16 before the kernel (tensor cores)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    # out_raw has shape [B, N, N, hidden_dim] and dtype float32
    out_raw = _trimul_outgoing_triton(left_fp16, right_fp16)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out_raw,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B,N,N,hidden_dim]

    out = out * out_gate                           # gated output
    to_out_weight = weights["to_out.weight"]       # [dim, hidden_dim]
    out = F.linear(out, to_out_weight)             # back to [B,N,N,dim]

    return out
