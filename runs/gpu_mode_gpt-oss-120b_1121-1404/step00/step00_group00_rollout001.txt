    j_range = j_off + tl.arange(0, BLOCK_J)

TriMul outgoing implementation (forward only)

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections → left_proj, right_proj  (shape [B,N,N,H]).
3. Linear + sigmoid → left_gate, right_gate, out_gate (shape [B,N,N,H]).
4. Fuse mask, left_gate and right_gate into left_proj/right_proj with a small

   separate PyTorch ops.
5. Compute the “triangle‑multiplicative” update:
        out[b,i,j,:] = Σ_k left_proj[b,i,k,:] * right_proj[b,j,k,:]

        out_h = left_h @ right_h^T.

   flattened batch of size B*H.
6. Layer‑norm the intermediate result, multiply by out_gate and apply the
   final linear projection back to the original dimensionality.
All operations are performed in float32 (the input is assumed to be float32;
if it is fp16 the same code works after a cast to float32).  Only the
element‑wise mask‑gate fusion is performed in Triton – the heavy N³ work is
handled by highly‑optimized cuBLAS batched GEMM.

The kernel is deliberately simple: it loads a 3‑D tile (i, j, hidden) of
`left_proj`, `right_proj`, the corresponding gates and the 2‑D mask tile,
applies `left = left * left_gate * mask` and `right = right * right_gate * mask`
in‑place and writes the results back.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: fuse mask + left/right gates into the projected tensors
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_mask_kernel(
    # pointers
    left_ptr, right_ptr,        # [B,N,N,H]
    left_gate_ptr, right_gate_ptr,
    mask_ptr,                   # [B,N,N]
    # sizes
    B, N, H,
    # strides (in elements)
    stride_lb, stride_li, stride_lj, stride_lh,
    stride_rb, stride_ri, stride_rj, stride_rh,
    stride_lgpb, stride_lgpi, stride_lgpj, stride_lgph,
    stride_rgpb, stride_rgpi, stride_rgpj, stride_rgph,
    stride_mb, stride_mi, stride_mj,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    For each (batch, i, j) tile, multiply:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
    The mask is broadcast over the hidden dimension.
    """

    batch_id = tl.program_id(0)
    i_tile_id = tl.program_id(1)
    j_tile_id = tl.program_id(2)

    i_start = i_tile_id * BLOCK_I
    j_start = j_tile_id * BLOCK_J

    # 1‑D indices for the tile
    i_idx = i_start + tl.arange(0, BLOCK_I)
    j_idx = j_start + tl.arange(0, BLOCK_J)

    # OOB masks for i and j
    i_ok = i_idx < N
    j_ok = j_idx < N
    ij_ok = i_ok[:, None] & j_ok[None, :]       # (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # Load the 2‑D mask tile (float32)
    # ------------------------------------------------------------------
    mask_offset = (batch_id * stride_mb
                 + i_idx[:, None] * stride_mi
                 + j_idx[None, :] * stride_mj)
    mask_tile = tl.load(mask_ptr + mask_offset,
                        mask=ij_ok,
                        other=0.0)            # shape (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # Loop over the hidden dimension in BLOCK_H sized chunks
    # ------------------------------------------------------------------
    for h_start in range(0, H, BLOCK_H):
        h_idx = h_start + tl.arange(0, BLOCK_H)
        h_ok = h_idx < H

        # broadcast i/j/h to a 3‑D tile
        i_tile = i_idx[:, None, None]          # (BLOCK_I,1,1)
        j_tile = j_idx[None, :, None]          # (1,BLOCK_J,1)
        h_tile = h_idx[None, None, :]          # (1,1,BLOCK_H)

        # Build offsets for left, right and the gates
        left_off = (batch_id * stride_lb
                  + i_tile * stride_li
                  + j_tile * stride_lj
                  + h_tile * stride_lh)

        right_off = (batch_id * stride_rb
                   + i_tile * stride_ri
                   + j_tile * stride_rj
                   + h_tile * stride_rh)

        left_gate_off = (batch_id * stride_lgpb
                       + i_tile * stride_lgpi
                       + j_tile * stride_lgpj
                       + h_tile * stride_lgph)

        right_gate_off = (batch_id * stride_rgpb
                        + i_tile * stride_rgpi
                        + j_tile * stride_rgpj
                        + h_tile * stride_rgph)

        # Combined validity mask (i,j,h)
        valid = ij_ok[:, :, None] & h_ok[None, None, :]

        # Load tensors
        left_val      = tl.load(left_ptr      + left_off,
                                 mask=valid,
                                 other=0.0)
        right_val     = tl.load(right_ptr     + right_off,
                                 mask=valid,
                                 other=0.0)
        left_gate_val = tl.load(left_gate_ptr + left_gate_off,
                                 mask=valid,
                                 other=0.0)
        right_gate_val = tl.load(right_gate_ptr + right_gate_off,
                                 mask=valid,
                                 other=0.0)

        # Broadcast mask to (i,j,1) and apply
        mask_exp = mask_tile[:, :, None]                      # (BLOCK_I,BLOCK_J,1)
        left_res  = left_val  * left_gate_val  * mask_exp
        right_res = right_val * right_gate_val * mask_exp

        # Store the results back in‑place
        tl.store(left_ptr  + left_off,  left_res,  mask=valid)
        tl.store(right_ptr + right_off, right_res, mask=valid)


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (C == config["dim"])
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : mapping of parameter names to torch.Tensors
        - config_dict  : contains "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]        # = H in the paper

    # --------------------------------------------------------------
    # 1) Input layer‑norm
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Gating values (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------
    # 4) Prepare mask (float, same dtype as tensors)
    # --------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(input_tensor.shape[:3], dtype=input_tensor.dtype,
                          device=input_tensor.device)
    else:
        mask = mask_tensor.to(dtype=input_tensor.dtype)
    # Ensure all tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    mask = mask.contiguous()

    B, N, _, _ = left.shape
    H = hidden_dim

    # --------------------------------------------------------------
    # 5) Fuse mask + gates into left / right using Triton
    # --------------------------------------------------------------
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_H = 32

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )
    _apply_gate_mask_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask,
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H,
    )
    # At this point: left = left_proj * left_gate * mask
    #                right = right_proj * right_gate * mask

    # --------------------------------------------------------------
    # 6) Triangle‑multiplicative update:
    #    out_h = left_h @ right_hᵀ   for every hidden channel h
    # --------------------------------------------------------------
    # Re‑arrange to (B, H, N, N) to use torch.bmm
    left_perm = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H, N, N)
    right_perm = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # (B*H, N, N)

    out_flat = torch.bmm(left_perm, right_perm.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)           # (B, N, N, H)

    # --------------------------------------------------------------
    # 7) Output layer‑norm, out‑gate, final linear projection
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
