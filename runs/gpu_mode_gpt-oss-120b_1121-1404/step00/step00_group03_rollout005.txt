def mask_gate_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr, HIDDEN: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute offsets
    i = pid_i
    j = pid_j

    # mask broadcast
    mask = tl.load(mask_ptr + pid_b * stride_mask_b + i * stride_mask_i + j * stride_mask_j, mask=(i < N) & (j < N))
    # mask has shape ()
    # load left vector
    offs_d = tl.arange(0, HIDDEN)
    left_offset = pid_b * stride_left_b + i * stride_left_i + j * stride_left_j + offs_d * stride_left_h

"""
custom_kernel
==============

Implements the **outgoing** TriMul update used in AlphaFold‑3 (and similar
protein‑structure models).  The kernel fuses the three elementwise steps
(masking, left/right gating) into a single Triton kernel, while the
computationally‑intensive ternary contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

is carried out with a highly‑optimised batched GEMM (torch.bmm) after
reshaping the tensors.  This gives a fast O(N³·C) implementation that
fits comfortably into the memory bandwidth and compute budget of an H100.

The function accepts a tuple ``(input, mask, weights, config)`` and
returns the processed tensor of shape ``[B, N, N, dim]``.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask * left_gate * left   and   mask * right_gate * right
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    # strides for left/left_gate/right (all have the same layout)
    stride_b, stride_i, stride_j, stride_h,
    # strides for mask (3‑D tensor)
    stride_mask_b, stride_mask_i, stride_mask_j,
    HIDDEN: tl.constexpr
):
    pid_b = tl.program_id(0)          # batch index
    pid_i = tl.program_id(1)          # row index i
    pid_j = tl.program_id(2)          # column index j

    i = pid_i
    j = pid_j

    # ------------------------------------------------------------------
    # bounds
    # ------------------------------------------------------------------
    valid = (i < N) & (j < N) & (pid_b < B)

    # ------------------------------------------------------------------
    # load mask scalar (broadcast to hidden dim later)
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr + pid_b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
        mask=valid,
        other=0.0
    )

    # ------------------------------------------------------------------
    # vector offset for the hidden dimension
    # ------------------------------------------------------------------
    d = tl.arange(0, HIDDEN)          # compile‑time length = hidden_dim

    # offsets for the 4‑D tensors (all strides are identical for left/right)
    base = pid_b * stride_b + i * stride_i + j * stride_j
    vec_off = base + d * stride_h       # shape (HIDDEN,)

    # ------------------------------------------------------------------
    # load data
    # ------------------------------------------------------------------
    left      = tl.load(left_ptr      + vec_off, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + vec_off, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + vec_off, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + vec_off, mask=valid, other=0.0)

    # ------------------------------------------------------------------
    # apply mask and gates
    # ------------------------------------------------------------------
    left_out  = left  * left_gate  * mask
    right_out = right * right_gate * mask

    # ------------------------------------------------------------------
    # write results back
    # ------------------------------------------------------------------
    tl.store(out_left_ptr  + vec_off, left_out,  mask=valid)
    tl.store(out_right_ptr + vec_off, right_out, mask=valid)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor:  torch.Tensor of shape [B, N, N] (or None)
            - weights_dict: mapping from module name to torch.Tensor
            - config_dict : must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (equivalent to self.norm)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gating logits + sigmoid
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Prepare mask (if config says no mask we treat it as all‑ones)
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask_tensor is None:
        mask = torch.ones((B, N, N), device=device, dtype=dtype)
    else:
        mask = mask_tensor.to(dtype)

    # ------------------------------------------------------------------
    # Allocate tensors for the masked & gated values
    # ------------------------------------------------------------------
    left_fused  = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # ------------------------------------------------------------------
    # Launch Triton kernel that does:
    #   left_fused  = left  * left_gate  * mask
    #   right_fused = right * right_gate * mask
    # ------------------------------------------------------------------
    stride_b, stride_i, stride_j, stride_h = left.stride()          # same for left/right
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

    # grid: (batch, row, col)
    grid = (B, N, N)
    _mask_gate_fuse_kernel[grid](
        left, left_gate, mask,
        right, right_gate,
        left_fused, right_fused,
        B, N, hidden_dim,
        stride_b, stride_i, stride_j, stride_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        HIDDEN=hidden_dim,
    )
    # ------------------------------------------------------------------
    # Core contraction: out = Σ_k left_fused[...,k,:] * right_fused[...,k,:]
    # Implemented as a batched GEMM after reshaping.
    #   left  : (B, N, N, H) -> (B*H, N, N)
    #   right : (B, N, N, H) -> (B*H, N, N)
    #   out   : (B, N, N, H)
    # ------------------------------------------------------------------
    B_, N_, _, H_ = left_fused.shape

    # reshape to (B*H, N, N) where each hidden slice is treated as a batch item
    left_flat  = left_fused.permute(0, 3, 1, 2).reshape(B_ * H_, N_, N_)
    right_flat = right_fused.permute(0, 3, 1, 2).reshape(B_ * H_, N_, N_)

    # batched matrix multiplication (i‑k) × (j‑k)ᵀ  =>  (i‑j) per hidden slice
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.reshape(B_, H_, N_, N_).permute(0, 2, 3, 1)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # Output LayerNorm (self.to_out_norm)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # ------------------------------------------------------------------
    # Apply out‑gate (elementwise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to dim (self.to_out)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
```
