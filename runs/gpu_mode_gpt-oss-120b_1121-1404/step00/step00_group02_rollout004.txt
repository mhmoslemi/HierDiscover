def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(0)  # batch*hidden
    i_tile = tl.program_id(1)
    j_tile = tl.program_id(2)
    
    # Compute batch and head index
    b = bh // hidden_dim
    d = bh % hidden_dim
    
    # Compute global row/col start offsets
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Mask for boundaries
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # Create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # iterate over k dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N
        
        # load A submatrix (M x K)
        a_ptrs = left_ptr + b*stride_l_b + d*stride_l_d + (offs_i[:, None] * stride_l_i) + (cur_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)
        
        # load B submatrix (K x N) where B corresponds to right[b, j, k, d] -> we need shape (K,N)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # Extract weights
    norm_weight = weights["norm.weight"].to(device)
    norm_bias = weights["norm.bias"].to(device)

    left_proj_weight = weights["left_proj.weight"].to(device).half()
    right_proj_weight = weights["right_proj.weight"].to(device).half()
    left_gate_weight = weights["left_gate.weight"].to(device).half()
    right_gate_weight = weights["right_gate.weight"].to(device).half()
    out_gate_weight = weights["out_gate.weight"].to(device).half()
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias = weights["to_out_norm.bias"].to(device)
    to_out_weight = weights["to_out.weight"].to(device)

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=eps)

    # Cast to half for projection and gates
    x_half = x.half()
    # Linear projections
    left = torch.nn.functional.linear(x_half, left_proj_weight)  # [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x_half, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_half, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_half, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_half, out_gate_weight))

    # Apply mask if present
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute triangular multiplication using Triton
    out = _tri_mul_outgoing(left, right)

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=eps)

    # Multiply by out_gate (cast to float32)
    out = out * out_gate.float()

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul “outgoing” implementation (AlphaFold‑3 style) using Triton.

The forward pass consists of:
1️⃣ Layer‑norm on the input tensor (shape [B,N,N,D]).
2️⃣ Linear projections → left, right (half‑precision) and three sigmoid‑gates.
3️⃣ Optional masking of left/right.
4️⃣ Element‑wise gating of left/right.
5️⃣ A batched “Gram‑style” multiplication:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]          (h = hidden dim)
   This O(N³·H) kernel is written in Triton and fused over the batch×head axis.
6️⃣ Layer‑norm on the hidden dimension, multiplication with the output gate,
   and a final linear projection back to D.

All heavy compute (step 5) runs on the GPU via a custom Triton kernel, while
the surrounding linear/normalisation steps use fast PyTorch operators.
The kernel works for any batch (B∈{1,2}), sequence length N (≤1024) and
hidden dimension H (≤384).  The implementation uses half‑precision for the
large tensors to reduce memory bandwidth, accumulates in FP32 for numeric
stability and returns a FP32 result.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication over the k‑dimension.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,                     # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program_id(0) – combined batch × hidden index
    bh = tl.program_id(0)
    pid_i = tl.program_id(1)   # tile index along the first N axis
    pid_j = tl.program_id(2)   # tile index along the second N axis

    # Decode batch and head from the combined axis
    b = bh // H
    h = bh % H

    # Global start coordinates for this tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # Boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- Load A = left[b, i, k, h] (M×K) ----------
        a_ptrs = (left_ptr
                  + b * stride_l_b
                  + h * stride_l_h
                  + offs_i[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----------- Load B = right[b, j, k, h] as (K×N) ----------
        # Note: we need the k‑dimension first, then the j‑dimension.
        b_ptrs = (right_ptr
                  + b * stride_r_b
                  + h * stride_r_h
                  + offs_k[:, None] * stride_r_k
                  + offs_j[None, :] * stride_r_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)      # (BLOCK_M×BLOCK_K)·(BLOCK_K×BLOCK_N)

    # ----------- Write back the tile ----------
    c_ptrs = (out_ptr
              + b * stride_o_b
              + h * stride_o_h
              + offs_i[:, None] * stride_o_i
              + offs_j[None, :] * stride_o_j)
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _tri_mul_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched Gram‑style multiplication implemented in Triton.
    left/right : [B, N, N, H] (half‑precision)
    Returns out : [B, N, N, H] (float‑32)
    """
    B, N, N2, H = left.shape
    assert N == N2, "left/right must be square in the two middle dimensions"
    left = left.contiguous()
    right = right.contiguous()

    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Strides (in elements, not bytes)
    ls = left.stride()
    rs = right.stride()
    os = out.stride()

    # Tile sizes – empirically good for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch×head, rows tiles, cols tiles)
    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        ls[0], ls[1], ls[2], ls[3],
        rs[0], rs[1], rs[2], rs[3],
        os[0], os[1], os[2], os[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data: tuple

    """

    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model parameters (move to correct device & dtype)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(device)
    norm_bias   = weights["norm.bias"].to(device)

    # Half‑precision for the large linear layers
    left_proj_weight  = weights["left_proj.weight"].to(device).half()
    right_proj_weight = weights["right_proj.weight"].to(device).half()
    left_gate_weight  = weights["left_gate.weight"].to(device).half()
    right_gate_weight = weights["right_gate.weight"].to(device).half()
    out_gate_weight   = weights["out_gate.weight"].to(device).half()

    # FP32 for the small post‑norm & output projection
    to_out_norm_weight = weights["to_out_norm.weight"].to(device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(device)
    to_out_weight      = weights["to_out.weight"].to(device)   # (D, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (FP32)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2‑4. Linear projections + sigmoid gates (half‑precision)
    # ------------------------------------------------------------------
    x_h = x.half()                            # [B,N,N,D] in FP16

    left = torch.nn.functional.linear(x_h, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_h, right_proj_weight)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_weight))  # FP16

    # ------------------------------------------------------------------
    # 5. Optional mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6. Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Core cubic multiplication (Triton)
    # ------------------------------------------------------------------
    out_hidden = _tri_mul_outgoing(left, right)      # [B,N,N,H] in FP32

    # ------------------------------------------------------------------
    # 8. Post‑norm, output gate and final projection (all FP32)
    # ------------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    out_hidden = out_hidden * out_gate.float()       # broadcast over hidden dim
    out = torch.nn.functional.linear(out_hidden, to_out_weight)  # → [B,N,N,D]

    return out
