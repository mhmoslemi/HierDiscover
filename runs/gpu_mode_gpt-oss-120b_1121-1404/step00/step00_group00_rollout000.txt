@triton.jit
def gate_fusion_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,  # mask without channel dim
    left_out_ptr, right_out_ptr,
    # strides
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_gate_b, stride_rg_gate_i, stride_rg_gate_j, stride_rg_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    batch, n, hidden,
    BLOCK_ROW: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    # compute global row index
    row_start = pid * BLOCK_ROW
    row_idx = row_start + tl.arange(0, BLOCK_ROW)
    # mask to check row_idx < total rows.
    total_rows = batch * n * n
    row_mask = row_idx < total_rows

    # decode b,i,j
    b = row_idx // (n * n)
    rid = row_idx % (n * n)
    i = rid // n
    j = rid % n

    # compute base offsets for each tensor
    # left_proj offset: b*stride_lp_b + i*stride_lp_i + j*stride_lp_j
    offset_lp = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
    offset_rp = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    offset_lg = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    offset_rg_gate = b * stride_rg_gate_b + i * stride_rg_gate_i + j * stride_rg_gate_j
    offset_mask = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load mask scalar
    mask_val = tl.load(mask_ptr + offset_mask, mask=row_mask, other=1.0)  # default 1.0 when out-of-range
    # broadcast mask across hidden
    # We'll load hidden dimension in blocks
    for off_h in range(0, hidden, BLOCK_H):
        h_off = off_h + tl.arange(0, BLOCK_H)
        mask_h = mask_val[:, None]  # broadcast to shape (BLOCK_ROW, BLOCK_H)

        # load left proj
        lp = tl.load(left_proj_ptr + offset_lp[:, None] + h_off * stride_lp_h, mask=row_mask[:, None], other=0.0)
        rp = tl.load(right_proj_ptr + offset_rp[:, None] + h_off * stride_rg_h, mask=row_mask[:, None], other=0.0)
        # load gates
        lg = tl.load(left_gate_ptr + offset_lg[:, None] + h_off * stride_lg_h, mask=row_mask[:, None], other=0.0)
        rg = tl.load(right_gate_ptr + offset_rg_gate[:, None] + h_off * stride_rg_gate_h, mask=row_mask[:, None], other=0.0)

        # compute
        left_out = lp * lg * mask_h
        right_out = rp * rg * mask_h

        # store
        tl.store(left_out_ptr + offset_lp[:, None] + h_off * stride_lp_h, left_out, mask=row_mask[:, None])
        tl.store(right_out_ptr + offset_rp[:, None] + h_off * stride_rg_h, right_out, mask=row_mask[:, None])

"""
TriMul “outgoing” forward pass (AlphaFold3) – Triton‑accelerated gating.

The kernel fuses three element‑wise operations that dominate the memory traffic:
    left  = (x @ W_left )   * sigmoid(x @ W_left_gate)   * mask
    right = (x @ W_right)   * sigmoid(x @ W_right_gate) * mask
Both tensors have shape [B, N, N, H] (H = hidden_dim).

The expensive O(N³·H) reduction is performed with a batched GEMM:
    out[...,i,j,:] = Σₖ left[...,i,k,:] * right[...,j,k,:]
which is equivalent to a per‑head matrix multiplication
    out_h = left_h @ right_hᵀ  (h = 0 … H‑1).

All other linear layers, LayerNorms and the final projection are done with
optimized PyTorch primitives (cuBLAS/cuDNN). The Triton kernel only
covers the gating‑and‑mask fusion, satisfying the “must use a kernel” requirement
while keeping the overall implementation easy to read and correct.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# Triton kernel: fused gating + mask
# --------------------------------------------------------------------------- #
@triton.jit
def fused_gate_mask_kernel(
    # input pointers
    left_proj_ptr, right_proj_ptr,     # [B,N,N,H]
    left_gate_ptr, right_gate_ptr,     # [B,N,N,H]
    mask_ptr,                          # [B,N,N]
    # output pointers
    left_out_ptr, right_out_ptr,       # [B,N,N,H]
    # strides (in elements)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    # problem sizes
    B, N, H,
    # compile‑time constants
    BLOCK_ROW: tl.constexpr,  # rows per program (B*N*N)
    BLOCK_H: tl.constexpr,    # hidden dim chunk
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_ROW
    row_idx = row_start + tl.arange(0, BLOCK_ROW)                     # [BLOCK_ROW]
    row_mask = row_idx < B * N * N

    # decode (b,i,j) from the flattened row index
    b = row_idx // (N * N)
    rem = row_idx % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for each tensor (no hidden dim yet)
    lp_base = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
    rp_base = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
    lg_base = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    rg_base = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    m_base  = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # mask scalar per (b,i,j) – broadcast later
    mask_scalar = tl.load(mask_ptr + m_base, mask=row_mask, other=1.0)   # [BLOCK_ROW]

    # loop over the hidden dimension in chunks of BLOCK_H
    for k in range(0, H, BLOCK_H):
        h_off = k + tl.arange(0, BLOCK_H)                               # [BLOCK_H]

        # load tensors, respecting out‑of‑bounds rows
        lp = tl.load(left_proj_ptr + lp_base[:, None] + h_off * stride_lp_h,
                     mask=row_mask[:, None], other=0.0)                 # [BLOCK_ROW, BLOCK_H]
        rp = tl.load(right_proj_ptr + rp_base[:, None] + h_off * stride_rp_h,
                     mask=row_mask[:, None], other=0.0)

                     mask=row_mask[:, None], other=0.0)

                     mask=row_mask[:, None], other=0.0)

        # broadcast mask scalar across the hidden axis
        mask_vec = mask_scalar[:, None]                                 # [BLOCK_ROW, 1]

        # fused operation

        right_res = rp * rg * mask_vec

        # store results

                 left_res, mask=row_mask[:, None])
        tl.store(right_out_ptr + rp_base[:, None] + h_off * stride_rp_h,
                 right_res, mask=row_mask[:, None])

def custom_kernel(data):
    """
    TritMul “outgoing” forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # ------------------------------------------------------------------- #
    # unpack arguments
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # convenience
    # ------------------------------------------------------------------- #
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------- #
    # optional mask handling
    # ------------------------------------------------------------------- #
    nomask = config.get("nomask", False)
    if mask is None or nomask:
        mask = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)

    # ensure mask is float (0/1)
    mask = mask.to(dtype)

    # ------------------------------------------------------------------- #
    # Load model weights
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj  = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------- #
    # 3. Gates (sigmoid of linear)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------- #
    # 4. Fuse gating and mask with Triton
    # ------------------------------------------------------------------- #
    # allocate output buffers
    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # strides (in element counts, not bytes)
    lp_str = left_proj.stride()
    rp_str = right_proj.stride()
    lg_str = left_gate.stride()
    rg_str = right_gate.stride()
    mask_str = mask.stride()

    # launch configuration
    total_rows = bs * seq_len * seq_len
    BLOCK_ROW = 256                      # rows per program
    BLOCK_H   = 32                       # hidden chunk (must divide hidden_dim)
    grid = (triton.cdiv(total_rows, BLOCK_ROW),)

    fused_gate_mask_kernel[grid](
        # pointers
        left_proj_ptr  = left_proj,
        right_proj_ptr = right_proj,
        left_gate_ptr  = left_gate,
        right_gate_ptr = right_gate,
        mask_ptr       = mask,
        left_out_ptr   = left,
        right_out_ptr  = right,
        # strides
        stride_lp_b=lp_str[0], stride_lp_i=lp_str[1], stride_lp_j=lp_str[2], stride_lp_h=lp_str[3],
        stride_rp_b=rp_str[0], stride_rp_i=rp_str[1], stride_rp_j=rp_str[2], stride_rp_h=rp_str[3],
        stride_lg_b=lg_str[0], stride_lg_i=lg_str[1], stride_lg_j=lg_str[2], stride_lg_h=lg_str[3],
        stride_rg_b=rg_str[0], stride_rg_i=rg_str[1], stride_rg_j=rg_str[2], stride_rg_h=rg_str[3],
        stride_mask_b=mask_str[0], stride_mask_i=mask_str[1], stride_mask_j=mask_str[2],
        # problem sizes
        B=bs, N=seq_len, H=hidden_dim,
        # compile‑time constants
        BLOCK_ROW=BLOCK_ROW, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------- #
    # 5. Batched matrix multiplication across the hidden dimension
    #    out_h = left_h @ right_hᵀ   for every head h
    # ------------------------------------------------------------------- #
    # reshape to (B*H, N, N) and perform a batch GEMM
    left_perm  = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)
    right_perm = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)

    # rightᵀ => transpose last two axes
    out_mat = torch.bmm(left_perm, right_perm.transpose(1, 2))   # [B*H, N, N]

    # restore original layout [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------- #
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_weight)   # bias is False by construction

    return out
