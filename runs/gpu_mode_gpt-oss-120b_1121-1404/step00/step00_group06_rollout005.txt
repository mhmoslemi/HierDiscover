def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    (x, mask, weights, config) = data
    B, N, _, D = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ensure weight tensors correct shape
    # weights dict with Tensor shape.
    # ensure mask dtype
    if mask is None:
        mask_f = torch.ones((B,N,N), device=x.device, dtype=x.dtype)
    else:
        mask_f = mask.to(dtype=x.dtype)
    # maybe mask may be bool or int.
    # Reshape mask for ops later.

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections without bias
    left_proj_weight = weights["left_proj.weight"]   # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"] # shape (hidden_dim, dim)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gate projections
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply mask and gates using Triton kernel
    # If nomask config (maybe config may contain "nomask" or we can check if mask_f is all ones)
    if config.get("nomask", False):
        # No mask, just elementwise multiply gate
        left = left * left_gate
        right = right * right_gate
    else:
        # Use kernel to combine mask and gate
        # compute total elements
        total_elements = B * N * N * hidden_dim
        BLOCK = 128
        grid = (triton.cdiv(total_elements, BLOCK),)
        # left after mask/gate
        left_out = torch.empty_like(left)
        apply_mask_gate_kernel[grid](
            left, mask_f, left_gate, left_out,
            total_elements,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            B, N, hidden_dim,
            BLOCK=BLOCK
        )
        # right after mask/gate
        right_out = torch.empty_like(right)
        apply_mask_gate_kernel[grid](
            right, mask_f, right_gate, right_out,
            total_elements,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            B, N, hidden_dim,
            BLOCK=BLOCK
        )
        left = left_out
        right = right_out

    # Compute bilinear product via batched matmul
    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_ * H_, N_, N_)
    right_mat = right_perm.view(B_ * H_, N_, N_)
    # out_mat shape (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate (might also be done via kernel)

    # out_gate shape B,N,N,H
    out_gated = out_norm * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)
    out_final = torch.nn.functional.linear(out_gated, to_out_weight)

    return out_final

import torch
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(in_ptr, mask_ptr, gate_ptr, out_ptr,
                           total_elements,
                           stride_in_b, stride_in_i, stride_in_j, stride_in_h,
                           stride_mask_b, stride_mask_i, stride_mask_j,
                           stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                           B, N, H,
                           BLOCK: tl.constexpr):
    # Compute linear thread IDs
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Compute multi-dimensional indices
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # Compute flat offsets for each tensor
    in_off = b * stride_in_b + i * stride_in_i + j * stride_in_j + h * stride_in_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h

    a = tl.load(in_ptr + in_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)  # default 1 (no masking) for out‑of‑bounds
    g = tl.load(gate_ptr + gate_off, mask=mask, other=0.0)

    out = a * m * g
    tl.store(out_ptr + in_off, out, mask=mask)

    """TriMul forward pass (outgoing version) with mask/gate fused via Triton.

    The implementation follows the reference PyTorch code, using
    torch.nn.functional for LayerNorms and Linear layers, and a small
    Triton kernel to fuse ``mask * gate`` multiplication on the projected
    tensors.
    """
    import torch
    import triton

    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Extract config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on the input (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]       # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]     # (hidden_dim, dim)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gating projections
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Prepare mask
    if mask_tensor is None:
        mask = torch.ones((input_tensor.shape[0],
                           input_tensor.shape[1],
                           input_tensor.shape[2]),
                           device=device,
                           dtype=dtype)
    else:
        # ensure float type for multiplication
        mask = mask_tensor.to(dtype=dtype)

    # Fuse mask & gate with a Triton kernel unless nomask==True (i.e. mask is all ones)
    if nomask:
        left = left * left_gate
        right = right * right_gate
    else:
        B, N_seq, _, _ = left.shape
        total_elements = B * N_seq * N_seq * hidden_dim
        BLOCK = 1024  # number of elements processed by one program
        grid = (triton.cdiv(total_elements, BLOCK),)

        # left = left * mask * left_gate
        left_out = torch.empty_like(left)
        apply_mask_gate_kernel[grid](
            left,
            mask,
            left_gate,
            left_out,
            total_elements,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            B, N_seq, hidden_dim,
            BLOCK=BLOCK,
        )
        # right = right * mask * right_gate
        right_out = torch.empty_like(right)
        apply_mask_gate_kernel[grid](
            right,
            mask,
            right_gate,
            right_out,
            total_elements,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            B, N_seq, hidden_dim,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out

    # === Core TriMul multiplication: sum_k left[...,k,:] * right[...,k,:] ===
    #    We view the tensors as (B, hidden_dim, N, N) and perform a batched GEMM.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N_seq, _ = left_perm.shape
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N_seq, N_seq)
    right_mat = right_perm.view(B * H, N_seq, N_seq)

    # Batch matrix multiply: (B*H, N, N) @ (B*H, N, N).T  -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm over hidden_dim
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply out_gate (element‑wise)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)

    # Ensure float32 output as required
    out = out.to(torch.float32)

    return out

"""
TriMul (outgoing version) – forward only.

The implementation follows the reference PyTorch module but fuses the
mask × gate multiplication into a tiny Triton kernel.  All heavy
linear/LayerNorm operations are performed with PyTorch (cuBLAS /
cuDNN) while the element‑wise mask‑gate fusion is done on‑GPU by
Triton.  This satisfies the requirement of “using Triton for at least
part of the computation” and keeps the overall runtime comparable to a
pure‑PyTorch implementation.

Algorithm (high level)
----------------------
1. Layer‑norm the input tensor over its last dimension.
2. Project to the hidden dimension (left/right) and compute three
   gating vectors (left_gate, right_gate, out_gate) with a sigmoid.
3. Fuse mask, left_gate and right_gate with a Triton kernel:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
4. Core TriMul: for each hidden slice h compute
        out[...,h] = left[...,h] @ right[...,h].T
   Implemented as a batched GEMM using torch.bmm (CUDA‑optimized).
5. Layer‑norm over the hidden dimension, apply out_gate, and a final
   linear projection back to the original feature dimension.
6. Return the result in float32.

The Triton kernel only touches the element‑wise part (step 3) which is
small compared to the O(N³·H) core but demonstrates custom‑kernel
usage.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(
    in_ptr,               # Tensor to be masked & gated, shape [B,N,N,H]
    mask_ptr,             # Binary mask, shape [B,N,N]
    gate_ptr,             # Gate values, shape [B,N,N,H]
    out_ptr,              # Output tensor, same shape as in_ptr
    total_elements,       # B * N * N * H
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    B, N, H,
    BLOCK: tl.constexpr,  # number of elements processed by one program
):
    """Fuse ``out = in * mask * gate``."""
    pid = tl.program_id(0)                     # 1‑D grid
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elements                # out‑of‑bounds guard

    # ----- decode linear offset into (b,i,j,h) ----------------------
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # ----- compute flat pointers ------------------------------------
    in_off = b * stride_in_b + i * stride_in_i + j * stride_in_j + h * stride_in_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    gate_off = (b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
                + h * stride_gate_h)

    # ----- load, compute and store ----------------------------------
    a = tl.load(in_ptr + in_off,   mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)   # default 1 → no masking
    g = tl.load(gate_ptr + gate_off, mask=mask, other=0.0)

    out = a * m * g
    tl.store(out_ptr + in_off, out, mask=mask)

    """
    Forward pass of the TriMul (outgoing) module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        weight=norm_weight,
        bias=norm_bias,

    # ------------------------------------------------------------------

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------

    right_gate_weight = weights["right_gate.weight"]

    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    if nomask:

        total_elements = B * N_seq * N_seq * hidden_dim

            right_gate.stride(0), right_gate.stride(1),
            right_gate.stride(2), right_gate.stride(3),
            B, N_seq, hidden_dim,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out

    # ------------------------------------------------------------------
    # 5. Core TriMul: out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    #    Implemented as a batched GEMM (cuBLAS) for each hidden slice.
    # ------------------------------------------------------------------
    #   shape → [B, H, N, N] for convenient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)              # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)            # (B*H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm over hidden dimension

    out = torch.nn.functional.layer_norm(

        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # 7. Apply out_gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    # Ensure final dtype is float32 as required
    out = out.to(torch.float32)

    return out
