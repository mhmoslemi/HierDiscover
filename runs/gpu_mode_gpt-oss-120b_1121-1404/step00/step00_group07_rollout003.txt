import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Triton‐accelerated outgoing TriMul (AlphaFold3 style).

    The kernel computes
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    for each hidden dimension d (hidden_dim) after layer‑norm,
    linear projections, masking and sigmoid gating.  The heavy
    O(N³·hidden_dim) contraction is performed in a custom Triton
    kernel; all other primitive operations are executed in PyTorch.
    """
    input_tensor, mask, weights, config = data
    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm over channel dimension `dim`.
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) -> shape [B,N,N,hidden_dim].
    # -----------------------------------------------------------------
    # NOTE: torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply optional mask (broadcasted on hidden_dim).
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of linear) and element‑wise gating.
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton‑based contraction:   out = Σ_k left[...,k,d] * right[...,k,d]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # Convert to half‑precision for the kernel (still accumulate in FP32)
    left_fp = left.to(torch.float16).contiguous()
    right_fp = right.to(torch.float16).contiguous()

    # Allocate output (FP32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Strides in elements (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Triton block sizes – tuned for H100 (fp16 loads, fp32 accumulation)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    # Grid: (i‑tiles, j‑tiles, batch * hidden_dim)
    grid = (triton.cdiv(N, BLOCK_I), triton.cdiv(N, BLOCK_J), B * hidden_dim)

    @triton.jit
    def trimul_outgoing_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        # Program IDs
        pid_i = tl.program_id(0)          # i‑tile
        pid_j = tl.program_id(1)          # j‑tile
        pid_bh = tl.program_id(2)         # combined batch & hidden index

        # Decode batch and hidden indices
        b = pid_bh // H
        h = pid_bh % H

        # Compute thread‑level offsets for I and J dimensions
        offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
        offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

        mask_i = offs_i < N
        mask_j = offs_j < N

        # Pointers that already include batch and hidden offsets
        left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
        right_base = right_ptr + b * stride_r_b + h * stride_r_h
        out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

        # Accumulator in FP32
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # Loop over the reduction dimension K
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # Load a tile of `left`  (shape: BLOCK_I × BLOCK_K)
            a_ptrs = left_base + offs_i[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
            a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

            # Load a tile of `right` transposed to (K × BLOCK_J)
            # right is stored as (j, k) but we need (k, j) for matmul
            b_ptrs = right_base + cur_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
            b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

            # Accumulate product
            acc += tl.dot(a, b)   # (BLOCK_I, BLOCK_J)

        # Write back
        out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # Launch kernel
    trimul_outgoing_kernel[grid](
        left_fp,
        right_fp,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Output normalization, gating and final linear projection.
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate   # element‑wise gating
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out

import triton.language as tl


def custom_kernel(data):
    # -----------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, hidden_dim]
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid of linear) + element‑wise modulation
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Triton kernel for the O(N³·hidden_dim) contraction
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # Work in fp16 inside the kernel (still accumulate in fp32)
    left_fp = left.to(torch.float16).contiguous()
    right_fp = right.to(torch.float16).contiguous()

    # Output buffer in float32
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # -------------------------------------------------------------
    # Triton kernel definition
    # -------------------------------------------------------------
    @triton.jit
    def trimul_outgoing_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        # program IDs
        pid_i = tl.program_id(0)           # tile over I (the first seq‑dim)
        pid_j = tl.program_id(1)           # tile over J (the second seq‑dim)
        pid_bh = tl.program_id(2)          # combined batch‑&‑hidden index

        # decode batch and hidden index
        b = pid_bh // H
        h = pid_bh % H

        # offsets for the current tile
        offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
        offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
        mask_i = offs_i < N
        mask_j = offs_j < N

        # base pointers already containing batch and hidden offsets
        left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
        right_base = right_ptr + b * stride_r_b + h * stride_r_h
        out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

        # accumulator in FP32
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # loop over the reduction dimension K
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            cur_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # A tile : left[i, k, h]  → shape (BLOCK_I, BLOCK_K)
            a_ptrs = left_base + offs_i[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
            a = tl.load(a_ptrs,
                        mask=mask_i[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

            # B tile : right[j, k, h] transposed to (k, j) for matmul
            b_ptrs = right_base + cur_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
            b = tl.load(b_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

            # accumulate: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)
            acc += tl.dot(a, b)

        # write results
        out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # -------------------------------------------------------------
    # Kernel launch
    # -------------------------------------------------------------
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * hidden_dim,
    )
    trimul_outgoing_kernel[grid](
        left_fp,
        right_fp,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Post‑processing: normalization, output gating and final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                       # element‑wise gating
    out = torch.nn.functional.linear(out, weights["to_out.weight"])   # → [B,N,N,dim]

    return out
