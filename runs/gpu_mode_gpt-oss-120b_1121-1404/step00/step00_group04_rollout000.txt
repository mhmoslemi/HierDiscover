"""
custom_kernel - Triangular Multiplicative Update (outgoing) using Triton

The algorithm follows the PyTorch reference of AlphaFold3's TriMul:
1. Layer‑norm the input tensor on the channel dimension.
2. Linear projections (`left_proj`, `right_proj`) and linear gates
   (`left_gate`, `right_gate`, `out_gate`) are applied (no bias).
3. The optional mask (shape [B,N,N]) is broadcasted and applied to the
   projected tensors before gating.
4. The core O(N³·C) operation
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is implemented as a batched matrix‑multiply with Triton.  By permuting
   the tensors we obtain a standard GEMM: for each batch‑hidden pair we
   compute `left_perm (N×N) @ right_perm.T (N×N)`.  The multiplication is
   fused into a single Triton kernel that tiles the output (64×64) and
   reduces over `k` in blocks of 64.
5. The result is layer‑normed again, gated by `out_gate`, and finally
   projected back to the original channel dimension.

Only the heavy matrix‑multiply is realised in Triton; the surrounding
operations (layer‑norm, linear layers, sigmoid gates) use PyTorch
functions for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A: (batch, M, K), B: (batch, K, N)
# ----------------------------------------------------------------------
@triton.jit
def _triton_batched_gemm(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix sizes
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs: (batch, block_m, block_n)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # accumulator (float32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K increments
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Masks for the k‑dimension
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_batch_ptr
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile: shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_batch_ptr
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate dot product
        acc += tl.dot(a, b)

    # Write back the result
    c = acc.to(tl.float32)
    tl.store(
        c_batch_ptr
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Arguments:
        data = (x, mask, weights, config)
            x      : Tensor[B, N, N, dim]   (float32)
            mask   : Tensor[B, N, N] or None
            weights: dict of model parameters (torch tensors)
            config : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        Tensor[B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs and configuration
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]               # input/output channel dimension
    hidden_dim = config["hidden_dim"] # intermediate dimension
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name].to(device=device, dtype=dtype)

    # LayerNorm parameters
    norm_weight = _w("norm.weight")
    norm_bias = _w("norm.bias")

    # Linear projection weights (bias=False)
    left_proj_weight = _w("left_proj.weight")   # (hidden_dim, dim)
    right_proj_weight = _w("right_proj.weight") # (hidden_dim, dim)

    # Gate weights
    left_gate_weight = _w("left_gate.weight")   # (hidden_dim, dim)
    right_gate_weight = _w("right_gate.weight") # (hidden_dim, dim)
    out_gate_weight = _w("out_gate.weight")     # (hidden_dim, dim)

    # Output norm + final projection
    to_out_norm_weight = _w("to_out_norm.weight")
    to_out_norm_bias = _w("to_out_norm.bias")
    to_out_weight = _w("to_out.weight")         # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm on channel dimension
    # ------------------------------------------------------------------
    # LayerNorm expects normalized_shape = (dim,)
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if nomask or mask is None:
        mask_tensor = torch.ones_like(x_norm[..., :1])  # shape (B,N,N,1)
    else:
        mask_tensor = mask.unsqueeze(-1).to(dtype)

    left = left * mask_tensor
    right = right * mask_tensor

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul via Triton GEMM
    #    We compute for each hidden slice d:
    #        out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
    #    This is equivalent to (left_perm @ right_perm.T) per hidden channel.
    # ------------------------------------------------------------------
    #   Permute to (B, hidden, N, N) and make right transposed on the last
    #   two dimensions, then flatten batch+hidden into a single batch for
    #   the Triton kernel.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (k ↔ j)

    B, H, N, _ = left_perm.shape
    batch_h = B * H                       # fused batch dimension

    # View as 3‑D tensors for the kernel
    A = left_perm.view(batch_h, N, N)      # (batch_h, M=N, K=N)
    B_mat = right_perm.view(batch_h, N, N) # (batch_h, K=N, N)

    # Allocate output buffer
    C = torch.empty_like(A)                # (batch_h, N, N)

    # Tile sizes – chosen to fit H100 registers and shared memory.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _triton_batched_gemm[grid](
        A,
        B_mat,
        C,
        N,      # M
        N,      # N
        N,      # K
        A.stride(0), A.stride(1), A.stride(2),
        B_mat.stride(0), B_mat.stride(1), B_mat.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
