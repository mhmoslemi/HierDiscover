import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH_H, N,
    stride_az, stride_ai, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets for output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer offsets
    a_ptrs = a_ptr + bh * stride_az + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + bh * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # load tiles
    a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < N), other=0.0).to(tl.float32)
    b = tl.load(b_ptrs, mask=(offs_k[:, None] < N) & mask_n[None, :], other=0.0).to(tl.float32)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets for current k tile
        cur_k = k * BLOCK_K
        # we need to recompute a and b for each k tile - but we can compute within loop
        # Instead we move pointers using pointer arithmetic inside loop.
        a_tile_ptrs = a_ptr + bh * stride_az + offs_m[:, None] * stride_ai + (cur_k + offs_k)[None, :] * stride_ak
        b_tile_ptrs = b_ptr + bh * stride_bz + (cur_k + offs_k)[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a_tile = tl.load(a_tile_ptrs, mask=mask_m[:, None] & ((cur_k + offs_k) < N), other=0.0).to(tl.float32)
        b_tile = tl.load(b_tile_ptrs, mask=((cur_k + offs_k)[:, None] < N) & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product tile
        acc += tl.dot(a_tile, b_tile)
    # store
    c_ptrs = c_ptr + bh * stride_cz + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

        right = right * mask_exp

    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()

"""
TriMul (outgoing) – a fused Triton kernel for the heavy N³ matrix multiplication.

The kernel implements
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
for every batch * hidden‑channel pair (b,d).  The implementation
* normalises the input (LayerNorm) and applies linear projections and
  sigmoid gates in PyTorch,
* packs the tensors so that the batch dimension and the hidden dimension
  become a single “batch‑page” dimension,
* calls a Triton kernel that performs a blocked GEMM of shape
  (N,N)×(N,N)ᵀ for each page,
* finishes the forward pass (LayerNorm, output‑gate, final linear)
  also in PyTorch.

The kernel works for any sequence length N (up to 1024 in the tests),
any batch size ≤2 and hidden size ≤128.  It uses float16 inside the
kernel for speed and accumulates in float32 to keep numeric stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    a_ptr,                     # const uint8*   (left)
    b_ptr,                     # const uint8*   (right)
    c_ptr,                     # uint8*         (output)
    batch_hidden, N,           # int32 scalars
    stride_az, stride_ai, stride_ak,   # strides for a (page, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for b (page, k‑dim, n‑dim)
    stride_cz, stride_ci, stride_cj,   # strides for c (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a block of C = A @ Bᵀ where
        A : (batch_hidden, N, N)  – row‑major
        B : (batch_hidden, N, N)  – row‑major, but accessed as Bᵀ
        C : (batch_hidden, N, N)  – row‑major
    """
    pid_z = tl.program_id(0)          # which (batch, hidden) page
    pid_m = tl.program_id(1)          # block row index
    pid_n = tl.program_id(2)          # block column index

    # ---- block offsets ----------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointer to the beginning of the current page
    a_page_ptr = a_ptr + pid_z * stride_az
    b_page_ptr = b_ptr + pid_z * stride_bz
    c_page_ptr = c_ptr + pid_z * stride_cz

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- iterate over the reduction dimension K ---------------------------
    cur_k = 0
    while cur_k < N:
        cur_k = tl.minimum(cur_k, N)          # guard for the last (partial) tile
        # tile offsets within the K‑dimension
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)

        # masks for the current K‑tile
        mask_k = cur_k_offs < N

        # ----- load A tile --------------------------------------------------
        a_ptrs = a_page_ptr \
                 + offs_m[:, None] * stride_ai \
                 + cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ----- load B tile (as Bᵀ) -----------------------------------------
        # Bᵀ has shape (K, N) → stride_bk = 1 (along K), stride_bn = N (along N)
        b_ptrs = b_page_ptr \
                 + cur_k_offs[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ----- multiply‑accumulate -----------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        cur_k += BLOCK_K

    # ---- write back --------------------------------------------------------
    c_ptrs = c_page_ptr \
             + offs_m[:, None] * stride_ci \
             + offs_n[None, :] * stride_cj
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul update.

    Arguments
    ---------
    data: tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------ unpack
    x, mask, weights, config = data
    # x : [B, N, N, dim]  (float32)
    # mask : [B, N, N]    (float32 or bool)
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------ LayerNorm on input
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------ Linear projections (no bias)
    left = F.linear(x_norm, weights["left_proj.weight"])      # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])    # [B,N,N,hidden]

    # ------------------------------------------------------------------ Mask (if any)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ Gating (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))     # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ Cast to half for the heavy kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # bring hidden dimension to the second axis -> [B, hidden, N, N] (contiguous)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # flatten (B * hidden) as the kernel's “batch” dimension
    BxH = B * hidden
    left_mat = left_h.view(BxH, N, N)
    right_mat = right_h.view(BxH, N, N)

    # allocate output buffer (half)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=x.device)

    # ----------------------------------------------------------- strides (in elements)
    stride_az = left_mat.stride(0)   # page stride = N*N
    stride_ai = left_mat.stride(1)   # row stride = N
    stride_ak = left_mat.stride(2)   # col stride = 1

    # right matrix is accessed as Bᵀ → k‑stride = col‑stride (1), n‑stride = row‑stride (N)
    stride_bz = right_mat.stride(0)  # same page stride
    stride_bk = right_mat.stride(2)  # =1  (stride over k)
    stride_bn = right_mat.stride(1)  # =N  (stride over output‑col j)

    stride_cz = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # ----------------------------------------------------------- launch config
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # small enough to fit in registers, good for N up to 1024

    grid = (BxH,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BxH,
        N,
        stride_az, stride_ai, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # GPU: H100 works well with 4‑warp kernels
    )

    # ----------------------------------------------------------- reshape back
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden]
    out = out.to(torch.float32)

    # ----------------------------------------------------------- LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ----------------------------------------------------------- output gate
    out = out * out_gate

    # ----------------------------------------------------------- final linear (to dim)
    out = F.linear(out, weights["to_out.weight"])

    return out
