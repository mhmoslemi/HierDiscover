import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # total tiles in output per hidden (M,N)
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_head = num_tiles_i * num_tiles_j
    # compute batch and head index
    bh = pid // tiles_per_head
    tile_id = pid % tiles_per_head
    tile_i = tile_id // num_tiles_j
    tile_j = tile_id % num_tiles_j

    # compute batch and hidden dim indices
    b = bh // H
    h = bh % H

    # offsets within tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A block (M, K)
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h \
                 + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block (K, N) – note the order of indices for the right tensor
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h \
                 + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute block multiplication
        acc += tl.dot(a, b)  # (M,K) @ (K,N) -> (M,N)

    # Write back
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # Unpack
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # optionally other config items
    # get weight tensors
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B,N,N2, D = input_tensor.shape
    assert N == N2 and D == dim

    # LayerNorm over last dim
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    if not nomask:
        # mask shape [B,N,N] -> [B,N,N,1]
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Launch Triton kernel
    # Ensure contiguous
    left_fp16 = left_fp16.contiguous()
    right_fp16 = right_fp16.contiguous()
    out = out.contiguous()

    # Get strides
    stride_left = left_fp16.stride()
    stride_right = right_fp16.stride()
    stride_out = out.stride()

    # Kernel block size
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid
    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * hidden_dim * num_tiles_i * num_tiles_j,)

    # Launch
    tri_mul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # dimensions of tile grid for each head (b,h)
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_head = num_tiles_i * num_tiles_j

    bh = pid // tiles_per_head            # index over (batch * hidden)
    tile_id = pid % tiles_per_head
    tile_i = tile_id // num_tiles_j
    tile_j = tile_id % num_tiles_j

    # compute batch and hidden index
    b = bh // H
    h = bh % H

    # offsets within tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A (left) tile: shape (M, K)
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h \
                 + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B (right) tile transposed: shape (K, N)
        # we read right[k, j] (k as row, j as column)
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h \
                 + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)  # (M,K) @ (K,N) -> (M,N)

    # Write result
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul “outgoing” update.

    This implementation performs the heavy three‑dimensional
    contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    using a fused batched GEMM kernel. All lightweight
    linear, bias, gating and layer‑norm operations are carried out
    with PyTorch for simplicity.
    """
    import torch
    import torch.nn.functional as F
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm over channel dim -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ----- Linear projections -----
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask -----
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- Gating -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Convert to fp16 and allocate output -----
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_f16.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_f16.device).contiguous()

    # ----- Strides (in elements) -----
    stride_l = left_f16.stride()
    stride_r = right_f16.stride()
    stride_o = out.stride()

    # ----- Kernel launch parameters -----
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * hidden_dim * num_tiles_i * num_tiles_j,)

    tri_mul_gemm_kernel[grid](
        left_f16,
        right_f16,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ----- Post‑processing (LayerNorm, gating, final linear) -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])
    return output

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the channel dimension.
2. Project the normalized tensor to a hidden dimension with two linear layers.
3. Apply an optional pairwise mask and three sigmoid gates.
4. Fuse the costly contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   into a batched GEMM using a custom Triton kernel.  The contraction is
   equivalent to a batch of matrix multiplications
       out[b,d] = left[b,:,:,d] @ right[b,:,:,d].T
   where each (b,d) pair is processed by one program that tiles the i‑ and
   j‑axes (64×64) and iterates over k in 32‑element blocks, using Tensor‑Core
   FP16 matrix units and accumulating in FP32.
5. Apply a second LayerNorm on the hidden dimension, multiply by the output
   gate, and finally project back to the original channel dimension.

Only the heavy inner product (step 4) is implemented in Triton; all other
operations use standard PyTorch functions.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr,                     # fp16 pointer, shape [B,N,N,H]
    right_ptr,                    # fp16 pointer, shape [B,N,N,H]
    out_ptr,                      # fp32 pointer, shape [B,N,N,H]
    B, N, H,                      # runtime integers
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid = tl.program_id(0)

    # Number of tiles for the i/j dimensions per (b,h) pair
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    tiles_per_head = num_tiles_i * num_tiles_j

    # Decode program id -> (batch, hidden, tile_i, tile_j)
    bh = pid // tiles_per_head               # index over B*H
    tile_id = pid % tiles_per_head
    tile_i = tile_id // num_tiles_j
    tile_j = tile_id % num_tiles_j

    b = bh // H                               # batch index
    h = bh % H                                # hidden‑dim index

    # Offsets inside the current tile
    offs_i = tile_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = tile_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis k in blocks of size BLOCK_K
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load a tile from left (shape M×K) -----
        a_ptrs = left_ptr + b * stride_left_b + h * stride_left_h \
                 + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # fp16

        # ----- Load a transposed tile from right (shape K×N) -----
        # right is indexed as (b, j, k, h); we need (k, j)
        b_ptrs = right_ptr + b * stride_right_b + h * stride_right_h \
                 + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # fp16

        # Matrix‑multiply‑accumulate (FP16 * FP16 → FP32)
        acc += tl.dot(a, b).to(tl.float32)    # (M,K) @ (K,N) → (M,N)

    # ----- Write the result tile -----
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h \
               + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])


# --------------------------------------------------------------
# Python entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights      : dict with all linear / layer‑norm parameters
        - config       : dict containing "dim", "hidden_dim" and "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional pairwise mask
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Triton kernel: the O(N³) contraction
    # ------------------------------------------------------------------
    # Cast to FP16 for the kernel (Tensor‑Core friendly)
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_f16.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32,
                      device=left_f16.device).contiguous()

    # Strides (in elements)
    stride_l = left_f16.stride()   # (B, i, k, h)
    stride_r = right_f16.stride()
    stride_o = out.stride()

    # Tiling parameters – empirically good for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B * hidden_dim * tiles_i * tiles_j,)

    tri_mul_gemm_kernel[grid](
        left_f16,
        right_f16,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])

    return output
