def custom_kernel(data):
    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm
    eps = 1e-5
    x = input_tensor
    # compute mean variance across dim
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # compute core TriMul using Triton
    out = tri_mul_triton(left, right)  # returns shape [B,N,N,hidden_dim]

    # to_out_norm LayerNorm across hidden_dim
    # compute mean,var across hidden_dim per (b,i,j) -> shape broadcast
    out_mean = out.mean(-1, keepdim=True)
    out_var = ((out - out_mean) ** 2).mean(-1, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # multiply by out_gate
    out_norm = out_norm * out_gate

    # final linear
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)

    return out_final
```

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH_M, M, N, K,
    stride_lbm, stride_lm, stride_lk,
    stride_rbm, stride_rn, stride_rk,
    stride_obm, stride_om, stride_on,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # compute pointers for left: shape (M,K)
        left_ptrs = left_ptr + batch_id * stride_lbm + offs_m[:, None] * stride_lm + offs_k[None, :] * stride_lk
        # compute pointers for right (transposed): we need elements right[batch, n, k]
        right_ptrs = right_ptr + batch_id * stride_rbm + offs_n[None, :] * stride_rn + offs_k[:, None] * stride_rk

        # load
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    stride_lbm, stride_lm, stride_lk,
    stride_rbm, stride_rn, stride_rk,
    stride_obm, stride_om, stride_on,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left pointers
        left_ptrs = left_ptr + batch_id * stride_lbm + offs_m[:, None] * stride_lm + offs_k[None, :] * stride_lk
        # right pointers (transposed)
        right_ptrs = right_ptr + batch_id * stride_rbm + offs_n[None, :] * stride_rn + offs_k[:, None] * stride_rk

        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # compute dot product in fp32 accumulator
        acc += tl.dot(left, right).to(tl.float32)

    # store result
    out_ptrs = out_ptr + batch_id * stride_obm + offs_m[:, None] * stride_om + offs_n[None, :] * stride_on

"""
TriMul (outgoing) Triton implementation.

We implement the forward of the TriMul module from AlphaFold3:
1. Layer‑norm the input.
2. Linear projections (left/right) and gating (left‑gate/right‑gate/out‑gate).
3. Apply optional pairwise mask.
4. Core operation:   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (einsum)
   – Implemented as many independent GEMMs (batch*heads) in a fused Triton kernel.
   – Left/right tensors are cast to float16 for bandwidth‑efficiency,
     accumulation is performed in float32.
5. Layer‑norm over the hidden dimension, multiply by out‑gate,
   and final linear projection back to the original dimension.

The Triton kernel processes tiles of the output matrix for each
(batch * head) slice, looping over the shared dimension `k`.
Masking and gating are handled in PyTorch; only the O(N³·H) “tri‑mul”
is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul einsum
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    M, N, K,
    # strides (in elements)
    stride_lbm, stride_lm, stride_lk,   # left  [batch, M, K]
    stride_rbm, stride_rn, stride_rk,   # right [batch, N, K] (to be read as [K,N] for transposed)
    stride_obm, stride_om, stride_on,   # out   [batch, M, N]
    # tile sizes (compile‑time constants)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each batch*head slice."""
    batch_id = tl.program_id(0)          # batch * head dimension
    pid_m = tl.program_id(1)             # output row block
    pid_n = tl.program_id(2)             # output col block

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to current tiles
        # left: [M, K]
        left_tile_ptr = left_ptr + batch_id * stride_lbm \
                        + offs_m[:, None] * stride_lm \
                        + offs_k[None, :] * stride_lk
        # right (transposed): we need right[batch, n, k] → tile [K, N]
        right_tile_ptr = right_ptr + batch_id * stride_rbm \
                         + offs_n[None, :] * stride_rn \
                         + offs_k[:, None] * stride_rk

        # load tiles (FP16) with out‑of‑bounds protection
        left = tl.load(left_tile_ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # block‑matrix multiply; cast result to FP32 before accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # --------------------------------------------------------------
    # write the result
    # --------------------------------------------------------------
    out_tile_ptr = out_ptr + batch_id * stride_obm \
                   + offs_m[:, None] * stride_om \
                   + offs_n[None, :] * stride_on

    tl.store(out_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper: GPU‑side TriMul (float16 inputs, float32 output)
# ----------------------------------------------------------------------
def _tri_mul_triton(left_fp16: torch.Tensor, right_fp16: torch.Tensor,
                    BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    left_fp16 / right_fp16 : [B, N, N, H] (float16)
    Returns out : [B, N, N, H] (float32)
    """
    B, N, _, H = left_fp16.shape
    # bring head dimension next to batch for easy flattening
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    BH = B * H
    left_flat = left_perm.view(BH, N, N)    # [BH, M, K]
    right_flat = right_perm.view(BH, N, N)  # [BH, N, K] (will be read transposed)

    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=left_fp16.device)

    # strides (in elements)
    stride_lbm = left_flat.stride(0)
    stride_lm  = left_flat.stride(1)
    stride_lk  = left_flat.stride(2)

    stride_rbm = right_flat.stride(0)
    stride_rn  = right_flat.stride(1)   # row (n) stride
    stride_rk  = right_flat.stride(2)   # col (k) stride

    stride_obm = out_flat.stride(0)
    stride_om  = out_flat.stride(1)
    stride_on  = out_flat.stride(2)

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,                     # M, N, K (all = seq_len)
        stride_lbm, stride_lm, stride_lk,
        stride_rbm, stride_rn, stride_rk,
        stride_obm, stride_om, stride_on,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, H] (float32)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module with a fused Triton GEMM.
    Input:
        data = (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]      (float32)
        - mask         : [B, N, N] (bool/float) or None
        - weights      : dict of model parameters
        - config       : dict with keys "dim", "hidden_dim"
    Output:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gating
    # ------------------------------------------------------------------
    # projections
    left_proj_w = weights["left_proj.weight"]      # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]    # [hidden_dim, dim]
    left = F.linear(x, left_proj_w)                # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # optional pairwise mask
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # gates (linear + sigmoid)
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Core TriMul (batched GEMM) – off‑loaded to Triton
    # ------------------------------------------------------------------
    # Cast to half‑precision for the GEMM kernel
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_hidden = _tri_mul_triton(left_fp16, right_fp16)   # [B,N,N,hidden_dim] FP32

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out_hidden,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    out = out * out_gate   # element‑wise gating

    to_out_w = weights["to_out.weight"]   # [dim, hidden_dim]
    out = F.linear(out, to_out_w)         # final projection to original dim

    return out
