@triton.jit
def final_linear_kernel(
    inp_ptr, weight_ptr, out_ptr,
    B, N, hidden_dim, out_dim,
    stride_inp_b, stride_inp_n1, stride_inp_n2, stride_inp_h,
    stride_weight_o, stride_weight_i,
    stride_out_b, stride_out_n1, stride_out_n2, stride_out_o,
    BLOCK_OUT: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0) # position index
    pid1 = tl.program_id(1) # out channel block

    total_positions = B * N * N
    if pid0 >= total_positions:
        return

    # decode position
    b = pid0 // (N * N)
    tmp = pid0 % (N * N)
    i = tmp // N
    j = tmp % N

    # output channel offsets
    offs_o = tl.arange(0, BLOCK_OUT) + pid1 * BLOCK_OUT
    mask_o = offs_o < out_dim

    # accumulator
    acc = tl.zeros((BLOCK_OUT,), dtype=tl.float32)

    # loop over hidden dim tiles
    num_k = tl.cdiv(hidden_dim, BLOCK_K)
    for kk in range(num_k):
        k_offset = kk * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + k_offset
        mask_k = offs_k < hidden_dim

        # load input vector slice
        inp = tl.load(inp_ptr + b*stride_inp_b + i*stride_inp_n1 + j*stride_inp_n2 + offs_k*stride_inp_h,
                      mask=mask_k, other=0.0)  # shape (BLOCK_K,)

        # load weight tile of shape (BLOCK_OUT, BLOCK_K)
        # weight index: out_idx * stride_weight_o + hidden_idx * stride_weight_i
        w = tl.load(weight_ptr + offs_o[:, None] * stride_weight_o + offs_k[None, :] * stride_weight_i,
                    mask=mask_o[:, None] & mask_k[None, :], other=0.0)  # shape (BLOCK_OUT, BLOCK_K)

        # accumulate dot product: sum over k: inp[k] * w[out, k]
        acc += tl.sum(w * inp[None, :], axis=1)

    # store result
    out_ptrs = out_ptr + b*stride_out_b + i*stride_out_n1 + j*stride_out_n2 + offs_o*stride_out_o
    tl.store(out_ptrs, acc, mask=mask_o)

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The algorithm follows the reference PyTorch implementation:
1. Layer‑norm the input tensor over the channel dimension.
2. Linear projections (left/right) and gating (left_gate, right_gate, out_gate).
3. Apply the optional mask and element‑wise gating.
4. Compute the core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiplication per hidden channel
   (left @ rightᵀ).  This uses torch.bmm for high‑performance GEMM.
5. Layer‑norm the contraction result (to_out_norm) and multiply by out_gate.
6. The final linear projection (hidden_dim → dim) is performed by a custom
   Triton kernel that fuses the per‑position dense multiplication:
       out_final[b,i,j,:] = Σ_h gated_out[b,i,j,h] * W[:,h]

The Triton kernel processes each (b,i,j) location as a program instance
and iterates over the hidden dimension in small tiles (BLOCK_K).  Output
channels are tiled with BLOCK_OUT to keep register pressure low.  All
reductions are performed in float32 and the result is written back in the
original dtype (float32).

The kernel ensures at least part of the heavy work is done on the GPU
with custom code while the rest of the pipeline uses PyTorch's highly
optimized primitives.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the final dense projection (hidden_dim → dim)
# ----------------------------------------------------------------------
@triton.jit
def final_linear_kernel(
    # pointers
    inp_ptr,          # gated tensor [B,N,N,hidden_dim]
    weight_ptr,       # linear weight [dim, hidden_dim]
    out_ptr,          # output tensor [B,N,N,dim]
    # shape parameters (int32 scalars)
    B, N, hidden_dim, out_dim,
    # strides (int32)
    stride_inp_b, stride_inp_n1, stride_inp_n2, stride_inp_h,
    stride_weight_o, stride_weight_i,
    stride_out_b, stride_out_n1, stride_out_n2, stride_out_o,
    # compile‑time tile sizes
    BLOCK_OUT: tl.constexpr,   # number of output channels per block
    BLOCK_K: tl.constexpr      # hidden‑dim tile size
):
    # ------------------------------------------------------------------
    # 1) identify the (b,i,j) position this program works on
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)                # position index
    pid1 = tl.program_id(1)                # output‑channel block index

    total_pos = B * N * N
    if pid0 >= total_pos:
        return

    # decode flattened position into (b,i,j)
    b = pid0 // (N * N)
    tmp = pid0 % (N * N)
    i = tmp // N
    j = tmp % N

    # ------------------------------------------------------------------
    # 2) offsets for the output channels handled by this block
    # ------------------------------------------------------------------
    offs_o = tl.arange(0, BLOCK_OUT) + pid1 * BLOCK_OUT
    mask_o = offs_o < out_dim                # out‑of‑bounds mask

    # accumulator for the output channels of this block
    acc = tl.zeros((BLOCK_OUT,), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) iterate over hidden dimension tiles
    # ------------------------------------------------------------------
    num_k = tl.cdiv(hidden_dim, BLOCK_K)
    for kk in range(num_k):
        k_offset = kk * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + k_offset
        mask_k = offs_k < hidden_dim

        # load a slice of the gated hidden vector for this (b,i,j)
        inp = tl.load(
            inp_ptr
            + b * stride_inp_b
            + i * stride_inp_n1
            + j * stride_inp_n2
            + offs_k * stride_inp_h,
            mask=mask_k,
            other=0.0
        )                                            # shape: (BLOCK_K,)

        # load the corresponding slice of the weight matrix
        # weight layout: [out_dim, hidden_dim]
        w = tl.load(
            weight_ptr
            + offs_o[:, None] * stride_weight_o   # out‑dim stride
            + offs_k[None, :] * stride_weight_i,  # hidden‑dim stride
            mask=mask_o[:, None] & mask_k[None, :],
            other=0.0
        )                                            # shape: (BLOCK_OUT, BLOCK_K)

        # accumulate dot products over the K‑tile
        # broadcasting inp (1, K) across the O dimension
        acc += tl.sum(w * inp[None, :], axis=1)

    # ------------------------------------------------------------------
    # 4) write the result back to the output tensor
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i * stride_out_n1
        + j * stride_out_n2
        + offs_o * stride_out_o
    )
    tl.store(out_ptrs, acc, mask=mask_o)


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor : torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights_dict: mapping from string names to torch.Tensor weight parameters
            - config_dict : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                 # input / output channel dimension
    hidden_dim = config["hidden_dim"]   # internal dimension used throughout

    # ------------------------------------------------------------------
    # convenience aliases for weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]          # shape [dim, hidden_dim]

    # ------------------------------------------------------------------
    # 1) input layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # 2) linear projections (no bias) -> [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # (out_features = hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(dtype).unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) gating networks (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched GEMM per hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # B and N are unchanged

    # permute to bring hidden_dim to the batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    # right needs its last two axes transposed to match the contraction
    right_T = right_perm.transpose(2, 3).contiguous()   # [B, hidden, N, N] with (k,i)

    # reshape to (B*hidden, N, N) for torch.bmm
    left_flat = left_perm.view(B * hidden_dim, N, N)
    right_flat = right_T.view(B * hidden_dim, N, N)

    # batched matrix multiplication (float32)
    out_flat = torch.bmm(left_flat, right_flat)          # [B*hidden, N, N]

    # reshape back to [B,N,N,hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) post‑contraction layer‑norm over hidden_dim
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 7) element‑wise gating by out_gate
    # ------------------------------------------------------------------
    gated = (out_norm * out_gate).contiguous()   # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 8) final linear projection using a fused Triton kernel
    #    (hidden_dim → dim).  No bias term for this projection.
    # ------------------------------------------------------------------
    out_final = torch.empty((B, N, N, dim), dtype=dtype, device=device)

    # compute strides for the tensors
    stride_inp_b, stride_inp_n1, stride_inp_n2, stride_inp_h = gated.stride()
    stride_out_b, stride_out_n1, stride_out_n2, stride_out_o = out_final.stride()
    # weight is stored as [dim, hidden_dim]
    weight = to_out_weight.contiguous()
    stride_weight_o, stride_weight_i = weight.stride()

    # tile sizes (chosen to fit in registers and shared memory on H100)
    BLOCK_OUT = 128   # max number of output channels per program
    BLOCK_K = 32      # hidden‑dim tile size

    total_positions = B * N * N
    out_blocks = (dim + BLOCK_OUT - 1) // BLOCK_OUT

    grid = (total_positions, out_blocks)

    final_linear_kernel[grid](
        gated,
        weight,
        out_final,
        B,
        N,
        hidden_dim,
        dim,
        stride_inp_b,
        stride_inp_n1,
        stride_inp_n2,
        stride_inp_h,
        stride_weight_o,
        stride_weight_i,
        stride_out_b,
        stride_out_n1,
        stride_out_n2,
        stride_out_o,
        BLOCK_OUT=BLOCK_OUT,
        BLOCK_K=BLOCK_K,
    )

    return out_final
