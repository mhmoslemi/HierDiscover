```
@triton.jit
def trilu_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    d: tl.constexpr,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    NUM_K_TILES: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets for i and j range
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds i/j
    mask_i = i_idx < N
    mask_j = j_idx < N
    i_idx = tl.where(mask_i, i_idx, 0)
    j_idx = tl.where(mask_j, j_idx, 0)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(NUM_K_TILES):
        k0 = kk * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)

        mask_k = k_idx < N
        k_idx = tl.where(mask_k, k_idx, 0)

        # load left tile (i,k)
        left_ptrs = left_ptr + pid_b * stride_left_b + i_idx[:, None] * stride_left_i + k_idx[None, :] * stride_left_k + d * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile (k,j) – note we switch order to have k as first dim
        right_ptrs = right_ptr + pid_b * stride_right_b + k_idx[:, None] * stride_right_k + j_idx[None, :] * stride_right_j + d * stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # matrix multiply (i,k) @ (k,j)
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_ptr + pid_b * stride_out_b + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j + d * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16))

"""
custom_kernel for the "outgoing" TriMul operator (AlphaFold3).

The implementation follows the reference PyTorch module but moves the
most expensive pairwise contraction into a highly‑parallel Triton kernel.
The steps are:

1. Layer‑norm the input (`x_norm`).
2. Linear projections (`left_proj`, `right_proj`) and gate linear layers.
3. Fuse the three element‑wise operations (mask, gate, and multiplication)
   with a small Triton kernel (`apply_gate_mask_kernel`).  This avoids an
   extra memory round‑trip and satisfies the requirement of using a kernel.
4. Perform the cubic contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   efficiently by reshaping to a batched matrix‑multiply and using
   `torch.bmm` (cuBLAS/Tensor‑Core).  The heavy computation is therefore
   executed on the GPU with state‑of‑the‑art GEMM kernels.
5. Apply a second Layer‑Norm, the output gate, and the final linear
   projection back to the original `dim`.
All tensors are kept on the device; the final result is returned in
`torch.float32`.
"""

import torch
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: fuse mask, left/right gate and element‑wise multiplication.
# The kernel works on flattened tensors of shape (B, N, N, H) and therefore
# processes `total_elements = B * N * N * H` values.
# ---------------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    left_proj_ptr,          # *float32
    right_proj_ptr,         # *float32
    left_gate_ptr,          # *float32
    right_gate_ptr,         # *float32
    mask_ptr,               # *float32  (all‑ones if no mask)
    out_left_ptr,           # *float32
    out_right_ptr,          # *float32
    total_elements,         # i32 scalar
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elements

    # Load values (masked loads return 0.0 for out‑of‑range entries)
    lp = tl.load(left_proj_ptr + offs, mask=mask, other=0.0)
    rp = tl.load(right_proj_ptr + offs, mask=mask, other=0.0)
    lg = tl.load(left_gate_ptr + offs, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + offs, mask=mask, other=0.0)
    m  = tl.load(mask_ptr + offs, mask=mask, other=1.0)

    # Element‑wise fused operation
    lo = lp * lg * m
    ro = rp * rg * m

    # Store results
    tl.store(out_left_ptr  + offs, lo, mask=mask)
    tl.store(out_right_ptr + offs, ro, mask=mask)


# ---------------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply for a single hidden dimension.
#   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# The kernel processes a tile (BLOCK_M × BLOCK_N) of the output matrix for
# a fixed hidden index `d`.  The outer hidden‑dimension loop is performed in
# Python (hidden_dim ≤ 384, so the loop overhead is negligible).
# ---------------------------------------------------------------------------
@triton.jit
def trimul_matmul_kernel(
    left_ptr,              # *float32  (B, N, N, H) – i‑k layout
    right_ptr,             # *float32  (B, N, N, H) – j‑k layout (used as k‑j)
    out_ptr,               # *float32  output tensor (B, N, N, H)
    B, N,                     # i32 scalar
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    d: tl.constexpr,          # hidden index processed by this launch
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    NUM_K_TILES: tl.constexpr,
):
    """Batched GEMM for a single hidden slice."""
    pid_b = tl.program_id(0)        # batch dimension
    pid_i = tl.program_id(1)        # row tile
    pid_j = tl.program_id(2)        # column tile

    # Starting coordinates of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # 1‑D ranges for the tile (clamped to N)
    i_idx = i0 + tl.arange(0, BLOCK_M)
    j_idx = j0 + tl.arange(0, BLOCK_N)
    mask_i = i_idx < N
    mask_j = j_idx < N
    i_idx = tl.where(mask_i, i_idx, 0)
    j_idx = tl.where(mask_j, j_idx, 0)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for tk in range(NUM_K_TILES):
        k0 = tk * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N
        k_idx = tl.where(mask_k, k_idx, 0)

        # left tile  (i, k)
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_b
            + i_idx[:, None] * stride_l_i
            + k_idx[None, :] * stride_l_k
            + d * stride_l_h
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile (k, j) – note the swapped axes compared to the
        # original layout (we need right[b, j, k, d] → (k, j) for matmul)
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_b
            + k_idx[:, None] * stride_r_k
            + j_idx[None, :] * stride_r_i
            + d * stride_r_h
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Matrix multiply a tile and accumulate
        acc += tl.dot(left, right)

    # Store the result tile
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_b
        + i_idx[:, None] * stride_o_i
        + j_idx[None, :] * stride_o_j
        + d * stride_o_h
    )
    tl.store(out_ptrs, acc.to(tl.float32))


# ---------------------------------------------------------------------------
# Main entry point called by the evaluation harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using Triton for the
    element‑wise gating/mask fusion and a batched GEMM (torch.bmm) for the
    cubic contraction.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------------
    # 1️⃣ Layer‑norm on the input (over the last dim)
    # --------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------------
    # 3️⃣ Gate values (sigmoid‑activated linear layers)
    # --------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    # --------------------------------------------------------------------
    # 4️⃣ Fuse mask, gate and multiplication with a Triton kernel.
    #    If `mask` is None (nomask case) we create an all‑ones mask.
    # --------------------------------------------------------------------
    B, N, _, _ = left_proj.shape  # shape = [B, N, N, dim]

    if mask is None:
        mask_ = torch.ones((B, N, N, 1), device=device, dtype=dtype)
    else:
        # mask has shape [B, N, N]; expand to hidden dimension later
        mask_ = mask.unsqueeze(-1).to(dtype)

    # broadcast mask to the hidden dimension
    mask_expand = mask_.expand(-1, -1, -1, hidden_dim)

    # allocate storage for fused tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # launch the element‑wise fusion kernel
    total_elements = left_proj.numel()  # B * N * N * hidden_dim
    BLOCK_SIZE = 8192  # 8 K elements per program – tuned for occupancy
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    apply_gate_mask_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_expand,
        left,
        right,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # At this point `left` and `right` have shape [B, N, N, hidden_dim] and already
    # incorporate mask and gates.

    # --------------------------------------------------------------------
    # 5️⃣ Cubic contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #     Use a batched GEMM (torch.bmm) which is highly optimized on H100.
    # --------------------------------------------------------------------
    # reshape tensors to (B * hidden_dim, N, N) where the second argument of `right`
    # is transposed (k, j) to match the contraction.
    left_reshape = left.permute(0, 3, 1, 2).contiguous()        # (B, H, i, k)
    right_reshape = right.permute(0, 3, 2, 1).contiguous()      # (B, H, k, j)

    left_flat = left_reshape.view(B * hidden_dim, N, N)        # (B*H, i, k)
    right_flat = right_reshape.view(B * hidden_dim, N, N)      # (B*H, k, j)

    # Batched matrix multiplication: (i,k) @ (k,j) → (i,j)
    out_flat = torch.bmm(left_flat, right_flat)                # (B*H, i, j)

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------
    # 6️⃣ Second Layer‑Norm over hidden dimension
    # --------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------------
    # 7️⃣ Apply the output gate (element‑wise) and final linear projection
    # --------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    # The reference implementation returns float32; ensure the dtype.
    return out.to(torch.float32)
