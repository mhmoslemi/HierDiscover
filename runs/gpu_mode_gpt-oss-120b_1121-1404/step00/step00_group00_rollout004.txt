@triton.jit
def trimul_full_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    # compute i and j block indices
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n
    
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # For each hidden tile within H:
    for d_start in range(0, H, BLOCK_D):
        # allocate accumulator and initialize zero
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)
        # loop over k
        for k_start in range(0, N, BLOCK_K):
            # compute valid k tile size
            cur_K = tl.minimum(BLOCK_K, N - k_start)
            offs_k = k_start + tl.arange(0, cur_K)

            # iterate over d within tile
            for di in range(0, tl.minimum(BLOCK_D, H - d_start)):
                d = d_start + di
                # load left (M,K)
                left_ptrs = left_ptr + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + d * stride_left_h + 0 * stride_left_b
                left = tl.load(left_ptrs, mask=(offs_i[:, None] < N) & (offs_k[None, :] < N), other=0.0).to(tl.float32) 
                # load right transposed (K,N) use stride_right_k for rows (k) and stride_right_j for columns (j)

                right = tl.load(right_ptrs, mask=(offs_k[:, None] < N) & (offs_j[None, :] < N), other=0.0).to(tl.float32)

                acc[:, :, di] += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)
        # store accumulator to output
```

"""
TriMul (outgoing) forward pass implementation.

The heavy 4‑D tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is executed with a custom Triton kernel.  The kernel works on
tiles of the (i,j) plane (BLOCK_M × BLOCK_N) and accumulates
over the k dimension (BLOCK_K) for a small block of the
hidden dimension (BLOCK_D).  The remaining network
operations (LayerNorm, linear projections, gates) are performed
with PyTorch for simplicity.

The implementation follows the reference PyTorch module
and supports optional masking, arbitrary batch size ≤2,
sequence lengths up to 1024, hidden dimensions 128,
384 or 768 and both float32 and float16 arithmetic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the Einstein‑summation:
#   out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,          # *Pointer to left tensor (B,N,N,H) in FP16
    right_ptr,         # *Pointer to right tensor (B,N,N,H) in FP16
    out_ptr,           # *Pointer to output tensor (B,N,N,H) in FP32
    N,                 # sequence length (int32)
    H,                 # hidden dimension (int32)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides (j = i dimension)
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # output strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Compute the TriMul contraction for one batch element."""

    # ------------------------------------------------------------------
    # Program IDs (i‑tile, j‑tile, batch)
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)          # block index over rows   (i)
    pid_n = tl.program_id(1)          # block index over cols   (j)
    pid_b = tl.program_id(2)          # batch index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds i/j
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Base pointers for the current batch
    left_base  = left_ptr  + pid_b * stride_l_b
    right_base = right_ptr + pid_b * stride_r_b
    out_base   = out_ptr   + pid_b * stride_o_b

    # Number of tiles along K and H
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    num_d_tiles = tl.cdiv(H, BLOCK_D)

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles
    # ------------------------------------------------------------------
    for d_tile in range(num_d_tiles):
        d_start = d_tile * BLOCK_D                     # first hidden index in this tile
        cur_D = tl.minimum(BLOCK_D, H - d_start)      # actual width of this tile

        # accumulator shape (BLOCK_M, BLOCK_N, BLOCK_D) – extra lanes are safe
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over K tiles
        # ------------------------------------------------------------------
        for k_tile in range(num_k_tiles):
            k_start = k_tile * BLOCK_K
            cur_K = tl.minimum(BLOCK_K, N - k_start)
            offs_k = k_start + tl.arange(0, cur_K)

            # ----------------------------------------------------------------
            # Loop over hidden lanes inside the tile (unrolled)
            # ----------------------------------------------------------------
            for di in range(BLOCK_D):
                d = d_start + di
                if d >= H:
                    continue

                # ----- left[i,k,d] -------------------------------------------------
                left_ptrs = (left_base
                             + offs_i[:, None] * stride_l_i
                             + offs_k[None, :] * stride_l_k
                             + d * stride_l_h)
                left = tl.load(left_ptrs,
                               mask=(mask_i[:, None] & (offs_k[None, :] < N)),
                               other=0.0,
                               dtype=tl.float16).to(tl.float32)

                # ----- right[j,k,d]  (we need the k‑major layout for dot) -----
                right_ptrs = (right_base
                              + offs_k[:, None] * stride_r_k
                              + offs_j[None, :] * stride_r_j
                              + d * stride_r_h)
                right = tl.load(right_ptrs,
                                mask=((offs_k[:, None] < N) & mask_j[None, :]),
                                other=0.0,
                                dtype=tl.float16).to(tl.float32)

                # ----- Accumulate -------------------------------------------------
                acc[:, :, di] += tl.dot(left, right)   # (BLOCK_M, BLOCK_N)

        # ------------------------------------------------------------------
        # Write the tile back to global memory
        # ------------------------------------------------------------------
        for di in range(BLOCK_D):
            d = d_start + di
            if d >= H:
                continue
            out_ptrs = (out_base
                        + offs_i[:, None] * stride_o_i
                        + offs_j[None, :] * stride_o_j
                        + d * stride_o_h)
            tl.store(out_ptrs, acc[:, :, di], mask=mask_ij)


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask_tensor  : torch.Tensor [B, N, N]  (may be None)
        - weights_dict : mapping from module parameter names to tensors
        - config_dict  : {'dim': int, 'hidden_dim': int}
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Helper: LayerNorm (mirrors nn.LayerNorm)
    # ------------------------------------------------------------------
    def layer_norm(x, weight, bias, eps=1e-5):
        # x is (..., C); weight / bias are (C,)
        mean = x.mean(dim=-1, keepdim=True)
        var = x.var(dim=-1, unbiased=False, keepdim=True)
        x_norm = (x - mean) * torch.rsqrt(var + eps)
        return x_norm * weight + bias

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"]
    )   # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"].to(x.dtype))
    right_proj = F.linear(x, weights["right_proj.weight"].to(x.dtype))

    # ------------------------------------------------------------------
    # 3) Optional masking
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)   # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid of a linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(x.dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(x.dtype)))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(x.dtype)))

    # Apply gates
    left = left_proj * left_gate          # [B,N,N,H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy contraction via Triton
    # ------------------------------------------------------------------
    # Cast the large tensors to half‑precision to reduce bandwidth.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (float32 as required by subsequent layers)
    out_buf = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_buf.stride()

    # Tile sizes – chosen to stay within register limits on H100.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 4   # hidden‑dim tiling

    grid = (
        triton.cdiv(N, BLOCK_M),   # program_id(0) – rows (i)
        triton.cdiv(N, BLOCK_N),   # program_id(1) – cols (j)
        B                          # program_id(2) – batch dimension
    )

    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_buf,
        N,
        hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating
    # ------------------------------------------------------------------
    out_norm = layer_norm(
        out_buf,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"]
    )   # still [B,N,N,H]

    # Apply the outgoing gate (broadcast on the hidden dimension)
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to model dimension
    # ------------------------------------------------------------------
    output = F.linear(out_norm, weights["to_out.weight"].to(out_norm.dtype))

    return output
