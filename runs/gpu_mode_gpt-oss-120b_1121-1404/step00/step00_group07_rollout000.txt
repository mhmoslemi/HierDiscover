@triton.jit
def triton_trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_ba, stride_ia, stride_ka,
    stride_bb, stride_jb, stride_kb,
    stride_bc, stride_ic, stride_jc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i block
    pid_n = tl.program_id(1)   # j block
    pid_z = tl.program_id(2)   # combined batch and hidden index

    # decode batch and hidden from pid_z
    num_batches = B
    # for use: batch = pid_z // H
    # hidden = pid_z % H
    batch = pid_z // H
    hidden = pid_z % H

    # Compute offsets for block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # off_k will be tiled in loop

    # mask out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers offset base
    a_ptrs = a_ptr + batch * stride_ba + hidden + offs_m[:, None] * stride_ia + tl.arange(0, BLOCK_K)[None, :] * stride_ka
    b_ptrs = b_ptr + batch * stride_bb + hidden + offs_n[:, None] * stride_jb + tl.arange(0, BLOCK_K)[None, :] * stride_kb

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load A and B tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & (k*BLOCK_K + tl.arange(0, BLOCK_K) < N), other=0.0)
        # dot product across k dimension (inner)
        # a shape: (BLOCK_M, BLOCK_K), b shape: (BLOCK_N, BLOCK_K)
        # we need sum over k: a[i,k] * b[j,k] -> perform outer product?
        # Use tl.dot on a @ b.T where b is transposed: we need b transposed shape (BLOCK_K, BLOCK_N)
        # with tl.dot supports matrix multiplication: a @ b.T (if b shape (BLOCK_N, BLOCK_K) we can use tl.dot(a, b, ...) with appropriate transposes)
        acc += tl.dot(a, b, trans_b=True)  # a (M,K), b (N,K) -> dot with trans_b => (M,N) sums over K
        # Move pointers
        a_ptrs += BLOCK_K * stride_ka
        b_ptrs += BLOCK_K * stride_kb

    # store result
    c_ptrs = c_ptr + batch * stride_bc + hidden + offs_m[:, None] * stride_ic + offs_n[None, :] * stride_jc
    # mask for store
    mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    # Step 1: LayerNorm on input
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]; maybe shape [dim]
    # compute normalized input: (x - mean)/sqrt(var+eps)*weight + bias. Use torch.nn.functional.layer_norm
    eps = 1e-5  # typical
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Step 2: linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # apply mask if needed
    if not config.get("nomask", True):
        mask = mask.to(left.dtype) # shape [B,N,N] -> unsqueeze -1 maybe
        left = left * mask.unsqueeze(-1)
        right = right * mask.unsqueeze(-1)
    else:
        # if no mask, mask may be None or default; (skip)
        pass
    # step3: gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # step4: triton kernel for out = left @ right^T per hidden dim
    # left, right are float32. Let's keep them.
    B, N, _, _ = left.shape  # shape B,N,N,hidden_dim
    # ensure contiguous to have expected stride (maybe .contiguous())
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]
    # call kernel
    # compute strides
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right.stride()
    stride_bc, stride_ic, stride_jc, stride_hc = out.stride()
    # configure block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B * hidden_dim)
    triton_trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_bl, stride_il, stride_kl,
        stride_br, stride_jr, stride_kr,
        stride_bc, stride_ic, stride_jc,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4, # typical
    )
    # step5: final layernorm over hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)
    # multiply by out_gate
    out = out * out_gate
    # step6: final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) – Triton‐accelerated forward pass.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C}
and computes for each hidden head h ∈ {0,…,H-1}:

    out[b,i,j,h] = Σ_k  ( (X·W_left)[b,i,k,h] ⊙ σ((X·W_left_gate)[b,i,k,h]) )
                  × ( (X·W_right)[b,j,k,h] ⊙ σ((X·W_right_gate)[b,j,k,h]) )

The heavy O(N³·H) part (the sum over k) is performed with a custom Triton
kernel that evaluates a batched matrix multiplication
      left_h @ right_hᵀ  for every (b, h).  The rest (LayerNorm,
projections, sigmoid gates and the final linear layer) are done in
regular PyTorch.

The kernel processes a tile of (M,N) output elements per program and
loops over K‑tiles.  The program grid is (⌈N/M⌉, ⌈N/N⌉, B·H) where the
third dimension fuses batch and head indices, avoiding any extra
synchronisation.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    a_ptr,                     # left  : [B, N, N, H]
    b_ptr,                     # right : [B, N, N, H]
    c_ptr,                     # out   : [B, N, N, H]
    B, N, H,                   # runtime sizes
    stride_ab, stride_ai, stride_ak,   # a strides (batch, i, k)
    stride_bb, stride_bj, stride_bk,   # b strides (batch, j, k)
    stride_cb, stride_ci, stride_cj,   # c strides (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for a given (batch, head) pair:
        C[i, j] = Σ_k A[i, k] * B[j, k]      (note the transpose on B)
    where A/B have shape (N, N) and the sum is over the second dimension.
    """
    pid_m = tl.program_id(0)               # i tile
    pid_n = tl.program_id(1)               # j tile
    pid_z = tl.program_id(2)               # fused batch * head

    # decode batch and head from pid_z
    batch = pid_z // H
    head  = pid_z % H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Base pointers that include batch and head offset (head stride is 1)
    a_base = a_ptr + batch * stride_ab + head
    b_base = b_ptr + batch * stride_bb + head
    c_base = c_ptr + batch * stride_cb + head

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = kp * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)          # k indices
        mask_k = offs_k < N

        # Pointers to the current K‑tile of A and B
        a_ptrs = a_base + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak
        b_ptrs = b_base + offs_n[:, None] * stride_bj + offs_k[None, :] * stride_bk

        # Load with masking (out‑of‑bounds entries are set to 0)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Perform dot product over K: a (M,K) * b (N,K).transpose() → (M,N)
        acc += tl.dot(a, b, trans_b=True)

    # ------------------------------------------------------------------
    # Write out the result
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    store_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=store_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N]  (may be ignored)
        - weights      : dict of torch.Tensors containing all linear / LN params
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    torch.cuda.set_device(torch.cuda.current_device())
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections to hidden dimension
    # --------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → broadcast on hidden dim
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4. Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy O(N³·H) multiplication via Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim
    left  = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape [B, N, N, hidden_dim]

    # Strides (in elements, not bytes)
    stride_ab, stride_ai, stride_ak, stride_ah = left.stride()
    stride_bb, stride_bj, stride_bk, stride_bh = right.stride()
    stride_cb, stride_ci, stride_cj, stride_ch = out.stride()

    # We only need batch, i, k strides for A and batch, j, k strides for B.
    # Hidden dimension stride is 1, so we add the head index directly.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    triton_trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_ab,
        stride_ai,
        stride_ak,
        stride_bb,
        stride_bj,
        stride_bk,
        stride_cb,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    torch.cuda.synchronize()

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # [B,N,N,hidden_dim]

    out = out * out_gate  # apply output gate

    to_out_w = weights["to_out.weight"]  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)  # final projection → [B,N,N,dim]

    return out
