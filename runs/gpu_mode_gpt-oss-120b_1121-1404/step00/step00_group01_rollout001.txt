```
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # block id along M
    pid_j = tl.program_id(1)  # block id along N
    pid_s = tl.program_id(2)  # slice id across B*H
    
    # Determine batch and hidden indices
    b = pid_s // H
    h = pid_s % H

    # Compute base offsets for i and j
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        # left_ptrs shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k + h * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed (K x N)
        # We want B_T shape (BLOCK_K, BLOCK_N) where rows = k, cols = j
        right_ptrs = right_ptr + b * stride_right_b + offs_j[None, :] * stride_right_i + offs_k[:, None] * stride_right_k + h * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

        # Compute block matmul
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + b * stride_out_b + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) with a custom Triton kernel.

The heavy part of the operator is the 4‑D Einstein sum

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is O(N³·H).  The kernel implements a tiled batched matrix‐multiply
(left @ rightᵀ) over the batch × hidden dimension (B·H) using
standard Triton block‑wise dot products.  All other linear layers,
LayerNorms and sigmoid gates are performed with PyTorch for simplicity,
but the core O(N³) contraction is fully fused in Triton.

The kernel works for any batch size (B∈{1,2}), sequence length N∈{128,…,1024},
model dimension `dim` and hidden dimension `hidden_dim` (typically 128).
Masking is optional (config["nomask"] = True/False).  The final output
has shape (B, N, N, dim) and dtype float32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr,     # *float16  [B, N, N, H]
    right_ptr,    # *float16  [B, N, N, H]
    out_ptr,      # *float32  [B, N, N, H]
    # sizes
    B, N, H,
    # strides for left  (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for output
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Tiling:
        i → BLOCK_M   (rows)
        j → BLOCK_N   (cols)
        k → BLOCK_K   (reduction)
    The third program‑id (PID 2) indexes the combined batch‑hidden dimension
    (slice = b * H + h).
    """
    pid_i = tl.program_id(0)          # tile over i (rows)
    pid_j = tl.program_id(1)          # tile over j (cols)
    pid_s = tl.program_id(2)          # slice = b*H + h

    # --------------------------------------------------------------
    # Decode slice into batch and hidden indices
    b = pid_s // H
    h = pid_s % H

    # --------------------------------------------------------------
    # Tile start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Offsets inside the current tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load left tile: shape (BLOCK_M, BLOCK_K)
        # left[b, i, k, h]
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)        # fp16

        # ----- Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # We need right_T[k, j] = right[b, j, k, h]
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_j[None, :] * stride_right_i    # j dimension uses stride_i
            + offs_k[:, None] * stride_right_k    # k dimension uses stride_k
            + h * stride_right_h
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)       # fp16

        # ----- Multiply–accumulate (fp16 → fp32)
        acc += tl.dot(left, right)   # (M×K)·(K×N) → (M×N)

    # --------------------------------------------------------------
    # Store the result
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Entry point for the TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask:        torch.Tensor [B, N, N] (bool/int/float) – optional
            - weights: dict of torch.Tensors containing all model parameters
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    input_tensor, mask, weights, config = data
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square in the pairwise dimensions"
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Layer‑norm on the input (dim‑wise)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (dim → hidden_dim) – no bias
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    if not config.get("nomask", True):
        # mask: [B, N, N] → [B, N, N, 1]
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating values (sigmoid) – keep them in float16 to save bandwidth
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # keep fp32

    # Apply gates
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the O(N³·H) contraction
    left_f16 = left.contiguous()
    right_f16 = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides in elements (not bytes)
    s_lb = left_f16.stride(0)
    s_li = left_f16.stride(1)
    s_lk = left_f16.stride(2)
    s_lh = left_f16.stride(3)

    s_rb = right_f16.stride(0)
    s_ri = right_f16.stride(1)
    s_rk = right_f16.stride(2)
    s_rh = right_f16.stride(3)

    s_ob = out.stride(0)
    s_oi = out.stride(1)
    s_oj = out.stride(2)
    s_oh = out.stride(3)

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * hidden_dim,                   # slice dimension (batch × hidden)
    )

    trimul_outgoing_kernel[grid](
        left_f16,
        right_f16,
        out,
        B, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                      # H100 works well with 8 warps per block
    )

    # ------------------------------------------------------------------
    # Post‑processing: norm → gate → linear
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # out_gate: [B, N, N, H] (fp32)

    out = F.linear(out, weights["to_out.weight"])  # (hidden_dim → dim)

    return out
