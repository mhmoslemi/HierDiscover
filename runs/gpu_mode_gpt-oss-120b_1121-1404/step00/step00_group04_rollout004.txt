def custom_kernel(data):
   input_tensor, mask, weights, config = data
   # extract dims
   dim = config["dim"]; hidden_dim = config["hidden_dim"]; seq_len = config["seqlen"]; bs = config["bs"]; nomask = config["nomask"]

   # Ensure input dtype = torch.float32 (if not, cast)
   x = input_tensor

   # LayerNorm
   # norm weight and bias of shape dim
   norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
   # use torch.nn.functional.layer_norm
   x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

   # Linear projections left_proj and right_proj: weight shape [hidden_dim, dim]
   left_proj_weight = weights["left_proj.weight"]
   right_proj_weight = weights["right_proj.weight"]
   left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape [B,N,N,hidden]
   right = torch.nn.functional.linear(x_norm, right_proj_weight)
   # Mask (if present)
   if not nomask:
        mask_ = mask.unsqueeze(-1).type_as(x_norm)
        left = left * mask_
        right = right * mask_
   else:
        # mask may be None or we ignore it
        pass

   # Gates
   left_gate_weight = weights["left_gate.weight"]
   right_gate_weight = weights["right_gate.weight"]
   out_gate_weight = weights["out_gate.weight"]
   left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
   right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
   out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

   left = left * left_gate
   right = right * right_gate

   # Optionally convert to fp16 for speed
   left_fp16 = left.to(torch.float16)
   right_fp16 = right.to(torch.float16)

   # TriMul kernel
   out_hidden_fp16 = tri_mul_forward(left_fp16, right_fp16) # returns [B,N,N,hidden] fp16

   out_hidden = out_hidden_fp16.to(torch.float32) # convert back

   # Apply to_out_norm (layernorm) with hidden_dim
   to_out_norm_weight = weights["to_out_norm.weight"]
   to_out_norm_bias = weights["to_out_norm.bias"]
   out_hidden_norm = torch.nn.functional.layer_norm(out_hidden, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

   # multiply by out_gate
   out_hidden_gate = out_hidden_norm * out_gate

   # Linear to out dim
   to_out_weight = weights["to_out.weight"]
   output = torch.nn.functional.linear(out_hidden_gate, to_out_weight)

   return output

@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, H, N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(2)  # batch*head id
    pid_m = tl.program_id(0)      # tile row
    pid_n = tl.program_id(1)      # tile col

    # compute batch index and head index
    batch_idx = pid_batch // H
    head_idx = pid_batch % H

    # offsets for output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-range
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch/head
    a_base = a_ptr + batch_idx * stride_a_batch + head_idx
    b_base = b_ptr + batch_idx * stride_b_batch + head_idx
    c_base = c_ptr + batch_idx * stride_c_batch + head_idx

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        # k offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # compute masks for k dimension
        mask_k = offs_k < N

        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B_T: shape (BLOCK_K, BLOCK_N)
        # B_T entry [k, j] = B[j, k] in original memory
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(a, b)

    # store result
    c_ptrs = c_base + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    tl.store(c_ptrs, acc.to(tl.float16) if tl.dtype(c_ptrs.element_type) == tl.float16 else acc,
             mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input (last dimension = dim).
2. Linear projections `left` and `right` (dim → hidden_dim) and apply the optional mask.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) and
   element‑wise multiply `left`/`right` with their respective gates.
4. Heavy part – the pairwise inner‑product over the sequence dimension:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication A @ Bᵀ for each (batch, head) pair.
   It is realised by a Triton kernel that tiles the matrices (BLOCK_M×BLOCK_N)
   and iterates over K‑tiles.
5. Layer‑norm the result over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.

The kernel works with arbitrary batch size (≤2), sequence length (≤1024) and
hidden dimension (≤384).  Input/weights are kept in FP32; the large matrix
multiplication is performed in FP16 for higher bandwidth while accumulating
in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: out = A @ Bᵀ   (A: [M,K], B: [N,K] → output [M,N])
# The kernel is launched with a 3‑D grid:
#   pid_m  – tile row index
#   pid_n  – tile column index
#   pid_bn – packed (batch, head) index = batch * hidden_dim + head
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    B, H, N,                            # B=batch, H=heads (hidden_dim), N=seq_len
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_bn = tl.program_id(2)         # packed batch*head

    # ------------------------------------------------------------------
    # Decode packed batch / head
    # ------------------------------------------------------------------
    batch_idx = pid_bn // H
    head_idx = pid_bn % H

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Base pointers for this (batch, head) pair
    # ------------------------------------------------------------------
    a_base = a_ptr + batch_idx * stride_a_batch + head_idx          # head dim stride = 1
    b_base = b_ptr + batch_idx * stride_b_batch + head_idx
    c_base = c_ptr + batch_idx * stride_c_batch + head_idx

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis K
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)   # cast to FP32

        # Bᵀ tile: we need Bᵀ[k, j] = B[j, k]
        # Load B with swapped strides: row = k (original col), col = j (original row)
        b_ptrs = b_base + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Dot product accumulation
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the tile
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_j
    # Output tensor dtype is the same as the input (fp16), so cast back
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_fp16(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the core TriMul operation in FP16 using the Triton kernel.
    `left` and `right` are tensors of shape [B, N, N, H] and dtype FP16.
    Returns a tensor of the same shape and dtype.
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides expressed in number of elements (not bytes)
    stride_a_batch = left.stride(0)
    stride_a_i = left.stride(1)
    stride_a_k = left.stride(2)

    stride_b_batch = right.stride(0)
    stride_b_i = right.stride(1)
    stride_b_k = right.stride(2)

    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)

    # Tile sizes – 64 works well on H100 for the tested range
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M ),   # rows
        ( (N + BLOCK_N - 1) // BLOCK_N ),   # cols
        B * H                               # batch*head
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, H, N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,  # H100 can handle many warps
    )
    return out


# ----------------------------------------------------------------------
# Main entry point expected by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask         : [B, N, N]       (bool / float, optional)
        - weights      : dict of tensors required by the module
        - config       : dict containing "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (last dim = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]        # [hidden_dim, dim]

    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim] (FP32)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid after linear, no bias)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 and invoke the Triton kernel
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_hidden_fp16 = _tri_mul_fp16(left_fp16, right_fp16)   # [B,N,N,hidden_dim] (fp16)

    # Cast back to FP32 for the remaining (LayerNorm, final linear)
    out_hidden = out_hidden_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_hidden = F.layer_norm(out_hidden,
                              normalized_shape=(hidden_dim,),
                              weight=to_out_norm_weight,
                              bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 7. Output gate multiplication
    # ------------------------------------------------------------------
    out_hidden = out_hidden * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]  # [dim, hidden_dim]
    output = F.linear(out_hidden, to_out_weight)  # [B, N, N, dim]

    return output
