"""
TriMul (outgoing) kernel implementation in Triton.

Algorithm
---------
1. Layer‑norm the input tensor over the last feature dimension.
2. Linear projections for the left/right tensors and separate linear
   layers for the left/right/out gates (all without bias).  Apply a
   sigmoid to the gates.
3. Mask (if provided) is applied to the projected tensors before gating.
4. Fuse the three element‑wise operations (proj * mask * gate) using
   normal PyTorch (the heavy part is the bilinear reduction, not the
   element‑wise ops).
5. Compute the core “TriMul” reduction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   for all batches and hidden channels d.
   This is a batched matrix multiplication where each head (d) performs
   N×N @ N×Nᵀ.  The kernel tiles the M (i) and N (j) axes and loops over
   K (k) in blocks, accumulating in fp32.
6. Layer‑norm the result over the hidden dimension, apply the out‑gate,
   and a final linear projection back to the original feature dimension.

The kernel works for any sequence length N (up to at least 1024) and
hidden dimension up to 128, and runs efficiently on an H100.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul reduction
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # sizes (all equal to seq_len)
    stride_am, stride_ak, stride_ah,    # A: (head, M, K) strides
    stride_bk, stride_bn, stride_bh,    # B: (head, K, N) strides
    stride_cm, stride_cn, stride_ch,    # C: (head, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each head compute C = A @ B   (A: M×K, B: K×N)."""
    pid_m = tl.program_id(0)            # tile over M dimension
    pid_n = tl.program_id(1)            # tile over N dimension
    pid_h = tl.program_id(2)            # head index (batch * hidden_dim)

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for the current head (no k offset yet)
    a_base = A_ptr + pid_h * stride_ah + offs_m[:, None] * stride_am
    b_base = B_ptr + pid_h * stride_bh + offs_n[None, :] * stride_bn

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in BLOCK_K steps
    # --------------------------------------------------------------
    k = 0
    while k < K:
        cur_k = k + offs_k                     # (BLOCK_K,)
        mask_k = cur_k < K

        # Load A block (M x K_tile)
        a = tl.load(
            a_base + cur_k[None, :] * stride_ak,
            mask=(offs_m[:, None] < M) & mask_k[None, :],
            other=0.0,
        )
        # Load B block (K_tile x N)
        b = tl.load(
            b_base + cur_k[:, None] * stride_bk,
            mask=mask_k[:, None] & (offs_n[None, :] < N),
            other=0.0,
        )
        # Rank‑K update
        acc += tl.dot(a, b)

        k += BLOCK_K

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    c_ptrs = C_ptr + pid_h * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)

        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gating logits
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Project
    left = torch.nn.functional.linear(x, left_proj_weight)      # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)    # [B,N,N,hidden]

    # Mask handling (if mask is None treat as all‑ones)
    if mask is None:
        mask_exp = torch.ones_like(left[..., :1])
    else:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)           # [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # Gates (sigmoid activation)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul reduction via Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # batch size and sequence length

    # Bring hidden dimension to the second axis and make tensors contiguous
    # Shape after permute: (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right on the last two dims (K ↔ J) as required by the einsum
    right_T = right_perm.transpose(-2, -1).contiguous()

    # Collapse batch and hidden dimensions into a single “head” dimension
    head = B * hidden_dim
    left_flat = left_perm.view(head, N, N)
    right_flat = right_T.view(head, N, N)
    out_flat = torch.empty_like(left_flat)   # will hold the result

    # Strides for the three tensors (head, M, K) / (head, K, N) / (head, M, N)
    stride_ah = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bh = right_flat.stride(0)
    stride_bk = right_flat.stride(1)   # K‑dimension
    stride_bn = right_flat.stride(2)   # N‑dimension

    stride_ch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)     # M‑dimension
    stride_cn = out_flat.stride(2)     # N‑dimension

    # Block sizes – chosen to fit in shared memory / registers on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (tiles over M, tiles over N, number of heads)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        head,
    )

    # Launch the kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,          # M
        N,          # N
        N,          # K
        stride_am,
        stride_ak,
        stride_ah,
        stride_bk,
        stride_bn,
        stride_bh,
        stride_cm,
        stride_cn,
        stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Post‑processing: normalization, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # back to [B,N,N,dim]

    return out
```
