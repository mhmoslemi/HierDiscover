"""
custom_kernel for the outgoing Triangular Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input pair representation (last dim = dim).
2. Linear projections → left/right (shape [B,N,N,H]) and three gate
   projections (shape [B,N,N,H]).
3. Apply the optional mask (broadcast on the hidden dimension) to left/right.
4. Apply sigmoid gates to left/right and multiply.
5. Compute the core tensor contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiply
        out_d = left_d @ right_d^T  for each hidden dimension d.
   This step is performed with a fused Triton GEMM kernel that works on
   FP16 inputs and accumulates in FP32.
6. Layer‑norm the intermediate result (hidden dim H).
7. Apply the output gate (sigmoid) and a final linear projection back to dim.

The kernel is written for NVIDIA H100 (Triton 3.3.1) and uses mixed‑precision
computations for maximum throughput while returning a FP32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ B   (A : (B, M, K), B : (B, K, N))
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,                     # *half  : (batch, M, K)
    B_ptr,                     # *half  : (batch, K, N)
    C_ptr,                     # *float : (batch, M, N)
    M, N, K,                   # matrix sizes (int)
    stride_batch_a, stride_m_a, stride_k_a,   # strides for A
    stride_batch_b, stride_k_b, stride_n_b,   # strides for B
    stride_batch_c, stride_m_c, stride_n_c,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    # Tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers for the current tile of A and B
        a_ptrs = (A_ptr
                  + batch_id * stride_batch_a
                  + offs_m[:, None] * stride_m_a
                  + offs_k[None, :] * stride_k_a)
        b_ptrs = (B_ptr
                  + batch_id * stride_batch_b
                  + offs_k[:, None] * stride_k_b
                  + offs_n[None, :] * stride_n_b)

        # Mask out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # Accumulate the product (FP16 × FP16 → FP32)
        acc += tl.dot(a, b)

    # Write the result back to C
    c_ptrs = (C_ptr
              + batch_id * stride_batch_c
              + offs_m[:, None] * stride_m_c
              + offs_n[None, :] * stride_n_c)
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


# ----------------------------------------------------------------------
# Helper: batched matmul for the TriMul contraction
# ----------------------------------------------------------------------
def triton_tri_mul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] using Triton.
    Parameters
    ----------
    left, right : torch.Tensor
        Shape (B, N, N, H) with dtype torch.float16.
    Returns
    -------
    out : torch.Tensor
        Shape (B, N, N, H) with dtype torch.float32.
    """
    B, N, _, H = left.shape
    # Permute to (B, H, N, N) and flatten the leading two dims
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(-1, N, N)              # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)            # (B*H, N, N)
    right_t = right_flat.transpose(1, 2).contiguous()  # (B*H, N, N)

    # Output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Strides
    stride_batch_a = left_flat.stride(0)
    stride_m_a = left_flat.stride(1)
    stride_k_a = left_flat.stride(2)

    stride_batch_b = right_t.stride(0)
    stride_k_b = right_t.stride(1)
    stride_n_b = right_t.stride(2)

    stride_batch_c = out_flat.stride(0)
    stride_m_c = out_flat.stride(1)
    stride_n_c = out_flat.stride(2)

    # Tile sizes – tuned for H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        left_flat.shape[0],                      # batch dimension (B*H)
        (N + BLOCK_M - 1) // BLOCK_M,            # rows
        (N + BLOCK_N - 1) // BLOCK_N,            # cols
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_t,
        out_flat,
        N,
        N,
        N,
        stride_batch_a,
        stride_m_a,
        stride_k_a,
        stride_batch_b,
        stride_k_b,
        stride_n_b,
        stride_batch_c,
        stride_m_c,
        stride_n_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Input:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Output:
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    # Expected config keys
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract model parameters (all tensors are on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Initial LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → left/right + gates
    # ------------------------------------------------------------------
    # Shape after projection: (B, N, N, hidden_dim)
    left = F.linear(x, left_proj_weight)          # (B,N,N,H)
    right = F.linear(x, right_proj_weight)        # (B,N,N,H)

    left_gate_pre = F.linear(x, left_gate_weight)   # (B,N,N,H)
    right_gate_pre = F.linear(x, right_gate_weight) # (B,N,N,H)
    out_gate_pre = F.linear(x, out_gate_weight)     # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if config.get("nomask", False):
        mask = torch.ones_like(mask_tensor, dtype=left.dtype, device=left.device)
    else:
        mask = mask_tensor.to(left.dtype)
    mask = mask.unsqueeze(-1)  # (B,N,N,1)

    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core triangular multiplicative update via Triton
    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy GEMM, keep the output FP32
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out = triton_tri_mul(left_fp16, right_fp16)   # (B,N,N,H), dtype=float32

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim (no bias)
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
