"""
TriMul “outgoing” operator (AlphaFold3) – fast Triton implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x`  (last dimension = dim).
2. Four linear projections (no bias) from dim → hidden_dim:
   * left_proj   = x @ W_left
   * right_proj  = x @ W_right
   * left_gate   = sigmoid(x @ W_left_gate)
   * right_gate  = sigmoid(x @ W_right_gate)
   * out_gate    = sigmoid(x @ W_out_gate)    # shape [B,N,N,hidden_dim]
3. Apply optional mask (broadcast on the channel dim) to the two
   “proj” tensors and multiply by the corresponding gates:
   left  = left_proj  * left_gate  * mask
   right = right_proj * right_gate * mask
4. Compute the core tensor contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix‑multiply:
       left  → (B, H, M=N, K=N)
       right → (B, H, K=N, N=N)   (transposed on the last two axes)
   We fuse this using a custom Triton kernel that processes one
   (batch * hidden) pair per program and uses block‑wise dot‑products.
5. Apply a second LayerNorm over the hidden dimension,
   multiply by `out_gate`, and finally project back to `dim`
   with a linear layer (hidden_dim → dim).

All heavy work (the N³ contraction) is performed by Triton; the
remaining linear layers, sigmoid, and layer‑norms are delegated to
highly‑optimised PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from torch.nn.functional import linear, layer_norm, sigmoid


@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul C = A @ B  (A: [batch,M,K], B: [batch,K,N], C: [batch,M,N])
    The batch dimension is supplied as the third program id."""
    pid_m = tl.program_id(1)               # row tile index
    pid_n = tl.program_id(0)               # column tile index
    pid_b = tl.program_id(2)               # batch*hidden index

    # Offsets for each tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for boundary handling
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute base pointers for this batch
    a_batch = pid_b * stride_a_batch
    b_batch = pid_b * stride_b_batch
    c_batch = pid_b * stride_c_batch

    # Accumulator in FP32 (works for FP16 inputs as well)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k
        mask_k = cur_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = (A_ptr + a_batch
                  + offs_m[:, None] * stride_a_m
                  + cur_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = (B_ptr + b_batch
                  + cur_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot‑product accumulation
        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = (C_ptr + c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module implemented with a
    Triton kernel for the N³ contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as
        the inputs).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # Layer‑norm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    # Projections
    left_proj  = linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = linear(x, weights["right_proj.weight"])

    # Gates (sigmoid)
    left_gate  = sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = sigmoid(linear(x, weights["out_gate.weight"]))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask handling
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        # No mask → use ones
        mask_ = torch.ones_like(input_tensor[..., 0:1])  # shape [B,N,N,1]
    else:
        mask_ = mask.unsqueeze(-1).to(dtype)  # broadcast on the channel dim

    left = left_proj * mask_ * left_gate
    right = right_proj * mask_ * right_gate

    # ------------------------------------------------------------------
    # Rearrange tensors for batched matmul:
    #   left  : (B, H, M=N, K=N)
    #   right : (B, H, K=N, N=N)  (note the transposition on the last two axes)
    # ------------------------------------------------------------------
    # left: [B,N,N,H] → [B,H,N,N]  (M=i, K=k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: [B,N,N,H] → [B,H,K=N,N] where K is the third axis (original k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape          # H == hidden_dim
    batch_h = B * H                        # fused batch * hidden dimension

    # Flatten the first two dimensions so that each (b, h) pair is a separate batch
    left_flat  = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)     # shape (batch_h, N, N)

    # ------------------------------------------------------------------
    # Triton kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_N - 1) // BLOCK_N,   # program_id(0) – column tiles
        (N + BLOCK_M - 1) // BLOCK_M,   # program_id(1) – row tiles
        batch_h                         # program_id(2) – batch*hidden
    )

    # Strides (in element counts, not bytes)
    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_flat.stride()
    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm over hidden_dim, gate, and final projection
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = linear(out, weights["to_out.weight"])

    return out
