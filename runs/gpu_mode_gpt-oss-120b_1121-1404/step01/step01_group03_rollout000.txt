"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The operator computes, for each batch element `b` and hidden channel `h`:

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are linear projections of the layer‑normalised input
followed by element‑wise sigmoidal gating and optional masking.
The heavy N³ reduction is performed by a batched GEMM kernel that treats the
pair (batch, hidden) as a single “page” dimension and multiplies a (N,N)
matrix with the transpose of another (N,N) matrix.  The kernel works in
float16 and accumulates in float16 (Tensor‑Core friendly), the result is cast
back to float32 for the remaining LayerNorm, gating and output projection.

The implementation keeps all non‑linear operations (LayerNorm, Linear,
sigmoid, final LayerNorm & projection) in PyTorch – only the cubic
multiplication is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_kernel(
    a_ptr,                     # const *half  (left)
    b_ptr,                     # const *half  (right)
    c_ptr,                     # *half        (output)
    batch_hidden, N,           # int32 scalar
    stride_az, stride_ai, stride_ak,   # strides for A (page, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (page, k, n) – B is accessed transposed
    stride_cz, stride_ci, stride_cj,   # strides for C (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a block of C = A @ Bᵀ for `batch_hidden` independent pages.
    A, B, C have shape (batch_hidden, N, N) in row‑major order.
    """
    pid_z = tl.program_id(0)          # page index   (batch * hidden)
    pid_m = tl.program_id(1)          # block‑row
    pid_n = tl.program_id(2)          # block‑col

    # -------------------------------------------------- tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this page
    a_page_ptr = a_ptr + pid_z * stride_az
    b_page_ptr = b_ptr + pid_z * stride_bz
    c_page_ptr = c_ptr + pid_z * stride_cz

    # accumulator (FP16 – uses Tensor‑Core)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # -------------------------------------------------- reduction over K
    cur_k = tl.int32(0)
    while cur_k < N:
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_offs < N

        # ---- load A[i,k]
        a_ptrs = a_page_ptr + \
                 offs_m[:, None] * stride_ai + \
                 cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=tl.float16(0.0))

        # ---- load B[j,k] (as Bᵀ, i.e. B[k,j])
        b_ptrs = b_page_ptr + \
                 cur_k_offs[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=tl.float16(0.0))

        # ---- accumulate
        acc += tl.dot(a, b)                # (BLOCK_M,BLOCK_K) @ (BLOCK_K,BLOCK_N)

        cur_k += BLOCK_K

    # -------------------------------------------------- write back
    c_ptrs = c_page_ptr + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------- unpack
    x, mask, weights, config = data
    # x : [B, N, N, dim]   (float32)
    # mask : [B, N, N]    (bool or float, may be None)
    B, N, _, dim = x.shape
    H = config["hidden_dim"]               # hidden dimension (c_z)

    # ------------------------------------------------------------------- LayerNorm on input
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------- Linear projections (no bias)
    left = F.linear(x_norm, weights["left_proj.weight"])      # [B, N, N, H]
    right = F.linear(x_norm, weights["right_proj.weight"])    # [B, N, N, H]

    # ------------------------------------------------------------------- Optional mask
    if mask is not None:
        # broadcast mask to hidden dim
        mask_exp = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- Gating
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))     # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- Cast to half for the cubic kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Move hidden dim to 2nd axis for page‑wise view: [B, H, N, N]
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # Collapse (B * H) → page dimension handled by Triton
    BH = B * H
    left_mat = left_h.view(BH, N, N)      # [BH, N, N]
    right_mat = right_h.view(BH, N, N)    # [BH, N, N]

    # Output buffer (half)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=x.device)

    # ------------------------------------------------------------------- strides (in elements)
    stride_az = left_mat.stride(0)   # page stride = N*N
    stride_ai = left_mat.stride(1)   # row stride  = N
    stride_ak = left_mat.stride(2)   # col stride  = 1

    stride_bz = right_mat.stride(0)  # same as left
    stride_bk = right_mat.stride(2)  # =1  (k‑stride after transposition)
    stride_bn = right_mat.stride(1)  # =N  (n‑stride after transposition)

    stride_cz = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # ------------------------------------------------------------------- launch configuration
    # Block sizes tuned for H100 Tensor‑Core (FP16) – enough registers for N≤1024
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64                     # larger K‑tile reduces loop count

    grid = (BH,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BH,
        N,
        stride_az, stride_ai, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                 # H100 prefers 8 warps per block
        num_stages=2,                # double‑buffering for latency hiding
    )

    # ------------------------------------------------------------------- reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)   # [B,N,N,H]

    # ------------------------------------------------------------------- final LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------- output gate
    out = out * out_gate

    # ------------------------------------------------------------------- final linear back to original dim
    out = F.linear(out, weights["to_out.weight"])

    return out
