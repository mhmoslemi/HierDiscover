@triton.jit
def apply_mask_gate_fused_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    left_out_ptr, right_out_ptr,
    total_elements,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_gate_b, stride_left_gate_i, stride_left_gate_j, stride_left_gate_h,
    stride_right_gate_b, stride_right_gate_i, stride_right_gate_j, stride_right_gate_h,
    B, N, H,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # decode offsets
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    left_off = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_off = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    left_gate_off = b * stride_left_gate_b + i * stride_left_gate_i + j * stride_left_gate_j + h * stride_left_gate_h
    right_gate_off = b * stride_right_gate_b + i * stride_right_gate_i + j * stride_right_gate_j + h * stride_right_gate_h

    a_left = tl.load(left_ptr + left_off, mask=mask, other=0.0)
    a_right = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)
    left_gate = tl.load(left_gate_ptr + left_gate_off, mask=mask, other=0.0)
    right_gate = tl.load(right_gate_ptr + right_gate_off, mask=mask, other=0.0)

    out_left = a_left * m * left_gate
    out_right = a_right * m * right_gate
    tl.store(left_out_ptr + left_off, out_left, mask=mask)
    tl.store(right_out_ptr + right_off, out_right, mask=mask)
```

"""
tri_mul_outgoing.py
-------------------
Optimized forward‑only implementation of the “outgoing” TriMul operation
(AlphaFold‑3).  

Key ideas
~~~~~~~~
*   All heavy linear / LayerNorm ops stay in PyTorch (cuBLAS/cuDNN) but are
    run in **float16** to cut memory traffic and exploit Tensor‑Cores.
*   The element‑wise mask‑gate fusion (``left = left * mask * left_gate`` and
    ``right = right * mask * right_gate``) is performed by a **single
    Triton kernel** – one launch for both tensors, thus avoiding two kernel
    dispatches and redundant loads.
*   The O(N³·H) core ``out = left @ rightᵀ`` is expressed as a batched GEMM
    over the combined batch‑hidden dimension (B·H).  Using the
    ``torch.matmul`` batched implementation (Tensor‑Core accelerated) is
    already optimal on H100.
*   The final LayerNorm, gating and projection are done in float16 and the
    result is cast back to float32 as required.

The function signature is exactly the one demanded by the test harness:

    custom_kernel(data) -> torch.Tensor
"""
import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask * left_gate * left  and  mask * right_gate * right
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_fused_kernel(
    left_ptr, right_ptr,               # [B,N,N,H] input tensors
    mask_ptr,                          # [B,N,N]   mask
    left_gate_ptr, right_gate_ptr,     # [B,N,N,H] gate tensors
    left_out_ptr, right_out_ptr,       # [B,N,N,H] output tensors
    total_elements,                    # B * N * N * H  (linear size)
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_m_b, stride_m_i, stride_m_j,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    B, N, H,
    BLOCK: tl.constexpr,               # # of linear indices per program
):
    """Fused mask & gate multiplication for left and right tensors."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # ---------- decode linear offset -----------------
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # ---------- compute flat pointers ----------------
    left_off  = b * stride_l_b + i * stride_l_i + j * stride_l_j + h * stride_l_h
    right_off = b * stride_r_b + i * stride_r_i + j * stride_r_j + h * stride_r_h
    mask_off  = b * stride_m_b + i * stride_m_i + j * stride_m_j          # no h dim
    lg_off    = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    rg_off    = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h

    # ---------- load -------------------------------------------------
    a_left   = tl.load(left_ptr  + left_off,  mask=mask, other=0.0)
    a_right  = tl.load(right_ptr + right_off, mask=mask, other=0.0)
    m        = tl.load(mask_ptr  + mask_off,  mask=mask, other=1.0)   # default: keep
    lg       = tl.load(left_gate_ptr  + lg_off, mask=mask, other=0.0)
    rg       = tl.load(right_gate_ptr + rg_off, mask=mask, other=0.0)

    # ---------- compute & store --------------------------------------
    out_l = a_left * m * lg
    out_r = a_right * m * rg
    tl.store(left_out_ptr  + left_off,  out_l, mask=mask)
    tl.store(right_out_ptr + right_off, out_r, mask=mask)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, C]
        - mask_tensor  : torch.Tensor  [B, N, N]  (may be None)
        - weights_dict : mapping of all linear / norm parameters
        - config_dict  : contains ``dim`` and ``hidden_dim`` (and optional
                         ``nomask`` flag)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------
    # 1) unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 2) LayerNorm on the input (float16 for speed, weight/bias casted)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"].to(torch.float16, device=device)
    norm_b = weights["norm.bias"].to(torch.float16, device=device)

    # the input is float32 by default; bring to fp16 after LN
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias) – all in fp16
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16, device=device)
    right_proj_w = weights["right_proj.weight"].to(torch.float16, device=device)

    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) – still fp16
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16, device=device)
    right_gate_w = weights["right_gate.weight"].to(torch.float16, device=device)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16, device=device)

    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5) Fuse mask + left_gate / right_gate with Triton
    # ------------------------------------------------------------------
    if nomask or mask_tensor is None:
        # No mask => simple element‑wise multiplication (two ops)
        left = left * left_gate
        right = right * right_gate
    else:
        # Ensure mask is fp16 (0 / 1)
        mask = mask_tensor.to(torch.float16, device=device)

        B, N_seq, _, _ = left.shape
        total_elements = B * N_seq * N_seq * hidden_dim
        BLOCK = 1024                    # ~1‑K elements per program
        grid = ( (total_elements + BLOCK - 1) // BLOCK, )

        # allocate fused outputs
        left_fused  = torch.empty_like(left)
        right_fused = torch.empty_like(right)

        apply_mask_gate_fused_kernel[grid](
            left, right,
            mask,
            left_gate, right_gate,
            left_fused, right_fused,
            total_elements,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            B, N_seq, hidden_dim,
            BLOCK=BLOCK,
        )
        left, right = left_fused, right_fused

    # ------------------------------------------------------------------
    # 6) Core TriMul: out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    #    Implemented as a batched GEMM over (B*H) matrices.
    # ------------------------------------------------------------------
    # bring hidden dim to the batch axis
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    left_mat  = left.view(B * H, N, N)              # (BH, N, N)
    right_mat = right.view(B * H, N, N)             # (BH, N, N)

    # (BH, N, N) @ (BH, N, N)^T  →  (BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16, device=device)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16, device=device)

    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 8) Apply out_gate (element‑wise) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate                                 # fp16

    to_out_w = weights["to_out.weight"].to(torch.float16, device=device)
    out = torch.nn.functional.linear(out, to_out_w)      # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9) Cast back to float32 as required by the spec
    # ------------------------------------------------------------------
    return out.to(torch.float32)
