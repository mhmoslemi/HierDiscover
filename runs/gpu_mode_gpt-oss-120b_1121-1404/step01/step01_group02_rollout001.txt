"""
TriMul (outgoing) – forward pass.

The expensive N³ operation
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is implemented as a batched matrix multiplication (A @ Bᵀ) using a
custom Triton kernel.  All surrounding LayerNorm/Linear/Gate ops are
performed with PyTorch (fp32).  The projections and gates are cast to
fp16 before the GEMM; the kernel uses an fp32 accumulator for numeric
stability.  The kernel is autotuned for several block‑size configurations
that work well on an H100.

The entry point `custom_kernel` receives:
    (input, mask, weights, config)
and returns the transformed tensor of shape [B, N, N, dim] (float32).
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from triton import Config, autotune


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ)
# ----------------------------------------------------------------------
@autotune(
    configs=[
        Config({'BLOCK_M': 64,  'BLOCK_N': 64,  'BLOCK_K': 32}, num_warps=4),
        Config({'BLOCK_M': 128, 'BLOCK_N': 128, 'BLOCK_K': 32}, num_warps=8),
        Config({'BLOCK_M': 64,  'BLOCK_N': 128, 'BLOCK_K': 64}, num_warps=8),
        Config({'BLOCK_M': 128, 'BLOCK_N': 64,  'BLOCK_K': 64}, num_warps=8),
    ],
    key=['M', 'N', 'K']
)
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    batch, M, N, K,               # dimensions
    stride_ah, stride_am, stride_ak,   # A strides (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # Bᵀ strides (batch, row(K), col(N))
    stride_ch, stride_cm, stride_cn,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : [batch, M, K]   (fp16)
    B : [batch, K, N]   (fp16)   (already transposed)
    C : [batch, M, N]   (fp32 accumulator, stored as fp32)
    """
    pid_batch = tl.program_id(0)   # which (b, hidden) pair
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for this batch element
    a_base = a_ptr + pid_batch * stride_ah
    b_base = b_ptr + pid_batch * stride_bh

    # Pointers for the row/col tiles (will be offset by k inside the loop)
    a_row_ptr = a_base + offs_m[:, None] * stride_am
    b_col_ptr = b_base + offs_n[None, :] * stride_bn

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    for cur_k in range(0, num_k_tiles):
        k_offset = cur_k * BLOCK_K

        # ------------------------------------------------------------------
        # Load A and B tiles with boundary checks
        # ------------------------------------------------------------------
        a_k_ptr = a_row_ptr + (k_offset + offs_k)[None, :] * stride_ak
        b_k_ptr = b_col_ptr + (k_offset + offs_k)[:, None] * stride_bk

        a_mask = (offs_m[:, None] < M) & ((k_offset + offs_k)[None, :] < K)
        b_mask = ((k_offset + offs_k)[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_k_ptr, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_k_ptr, mask=b_mask, other=0.0)   # fp16

        # Accumulate
        acc += tl.dot(a, b)  # fp32

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_row_ptr = c_ptr + pid_batch * stride_ch
    c_ptrs = c_row_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using Triton for the
    N³ matrix‑multiplication part.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (bias=False)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Apply mask if present
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; broadcast over hidden dimension
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Compute gates (sigmoid) and apply them
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to fp16 for the heavy GEMM
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6) Reshape to (batch*hidden, N, N) – each (b,d) pair is an
    #    independent matrix multiplication.
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left_perm  = left_f16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_f16.permute(0, 3, 1, 2).contiguous()

    left_mat  = left_perm.view(B * hidden_dim, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)  # [B*H, N, N]

    # Transpose the second matrix so that the GEMM computes
    #   left @ rightᵀ  ->  Σ_k left[i,k] * right[j,k]
    right_mat_T = right_mat.permute(0, 2, 1).contiguous()   # [B*H, N, N]

    # Output buffer (fp32 because accumulator is fp32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7) Triton batched GEMM launch
    # ------------------------------------------------------------------
    # Strides in element units (not bytes)
    stride_ah = left_mat.stride(0)
    stride_am = left_mat.stride(1)
    stride_ak = left_mat.stride(2)

    stride_bh = right_mat_T.stride(0)
    stride_bk = right_mat_T.stride(1)
    stride_bn = right_mat_T.stride(2)

    stride_ch = out_mat.stride(0)
    stride_cm = out_mat.stride(1)
    stride_cn = out_mat.stride(2)

    # Grid: (batch, rows, cols)
    grid = (
        B * hidden_dim,
        (N + 128 - 1) // 128,
        (N + 128 - 1) // 128,
    )

    # Launch kernel – the autotuned configs will pick the best block sizes
    batched_matmul_kernel[grid](
        left_mat,
        right_mat_T,
        out_mat,
        B * hidden_dim,
        N,
        N,
        N,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bk,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        # The following constants are overwritten by the autotune configs.
        # They are kept here for clarity but are not used directly.
        BLOCK_M=128,
        BLOCK_N=128,
        BLOCK_K=32,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 9) Output LayerNorm + gating + final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to dim

    return out
