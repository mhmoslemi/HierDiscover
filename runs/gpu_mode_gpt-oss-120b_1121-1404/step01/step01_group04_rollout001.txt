def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim=config["dim"]; hidden_dim=config["hidden_dim"]; nomask=config.get("nomask", True)
    # Determine dtype: if input is float16 then keep else use float16 for compute (cast)
    compute_dtype = torch.float16
    # 1. LayerNorm: cast to compute dtype.
    x = F.layer_norm(input_tensor.to(compute_dtype),
                     normalized_shape=(dim,),
                     weight=weights["norm.weight"].to(compute_dtype),
                     bias=weights["norm.bias"].to(compute_dtype))
    # x shape B,N,N,dim half
    # Linear projections halves
    left_proj_w = weights["left_proj.weight"].to(compute_dtype)  # hidden_dim x dim
    right_proj_w = weights["right_proj.weight"].to(compute_dtype)
    left = F.linear(x, left_proj_w)  # half -> B,N,N,hidden_dim
    right = F.linear(x, right_proj_w)
    # Gates
    left_gate_w = weights["left_gate.weight"].to(compute_dtype)
    right_gate_w = weights["right_gate.weight"].to(compute_dtype)
    out_gate_w = weights["out_gate.weight"].to(compute_dtype)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))
    if not nomask:
        # mask float half of shape B,N,N
        mask_fp = mask_tensor.to(compute_dtype)
        # fused kernel
        BLOCK = 128
        total = left.numel()
        grid = (triton.cdiv(total, BLOCK), )
        # compute stride
        stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h = left_gate.stride()
        stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h = right_gate.stride()
        stride_m_b, stride_m_i, stride_m_j = mask_fp.stride()
        fused_mask_gate_kernel[grid](
            left_ptr=left,
            left_gate_ptr=left_gate,
            right_ptr=right,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp,
            B=input_tensor.shape[0],
            N=input_tensor.shape[1],
            H=hidden_dim,
            stride_l_b=stride_l_b,
            stride_l_i=stride_l_i,
            stride_l_j=stride_l_j,
            stride_l_h=stride_l_h,
            stride_lg_b=stride_lg_b,
            stride_lg_i=stride_lg_i,
            stride_lg_j=stride_lg_j,
            stride_lg_h=stride_lg_h,
            stride_r_b=stride_r_b,
            stride_r_i=stride_r_i,
            stride_r_j=stride_r_j,
            stride_r_h=stride_r_h,
            stride_rg_b=stride_rg_b,
            stride_rg_i=stride_rg_i,
            stride_rg_j=stride_rg_j,
            stride_rg_h=stride_rg_h,
            stride_m_b=stride_m_b,
            stride_m_i=stride_m_i,
            stride_m_j=stride_m_j,
            BLOCK=BLOCK,
        )
    else:
        left = left * left_gate
        right = right * right_gate

    # Contract over k: left[b,i,k,h] * right[b,j,k,h]
    # Permute to (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # matmul: (B,H,N,N) x (B,H,N,N).transposed last two dims
    # Use torch.matmul (batched)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    # shape B,H,N,N -> permute back
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # Final to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_w = weights["to_out_norm.weight"].to(compute_dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(compute_dtype)
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)
    # gating
    out = out * out_gate
    # final linear
    to_out_w = weights["to_out.weight"].to(compute_dtype)
    out = F.linear(out, to_out_w)

    # Cast back to input dtype
    out = out.to(input_tensor.dtype)
    return out

"""
TriMul (outgoing) – Triton‑fused implementation
================================================

The forward pass consists of

1. **LayerNorm** over the last feature dimension (`dim`).
2. **Linear projections** to a hidden dimension (`hidden_dim`) for the
   left‑hand side, the right‑hand side and three elementwise gates.
   All heavy work is performed in **float16** to halve memory traffic.
3. **Mask & gate fusion** – a tiny Triton kernel multiplies the left
   and right projections with their corresponding sigmoid gates and with
   the optional binary mask (in‑place, no extra temporaries).
4. **Tri‑multiplicative contraction**  

      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   which is a batched matrix multiplication for each hidden channel.
   By permuting the tensors we obtain a `(B*hidden_dim, N, N)` batch and
   call `torch.matmul`, which uses cuBLAS on the H100.
5. **Output LayerNorm → gate → final linear** – again performed in
   float16 and cast back to the original dtype before returning.

The kernel `fused_mask_gate_kernel` is the only mandatory Triton kernel;
all other operations stay in PyTorch (cuBLAS / CUDA kernels) to keep the
implementation simple yet fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def fused_mask_gate_kernel(
    left_ptr,          # [B,N,N,H]  left  (in‑place)
    left_gate_ptr,     # [B,N,N,H]  left_gate
    right_ptr,         # [B,N,N,H]  right (in‑place)
    right_gate_ptr,    # [B,N,N,H]  right_gate
    mask_ptr,          # [B,N,N]    mask (0/1)
    B, N, H,           # problem sizes
    # strides for left
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    # strides for mask (no hidden dim)
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    valid = offs < total

    # ---- decode linear index into (b,i,j,h) ----
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # mask invalid coordinates (should never happen for valid)
    valid = valid & (b < B) & (i < N) & (j < N) & (h < H)

    # compute pointers
    left_off   = left_ptr   + b * stride_l_b   + i * stride_l_i   + j * stride_l_j   + h * stride_l_h
    leftg_off  = left_gate_ptr + b * stride_lg_b  + i * stride_lg_i  + j * stride_lg_j  + h * stride_lg_h
    right_off  = right_ptr  + b * stride_r_b   + i * stride_r_i   + j * stride_r_j   + h * stride_r_h
    rightg_off = right_gate_ptr + b * stride_rg_b  + i * stride_rg_i  + j * stride_rg_j  + h * stride_rg_h
    mask_off   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + j * stride_m_j

    # load values
    left      = tl.load(left_off,   mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(leftg_off,  mask=valid, other=tl.float16(0.0))
    right     = tl.load(right_off,  mask=valid, other=tl.float16(0.0))
    right_gate= tl.load(rightg_off, mask=valid, other=tl.float16(0.0))
    mask_val  = tl.load(mask_off,   mask=valid, other=tl.float16(1.0))

    # fused mask + gate (sigmoid already applied upstream)
    left  = left  * left_gate  * mask_val
    right = right * right_gate * mask_val

    # store back in‑place
    tl.store(left_off,  left,  mask=valid)
    tl.store(right_off, right, mask=valid)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  Cast everything to float16 for the heavy path
    # ------------------------------------------------------------------
    compute_dtype = torch.float16
    x = input_tensor.to(compute_dtype)

    # ------------------------------------------------------------------
    # 2️⃣  LayerNorm over `dim` (float16)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(compute_dtype),
        bias=weights["norm.bias"].to(compute_dtype),
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 3️⃣  Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(compute_dtype))   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"].to(compute_dtype))

    # ------------------------------------------------------------------
    # 4️⃣  Gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(compute_dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(compute_dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(compute_dtype)))

    # ------------------------------------------------------------------
    # 5️⃣  Fuse mask + gate with a tiny Triton kernel (in‑place)
    # ------------------------------------------------------------------
    if not nomask:
        # ensure mask is float16 (0.0 or 1.0)
        mask_fp = mask_tensor.to(compute_dtype)

        BLOCK = 1024
        total_el = left.numel()                     # B * N * N * hidden_dim
        grid = (triton.cdiv(total_el, BLOCK),)

        # strides for all tensors
        stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h = left_gate.stride()
        stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h = right_gate.stride()
        stride_m_b, stride_m_i, stride_m_j = mask_fp.stride()

        fused_mask_gate_kernel[grid](
            left_ptr=left,
            left_gate_ptr=left_gate,
            right_ptr=right,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp,
            B=input_tensor.shape[0],
            N=input_tensor.shape[1],
            H=hidden_dim,
            stride_l_b=stride_l_b,
            stride_l_i=stride_l_i,
            stride_l_j=stride_l_j,
            stride_l_h=stride_l_h,
            stride_lg_b=stride_lg_b,
            stride_lg_i=stride_lg_i,
            stride_lg_j=stride_lg_j,
            stride_lg_h=stride_lg_h,
            stride_r_b=stride_r_b,
            stride_r_i=stride_r_i,
            stride_r_j=stride_r_j,
            stride_r_h=stride_r_h,
            stride_rg_b=stride_rg_b,
            stride_rg_i=stride_rg_i,
            stride_rg_j=stride_rg_j,
            stride_rg_h=stride_rg_h,
            stride_m_b=stride_m_b,
            stride_m_i=stride_m_i,
            stride_m_j=stride_m_j,
            BLOCK=BLOCK,
        )
    else:
        # only apply gates (no mask)
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣  Tri‑multiplicative contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #     → batched GEMM over the (i,k) × (j,k) matrices for every hidden channel.
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) – hidden dimension becomes batch axis
    left_perm = left.permute(0, 3, 1, 2)    # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2)  # (B, H, N, N)

    # batched matrix multiply (cuBLAS) in float16
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # (B, H, N, N)

    # restore original ordering → (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1)    # (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7️⃣  Output LayerNorm (over hidden_dim), gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"].to(compute_dtype),
        bias=weights["to_out_norm.bias"].to(compute_dtype),
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"].to(compute_dtype))

    # ------------------------------------------------------------------
    # 8️⃣  Cast back to the input dtype (usually float32) and return
    # ------------------------------------------------------------------
    return out.to(input_tensor.dtype)
