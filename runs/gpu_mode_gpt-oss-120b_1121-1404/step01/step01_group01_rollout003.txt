@triton.jit
def fuse_mul_kernel(a_ptr, b_ptr, mask_ptr, out_ptr, total_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE

    mask = tl.load(mask_ptr + offsets, mask=offsets < total_elements, other=0.0)
    a = tl.load(a_ptr + offsets, mask=offsets < total_elements, other=0.0)
    b = tl.load(b_ptr + offsets, mask=offsets < total_elements, other=0.0)

Outgoing TriMul (AlphaFold‑3) – fast forward implementation

Algorithm
---------
1. Layer‑norm the input tensor.
2. Two linear projections (left / right) and three gating projections

    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)

    # bounds
    mask = offs < total_elems

    a = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    b = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)   # default 1 → no effect

    tl.store(out_ptr + offs, a * b * m, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – flatten to (B·N·N, dim)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                     # (B·N·N, dim)

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = F.linear(x_flat, left_proj_weight).view(bs, N, N, hidden_dim)
    right_proj = F.linear(x_flat, right_proj_weight).view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid(linear))
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight)).view(bs, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4. Cast projections & gates to fp16 (Tensor‑Core friendly)
    # ------------------------------------------------------------------
    left_proj_fp16 = left_proj.to(torch.float16)
    right_proj_fp16 = right_proj.to(torch.float16)
    left_gate_fp16 = left_gate.to(torch.float16)
    right_gate_fp16 = right_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Optional mask + gate fusion (Triton)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: (bs, N, N) → (bs, N, N, hidden_dim) broadcasted
        mask_exp = mask.to(x.dtype).unsqueeze(-1)                 # (bs, N, N, 1)
        mask_exp = mask_exp.expand(-1, -1, -1, hidden_dim).contiguous()
        mask_fp16 = mask_exp.to(torch.float16)

        total = left_proj_fp16.numel()
        BLOCK = 1024
        grid = ((total + BLOCK - 1) // BLOCK,)

        left = torch.empty_like(left_proj_fp16)
        right = torch.empty_like(right_proj_fp16)

        fuse_mul_kernel[grid](
            left_proj_fp16,
            left_gate_fp16,
            mask_fp16,
            left,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
        fuse_mul_kernel[grid](
            right_proj_fp16,
            right_gate_fp16,
            mask_fp16,
            right,
            total,
            BLOCK_SIZE=BLOCK,
            num_warps=4,
        )
    else:
        # No mask – just element‑wise multiplication
        left = left_proj_fp16 * left_gate_fp16
        right = right_proj_fp16 * right_gate_fp16

    # ------------------------------------------------------------------
    # 6. Batched GEMM (left @ rightᵀ) – all in fp16
    # ------------------------------------------------------------------
    # reshape to (B·H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)  (j,k) → (k,j)

    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, N, N)                # (BH, M, K)
    right_flat = right_perm.view(BH, N, N)              # (BH, K, N) – already transposed

    # torch.bmm uses cuBLAS and fully leverages Tensor Cores on H100
    out_flat_fp16 = torch.bmm(left_flat, right_flat)    # (BH, N, N)

    # ------------------------------------------------------------------
    # 7. Restore shape (B, N, N, H) and convert to fp32
    # ------------------------------------------------------------------
    out = out_flat_fp16.view(bs, hidden_dim, N, N)      # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).to(torch.float32)    # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8. Output LayerNorm
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 9. Apply the output gate (still in fp32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 10. Final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out.view(-1, hidden_dim), to_out_weight).view(bs, N, N, dim)

    return out
