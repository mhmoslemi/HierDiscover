    left_out_ptr, right_out_ptr,

    stride_lpw_h, stride_lpw_d,
    stride_lgw_h, stride_lgw_d,
    stride_rpw_h, stride_rpw_d,
    stride_rgw_h, stride_rgw_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lout_b, stride_lout_i, stride_lout_j, stride_lout_h,
    stride_rout_b, stride_rout_i, stride_rout_j, stride_rout_h,
    B, N, D, H,
    BLOCK_ROW: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):

    row_start = pid * BLOCK_ROW
    rows = row_start + tl.arange(0, BLOCK_ROW)
    row_mask = rows < B * N * N

    rem = rows % (N * N)
    i = rem // N
    j = rem % N

    acc_lg = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)

                       mask=d_mask[:, None] & h_mask[None, :],
                       other=0.0).to(tl.float32)

        w_rp = tl.load(right_proj_w_ptr + d_idx[:, None] * stride_rpw_d + h_idx[None, :] * stride_rpw_h,
                       mask=d_mask[:, None] & h_mask[None, :],

        w_rg = tl.load(right_gate_w_ptr + d_idx[:, None] * stride_rgw_d + h_idx[None, :] * stride_rgw_h,
                       mask=d_mask[:, None] & h_mask[None, :],
                       other=0.0).to(tl.float32)

        # dot
        acc_lp += tl.dot(x, w_lp)   # shape (BLOCK_ROW, BLOCK_H)
        acc_lg += tl.dot(x, w_lg)
        acc_rp += tl.dot(x, w_rp)
        acc_rg += tl.dot(x, w_rg)

    left_gate = tl.sigmoid(acc_lg)
    right_gate = tl.sigmoid(acc_rg)
    left_out = acc_lp * left_gate * mask_vec
    right_out = acc_rp * right_gate * mask_vec

    # store
    tl.store(left_out_ptr + b * stride_lout_b + i * stride_lout_i + j * stride_lout_j + h_idx * stride_lout_h,
             left_out, mask=row_mask[:, None] & (h_idx < H)[None, :])

             right_out, mask=row_mask[:, None] & (h_idx < H)[None, :])

import torch.nn.functional as F
import triton
import triton.language as tl

def sigmoid(x):
    return 1.0 / (1.0 + tl.exp(-x))

@triton.jit
def fused_proj_gate_mask_kernel(
    x_ptr,                # [B,N,N,D]
    left_proj_w_ptr, left_gate_w_ptr, right_proj_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    # strides
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_lpw_h, stride_lpw_d,
    stride_lgw_h, stride_lgw_d,
    stride_rpw_h, stride_rpw_d,
    stride_rgw_h, stride_rgw_d,
    stride_ogw_h, stride_ogw_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lout_b, stride_lout_i, stride_lout_j, stride_lout_h,
    stride_rout_b, stride_rout_i, stride_rout_j, stride_rout_h,
    stride_ogout_b, stride_ogout_i, stride_ogout_j, stride_ogout_h,
    B, N, D, H,
    BLOCK_ROW: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_ROW
    rows = row_start + tl.arange(0, BLOCK_ROW)
    row_mask = rows < B * N * N

    # Decode (b,i,j)
    b = rows // (N * N)
    rem = rows % (N * N)
    i = rem // N
    j = rem % N

    # Base offset for x
    x_base = b * stride_xb + i * stride_xi + j * stride_xj

    # Load mask scalar
    mask_val = tl.load(
        mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
        mask=row_mask,
        other=1.0,
    )
    mask_vec = mask_val[:, None]  # (BLOCK_ROW, 1)

    # accumulators
    acc_lp = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)

    # hidden indices
    h_idx = tl.arange(0, BLOCK_H)

    # Loop over input dim
    for d in range(0, D, BLOCK_D):
        d_idx = d + tl.arange(0, BLOCK_D)
        d_mask = d_idx < D

        # Load x slice
        x = tl.load(
            x_ptr + x_base[:, None] + d_idx[None, :] * stride_xd,
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0,
        ).to(tl.float32)  # (BLOCK_ROW, BLOCK_D)

        # load weight slices (transposed: [BLOCK_D, BLOCK_H])
        w_lp = tl.load(
            left_proj_w_ptr + d_idx[:, None] * stride_lpw_d + h_idx[None, :] * stride_lpw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        w_lg = tl.load(
            left_gate_w_ptr + d_idx[:, None] * stride_lgw_d + h_idx[None, :] * stride_lgw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        w_rp = tl.load(
            right_proj_w_ptr + d_idx[:, None] * stride_rpw_d + h_idx[None, :] * stride_rpw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        w_rg = tl.load(
            right_gate_w_ptr + d_idx[:, None] * stride_rgw_d + h_idx[None, :] * stride_rgw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        w_og = tl.load(
            out_gate_w_ptr + d_idx[:, None] * stride_ogw_d + h_idx[None, :] * stride_ogw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        # Dot products
        acc_lp += tl.dot(x, w_lp)   # (BLOCK_ROW, BLOCK_H)
        acc_lg += tl.dot(x, w_lg)
        acc_rp += tl.dot(x, w_rp)
        acc_rg += tl.dot(x, w_rg)
        acc_og += tl.dot(x, w_og)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    """ TritMul outgoing forward pass fused projection, gating, and mask.

    Kernel fuses:
        left = (x @ W_left)          * sigmoid(x @ W_left_gate) * mask
        right = (x @ W_right)        * sigmoid(x @ W_right_gate) * mask
        out_gate = sigmoid(x @ W_out_gate)

    All three linear layers are computed inside a single Triton kernel to
    avoid multiple reads of the large input tensor. The main O(N³·H) reduction
    is performed with a batched GEMM using torch.bmm, which leverages the
    Tensor Core implementation (fp16). The rest of the network (LayerNorms,
    final projection) is executed with PyTorch primitives.
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # ----------   Prepare tensors (fp16 where possible)   ----------
    # Input LayerNorm (keep in float32 for numeric stability)
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Cast to fp16 for the heavy part
    x_fp16 = x.to(torch.float16)

    # Masks
    if mask is None or nomask:
        mask_fp16 = torch.ones((bs, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # -----------   Cast model weights to fp16 for kernel   ----------
    def to_fp16(t):
        return t.to(torch.float16)

    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])

    # Output buffers
    left = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)  # same shape as left/right

    # ----------   Strides (element counts)   ----------
    # Input tensor x_fp16 [B,N,N,D]
    stride_x = x_fp16.stride()  # (b,i,j,d)

    # Weight tensors [H,D]
    stride_lpw = left_proj_w.stride()  # (h,d)
    stride_lgw = left_gate_w.stride()
    stride_rpw = right_proj_w.stride()
    stride_rgw = right_gate_w.stride()
    stride_ogw = out_gate_w.stride()

    # Mask tensor [B,N,N]
    stride_mask = mask_fp16.stride()

    # Output tensors [B,N,N,H]
    stride_lout = left.stride()
    stride_rout = right.stride()
    stride_ogout = out_gate.stride()

    # ----------   Kernel launch configuration   ----------
    total_rows = bs * seq_len * seq_len
    BLOCK_ROW = 64   # rows processed per program
    BLOCK_D = 32     # chunk of the input dimension
    BLOCK_H = 32     # chunk of the hidden dimension

    grid = (triton.cdiv(total_rows, BLOCK_ROW),)

    fused_proj_gate_mask_kernel[grid](
        # pointers
        x_fp16,
        left_proj_w, left_gate_w, right_proj_w, right_gate_w,
        out_gate_w,
        mask_fp16,
        left, right, out_gate,
        # strides
        stride_x[0], stride_x[1], stride_x[2], stride_x[3],
        stride_lpw[0], stride_lpw[1],
        stride_lgw[0], stride_lgw[1],
        stride_rpw[0], stride_rpw[1],
        stride_rgw[0], stride_rgw[1],
        stride_ogw[0], stride_ogw[1],
        stride_mask[0], stride_mask[1], stride_mask[2],
        stride_lout[0], stride_lout[1], stride_lout[2], stride_lout[3],
        stride_rout[0], stride_rout[1], stride_rout[2], stride_rout[3],
        stride_ogout[0], stride_ogout[1], stride_ogout[2], stride_ogout[3],
        # problem sizes
        bs, seq_len, dim, hidden_dim,
        # compile‑time constants
        BLOCK_ROW=BLOCK_ROW, BLOCK_D=BLOCK_D, BLOCK_H=BLOCK_H,
    )

    # ----------   Batched GEMM over the hidden dimension   ----------
    # shape: (B, N, N, H) -> (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)
    right_perm = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)

    # Matrix multiplication using fp16 tensor cores
    out_mat = torch.bmm(left_perm, right_perm.transpose(1, 2))   # [B*H, N, N]

    # Restore original layout [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).float()  # fp32

    # ----------   Output LayerNorm + gating   ----------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"])

    # out_gate is currently fp16, bring to fp32
    out_gate_fp32 = out_gate.float()
    out = out * out_gate_fp32

    # ----------   Final linear projection back to dim   ----------
    out = F.linear(out, weights["to_out.weight"])   # bias is omitted by construction

    return out

@triton.jit
def fused_proj_gate_mask_kernel(
    # pointers
    x_ptr,                     # [B,N,N,D]
    left_proj_w_ptr, left_gate_w_ptr,
    right_proj_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    mask_ptr,                  # [B,N,N]
    left_out_ptr, right_out_ptr, out_gate_out_ptr,   # [B,N,N,H]
    # strides (in elements)
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_lpw_h, stride_lpw_d,
    stride_lgw_h, stride_lgw_d,
    stride_rpw_h, stride_rpw_d,
    stride_rgw_h, stride_rgw_d,
    stride_ogw_h, stride_ogw_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lout_b, stride_lout_i, stride_lout_j, stride_lout_h,
    stride_rout_b, stride_rout_i, stride_rout_j, stride_rout_h,
    stride_ogout_b, stride_ogout_i, stride_ogout_j, stride_ogout_h,
    # problem sizes
    B, N, D, H,
    # compile‑time constants
    BLOCK_ROW: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_ROW
    rows = row_start + tl.arange(0, BLOCK_ROW)                # [BLOCK_ROW]
    row_mask = rows < B * N * N

    # Decode (b,i,j) from row index
    b = rows // (N * N)
    rem = rows % (N * N)
    i = rem // N
    j = rem % N

    # Base offset for the input feature vector x[b,i,j,:]
    x_base = b * stride_xb + i * stride_xi + j * stride_xj

    # Load mask scalar (broadcast later)
    mask_val = tl.load(
        mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
        mask=row_mask,
        other=1.0,
    )
    mask_vec = mask_val[:, None]        # shape (BLOCK_ROW, 1)

    # Accumulators for the four projections + out‑gate (float32 for precision)
    acc_lp = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)

    # Hidden‑dimension indices (same for all accumulators)
    h_idx = tl.arange(0, BLOCK_H)

    # Loop over the input dimension D in tiles of BLOCK_D
    for d in range(0, D, BLOCK_D):
        d_idx = d + tl.arange(0, BLOCK_D)
        d_mask = d_idx < D

        # Load a tile of the input vector: shape (BLOCK_ROW, BLOCK_D)
        x = tl.load(
            x_ptr + x_base[:, None] + d_idx[None, :] * stride_xd,
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0,
        ).to(tl.float32)

        # Load corresponding slices of the four weight matrices (transposed)
        w_lp = tl.load(
            left_proj_w_ptr + d_idx[:, None] * stride_lpw_d + h_idx[None, :] * stride_lpw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        w_lg = tl.load(
            left_gate_w_ptr + d_idx[:, None] * stride_lgw_d + h_idx[None, :] * stride_lgw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        w_rp = tl.load(
            right_proj_w_ptr + d_idx[:, None] * stride_rpw_d + h_idx[None, :] * stride_rpw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        w_rg = tl.load(
            right_gate_w_ptr + d_idx[:, None] * stride_rgw_d + h_idx[None, :] * stride_rgw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        w_og = tl.load(
            out_gate_w_ptr + d_idx[:, None] * stride_ogw_d + h_idx[None, :] * stride_ogw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        # Accumulate dot‑products for each projection
        acc_lp += tl.dot(x, w_lp)   # (BLOCK_ROW, BLOCK_H)
        acc_lg += tl.dot(x, w_lg)
        acc_rp += tl.dot(x, w_rp)
        acc_rg += tl.dot(x, w_rg)
        acc_og += tl.dot(x, w_og)

    # Apply sigmoid to gating tensors
    left_gate = 1.0 / (1.0 + tl.exp(-acc_lg))
    right_gate = 1.0 / (1.0 + tl.exp(-acc_rg))
    out_gate = 1.0 / (1.0 + tl.exp(-acc_og))

    # Fuse mask and cast back to fp16 for storage
    left_res = (acc_lp * left_gate * mask_vec).to(tl.float16)
    right_res = (acc_rp * right_gate * mask_vec).to(tl.float16)
    out_gate_res = out_gate.to(tl.float16)

    # Compute output base offsets
    out_base_l = b * stride_lout_b + i * stride_lout_i + j * stride_lout_j
    out_base_r = b * stride_rout_b + i * stride_rout_i + j * stride_rout_j
    out_base_g = b * stride_ogout_b + i * stride_ogout_i + j * stride_ogout_j

    # Store results (hidden dimension is vectorised)
    store_mask = row_mask[:, None] & (h_idx[None, :] < H)

    tl.store(
        left_out_ptr + out_base_l[:, None] + h_idx[None, :] * stride_lout_h,
        left_res,
        mask=store_mask,
    )
    tl.store(
        right_out_ptr + out_base_r[:, None] + h_idx[None, :] * stride_rout_h,
        right_res,
        mask=store_mask,
    )
    tl.store(
        out_gate_out_ptr + out_base_g[:, None] + h_idx[None, :] * stride_ogout_h,
        out_gate_res,
        mask=store_mask,
    )

"""
TriMul “outgoing” forward pass – Triton‑fused projection, gating and mask.

The heavy O(N³·H) work is split into two stages:
1️⃣  A single Triton kernel computes the four linear projections
    (left/right × proj/gate) plus the out‑gate in one pass over the
    input tensor.  The kernel fuses:
        left  = (x·W_left )   * sigmoid(x·W_left_gate)   * mask
        right = (x·W_right)   * sigmoid(x·W_right_gate)  * mask
        out_gate = sigmoid(x·W_out_gate)
    All operations are done in fp16, while the accumulators use fp32
    for extra precision.
2️⃣  The resulting tensors have shape [B, N, N, H].  A batched GEMM
    (torch.bmm) computes
        out_h = left_h @ right_hᵀ      for each hidden index h,
    which is exactly the einsum
        out[..., i, j, :] = Σₖ left[..., i, k, :] * right[..., j, k, :].

The rest of the TF‑style TriMul (LayerNorms, final projection) is
implemented with PyTorch primitives, keeping the code simple yet fast.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------- #
# Triton kernel: fused proj‑gate‑mask + out‑gate
# --------------------------------------------------------------- #
@triton.jit
def fused_proj_gate_mask_kernel(
    # pointers
    x_ptr,                                 # [B,N,N,D]                  (fp16)
    left_proj_w_ptr, left_gate_w_ptr,      # [H,D] each                 (fp16)
    right_proj_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    mask_ptr,                              # [B,N,N]                    (fp16)
    left_out_ptr, right_out_ptr, out_gate_out_ptr,  # [B,N,N,H] (fp16)

    # strides (in elements)
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_lpw_h, stride_lpw_d,
    stride_lgw_h, stride_lgw_d,
    stride_rpw_h, stride_rpw_d,
    stride_rgw_h, stride_rgw_d,
    stride_ogw_h, stride_ogw_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lout_b, stride_lout_i, stride_lout_j, stride_lout_h,
    stride_rout_b, stride_rout_i, stride_rout_j, stride_rout_h,
    stride_ogout_b, stride_ogout_i, stride_ogout_j, stride_ogout_h,

    # problem sizes
    B, N, D, H,

    # compile‑time constants
    BLOCK_ROW: tl.constexpr,
    BLOCK_D:   tl.constexpr,
    BLOCK_H:   tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_ROW
    rows = row_start + tl.arange(0, BLOCK_ROW)               # [BLOCK_ROW]
    row_mask = rows < B * N * N

    # ---------------------------------------------------------------- #
    # decode (b,i,j) from flattened row index
    # ---------------------------------------------------------------- #
    b = rows // (N * N)
    rem = rows % (N * N)
    i = rem // N
    j = rem % N

    # base address for the input vector x[b,i,j,:]
    x_base = b * stride_xb + i * stride_xi + j * stride_xj

    # mask scalar (1.0 if no mask)
    mask_val = tl.load(
        mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
        mask=row_mask,
        other=1.0,
    )
    mask_vec = mask_val[:, None]                     # (BLOCK_ROW,1)

    # ---------------------------------------------------------------- #
    # accumulators (fp32 for numerical stability)
    # ---------------------------------------------------------------- #
    acc_lp = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_lg = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_rp = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_rg = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)
    acc_og = tl.zeros((BLOCK_ROW, BLOCK_H), dtype=tl.float32)

    # hidden‑dim indices (vectorised)
    h_idx = tl.arange(0, BLOCK_H)

    # ---------------------------------------------------------------- #
    # loop over the input dimension D in tiles of BLOCK_D
    # ---------------------------------------------------------------- #
    for d in range(0, D, BLOCK_D):
        d_idx = d + tl.arange(0, BLOCK_D)
        d_mask = d_idx < D

        # load a tile of the input feature vector
        x = tl.load(
            x_ptr + x_base[:, None] + d_idx[None, :] * stride_xd,
            mask=row_mask[:, None] & d_mask[None, :],
            other=0.0,
        ).to(tl.float32)               # (BLOCK_ROW, BLOCK_D)

        # load transposed weight tiles: shape (BLOCK_D, BLOCK_H)
        w_lp = tl.load(
            left_proj_w_ptr + d_idx[:, None] * stride_lpw_d + h_idx[None, :] * stride_lpw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)
        w_lg = tl.load(
            left_gate_w_ptr + d_idx[:, None] * stride_lgw_d + h_idx[None, :] * stride_lgw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)
        w_rp = tl.load(
            right_proj_w_ptr + d_idx[:, None] * stride_rpw_d + h_idx[None, :] * stride_rpw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)
        w_rg = tl.load(
            right_gate_w_ptr + d_idx[:, None] * stride_rgw_d + h_idx[None, :] * stride_rgw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)
        w_og = tl.load(
            out_gate_w_ptr + d_idx[:, None] * stride_ogw_d + h_idx[None, :] * stride_ogw_h,
            mask=d_mask[:, None] & (h_idx[None, :] < H),
            other=0.0,
        ).to(tl.float32)

        # dot products – tl.dot expects (M,K) @ (K,N)
        acc_lp += tl.dot(x, w_lp)
        acc_lg += tl.dot(x, w_lg)
        acc_rp += tl.dot(x, w_rp)
        acc_rg += tl.dot(x, w_rg)
        acc_og += tl.dot(x, w_og)

    # ---------------------------------------------------------------- #
    # apply sigmoid gating and mask, cast back to fp16 for storage
    # ---------------------------------------------------------------- #
    left_gate  = 1.0 / (1.0 + tl.exp(-acc_lg))
    right_gate = 1.0 / (1.0 + tl.exp(-acc_rg))
    out_gate   = 1.0 / (1.0 + tl.exp(-acc_og))

    left_res  = (acc_lp * left_gate  * mask_vec).to(tl.float16)
    right_res = (acc_rp * right_gate * mask_vec).to(tl.float16)
    out_gate_res = out_gate.to(tl.float16)

    # ---------------------------------------------------------------- #
    # store results (vectorised over the hidden dimension)
    # ---------------------------------------------------------------- #
    out_base_l = b * stride_lout_b + i * stride_lout_i + j * stride_lout_j
    out_base_r = b * stride_rout_b + i * stride_rout_i + j * stride_rout_j
    out_base_g = b * stride_ogout_b + i * stride_ogout_i + j * stride_ogout_j

    store_mask = row_mask[:, None] & (h_idx[None, :] < H)

    tl.store(
        left_out_ptr + out_base_l[:, None] + h_idx[None, :] * stride_lout_h,
        left_res,
        mask=store_mask,
    )
    tl.store(
        right_out_ptr + out_base_r[:, None] + h_idx[None, :] * stride_rout_h,
        right_res,
        mask=store_mask,
    )
    tl.store(
        out_gate_out_ptr + out_base_g[:, None] + h_idx[None, :] * stride_ogout_h,
        out_gate_res,
        mask=store_mask,
    )


# --------------------------------------------------------------- #
# Python entry‑point
# --------------------------------------------------------------- #
def custom_kernel(data):
    """
    TritMul “outgoing” forward pass (AlphaFold3).

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (keep fp32 for stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2️⃣ Cast to fp16 – the rest of the heavy work runs in half‑precision
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    if mask is None or nomask:
        mask_fp16 = torch.ones((bs, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # -----------------------------------------------------------------
    # 3️⃣ Cast all weights that the Triton kernel uses
    # -----------------------------------------------------------------
    def to_fp16(t): return t.to(torch.float16)

    left_proj_w = to_fp16(weights["left_proj.weight"])
    right_proj_w = to_fp16(weights["right_proj.weight"])
    left_gate_w = to_fp16(weights["left_gate.weight"])
    right_gate_w = to_fp16(weights["right_gate.weight"])
    out_gate_w = to_fp16(weights["out_gate.weight"])

    # -----------------------------------------------------------------
    # 4️⃣ Allocate output buffers for left/right/out_gate
    # -----------------------------------------------------------------
    left = torch.empty((bs, seq_len, seq_len, hidden_dim),
                       dtype=torch.float16, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)  # same shape as left/right

    # -----------------------------------------------------------------
    # 5️⃣ Gather strides (element counts, not bytes)
    # -----------------------------------------------------------------
    sx = x_fp16.stride()               # (b,i,j,d)
    sw_lproj = left_proj_w.stride()    # (h,d)  – weight layout of nn.Linear
    sw_lgate = left_gate_w.stride()
    sw_rproj = right_proj_w.stride()
    sw_rgate = right_gate_w.stride()
    sw_ogate = out_gate_w.stride()
    sm = mask_fp16.stride()            # (b,i,j)
    sl = left.stride()                 # (b,i,j,h)
    sr = right.stride()
    sog = out_gate.stride()

    # -----------------------------------------------------------------
    # 6️⃣ Triton launch configuration
    # -----------------------------------------------------------------
    total_rows = bs * seq_len * seq_len
    BLOCK_ROW = 64   # rows processed per program
    BLOCK_D   = 32   # tile over input dimension
    BLOCK_H   = 32   # tile over hidden dimension

    grid = (triton.cdiv(total_rows, BLOCK_ROW),)

    fused_proj_gate_mask_kernel[grid](
        # pointers
        x_fp16,
        left_proj_w, left_gate_w, right_proj_w, right_gate_w,
        out_gate_w,
        mask_fp16,
        left, right, out_gate,
        # strides
        sx[0], sx[1], sx[2], sx[3],
        sw_lproj[0], sw_lproj[1],
        sw_lgate[0], sw_lgate[1],
        sw_rproj[0], sw_rproj[1],
        sw_rgate[0], sw_rgate[1],
        sw_ogate[0], sw_ogate[1],
        sm[0], sm[1], sm[2],
        sl[0], sl[1], sl[2], sl[3],
        sr[0], sr[1], sr[2], sr[3],
        sog[0], sog[1], sog[2], sog[3],
        # problem sizes
        bs, seq_len, dim, hidden_dim,
        # compile‑time constants
        BLOCK_ROW=BLOCK_ROW, BLOCK_D=BLOCK_D, BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # 7️⃣ Batched matrix multiplication over the hidden dimension
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)
    right_perm = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)

    # fp16 GEMM (Tensor Cores)
    out_mat = torch.bmm(left_perm, right_perm.transpose(1, 2))   # [B*H, N, N]

    # back to [B, N, N, H] (float32 for the following LayerNorm)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).float()

    # -----------------------------------------------------------------
    # 8️⃣ Output LayerNorm + out_gate scaling
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.float()   # out_gate was produced in fp16

    # -----------------------------------------------------------------
    # 9️⃣ Final linear projection back to the original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # bias is absent by construction

    return out
