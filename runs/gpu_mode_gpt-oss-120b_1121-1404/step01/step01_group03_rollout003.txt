"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy part of the module is a batched 3‑tensor contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
for every batch `b` and hidden channel `d`.  After layer‑norm and linear
projections + sigmoid gates (still done in PyTorch) the operands `left`
and `right` have shape [B, N, N, H] (H = hidden_dim).  By reshaping to
[B*H, N, N] each (b,d) pair becomes a separate matrix multiplication
C = A @ Bᵀ.  The Triton kernel computes one block of C at a time using
FP16 tiles and FP32 accumulation (Tensor‑Core friendly).  The output is
written back as FP16, then the remaining post‑processing (output‑norm,
gating, final linear) is performed in PyTorch.

The kernel works for any sequence length N up to 1024, batch ≤2 and hidden
≤128 and handles the optional mask (applied before the kernel).  The
final tensor is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  with A,B ∈ (batch_hidden, N, N)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    a_ptr,                # const *half   (left)
    b_ptr,                # const *half   (right)
    c_ptr,                # *half         (output)
    batch_hidden, N,      # i32 scalars
    stride_az, stride_ai, stride_ak,   # strides for A (page, row, col)
    stride_bz, stride_bk, stride_bn,   # strides for B (page, k‑dim, n‑dim)
    stride_cz, stride_ci, stride_cj,   # strides for C (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a tile of C = A @ Bᵀ for a single (batch, hidden) page.
    All tensors are stored in row‑major order.
    """
    pid_z = tl.program_id(0)          # which (batch, hidden) page
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # ------------------------------------------------------------------
    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this page
    a_page = a_ptr + pid_z * stride_az
    b_page = b_ptr + pid_z * stride_bz
    c_page = c_ptr + pid_z * stride_cz

    # accumulator (FP32 for numeric stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over K dimension
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)            # [BLOCK_K]
        mask_k = cur_k < N

        # ---- load A tile ------------------------------------------------
        a_ptrs = a_page + \
                 offs_m[:, None] * stride_ai + \
                 cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                     # half → stays half

        # ---- load B tile as Bᵀ -----------------------------------------
        # Bᵀ: row = k, col = n  → stride over k = stride_bk (col‑stride of B)
        #              stride over n = stride_bn (row‑stride of B)
        b_ptrs = b_page + \
                 cur_k[:, None] * stride_bk + \
                 offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                     # half

        # ---- multiply‑accumulate ----------------------------------------
        acc += tl.dot(a, b)                         # FP16 dot → FP32 acc

        k += BLOCK_K

    # ------------------------------------------------------------------
    # write C tile (convert back to half)
    c_ptrs = c_page + \
             offs_m[:, None] * stride_ci + \
             offs_n[None, :] * stride_cj
    c = acc.to(tl.float16)
    tl.store(c_ptrs,
             c,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, config = data
    # x : [B, N, N, dim]   (float32)
    # mask : [B, N, N]   (bool/float)  or None
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B, N, N, hidden]

    # --------------------------------------------------------------
    # 3) Optional mask (mask is broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None:
        # ensure same dtype for multiplication
        mask_exp = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4) Gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))    # [B,N,N,hidden]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Cast to half and reshape for the Triton kernel
    # --------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # (B, N, N, hidden) -> (B, hidden, N, N)  (contiguous)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & hidden into a single page dimension
    BxH = B * hidden
    left_mat = left_h.view(BxH, N, N)      # [BxH, N, N], half
    right_mat = right_h.view(BxH, N, N)    # [BxH, N, N], half

    # Output buffer (half)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # --------------------------------------------------------------
    # 6) Strides (in elements)
    # --------------------------------------------------------------
    stride_az = left_mat.stride(0)   # page stride = N*N
    stride_ai = left_mat.stride(1)   # row stride = N
    stride_ak = left_mat.stride(2)   # col stride = 1

    # right is accessed as Bᵀ
    stride_bz = right_mat.stride(0)  # page stride = N*N
    stride_bk = right_mat.stride(2)  # =1  (stride over k after transposition)
    stride_bn = right_mat.stride(1)  # =N  (stride over output col j)

    stride_cz = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # --------------------------------------------------------------
    # 7) Launch Triton kernel
    # --------------------------------------------------------------
    # Block sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BxH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BxH,
        N,
        stride_az,
        stride_ai,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps per block = 256 threads
        num_stages=3,         # double‑buffering for latency hiding
    )

    # --------------------------------------------------------------
    # 8) Reshape back to original layout, convert to float32
    # --------------------------------------------------------------
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 9) Output LayerNorm over hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 10) Output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 11) Final linear (to original dim)
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```

**Explanation of key optimisations**

* **Half‑precision tiles & FP32 accumulation** – the kernel loads `left` and `right` as `float16` and accumulates in `float32` via `tl.dot`, which maps to NVIDIA Tensor‑Cores on an H100.
* **Large 128×128 tiles** – each thread‑block computes a relatively big output tile, reducing launch overhead while staying within the register budget.
* **Triple‑buffered pipeline (`num_stages=3`)** – overlaps memory loads with compute.
* **8 warps per block** – matches the occupancy characteristics of H100 for the chosen tile size.
* **Minimal host‑side work** – only the cheap linear/gating/LayerNorm operations are performed in PyTorch; the N³ contraction is fully off‑loaded to Triton.
* **Mask handling outside the kernel** – the optional mask is applied before the kernel to keep the kernel simple and fast.

The function `custom_kernel` complies with the required signature and returns a `torch.float32` tensor of shape `[bs, seq_len, seq_len, dim]`.
