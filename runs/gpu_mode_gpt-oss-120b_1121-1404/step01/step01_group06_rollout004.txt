@triton.jit
def proj_gate_mask_kernel(
    x_ptr, mask_ptr,
    w_proj_left_ptr, w_gate_left_ptr,
    w_proj_right_ptr, w_gate_right_ptr,
    left_out_ptr, right_out_ptr,
    # strides: x stride [b,i,j,d], mask stride [b,i,j],
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_wproj_ld, stride_wproj_hd, # weight left proj transposed: (dim, hidden)
    stride_wgate_ld, stride_wgate_hd,
    stride_wrproj_ld, stride_wrproj_hd,
    stride_wrgate_ld, stride_wrgate_hd,
    # problem sizes:
    B, N, D, H,
    # compile-time constants:
    BLOCK_ROW: tl.constexpr, BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_ROW
    row_idx = row_start + tl.arange(0, BLOCK_ROW)
    mask = row_idx < B * N * N

    # decode (b,i,j)
    b = row_idx // (N * N)
    rem = row_idx % (N * N)
    i = rem // N
    j = rem % N

    # base offset for x
    x_base = b*stride_x_b + i*stride_x_i + j*stride_x_j

    # mask scalar
    mask_val = tl.load(mask_ptr + b*stride_mask_b + i*stride_mask_i + j*stride_mask_j, mask=mask, other=1.0)

    # allocate accumulators
    for h_off in range(0, H, BLOCK_H):
        # init accum
        acc_left_proj = tl.zeros((BLOCK_H,), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)
        acc_right_proj = tl.zeros((BLOCK_H,), dtype=tl.float32)
        acc_right_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)

        # loop over d dimension
        for d_off in range(0, D, BLOCK_D):
            # compute offsets
            d_idx = d_off + tl.arange(0, BLOCK_D)  # vector of dim indices
            # load x fragment
            x = tl.load(x_ptr + x_base + d_idx * stride_x_d, mask=mask[:, None], other=0.0)

            # load weight fragments
            # left_proj weight fragment shape (BLOCK_D, BLOCK_H)
            w_lproj = tl.load(w_proj_left_ptr + d_idx[:, None] * stride_wproj_ld + (h_off + tl.arange(0,BLOCK_H)) * stride_wproj_hd, mask=mask[:, None], other=0.0)
            w_lgate = tl.load(w_gate_left_ptr + d_idx[:, None] * stride_wgate_ld + (h_off + tl.arange(0,BLOCK_H)) * stride_wgate_hd, mask=mask[:, None], other=0.0)
            w_rproj = tl.load(w_proj_right_ptr + d_idx[:, None] * stride_wrproj_ld + (h_off + tl.arange(0,BLOCK_H)) * stride_wrproj_hd, mask=mask[:, None], other=0.0)
            w_rgate = tl.load(w_gate_right_ptr + d_idx[:, None] * stride_wrgate_ld + (h_off + tl.arange(0,BLOCK_H)) * stride_wrgate_hd, mask=mask[:, None], other=0.0)

            # dot: x (1, BLOCK_D) dot w (BLOCK_D, BLOCK_H) => (1,BLOCK_H)
            # Using tl.dot: input shapes (M,K) and (K,N)
            acc_left_proj += tl.dot(x[None, :], w_lproj)[0]  # shape (BLOCK_H)
            acc_left_gate += tl.dot(x[None, :], w_lgate)[0]
            acc_right_proj += tl.dot(x[None, :], w_rproj)[0]
            acc_right_gate += tl.dot(x[None, :], w_rgate)[0]

        # after D loop, apply sigmoid to gate accum
        left_gate = tl.sigmoid(acc_left_gate)
        right_gate = tl.sigmoid(acc_right_gate)

        # fuse
        left_val = acc_left_proj * left_gate * mask_val[:, None]  # broadcast mask per hidden
        right_val = acc_right_proj * right_gate * mask_val[:, None]

        # store to output
        left_ptr = left_out_ptr + b*stride_left_out_b + i*stride_left_out_i + j*stride_left_out_j + (h_off + tl.arange(0, BLOCK_H)) * stride_left_out_h
        right_ptr = right_out_ptr + b*stride_right_out_b + i*stride_right_out_i + j*stride_right_out_j + (h_off + tl.arange(0, BLOCK_H)) * stride_right_out_h

        tl.store(left_ptr, left_val, mask=mask[:, None])
        tl.store(right_ptr, right_val, mask=mask[:, None])

@triton.jit
def proj_gate_mask_kernel(
    x_ptr, mask_ptr,
    wproj_l_ptr, wgate_l_ptr,
    wproj_r_ptr, wgate_r_ptr,
    left_out_ptr, right_out_ptr,
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_wproj_l_d, stride_wproj_l_h,
    stride_wgate_l_d, stride_wgate_l_h,
    stride_wproj_r_d, stride_wproj_r_h,
    stride_wgate_r_d, stride_wgate_r_h,
    B, N, D, H,
    BLOCK_H: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode indices
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offset for x and mask
    x_base = b*stride_x_b + i*stride_x_i + j*stride_x_j
    mask_offset = b*stride_mask_b + i*stride_mask_i + j*stride_mask_j

    mask_val = tl.load(mask_ptr + mask_offset)  # scalar

    # Prepare outputs
    # Loop over hidden dimension
    for h_off in range(0, H, BLOCK_H):
        # accumulators
        acc_left_proj = tl.zeros((BLOCK_H,), dtype=tl.float32)
        acc_left_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)
        acc_right_proj = tl.zeros((BLOCK_H,), dtype=tl.float32)
        acc_right_gate = tl.zeros((BLOCK_H,), dtype=tl.float32)

        # loop over input dim
        for d_off in range(0, D, BLOCK_D):
            # dimension index vector
            d_range = d_off + tl.arange(0, BLOCK_D)
            d_mask = d_range < D

            # load x fragment
            x = tl.load(x_ptr + x_base + d_range * stride_x_d, mask=d_mask, other=0.0)  # shape (BLOCK_D)

            # weight fragments (BLOCK_D, BLOCK_H)
            w_lproj = tl.load(wproj_l_ptr + d_range[:, None] * stride_wproj_l_d + (h_off + tl.arange(0, BLOCK_H)) * stride_wproj_l_h, mask=(d_mask[:, None]), other=0.0)
            w_lgate = tl.load(wgate_l_ptr + d_range[:, None] * stride_wgate_l_d + (h_off + tl.arange(0, BLOCK_H)) * stride_wgate_l_h, mask=(d_mask[:, None]), other=0.0)
            w_rproj = tl.load(wproj_r_ptr + d_range[:, None] * stride_wproj_r_d + (h_off + tl.arange(0, BLOCK_H)) * stride_wproj_r_h, mask=(d_mask[:, None]), other=0.0)
            w_rgate = tl.load(wgate_r_ptr + d_range[:, None] * stride_wgate_r_d + (h_off + tl.arange(0, BLOCK_H)) * stride_wgate_r_h, mask=(d_mask[:, None]), other=0.0)

            # dot product
            # reshape x to (1,BLOCK_D) for dot
            acc_left_proj += tl.dot(x[None, :], w_lproj)[0]
            acc_left_gate += tl.dot(x[None, :], w_lgate)[0]
            acc_right_proj += tl.dot(x[None, :], w_rproj)[0]
            acc_right_gate += tl.dot(x[None, :], w_rgate)[0]

        # apply sigmoid gate
        left_gate = tl.sigmoid(acc_left_gate)
        right_gate = tl.sigmoid(acc_right_gate)

        # mask broadcast to hidden dim
        mask_vec = tl.broadcast_to(mask_val, (BLOCK_H,))  # shape (BLOCK_H)
        left_val = acc_left_proj * left_gate * mask_vec
        right_val = acc_right_proj * right_gate * mask_vec

        # store
        h_range = h_off + tl.arange(0, BLOCK_H)
        h_mask = h_range < H

        # compute output pointers offset
        left_ptr = left_out_ptr + b*stride_left_out_b + i*stride_left_out_i + j*stride_left_out_j + h_range * stride_left_out_h
        right_ptr = right_out_ptr + b*stride_right_out_b + i*stride_right_out_i + j*stride_right_out_j + h_range * stride_right_out_h

        tl.store(left_ptr, left_val.to(tl.float16), mask=h_mask)
        tl.store(right_ptr, right_val.to(tl.float16), mask=h_mask)

"""
TriMul “outgoing’’ forward pass (AlphaFold‑3) – Triton‑accelerated linear,
gating and mask fusion.

Algorithm
---------
1. Layer‑norm the input tensor (float32) → x.
2. For every pair (b,i,j) compute in a single Triton kernel
       left  = (x·W_left_proj ) * sigmoid(x·W_left_gate ) * mask
       right = (x·W_right_proj) * sigmoid(x·W_right_gate) * mask
   The kernel loads the input vector once per dimension block and
   performs four dot‑products (proj & gate for left & right) while
   accumulating in FP32.  All results are written out as FP32 tensors
   `left` and `right` of shape [B,N,N,H] (H = hidden_dim).

3. Batched matrix‑multiplication across the hidden dimension:
       out_h = left_h @ right_hᵀ   (h = 0 … H‑1)
   Implemented with a single torch.bmm on the reshaped tensors.

4. Layer‑norm + per‑head gating (`out_gate`).

5. Final linear projection back to `dim`.

The only non‑trivial computation that is not a BLAS call is the
fused projection/gate/mask kernel; everything else uses highly
optimised cuBLAS/cuDNN kernels.  The implementation works for
both masked and un‑masked inputs and for all shapes listed in the
test matrix.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fused projection, gating and mask
# --------------------------------------------------------------
@triton.jit
def proj_gate_mask_kernel(
    # ------------------------------------------------------------------
    # Pointers
    # ------------------------------------------------------------------
    x_ptr,               # [B,N,N,D]  input after LayerNorm
    mask_ptr,            # [B,N,N]    0/1 mask (float)
    wproj_l_ptr, wgate_l_ptr,   # [D,H] left  proj / gate (transposed)
    wproj_r_ptr, wgate_r_ptr,   # [D,H] right proj / gate (transposed)
    left_out_ptr, right_out_ptr,  # [B,N,N,H] outputs
    # ------------------------------------------------------------------
    # Strides (in elements, NOT bytes)
    # ------------------------------------------------------------------
    stride_x_b, stride_x_i, stride_x_j, stride_x_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_wproj_l_d, stride_wproj_l_h,
    stride_wgate_l_d, stride_wgate_l_h,
    stride_wproj_r_d, stride_wproj_r_h,
    stride_wgate_r_d, stride_wgate_r_h,
    stride_left_out_b, stride_left_out_i, stride_left_out_j, stride_left_out_h,
    stride_right_out_b, stride_right_out_i, stride_right_out_j, stride_right_out_h,
    # ------------------------------------------------------------------
    # Problem sizes (runtime)
    # ------------------------------------------------------------------
    B, N,
    # ------------------------------------------------------------------
    # Compile‑time constants
    # ------------------------------------------------------------------
    D: tl.constexpr,    # input channel dimension
    H: tl.constexpr,    # hidden dimension (≤ 128 in all tests)
    BLOCK_D: tl.constexpr,  # tile along D (e.g. 32)
):
    """Fused linear‑projection + sigmoid‑gate + mask.
    One program processes a single (b,i,j) pair.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # ---------- decode (b,i,j) ----------
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # ---------- mask ----------
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_off)          # scalar FP32

    # ---------- input base ----------
    x_base = b * stride_x_b + i * stride_x_i + j * stride_x_j

    # ---------- accumulators ----------
    acc_lproj = tl.zeros((H,), dtype=tl.float32)
    acc_lgate = tl.zeros((H,), dtype=tl.float32)
    acc_rproj = tl.zeros((H,), dtype=tl.float32)
    acc_rgate = tl.zeros((H,), dtype=tl.float32)

    # hidden index vector (compile‑time)
    hid = tl.arange(0, H)

    # ---------- loop over input dimension ----------
    for d_off in range(0, D, BLOCK_D):
        # mask for the current D‑tile
        d = d_off + tl.arange(0, BLOCK_D)
        d_mask = d < D

        # load input tile x[b,i,j,d]
        x_vec = tl.load(
            x_ptr + x_base + d * stride_x_d,
            mask=d_mask,
            other=0.0,
        )                                            # (BLOCK_D,)

        # ----- weight tiles -----
        # shape (BLOCK_D, H)  –  column‑major because weight is stored transposed (D,H)
        w_lproj = tl.load(
            wproj_l_ptr + d[:, None] * stride_wproj_l_d + hid[None, :] * stride_wproj_l_h,
            mask=(d_mask[:, None]),
            other=0.0,
        )
        w_lgate = tl.load(
            wgate_l_ptr + d[:, None] * stride_wgate_l_d + hid[None, :] * stride_wgate_l_h,
            mask=(d_mask[:, None]),
            other=0.0,
        )
        w_rproj = tl.load(
            wproj_r_ptr + d[:, None] * stride_wproj_r_d + hid[None, :] * stride_wproj_r_h,
            mask=(d_mask[:, None]),
            other=0.0,
        )
        w_rgate = tl.load(
            wgate_r_ptr + d[:, None] * stride_wgate_r_d + hid[None, :] * stride_wgate_r_h,
            mask=(d_mask[:, None]),
            other=0.0,
        )

        # ----- dot products (1×BLOCK_D)·(BLOCK_D×H) → (1×H) -----
        # tl.dot works on 2‑D matrices, so we view the vector as a 1×BLOCK_D row.
        p = tl.dot(x_vec[None, :], w_lproj)        # (1, H)
        acc_lproj += p[0]
        p = tl.dot(x_vec[None, :], w_lgate)
        acc_lgate += p[0]
        p = tl.dot(x_vec[None, :], w_rproj)
        acc_rproj += p[0]
        p = tl.dot(x_vec[None, :], w_rgate)
        acc_rgate += p[0]

    # ---------- apply sigmoid gating ----------
    left_gate  = 1.0 / (1.0 + tl.exp(-acc_lgate))
    right_gate = 1.0 / (1.0 + tl.exp(-acc_rgate))

    # ---------- mask broadcast ----------
    mask_vec = mask_val * tl.ones((H,), dtype=tl.float32)

    # ---------- final fused values ----------
    left_val  = acc_lproj * left_gate  * mask_vec
    right_val = acc_rproj * right_gate * mask_vec

    # ---------- store results ----------
    left_ptr = (
        left_out_ptr
        + b * stride_left_out_b
        + i * stride_left_out_i
        + j * stride_left_out_j
        + hid * stride_left_out_h
    )
    right_ptr = (
        right_out_ptr
        + b * stride_right_out_b
        + i * stride_right_out_i
        + j * stride_right_out_j
        + hid * stride_right_out_h
    )
    tl.store(left_ptr,  left_val,  mask=None)
    tl.store(right_ptr, right_val, mask=None)


# --------------------------------------------------------------
# TriMul entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Mask handling (mask may be None or omitted when nomask=True)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)
    mask = mask.to(dtype)               # ensure same dtype as input

    # ------------------------------------------------------------------
    # Pull weights (all weights are stored as FP32 in the tests)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]      # (hidden, dim)
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep FP32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B,N,N,dim] – FP32

    # ------------------------------------------------------------------
    # 2. Compute out‑gate (used after the big GEMM)
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Fuse projection, gating and mask with Triton
    # ------------------------------------------------------------------
    # Transpose weight matrices so that they are stored as (D, H)
    wproj_l_T = left_proj_weight.t().contiguous()
    wgate_l_T = left_gate_weight.t().contiguous()
    wproj_r_T = right_proj_weight.t().contiguous()
    wgate_r_T = right_gate_weight.t().contiguous()

    # Allocate output buffers (FP32)
    left  = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=device)
    right = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in element counts)
    x_str   = input_tensor.stride()
    mask_str= mask.stride()
    wpl_str = wproj_l_T.stride()     # (D stride, H stride)
    wgl_str = wgate_l_T.stride()
    wpr_str = wproj_r_T.stride()
    wgr_str = wgate_r_T.stride()
    left_str  = left.stride()
    right_str = right.stride()

    total_rows = bs * seq_len * seq_len
    grid = (triton.cdiv(total_rows, 1),)   # one program per (b,i,j)

    proj_gate_mask_kernel[grid](
        # pointers
        x_ptr               = x,
        mask_ptr            = mask,
        wproj_l_ptr         = wproj_l_T,
        wgate_l_ptr         = wgate_l_T,
        wproj_r_ptr         = wproj_r_T,
        wgate_r_ptr         = wgate_r_T,
        left_out_ptr        = left,
        right_out_ptr       = right,
        # strides
        stride_x_b          = x_str[0], stride_x_i = x_str[1], stride_x_j = x_str[2], stride_x_d = x_str[3],
        stride_mask_b       = mask_str[0], stride_mask_i = mask_str[1], stride_mask_j = mask_str[2],
        stride_wproj_l_d    = wpl_str[0], stride_wproj_l_h = wpl_str[1],
        stride_wgate_l_d    = wgl_str[0], stride_wgate_l_h = wgl_str[1],
        stride_wproj_r_d    = wpr_str[0], stride_wproj_r_h = wpr_str[1],
        stride_wgate_r_d    = wgr_str[0], stride_wgate_r_h = wgr_str[1],
        stride_left_out_b   = left_str[0], stride_left_out_i = left_str[1],
        stride_left_out_j   = left_str[2], stride_left_out_h = left_str[3],
        stride_right_out_b  = right_str[0], stride_right_out_i = right_str[1],
        stride_right_out_j  = right_str[2], stride_right_out_h = right_str[3],
        # problem sizes
        B = bs, N = seq_len,
        # compile‑time constants
        D = dim, H = hidden_dim, BLOCK_D = 32,
    )

    # ------------------------------------------------------------------
    # 4. Batched GEMM across the hidden dimension
    #    out_h = left_h @ right_hᵀ   (h = 0 … hidden_dim‑1)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)
    right_perm = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)

    # GEMM – fp32 (uses Tensor‑cores on H100)
    out_mat = torch.bmm(left_perm, right_perm.transpose(1, 2))   # (B*H, N, N)

    # restore original layout [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm + per‑head gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 6. Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    # --------------------------------------------------
    # Return FP32 tensor
    # --------------------------------------------------
    return out
