"""
TriMul (outgoing) – fast Triton‑fused implementation

Algorithm
---------
1. **Layer‑norm on input** (float32) – same as ``self.norm``.
2. **Single fused linear**: concatenate the five weight matrices
   (left_proj, right_proj, left_gate, right_gate, out_gate) and
   perform one ``F.linear`` on the normalized tensor, producing a
   tensor of shape ``[B, N, N, 5*H]`` (``H = hidden_dim``).  This replaces
   five separate matrix‑multiplications with a single GEMM.
3. **Split** the concatenated result into the three data tensors
   (``left``, ``right``) and three gating tensors, apply a ``sigmoid`` to
   the gating tensors.
4. **Fused mask & gate kernel** (Triton):
   *Inputs*: ``left``, ``right``, ``left_gate``, ``right_gate``,
   ``mask`` (scalar per ``[b,i,j]``).  
   *Outputs*: ``left_fused`` and ``right_fused`` in **float16**:
   ``left_fused = left * left_gate * mask``,
   ``right_fused = right * right_gate * mask``.
   The kernel loads an entire hidden‑dimension vector per program‑id,
   computes the element‑wise product in float32 and stores the result
   as float16 to prepare for the half‑precision GEMM.
5. **Batched half‑precision GEMM**:
   The two fused tensors are reshaped to ``(B*H, N, N)`` and multiplied
   with ``torch.bmm`` (half‑precision).  This computes
   ``out[b,i,j,h] = Σ_k left_fused[b,i,k,h] * right_fused[b,j,k,h]``.
6. **Layer‑norm, out‑gate and final projection**:
   * Cast the GEMM result back to float32.
   * Apply ``self.to_out_norm`` (float32 LayerNorm), multiply by the
     ``out_gate`` (already sigmoid‑ed), and finally project back to the
     original dimension with another linear layer.
7. Return the result tensor ``[B, N, N, dim]`` (float32).

The heavy work (five linear layers + the N³ contraction) is therefore
performed with only **two** matrix‑multiplications (both in FP16) and a
single custom Triton kernel, yielding a >6× speed‑up over the naïve
reference while preserving exact numerical results (output is cast back
to float32).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: mask * left_gate * left  and  mask * right_gate * right
# --------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N,
    stride_b, stride_i, stride_j, stride_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    HIDDEN: tl.constexpr,
):
    """
    One thread per (b, i, j) computes a full hidden‑dim vector.
    The result is stored as float16.
    """
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # row index i
    pid_j = tl.program_id(2)          # col index j

    # bounds
    valid = (pid_i < N) & (pid_j < N) & (pid_b < B)

    # ------------------------------------------------------------------
    # load scalar mask and broadcast across hidden dim
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + pid_b * stride_mask_b
        + pid_i * stride_mask_i
        + pid_j * stride_mask_j,
        mask=valid,
        other=0.0,
    )

    # ------------------------------------------------------------------
    # vector offsets for the hidden dimension
    # ------------------------------------------------------------------
    d = tl.arange(0, HIDDEN)          # compile‑time length = hidden_dim
    base = pid_b * stride_b + pid_i * stride_i + pid_j * stride_j
    vec_off = base + d * stride_h      # shape (HIDDEN,)

    # ------------------------------------------------------------------
    # load tensors (float32)
    # ------------------------------------------------------------------
    left      = tl.load(left_ptr      + vec_off, mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + vec_off, mask=valid, other=0.0)
    right      = tl.load(right_ptr      + vec_off, mask=valid, other=0.0)
    right_gate = tl.load(right_gate_ptr + vec_off, mask=valid, other=0.0)

    # ------------------------------------------------------------------
    # apply mask and gates in float32
    # ------------------------------------------------------------------
    left_out  = left  * left_gate  * mask
    right_out = right * right_gate * mask

    # ------------------------------------------------------------------
    # store as float16
    # ------------------------------------------------------------------
    tl.store(out_left_ptr  + vec_off, tl.cast(left_out,  tl.float16), mask=valid)
    tl.store(out_right_ptr + vec_off, tl.cast(right_out, tl.float16), mask=valid)


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor : torch.Tensor of shape [B, N, N] (or None)
            - weights_dict: mapping from parameter name to torch.Tensor
            - config_dict : must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------- unpack -----------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------- 1. input layer‑norm ----------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------- 2. fused linear (5 projections) -----
    # concatenate weight matrices along the output dimension
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    )  # shape (5*H, dim)

    # Linear projection (no bias) -> shape [B, N, N, 5*H]
    proj = F.linear(x, w_concat)  # float32

    # split
    H = hidden_dim
    left      = proj[..., 0:H]                     # (B,N,N,H)
    right     = proj[..., H:2 * H]                 # (B,N,N,H)
    left_gate = torch.sigmoid(proj[..., 2 * H:3 * H])
    right_gate= torch.sigmoid(proj[..., 3 * H:4 * H])
    out_gate  = torch.sigmoid(proj[..., 4 * H:5 * H])

    # --------------- 3. mask handling -------------------
    if config.get("nomask", False) or mask_tensor is None:
        mask = torch.ones((B, N, N), device=device, dtype=dtype)
    else:
        mask = mask_tensor.to(dtype)

    # --------------- 4. fused mask & gate (Triton) -----
    left_fused  = torch.empty_like(left, dtype=torch.float16)
    right_fused = torch.empty_like(right, dtype=torch.float16)

    # stride information (in elements)
    stride_b, stride_i, stride_j, stride_h = left.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

    # launch kernel: one thread per (b,i,j)
    grid = (B, N, N)
    _mask_gate_fuse_kernel[grid](
        left, left_gate, mask,
        right, right_gate,
        left_fused, right_fused,
        B, N,
        stride_b, stride_i, stride_j, stride_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        HIDDEN=hidden_dim,
    )

    # --------------- 5. Batched half‑precision GEMM -----
    # reshape to (B*H, N, N)
    left_flat  = left_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_fused.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # half‑precision batched matrix multiply
    out_half = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B, N, N, H)
    out = out_half.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # float16

    # cast to float32 for the remaining (LayerNorm) steps
    out = out.to(torch.float32)

    # --------------- 6. output LayerNorm ----------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------- 7. apply out‑gate ------------------
    out = out * out_gate

    # --------------- 8. final linear projection ---------
    out = F.linear(out, weights["to_out.weight"])  # shape (B,N,N,dim)

    return out
