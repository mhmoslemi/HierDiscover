import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing version) – forward pass only.

The core N³ operation is implemented as a batched GEMM in Triton:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are the projected & gated tensors of shape
    [B, N, N, H]   (H = hidden_dim).

The kernel works on a batch of independent matrix multiplications
    (B*H) × (N×N) @ (N×N)ᵀ
using half‑precision for the inputs and a float32 accumulator.
All surrounding LayerNorm / Linear / Sigmoid work is performed with
PyTorch primitives.
"""

import math
import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch, M, N, K,                          # dimensions
    stride_ah, stride_am, stride_ak,         # A strides
    stride_bh, stride_bk, stride_bn,         # Bᵀ strides
    stride_ch, stride_cm, stride_cn,         # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ Bᵀ for a batch of matrices.
    A : [batch, M, K]   (float16)
    B : [batch, K, N]   (float16)  == rightᵀ
    C : [batch, M, N]   (float32)
    """
    pid_batch = tl.program_id(0)                 # which (b,d) pair
    pid_m = tl.program_id(1)                     # tile row
    pid_n = tl.program_id(2)                     # tile col

    # ------------------------------------------------------------------
    # Offsets for the current block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of A and Bᵀ blocks
    a_ptrs = a_ptr + pid_batch * stride_ah \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * stride_bh \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for cur_k in range(0, num_k_blocks):
        # Global k offset for masking
        k_offset = cur_k * BLOCK_K

        # Boundary masks
        a_mask = (offs_m < M)[:, None] & ((k_offset + offs_k) < K)[None, :]
        b_mask = ((k_offset + offs_k) < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)     # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)     # fp16

        acc += tl.dot(a, b)                            # fp32 accumulation

        # Advance pointers to next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store the result (apply mask for the trailing edges)
    c_ptrs = c_ptr + pid_batch * stride_ch \
                 + offs_m[:, None] * stride_cm \
                 + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using Triton for the
    N³ matrix‑multiplication part.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]      (float32 or bool)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # retrieve weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right = F.linear(x, right_proj_weight) # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Apply mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure same dtype as the projections (float32 at this point)
        mask = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to half‑precision for the heavy GEMM
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # reshape to (batch * hidden_dim, N, N) = independent matrices
    left_mat  = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # we need rightᵀ for the multiplication
    right_mat_T = right_mat.transpose(1, 2).contiguous()   # shape (B*H, N, N)

    # output buffer (float32, because the accumulator of the kernel is fp32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 6) Triton batched GEMM launch
    # ------------------------------------------------------------------
    # Tuning parameters – these work well for the target sizes on an H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # strides (in elements, not bytes)
    stride_ah = left_mat.stride(0)   # batch stride
    stride_am = left_mat.stride(1)   # row stride (i)
    stride_ak = left_mat.stride(2)   # column stride (k)

    stride_bh = right_mat_T.stride(0)   # batch stride
    stride_bk = right_mat_T.stride(1)   # K‑dimension stride (after transpose => 1)
    stride_bn = right_mat_T.stride(2)   # N‑dimension stride

    stride_ch = out_mat.stride(0)   # batch stride
    stride_cm = out_mat.stride(1)   # row stride
    stride_cn = out_mat.stride(2)   # col stride

    grid = (
        B * hidden_dim,                     # batch dimension (B * H)
        (N + BLOCK_M - 1) // BLOCK_M,       # rows
        (N + BLOCK_N - 1) // BLOCK_N,       # cols
    )

    batched_matmul_kernel[grid](
        left_mat, right_mat_T, out_mat,
        B * hidden_dim, N, N, N,
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8) Output LayerNorm + gating + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # final projection back to dim

    return out
