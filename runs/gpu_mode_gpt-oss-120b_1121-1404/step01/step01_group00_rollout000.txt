"""
TriMul “outgoing” kernel (AlphaFold3) – fast fused reduction.

The forward pass follows the reference PyTorch module but replaces the
O(N³) reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with a single Triton kernel that performs a batched matrix‑multiply where the
second operand is transposed:
    out[p] = left[p] @ right[p].T      (p = batch * hidden_dim)

Both operands are loaded as FP16 (Tensor‑Core friendly) while the
accumulator stays in FP32 for numeric stability.  The kernel tiles the
M×N output with BLOCK_M×BLOCK_N work‑items and traverses the K‑dimension in
BLOCK_K steps, using `tl.dot` which maps to WMMA on an H100.  After the
reduction we reshape, apply a second LayerNorm, gating and a final linear
projection – all done in PyTorch to keep the implementation simple.

The code works for any batch size (1‑2), sequence length up to 1024,
hidden dimension 128‑384 and optional masking.
"""

import torch
import triton
import triton.language as tl
import math

# ----------------------------------------------------------------------
# Triton kernel: batched matmul with the second operand transposed.
# out[p,i,j] = Σ_k left[p,i,k] * right[p,j,k]
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr,                      # const *[fp16] left  (P, M, K)
    b_ptr,                      # const *[fp16] right (P, N, K)
    c_ptr,                      #       *[fp32] out   (P, M, N)
    batch, M, N, K,             # int32 shape parameters
    stride_apb, stride_amn, stride_ak,   # int64 strides for a
    stride_bpb, stride_bmn, stride_bk,   # int64 strides for b
    stride_cpb, stride_cmn, stride_cnk,   # int64 strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # --------------------------------------------------------------
    # program ids
    # --------------------------------------------------------------
    pid_b = tl.program_id(2)          # batch * hidden index
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column

    # --------------------------------------------------------------
    # offsets within the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K-loop – iterate over blocks of the reduction dimension
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left matrix block : (BLOCK_M, BLOCK_K)
        a_ptrs = (a_ptr
                  + pid_b * stride_apb
                  + offs_m[:, None] * stride_amn
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # right matrix transposed block : (BLOCK_K, BLOCK_N)
        b_ptrs = (b_ptr
                  + pid_b * stride_bpb
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bmn)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # Tensor‑Core dot (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write out result
    # --------------------------------------------------------------
    c_ptrs = (c_ptr
              + pid_b * stride_cpb
              + offs_m[:, None] * stride_cmn
              + offs_n[None, :] * stride_cnk)
    tl.store(c_ptrs, acc,
             mask=(mask_m[:, None] & mask_n[None, :]))

# Main entry point – the custom kernel expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]   (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model parameters (see keys below)
        - config       : dict containing at least
                         "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5
    nomask     = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        m = mask[..., None].to(left.dtype)           # [B,N,N,1]
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gating – sigmoid of a linear projection of the *pre‑norm* tensor
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul reduction (batched matmul, right operand transposed)
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, N) and fuse batch & hidden dims
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H                         # combined batch*head dimension

    # flatten to (P, N, N)
    left_flat  = left.view(P, N, N).contiguous()
    right_flat = right.view(P, N, N).contiguous()

    # cast to FP16 for Tensor‑Core friendly loads
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # output buffer (FP32 accumulator)
    out_fp32 = torch.empty((P, N, N), dtype=torch.float32, device=device)

    # --------------------------------------------------------------
    # auto‑tune block sizes (empirically good for the tested ranges)
    # --------------------------------------------------------------
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
    else:
        BLOCK_M = BLOCK_N = 128
    BLOCK_K = 32   # must be a multiple of 16 for Tensor Cores

    grid = (
        triton.cdiv(N, BLOCK_M),   # M‑tiles
        triton.cdiv(N, BLOCK_N),   # N‑tiles
        P,                         # batch*head
    )

    _batched_matmul_transpose_kernel[grid](
        # pointers
        left_fp16,
        right_fp16,
        out_fp32,
        # shape parameters
        P, N, N, N,
        # strides (in elements, not bytes)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2),
        # compile‑time tile sizes
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # runtime launch configs (tuned for H100)
        num_warps=8,
        num_stages=2,
    )

    # reshape back to original layout: (B, N, N, hidden)
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Final LayerNorm, gating and output projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcast gate over hidden dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
