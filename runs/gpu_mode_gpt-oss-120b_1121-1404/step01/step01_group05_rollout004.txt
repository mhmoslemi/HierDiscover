@triton.jit
def _tri_mul_fused_kernel(
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    B, N, HIDDEN,
    # strides
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_lgb, stride_lgi, stride_lgk, stride_lgh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_rgb, stride_rgi, stride_rgk, stride_rgh,
    stride_mb, stride_mi, stride_mk,
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile-time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, HIDDEN: tl.constexpr
):
    bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch and head
    b = bh // HIDDEN
    h = bh - b * HIDDEN  # modulo

    # row and column indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left projection and gate and mask for (i,k)
        left = tl.load(
            left_ptr + b * stride_lb + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk + h * stride_lh,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr + b * stride_lgb + offs_m[:, None] * stride_lgi + offs_k[None, :] * stride_lgk + h * stride_lgh,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        mask_left = tl.load(
            mask_ptr + b * stride_mb + offs_m[:, None] * stride_mi + offs_k[None, :] * stride_mk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # fused left tile (M, K) fp16
        left_fused = left * left_gate * mask_left

        # Load right projection and gate and mask for (j,k) and transpose to (K,N)
        # Load as (K,N) directly:
        right_T = tl.load(
            right_ptr + b * stride_rb + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj + h * stride_rh,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        right_gate_T = tl.load(
            right_gate_ptr + b * stride_rgb + offs_k[:, None] * stride_rgk + offs_n[None, :] * stride_rgi + h * stride_rgh,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        mask_right_T = tl.load(
            mask_ptr + b * stride_mb + offs_n[None, :] * stride_mi + offs_k[:, None] * stride_mk,
            mask=mask_n[None, :] & mask_k[:, None],
            other=0.0,
        )
        right_fused_T = right_T * right_gate_T * mask_right_T

        # dot product (M,N) accumulation
        acc += tl.dot(left_fused, right_fused_T)

    # store result
    out_ptrs = out_ptr + b * stride_ob + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj + h * stride_oh
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul “outgoing” update – highly‑fused Triton implementation.

The forward pass consists of:
1. Layer‑norm on the input tensor.
2. Linear projections and gating logits (all cast to FP16).
3. A single Triton kernel that fuses:
   • element‑wise mask multiplication,
   • left‑gate * left‑proj,
   • right‑gate * right‑proj,
   • the ternary contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   The contraction is performed as a tiled batched GEMM (one GEMM per
   (batch, hidden) pair) with FP32 accumulation.
4. FP16 → FP32 conversion, a second Layer‑norm, out‑gate multiplication,
   and a final linear projection back to the model dimension.

The kernel is written for H100 (CUDA 12, Triton 3.3.1) and uses
BLOCK_M = BLOCK_N = 64, BLOCK_K = 32 – a good trade‑off between
shared‑memory usage and occupancy for the supported sequence lengths
(256‑1024).  All heavy work (the N³ contraction) is done on‑GPU,
leaving only cheap point‑wise ops in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask, gating and the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_fused_kernel(
    # pointers
    left_ptr, left_gate_ptr, right_ptr, right_gate_ptr,
    mask_ptr, out_ptr,
    # runtime arguments
    B, N,
    # strides for the four 4‑D tensors (B,N,N,H) – all are contiguous
    stride_lb, stride_li, stride_lk, stride_lh,      # left_proj
    stride_lgb, stride_lgi, stride_lgk, stride_lgh, # left_gate
    stride_rb, stride_rj, stride_rk, stride_rh,     # right_proj
    stride_rgb, stride_rgi, stride_rgk, stride_rgh, # right_gate
    stride_mb, stride_mi, stride_mk,                # mask (B,N,N)
    stride_ob, stride_oi, stride_oj, stride_oh,     # output
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,                           # hidden dimension (constexpr)
):
    bh = tl.program_id(0)          # packed (batch, hidden) index
    pid_m = tl.program_id(1)       # row block
    pid_n = tl.program_id(2)       # col block

    # unpack batch and hidden index from the packed program id
    b = bh // HIDDEN               # integer division, batch index
    h = bh - b * HIDDEN            # remainder -> hidden index

    # -------------------------- tile indices --------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑indices

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------- k‑loop (inner reduction) --------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left side (i,k) -------------------------------------------------
        left = tl.load(
            left_ptr
            + b * stride_lb
            + offs_m[:, None] * stride_li
            + offs_k[None, :] * stride_lk
            + h * stride_lh,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        left_gate = tl.load(
            left_gate_ptr
            + b * stride_lgb
            + offs_m[:, None] * stride_lgi
            + offs_k[None, :] * stride_lgk
            + h * stride_lgh,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        mask_left = tl.load(
            mask_ptr
            + b * stride_mb
            + offs_m[:, None] * stride_mi
            + offs_k[None, :] * stride_mk,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # fused left tile, still FP16
        left_fused = left * left_gate * mask_left

        # ---- right side (j,k) – loaded transposed as (k,j) --------------------
        #   We need right_T[k, n] where n == j
        right_T = tl.load(
            right_ptr
            + b * stride_rb
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rj
            + h * stride_rh,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        right_gate_T = tl.load(
            right_gate_ptr
            + b * stride_rgb
            + offs_k[:, None] * stride_rgk
            + offs_n[None, :] * stride_rgi
            + h * stride_rgh,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        mask_right_T = tl.load(
            mask_ptr
            + b * stride_mb
            + offs_n[None, :] * stride_mi
            + offs_k[:, None] * stride_mk,
            mask=mask_n[None, :] & mask_k[:, None],
            other=0.0,
        )
        right_fused_T = right_T * right_gate_T * mask_right_T

        # ---- accumulate -------------------------------------------------------
        acc += tl.dot(left_fused, right_fused_T)   # (M,N)  FP32

    # -------------------------- write result --------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_m[:, None] * stride_oi
        + offs_n[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point – the Python side of the TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the **outgoing** TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N]   (bool/float, may be None)
        - weights_dict : dict mapping weight names to torch.Tensor
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------- unpack -------------------------------------------------
    if len(data) == 4:
        input_tensor, mask_tensor, weights, config = data
    else:           # graceful fallback
        input_tensor, weights, config = data
        mask_tensor = None

    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape

    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------- 1️⃣ LayerNorm on the input ----------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]  (float32)

    # cast once for all the cheap linear/gate operations
    x_f16 = x.to(torch.float16)

    # ------------------- 2️⃣ Linear projections (FP16) -------------------------
    left_proj = F.linear(x_f16, weights["left_proj.weight"].to(torch.float16)).contiguous()
    right_proj = F.linear(x_f16, weights["right_proj.weight"].to(torch.float16)).contiguous()

    # ------------------- 3️⃣ Gating logits + sigmoid (FP16) -------------------
    left_gate = torch.sigmoid(
        F.linear(x_f16, weights["left_gate.weight"].to(torch.float16))
    ).contiguous()
    right_gate = torch.sigmoid(
        F.linear(x_f16, weights["right_gate.weight"].to(torch.float16))
    ).contiguous()
    out_gate = torch.sigmoid(
        F.linear(x_f16, weights["out_gate.weight"].to(torch.float16))
    ).contiguous()   # shape [B, N, N, hidden]

    # ------------------- 4️⃣ Mask (FP16) ---------------------------------------
    if nomask or mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask = mask_tensor.to(dtype=torch.float16, device=device)

    # ------------------- 5️⃣ Allocate output buffer (FP16) --------------------
    out_f16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # ------------------- 6️⃣ Load strides for the kernel -----------------------
    # all tensors are contiguous ⇒ stride pattern (B,N,N,H)
    stride_lb, stride_li, stride_lk, stride_lh = left_proj.stride()
    stride_lgb, stride_lgi, stride_lgk, stride_lgh = left_gate.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_proj.stride()
    stride_rgb, stride_rgi, stride_rgk, stride_rgh = right_gate.stride()
    stride_mb, stride_mi, stride_mk = mask.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out_f16.stride()

    # ------------------- 7️⃣ Triton launch ------------------------------------
    # One GEMM per (batch, hidden) pair
    num_bh = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (num_bh, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_fused_kernel[grid](
        left_proj,
        left_gate,
        right_proj,
        right_gate,
        mask,
        out_f16,
        B,
        N,
        # strides
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_lgb, stride_lgi, stride_lgk, stride_lgh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_rgb, stride_rgi, stride_rgk, stride_rgh,
        stride_mb, stride_mi, stride_mk,
        stride_ob, stride_oi, stride_oj, stride_oh,
        # compile‑time constants
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        HIDDEN=hidden_dim,
    )
    # ---------------------------------------------------------------
    # 8️⃣ Cast back to FP32 for the remaining (layer‑norm, linear) steps
    out = out_f16.to(torch.float32)

    # 9️⃣ Output LayerNorm (self.to_out_norm)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # 10️⃣ Apply out‑gate (broadcasted over the hidden dimension)
    out = out * out_gate.to(torch.float32)

    # 11️⃣ Final linear projection back to model dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
