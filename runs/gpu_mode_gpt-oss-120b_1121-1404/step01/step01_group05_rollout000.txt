TriMul (outgoing) – fast forward implementation

Algorithm
---------
1.  Layer‑norm on the input tensor (float32).
2.  Cast to FP16 for the heavy linear/gating work.
3.  Linear projections → left, right (FP16).
4.  Linear gates → left_gate, right_gate, out_gate (FP16, sigmoid).
5.  Optional mask (FP16).  A single Triton kernel fuses:
        left_fused  = left  * left_gate  * mask
        right_fused = right * right_gate * mask
6.  Contract over the middle sequence dimension k:
        out[b,i,j,d] = Σₖ left_fused[b,i,k,d] * right_fused[b,j,k,d]
    Implemented as a batched GEMM on reshaped tensors
    (half‑precision torch.bmm, which on H100 is ~3‑5× faster than a
    naïve triple loop).
7.  Layer‑norm on the hidden dimension (FP16 → up‑casted to FP32 inside
    `torch.nn.functional.layer_norm` for stability).
8.  Apply the output gate (element‑wise, still FP16) and a final
    linear projection back to the model dimension.
9.  Cast the result to float32 (the required output dtype).

Only the mask‑gate fusion is performed in a custom Triton kernel; all
other operations use highly‑optimised PyTorch kernels (FP16 GEMM,
layer‑norm, etc.).  The implementation works for all test configurations
(`bs ∈ {1,2}`, `seq_len ∈ {256,512,768,1024}`, `dim ∈ {128,384}`,
`hidden_dim = 128`) and supports both masked and un‑masked inputs.

    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # row  i
    pid_j = tl.program_id(2)   # col  j

    # --------------------------------------------------------------
    # bounds
    # --------------------------------------------------------------
    valid = (pid_i < N) & (pid_j < N) & (pid_b < B)

    # --------------------------------------------------------------
    # load scalar mask and broadcast to the hidden dimension
    # --------------------------------------------------------------
    mask_val = tl.load(
        mask_ptr
        + pid_b * stride_mask_b
        + pid_i * stride_mask_i
        + pid_j * stride_mask_j,
        mask=valid,
        other=tl.float16(0.0)   # will be multiplied = 0 → no contribution
    )

    # --------------------------------------------------------------
    # offsets for the hidden dimension (vector of length HIDDEN)
    # --------------------------------------------------------------
    d = tl.arange(0, HIDDEN)                     # (HIDDEN,)
    base = pid_b * stride_b + pid_i * stride_i + pid_j * stride_j
    vec_off = base + d * stride_h                 # (HIDDEN,)

    # --------------------------------------------------------------
    # load vectors
    # --------------------------------------------------------------
    left      = tl.load(left_ptr      + vec_off, mask=valid, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + vec_off, mask=valid, other=tl.float16(0.0))
    right     = tl.load(right_ptr     + vec_off, mask=valid, other=tl.float16(0.0))
    right_gate= tl.load(right_gate_ptr+ vec_off, mask=valid, other=tl.float16(0.0))

    # --------------------------------------------------------------
    # apply mask and gates
    # --------------------------------------------------------------
    left_out  = left  * left_gate  * mask_val
    right_out = right * right_gate * mask_val

    # --------------------------------------------------------------
    # store results
    # --------------------------------------------------------------
    tl.store(out_left_ptr  + vec_off, left_out,  mask=valid)
    tl.store(out_right_ptr + vec_off, right_out, mask=valid)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
            - mask_tensor  : torch.Tensor[B, N, N]       (float32) or None
            - weights_dict : mapping from weight names to tensors
            - config_dict  : must contain "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim] float32

    # ------------------------------------------------------------------
    # 2) Cast to FP16 for heavy work
    # ------------------------------------------------------------------
    x_fp16 = x.half()

    # ------------------------------------------------------------------
    # 3) Linear projections (no bias) – FP16
    # ------------------------------------------------------------------
    left  = F.linear(x_fp16, weights["left_proj.weight"].half())   # [B,N,N,hidden]
    right = F.linear(x_fp16, weights["right_proj.weight"].half())

    # ------------------------------------------------------------------
    # 4) Gating logits + sigmoid – FP16
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_fp16, weights["left_gate.weight"].half()))
    right_gate = torch.sigmoid(F.linear(x_fp16, weights["right_gate.weight"].half()))
    out_gate   = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].half()))  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 5) Prepare mask (FP16) – 1.0 if nomask or None
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask_tensor is None:
        mask = torch.ones((B, N, N), device=device, dtype=torch.float16)
    else:
        mask = mask_tensor.to(dtype=torch.float16)

    # ------------------------------------------------------------------
    # 6) Allocate fused tensors
    # ------------------------------------------------------------------
    left_fused  = torch.empty_like(left)   # FP16
    right_fused = torch.empty_like(right)  # FP16

    # ------------------------------------------------------------------
    # 7) Triton kernel – fuse mask and gating
    # ------------------------------------------------------------------
    stride_b, stride_i, stride_j, stride_h = left.stride()          # same for left/right
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

    # grid: (batch, row, col)
    grid = (B, N, N)
    _mask_gate_fuse_kernel[grid](
        left, left_gate, mask,
        right, right_gate,
        left_fused, right_fused,
        B, N, hidden_dim,
        stride_b, stride_i, stride_j, stride_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        HIDDEN=hidden_dim,
    )

    # ------------------------------------------------------------------
    # 8) Core contraction: Σₖ left_fused[...,k,:] * right_fused[...,k,:]
    #    Implemented as batched GEMM in FP16.
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat  = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication (i‑k) × (j‑k)ᵀ → (i‑j) per hidden slice
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 9) Output LayerNorm (apply in float32 for numerical stability)
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B,N,N,hidden]  (float32)

    # ------------------------------------------------------------------
    # 10) Apply output gate (convert gate to float32) and final projection
    # ------------------------------------------------------------------
    out_gate_fp32 = out_gate.float()
    gated = out_norm * out_gate_fp32                     # still float32

    # final linear back to model dim – compute in FP16 and cast back
    gated_fp16 = gated.half()
    out_proj = F.linear(gated_fp16, weights["to_out.weight"].half())   # [B,N,N,dim] FP16
    out_final = out_proj.float()                                      # cast to float32

    return out_final
