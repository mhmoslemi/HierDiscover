"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated gating.

Algorithm
---------
1. Layer‑norm the input (FP16)   :  x = LN(x)
2. Fuse the two linear projections and their gates:
       left_proj  , left_gate = split( x @ [W_left ; W_left_gate] )
       right_proj , right_gate= split( x @ [W_right; W_right_gate] )
   (single matmul per side → 2 matmuls instead of 4)
3. Compute the out‑gate: out_gate = sigmoid( x @ W_out_gate )
4. Fuse element‑wise mask & sigmoid using a Triton kernel:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
5. Batched GEMM across the hidden dimension (H) in FP16:
       out_h = left_h @ right_hᵀ   for every h ∈ [0, H)
   (implemented with torch.bmm on a (B·H, N, N) view)
6. Cast to FP32, apply the output layer‑norm, the out‑gate,
   and the final linear projection back to the model dimension.
All heavy memory‑traffic‑bound work (steps 2–5) runs in half‑precision,
dramatically reducing bandwidth and allowing the H100 Tensor Cores to
drive the O(N³·H) reduction within the 1 ms target.

The kernel `fused_gate_mask_kernel` performs the element‑wise
multiplication of the projected tensors with their sigmoid gates and
the optional binary mask in a single pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: fuse gate (already after sigmoid) + mask
# --------------------------------------------------------------
@triton.jit
def fused_gate_mask_kernel(
    # input pointers
    left_proj_ptr, right_proj_ptr,      # [B,N,N,H] (FP16)
    left_gate_ptr, right_gate_ptr,      # [B,N,N,H] (FP16, already sigmoid)
    mask_ptr,                           # [B,N,N]   (FP16, 0/1)

    # output pointers
    left_out_ptr, right_out_ptr,        # [B,N,N,H] (FP16)

    # strides (in elements, not bytes)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,

    # problem sizes
    B, N, H,

    # compile‑time constants
    BLOCK_ROW: tl.constexpr,   # rows processed per program (flattened B·N·N)
    BLOCK_H: tl.constexpr,     # chunk of the hidden dimension
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_ROW
    # flattened index of rows handled by this program
    row_idx = row_start + tl.arange(0, BLOCK_ROW)
    row_mask = row_idx < B * N * N

    # decode (b, i, j) from the flattened row index
    b = row_idx // (N * N)
    rem = row_idx % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for each tensor (no hidden dim yet)
    lp_base = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
    rp_base = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
    lg_base = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    rg_base = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    m_base  = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # mask scalar per (b,i,j) → broadcast across the hidden axis
    mask_scalar = tl.load(mask_ptr + m_base,
                          mask=row_mask,
                          other=tl.float16(1.0))   # 1.0 as FP16

    # iterate over the hidden dimension in BLOCK_H sized chunks
    for k in range(0, H, BLOCK_H):
        h_off = k + tl.arange(0, BLOCK_H)               # [BLOCK_H]

        # load tensors; out‑of‑bounds rows are masked to zero
        lp = tl.load(left_proj_ptr + lp_base[:, None] + h_off * stride_lp_h,
                     mask=row_mask[:, None],
                     other=tl.float16(0.0))               # [BLOCK_ROW, BLOCK_H]
        rp = tl.load(right_proj_ptr + rp_base[:, None] + h_off * stride_rp_h,
                     mask=row_mask[:, None],
                     other=tl.float16(0.0))
        lg = tl.load(left_gate_ptr + lg_base[:, None] + h_off * stride_lg_h,
                     mask=row_mask[:, None],
                     other=tl.float16(0.0))
        rg = tl.load(right_gate_ptr + rg_base[:, None] + h_off * stride_rg_h,
                     mask=row_mask[:, None],
                     other=tl.float16(0.0))

        # broadcast mask across the hidden axis
        mask_vec = mask_scalar[:, None]

        # fused element‑wise operation
        left_res  = lp * lg * mask_vec
        right_res = rp * rg * mask_vec

        # store results back
        tl.store(left_out_ptr + lp_base[:, None] + h_off * stride_lp_h,
                 left_res,
                 mask=row_mask[:, None])
        tl.store(right_out_ptr + rp_base[:, None] + h_off * stride_rp_h,
                 right_res,
                 mask=row_mask[:, None])


# --------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack and basic constants
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    # work in half‑precision for everything that does most of the work
    hp = torch.float16

    # -----------------------------------------------------------------
    # Cast weights to the required precision
    # -----------------------------------------------------------------
    # keep FP32 copies for the final LN & linear (higher precision)
    norm_weight_f = weights["norm.weight"]
    norm_bias_f   = weights["norm.bias"]
    to_out_norm_weight_f = weights["to_out_norm.weight"]
    to_out_norm_bias_f   = weights["to_out_norm.bias"]
    to_out_weight_f      = weights["to_out.weight"]

    # half‑precision copies for the matmuls
    norm_weight_h = norm_weight_f.to(hp)
    norm_bias_h   = norm_bias_f.to(hp)

    left_proj_weight_h  = weights["left_proj.weight"].to(hp)
    right_proj_weight_h = weights["right_proj.weight"].to(hp)
    left_gate_weight_h  = weights["left_gate.weight"].to(hp)
    right_gate_weight_h = weights["right_gate.weight"].to(hp)
    out_gate_weight_h   = weights["out_gate.weight"].to(hp)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (FP16)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor.to(hp), (dim,),
        weight=norm_weight_h,
        bias=norm_bias_h,
    )   # -> [B,N,N,dim]  fp16

    # -----------------------------------------------------------------
    # 2. Fused linear‑+‑gate projections (2 matmuls instead of 4)
    # -----------------------------------------------------------------
    # Concatenate projection & gate weights so that a single linear call
    # yields both the raw projection and the sigmoid gate.
    left_concat_w = torch.cat([left_proj_weight_h, left_gate_weight_h], dim=0)   # (2H, dim)
    right_concat_w = torch.cat([right_proj_weight_h, right_gate_weight_h], dim=0)  # (2H, dim)

    # shape: [B,N,N,2H]  (fp16)
    left_cat = F.linear(x, left_concat_w)
    right_cat = F.linear(x, right_concat_w)

    # split back into projection & gate tensors
    left_proj, left_gate_logit = torch.split(left_cat, hidden_dim, dim=-1)
    right_proj, right_gate_logit = torch.split(right_cat, hidden_dim, dim=-1)

    # sigmoid the gate logits (still fp16)
    left_gate = torch.sigmoid(left_gate_logit)
    right_gate = torch.sigmoid(right_gate_logit)

    # -----------------------------------------------------------------
    # 3. Out‑gate (separate linear)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight_h))   # [B,N,N,H]  fp16

    # -----------------------------------------------------------------
    # 4. Mask handling (fp16)
    # -----------------------------------------------------------------
    if mask is None or nomask:
        mask_fp = torch.ones((bs, seq_len, seq_len), dtype=hp, device=device)
    else:
        mask_fp = mask.to(hp)

    # -----------------------------------------------------------------
    # 5. Fuse gate + mask with Triton
    # -----------------------------------------------------------------
    left = torch.empty_like(left_proj)    # fp16, final gated left
    right = torch.empty_like(right_proj)  # fp16, final gated right

    # strides in element counts
    lp_str = left_proj.stride()
    rp_str = right_proj.stride()
    lg_str = left_gate.stride()
    rg_str = right_gate.stride()
    mask_str = mask_fp.stride()

    total_rows = bs * seq_len * seq_len
    BLOCK_ROW = 256                     # rows per program (tunable)
    BLOCK_H   = 32                      # hidden‑dim chunk (must divide hidden_dim)

    grid = (triton.cdiv(total_rows, BLOCK_ROW),)

    fused_gate_mask_kernel[grid](
        left_proj_ptr   = left_proj,
        right_proj_ptr  = right_proj,
        left_gate_ptr   = left_gate,
        right_gate_ptr  = right_gate,
        mask_ptr        = mask_fp,
        left_out_ptr    = left,
        right_out_ptr   = right,
        # strides
        stride_lp_b=lp_str[0], stride_lp_i=lp_str[1], stride_lp_j=lp_str[2], stride_lp_h=lp_str[3],
        stride_rp_b=rp_str[0], stride_rp_i=rp_str[1], stride_rp_j=rp_str[2], stride_rp_h=rp_str[3],
        stride_lg_b=lg_str[0], stride_lg_i=lg_str[1], stride_lg_j=lg_str[2], stride_lg_h=lg_str[3],
        stride_rg_b=rg_str[0], stride_rg_i=rg_str[1], stride_rg_j=rg_str[2], stride_rg_h=rg_str[3],
        stride_mask_b=mask_str[0], stride_mask_i=mask_str[1], stride_mask_j=mask_str[2],
        # problem sizes
        B=bs, N=seq_len, H=hidden_dim,
        # compile‑time constants
        BLOCK_ROW=BLOCK_ROW, BLOCK_H=BLOCK_H,
    )

    # -----------------------------------------------------------------
    # 6. Batched GEMM across the hidden dimension (FP16)
    # -----------------------------------------------------------------
    # reshape to (B*H, N, N) and perform a batch matrix multiplication
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(bs * hidden_dim, seq_len, seq_len)   # (B·H, N, N)
    right_flat = right_perm.view(bs * hidden_dim, seq_len, seq_len) # (B·H, N, N)

    # out_h = left_h @ right_hᵀ   for every head h
    out_mat = torch.bmm(left_flat, right_flat.transpose(1, 2))     # (B·H, N, N)

    # restore the original layout [B, N, N, H] (still fp16)
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 7. Final LayerNorm, out‑gate, and projection (back to FP32)
    # -----------------------------------------------------------------
    out_fp32 = out.float()                       # cast to FP32 for higher‑precision LN
    out_gate_fp32 = out_gate.float()

    # output LayerNorm (FP32)
    out_norm = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight_f,
        bias=to_out_norm_bias_f,
    )

    # gating
    out_gated = out_norm * out_gate_fp32

    # final linear projection back to model dimension (FP32)
    final_out = F.linear(out_gated, to_out_weight_f)   # bias = False by construction

    return final_out
