#!/usr/bin/env python
"""
TriMul outgoing update (AlphaFold3 style) – fast forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Four linear projections (no bias) are computed with `torch.nn.functional.linear`:
   * left  = x_norm @ W_left_proj      → [B,N,N,H]
   * right = x_norm @ W_right_proj     → [B,N,N,H]
   * left_gate  = sigmoid(x_norm @ W_left_gate)   → [B,N,N,H]
   * right_gate = sigmoid(x_norm @ W_right_gate)  → [B,N,N,H]
   * out_gate   = sigmoid(x_norm @ W_out_gate)    → [B,N,N,H]
3. (Optional) apply the pair‑wise mask to `left`/`right`.
4. Fuse the element‑wise products `left*left_gate` and
   `right*right_gate` in a tiny Triton kernel (half‑precision).
5. Rearrange the tensors to (B·H, N, N) and perform a batched
   matrix‑multiply `C = A @ Aᵀ` with `torch.bmm`.  This uses cuBLAS/
   Tensor‑Cores and is far faster than a hand‑written Triton GEMM.
6. Reshape the result back to [B,N,N,H] (float32) and apply a second
   LayerNorm (`to_out_norm`).
7. Multiply by the output gate, then a final linear projection to the
   original feature dimension `D`.
8. Return the tensor of shape [B,N,N,D] (float32).

Only the element‑wise gating step is implemented in Triton; the heavy
N³ matrix multiplication is handled by the highly‑optimised cuBLAS batched
GEMM, giving sub‑millisecond runtimes on an H100 for the target
configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse left/right gating (half‑precision)
# ----------------------------------------------------------------------
@triton.jit
def gate_fuse_kernel(
    left_ptr,          # half*  (left projection)
    right_ptr,         # half*  (right projection)
    left_gate_ptr,     # half*  (left gate)
    right_gate_ptr,    # half*  (right gate)
    out_left_ptr,      # half*  (output left)
    out_right_ptr,     # half*  (output right)
    n_elem,            # total number of elements = B*N*N*H
    BLOCK_SIZE: tl.constexpr,
):
    """out_left = left * left_gate
    out_right = right * right_gate
    All tensors are flattened and contiguous."""
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offset < n_elem

    # load
    l = tl.load(left_ptr + offset, mask=mask)
    r = tl.load(right_ptr + offset, mask=mask)
    lg = tl.load(left_gate_ptr + offset, mask=mask)
    rg = tl.load(right_gate_ptr + offset, mask=mask)

    # store fused results
    tl.store(out_left_ptr + offset, l * lg, mask=mask)
    tl.store(out_right_ptr + offset, r * rg, mask=mask)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype torch.float32)
    """
    # ------------------------------------------------- unpack arguments
    x, mask, weights, config = data
    # x : [B, N, N, D]  (float32)
    # mask : [B, N, N] or None
    B, N, _, D = x.shape
    H = config["hidden_dim"]                # internal hidden dimension

    # ------------------------------------------------- LayerNorm on input
    x_norm = F.layer_norm(
        x,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, D]  (float32)

    # ------------------------------------------------- Linear projections (no bias)
    # All linear ops are cheap (they are a single cuBLAS GEMM per tensor)
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]  (float32)
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B,N,N,H]

    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))     # [B,N,N,H]

    # ------------------------------------------------- Optional mask
    if mask is not None:
        # mask broadcast over hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------- Cast everything to half for the heavy part
    left_h   = left.to(torch.float16).contiguous()
    right_h  = right.to(torch.float16).contiguous()
    left_g_h  = left_gate.to(torch.float16).contiguous()
    right_g_h = right_gate.to(torch.float16).contiguous()

    # ------------------------------------------------- Fuse gating with a tiny Triton kernel
    # allocate output buffers
    left_gated  = torch.empty_like(left_h)
    right_gated = torch.empty_like(right_h)

    n_elem = left_h.numel()                # B * N * N * H
    BLOCK_SIZE = 65536                     # 64 K elements per program
    grid = (triton.cdiv(n_elem, BLOCK_SIZE),)

    gate_fuse_kernel[grid](
        left_h,
        right_h,
        left_g_h,
        right_g_h,
        left_gated,
        right_gated,
        n_elem,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )
    # ------------------------------------------------- Prepare for batched matmul
    # reshape to (B*H, N, N)  – each page is an independent GEMM
    left_perm  = left_gated.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right_gated.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    BH = B * H
    left_mat  = left_perm.view(BH, N, N)   # [BH, N, N]   half
    right_mat = right_perm.view(BH, N, N)  # [BH, N, N]   half

    # ------------------------------------------------- Batched GEMM  C = A @ Aᵀ
    # Tensor‑Core GEMM (FP16→FP32 accumulation) is used internally.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [BH, N, N] half

    # ------------------------------------------------- Reshape back to [B,N,N,H] (float32)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,H]

    # ------------------------------------------------- LayerNorm over hidden dim
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # [B,N,N,H]

    # ------------------------------------------------- Output gate (keep FP32)
    out = out * out_gate

    # ------------------------------------------------- Final linear projection back to D
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,D]  (float32)

    return out
