"""
custom_kernel
==============

Implements the **outgoing** TriMul operation (AlphaFold‑3) with a fused
Triton GEMM.  The kernel computes for every batch `b`, hidden channel `h`
and pair of positions `(i, j)`:

    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

All heavy arithmetic (the N³ reduction) is performed in a single Triton
kernel that:

* works on fp16 inputs and accumulates in fp32 (Tensor‑Core friendly);
* tiles the multiplication with 128 × 128 output tiles and a 64‑wide
  reduction tile (multiple of 16, optimal for H100 tensor cores);
* pipelines the loads with ``num_stages=4`` and runs 8 warps per CTA,
  providing enough parallelism for the required workloads (N up to 1024,
  hidden ≤ 128, batch ≤ 2).

All other operations (LayerNorm, linear projections, sigmoid gates,
output gating) are cheap compared to the N³ core and stay in PyTorch.
The final result is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    a_ptr,                     # left  (BFH × N × N) – fp16
    b_ptr,                     # right (BFH × N × N) – fp16
    c_ptr,                     # output (BFH × N × N) – fp16
    batch_hidden, N,           # scalar dimensions
    stride_az, stride_ai, stride_ak,   # strides of A (page, row, col)
    stride_bz, stride_bk, stride_bn,   # strides of B (page, k, n)
    stride_cz, stride_ci, stride_cj,   # strides of C (page, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute a block of C = A @ Bᵀ

    A : (batch_hidden, N, N) row‑major
    B : (batch_hidden, N, N) row‑major – accessed as transposed
    C : (batch_hidden, N, N) row‑major
    """

    pid_z = tl.program_id(0)          # which (batch, hidden) page
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # ------------------------------------------------- tile coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this page
    a_page_ptr = a_ptr + pid_z * stride_az
    b_page_ptr = b_ptr + pid_z * stride_bz
    c_page_ptr = c_ptr + pid_z * stride_cz

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------- reduction over K
    cur_k = 0
    # pipeline: we pre‑load the first tile outside the loop
    while cur_k < N:
        cur_k_offs = cur_k + tl.arange(0, BLOCK_K)
        mask_k = cur_k_offs < N

        # ---- load A tile -------------------------------------------------
        a_ptrs = a_page_ptr \
                 + offs_m[:, None] * stride_ai \
                 + cur_k_offs[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- load B tile (as Bᵀ) ----------------------------------------
        # Bᵀ has shape (K, N) → stride_bk advances K, stride_bn advances N
        b_ptrs = b_page_ptr \
                 + cur_k_offs[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ---- multiply‑accumulate ----------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

        cur_k += BLOCK_K

    # ------------------------------------------------- write out
    c_ptrs = c_page_ptr \
             + offs_m[:, None] * stride_ci \
             + offs_n[None, :] * stride_cj
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------ unpack
    x, mask, weights, config = data
    # x : [B, N, N, dim]   (float32)
    # mask : [B, N, N]     (bool/float or None)
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------ LayerNorm on the input
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # -> [B, N, N, dim]

    # ------------------------------------------------------------------ Linear projections (no bias)
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B, N, N, hidden]

    # ------------------------------------------------------------------ Apply optional mask (broadcast on hidden dim)
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"])) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))     # [B,N,N,hidden]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ Cast to half for the heavy kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ------------------------------------------------------------------ Layout for the kernel:
    #   (B, hidden, N, N) → (B*hidden, N, N)
    left_h = left_h.permute(0, 3, 1, 2).contiguous()
    right_h = right_h.permute(0, 3, 1, 2).contiguous()

    BxH = B * hidden
    left_mat = left_h.view(BxH, N, N)   # shape (pages, N, N)
    right_mat = right_h.view(BxH, N, N)

    # ------------------------------------------------------------------ Output buffer (fp16)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=x.device)

    # ------------------------------------------------------------------ Strides in **elements**
    stride_az = left_mat.stride(0)   # page stride = N * N
    stride_ai = left_mat.stride(1)   # row stride = N
    stride_ak = left_mat.stride(2)   # col stride = 1

    stride_bz = right_mat.stride(0)  # same page stride
    stride_bk = right_mat.stride(2)  # stride over k (col) = 1
    stride_bn = right_mat.stride(1)  # stride over n (row) = N

    stride_cz = out_mat.stride(0)
    stride_ci = out_mat.stride(1)
    stride_cj = out_mat.stride(2)

    # ------------------------------------------------------------------ Tunable tile sizes
    # 128×128 output tiles, 64‑wide reduction (multiple of 16 → tensor cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64   # enlarge K‑tile to halve loop count (still fits registers)

    grid = (
        BxH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch kernel
    trimul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        BxH,
        N,
        stride_az, stride_ai, stride_ak,
        stride_bz, stride_bk, stride_bn,
        stride_cz, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 benefits from 8 warps per CTA
        num_stages=4,         # deep pipeline for overlapping loads / compute
    )

    # ------------------------------------------------------------------ Reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------ LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------ Output gate
    out = out * out_gate

    # ------------------------------------------------------------------ Final linear projection to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
