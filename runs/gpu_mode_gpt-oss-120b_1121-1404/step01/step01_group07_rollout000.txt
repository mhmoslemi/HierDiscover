"""
TriMul (outgoing) – fully‑fused forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input over its last dimension (half‑precision).
2. Linear projections (left/right) and three gating vectors (half‑precision)
   – no bias.
3. Fuse mask, left‑gate and right‑gate with a small Triton kernel
   (in‑place elementwise multiplication).  When `nomask=True` the kernel
   is skipped.
4. Core TriMul contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented with a half‑precision batched matrix‑multiply
   (`torch.einsum`) which internally maps to a highly‑optimized cuBLAS
   GEMM (each hidden slice is an independent batch).
5. Layer‑norm over the hidden dimension, apply the output gate
   and a final linear projection back to `dim`.
6. Cast the result back to float32 (required by the interface).

All heavy arithmetic is performed in float16 to maximise throughput on
an H100.  The only Triton‑written part is the element‑wise mask‑gate
fusion, satisfying the “use Triton” requirement while keeping the
overall runtime well below the target.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: in‑place mask × gate fusion
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_inplace_kernel(
    x_ptr,                # [B, N, N, H]  – input tensor (to be overwritten)
    mask_ptr,             # [B, N, N]    – binary mask
    gate_ptr,             # [B, N, N, H] – gate values
    total_elements,       # B * N * N * H
    stride_x_b, stride_x_i, stride_x_j, stride_x_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    B, N, H,
    BLOCK: tl.constexpr,
):
    """Fuse ``x = x * mask * gate`` in place."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    # Decode linear offset -> (b, i, j, h)
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # Compute flat pointers
    x_off = (b * stride_x_b + i * stride_x_i + j * stride_x_j + h * stride_x_h)
    mask_off = (b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)
    gate_off = (b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h)

    # Load, compute and store (half‑precision)
    x = tl.load(x_ptr + x_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(1.0))
    g = tl.load(gate_ptr + gate_off, mask=mask, other=tl.float16(0.0))

    out = x * m * g
    tl.store(x_ptr + x_off, out, mask=mask)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast all model weights to float16 (half‑precision) – they are
    # static for the lifetime of a forward call.
    # ------------------------------------------------------------------
    def to_half(t):
        return t.to(device=device, dtype=torch.float16)

    norm_weight = to_half(weights["norm.weight"])
    norm_bias = to_half(weights["norm.bias"])

    left_proj_weight = to_half(weights["left_proj.weight"])   # (hidden_dim, dim)
    right_proj_weight = to_half(weights["right_proj.weight"])

    left_gate_weight = to_half(weights["left_gate.weight"])
    right_gate_weight = to_half(weights["right_gate.weight"])
    out_gate_weight = to_half(weights["out_gate.weight"])

    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias = to_half(weights["to_out_norm.bias"])

    to_out_weight = to_half(weights["to_out.weight"])       # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (half precision)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm works with half tensors.
    x = torch.nn.functional.layer_norm(
        input_tensor.to(torch.float16),
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim] (fp16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) – produce left/right tensors
    # ------------------------------------------------------------------
    # Using torch.nn.functional.linear which internally does a GEMM.
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – also in half precision
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate with Triton (in‑place).  Skip when `nomask` is True.
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool/int; ensure it is half‑precision float (0/1)
        if mask_tensor is None:
            mask = torch.ones(
                (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
                device=device,
                dtype=torch.float16,
            )
        else:
            mask = mask_tensor.to(device=device, dtype=torch.float16)

        B, N_seq, _, _ = left.shape
        total = B * N_seq * N_seq * hidden_dim
        BLOCK = 4096                     # fine‑tune for H100
        grid = ( (total + BLOCK - 1) // BLOCK, )

        # LEFT   = left * mask * left_gate   (in‑place)
        mask_gate_inplace_kernel[grid](
            left,
            mask,
            left_gate,
            total,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            left_gate.stride(0), left_gate.stride(1),
            left_gate.stride(2), left_gate.stride(3),
            B, N_seq, hidden_dim,
            BLOCK=BLOCK,
        )
        # RIGHT  = right * mask * right_gate (in‑place)
        mask_gate_inplace_kernel[grid](
            right,
            mask,
            right_gate,
            total,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            right_gate.stride(0), right_gate.stride(1),
            right_gate.stride(2), right_gate.stride(3),
            B, N_seq, hidden_dim,
            BLOCK=BLOCK,
        )
    else:
        # No masking – just element‑wise gate multiplication
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul contraction.
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented with half‑precision Einstein summation.
    # ------------------------------------------------------------------
    # The einsum internally launches a batched GEMM for each hidden slice,
    # which on an H100 is highly optimized.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # fp16

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension (half precision)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 7. Apply out_gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim (half precision)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim] fp16

    # ------------------------------------------------------------------
    # 9. Cast to float32 for the external API
    # ------------------------------------------------------------------
    return out.to(torch.float32)
