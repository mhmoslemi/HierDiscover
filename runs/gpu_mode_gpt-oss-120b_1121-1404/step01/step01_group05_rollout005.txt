    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, HIDDEN: tl.constexpr

    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch_h = pid
    batch = batch_h // HIDDEN
    hid = batch_h - batch * HIDDEN

    # compute i, j base indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

"""
TriMul (outgoing) – fast Triton implementation
------------------------------------------------
The forward pass consists of three steps:

1️⃣  Layer‑norm + linear projections + sigmoid gates  
    (all done in FP16 to reduce memory traffic).

2️⃣  A *single* Triton kernel that fuses  
        • mask multiplication  
        • left‑gate & right‑gate application  
        • the 3‑tensor contraction  

    The contraction is
        out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h] ,
    i.e. a batched GEMM (A @ Bᵀ) for every hidden slice *h*.
    We flatten the batch and hidden dimensions (B·H) and compute a
    tiled mat‑mul (M×K)·(K×N) → (M×N) with FP32 accumulation.
    The kernel avoids materialising the intermediate tensors
    `left_fused` / `right_fused`, saving a large amount of memory
    bandwidth.

3️⃣  Second Layer‑norm, out‑gate, and the final linear projection
    back to the original feature dimension.

The kernel is written for Triton 3.3.1 and targets an H100 GPU.
A block size of 64×64×64 works well for all test sizes
(N∈{256,512,768,1024}) and hidden dimensions (≤ 768).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused mask·gate·contraction
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_gemm_kernel(
    # pointers
    left_ptr, left_gate_ptr,               # left tensor & its gate
    right_ptr, right_gate_ptr,             # right tensor & its gate
    mask_ptr, out_ptr,                     # mask and output
    # problem size
    B, N, H,                               # batch, seq_len, hidden_dim
    # strides for left (and right) 4‑D layout [B, N, N, H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for left_gate (identical layout)
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    # strides for right (and its gate)
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_right_gate_b, stride_right_gate_i, stride_right_gate_k, stride_right_gate_h,
    # strides for mask [B, N, N]
    stride_mask_b, stride_mask_i, stride_mask_k,
    # strides for output (same layout as left/right)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN: tl.constexpr,                         # compile‑time hidden dim
):
    # ------------------------------------------------------------------
    # 1️⃣  Grid layout – we flatten (batch, hidden) -> pid0
    # ------------------------------------------------------------------
    pid = tl.program_id(0)           # = batch * hidden + hidden_idx
    pid_i = tl.program_id(1)         # output row‑tile index
    pid_j = tl.program_id(2)         # output col‑tile index

    batch_h = pid
    batch = batch_h // HIDDEN
    hid   = batch_h - batch * HIDDEN               # hidden index for this program

    # ------------------------------------------------------------------
    # 2️⃣  Compute tile coordinates
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # rows i … i+BLOCK_M‑1
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # cols j … j+BLOCK_N‑1

    mask_i = i < N
    mask_j = j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # ------------------------------------------------------------------
    # 3️⃣  Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4️⃣  Loop over the reduction dimension k
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # --------------------------------------------------------------
        # Load left tile   [i, k, hid]
        # --------------------------------------------------------------
        a_ptrs = left_ptr \
            + batch * stride_left_b \
            + i[:, None] * stride_left_i \
            + k[None, :] * stride_left_k \
            + hid * stride_left_h
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # left gate tile
        a_gate_ptrs = left_gate_ptr \
            + batch * stride_gate_b \
            + i[:, None] * stride_gate_i \
            + k[None, :] * stride_gate_k \
            + hid * stride_gate_h
        a_gate = tl.load(a_gate_ptrs,
                         mask=mask_i[:, None] & mask_k[None, :],
                         other=0.0)

        # mask tile for the left side (mask[b,i,k])
        mask_a_ptrs = mask_ptr \
            + batch * stride_mask_b \
            + i[:, None] * stride_mask_i \
            + k[None, :] * stride_mask_k
        mask_a = tl.load(mask_a_ptrs,
                         mask=mask_i[:, None] & mask_k[None, :],
                         other=0.0)

        a_fused = a * a_gate * mask_a                     # [M, K]

        # --------------------------------------------------------------
        # Load right tile  [j, k, hid]  → we need (K, N) layout for dot
        # --------------------------------------------------------------
        b_ptrs = right_ptr \
            + batch * stride_right_b \
            + k[:, None] * stride_right_k \
            + j[None, :] * stride_right_i \
            + hid * stride_right_h
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # right gate tile
        b_gate_ptrs = right_gate_ptr \
            + batch * stride_right_gate_b \
            + k[:, None] * stride_right_gate_k \
            + j[None, :] * stride_right_gate_i \
            + hid * stride_right_gate_h
        b_gate = tl.load(b_gate_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        # mask tile for the right side (mask[b,j,k])
        mask_b_ptrs = mask_ptr \
            + batch * stride_mask_b \
            + j[None, :] * stride_mask_i \
            + k[:, None] * stride_mask_k
        mask_b = tl.load(mask_b_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)

        b_fused = b * b_gate * mask_b                     # [K, N]

        # --------------------------------------------------------------
        # Matrix multiply & accumulate
        # --------------------------------------------------------------
        acc += tl.dot(a_fused, b_fused)                   # (M,N) ← (M,K)·(K,N)

    # ------------------------------------------------------------------
    # 5️⃣  Write the result back (convert to FP16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr \
        + batch * stride_out_b \
        + i[:, None] * stride_out_i \
        + j[None, :] * stride_out_j \
        + hid * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 2️⃣  Prepare mask (fp16) – if nomask we use a tensor of ones
    # --------------------------------------------------------------
    if nomask or mask_tensor is None:
        mask = torch.ones((B, N, N), device=device, dtype=dtype_fp16)
    else:
        mask = mask_tensor.to(dtype_fp16)

    # --------------------------------------------------------------
    # 3️⃣  Allocate output tensor (fp16)
    # --------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=dtype_fp16)

    # --------------------------------------------------------------
    # 4️⃣  Compute strides (in elements, as expected by Triton)
    # --------------------------------------------------------------
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h = left_gate.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_right_gate_b, stride_right_gate_i, stride_right_gate_k, stride_right_gate_h = right_gate.stride()
    stride_mask_b, stride_mask_i, stride_mask_k = mask.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # --------------------------------------------------------------
    # 5️⃣  Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # first axis = batch * hidden  (flattened)
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _tri_mul_gemm_kernel[grid](
        left, left_gate,
        right, right_gate,
        mask, out,
        B, N, hidden_dim,
        # strides for left
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        # strides for left_gate (identical layout)
        stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
        # strides for right
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        # strides for right_gate
        stride_right_gate_b, stride_right_gate_i, stride_right_gate_k, stride_right_gate_h,
        # mask strides
        stride_mask_b, stride_mask_i, stride_mask_k,
        # output strides
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, HIDDEN=hidden_dim
    )

    # --------------------------------------------------------------
    # 6️⃣  Second LayerNorm, out‑gate, final projection (all in fp16)
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate                                              # element‑wise gate

    # final linear back to `dim`
    to_out_w = weights["to_out.weight"].to(dtype_fp16)
    out = F.linear(out, to_out_w).to(torch.float32)                    # back to fp32

    return out
