import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul “outgoing” kernel (AlphaFold3).

The implementation follows the reference PyTorch module but replaces the costly
pair‑wise reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

with a fused Triton kernel.  After LayerNorm, linear projections and gating the
tensor is reshaped to [B*H, N, N] (H = hidden_dim).  For each batch element a
matrix‑multiply‑accumulate computes `left @ right.T` using Tensor‑Core friendly
tiles (BLOCK_M×BLOCK_N) while iterating over the K‑dimension in blocks of
BLOCK_K.  The kernel works in FP16 for the loads (to cut bandwidth) and accumulates
in FP32 for numeric stability.  The result is reshaped back, normalized,
gated and linearly projected to the original dimension.

The kernel is deliberately simple, uses three program‑id axes (M‑tile, N‑tile,
batch) and respects arbitrary stride layouts, enabling it to be used with the
contiguous tensors produced in the surrounding PyTorch code.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul where the second operand is transposed.
#   out[p,i,j] = Σ_k left[p,i,k] * right[p,j,k]
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_transpose_kernel(
    a_ptr,                      # const *[fp16]   left  matrix   (P, M, K)
    b_ptr,                      # const *[fp16]   right matrix   (P, N, K)
    c_ptr,                      #       *[fp32]   output matrix (P, M, N)
    batch, M, N, K,             # int32 shape parameters
    stride_apb, stride_amn, stride_ak,   # int64 strides for a
    stride_bpb, stride_bmn, stride_bk,   # int64 strides for b
    stride_cpb, stride_cmn, stride_cnk,   # int64 strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)         # batch*hidden index
    pid_m = tl.program_id(0)         # block row index
    pid_n = tl.program_id(1)         # block column index

    # --------------------------------------------------------------
    # offsets within the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # guards for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # a: (BLOCK_M, BLOCK_K)  – row‑major layout
        a_ptrs = (a_ptr
                  + pid_b * stride_apb
                  + offs_m[:, None] * stride_amn
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # b: (BLOCK_K, BLOCK_N) – note the transposed view
        b_ptrs = (b_ptr
                  + pid_b * stride_bpb
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bmn)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # dot produces FP32 accumulation automatically (Tensor Cores)
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result
    # --------------------------------------------------------------
    c_ptrs = (c_ptr
              + pid_b * stride_cpb
              + offs_m[:, None] * stride_cmn
              + offs_n[None, :] * stride_cnk)
    tl.store(c_ptrs, acc,
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Entrypoint used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be None or ignored when config["nomask"] is True)
            - weights: dict of model parameters (see code for required keys)
            - config: dict containing at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device as input_tensor)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1.) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2.) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3.) Optional mask (broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_broadcast = mask[..., None].to(left.dtype)  # [B, N, N, 1]
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4.) Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5.) Core TriMul reduction: Σ_k left[...,i,k,d] * right[...,j,k,d]
    #    Implemented as a batched matmul with the second operand transposed.
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] and fuse the batch & hidden dimensions
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H  # total “batch” dimension for the Triton kernel

    # flatten batch*hidden -> (P, N, N)
    left_flat = left.view(P, N, N)
    right_flat = right.view(P, N, N)

    # cast to FP16 for the kernel (Tensor Cores) – accumulation stays FP32
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    out_fp32 = torch.empty((P, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows per block
    BLOCK_N = 64   # cols per block
    BLOCK_K = 32   # reduction step

    grid = (
        triton.cdiv(N, BLOCK_M),   # number of blocks along M
        triton.cdiv(N, BLOCK_N),   # number of blocks along N
        P,                         # batched dimension (batch * hidden)
    )

    _batched_matmul_transpose_kernel[grid](
        # pointers
        left_fp16,
        right_fp16,
        out_fp32,
        # shape parameters
        P, N, N, N,
        # strides (in elements)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2),
        # compile‑time tile sizes
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6.) Final normalization + gating + linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                      # broadcasted gating
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
