11. final linear to_out (hidden_dim->dim) bias False.

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask and mask is not None:
        # mask shape [B,N,N] broadcast to hidden dim
        m = mask.unsqueeze(-1).to(left.dtype)
        left = left * m
        right = right * m

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # reshape for batched matmul
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H

    left_flat = left.view(P, N, N)
    right_flat = right.view(P, N, N)

    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    out_fp32 = torch.empty((P, N, N), dtype=torch.float32, device=device)

    # kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), P)

    _batched_gemm_transpose_fp16_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        P, N, N, N,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Final norm, gating, projection
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul “outgoing” implementation (AlphaFold3).

The forward pass consists of:
1️⃣ Layer‑norm over the last channel dimension.
2️⃣ Two bias‑free linear projections (left / right) and three gating
   linear layers (left‑gate, right‑gate, out‑gate) with sigmoid.
3️⃣ Optional element‑wise mask applied to the projected tensors.
4️⃣ Element‑wise gating of the projected tensors.
5️⃣ Core triangular reduction: for every hidden channel we need
   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d].
   This is a batched matrix‑multiply (left @ rightᵀ) where each hidden
   channel is an independent GEMM of shape (N,N).  The heavy computation
   is performed by a custom Triton kernel that uses Tensor‑Core‐friendly
   tiles (BLOCK_M×BLOCK_N) and accumulates in FP32 while loading in FP16.
6️⃣ Layer‑norm + out‑gate + final linear projection back to the original
   channel dimension.

The Triton kernel is deliberately small but highly tuned:
- TILE_M = TILE_N = 128, TILE_K = 64 give few program‑ids per batch
  (large compute per launch → less launch overhead).
- All loads are contiguous; masks for out‑of‑bounds tiles are applied.
- Accumulation is in FP32 for numerical stability; the final result is
  stored as FP32 because the following LayerNorm expects float32.

The surrounding PyTorch code handles LayerNorms, linear layers, and
sigmoids – operations that are already efficiently vectorised.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Batched GEMM where the second operand is implicitly transposed.
#   out[p, i, j] = Σₖ left[p, i, k] * right[p, j, k]
#   left  : (P, M, K)  – FP16
#   right : (P, N, K)  – FP16 (treated as transposed view (K, N))
#   out   : (P, M, N)  – FP32
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_transpose_fp16_kernel(
    a_ptr,                           # const *[fp16] left  matrix   (P, M, K)
    b_ptr,                           # const *[fp16] right matrix   (P, N, K)
    c_ptr,                           #       *[fp32] output matrix (P, M, N)
    batch, M, N, K,                  # shape parameters (int32)
    stride_apb, stride_amn, stride_ak,   # strides for a
    stride_bpb, stride_bk, stride_bn,    # strides for b (treated as transposed)
    stride_cpb, stride_cmn, stride_cnk,  # strides for c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch × hidden index
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column

    # ------------------------------------------------------------------
    # Offsets inside this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension in BLOCK_K steps
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- load a : (BLOCK_M, BLOCK_K) -----
        a_ptrs = (a_ptr
                  + pid_b * stride_apb
                  + offs_m[:, None] * stride_amn
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0)

        # ----- load b (transposed view) : (BLOCK_K, BLOCK_N) -----
        b_ptrs = (b_ptr
                  + pid_b * stride_bpb
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0)

        # ----- dot product (Tensor‑Core) -----
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (c_ptr
              + pid_b * stride_cpb
              + offs_m[:, None] * stride_cmn
              + offs_n[None, :] * stride_cnk)
    tl.store(c_ptrs, acc,
             mask=(mask_m[:, None] & mask_n[None, :]))


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (optional, may be None)
            - weights: dict with the model parameters (see code for keys)
            - config: dict containing at least:
                * "dim" (int) – input channel dimension
                * "hidden_dim" (int) – hidden dimension used inside TriMul
                * "nomask" (bool, optional) – if True mask is ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32) on the same device
        as ``input_tensor``.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is [B, N, N]; expand to [..., 1] to broadcast over hidden_dim
        mask_broadcast = mask.unsqueeze(-1).to(left.dtype)
        left  = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul reduction (batched GEMM with transposed second operand)
    # ------------------------------------------------------------------
    #   left/right : [B, N, N, hidden_dim]  →  [B, hidden_dim, N, N]
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H                                 # fused batch × hidden dimension

    # flatten (B, H) → P
    left_flat  = left.view(P, N, N)
    right_flat = right.view(P, N, N)

    # convert to fp16 to exploit Tensor Cores
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # output buffer (FP32 accumulator)
    out_fp32 = torch.empty((P, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton launch configuration (tuned for H100)
    # ------------------------------------------------------------------
    BLOCK_M = 128   # rows per block
    BLOCK_N = 128   # cols per block
    BLOCK_K = 64    # reduction step

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles along M
        triton.cdiv(N, BLOCK_N),   # tiles along N
        P,                         # fused batch × hidden dimension
    )

    _batched_gemm_transpose_fp16_kernel[grid](
        # pointers
        left_fp16,
        right_fp16,
        out_fp32,
        # shape parameters
        P, N, N, N,
        # strides (in elements, not bytes)
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2),
        # compile‑time tile sizes
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Final normalisation, out‑gate, linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # broadcasted gating
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
