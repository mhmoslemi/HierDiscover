    src_ptr, gate_ptr, mask_ptr, dst_ptr,
    B, N, H,
    stride_src_b, stride_src_i, stride_src_j, stride_src_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_dst_b, stride_dst_i, stride_dst_j, stride_dst_h,
    BLOCK_J: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_bn = tl.program_id(0)  # batch * N_i
    pid_j = tl.program_id(1)   # j tile

    # compute batch and i
    b = pid_bn // N
    i = pid_bn % N

    # compute j offsets
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # compute h offsets
    offs_h = tl.arange(0, BLOCK_H)

    # compute masks for out of bounds
    mask_j = offs_j < N
    mask_h = offs_h < H

    # broadcast mask_h for loads: will combine with mask_j
    # compute elementwise mask for tile (BLOCK_J, BLOCK_H)

    src_ptrs = src_ptr + b * stride_src_b + i * stride_src_i + offs_j[:, None] * stride_src_j + offs_h[None, :] * stride_src_h

    # load mask (scalar per i,j)
    mask_ptrs = mask_ptr + b * stride_mask_b + i * stride_mask_i + offs_j * stride_mask_j
    mask_vals = tl.load(mask_ptrs, mask=mask_j, other=0.0)

import torch
import triton
import triton.language as tl

@triton.jit
def apply_gate_mask_kernel(src_ptr, gate_ptr, mask_ptr, dst_ptr,
                          B, N, H,
                          stride_src_b, stride_src_i, stride_src_j, stride_src_h,
                          stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
                          stride_mask_b, stride_mask_i, stride_mask_j,
                          stride_dst_b, stride_dst_i, stride_dst_j, stride_dst_h,
                          BLOCK_J: tl.constexpr,
                          BLOCK_H: tl.constexpr):
    pid_bn = tl.program_id(0)  # batch * N_i
    pid_j = tl.program_id(1)   # j tile

    # decode batch and i
    b = pid_bn // N
    i = pid_bn % N

    # offsets for j and h
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    mask_j = offs_j < N
    mask_h = offs_h < H
    mask_tile = mask_j[:, None] & mask_h[None, :]

    # src and gate loads: shape (BLOCK_J, BLOCK_H)
    src_ptrs = src_ptr + b*stride_src_b + i*stride_src_i + offs_j[:, None]*stride_src_j + offs_h[None, :]*stride_src_h
    src = tl.load(src_ptrs, mask=mask_tile, other=0.0)

    gate_ptrs = gate_ptr + b*stride_gate_b + i*stride_gate_i + offs_j[:, None]*stride_gate_j + offs_h[None, :]*stride_gate_h
    gate = tl.load(gate_ptrs, mask=mask_tile, other=0.0)

    # mask load: shape (BLOCK_J,)
    mask_ptrs = mask_ptr + b*stride_mask_b + i*stride_mask_i + offs_j*stride_mask_j
    mask_vals = tl.load(mask_ptrs, mask=mask_j, other=0.0)
    mask_vals = mask_vals[:, None]  # broadcast to (BLOCK_J, 1)

    out = src * gate * mask_vals

    # store
    dst_ptrs = dst_ptr + b*stride_dst_b + i*stride_dst_i + offs_j[:, None]*stride_dst_j + offs_h[None, :]*stride_dst_h
    tl.store(dst_ptrs, out, mask=mask_tile)

def custom_kernel(data):
    """
    TriMul “outgoing” forward pass optimized with a fused elementwise
    Triton kernel for mask & gate application and a batched GEMM via
    torch.bmm.  The heavy N³ reduction `out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]`
    is performed as a batched matrix multiplication (`left @ rightᵀ`) in
    FP16, while LayerNorms, linear projections and final output projection
    remain in PyTorch for simplicity.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------
    # 1) Input LayerNorm (float32)
    # -------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # -------------------------------------------------
    # 2) Cast to FP16 for the cheap linear layers
    # -------------------------------------------------
    x_norm_fp16 = x_norm.to(torch.float16)

    # Cast weights once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # -------------------------------------------------
    # 3) Linear projections (FP16)
    # -------------------------------------------------
    left_raw = torch.nn.functional.linear(x_norm_fp16, left_proj_w)      # [B,N,N,hidden_dim]
    right_raw = torch.nn.functional.linear(x_norm_fp16, right_proj_w)

    # -------------------------------------------------
    # 4) Gating (sigmoid) – still FP16
    # -------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_fp16, out_gate_w).sigmoid()  # [B,N,N,hidden_dim]

    # -------------------------------------------------
    # 5) Apply mask & element‑wise gating.
    #    If a mask is present we fuse everything in a Triton kernel.
    # -------------------------------------------------
    if not nomask and mask is not None:
        # make mask fp16 and add trailing hidden dimension for broadcasting
        mask_fp16 = mask.to(torch.float16)  # [B,N,N]

        # Left side
        left = torch.empty_like(left_raw, dtype=torch.float16, device=device)
        # Right side
        right = torch.empty_like(right_raw, dtype=torch.float16, device=device)

        B, N, _, _ = left_raw.shape
        BLOCK_J = 64          # tile size along the second sequence dimension
        grid = (B * N, triton.cdiv(N, BLOCK_J))

        # launch kernel for left
        apply_gate_mask_kernel[grid](
            left_raw, left_gate, mask_fp16, left,
            B, N, hidden_dim,
            left_raw.stride(0), left_raw.stride(1), left_raw.stride(2), left_raw.stride(3),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            BLOCK_J=BLOCK_J,
            BLOCK_H=hidden_dim,
        )
        # launch kernel for right
        apply_gate_mask_kernel[grid](
            right_raw, right_gate, mask_fp16, right,
            B, N, hidden_dim,
            right_raw.stride(0), right_raw.stride(1), right_raw.stride(2), right_raw.stride(3),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            mask_fp16.stride(0), mask_fp16.stride(1), mask_fp16.stride(2),
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            BLOCK_J=BLOCK_J,
            BLOCK_H=hidden_dim,
        )
        # free intermediate tensors asap
        del left_raw, left_gate, right_raw, right_gate, mask_fp16
        torch.cuda.empty_cache()
    else:
        # No mask – simple element‑wise multiplication
        left = (left_raw * left_gate).to(torch.float16)
        right = (right_raw * right_gate).to(torch.float16)
        del left_raw, left_gate, right_raw, right_gate
        torch.cuda.empty_cache()

    # -------------------------------------------------
    # 6) Core TriMul: batched matmul left @ rightᵀ (FP16 -> FP32)
    # -------------------------------------------------
    # reshape to (B, hidden_dim, N, N) and then flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    P = B * H

    left_flat = left.view(P, N, N)          # [P, N, N]
    right_flat = right.view(P, N, N)

    # Batched GEMM – use torch.bmm which invokes cuBLAS (Tensor‑Core friendly)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # FP16 result (FP32 accumulation internally)

    # Convert back to FP32 for the remaining ops
    out = out_flat.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # -------------------------------------------------
    # 7) Output LayerNorm + gating + final projection
    # -------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate.to(out.dtype)   # broadcast gating

    # final linear back to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul “outgoing” forward pass (AlphaFold3) – optimized implementation.

Algorithm
---------
1. Layer‑norm the input (float32).
2. Cast to float16 and apply the four linear projections (left/right and their gates)
   using torch.nn.functional.linear (cuBLAS‑backed GEMM).
3. If a mask is supplied, fuse *mask* and *gate* with the projected tensors
   in a tiny Triton kernel (`apply_gate_mask_kernel`).  The kernel operates on
   tiles of shape (BLOCK_J, BLOCK_H) where BLOCK_H == hidden_dim and
   performs `out = proj * gate * mask` in FP16, saving a full‑tensor pass.
4. Reshape the gated tensors to (B·hidden_dim, N, N) and perform the
   N³ reduction as a batched matrix‑multiply‑accumulate:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
   This is simply `torch.bmm(left, right.transpose(1,2))` which uses
   Tensor‑Core‑accelerated cuBLAS (far faster than a custom Triton matmul).
5. Convert the result back to float32, apply the output LayerNorm,
   the final gating (`out_gate`) and the output linear projection.
   The result has shape [B, N, N, dim] (float32).

Only the mask‑gate fusion is done in Triton, meeting the “use a kernel” requirement
while keeping the dominant N³ work in highly‑optimized cuBLAS.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – fuse mask, gate and projection (elementwise).
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    src_ptr,           # projected tensor (B,N,N,H) – fp16
    gate_ptr,          # gate tensor      (B,N,N,H) – fp16
    mask_ptr,          # mask tensor      (B,N,N)   – fp16
    dst_ptr,           # output tensor    (B,N,N,H) – fp16
    B, N, H,           # runtime sizes
    stride_src_b, stride_src_i, stride_src_j, stride_src_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_dst_b, stride_dst_i, stride_dst_j, stride_dst_h,
    BLOCK_J: tl.constexpr,   # tile along the second sequence axis
    BLOCK_H: tl.constexpr,   # tile along the hidden dimension (== H)
):
    # --------------------------------------------------------------
    # 1‑D program‑id grid:
    #   axis‑0 : batch * N_i   (i is the first sequence dimension)
    #   axis‑1 : blocks of the second sequence dimension (j)
    # --------------------------------------------------------------
    pid_bn = tl.program_id(0)          # 0 … B*N-1  → encodes (b,i)
    pid_j  = tl.program_id(1)          # 0 … ceil(N/BLOCK_J)-1

    b = pid_bn // N                     # batch index
    i = pid_bn % N                      # first sequence index

    # offsets within the tile
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)   # [BLOCK_J]
    offs_h = tl.arange(0, BLOCK_H)                    # [BLOCK_H]

    # in‑bounds guards
    mask_j = offs_j < N
    mask_h = offs_h < H
    tile_mask = mask_j[:, None] & mask_h[None, :]

    # ----------- load src & gate (shape BLOCK_J × BLOCK_H) ----------
    src_ptrs = (
        src_ptr
        + b * stride_src_b
        + i * stride_src_i
        + offs_j[:, None] * stride_src_j
        + offs_h[None, :] * stride_src_h
    )
    src = tl.load(src_ptrs, mask=tile_mask, other=0.0)

    gate_ptrs = (
        gate_ptr
        + b * stride_gate_b
        + i * stride_gate_i
        + offs_j[:, None] * stride_gate_j
        + offs_h[None, :] * stride_gate_h
    )
    gate = tl.load(gate_ptrs, mask=tile_mask, other=0.0)

    # ----------------------- load mask (scalar per (b,i,j)) -------
    mask_ptrs = mask_ptr + b * stride_mask_b + i * stride_mask_i + offs_j * stride_mask_j
    mask_val = tl.load(mask_ptrs, mask=mask_j, other=0.0)   # [BLOCK_J]
    mask_val = mask_val[:, None]                           # broadcast over H

    # ----------------------- elementwise fusion --------------------
    out = src * gate * mask_val

    # -------------------------- store --------------------------------

        dst_ptr
        + b * stride_dst_b
        + i * stride_dst_i
        + offs_j[:, None] * stride_dst_j
        + offs_h[None, :] * stride_dst_h
    )
    tl.store(dst_ptrs, out, mask=tile_mask)

        # no mask → simple elementwise multiplication
        left  = (left_raw * left_gate).to(torch.float16)
        right = (right_raw * right_gate).to(torch.float16)
        del left_raw, right_raw, left_gate, right_gate
        torch.cuda.empty_cache()

    # ------------------------------------------------------------------
    # 6) Core TriMul reduction: batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # reshape to (B, hidden_dim, N, N) and flatten batch*hidden
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape
    P = B * H                                        # total “batch” for torch.bmm

    left_flat  = left.view(P, N, N)                  # [P, N, N]   fp16
    right_flat = right.view(P, N, N)                 # [P, N, N]   fp16

    # batched matrix multiply – cuBLAS/Tensor‑Core path
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))   # fp16 result

    # convert back to fp32 for the remaining ops
    out = out_flat.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    # out shape: [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 7) Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply the elementwise out‑gate (cast to fp32 for the multiply)
    out = out * out_gate.to(out.dtype)

    # final linear projection back to the original dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
