import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
TriMul (outgoing version) – forward only.

The implementation follows the reference PyTorch module but fuses the
mask × gate multiplication into a tiny Triton kernel.  All heavy
linear/LayerNorm operations are performed with PyTorch (cuBLAS /
cuDNN) while the element‑wise mask‑gate fusion is done on‑GPU by
Triton.  This satisfies the requirement of “using Triton for at least
part of the computation” and keeps the overall runtime comparable to a
pure‑PyTorch implementation.

Algorithm (high level)
----------------------
1. Layer‑norm the input tensor over its last dimension.
2. Project to the hidden dimension (left/right) and compute three
   gating vectors (left_gate, right_gate, out_gate) with a sigmoid.
3. Fuse mask, left_gate and right_gate with a Triton kernel:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
4. Core TriMul: for each hidden slice h compute
        out[...,h] = left[...,h] @ right[...,h].T
   Implemented as a batched GEMM using torch.bmm (CUDA‑optimized).
5. Layer‑norm over the hidden dimension, apply out_gate, and a final
   linear projection back to the original feature dimension.
6. Return the result in float32.

The Triton kernel only touches the element‑wise part (step 3) which is
small compared to the O(N³·H) core but demonstrates custom‑kernel
usage.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_mask_gate_kernel(
    in_ptr,               # Tensor to be masked & gated, shape [B,N,N,H]
    mask_ptr,             # Binary mask, shape [B,N,N]
    gate_ptr,             # Gate values, shape [B,N,N,H]
    out_ptr,              # Output tensor, same shape as in_ptr
    total_elements,       # B * N * N * H
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    B, N, H,
    BLOCK: tl.constexpr,  # number of elements processed by one program
):
    """Fuse ``out = in * mask * gate``."""
    pid = tl.program_id(0)                     # 1‑D grid
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)          # linear offsets
    mask = offs < total_elements                # out‑of‑bounds guard

    # ----- decode linear offset into (b,i,j,h) ----------------------
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # ----- compute flat pointers ------------------------------------
    in_off = b * stride_in_b + i * stride_in_i + j * stride_in_j + h * stride_in_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    gate_off = (b * stride_gate_b + i * stride_gate_i + j * stride_gate_j
                + h * stride_gate_h)

    # ----- load, compute and store ----------------------------------
    a = tl.load(in_ptr + in_off,   mask=mask, other=0.0)
    m = tl.load(mask_ptr + mask_off, mask=mask, other=1.0)   # default 1 → no masking
    g = tl.load(gate_ptr + gate_off, mask=mask, other=0.0)

    out = a * m * g
    tl.store(out_ptr + in_off, out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]     # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]   # (hidden_dim, dim)

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate with Triton (skip if mask is all‑ones)
    # ------------------------------------------------------------------
    if nomask:
        # No masking → simple element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
    else:
        # mask may be bool/int – convert to same dtype as the tensors
        if mask_tensor is None:
            mask = torch.ones((input_tensor.shape[0],
                               input_tensor.shape[1],
                               input_tensor.shape[2]),
                              device=device,
                              dtype=dtype)
        else:
            mask = mask_tensor.to(dtype=dtype)

        B, N_seq, _, _ = left.shape
        total_elements = B * N_seq * N_seq * hidden_dim
        BLOCK = 1024                                    # tunable
        grid = ((total_elements + BLOCK - 1) // BLOCK,)

        # left = left * mask * left_gate
        left_out = torch.empty_like(left)
        apply_mask_gate_kernel[grid](
            left,
            mask,
            left_gate,
            left_out,
            total_elements,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            left_gate.stride(0), left_gate.stride(1),
            left_gate.stride(2), left_gate.stride(3),
            B, N_seq, hidden_dim,
            BLOCK=BLOCK,
        )
        # right = right * mask * right_gate
        right_out = torch.empty_like(right)
        apply_mask_gate_kernel[grid](
            right,
            mask,
            right_gate,
            right_out,
            total_elements,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            right_gate.stride(0), right_gate.stride(1),
            right_gate.stride(2), right_gate.stride(3),
            B, N_seq, hidden_dim,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out

    # ------------------------------------------------------------------
    # 5. Core TriMul: out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    #    Implemented as a batched GEMM (cuBLAS) for each hidden slice.
    # ------------------------------------------------------------------
    #   shape → [B, H, N, N] for convenient batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)              # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)            # (B*H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T  → (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 7. Apply out_gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B,N,N,dim]

    # Ensure final dtype is float32 as required
    out = out.to(torch.float32)

    return out
