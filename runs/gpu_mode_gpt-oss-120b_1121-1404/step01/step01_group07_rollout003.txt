@triton.jit
def apply_mask_gate_kernel_half(
    in_ptr, mask_ptr, gate_ptr, out_ptr,
    total_elements,
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    B, N, H,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elements

    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    in_off = b * stride_in_b + i * stride_in_i + j * stride_in_j + h * stride_in_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h

    a = tl.load(in_ptr + in_off, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=mask, other=tl.float16(1.0))
    g = tl.load(gate_ptr + gate_off, mask=mask, other=tl.float16(0.0))

    out = a * m * g
    tl.store(out_ptr + in_off, out, mask=mask)

        # right result
        right_out = torch.empty_like(right)
        apply_mask_gate_kernel_half[grid](
            right,
            mask,
            right_gate,
            right_out,
            total_elements,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            B, N_seq, H,
            BLOCK=BLOCK,
        )
        left = left_out
        right = right_out
    else:
        left = left * left_gate
        right = right * right_gate

    # 5. Core TriMul using batched half matmul
    # left: (B,N,N,H) -> (B*H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # compute batched matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # 6. Output LayerNorm over hidden_dim (half)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight_f16,
        bias=to_out_norm_bias_f16,
    )

    # 7. Apply out_gate (still half)
    out = out * out_gate

    # 8. Final linear projection back to dim (half)
    out = torch.nn.functional.linear(out, to_out_weight_f16)  # output half

    # Cast to float32 as required
    out = out.to(torch.float32)

    return out

"""
TriMul (outgoing) – high‑performance forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 – good numeric stability).
2. Cast the normalized tensor to float16 and run all linear projections /
   sigmoid gates in half‑precision (TensorCore friendly).
3. Fuse mask × gate × tensor (element‑wise) with a tiny Triton kernel
   (works on float16).
4. Core TriMul:  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision GEMM (B·H independent
   mat‑muls of size N×N) using torch.bmm which invokes cuBLAS Tensor
   Cores.
5. Layer‑norm over the hidden dimension in half‑precision, apply the
   output gate, and a final linear projection back to the original
   feature dimension (also in half‑precision).
6. Cast the result back to float32 (as required by the API).

All heavy arithmetic is done in float16, dramatically reducing memory
traffic and enabling Tensor Core acceleration while still delivering
the exact same functional result (within FP16 rounding error, which is
acceptable for inference‑only TriMul).
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel – half precision fusion of mask, gate and tensor
# ----------------------------------------------------------------------
@triton.jit
def apply_mask_gate_fp16_kernel(
    in_ptr,          # [B, N, N, H]   (float16)
    mask_ptr,        # [B, N, N]      (float16, 0/1)
    gate_ptr,        # [B, N, N, H]   (float16)
    out_ptr,         # [B, N, N, H]   (float16)
    total_elements,  # B * N * N * H  (int64)
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    B, N, H,
    BLOCK: tl.constexpr,
):
    """out = in * mask * gate  (all float16)."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    in_bounds = offs < total_elements

    # Linear index → (b,i,j,h)
    idx = offs
    b = idx // (N * N * H)
    idx = idx % (N * N * H)
    i = idx // (N * H)
    idx = idx % (N * H)
    j = idx // H
    h = idx % H

    # Compute flat pointers
    in_off  = b * stride_in_b  + i * stride_in_i  + j * stride_in_j  + h * stride_in_h
    mask_off= b * stride_mask_b+ i * stride_mask_i+ j * stride_mask_j
    gate_off= b * stride_gate_b+ i * stride_gate_i+ j * stride_gate_j+ h * stride_gate_h

    a = tl.load(in_ptr   + in_off,   mask=in_bounds, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=in_bounds, other=tl.float16(1.0))
    g = tl.load(gate_ptr + gate_off, mask=in_bounds, other=tl.float16(0.0))

    tl.store(out_ptr + in_off, a * m * g, mask=in_bounds)


# ----------------------------------------------------------------------
# Entrypoint – pure python + Triton
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # --------------------------------------------------------------
    # weight tensors – keep a half‑precision copy for heavy ops
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # linear / gate weights (store in half for TensorCore use)
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)

    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2. Cast to fp16 – all subsequent heavy ops run in half precision
    # --------------------------------------------------------------
    x = x.to(torch.float16)

    # --------------------------------------------------------------
    # 3. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim] fp16
    right = torch.nn.functional.linear(x, right_proj_w)

    # --------------------------------------------------------------
    # 4. Gates (sigmoid) – still fp16
    # --------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # --------------------------------------------------------------
    # 5. Fuse mask×gate (skip if nomask)
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool/int – cast to fp16 (0.0/1.0)
        if mask_tensor is None:
            mask = torch.ones(
                (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
                dtype=torch.float16,
                device=device,
            )
        else:
            mask = mask_tensor.to(dtype=torch.float16, device=device)

        B, N_seq, _, H = left.shape
        total = B * N_seq * N_seq * H
        BLOCK = 131072                     # ~128 k elements per program (tunable)
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left = left * mask * left_gate
        left_out = torch.empty_like(left)
        apply_mask_gate_fp16_kernel[grid](
            left,
            mask,
            left_gate,
            left_out,
            total,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            B, N_seq, H,
            BLOCK=BLOCK,
        )
        # right = right * mask * right_gate
        right_out = torch.empty_like(right)
        apply_mask_gate_fp16_kernel[grid](
            right,
            mask,
            right_gate,
            right_out,
            total,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            B, N_seq, H,
            BLOCK=BLOCK,
        )
        left  = left_out
        right = right_out
    else:
        # No mask → simple element‑wise fusion
        left  = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 6. Core TriMul – batched half‑precision GEMM:
    #    out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    # --------------------------------------------------------------
    # reshape to (B*H, N, N) for torch.bmm
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    left_mat  = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)  # (B*H, N, N)

    # (B*H, N, N) @ (B*H, N, N).T -> (B*H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7. Output LayerNorm (hidden_dim) – still half precision
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # --------------------------------------------------------------
    # 8. Apply out_gate (element‑wise)
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9. Final linear projection back to original dim (half precision)
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim] fp16

    # --------------------------------------------------------------
    # 10. Cast to float32 for API compatibility
    # --------------------------------------------------------------
    out = out.to(torch.float32)

    return out
