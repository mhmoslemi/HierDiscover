Optimized “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1. Layer‑norm the input (float32 → float16 after norm).
2. Four linear projections (dim → hidden_dim) to obtain:

    j = rem % N

    # base offsets (no hidden dim)
    lp_base = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
    rp_base = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
    lg_base = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    rg_base = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    mask_base = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load mask (scalar per (b,i,j)), broadcast later
    mask_scalar = tl.load(mask_ptr + mask_base,
                          mask=row_mask,
                          other=tl.full([1], 1.0, tl.float16)).to(tl.float16)   # [BLOCK_ROW]

    for k in range(0, H, BLOCK_H):
        h_off = k + tl.arange(0, BLOCK_H)                     # [BLOCK_H]

        # Load tensors (shape: [BLOCK_ROW, BLOCK_H])
        lp = tl.load(left_proj_ptr + lp_base[:, None] + h_off * stride_lp_h,
                     mask=row_mask[:, None], other=tl.float16(0.0))
        rp = tl.load(right_proj_ptr + rp_base[:, None] + h_off * stride_rp_h,
                     mask=row_mask[:, None], other=tl.float16(0.0))
        lg = tl.load(left_gate_ptr + lg_base[:, None] + h_off * stride_lg_h,
                     mask=row_mask[:, None], other=tl.float16(0.0))
        rg = tl.load(right_gate_ptr + rg_base[:, None] + h_off * stride_rg_h,
                     mask=row_mask[:, None], other=tl.float16(0.0))

        # broadcast mask across hidden axis
        mask_vec = mask_scalar[:, None]

        # fused operation: proj * sigmoid(gate) * mask
        # sigmoid(x) ≈ 1 / (1 + exp(-x)); using tl.exp for FP16 works fine
        left_res  = lp * (tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-lg))) * mask_vec
        right_res = rp * (tl.float16(1.0) / (tl.float16(1.0) + tl.exp(-rg))) * mask_vec

        # store results
        tl.store(left_out_ptr + lp_base[:, None] + h_off * stride_lp_h,
                 left_res, mask=row_mask[:, None])
        tl.store(right_out_ptr + rp_base[:, None] + h_off * stride_rp_h,
                 right_res, mask=row_mask[:, None])

# --------------------------------------------------------------
# Triton‑accelerated TriMul “outgoing” forward pass
# --------------------------------------------------------------
def custom_kernel(data):
    """
    TritMul “outgoing” forward pass (AlphaFold‑3).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Optional mask handling
    # ------------------------------------------------------------------
    nomask = config.get("nomask", False)
    if mask is None or nomask:
        mask = torch.ones((bs, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # Load weights (convert to FP16 once)
    # ------------------------------------------------------------------
    # LayerNorm on the input
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    # Linear projections (dim → hidden_dim)
    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    # Gates (dim → hidden_dim)
    left_gate_weight  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(torch.float16)

    # Output norm & projection
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (keep FP32 for numerical stability, then cast)
    # ------------------------------------------------------------------
    x_fp32 = F.layer_norm(input_tensor,
                          normalized_shape=(dim,),
                          weight=norm_weight.float(),
                          bias=norm_bias.float())
    x = x_fp32.to(torch.float16)           # [B,N,N,dim] in FP16

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (all in FP16)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_weight)          # [B,N,N,H]
    right_proj = F.linear(x, right_proj_weight)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Fuse projection, gate, and mask with Triton
    # ------------------------------------------------------------------
    left  = torch.empty_like(left_proj)   # FP16 buffer
    right = torch.empty_like(right_proj)

    # Strides (in elements, not bytes)
    lp_str = left_proj.stride()
    rp_str = right_proj.stride()
    lg_str = left_gate.stride()
    rg_str = right_gate.stride()
    mask_str = mask.stride()

    total_rows = bs * seq_len * seq_len
    BLOCK_ROW = 256                # rows per program (tunable)
    BLOCK_H   = 32                 # hidden‑dim chunk; 32 divides 128/384

    grid = (triton.cdiv(total_rows, BLOCK_ROW),)

    fused_gate_mask_kernel[grid](
        left_proj_ptr   = left_proj,
        right_proj_ptr  = right_proj,
        left_gate_ptr   = left_gate,
        right_gate_ptr  = right_gate,
        mask_ptr        = mask,
        left_out_ptr    = left,
        right_out_ptr   = right,
        # strides
        stride_lp_b=lp_str[0], stride_lp_i=lp_str[1], stride_lp_j=lp_str[2], stride_lp_h=lp_str[3],
        stride_rp_b=rp_str[0], stride_rp_i=rp_str[1], stride_rp_j=rp_str[2], stride_rp_h=rp_str[3],
        stride_lg_b=lg_str[0], stride_lg_i=lg_str[1], stride_lg_j=lg_str[2], stride_lg_h=lg_str[3],
        stride_rg_b=rg_str[0], stride_rg_i=rg_str[1], stride_rg_j=rg_str[2], stride_rg_h=rg_str[3],
        stride_mask_b=mask_str[0], stride_mask_i=mask_str[1], stride_mask_j=mask_str[2],
        # problem sizes
        B=bs, N=seq_len, H=hidden_dim,
        BLOCK_ROW=BLOCK_ROW, BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 4. Batched GEMM across the hidden dimension (FP16)
    #    out_h = left_h @ right_hᵀ   for each hidden channel
    # ------------------------------------------------------------------
    # Reshape to (B*H, N, N) for a single bmm call
    left_mat  = left.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)
    right_mat = right.permute(0, 3, 1, 2).reshape(bs * hidden_dim, seq_len, seq_len)

    # rightᵀ  →  (B*H, N, N) @ (B*H, N, N).T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16, shape [B*H, N, N]

    # Restore original layout [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)  # FP16

    # ------------------------------------------------------------------
    # 5. Output LayerNorm + gating (still FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 6. Final linear projection back to dim (FP16) and cast to FP32
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim] FP16
    out = out.to(torch.float32)

    return out
