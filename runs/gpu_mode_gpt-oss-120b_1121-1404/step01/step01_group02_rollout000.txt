"""
custom_kernel
==============

Forward pass of the *outgoing* TriMul module (AlphaFold‑3) with the
N³ “pairwise‑multiplication” carried out by a custom Triton kernel.

Algorithm
---------
1. Layer‑norm on the input tensor ``x``.
2. Two linear projections → ``left`` and ``right`` (shape B×N×N×H).
3. Optional mask (broadcasted on the feature dimension) is applied.
4. Gating:
     left_gate  = sigmoid(Linear(x, left_gate_weight))
     right_gate = sigmoid(Linear(x, right_gate_weight))
     out_gate   = sigmoid(Linear(x, out_gate_weight))
   ````left = left * left_gate`` , ``right = right * right_gate``.
5. Cast ``left`` / ``right`` to FP16 and reshape to a *batch of independent
   matrix‑multiplications*:
        (B·H) × (N×N)  @  (N×N)ᵀ   →   (B·H) × N × N
   The heavy N³ computation is performed by ``batched_matmul_kernel`` which
   uses a classic blocked GEMM (fp16 inputs, fp32 accumulator).
6. Reshape back to B×N×N×H.
7. Layer‑norm on the hidden dimension, element‑wise multiply by ``out_gate``,
   and a final linear projection back to the original feature dimension.

The kernel is deliberately kept simple (no shared‑memory tricks) but uses
large block sizes (128×128×32) that saturate H100 Tensor‑cores for the
target shapes (N up to 1024, H = 128).  All surrounding ops are done with
PyTorch for ease of use and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) – one independent GEMM per (batch,head)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,              # pointers to A, Bᵀ, C
    batch, M, N, K,                   # logical dimensions
    stride_ah, stride_am, stride_ak,  # A (batch, M, K) strides
    stride_bh, stride_bk, stride_bn,  # B (batch, K, N) strides
    stride_ch, stride_cm, stride_cn,  # C (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for a batch of matrices.
    A : [batch, M, K]   (fp16)
    B : [batch, K, N]   (fp16)   – already transposed w.r.t. the original
                                 “right” tensor.
    C : [batch, M, N]   (fp32 accumulator → fp32 output)
    """

    pid_batch = tl.program_id(0)   # which (b, head) pair
    pid_m = tl.program_id(1)        # block row index
    pid_n = tl.program_id(2)        # block col index

    # --------------------------------------------------------------
    # Compute offsets for the current block
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for the current tile of A and Bᵀ
    a_ptrs = a_ptr + pid_batch * stride_ah \
                   + offs_m[:, None] * stride_am \
                   + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_batch * stride_bh \
                   + offs_k[:, None] * stride_bk \
                   + offs_n[None, :] * stride_bn

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in BLOCK_K sized chunks
    # --------------------------------------------------------------
    for cur_k in range(tl.cdiv(K, BLOCK_K)):
        k_offset = cur_k * BLOCK_K

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & ((k_offset + offs_k) < K)[None, :]
        b_mask = ((k_offset + offs_k) < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp16
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp16

        # FP16×FP16 → FP32 accumulation (Tensor‑core friendly)
        acc += tl.dot(a, b)   # shape (BLOCK_M, BLOCK_N)

        # Advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # --------------------------------------------------------------
    # Store the result (apply edge masks)
    # --------------------------------------------------------------
    c_ptrs = c_ptr + pid_batch * stride_ch \
                  + offs_m[:, None] * stride_cm \
                  + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Entrypoint: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] or ``None`` (bool/float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # Model weights (bias‑free linear layers except LayerNorms)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]   # (H, dim)
    right_proj_weight = weights["right_proj.weight"]  # (H, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (H, dim)
    right_gate_weight = weights["right_gate.weight"]  # (H, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (H, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]  # (H,)
    to_out_norm_bias   = weights["to_out_norm.bias"]    # (H,)
    to_out_weight      = weights["to_out.weight"]       # (dim, H)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (dim‑wise)
    # --------------------------------------------------------------
    with torch.no_grad():
        x = F.layer_norm(
            input_tensor,
            (dim,),
            weight=norm_weight,
            bias=norm_bias,
        )   # [B, N, N, dim]  float32

        # ----------------------------------------------------------
        # 2) Linear projections (no bias) → hidden_dim
        # ----------------------------------------------------------
        left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
        right = F.linear(x, right_proj_weight)  # [B, N, N, H]

        # ----------------------------------------------------------
        # 3) Mask (optional)
        # ----------------------------------------------------------
        if mask is not None:
            # ensure mask has same dtype as projections
            m = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
            left  = left * m
            right = right * m

        # ----------------------------------------------------------
        # 4) Gating + element‑wise multiplication
        # ----------------------------------------------------------
        left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
        right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]
        out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B,N,N,H]

        left  = left * left_gate
        right = right * right_gate

        # ----------------------------------------------------------
        # 5) Cast to FP16 for the heavy N³ GEMM
        # ----------------------------------------------------------
        left_f16  = left.to(torch.float16)
        right_f16 = right.to(torch.float16)

        # Reshape to (batch * hidden_dim, N, N) → each (i,k) matrix is independent
        left_mat  = left_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
        right_mat = right_f16.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

        # Right matrix needs to be transposed so that we compute left @ rightᵀ
        right_mat_T = right_mat.permute(0, 2, 1).contiguous()   # (batch*H, K, N)

        # Output buffer – fp32 because the kernel accumulates in fp32
        out_mat = torch.empty_like(left_mat, dtype=torch.float32, device=device)

        # ----------------------------------------------------------
        # 6) Triton batched GEMM launch
        # ----------------------------------------------------------
        # Block sizes tuned for H100 Tensor‑cores (empirically good)
        BLOCK_M = 128
        BLOCK_N = 128
        BLOCK_K = 32

        # Strides in elements (not bytes)
        stride_ah = left_mat.stride(0)
        stride_am = left_mat.stride(1)
        stride_ak = left_mat.stride(2)

        stride_bh = right_mat_T.stride(0)
        stride_bk = right_mat_T.stride(1)
        stride_bn = right_mat_T.stride(2)

        stride_ch = out_mat.stride(0)
        stride_cm = out_mat.stride(1)
        stride_cn = out_mat.stride(2)

        grid = (
            B * hidden_dim,                     # batch dimension (b * head)
            (N + BLOCK_M - 1) // BLOCK_M,       # rows
            (N + BLOCK_N - 1) // BLOCK_N,       # cols
        )

        batched_matmul_kernel[grid](
            left_mat,
            right_mat_T,
            out_mat,
            B * hidden_dim, N, N, N,
            stride_ah, stride_am, stride_ak,
            stride_bh, stride_bk, stride_bn,
            stride_ch, stride_cm, stride_cn,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

        # ----------------------------------------------------------
        # 7) Reshape back to [B, N, N, hidden_dim]
        # ----------------------------------------------------------
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

        # ----------------------------------------------------------
        # 8) Output LayerNorm, gating and final linear projection
        # ----------------------------------------------------------
        out = F.layer_norm(
            out,
            (hidden_dim,),
            weight=to_out_norm_weight,
            bias=to_out_norm_bias,
        )
        out = out * out_gate               # element‑wise gating
        out = F.linear(out, to_out_weight)  # [B,N,N,dim]  float32

    return out
