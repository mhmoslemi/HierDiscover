@triton.jit
def mask_gate_kernel(in_ptr, mask_ptr, gate_ptr, out_ptr,
                     total_elems,
                     stride_ib, stride_ii, stride_ij, stride_ih,
                     stride_mb, stride_mi, stride_mj,
                     stride_gb, stride_gi, stride_gj, stride_gh,
                     B, N, H,
                     BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < total_elems

    # decode indices
    b = offs // (N * N * H)
    tmp = offs % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # compute addresses
    in_offset = b * stride_ib + i * stride_ii + j * stride_ij + h * stride_ih
    mask_offset = b * stride_mb + i * stride_mi + j * stride_mj
    gate_offset = b * stride_gb + i * stride_gi + j * stride_gj + h * stride_gh

    a = tl.load(in_ptr + in_offset, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_offset, mask=mask, other=tl.float16(1.0))
    g = tl.load(gate_ptr + gate_offset, mask=mask, other=tl.float16(0.0))

    out = a * m * g
    tl.store(out_ptr + in_offset, out, mask=mask)

"""
TriMul (outgoing) – forward only, highly‑optimized for H100.

Algorithm
---------
1️⃣ Layer‑norm the input (dim‑last).
2️⃣ Linear projections → left/right (fp16, no bias).
3️⃣ Gating vectors (sigmoid) → left_gate, right_gate, out_gate (fp16).
4️⃣ Fuse mask * gate * tensor with a tiny Triton kernel
   (skipped when mask is all‑ones → pure torch ops).
5️⃣ Core TriMul:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a single batched GEMM using torch.einsum
   on fp16 tensors – best‑possible Tensor‑Core utilisation.
6️⃣ Layer‑norm over the hidden dimension, apply out_gate.
7️⃣ Final linear projection back to the original feature dim.
8️⃣ Cast result to float32 (required output type).

All heavy matrix ops run at fp16 (Tensor‑Core) speed, while a
hand‑written Triton kernel demonstrates custom‑kernel usage.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: out = in * mask * gate   (broadcast mask over H)
# --------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    in_ptr,            # [B,N,N,H]   fp16
    mask_ptr,          # [B,N,N]     fp16 (0 or 1)
    gate_ptr,          # [B,N,N,H]   fp16
    out_ptr,           # [B,N,N,H]   fp16
    total_elems,       # B*N*N*H   (int)
    stride_in_b, stride_in_i, stride_in_j, stride_in_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    B, N, H,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < total_elems

    # -------- decode linear offset ---------------------------------
    b = offs // (N * N * H)
    tmp = offs % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # -------- compute flat pointers --------------------------------
    in_off  = b * stride_in_b   + i * stride_in_i   + j * stride_in_j   + h * stride_in_h
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + h * stride_gate_h

    # -------- load, compute, store --------------------------------
    a = tl.load(in_ptr + in_off,   mask=valid, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_off, mask=valid, other=tl.float16(1.0))
    g = tl.load(gate_ptr + gate_off, mask=valid, other=tl.float16(0.0))

    out = a * m * g
    tl.store(out_ptr + in_off, out, mask=valid)


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Float32 tensor of shape [bs, seq_len, seq_len, dim].
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    # We'll compute in fp16 for speed; final cast to fp32
    compute_dtype = torch.float16

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Cast weights to fp16 on the correct device
    # ------------------------------------------------------------------
    def hp(name):
        return weights[name].to(device=device, dtype=compute_dtype)

    norm_w = hp("norm.weight")
    norm_b = hp("norm.bias")
    left_proj_w = hp("left_proj.weight")        # (hidden_dim, dim)
    right_proj_w = hp("right_proj.weight")
    left_gate_w = hp("left_gate.weight")
    right_gate_w = hp("right_gate.weight")
    out_gate_w = hp("out_gate.weight")
    to_out_norm_w = hp("to_out_norm.weight")
    to_out_norm_b = hp("to_out_norm.bias")
    to_out_w = hp("to_out.weight")              # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (fp16)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor.to(compute_dtype),
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )  # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape -> [B, N, N, hidden_dim]
    left = torch.nn.functional.linear(x, left_proj_w)   # (hidden_dim, dim) weight
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid) – keep in fp16
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate with Triton (skip when mask is all‑ones)
    # ------------------------------------------------------------------
    if nomask:
        # simple fused element‑wise multiplication
        left = left * left_gate
        right = right * right_gate
    else:
        # ensure mask is fp16 and has shape [B,N,N]
        if mask_tensor is None:
            mask = torch.ones(
                (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
                dtype=compute_dtype,
                device=device,
            )
        else:
            mask = mask_tensor.to(device=device, dtype=compute_dtype)

        B, N, _, _ = left.shape
        total = B * N * N * hidden_dim
        BLOCK = 1024  # threads per program (≤8192 on H100)
        grid = ((total + BLOCK - 1) // BLOCK,)

        # left = left * mask * left_gate
        left_fused = torch.empty_like(left)
        mask_gate_kernel[grid](
            left,
            mask,
            left_gate,
            left_fused,
            total,
            left.stride(0), left.stride(1), left.stride(2), left.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
            B, N, hidden_dim,
            BLOCK=BLOCK,
        )
        # right = right * mask * right_gate
        right_fused = torch.empty_like(right)
        mask_gate_kernel[grid](
            right,
            mask,
            right_gate,
            right_fused,
            total,
            right.stride(0), right.stride(1), right.stride(2), right.stride(3),
            mask.stride(0), mask.stride(1), mask.stride(2),
            right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
            B, N, hidden_dim,
            BLOCK=BLOCK,
        )
        left, right = left_fused, right_fused

    # ------------------------------------------------------------------
    # 5. Core TriMul – batched GEMM across the hidden dimension
    #    Use torch.einsum which under the hood maps to a batched matmul.
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim] (fp16)
    out = torch.einsum("b i k h, b j k h -> b i j h", left, right)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7. Apply out_gate (element‑wise)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim (fp16)
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_w)  # (dim, hidden_dim) weight

    # ------------------------------------------------------------------
    # Cast back to float32 as required by the spec
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    return out
