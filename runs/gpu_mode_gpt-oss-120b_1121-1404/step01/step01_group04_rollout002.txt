@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)  # combined B*H

    # decode b and h from pid_z
    b = pid_z // H
    h = pid_z % H

    # compute starting indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # mask for valid i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, N, BLOCK_K):
        cur_k = k
        K_eff = tl.minimum(N - cur_k, BLOCK_K)  # Not needed for offset generation?

        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile (M, K)
        left_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + offs_i[:,None] * stride_l_i + offs_k[None,:] * stride_l_k
        left_mask = mask_i[:,None] & mask_k[None,:]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # load right tile transposed (K, N)
        # right shape: (b, h, j, k)
        # want (K, N) with axes (k, j)
        right_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + offs_k[:,None] * stride_r_k + offs_j[None,:] * stride_r_j
        right_mask = mask_k[:,None] & mask_j[None,:]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + offs_i[:,None] * stride_o_i + offs_j[None,:] * stride_o_j
    out_mask = mask_i[:,None] & mask_j[None,:]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)

"""
Triton‑accelerated outgoing TriMul (AlphaFold3)

Algorithm
---------
1. Cast everything to fp16 (the H100 delivers >200 TFLOP/s in fp16, so the
   N³ contraction dominates the runtime).
2. Layer‑norm on the input (dim‑wise) – performed with Torch’s FP16
   implementation.
3. Linear projections to the hidden dimension (left, right) and three
   gating linear layers (left_gate, right_gate, out_gate).  All weights are
   cast to fp16.
4. Optional mask+gate fusion: a tiny Triton kernel multiplies
   ``left = left * left_gate * mask`` and ``right = right * right_gate * mask``
   in‑place.
5. Core contraction  

        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

   This is a batched matrix‑multiply for each hidden channel.  A custom
   Triton kernel implements a blocked GEMM (BLOCK_M×BLOCK_N output tiles,
   BLOCK_K reduction) with the hidden dimension folded into the third grid
   dimension (B·H).  Accumulators are in fp32 for numerical stability,
   the final result is stored as fp16.
6. Layer‑norm over the hidden dimension, element‑wise gating by ``out_gate``,
   and a final linear projection back to ``dim`` (still in fp16).
7. Cast the result to fp32 before returning.

Only the mask‑gate fusion and the N³ contraction are written in Triton – the
remaining ops use highly‑optimized Torch kernels, which together give a large
speed‑up over the pure‑PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣  Mask + gate fusion (used only when a mask is present)
# ----------------------------------------------------------------------
@triton.jit
def fused_mask_gate_kernel(
    left_ptr,          # [B,N,N,H]  left  (in‑place)
    left_gate_ptr,     # [B,N,N,H]  left_gate
    right_ptr,         # [B,N,N,H]  right (in‑place)
    right_gate_ptr,    # [B,N,N,H]  right_gate
    mask_ptr,          # [B,N,N]    mask (0/1)
    B, N, H,           # problem sizes
    # strides for left
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    # strides for right
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    # strides for mask
    stride_m_b, stride_m_i, stride_m_j,
    BLOCK: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    valid = offs < total

    # ---- decode linear index into (b,i,j,h) ----
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # bounds check (should be true for all valid)
    valid = valid & (b < B) & (i < N) & (j < N) & (h < H)

    # pointers
    left_off   = left_ptr   + b * stride_l_b   + i * stride_l_i   + j * stride_l_j   + h * stride_l_h
    leftg_off  = left_gate_ptr + b * stride_lg_b + i * stride_lg_i + j * stride_lg_j + h * stride_lg_h
    right_off  = right_ptr  + b * stride_r_b   + i * stride_r_i   + j * stride_r_j   + h * stride_r_h
    rightg_off = right_gate_ptr + b * stride_rg_b + i * stride_rg_i + j * stride_rg_j + h * stride_rg_h
    mask_off   = mask_ptr   + b * stride_m_b   + i * stride_m_i   + j * stride_m_j

    left      = tl.load(left_off,   mask=valid, other=0.0)
    left_gate = tl.load(leftg_off,  mask=valid, other=0.0)
    right     = tl.load(right_off,  mask=valid, other=0.0)
    right_gate= tl.load(rightg_off, mask=valid, other=0.0)
    mask_val  = tl.load(mask_off,   mask=valid, other=1.0)   # 1 → no masking

    # fused mask+gate
    left  = left  * left_gate  * mask_val
    right = right * right_gate * mask_val

    tl.store(left_off,  left,  mask=valid)
    tl.store(right_off, right, mask=valid)


# ----------------------------------------------------------------------
# 2️⃣  Core N³ contraction (batched GEMM per hidden channel)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,   # [B,H,N,N]   left  (M×K)
    right_ptr,  # [B,H,N,N]   right (N×K) – we read it transposed as K×N
    out_ptr,    # [B,H,N,N]   output (M×N)
    B, H, N,    # problem sizes
    # strides
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    # block sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # tile over i (rows)
    pid_j = tl.program_id(1)   # tile over j (cols)
    pid_z = tl.program_id(2)   # combined batch * hidden channel

    # decode batch and hidden index
    b = pid_z // H
    h = pid_z % H

    # starting indices of the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets within the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # validity masks for the output tile
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32 for accuracy
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        # k‑range for this iteration
        cur_k = k
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M × K) -----
        left_ptrs = (left_ptr
                     + b * stride_l_b
                     + h * stride_l_h
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k)
        left_mask = mask_i[:, None] & mask_k[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # ----- load right tile transposed (K × N) -----
        # right layout: (b, h, j, k) -> we need (k, j)
        right_ptrs = (right_ptr
                      + b * stride_r_b
                      + h * stride_r_h
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_j)
        right_mask = mask_k[:, None] & mask_j[None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ----- dot product accumulation -----
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # Store the result (cast back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + h * stride_o_h
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=out_mask)


# ----------------------------------------------------------------------
# 3️⃣  Entry point – combines Torch ops with the Triton kernels above
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Cast everything to fp16 (fast on H100)
    # --------------------------------------------------------------
    dtype = torch.float16
    x = input_tensor.to(dtype)

    # --------------------------------------------------------------
    # 1️⃣  Input LayerNorm (dim‑wise)
    # --------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"].to(dtype),
        bias=weights["norm.bias"].to(dtype),
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) → hidden_dim
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(dtype))   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"].to(dtype))

    # --------------------------------------------------------------
    # 3️⃣  Gates (sigmoid after linear)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"].to(dtype)))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"].to(dtype)))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"].to(dtype)))  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 4️⃣  Optional mask + gate fusion (Triton)
    # --------------------------------------------------------------
    if not nomask:
        # cast mask to same dtype (0/1)
        mask_fp16 = mask_tensor.to(dtype)

        BLOCK = 128
        total = left.numel()                     # B * N * N * hidden_dim

        # Strides for all tensors (required by Triton)
        stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h = left_gate.stride()
        stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h = right_gate.stride()
        stride_m_b, stride_m_i, stride_m_j = mask_fp16.stride()

        grid = (triton.cdiv(total, BLOCK),)

        fused_mask_gate_kernel[grid](
            left_ptr=left,
            left_gate_ptr=left_gate,
            right_ptr=right,
            right_gate_ptr=right_gate,
            mask_ptr=mask_fp16,
            B=x.shape[0],
            N=x.shape[1],
            H=hidden_dim,
            stride_l_b=stride_l_b,
            stride_l_i=stride_l_i,
            stride_l_j=stride_l_j,
            stride_l_h=stride_l_h,
            stride_lg_b=stride_lg_b,
            stride_lg_i=stride_lg_i,
            stride_lg_j=stride_lg_j,
            stride_lg_h=stride_lg_h,
            stride_r_b=stride_r_b,
            stride_r_i=stride_r_i,
            stride_r_j=stride_r_j,
            stride_r_h=stride_r_h,
            stride_rg_b=stride_rg_b,
            stride_rg_i=stride_rg_i,
            stride_rg_j=stride_rg_j,
            stride_rg_h=stride_rg_h,
            stride_m_b=stride_m_b,
            stride_m_i=stride_m_i,
            stride_m_j=stride_m_j,
            BLOCK=BLOCK,
        )
    else:
        # No mask → apply only the gates
        left = left * left_gate
        right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣  Core contraction: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a blocked GEMM in Triton (fp16, accumulation fp32)
    # --------------------------------------------------------------
    # Permute to (B, H, N, N) – makes the hidden dim the second axis,
    # which maps nicely to the Triton grid.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape

    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_j, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_perm.stride()

    # Block sizes – chosen to fit in registers while keeping enough
    # parallelism.  Tweaking these may further improve performance.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H,                     # batch*hidden channel
    )
    trimul_outgoing_kernel[grid](
        left_ptr=left_perm,
        right_ptr=right_perm,
        out_ptr=out_perm,
        B=B,
        H=H,
        N=N,
        stride_l_b=stride_l_b,
        stride_l_h=stride_l_h,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_r_b=stride_r_b,
        stride_r_h=stride_r_h,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_o_b=stride_o_b,
        stride_o_h=stride_o_h,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore the original layout (B, N, N, H)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣  Output LayerNorm, gating, and final projection back to dim
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),

    )

    out = F.linear(out, weights["to_out.weight"].to(dtype))

    # --------------------------------------------------------------
    # Cast back to float32 (the model expects fp32 downstream)
    # --------------------------------------------------------------
    return out.to(torch.float32)
