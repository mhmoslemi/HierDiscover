    right_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,

    stride_l_b, stride_l_i, stride_l_k, stride_l_h,

    stride_r_b, stride_r_j, stride_r_k, stride_r_h,

    pid_n = tl.program_id(2)

```python
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def trimul_fused_gemm_kernel(
    # pointers
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    # stride pointers
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_rgate_b, stride_rgate_j, stride_rgate_k, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # problem sizes
    B, N, H,
    # compile time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)  # total B*H
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decode batch index
    b = batch // H
    h = batch % H

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this (b,h) slice
    base_left = b * stride_left_b + h * stride_left_h
    base_left_gate = b * stride_lgate_b + h * stride_lgate_h
    base_right = b * stride_right_b + h * stride_right_h
    base_right_gate = b * stride_rgate_b + h * stride_rgate_h
    base_out = b * stride_out_b + h * stride_out_h

    # accumulator (use fp32 for better accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M,K) and apply gate+mask
        a_ptrs = left_ptr + base_left + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a_gate_ptrs = left_gate_ptr + base_left_gate + offs_m[:, None] * stride_lgate_i + offs_k[None, :] * stride_lgate_k
        mask_i_ptrs = mask_ptr + b * stride_mask_b + offs_m[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        a_gate = tl.load(a_gate_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        mask_i = tl.load(mask_i_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        a = a * a_gate * mask_i

        # Load right tile (K,N) after transposition
        # We need right[b, j, k, h] -> for each k (row) and j (col)
        b_ptrs = right_ptr + base_right + offs_k[:, None] * stride_right_j + offs_n[None, :] * stride_right_k
        b_gate_ptrs = right_gate_ptr + base_right_gate + offs_k[:, None] * stride_rgate_j + offs_n[None, :] * stride_rgate_k
        mask_j_ptrs = mask_ptr + b * stride_mask_b + offs_n[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k

        b_val = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        b_gate = tl.load(b_gate_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)
        mask_j = tl.load(mask_j_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        b_val = b_val * b_gate * mask_j

        # Accumulate
        acc += tl.dot(a, b_val)   # a: (M,K), b_val: (K,N) => (M,N)

    # Store result
    out_ptrs = out_ptr + base_out + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul “outgoing” forward pass – fully fused Triton implementation.

The kernel fuses the three dominant memory‑bound operations:
  * left  = (x @ W_left )   * sigmoid(x @ W_left_gate )   * mask
  * right = (x @ W_right)   * sigmoid(x @ W_right_gate)   * mask
and the O(N³·H) reduction
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

All arithmetic is performed in FP16 (the H100 tensor‑cores are used)
while the accumulation is done in FP32 for numerical stability.
The rest of the module (LayerNorms, final projection) is executed with
regular PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: fused gating, mask and batched GEMM (outgoing variant)
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_outgoing_kernel(
    # tensors
    left_ptr, left_gate_ptr,
    right_ptr, right_gate_ptr,
    mask_ptr,
    out_ptr,
    # strides (in elements, NOT bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_lg_b, stride_lg_i, stride_lg_k, stride_lg_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_rg_b, stride_rg_i, stride_rg_k, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # problem size
    B, N, H,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batch‑MATMUL across the hidden dimension with fused element‑wise
    gate · mask multiplications.
    The launch grid is (batch = B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N)).
    """
    batch_id = tl.program_id(0)          # 0 … B*H-1
    pid_m    = tl.program_id(1)          # block row
    pid_n    = tl.program_id(2)          # block column

    # decode batch → (b, h)
    b = batch_id // H
    h = batch_id % H

    # block coordinates in the i‑ and j‑dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointer for the current (b, h) slice
    base_l   = b * stride_l_b   + h * stride_l_h
    base_lg  = b * stride_lg_b  + h * stride_lg_h
    base_r   = b * stride_r_b   + h * stride_r_h
    base_rg  = b * stride_rg_b  + h * stride_rg_h
    base_out = b * stride_out_b + h * stride_out_h

    # accumulator in FP32 (more accurate than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # K‑loop (the summed sequence dimension)
    # ------------------------------------------------------------------- #
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- left tile : shape (M, K) ----------
        #   left   = proj * gate * mask
        a_ptrs      = left_ptr      + base_l  + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a_gate_ptrs = left_gate_ptr + base_lg + offs_m[:, None] * stride_lg_i + offs_k[None, :] * stride_lg_k
        mask_i_ptrs = mask_ptr       + b * stride_mask_b + offs_m[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k

        a      = tl.load(a_ptrs,      mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a_gate = tl.load(a_gate_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        mask_i = tl.load(mask_i_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        a = a * a_gate * mask_i          # FP16 arithmetic

        # ---------- right tile (transposed) : shape (K, N) ----------
        #   right = proj * gate * mask   (the mask uses the (j,k) entry)
        b_ptrs      = right_ptr      + base_r  + offs_k[:, None] * stride_r_i + offs_n[None, :] * stride_r_k
        b_gate_ptrs = right_gate_ptr + base_rg + offs_k[:, None] * stride_rg_i + offs_n[None, :] * stride_rg_k
        mask_j_ptrs = mask_ptr        + b * stride_mask_b + offs_n[:, None] * stride_mask_i + offs_k[None, :] * stride_mask_k

        b_val = tl.load(b_ptrs,      mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b_gate= tl.load(b_gate_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        mask_j= tl.load(mask_j_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        b_val = b_val * b_gate * mask_j   # FP16

        # ---------- accumulate ----------
        # tl.dot works on FP16 → returns FP16; cast to FP32 for the sum
        acc += tl.dot(a, b_val).to(tl.float32)

    # ------------------------------------------------------------------- #
    # Write result (FP16) – layout (B,N,N,H)
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + base_out + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated TriMul “outgoing” forward pass.
    Input:  (input_tensor, mask_tensor, weights_dict, config_dict)
    Output: torch.Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------- #
    # Optional mask handling
    # ------------------------------------------------------------------- #
    nomask = config.get("nomask", False)
    if mask is None or nomask:
        mask = torch.ones((bs, seq_len, seq_len), dtype=torch.float16, device=device)
    else:
        mask = mask.to(dtype=torch.float16, device=device)

    # ------------------------------------------------------------------- #
    # Load weights & cast to FP16 where sensible
    # ------------------------------------------------------------------- #
    norm_w   = weights["norm.weight"].to(torch.float16)
    norm_b   = weights["norm.bias"].to(torch.float16)
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w      = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (FP16)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor.to(torch.float16),
        (dim,),
        weight=norm_w,
        bias=norm_b,
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias) – stay in FP16
    # ------------------------------------------------------------------- #
    left_proj  = F.linear(x, left_proj_w)   # [B,N,N,H]
    right_proj = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------- #
    # 3. Gates (sigmoid of linear) – FP16
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H]

    # ------------------------------------------------------------------- #
    # 4. Fused GEMM + gate & mask (TRITON)
    # ------------------------------------------------------------------- #
    out = torch.empty((bs, seq_len, seq_len, hidden_dim),
                      dtype=torch.float16, device=device)

    # Strides (in element counts)
    lp_s = left_proj.stride()      # (B, N, N, H)
    rp_s = right_proj.stride()
    lg_s = left_gate.stride()
    rg_s = right_gate.stride()
    m_s  = mask.stride()
    o_s  = out.stride()

    # Tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        bs * hidden_dim,                       # batch × hidden
        triton.cdiv(seq_len, BLOCK_M),         # rows
        triton.cdiv(seq_len, BLOCK_N),         # columns
    )

    trimul_outgoing_kernel[grid](
        # data pointers
        left_ptr=left_proj,
        left_gate_ptr=left_gate,
        right_ptr=right_proj,
        right_gate_ptr=right_gate,
        mask_ptr=mask,
        out_ptr=out,
        # strides
        stride_l_b=lp_s[0], stride_l_i=lp_s[1], stride_l_k=lp_s[2], stride_l_h=lp_s[3],
        stride_lg_b=lg_s[0], stride_lg_i=lg_s[1], stride_lg_k=lg_s[2], stride_lg_h=lg_s[3],
        stride_r_b=rp_s[0], stride_r_i=rp_s[1], stride_r_k=rp_s[2], stride_r_h=rp_s[3],
        stride_rg_b=rg_s[0], stride_rg_i=rg_s[1], stride_rg_k=rg_s[2], stride_rg_h=rg_s[3],
        stride_mask_b=m_s[0], stride_mask_i=m_s[1], stride_mask_k=m_s[2],
        stride_out_b=o_s[0], stride_out_i=o_s[1], stride_out_j=o_s[2], stride_out_h=o_s[3],
        # problem sizes
        B=bs, N=seq_len, H=hidden_dim,
        # compile‑time constants
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,               # good default for fp16 on H100
    )

    # ------------------------------------------------------------------- #
    # 5. Output LayerNorm + final gating
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate   # element‑wise gate (FP16)

    # ------------------------------------------------------------------- #
    # 6. Final linear projection back to model dimension
    # ------------------------------------------------------------------- #
    out = F.linear(out, to_out_w)   # bias == False

    # Return as FP32 (the rest of the model expects FP32)
    return out.to(torch.float32)
