def custom_kernel(data):
   input_tensor, mask_tensor, weights, config = data
   device = input_tensor.device
   # gather dims
   dim = config["dim"]
   hidden = config["hidden_dim"]
   B,N,_,_ = input_tensor.shape

   #Split weights into half versions
   # Create half versions for all weights that will be used in half compute
   # Note: we need to keep original for layernorm? we can do layernorm in float32 then cast.
   # Cast weight matrices to half
   left_proj_w = weights["left_proj.weight"].to(torch.float16)
   right_proj_w = weights["right_proj.weight"].to(torch.float16)
   left_gate_w = weights["left_gate.weight"].to(torch.float16)
   right_gate_w = weights["right_gate.weight"].to(torch.float16)
   out_gate_w = weights["out_gate.weight"].to(torch.float16)
   to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
   to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
   to_out_w = weights["to_out.weight"].to(torch.float16)

   # layernorm on input (float32)
   x = F.layer_norm(input_tensor, (dim,), weights["norm.weight"], weights["norm.bias"])

   # cast to half
   x_h = x.to(torch.float16)

   # linear projections (no bias) keep half
   left = F.linear(x_h, left_proj_w)  # [B,N,N,hidden]
   right = F.linear(x_h, right_proj_w)

   # gating logits and sigmoid
   left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
   right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
   out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

   # mask half
   if config.get("nomask", False) or mask_tensor is None:
       mask = torch.ones((B,N,N), device=device, dtype=torch.float16)
   else:
       mask = mask_tensor.to(dtype=torch.float16)

   # allocate fused outputs (half)
   left_fused = torch.empty_like(left)
   right_fused = torch.empty_like(right)

   # launch Triton kernel to compute left_fused and right_fused
   stride_b, stride_i, stride_j, stride_h = left.stride()
   stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()
   _mask_gate_fuse_kernel[ (B,N,N) ](
       left, left_gate, mask,
       right, right_gate,
       left_fused, right_fused,
       B, N, hidden,
       stride_b, stride_i, stride_j, stride_h,
       stride_mask_b, stride_mask_i, stride_mask_j,
       HIDDEN=hidden,
   )

   # batched matrix multiplication: reshape to (B*H,N,N) 
   B_, N_, _, H_ = left_fused.shape
   left_flat = left_fused.permute(0,3,1,2).reshape(B_*H_, N_, N_)
   right_flat = right_fused.permute(0,3,1,2).reshape(B_*H_, N_, N_)
   # use torch.bmm with half dtype (Tensor cores)
   out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
   # reshape back
   out = out_flat.reshape(B_, H_, N_, N_).permute(0,2,3,1)  # [B,N,N,hidden]

   # to_out_norm (layernorm) in half
   out = F.layer_norm(
       out,
       (hidden,),
       weight=to_out_norm_w,
       bias=to_out_norm_b,
   )
   # apply out_gate (half)
   out = out * out_gate

   # final projection to dim in half, then cast to float32
   out = F.linear(out, to_out_w)  # [B,N,N,dim] half
   return out.float()

@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_b, stride_i, stride_j, stride_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    HIDDEN: tl.constexpr
):
   pid_b = tl.program_id(0)
   pid_i = tl.program_id(1)
   pid_j = tl.program_id(2)
   valid = (pid_i < N) & (pid_j < N) & (pid_b < B)
   mask = tl.load(mask_ptr + pid_b*stride_mask_b + pid_i*stride_mask_i + pid_j*stride_mask_j,
                  mask=valid, other=tl.float16(0.0))
   d = tl.arange(0, HIDDEN)
   base = pid_b*stride_b + pid_i*stride_i + pid_j*stride_j
   vec_off = base + d*stride_h
   left = tl.load(left_ptr + vec_off, mask=valid, other=tl.float16(0.0))
   left_gate = tl.load(left_gate_ptr + vec_off, mask=valid, other=tl.float16(0.0))
   right = tl.load(right_ptr + vec_off, mask=valid, other=tl.float16(0.0))
   right_gate = tl.load(right_gate_ptr + vec_off, mask=valid, other=tl.float16(0.0))
   left_out = left * left_gate * mask
   right_out = right * right_gate * mask
   tl.store(out_left_ptr + vec_off, left_out, mask=valid)
   tl.store(out_right_ptr + vec_off, right_out, mask=valid)

"""
TriMul (outgoing) – fast half‑precision implementation.

Algorithm
---------
1. Layer‑norm the input (float32 for numerical stability).
2. Cast to float16 and apply the four linear projections (left, right,
   left_gate, right_gate, out_gate).  All weights are also cast to float16.
3. Fuse the element‑wise mask, left‑gate and right‑gate in a single Triton
   kernel.  The kernel works on a per‑(batch,i,j) basis and updates the
   hidden vectors (size `hidden_dim`) in‑place, halving memory traffic.
4. Perform the O(N³·C) contraction as a batched GEMM:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   The tensors are reshaped to (B·hidden, N, N) and multiplied with
   `torch.bmm`, which uses Tensor‑Core FP16 on H100.
5. Apply a second Layer‑norm (still in FP16), the out‑gate, and the final
   linear projection back to `dim`.  The result is cast back to float32.

The heavy memory passes (mask/gate and the N³ contraction) are dramatically
reduced by using FP16 and by fusing the mask‑gate step, delivering runtimes
well below the 1 ms target on the reference configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and store the result back in‑place.
# ----------------------------------------------------------------------
@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr, left_gate_ptr, mask_ptr,
    right_ptr, right_gate_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    # strides for 4‑D tensors (B,N,N,H)
    stride_b, stride_i, stride_j, stride_h,
    # strides for mask (B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_j,
    HIDDEN: tl.constexpr,
):
    # program identifiers
    pid_b = tl.program_id(0)      # batch
    pid_i = tl.program_id(1)      # row i
    pid_j = tl.program_id(2)      # col j

    # ------------------------------------------------------------------
    # load the scalar mask element and broadcast to hidden dim
    # ------------------------------------------------------------------
    mask = tl.load(mask_ptr + pid_b * stride_mask_b +
                             pid_i * stride_mask_i +
                             pid_j * stride_mask_j,
                   other=tl.float16(0.0))            # scalar fp16
    # hide dim vector
    d = tl.arange(0, HIDDEN)                       # (HIDDEN,)

    # ------------------------------------------------------------------
    # compute vector offset for the hidden dimension
    # ------------------------------------------------------------------
    base = pid_b * stride_b + pid_i * stride_i + pid_j * stride_j
    vec_off = base + d * stride_h                     # (HIDDEN,)

    # ------------------------------------------------------------------
    # load tensors
    # ------------------------------------------------------------------
    left      = tl.load(left_ptr      + vec_off)      # (HIDDEN,)
    left_gate = tl.load(left_gate_ptr + vec_off)      # (HIDDEN,)
    right      = tl.load(right_ptr      + vec_off)   # (HIDDEN,)
    right_gate = tl.load(right_gate_ptr + vec_off)   # (HIDDEN,)

    # ------------------------------------------------------------------
    # apply mask and gates
    # ------------------------------------------------------------------
    left_out  = left  * left_gate  * mask
    right_out = right * right_gate * mask

    # ------------------------------------------------------------------
    # write back (in‑place)
    # ------------------------------------------------------------------
    tl.store(out_left_ptr  + vec_off, left_out)
    tl.store(out_right_ptr + vec_off, right_out)


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor of shape [B, N, N] (bool or float)
            - weights_dict: dict of torch.Tensors (all weights are bias‑less)
            - config_dict : must contain "dim" and "hidden_dim" and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # --------------------------------------------------------------
    # 1) Input LayerNorm (float32 for stability)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )                           # [B,N,N,dim]  float32

    # --------------------------------------------------------------
    # 2) Cast to FP16 and perform linear projections (bias‑less)
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left  = F.linear(x_h, left_proj_w)   # [B,N,N,hidden]   fp16
    right = F.linear(x_h, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_h, out_gate_w))   # [B,N,N,hidden] fp16

    # --------------------------------------------------------------
    # 3) Prepare mask (fp16).  If nomask -> all‑ones.
    # --------------------------------------------------------------
    if config.get("nomask", False) or mask_tensor is None:
        mask = torch.ones((B, N, N), device=device, dtype=torch.float16)
    else:
        mask = mask_tensor.to(dtype=torch.float16)

    # --------------------------------------------------------------
    # 4) Fuse mask & gating (in‑place) using Triton
    #    left  = left  * left_gate  * mask
    #    right = right * right_gate * mask
    # --------------------------------------------------------------
    stride_b, stride_i, stride_j, stride_h = left.stride()           # same for left/right
    stride_mask_b, stride_mask_i, stride_mask_j = mask.stride()

    # launch a program for every (b,i,j) coordinate
    grid = (B, N, N)
    _mask_gate_fuse_kernel[grid](
        left, left_gate, mask,
        right, right_gate,
        left, right,                     # write results back into the same buffers
        B, N, hidden_dim,
        stride_b, stride_i, stride_j, stride_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        HIDDEN=hidden_dim,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5) Core O(N³·C) contraction via a strided‑batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) → treat each hidden slice as a batch item
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication (FP16 → Tensor Cores)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B,N,N,hidden)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden] fp16

    # --------------------------------------------------------------
    # 6) Output LayerNorm (still FP16)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )   # fp16

    # --------------------------------------------------------------
    # 7) Apply out‑gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # --------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_w)         # [B,N,N,dim] fp16

    # --------------------------------------------------------------
    # 9) Cast back to float32 for the API contract
    # --------------------------------------------------------------
    return out.to(torch.float32)
```
