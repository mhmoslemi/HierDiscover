def custom_kernel(data):
   input_tensor, mask, weights, config = data
   dim = config["dim"]; hidden_dim = config["hidden_dim"]
   use_mask = not config.get("nomask", True)
   # Compute layernorm of input
    # torch.nn.functional.layer_norm for dim (dim)
   eps = 1e-5
   norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"];
   x = (input_tensor - input_tensor.mean(-1, keepdim=True)) / torch.sqrt(input_tensor.var(-1, unbiased=False, keepdim=True) + eps)
   x = x * norm_weight + norm_bias

   # Proj left, right
   # linear: weight shape (hidden_dim, dim)
   left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
   right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
   left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
   right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
   out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
   left = left_proj * left_gate
   right = right_proj * right_gate
   if use_mask:
       mask_exp = mask.unsqueeze(-1).to(left.dtype)
       left = left * mask_exp
       right = right * mask_exp
   # Use triton for the pairwise multiplication-sum
   out = triton_tri_mul(left, right, config)   # custom kernel returns shape (B,N,N,hidden_dim)
   # to_out_norm (layernorm)
   to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
   out_norm = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + eps)
   out_norm = out_norm * to_out_norm_weight + to_out_norm_bias
   out_norm = out_norm * out_gate
   # final linear to dimension dim
   output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
   return output

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, H,
                  stride_lb, stride_li, stride_lk, stride_lh,
                  stride_rb, stride_rj, stride_rk, stride_rh,
                  stride_ob, stride_oi, stride_oj, stride_oh,
                  BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    # grid dims: we will launch over i,j,b,h? but we can flatten.
    # Let total blocks = B * ceil(N/BLOCK_I) * ceil(N/BLOCK_J) * ceil(H/BLOCK_H)
    # decompose pid
    # Actually we define grid = (B*ceil(N/BLOCK_I)*ceil(N/BLOCK_J)*ceil(H/BLOCK_H),)
    # Compute block indices:
    total_i = tl.cdiv(N, BLOCK_I)
    total_j = tl.cdiv(N, BLOCK_J)
    total_h = tl.cdiv(H, BLOCK_H)
    # block linear id across dimensions
    # Order: b * total_i * total_j * total_h + bi * total_j * total_h + bj * total_h + bh
    # use integer division/mod
    b = pid // (total_i * total_j * total_h)
    residual = pid % (total_i * total_j * total_h)
    bi = residual // (total_j * total_h)
    residual2 = residual % (total_j * total_h)
    bj = residual2 // total_h
    bh = residual2 % total_h

    # compute offsets
    i_start = bi * BLOCK_I
    j_start = bj * BLOCK_J
    h_start = bh * BLOCK_H

    # offsets within tile
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # create broadcast offsets for k dimension
    offs_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):

        k_start = k * BLOCK_K
        curr_k = k_start + offs_k
        mask_k = curr_k < N

        # We need left[b, i, k, h] for each i in tile, k, h in tile.

        # right pointer dims: [b, j, k, h]
        right_ptrs = right_ptr + b * stride_rb + offs_j[:, None] * stride_rj + curr_k[None, :] * stride_rk + offs_h[None, :] * stride_rh

        # Load with masks
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_h[None, :] & mask_k[None, :], other=0.0) # shape (BLOCK_I, BLOCK_H)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_h[None, :] & mask_k[None, :], other=0.0) # shape (BLOCK_J, BLOCK_H)

        # Need to multiply left[i,h] * right[j,h] over k, accumulate into acc[i,j,h].
        # Actually left and right per k contain per h values. We need cross product across i and j.
        # For each h, we need outer product of left over i with right over j (both have size K?). However we are looping over k, so at each k we have left_i_h = left[i, h] (scalar for each i,h) and right_j_h = right[j, h] (scalar for each j,h). Multiplying left_i_h * right_j_h yields shape (BLOCK_I, BLOCK_J, BLOCK_H) if we broadcast over i and j.

        # But we need to keep dimension ordering correct: left: (i,h), right: (j,h)
        # Use:
        # left_exp = left[:, None, :] # shape (BLOCK_I,1,BLOCK_H)
        # right_exp = right[None, :, :] # shape (1,BLOCK_J,BLOCK_H)
        # product = left_exp * right_exp # shape (BLOCK_I, BLOCK_J, BLOCK_H)

        prod = left[:, None, :] * right[None, :, :]  # broadcasting

"""
custom_kernel – Outgoing TriMul (AlphaFold3)

The kernel implements the heavy pairwise reduction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
for a 4‑D tensor [B, N, N, H] (H = hidden_dim).  An outer
Triton kernel computes this reduction tile‑wise over the
sequence dimensions (i, j) while iterating over k in blocks.
All other operations (LayerNorm, linear projections, gating,
masking, final norm and linear) are performed with PyTorch
calls on the same device.  The implementation works for any
batch size ≤ 2, sequence length up to 1024 and hidden dimensions
up to 128.  No gradients are required – the function is
purely a forward pass.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: compute the pair‑wise sum over the third dimension.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # const *float   [B, N, N, H]
    right_ptr,         # const *float   [B, N, N, H]
    out_ptr,           # *float         [B, N, N, H]
    B: tl.int32,
    N: tl.int32,
    H: tl.int32,
    # strides for left
    stride_lb, stride_li, stride_lk, stride_lh,
    # strides for right
    stride_rb, stride_rj, stride_rk, stride_rh,
    # strides for out
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time block sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]"""

    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decompose program id into (b, i_tile, j_tile, h) indices.
    # ------------------------------------------------------------------
    total_i = tl.cdiv(N, BLOCK_I)          # number of i‑tiles
    total_j = tl.cdiv(N, BLOCK_J)          # number of j‑tiles
    total_h = H                             # one kernel per channel h

    b = pid // (total_i * total_j * total_h)
    residual = pid % (total_i * total_j * total_h)
    bi = residual // (total_j * total_h)
    residual2 = residual % (total_j * total_h)
    bj = residual2 // total_h
    h = residual2 % total_h

    # tile start positions
    i_start = bi * BLOCK_I
    j_start = bj * BLOCK_J

    # ------------------------------------------------------------------
    # Offsets within the tile.
    # ------------------------------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the tile (BLOCK_I x BLOCK_J)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in blocks of size BLOCK_K.
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + cur_k[None, :] * stride_lk
            + h * stride_lh
        )
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # right tile: shape (BLOCK_J, BLOCK_K)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + offs_j[:, None] * stride_rj
            + cur_k[None, :] * stride_rk
            + h * stride_rh
        )
        right_tile = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Partial result for this k‑block:
        #   left_tile @ right_tile.T  -> (BLOCK_I, BLOCK_J)
        acc += tl.dot(left_tile, tl.trans(right_tile, (1, 0)))

    # ------------------------------------------------------------------
    # Store the accumulated tile into the output tensor.
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel.
# ----------------------------------------------------------------------
def tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H]  (float32, same device)
    returns   : out   : [B, N, N, H]
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    # Output buffer
    out = torch.empty_like(left)

    # Block sizes – chosen to fit comfortably in registers/shared memory.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    total_i = (N + BLOCK_I - 1) // BLOCK_I
    total_j = (N + BLOCK_J - 1) // BLOCK_J
    # One program for each (b, i_tile, j_tile, h)
    grid = (B * total_i * total_j * H,)

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        # strides (int64)
        stride_lb=left.stride(0),
        stride_li=left.stride(1),
        stride_lk=left.stride(2),
        stride_lh=left.stride(3),
        stride_rb=right.stride(0),
        stride_rj=right.stride(1),
        stride_rk=right.stride(2),
        stride_rh=right.stride(3),
        stride_ob=out.stride(0),
        stride_oi=out.stride(1),
        stride_oj=out.stride(2),
        stride_oh=out.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entrypoint required by the problem statement.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    data = (input_tensor, mask, weights, config)
    Returns the forward pass of the outgoing TriMul module.
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Configuration
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all are float32 tensors on the same device as input)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over last dimension = dim)
    # ------------------------------------------------------------------
    # Manual LayerNorm for full control of weight/bias.
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * norm_weight + norm_bias   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) + mask
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    if use_mask:
        # mask: [B, N, N] -> [B, N, N, 1] -> broadcast
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid) and apply element‑wise
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core reduction via Triton: out = Σₖ left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    out = tri_mul_outgoing_triton(left, right)   # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5) Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # 6) Apply output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)   # [B,N,N,dim]

    return output
