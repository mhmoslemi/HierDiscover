def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # dtype conversion to float16 for compute
    dtype = input_tensor.dtype
    # we will cast to float16 if it's float32
    x = input_tensor
    # LayerNorm
    # For layer norm across last dim: use weights "norm.weight", "norm.bias"
    # compute mean and var per element across dim
    # Use torch's layer_norm: (bs,N,N,dim)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Now compute linear projections
    # left_proj: linear with weight left_proj_weight shape (hidden_dim, dim)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # apply mask if needed
    if mask is not None and mask.numel() > 0 and not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape (bs,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # convert to fp16 for faster triton kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare output buffer
    out = torch.empty_like(left, dtype=torch.float16)  # same shape
    # Compute strides
    # left.stride(): (stride_b, stride_i, stride_k, stride_h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = lambda meta: (
        triton.cdiv(N, meta['BLOCK_M']),
        triton.cdiv(N, meta['BLOCK_N']),

    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    mask_n = offs_n < N

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = `dim`).
2. Linear projections (`left_proj`, `right_proj`) → shape (B,N,N,hidden_dim).
3. (Optional) apply the pairwise mask on the projected tensors.
4. Compute three gating vectors (`left_gate`, `right_gate`, `out_gate`) via
   linear + sigmoid and multiply them into the projections.
5. Core operation – a batched matrix multiplication across the sequence
   dimension:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly the einsum `'... i k d, ... j k d -> ... i j d'`.
   It is implemented in Triton as a GEMM tiled over M (i), N (j) and K (k)
   for every (batch, hidden) pair.
6. Layer‑norm on the hidden dimension, gate with `out_gate`, and a final
   linear projection back to `dim`.

Only the heavy O(N³) multiplication is performed on the GPU with Triton;
all other cheap operations are done with PyTorch.  Computations inside the
kernel use fp16 inputs with fp32 accumulation (TensorCore‑accelerated),
while the final output is returned in fp32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel for the triple‑multiplication (einsum)                       #
# --------------------------------------------------------------------------- #
@triton.jit
def _trigemm_kernel(
    left_ptr,     # *fp16 left  [B,N,N,H]
    right_ptr,    # *fp16 right [B,N,N,H]
    out_ptr,      # *fp16 out   [B,N,N,H]

    B, N, H,
    # strides for left (b,i,k,h)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right (b,j,k,h)   – we will load as (k,j)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out (b,i,j,h)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    The kernel is launched with a 3‑D grid:
        pid_m : block index over i (rows)
        pid_n : block index over j (cols)
        pid_h : combined (batch, hidden) dimension
    """
    pid_h = tl.program_id(2)                     # batch * hidden
    pid_m = tl.program_id(0)                     # i‑tile
    pid_n = tl.program_id(1)                     # j‑tile

    # ------------------------------------------------------------------- #
    # Decode batch and hidden indices from pid_h
    # ------------------------------------------------------------------- #
    batch_idx = pid_h // H
    hidden_idx = pid_h % H

    # ------------------------------------------------------------------- #
    # Tile offsets
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # j indices
    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------- #
    # Accumulator in FP32
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------- #
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)   # k indices
        mask_k = offs_k < N

        # ----- left tile : shape (M, K)   (i,k)
        left_base = left_ptr \
            + batch_idx * stride_left_b \
            + hidden_idx * stride_left_h
        left_ptrs = left_base \
            + offs_m[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        # ----- right tile : shape (K, N)   (k,j)  – note the order
        right_base = right_ptr \
            + batch_idx * stride_right_b \
            + hidden_idx * stride_right_h
        right_ptrs = right_base \
            + offs_k[:, None] * stride_right_k \
            + offs_n[None, :] * stride_right_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)

        # ----- FP16 × FP16 → FP32 accumulation (TensorCore)
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------- #
    # Store the result
    # ------------------------------------------------------------------- #
    out_base = out_ptr \
        + batch_idx * stride_out_b \
        + hidden_idx * stride_out_h
    out_ptrs = out_base \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Public entry point used by the evaluation harness                           #
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape (bs, seq_len, seq_len, dim) (dtype = float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------- #
    # 1️⃣ LayerNorm on the input (dim axis)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )  # shape (B,N,N,dim), dtype = float32

    # ------------------------------------------------------------------- #
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])    # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3️⃣ Optional pairwise mask (mask shape (B,N,N))
    # ------------------------------------------------------------------- #
    if mask_tensor is not None and not config.get("nomask", True):
        # mask is broadcasted onto the hidden dimension
        mask_exp = mask_tensor.unsqueeze(-1).to(left.dtype)  # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 4️⃣ Gating vectors (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Cast gates to the compute dtype (fp16) to avoid up‑casting later
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate = out_gate.to(torch.float16)

    # ------------------------------------------------------------------- #
    # 5️⃣ Apply gates
    # ------------------------------------------------------------------- #
    left = left.to(torch.float16) * left_gate
    right = right.to(torch.float16) * right_gate

    # ------------------------------------------------------------------- #
    # 6️⃣ Core triple‑multiplication via Triton
    # ------------------------------------------------------------------- #
    # Allocate output buffer in fp16 (will be up‑cast later)
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # Strides (in number of elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – tuned for H100 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (M‑tiles, N‑tiles, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        bs * hidden_dim,
    )

    _trigemm_kernel[grid](
        left, right, out,
        bs, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 7️⃣ Post‑processing (LayerNorm, out_gate, final linear)
    # ------------------------------------------------------------------- #
    # Convert to FP32 for the remaining LayerNorm / Linear ops
    out = out.to(torch.float32)

    # LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Apply the output gate (already fp16 → cast to fp32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
