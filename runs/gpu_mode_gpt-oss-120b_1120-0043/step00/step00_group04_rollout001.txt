                   M, N, K,
                   stride_az, stride_am, stride_ak,

                   stride_cz, stride_cm, stride_cn,

    num_n = tl.cdiv(N, BLOCK_N)

    pid_n = pid % num_n

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` on the channel dimension `dim`.
2. Linear projections from `dim → hidden_dim` for the left/right tensors.
3. Optional pairwise mask (broadcasted on the channel dimension) is applied.
4. Gating scalars (sigmoid output of three additional linear layers) are
   computed from the normalized input and applied to `left`, `right` and the
   final output.
5. The core “tri‑multiplicative” update:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is realised as many independent matrix multiplications
   `C = A @ Bᵀ` where `A` and `B` have shape `[B*hidden_dim, N, N]`.
   A custom Triton kernel (`tri_mul_kernel`) performs the
   batched GEMM with FP32 accumulation.
6. Layer‑norm over the hidden dimension, gating, and the final linear
   projection back to `dim` produce the output tensor.

The kernel only handles the O(N³·hidden_dim) part; all other ops are
performed with native PyTorch which already runs at near‑optimal speed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    a_ptr,                # A matrix pointer   (B*H, M, K)
    b_ptr,                # B matrix pointer   (B*H, K, N) – interpreted as transposed view of (N, K)
    c_ptr,                # C matrix pointer   (B*H, M, N)
    M, N, K,              # problem sizes
    stride_az, stride_am, stride_ak,   # strides for A: batch, row (M), col (K)
    stride_bz, stride_bk, stride_bn,   # strides for B (transposed view): batch, row‑k, col‑n
    stride_cz, stride_cm, stride_cn,   # strides for C: batch, row (M), col (N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ.
    A shape   : (batch, M, K)   row‑major
    B shape   : (batch, N, K)   row‑major  (will be accessed as Bᵀ)
    C shape   : (batch, M, N)   row‑major
    """
    pid = tl.program_id(0)          # tile id within a batch
    batch = tl.program_id(1)        # which (batch * hidden_dim) slice

    # -------------------------------------------------------------
    # Tile indices
    # -------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_n            # block row index
    pid_n = pid %  num_n            # block column index

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # -----------------------------------------------------------------
    # Pointers shifted to the current batch
    # -----------------------------------------------------------------
    a_batch_ptr = a_ptr + batch * stride_az
    b_batch_ptr = b_ptr + batch * stride_bz
    c_batch_ptr = c_ptr + batch * stride_cz

    # -----------------------------------------------------------------
    # Accumulator (FP32 for numerical stability)
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Main K‑loop
    # -----------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile  (M × K)
        a = tl.load(
            a_batch_ptr
            + (offs_m[:, None] * stride_am)
            + (cur_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile as transposed view (K × N)
        # Note: stride_bk corresponds to the K‑dimension (contiguous = 1)
        #       stride_bn corresponds to the N‑dimension (stride = N)
        b = tl.load(
            b_batch_ptr
            + (cur_k[:, None] * stride_bk)
            + (offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Compute dot product for the current K‑slice
        acc += tl.dot(a, b)                      # acc is FP32

    # -----------------------------------------------------------------
    # Store the result tile
    # -----------------------------------------------------------------
    tl.store(
        c_batch_ptr
        + (offs_m[:, None] * stride_cm)
        + (offs_n[None, :] * stride_cn),
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # Linear projections (dim → hidden_dim) – bias is omitted in the model
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Optional pairwise mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # Gating scalars (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Prepare tensors for the Triton kernel
    #   Shape required: (batch * hidden_dim, N, N)
    # -----------------------------------------------------------------
    #   [bs, N, N, hidden_dim] --> [bs, hidden_dim, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = bs * hidden_dim
    left_flat = left.view(batch_hidden, seq_len, seq_len).contiguous()
    right_flat = right.view(batch_hidden, seq_len, seq_len).contiguous()
    out_flat = torch.empty_like(left_flat)

    # -----------------------------------------------------------------
    # Kernel launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_m = (seq_len + BLOCK_M - 1) // BLOCK_M
    num_n = (seq_len + BLOCK_N - 1) // BLOCK_N
    total_tiles = num_m * num_n

    grid = (total_tiles, batch_hidden)   # (tiles per matrix, number of independent matrices)

    # Strides for A (left)
    stride_az = left_flat.stride(0)
    stride_am = left_flat.stride(1)   # row stride (M)
    stride_ak = left_flat.stride(2)   # col stride (K)

    # Strides for B (right) – we interpret B as (K, N) = (seq_len, seq_len) transposed view
    stride_bz = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # K‑dimension (contiguous = 1)
    stride_bn = right_flat.stride(1)   # N‑dimension (stride = N)

    # Strides for C (output)
    stride_cz = out_flat.stride(0)
    stride_cm = out_flat.stride(1)   # row stride (M)
    stride_cn = out_flat.stride(2)   # col stride (N)

    # -----------------------------------------------------------------
    # Run the fused kernel
    # -----------------------------------------------------------------
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,
        seq_len,
        seq_len,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Reshape back to [bs, N, N, hidden_dim]
    # -----------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)
    out = out.permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # Output post‑processing: LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
