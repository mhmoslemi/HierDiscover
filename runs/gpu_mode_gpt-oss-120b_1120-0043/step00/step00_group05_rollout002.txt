def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5 # from alpha fold?
    # Norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=1e-5)
    # Linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight']) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    # Mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['out_gate.weight']))
    left = left * left_gate
    right = right * right_gate

    # Convert to float16 for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    # Reshape to (B*hidden_dim, N, N)
    left_reshaped = left_fp16.permute(0,3,1,2).contiguous().view(-1, N, N)
    right_reshaped = right_fp16.permute(0,3,1,2).contiguous().view(-1, N, N)
    # Create output buffer
    out_reshaped = torch.empty_like(left_reshaped, dtype=torch.float16)

    # Call Triton kernel
    matmul_triton(left_reshaped, right_reshaped, out_reshaped, B*hidden_dim, N, N)

    # Reshape output back
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous() # shape [B,N,N,hidden_dim]

    # to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=1e-5)
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights['to_out.weight'])
    return out
```

import triton
import triton.language as tl

@triton.jit
def batch_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for block start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks to avoid out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for each batch
    a_ptrs = A + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B + pid_batch * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K - k*BLOCK_K) & mask_n[None, :], other=0.0)
        # dot product (matrix multiply block)
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write back
    c_ptrs = C + pid_batch * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul “outgoing” kernel (AlphaFold3).

The forward pass consists of:
1. Layer‑norm over the last dimension (dim).
2. Linear projections → left/right (dim → hidden_dim) and three gating
   linear layers (dim → hidden_dim).  Gates are passed through sigmoid.
3. Optional mask (broadcast on the hidden dimension) is applied before
   gating.
4. The core “TriMul” contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. a batched matrix‑multiply of shape (N×N) * (N×N)^T for each
   hidden channel d.  We fuse this using a custom Triton kernel that
   treats every (batch, hidden) pair as an independent matrix‑multiply.
   The kernel works in FP16 (accumulated in FP32) to exploit Tensor‑Cores.
5. Layer‑norm over the hidden dimension, gating with out_gate and a final
   linear projection back to dim.

Only the heavy N³ contraction is implemented in Triton; the surrounding
operations are performed with PyTorch (still highly vectorised).  The
kernel processes a 3‑D grid:
    program_id(0) → batch * hidden_dim  (BH)
    program_id(1) → block row (i)
    program_id(2) → block col (j)
Tile sizes (BLOCK_M/N/K) = 64 are chosen to fit inside H100 registers
while keeping high occupancy.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of (M×K) @ (K×N) for many batches.
# Here M = N (seq_len), K = N, and each batch corresponds to one
# hidden channel (batch * hidden_dim).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dims (all = seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, row, col) – note B is already transposed
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)      # batch * hidden_dim index
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # ---- compute the base offsets for this block ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current batch
    a_batch = a_ptr + pid_batch * stride_ab
    b_batch = b_ptr + pid_batch * stride_bb
    c_batch = c_ptr + pid_batch * stride_cb

    # accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K dimension in tiles ----
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M×K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Load B block (K×N) – note B is already transposed
        b = tl.load(
            b_batch + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        # Dot product over the K dimension
        acc += tl.dot(a, b)   # result is FP32

    # ---- write back the result (cast to FP16) ----
    c = acc.to(tl.float16)
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Callable entry point used by the evaluation harness.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]  (float32)
        - mask         : torch.Tensor [B, N, N] (bool/int) or None
        - weights      : dict of torch.Tensor, see code for required keys
        - config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # --------------------------------------------------------------
    # 2) Linear projections (dim -> hidden_dim) – no bias
    # --------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]

    # 3) Optional mask (skip if config["nomask"] is True)
    # --------------------------------------------------------------
    if not config.get("nomask", True) and mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)     # broadcast to hidden_dim
        left  = left * mask_
        right = right * mask_

    # --------------------------------------------------------------
    # 4) Gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # fp32
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # will be used later

    # Cast gates to half for the element‑wise multiplication
    left_gate  = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)

    # Apply gates
    left  = (left * left_gate).to(torch.float16)   # now FP16
    right = (right * right_gate).to(torch.float16)

    #    Shape → (B*hidden_dim, N, N)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # Move hidden_dim next to batch and flatten batch*hidden_dim
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)

    # Output buffer (FP16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # --------------------------------------------------------------
    # 6) Launch the Triton kernel
    # --------------------------------------------------------------
    BH = B * hidden_dim                                 # total number of independent matmuls
    BLOCK_M = BLOCK_N = BLOCK_K = 64                     # tile size (tunable)
    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides in elements (not bytes)
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bk, stride_bn = right_flat.stride()
    # NOTE: we need B transposed → (k, n) layout, so we swap the last two strides
    #       (row = original column, col = original row)
    stride_bk, stride_bn = right_flat.stride(2), right_flat.stride(1)

    stride_cb, stride_cm, stride_cn = out_flat.stride()

    tri_mul_kernel[
        grid
    ](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # --------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden_dim]
    # --------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # out is still FP16 → cast to FP32 for the remaining ops
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 8) Output LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                      # out_gate is FP32
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim] (FP32)

    return out
```
