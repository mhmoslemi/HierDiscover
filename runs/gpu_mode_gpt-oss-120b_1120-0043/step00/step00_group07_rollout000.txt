    hidden_dim = config["hidden_dim"]

    right = linear_triton(x, weights["right_proj.weight"])

        right = right * mask_exp

    right_gate = torch.sigmoid(linear_triton(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(linear_triton(x, weights["out_gate.weight"]))

    out = out * out_gate
    # final linear
    out = linear_triton(out, weights["to_out.weight"])  # shape (B,N,N,dim)
    return out

def layernorm_fwd_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    stride_xm, stride_xd,
    stride_ym, stride_yd,
    stride_wd, stride_bd,
    M, D, eps,
    BLOCK_D: tl.constexpr,
):
    row = tl.program_id(0)
    # Offsets for this row start
    x_ptrs = x_ptr + row * stride_xm
    y_ptrs = y_ptr + row * stride_ym
    # Compute sum and sumsq in multiple steps
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)
    # loop over d blocks
    for d in range(0, tl.cdiv(D, BLOCK_D)):
        offs = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = offs < D
        x = tl.load(x_ptrs + offs * stride_xd, mask=mask, other=0.0)

TriMul (outgoing) forward pass implemented with Triton fusions.

The kernel performs three main fused element‑wise steps:
  1) Layer‑norm on the input tensor (per (b,i,j) vector).
  2) Element‑wise multiplication of projected tensors with their sigmoid gates
     and an optional mask (fused in a single Triton kernel).
  3) A second layer‑norm on the intermediate result.

The expensive pairwise reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is carried out with a batched matrix multiplication that leverages cuBLAS
(the hidden dimension is treated as a batch dimension, i.e.
    out_d = left_d @ right_dᵀ ).
All linear projections and the final output projection are performed with
`torch.nn.functional.linear` (which is already highly optimized).

The implementation works for all configurations listed in the problem
statement (varying batch size, sequence length, dim and hidden_dim) and
uses float32 throughout.

import triton.language as tl
from torch.nn import functional as F

# -------------------------------------------------------------------------- #
# Triton kernels
# -------------------------------------------------------------------------- #

@triton.jit
def layernorm_fwd_kernel(
    x_ptr, y_ptr,               # input / output pointers (2‑D flatten)
    w_ptr, b_ptr,               # weight & bias (1‑D)

    stride_ym, stride_yd,       # strides for output matrix
    stride_wd, stride_bd,       # strides for weight / bias (always 1)
    M, D, eps,                  # matrix dims and epsilon
    BLOCK_D: tl.constexpr,      # compile‑time block size for the feature dim
):

    x_row = x_ptr + row * stride_xm              # start of the row
    y_row = y_ptr + row * stride_ym

    # ---------- first pass: compute mean & variance ----------

    sum_sq  = tl.zeros([1], dtype=tl.float32)

    for off in range(0, tl.cdiv(D, BLOCK_D)):
        d = off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = d < D
        x = tl.load(x_row + d * stride_xd, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    var  = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: write normalized output ----------
    for off in range(0, tl.cdiv(D, BLOCK_D)):
        d = off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = d < D
        x = tl.load(x_row + d * stride_xd, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        w = tl.load(w_ptr + d * stride_wd, mask=mask, other=1.0)
        b = tl.load(b_ptr + d * stride_bd, mask=mask, other=0.0)
        y = norm * w + b
        tl.store(y_row + d * stride_yd, y, mask=mask)


@triton.jit
def fused_mul_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,   # a*tensor, b*gates, c*mask, output
    stride_a, stride_b, stride_c, stride_out,
    N,                              # total number of elements
    BLOCK_SIZE: tl.constexpr,       # thread‑block size
):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N

    a = tl.load(a_ptr + offs * stride_a, mask=mask)
    b = tl.load(b_ptr + offs * stride_b, mask=mask)
    c = tl.load(c_ptr + offs * stride_c, mask=mask)

    out = a * b * c
    tl.store(out_ptr + offs * stride_out, out, mask=mask)


# -------------------------------------------------------------------------- #
# Helper functions (Python side)
# -------------------------------------------------------------------------- #

def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm over the last dimension using Triton.
    Arguments:
        x    – (..., D) tensor
        weight, bias – (D,) tensors
    Returns:
        normalized tensor with same shape as ``x`` (float32)
    """
    D = weight.shape[0]
    # flatten to 2‑D (rows, D) for the kernel
    x_flat = x.reshape(-1, D).contiguous()
    out_flat = torch.empty_like(x_flat)

    M = x_flat.shape[0]
    BLOCK_D = 128                     # enough for the supported D ≤ 384

    grid = (M,)
    layernorm_fwd_kernel[grid](
        x_flat, out_flat,
        weight, bias,
        x_flat.stride(0), x_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1),
        weight.stride(0), bias.stride(0),
        M, D, eps,
        BLOCK_D=BLOCK_D,
    )
    return out_flat.view_as(x)


def fused_gate_mask(tensor: torch.Tensor,
                    gate: torch.Tensor,
                    mask: torch.Tensor) -> torch.Tensor:
    """
    Apply ``tensor * gate * mask`` using a single Triton kernel.
    All tensors must have identical shape and be contiguous.
    """
    total = tensor.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total, BLOCK),)

    out = torch.empty_like(tensor)
    fused_mul_kernel[grid](
        tensor, gate, mask, out,
        1, 1, 1, 1,          # all strides are 1 after flattening
        total,
        BLOCK_SIZE=BLOCK,
    )
    return out


# -------------------------------------------------------------------------- #
# Main entry point
# -------------------------------------------------------------------------- #

def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.

    data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    # ------------------------------------------------------------------ #
    # unpack
    # ------------------------------------------------------------------ #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------ #
    x = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2️⃣ Linear projections (no bias) – using torch.nn.functional.linear
    # ------------------------------------------------------------------ #
    B, N, _, _ = x.shape
    x_flat = x.reshape(-1, dim)                         # (B*N*N, dim)

    left  = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------ #
    # 3️⃣ Gates (sigmoid) – also linear + sigmoid
    # ------------------------------------------------------------------ #
    left_gate  = torch.sigmoid(
        F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim)
    )
    right_gate = torch.sigmoid(
        F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim)
    )
    out_gate   = torch.sigmoid(
        F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim)
    )

    # ------------------------------------------------------------------ #
    # 4️⃣ Apply mask (if any) and gates using a fused Triton kernel
    # ------------------------------------------------------------------ #
    if mask_tensor is None:
        mask_exp = torch.ones_like(left, dtype=dtype, device=device)
    else:
        # mask shape: [B, N, N] → broadcast over hidden_dim
        mask_exp = mask_tensor.unsqueeze(-1).expand(-1, -1, -1, hidden_dim)

    left  = fused_gate_mask(left,  left_gate,  mask_exp)
    right = fused_gate_mask(right, right_gate, mask_exp)

    # ------------------------------------------------------------------ #
    # 5️⃣ Core TriMul reduction:
    #     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #     Implemented as a batched matrix multiplication per hidden channel.
    # ------------------------------------------------------------------ #
    # reshape to (B*H, N, N)
    left_reshape  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # MatMul: left @ rightᵀ   → shape (B*H, N, N)
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------ #
    # 6️⃣ LayerNorm on the hidden dimension (to_out_norm)
    # ------------------------------------------------------------------ #
    out = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps,
    )

    # ------------------------------------------------------------------ #
    # 7️⃣ Apply the output gate (element‑wise)
    # ------------------------------------------------------------------ #
    out = out * out_gate

    # ------------------------------------------------------------------ #
    # 8️⃣ Final linear projection back to ``dim``
    # ------------------------------------------------------------------ #
    out_flat = out.reshape(-1, hidden_dim)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
