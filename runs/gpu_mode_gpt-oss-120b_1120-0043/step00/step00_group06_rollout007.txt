def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)
    batch = pid_z // tl.cdiv(H, BLOCK_D)
    d_block = pid_z % tl.cdiv(H, BLOCK_D)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = d_block * BLOCK_D

    # Offsets:
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k0 = k_iter * BLOCK_K
        cur_k = k0 + offs_k
        mask_k = cur_k < N

        # left offset compute: (batch, i, k, d)
        # left_ptr + batch*stride_left_b + offs_i[:,None,None]*stride_left_i + cur_k[None,:,None]*stride_left_k + offs_d[None,None,:]*stride_left_h

        left_off = batch * stride_left_b + offs_i[:, None, None] * stride_left_i + cur_k[None, :, None] * stride_left_k + offs_d[None, None, :] * stride_left_h
        left = tl.load(left_ptr + left_off, mask=mask_i[:, None, None] & mask_k[None, :] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # right offset compute as (batch, k, j, d) (transposed)
        right_off = batch * stride_right_b + cur_k[:, None, None] * stride_right_k + offs_j[None, :, None] * stride_right_j + offs_d[None, None, :] * stride_right_h
        right = tl.load(right_ptr + right_off, mask=mask_k[:, None, None] & mask_j[None, :] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # compute dot for each d
        for d in range(0, BLOCK_D):
            if not mask_d[d]:
                continue
            a = left[:, :, d]   # shape (BLOCK_M, BLOCK_K)
            b = right[:, :, d]  # shape (BLOCK_K, BLOCK_N)
            acc[:, :, d] += tl.dot(a, b)
    # end for k

    # write back
    out_off = batch * stride_out_b + offs_i[:, None, None] * stride_out_i + offs_j[None, :, None] * stride_out_j + offs_d[None, None, :] * stride_out_h
    tl.store(out_ptr + out_off, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])
```

    hidden_dim = config["hidden_dim"]

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # number of hidden blocks
    num_h_blocks = tl.cdiv(H, BLOCK_D)

    batch = pid_z // num_h_blocks
    d_block = pid_z % num_h_blocks

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = d_block * BLOCK_D

    # ranges
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator for each hidden slice
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k0 = k_iter * BLOCK_K
        cur_k = k0 + offs_k
        mask_k = cur_k < N

                   offs_i[:, None, None] * stride_left_i + \
                   cur_k[None, :, None] * stride_left_k + \
                   offs_d[None, None, :] * stride_left_h
        left = tl.load(left_ptr + left_off,
                       mask=mask_i[:, None, None] & mask_k[None, :] & mask_d[None, None, :],
                       other=0.0)

                        mask=mask_k[:, None, None] & mask_j[None, :] & mask_d[None, None, :],
                        other=0.0)

            a = left[:, :, d]   # (M, K)
            b = right[:, :, d]  # (K, N)
            acc[:, :, d] += tl.dot(a, b)

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # LayerNorm over the last dim
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections (bias=False)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1).type_as(left)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating projections and sigmoid
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Convert to half for the kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements)
    s_left_b, s_left_i, s_left_k, s_left_h = left_fp16.stride()
    s_right_b, s_right_j, s_right_k, s_right_h = right_fp16.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 8   # hidden block

        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,

"""
TriMul (outgoing) forward pass – Triton‑accelerated N³ contraction.

The heavy part of the module is the tensor contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
For each hidden slice d this is a matrix multiplication
    (N×N) @ (N×N).T.
The kernel tiles the i, j dimensions (BLOCK_M × BLOCK_N) and the
summation dimension k (BLOCK_K).  A small hidden‑dimension tile
(BLOCK_D) is processed together, keeping an accumulator of shape
(M, N, D).  All arithmetic is performed in float32 while the
input tensors are stored as float16 to exploit the H100 tensor‑cores.
The surrounding LayerNorm, linear projections and gating are handled
in PyTorch – only the O(B·N³·H) contraction is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Arguments:
        left_ptr, right_ptr, out_ptr   – pointers to the tensors (float16 for left/right,
                                         float32 for out)
        B, N, H                        – batch size, sequence length, hidden dimension
        stride_*                       – strides (in elements) for each dimension of the
                                         corresponding tensor
        BLOCK_*                        – compile‑time tile sizes
    """
    pid_i = tl.program_id(0)      # i‑tile
    pid_j = tl.program_id(1)      # j‑tile
    pid_z = tl.program_id(2)      # combined batch + hidden‑tile

    # decode batch index and hidden‑tile index
    num_h_blocks = tl.cdiv(H, BLOCK_D)
    batch = pid_z // num_h_blocks
    d_block = pid_z % num_h_blocks

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = d_block * BLOCK_D

    # 1‑D aranges for each tile dimension
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator for the current (i,j,block_d) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k_blocks):
        k0 = k_iter * BLOCK_K
        cur_k = k0 + offs_k
        mask_k = cur_k < N

        # ---- load left[b,i,k,d]  (M,K,D) ---------------------------------
        left_off = (
            batch * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + cur_k[None, :, None] * stride_left_k
            + offs_d[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None, None] & mask_k[None, :] & mask_d[None, None, :],
            other=0.0,
        )
        left = left.to(tl.float32)                # promote to fp32 for accumulation

        # ---- load right[b,j,k,d] and transpose to (K,N,D) ----------------
        right_off = (
            batch * stride_right_b
            + cur_k[:, None, None] * stride_right_k   # K dimension first
            + offs_j[None, :, None] * stride_right_j   # N dimension second (original j)
            + offs_d[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptr + right_off,
            mask=mask_k[:, None, None] & mask_j[None, :] & mask_d[None, None, :],
            other=0.0,
        )
        right = right.to(tl.float32)

        # ---- multiply‑accumulate for each hidden slice in the tile -------
        for d in range(0, BLOCK_D):
            if not mask_d[d]:
                continue
            a = left[:, :, d]   # (M, K)
            b = right[:, :, d]  # (K, N)
            acc[:, :, d] += tl.dot(a, b)

    # ---- store the result -------------------------------------------------
    out_off = (
        batch * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_d[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptr + out_off,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module using a Triton kernel
    for the N³ contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the module output.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # -----------------------------------------------------------------
    # 1) LayerNorm over the feature dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (mask shape: [B, N, N])
    # -----------------------------------------------------------------
    if not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1).type_as(left)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Cast to half‑precision and launch the Triton kernel
    # -----------------------------------------------------------------
    left_half = left.to(torch.float16).contiguous()
    right_half = right.to(torch.float16).contiguous()

    B, N, _, _ = left_half.shape
    out = torch.empty(
        (B, N, N, hidden_dim), dtype=torch.float32, device=device
    )  # result of the contraction (fp32)

    # strides in elements (not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left_half.stride()
    s_r_b, s_r_j, s_r_k, s_r_h = right_half.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Tile sizes – chosen to fit the H100 and keep register pressure low
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_D = 8   # hidden‑tile size (must divide hidden_dim reasonably)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * ((hidden_dim + BLOCK_D - 1) // BLOCK_D),
    )

    trimul_einsum_kernel[grid](
        left_half,
        right_half,
        out,
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_j,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # 6) Post‑processing: LayerNorm, output gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
