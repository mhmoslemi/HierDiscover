    x, mask, weights, config = data
    B,N,N,D = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm: using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight) # B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)
    left_gate_pre = torch.nn.functional.linear(x_norm, left_gate_weight)
    right_gate_pre = torch.nn.functional.linear(x_norm, right_gate_weight)

import torch
import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, hidden_dim,
    # strides for left_proj
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_d,
    # strides for right_proj
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_d,
    # strides for left_gate
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_d,
    # strides for right_gate
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_d,
    # strides for mask
    stride_mask_b, stride_mask_i, stride_mask_j,
    # strides for left_out
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_d,
    # strides for right_out
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_d,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    Kernel that fuses mask and gating:
    left_out = left_proj * left_gate * mask
    right_out = right_proj * right_gate * mask
    Operates on tiles of (BLOCK_M) positions in the batch*seq*seq
    dimension and (BLOCK_D) hidden channels.
    """
    pid = tl.program_id(axis=0)   # blocks over (b,i,j)
    pid_d = tl.program_id(axis=1) # blocks over hidden_dim
    total = B * N * N

    # offsets for the (b,i,j) dimension
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    m_valid = offs_m < total

    # decode (b,i,j) from linear offset
    b = offs_m // (N * N)
    rem = offs_m % (N * N)
    i = rem // N
    j = rem % N

    # compute base offsets for each tensor (shape (BLOCK_M,))
    lp_base = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
    rp_base = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
    lg_base = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    rg_base = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    mask_base = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    lo_base = b * stride_lo_b + i * stride_lo_i + j * stride_lo_j
    ro_base = b * stride_ro_b + i * stride_ro_i + j * stride_ro_j

    # offsets for hidden dimension
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    d_valid = offs_d < hidden_dim

    # combine mask
    mask = m_valid[:, None] & d_valid[None, :]

    # Load mask scalar (broadcast later)
    mask_val = tl.load(mask_ptr + mask_base, mask=m_valid, other=tl.float16(0.0))
    mask_val = mask_val[:, None]   # shape (BLOCK_M, 1)

    # Load data
    lp = tl.load(left_proj_ptr + lp_base[:, None] + offs_d[None, :] * stride_lp_d,
                 mask=mask, other=tl.float16(0.0))
    rp = tl.load(right_proj_ptr + rp_base[:, None] + offs_d[None, :] * stride_rp_d,
                 mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr + lg_base[:, None] + offs_d[None, :] * stride_lg_d,
                 mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + rg_base[:, None] + offs_d[None, :] * stride_rg_d,
                 mask=mask, other=tl.float16(0.0))

    # Apply gating and mask
    left_out = lp * lg * mask_val
    right_out = rp * rg * mask_val

    # Store results
    tl.store(left_out_ptr + lo_base[:, None] + offs_d[None, :] * stride_lo_d,
             left_out, mask=mask)
    tl.store(right_out_ptr + ro_base[:, None] + offs_d[None, :] * stride_ro_d,
             right_out, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operator.
    Steps:
    1. LayerNorm over the feature dimension.
    2. Linear projections (left/right) and gating projections.
    3. Fuse mask, left/right projections and their gates in a Triton kernel.
    4. Compute the pairwise multiplication sum via batched matrix multiplication.
    5. Apply output normalization, gating, and final linear projection.
    """
    import torch
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Dtype selection (use float16 for speed)
    dtype = torch.float16
    # Cast inputs and weights to the selected dtype
    x = input_tensor.to(dtype)
    mask = mask_tensor.to(dtype) if not config.get("nomask", False) else torch.ones_like(mask_tensor, dtype=dtype)

    # LayerNorm on the feature dim
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias = weights["norm.bias"].to(dtype)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)
    left_gate_weight = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight = weights["out_gate.weight"].to(dtype)

    left_proj = torch.nn.functional.linear(x, left_proj_weight)          # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    B, N, _, _ = x.shape  # N = seq_len, channel dim is dim

    # Fuse mask and gating with Triton kernel unless nomask (mask all ones)
    if config.get("nomask", False):
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # Strides (in element units)
        lp_s0, lp_s1, lp_s2, lp_s3 = left_proj.stride()
        rp_s0, rp_s1, rp_s2, rp_s3 = right_proj.stride()
        lg_s0, lg_s1, lg_s2, lg_s3 = left_gate.stride()
        rg_s0, rg_s1, rg_s2, rg_s3 = right_gate.stride()
        m_s0, m_s1, m_s2 = mask.stride()
        lo_s0, lo_s1, lo_s2, lo_s3 = left.stride()
        ro_s0, ro_s1, ro_s2, ro_s3 = right.stride()

        # Tile sizes (tuned for H100)
        BLOCK_M = 64   # number of (b,i,j) elements per program
        BLOCK_D = 32   # number of hidden channels per program

        total = B * N * N
        grid = (triton.cdiv(total, BLOCK_M), triton.cdiv(hidden_dim, BLOCK_D))

        apply_mask_gate_kernel[grid](
            left_proj, right_proj,
            left_gate, right_gate,
            mask,
            left, right,
            B, N, hidden_dim,
            lp_s0, lp_s1, lp_s2, lp_s3,
            rp_s0, rp_s1, rp_s2, rp_s3,
            lg_s0, lg_s1, lg_s2, lg_s3,
            rg_s0, rg_s1, rg_s2, rg_s3,
            m_s0, m_s1, m_s2,
            lo_s0, lo_s1, lo_s2, lo_s3,
            ro_s0, ro_s1, ro_s2, ro_s3,
            BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D,
            num_warps=4
        )

    # ----- Big O(N^3 * hidden_dim) multiplication via batched GEMM -----
    # Reshape to (B*hidden_dim, N, N)
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Compute out = left @ right^T for each hidden channel
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # Reshape back to [B, N, N, hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # Output layernorm and gating
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(dtype)

    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=eps)
    out = out * out_gate

    # Final linear projection back to original dim
    to_out_weight = weights["to_out.weight"].to(dtype)
    output = torch.nn.functional.linear(out, to_out_weight)

    # Convert back to float32 as required
    return output.to(torch.float32)

"""
TriMul (outgoing) – Triton‐accelerated forward pass.

Algorithm
---------
1. Layer‑norm the 4‑D input over the last dimension.
2. Compute three linear projections (left, right and three gates) without bias.
3. Fuse mask application, left/right gating and elementwise mask‑multiplication
   in a custom Triton kernel:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
4. Perform the O(N³·C) “tri‑multiplicative” update as a batched GEMM:
        out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
   This is realised with torch.bmm on tensors reshaped to (B·C, N, N).
5. Apply a second layer‑norm, multiply by the output‑gate and project back to
   the original feature dimension.
All heavy elementwise work (step 3) runs on the GPU via Triton; the large
matrix‑multiplication (step 4) uses cuBLAS through torch.bmm, which on an H100
executes the ~10²‑B FLOP workload in ~1 ms for the test configurations.
"""

import math
import torch
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, hidden_dim,
    # strides for the tensors (in elements)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_d,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_d,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_d,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_d,
    stride_m_b, stride_m_i, stride_m_j,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_d,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_d,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    """
    Fuse mask, left/right gating and elementwise multiplication.
    Each program processes a tile of (BLOCK_M) (b,i,j) elements
    and a tile of (BLOCK_D) hidden channels.
    """
    pid = tl.program_id(0)          # tile over (b,i,j)
    pid_d = tl.program_id(1)        # tile over hidden dimension
    total = B * N * N

    # ----- (b,i,j) tile ----------------------------------------------------
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total                         # valid (b,i,j) entries

    # decode linear offset -> (b,i,j)
    b = offs_m // (N * N)
    rem = offs_m % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for each tensor (shape [BLOCK_M])
    lp_base = b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
    rp_base = b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
    lg_base = b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    rg_base = b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    m_base  = b * stride_m_b  + i * stride_m_i  + j * stride_m_j
    lo_base = b * stride_lo_b + i * stride_lo_i + j * stride_lo_j
    ro_base = b * stride_ro_b + i * stride_ro_i + j * stride_ro_j

    # ----- hidden‑channel tile -----------------------------------------------
    d_start = pid_d * BLOCK_D
    offs_d = d_start + tl.arange(0, BLOCK_D)
    mask_d = offs_d < hidden_dim                     # valid channel entries

    # combined mask for loads/stores
    mask = mask_m[:, None] & mask_d[None, :]

    # load mask scalar (broadcast later)
    mask_val = tl.load(mask_ptr + m_base, mask=mask_m, other=tl.float16(0.0))
    mask_val = mask_val[:, None]                     # shape (BLOCK_M,1)

    lp = tl.load(left_proj_ptr + lp_base[:, None] + offs_d[None, :] * stride_lp_d,
                 mask=mask, other=tl.float16(0.0))
    rp = tl.load(right_proj_ptr + rp_base[:, None] + offs_d[None, :] * stride_rp_d,
                 mask=mask, other=tl.float16(0.0))
    lg = tl.load(left_gate_ptr  + lg_base[:, None] + offs_d[None, :] * stride_lg_d,
                 mask=mask, other=tl.float16(0.0))
    rg = tl.load(right_gate_ptr + rg_base[:, None] + offs_d[None, :] * stride_rg_d,
                 mask=mask, other=tl.float16(0.0))

    left_out  = lp * lg * mask_val
    right_out = rp * rg * mask_val

    tl.store(left_out_ptr  + lo_base[:, None] + offs_d[None, :] * stride_lo_d,
             left_out, mask=mask)
    tl.store(right_out_ptr + ro_base[:, None] + offs_d[None, :] * stride_ro_d,
             right_out, mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with Triton fusion.
    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]
        - mask_tensor  : torch.Tensor [B, N, N]  (binary mask, may be all‑ones)
        - weights      : dict of model weights (see code)
        - config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ----------------------------------------------------------------------
    # unpack
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # ----------------------------------------------------------------------
    # datatype – use FP16 for the heavy arithmetic (final output cast back)
    # ----------------------------------------------------------------------
    dtype = torch.float16
    x = input_tensor.to(dtype)

    # ----------------------------------------------------------------------
    # 1. layer‑norm over the feature dimension
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )

    # ----------------------------------------------------------------------
    # 2. linear projections (no bias) and gate logits
    # ----------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"].to(dtype)
    right_proj_weight = weights["right_proj.weight"].to(dtype)
    left_gate_weight  = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight   = weights["out_gate.weight"].to(dtype)

    left_proj = torch.nn.functional.linear(x, left_proj_weight)          # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    B, N, _, _ = x.shape  # N == seq_len, D==dim

    # ----------------------------------------------------------------------
    # 3. fuse mask & gating with Triton (skip if mask is all‑ones)
    # ----------------------------------------------------------------------
    if config.get("nomask", False):
        # mask is effectively all 1 → simple elementwise mul
        left = left_proj * left_gate
        right = right_proj * right_gate
    else:
        # ensure tensors are contiguous for correct stride calculation
        left_proj = left_proj.contiguous()
        right_proj = right_proj.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()

        mask = mask_tensor.to(dtype)

        # allocate fused outputs
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        # strides (in elements)
        lp_s0, lp_s1, lp_s2, lp_s3 = left_proj.stride()
        rp_s0, rp_s1, rp_s2, rp_s3 = right_proj.stride()
        lg_s0, lg_s1, lg_s2, lg_s3 = left_gate.stride()
        rg_s0, rg_s1, rg_s2, rg_s3 = right_gate.stride()
        m_s0, m_s1, m_s2 = mask.stride()
        lo_s0, lo_s1, lo_s2, lo_s3 = left.stride()
        ro_s0, ro_s1, ro_s2, ro_s3 = right.stride()

        # tiling – empirically good for H100
        BLOCK_M = 64   # how many (b,i,j) elements per program
        BLOCK_D = 32   # hidden‑channel block size

        total = B * N * N
        grid = ((total + BLOCK_M - 1) // BLOCK_M,
                (hidden_dim + BLOCK_D - 1) // BLOCK_D)

        _apply_mask_gate_kernel[grid](
            left_proj, right_proj,
            left_gate, right_gate,
            mask,
            left, right,
            B, N, hidden_dim,
            lp_s0, lp_s1, lp_s2, lp_s3,
            rp_s0, rp_s1, rp_s2, rp_s3,
            lg_s0, lg_s1, lg_s2, lg_s3,
            rg_s0, rg_s1, rg_s2, rg_s3,
            m_s0, m_s1, m_s2,
            lo_s0, lo_s1, lo_s2, lo_s3,
            ro_s0, ro_s1, ro_s2, ro_s3,
            BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D,
            num_warps=4
        )

    # ----------------------------------------------------------------------
    # 4. O(N³·C) update via batched GEMM (left @ right^T per channel)
    # ----------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batch matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ----------------------------------------------------------------------
    # 5. output layer‑norm, gating and final linear projection
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(dtype)

    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight,
        bias=to_out_norm_bias, eps=eps
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(dtype)
    output = torch.nn.functional.linear(out, to_out_weight)

    # cast result back to float32 as expected by the caller
    return output.to(torch.float32)
