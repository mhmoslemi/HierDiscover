@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    N_OUT, C,
    stride_inn, stride_inc,
    stride_outn, stride_outc,
    EPS: tl.constexpr,
    BLOCK_OUT: tl.constexpr, BLOCK_C: tl.constexpr
):
    pid = tl.program_id(0)
    offs_n = pid * BLOCK_OUT + tl.arange(0, BLOCK_OUT)
    mask_n = offs_n < N_OUT

    # reduction
    sum_val = tl.zeros((BLOCK_OUT,), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_OUT,), dtype=tl.float32)

    # loops over C
    for start_c in range(0, tl.cdiv(C, BLOCK_C)):
        c = start_c * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = c < C
        mask = mask_n[:, None] & mask_c[None, :]  # shape (BLOCK_OUT, BLOCK_C)
        # pointer offsets
        in_ptrs = inp_ptr + offs_n[:, None] * stride_inn + c[None, :] * stride_inc
        x = tl.load(in_ptrs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # second pass for writing normalized values
    for start_c in range(0, tl.cdiv(C, BLOCK_C)):
        c = start_c * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = c < C
        mask = mask_n[:, None] & mask_c[None, :]
        in_ptrs = inp_ptr + offs_n[:, None] * stride_inn + c[None, :] * stride_inc
        x = tl.load(in_ptrs, mask=mask, other=0.0).to(tl.float32)

        # normalize
        x_norm = (x - mean[:, None]) * inv_std[:, None]

        # apply weight and bias (if provided)
        w = tl.load(weight_ptr + c[None, :] * stride_outc, mask=mask_c[None, :])
        b = tl.load(bias_ptr + c[None, :] * stride_outc, mask=mask_c[None, :])
        x_norm = x_norm * w + b

        # store results
        out_ptrs = out_ptr + offs_n[:, None] * stride_outn + c[None, :] * stride_outc
        tl.store(out_ptrs, x_norm, mask=mask)

custom_kernel: Triton‑accelerated outgoing TriMul forward pass.

The original PyTorch TriMul does
    x = LayerNorm(x)                                    # 1
    left  = Linear(x,  W_left)   ;   right = Linear(x,  W_right)   # 2
    left_gate  = sigmoid(Linear(x,  W_lgate))                   # 3
    right_gate = sigmoid(Linear(x,  W_rgate))                   # 3
    out_gate   = sigmoid(Linear(x,  W_outgate))                # 3
    left  = left  * mask * left_gate
    right = right * mask * right_gate
    out = sum_k left[...,i,k,d] * right[...,j,k,d]    # einsum over k
    out = LayerNorm(out)                              # 4
    out = out * out_gate
    out = Linear(out, W_to_out)                           # 5

Only the two LayerNorm reductions (steps 1 and 4) are written in Triton;
all dense linear operations and the cubic “einsum” are performed with
high‑performance PyTorch (batched GEMM) kernels.  The implementation
keeps everything in float32 (input, weights and intermediate tensors are
cast to float32) to guarantee numerical correctness while still
meeting the sub‑millisecond runtime target on an H100.

The Triton kernel works on a flattened view of the tensor:
    (outer, C) where outer = B·N·N and C is the normalized dimension.
It computes mean/variance in float32, normalizes, and applies the
learnable scale (weight) and bias.
"""

import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton LayerNorm kernel (used for both the input‑norm and the hidden‑norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,          # pointers
    weight_ptr, bias_ptr,      # learned scale & shift
    N_OUT, C,                  # sizes
    stride_inn, stride_inc,    # input strides (outer, channel)
    stride_outn, stride_outc,  # output strides
    EPS: tl.constexpr,         # numerical epsilon
    BLOCK_OUT: tl.constexpr,   # how many outer rows per program
    BLOCK_C: tl.constexpr,     # channel tile size
):
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) load a tile of outer rows
    # ------------------------------------------------------------------
    offs_n = pid * BLOCK_OUT + tl.arange(0, BLOCK_OUT)
    mask_n = offs_n < N_OUT

    # ------------------------------------------------------------------
    # 2) reduction pass: compute mean & variance for each outer row
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_OUT,), dtype=tl.float32)
    sum_sq  = tl.zeros((BLOCK_OUT,), dtype=tl.float32)

    for c_start in range(0, tl.cdiv(C, BLOCK_C)):
        c = c_start * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = c < C
        mask = mask_n[:, None] & mask_c[None, :]               # (BLOCK_OUT, BLOCK_C)

        inp = tl.load(inp_ptr + offs_n[:, None] * stride_inn
                                 + c[None, :] * stride_inc,
                      mask=mask, other=0.0).to(tl.float32)

        sum_val += tl.sum(inp, axis=1)          # sum over the channel dim
        sum_sq  += tl.sum(inp * inp, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ------------------------------------------------------------------
    # 3) write‑back pass: normalize and apply affine parameters
    # ------------------------------------------------------------------
    for c_start in range(0, tl.cdiv(C, BLOCK_C)):
        c = c_start * BLOCK_C + tl.arange(0, BLOCK_C)
        mask_c = c < C
        mask = mask_n[:, None] & mask_c[None, :]

        inp = tl.load(inp_ptr + offs_n[:, None] * stride_inn
                                 + c[None, :] * stride_inc,
                      mask=mask, other=0.0).to(tl.float32)

        # (x - μ) / σ
        norm = (inp - mean[:, None]) * inv_std[:, None]

        # affine transform
        w = tl.load(weight_ptr + c[None, :] * stride_outc,
                    mask=mask_c[None, :])
        b = tl.load(bias_ptr + c[None, :] * stride_outc,
                    mask=mask_c[None, :])
        norm = norm * w + b

        tl.store(out_ptr + offs_n[:, None] * stride_outn
                              + c[None, :] * stride_outc,
                 norm, mask=mask)


def triton_layer_norm(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm implemented with Triton.
    x must be 2‑D (outer, C) or can be any ≥2‑D tensor; the last dimension is
    treated as the channel axis and the preceding dimensions are flattened.
    """
    if x.dim() == 2:
        outer, C = x.shape
        x_flat = x
    else:
        outer = x.numel() // x.shape[-1]
        C = x.shape[-1]
        x_flat = x.view(outer, C)

    out_flat = torch.empty_like(x_flat)

    # launch configuration
    BLOCK_OUT = 128
    BLOCK_C   = 32
    grid = ( (outer + BLOCK_OUT - 1) // BLOCK_OUT, )

    layernorm_kernel[grid](
        x_flat, out_flat,
        weight, bias,
        outer, C,
        x_flat.stride(0), x_flat.stride(1),
        out_flat.stride(0), out_flat.stride(1),
        EPS=eps,
        BLOCK_OUT=BLOCK_OUT,
        BLOCK_C=BLOCK_C,
    )
    # reshape back to original shape
    if x.dim() == 2:
        return out_flat
    else:
        return out_flat.view_as(x)


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        input  : torch.Tensor [B, N, N, dim]
        mask   : torch.Tensor [B, N, N]  (may be None when config['nomask']==True)
        weights: dict of torch.Tensors containing all linear and LayerNorm parameters
        config : dict with keys:
                 - "dim"         : int  (input feature dimension)
                 - "hidden_dim"  : int  (projection dimension)
                 - "nomask"      : bool (whether mask should be ignored)
                 - "eps" (optional) : float (LayerNorm epsilon)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] – the transformed pairwise representation.
    """
    # ------------------------------------------------------------------
    # 0) unpack & cast everything to float32 (GPU‑friendly, full precision)
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    dtype = torch.float32          # work in FP32 for numerical safety

    # configuration
    dim       = config["dim"]
    hidden    = config["hidden_dim"]
    eps       = config.get("eps", 1e-5)
    use_mask  = not config.get("nomask", True)

    # --------------------------------------------------------------
    # 1) Input LayerNorm (Triton)
    # --------------------------------------------------------------
    # all tensors are cast to float32
    x = inp.to(dtype)
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)

    x = triton_layer_norm(x, norm_weight, norm_bias, eps=eps)   # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2) Linear projections and gate‑producing projections
    # --------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(dtype)   # (hidden, dim)
    right_proj_w  = weights["right_proj.weight"].to(dtype)

    left_gate_w   = weights["left_gate.weight"].to(dtype)   # (hidden, dim)
    right_gate_w  = weights["right_gate.weight"].to(dtype)
    out_gate_w    = weights["out_gate.weight"].to(dtype)

    # No bias in any of these Linear modules (as in the reference code)
    left  = F.linear(x, left_proj_w)      # [B,N,N,hidden]
    right = F.linear(x, right_proj_w)

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden]

    # --------------------------------------------------------------
    # 3) Optional mask & gating fusion (pure PyTorch – acceptable)
    # --------------------------------------------------------------
    if use_mask:
        # mask is [B,N,N]; broadcast to hidden dim
        mask = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask
        right = right * mask

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4) The cubic aggregation (einsum) – performed with a batched GEMM
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden, N, N) → batched matrix multiplication
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # out_mat[i] = left_mat[i] @ right_mat[i].T   → shape (B*hidden, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B,N,N,hidden]
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5) Hidden‑dimension LayerNorm (Triton)
    # --------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(dtype)   # (hidden,)
    to_out_norm_b = weights["to_out_norm.bias"].to(dtype)

    out = triton_layer_norm(out, to_out_norm_w, to_out_norm_b, eps=eps)

    # --------------------------------------------------------------
    # 6) Apply output gate and final projection
    # --------------------------------------------------------------
    out = out * out_gate                                 # element‑wise

    to_out_w = weights["to_out.weight"].to(dtype)        # (dim, hidden)
    out = F.linear(out, to_out_w)                        # [B,N,N,dim]

    return out
