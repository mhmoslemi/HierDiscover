```python
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_x = tl.program_id(0)
    pid_y = tl.program_id(1)
    pid_z = tl.program_id(2)
    
    # Determine batch and d block
    num_d_blocks = tl.cdiv(hidden_dim, BLOCK_D)
    batch = pid_z // num_d_blocks
    d_block = pid_z % num_d_blocks
    
    # Compute start indices
    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N
    d0 = d_block * BLOCK_D
    
    # Range offsets
    i_offset = i0 + tl.arange(0, BLOCK_M)
    j_offset = j0 + tl.arange(0, BLOCK_N)
    d_offset = d0 + tl.arange(0, BLOCK_D)
    
    # Bounds mask
    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < hidden_dim
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)
    
    # Loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        # Compute actual K block size (triton's range masked)
        curk = k0 + tl.arange(0, BLOCK_K)
        k_mask = curk < N
        
        # left pointer offsets
        # left: (batch, i, k, d)
        left_ptrs = left_ptr + batch * stride_lb \
                    + i_offset[:, None, None] * stride_li \
                    + curk[None, :, None] * stride_lk \
                    + d_offset[None, None, :] * stride_ld
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)
        
        # right pointer offsets (batch, j, k, d)
        right_ptrs = right_ptr + batch * stride_rb \
                     + j_offset[:, None, None] * stride_rj \
                     + curk[None, :, None] * stride_rk \
                     + d_offset[None, None, :] * stride_rd
        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)
        
        # Loop over d slice
        for d in range(BLOCK_D):
            # shape (BLOCK_M, BLOCK_K) and (BLOCK_N, BLOCK_K)
            left_d = left[:, :, d]
            right_d = right[:, :, d]
            # Compute dot product left_d (M,K) @ right_d.T (K,N)
            acc[:, :, d] += tl.dot(left_d, tl.trans(right_d))
    
    # Write out
    out_ptrs = out_ptr + batch * stride_ob \
               + i_offset[:, None, None] * stride_oi \
               + j_offset[None, :, None] * stride_oj \
               + d_offset[None, None, :] * stride_od
    tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
    # 1. LayerNorm on input
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    eps = 1e-5  # default
    # Using torch.nn.functional.layer_norm:
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # 2. Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)    # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    
    # Optionally cast to fp16
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    
    # Mask
    if not nomask:
        # mask shape [B,N,N]
        mask = mask_tensor.to(left.dtype)
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask
    
    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid().to(torch.float16)
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid().to(torch.float16)
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid().to(torch.float16)
    
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    
    # Prepare output tensor
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    
    # Call Triton kernel
    trimul_triton(left, right, out)
    
    # LayerNorm on out (hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    
    # Multiply by out_gate
    out = out * out_gate
    
    # Final Linear to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # shape [B,N,N,dim]
    # Cast to float32
    out = out.to(torch.float32)
    return out

def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim, num_d_blocks,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):

    pid_y = tl.program_id(1)
    pid_z = tl.program_id(2)
    batch = pid_z // num_d_blocks
    d_block = pid_z % num_d_blocks
    # start indexes
    i0 = pid_x * BLOCK_M
    j0 = pid_y * BLOCK_N
    d0 = d_block * BLOCK_D
    
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    d = d0 + tl.arange(0, BLOCK_D)
    k = tl.arange(0, BLOCK_K)

    j_mask = j < N
    d_mask = d < hidden_dim

        cur_k = k0 + k
        k_mask = cur_k < N

        # left load
        left_ptrs = left_ptr + batch * stride_lb \
                    + i[:, None, None] * stride_li \
                    + cur_k[None, :, None] * stride_lk \
                    + d[None, None, :] * stride_ld

                       mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],

"""
custom_kernel for the outgoing TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the input tensor `x` over its last dimension `dim`.
2. Linear projections (no bias) to the hidden dimension:
      left  = x @ W_left   (shape [B,N,N,H])
      right = x @ W_right
   where `H = hidden_dim`.
3. If a mask is present (config["nomask"] == False) multiply both projected
   tensors by the mask (broadcasted on the hidden dimension).
4. Compute three sigmoid gates from the normalized `x`:
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
5. Apply the two inner gates to the projected tensors:
      left  = left  * left_gate
      right = right * right_gate
   (All tensors are cast to float16 for the heavy‐weight part.)
6. Compute the core “tri‑multiplicative” update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This dense O(N³·H) contraction is performed by a tiled Triton kernel.
   The kernel processes a 2‑D tile of the (i,j) plane and a small slice
   of the hidden dimension (`BLOCK_D`).  It accumulates in float32 while
   loading in float16.
7. Layer‑normalize the result over the hidden dimension using learned
   `to_out_norm` parameters.
8. Apply the outer gate `out_gate`.
9. Final linear projection back to the original dimension `dim`
   (`to_out` weight).  The final output is returned as float32.

The implementation uses Triton for step 6 (the dominant N³ work) and
PyTorch for the remaining linear / LayerNorm / sigmoid operations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: tiled multiplication with a hidden‑dimension slice
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,         # const float16 *
    right_ptr,        # const float16 *
    out_ptr,          # float32 *
    B, N, H,          # runtime sizes
    num_d_blocks,     # number of hidden‑dim blocks = ceil(H / BLOCK_D)
    # strides (in elements)
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_z = tl.program_id(2)   # combined (batch, hidden‑block) tile

    # ----- decode batch and hidden‑block --------------------------------
    batch = pid_z // num_d_blocks
    d_block = pid_z % num_d_blocks

    # ----- start indices for this tile ----------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = d_block * BLOCK_D

    # ----- index vectors -------------------------------------------------
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    d = d0 + tl.arange(0, BLOCK_D)
    k = tl.arange(0, BLOCK_K)

    i_valid = i < N
    j_valid = j < N
    d_valid = d < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- loop over reduction axis k ------------------------------------
    for kk in range(0, N, BLOCK_K):
        cur_k = kk + k
        k_valid = cur_k < N

        # ----- load left tile (M, K, D) ---------------------------------
        left_offsets = (
            batch * stride_lb +
            i[:, None, None] * stride_li +
            cur_k[None, :, None] * stride_lk +
            d[None, None, :] * stride_ld
        )
        left = tl.load(
            left_ptr + left_offsets,
            mask=i_valid[:, None, None] & k_valid[None, :, None] & d_valid[None, None, :],
            other=0.0
        )   # shape (M, K, D), FP16

        # ----- load right tile (N, K, D) --------------------------------
        right_offsets = (
            batch * stride_rb +
            j[:, None, None] * stride_rj +
            cur_k[None, :, None] * stride_rk +
            d[None, None, :] * stride_rd
        )
        right = tl.load(
            right_ptr + right_offsets,
            mask=j_valid[:, None, None] & k_valid[None, :, None] & d_valid[None, None, :],
            other=0.0
        )   # shape (N, K, D), FP16

        # ----- inner product for each d slice ----------------------------
        for d_idx in range(BLOCK_D):
            # (M, K) @ (K, N) -> (M, N)
            left_d  = left[:, :, d_idx]          # (M, K)
            right_d = right[:, :, d_idx]         # (N, K)
            acc[:, :, d_idx] += tl.dot(left_d, tl.trans(right_d))

    # ----- store the result ----------------------------------------------
    out_offsets = (
        batch * stride_ob +
        i[:, None, None] * stride_oi +
        j[None, :, None] * stride_oj +
        d[None, None, :] * stride_od
    )
    tl.store(
        out_ptr + out_offsets,
        acc,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & d_valid[None, None, :]
    )


def _trimul_triton(left: torch.Tensor,
                   right: torch.Tensor,
                   out: torch.Tensor) -> None:
    """
    Wrapper that launches the tiled Triton kernel.
    left/right: [B, N, N, H]  (float16, contiguous)
    out:        [B, N, N, H]  (float32, contiguous)
    """
    B, N, _, H = left.shape

    # strides in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tiling constants – chosen to stay within the ~64KB register budget
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 4

    num_d_blocks = (H + BLOCK_D - 1) // BLOCK_D

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,    # i‑tiles
             (N + BLOCK_N - 1) // BLOCK_N,    # j‑tiles
             B * num_d_blocks )               # (batch, hidden‑block) tiles

    _trimul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        num_d_blocks,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    # Synchronize to make sure the kernel has finished before we use `out`
    torch.cuda.synchronize()


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a fused Triton kernel
    for the N³ contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (may be unused)
        - weights_dict : dictionary with the model parameters
        - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )                        # [B,N,N,dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) -> hidden dimension
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    left = F.linear(x, left_proj_w)      # [B,N,N,hidden_dim] float32
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is 0/1; cast to the proj dtype and broadcast
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gate computation (sigmoid) – also projected from x
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 5. Apply inner gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for the heavy kernel (reduces bandwidth)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out_gate = out_gate.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 6. Tri‑multiplicative contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=device)  # accumulator in FP32
    _trimul_triton(left, right, out)   # out = Σ_k left[...,k] * right[...,k]

    # ------------------------------------------------------------------
    # 7. Output layer‑norm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # Cast weight/bias to match out dtype (float32 → float32 is fine)
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8. Apply outer gate
    # ------------------------------------------------------------------
    out = out * out_gate  # both in float16 -> result promoted to float32

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    # Cast final weight to float16 for consistency (minor speedup)
    to_out_w_fp16 = to_out_w.to(torch.float16)
    out = F.linear(out.to(torch.float16), to_out_w_fp16)   # [B,N,N,dim], still fp16
    out = out.to(torch.float32)                           # return float32 as requested

    return out
```

**Explanation of key points**

* **Triton kernel** – Implements the core contraction `out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]`.  
  * Tiles the `(i, j)` plane with `BLOCK_M × BLOCK_N` threads.  
  * Processes the hidden dimension in small blocks (`BLOCK_D = 4`) to keep register usage below Triton’s limit.  
  * Loads the three‑dimensional tiles of `left` and `right` (`M × K × D` and `N × K × D`) in FP16, performs a dot product per hidden slice with `tl.dot`, and accumulates in FP32.

* **Host‑side PyTorch** – Handles all linear layers, sigmoid gates, and LayerNorms, which are cheap compared to the `O(N³·H)` work and are efficiently executed by cuBLAS / cuDNN.

* **Precision** – All heavy memory‑bound work (projections, gating, the contraction) is performed in `float16` to halve bandwidth. Accumulation is done in `float32` to preserve numerical stability. The final output is cast back to `float32`.

* **Mask handling** – When `config["nomask"]` is `False` the mask is applied (broadcasted on the hidden dimension) before the inner gates, exactly matching the reference implementation.

The kernel runs on a single launch with a grid of size  
`(ceil(N/BLOCK_M), ceil(N/BLOCK_N), B * ceil(H/BLOCK_D))`,  
which comfortably fits the test configurations and meets the target runtime on an H100.
