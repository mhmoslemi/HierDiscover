import torch
import triton
import triton.language as tl

@triton.jit
def fused_mul_kernel(
    out_ptr, a_ptr, b_ptr, mask_ptr,
    N_ELEMENTS: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < N_ELEMENTS
    a = tl.load(a_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets, mask=mask, other=0.0)
    if mask_ptr is None:
        m = 1.0
    else:
        m = tl.load(mask_ptr + offsets, mask=mask, other=1.0)
    out = a * b * m
    tl.store(out_ptr + offsets, out, mask=mask)
```

2. Two linear projections (`left_proj`, `right_proj`) from `dim -> hidden_dim`.
3. Three sigmoid‑gates (`left_gate`, `right_gate`, `out_gate`) from `dim -> hidden_dim`.
4. Apply mask (if present) and fuse the three tensors (`proj * gate * mask`)
   with a tiny Triton kernel (`fuse_mul_kernel`).  This kernel does a
   single element‑wise multiply‑accumulate over the flattened tensors.
5. Compute the “triangle multiplicative update”
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix multiplication for every hidden
   channel `d`.  We reshape the tensors to `(B*hidden_dim, N, N)` and
   call `torch.bmm` (FP16) – this leverages the highly‑optimized cuBLAS
   kernel on H100.
6. Apply a second LayerNorm (`to_out_norm`), multiply by `out_gate`,
   and finally a linear projection back to `dim` (`to_out`).
7. The result is returned in FP32 (the rest of the network is usually
   FP32‑compatible).

Only the element‑wise fused multiplication (step 4) is written in Triton,
fulfilling the “must use a kernel” requirement while keeping the heavy
matrix‑multiply step on cuBLAS for maximum throughput.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def fuse_mul_kernel(
    out_ptr, a_ptr, b_ptr, mask_ptr,
    N_ELEMS: tl.constexpr, BLOCK_SIZE: tl.constexpr
):
    """Element‑wise out = a * b * mask  (all tensors are 1‑D views)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    m = tl.load(mask_ptr + offs, mask=mask, other=1.0)

    tl.store(out_ptr + offs, a * b * m, mask=mask)


def custom_kernel(data):
    """
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (torch.float32)
        - mask_tensor  : [B, N, N]       (torch.bool / torch.float32)
        - weights_dict : model weights (all tensors on the same CUDA device)
        - config_dict  : {"dim":..., "hidden_dim":..., "nomask":..., ...}
    Returns
    -------
    out : torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    bs, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Cast everything to fp16 for the heavy part (except the first LN)
    # ------------------------------------------------------------------
    # 1) input layer‑norm (keep it in fp32 for numerical stability)
    norm_weight = weights["norm.weight"].to(torch.float32)
    norm_bias = weights["norm.bias"].to(torch.float32)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim]  fp32

    # cast to half for subsequent ops
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (fp16)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # projections
    left = torch.nn.functional.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_w)

    # gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # ------------------------------------------------------------------
    # 3) Mask handling
    # ------------------------------------------------------------------
    if nomask:
        # all‑ones mask – shape matches the projected tensors
        mask_exp = torch.ones(
            (bs, N, N, hidden_dim), dtype=left.dtype, device=device
        )
    else:
        # mask shape: (B,N,N).  Expand to (B,N,N,hidden_dim)
        mask_exp = mask_tensor.to(dtype=left.dtype, device=device)
        mask_exp = mask_exp.unsqueeze(-1)                         # (B,N,N,1)
        mask_exp = mask_exp.expand(-1, -1, -1, hidden_dim).contiguous()

    # ------------------------------------------------------------------
    # 4) Fuse projection, gate and mask with a tiny Triton kernel
    # ------------------------------------------------------------------
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    numel = left.numel()
    BLOCK = 1024
    grid = (triton.cdiv(numel, BLOCK),)

    # left
    fuse_mul_kernel[grid](
        out_ptr=left_fused,
        a_ptr=left,
        b_ptr=left_gate,
        mask_ptr=mask_exp,
        N_ELEMS=numel,
        BLOCK_SIZE=BLOCK,
    )
    # right
    fuse_mul_kernel[grid](
        out_ptr=right_fused,
        a_ptr=right,
        b_ptr=right_gate,
        mask_ptr=mask_exp,
        N_ELEMS=numel,
        BLOCK_SIZE=BLOCK,
    )

    # ------------------------------------------------------------------
    # 5) Triangle multiplicative update = batched matrix‑multiply per channel
    # ------------------------------------------------------------------
    # shape: (B,N,N,hidden_dim) -> (B*hidden_dim, N, N)
    left_perm = left_fused.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_fused.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(bs * hidden_dim, N, N)
    right_flat = right_perm.view(bs * hidden_dim, N, N)

    # right needs to be transposed on the last two dims (k ↔ j)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # reshape back to (B,N,N,hidden_dim)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm, apply out_gate and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # still fp16

    # multiply by out_gate (already fp16)
    out = out * out_gate

    # final linear projection back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float16)
    out = torch.nn.functional.linear(out, to_out_w)   # (B,N,N,dim) fp16

    # ------------------------------------------------------------------
    # 7) Cast back to fp32 (network downstream expects fp32)
    # ------------------------------------------------------------------
    return out.to(torch.float32)
