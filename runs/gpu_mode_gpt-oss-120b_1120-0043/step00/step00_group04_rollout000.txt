@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offsets
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask within bounds
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # Load A and B
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float32), mask=c_mask)

    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    device = input_tensor.device

    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Optionally cast to float16 for speed in later compute
    x = x.to(torch.float16)

    # Projections
    left_proj_weight = weights["left_proj.weight"] # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)

    # Linear projections without bias
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:
        # mask shape [B, N, N]; unsqueeze -1 to broadcast
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    right = right.permute(0, 3, 1, 2).contiguous()

        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,

    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape (B,N,N,hidden)

"""
Outgoing TriMul (TriMul‑out) implementation.

The module normalises the 4‑D input, applies two linear projections (left/right)
with per‑position gating, optionally masks the projected tensors and finally
computes the core N×N matrix multiplication summed over the third sequence
dimension:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

We flatten the batch and hidden dimension (B·H, N, N) and execute this
batched GEMM with a custom Triton kernel that performs
C = A @ Bᵀ in float32 (A/B are stored as float16 for memory‑bandwidth
efficiency).  The surrounding LayerNorms, projections, gates and the final
linear projection are performed with PyTorch for brevity – the kernel
covers the dominant O(N³·H) work.

The kernel is written for Triton 3.3.1 and targets NVIDIA H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A : (batch, M, K)   – here batch = B * hidden_dim, M = N, K = N
# B : (batch, N, K)   – B already transposed on the host (rightᵀ)
# C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 3‑D grid: (batch, block_row, block_col)
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # ---- block offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K --------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers with proper strides
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        # masked loads; cast to FP32 for the dot
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ---- write result --------------------------------------------------
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: [B, N, N, dim]    (float32)
            mask_tensor : [B, N, N]       (optional, bool or float)
            weights_dict: model parameters (LayerNorm, Linear weights, etc.)
            config_dict : contains "dim", "hidden_dim", "nomask"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)                     # (B, N, N, dim)  FP32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    left = F.linear(x, left_proj_w)                    # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the heavy N³ work
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6) Reshape to (B*H, N, N) and prepare rightᵀ
    # ------------------------------------------------------------------
    # (B, N, N, hidden) -> (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    left_flat = left.view(B * hidden_dim, N, N)            # (B*H, N, N)
    right_flat = right.view(B * hidden_dim, N, N)          # (B*H, N, N)
    right_t = right_flat.transpose(1, 2).contiguous()     # (B*H, N, N) = Bᵀ

    # output container (FP32 accumulator)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # ------------------------------------------------------------------
    # 7) Launch Triton batched GEMM kernel
    # ------------------------------------------------------------------
    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_t.stride()
    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    # Kernel block sizes – tuned for H100 (shared‑memory friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_gemm_kernel[grid](
        left_flat, right_t, out_flat,
        B * hidden_dim, N, N, N,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8) Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 9) Final LayerNorm, out‑gate and linear projection to `dim`
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)                # (B, N, N, hidden)

    out = out * out_gate                                   # broadcast (B,N,N,hidden)

    to_out_w = weights["to_out.weight"]                     # (dim, hidden)
    out = F.linear(out, to_out_w)                          # (B, N, N, dim)

    return out
```
