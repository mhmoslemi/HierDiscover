import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # all on cuda
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # get linear weights
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"] # shape [dim, hidden_dim]

    # Normalization: LayerNorm over last dim
    # Using torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)
    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # allocate output tensor (float32)
    out = torch.empty_like(left, dtype=torch.float32) # [B,N,N,hidden_dim]

    # call kernel
    tri_mul_forward(left_h, right_h, out)

    # after kernel, out is float32
    # to_out_norm layernorm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # out gating
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, to_out_weight) # shape [B,N,N,dim]

    return out
```

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)  # combined batch*hidden
    pid_m = tl.program_id(0)  # tile row (i)
    pid_n = tl.program_id(1)  # tile col (j)
    # get batch index and hidden index
    b = pid_z // D
    d = pid_z % D

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # Pointers base for this (b,d)
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base = out_ptr + b * stride_ob + d * stride_od

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + offs_k
        mask_k_tile = k_offset < N

        # Load left tile (M x K)
        left_ptrs = left_base + offs_m[:, None] * stride_li + k_offset[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k_tile[None, :], other=0.0)

        # Load right tile (N x K)
        right_ptrs = right_base + offs_n[:, None] * stride_ri + k_offset[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k_tile[None, :], other=0.0)

        # Compute dot product: (M,K) x (K,N) -> (M,N)
        # right currently shape (BLOCK_N, BLOCK_K) -> need transpose to (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right.T)

    # Store result
    out_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Linear projections (bias‑free) to a hidden dimension:
       left = x @ W_left   ,   right = x @ W_right
3. Optional pair‑mask (broadcasted on the hidden dim) is applied.
4. Compute three sigmoid gates from the normalized input:
       left_gate  = sigmoid(x @ W_left_gate)
       right_gate = sigmoid(x @ W_right_gate)
       out_gate   = sigmoid(x @ W_out_gate)
   Apply the first two gates element‑wise to *left* and *right*.
5. Cast *left* and *right* to FP16 and launch a Triton kernel that
   computes, for every batch *b* and hidden channel *d*:

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   (i.e. a batched matrix‑multiply L @ Rᵀ for each hidden channel).
   The kernel accumulates in FP32 and writes the result in FP32.
6. Apply a second layer‑norm over the hidden dimension, multiply by *out_gate*,
   and finally a bias‑free linear projection back to the original dimension.

Only the O(N³) reduction (step 5) is performed in Triton; the remaining
operations use regular PyTorch which already runs efficiently on the GPU.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: weighted batched matrix multiplication for the TriMul step
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # FP16, shape [B, N, N, D]
    right_ptr,         # FP16, shape [B, N, N, D]
    out_ptr,           # FP32, shape [B, N, N, D]
    B, N, D,           # runtime sizes
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides (b,i,k,d)
    stride_rb, stride_ri, stride_rk, stride_rd,   # right strides (b,i,k,d)
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides (b,i,j,d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""
    pid_z = tl.program_id(2)   # combined batch * hidden channel
    pid_m = tl.program_id(0)   # tile row (i)
    pid_n = tl.program_id(1)   # tile col (j)

    # De‑compose pid_z --> batch index b and hidden channel d
    b = pid_z // D
    d = pid_z % D

    # 1‑D offsets inside the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_k = offs_k < N

    # Base pointers for this (b, d)
    left_base  = left_ptr  + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base   = out_ptr   + b * stride_ob + d * stride_od

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction dimension k
    for block_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = block_k * BLOCK_K + offs_k
        cur_mask_k = cur_k < N

        # left[i, k]   -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + \
            offs_i[:, None] * stride_li + \
            cur_k[None, :] * stride_lk
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & cur_mask_k[None, :],
                            other=0.0)

        # right[j, k]  -> shape (BLOCK_N, BLOCK_K)
        right_ptrs = right_base + \
            offs_j[:, None] * stride_ri + \
            cur_k[None, :] * stride_rk
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & cur_mask_k[None, :],
                             other=0.0)

        # Accumulate dot product: (M,K) × (K,N) → (M,N)
        acc += tl.dot(left_tile, right_tile.T)

    # Store the resulting block
    out_ptrs = out_base + \
        offs_i[:, None] * stride_oi + \
        offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
              acc,
              mask=mask_i[:, None] & mask_j[None, :])


def tri_mul_forward(left_fp16: torch.Tensor, right_fp16: torch.Tensor, out_fp32: torch.Tensor) -> None:
    """
    Wrapper that launches the Triton kernel.
    left/right: [B, N, N, D] in FP16
    out:       [B, N, N, D] in FP32 (pre‑allocated)
    """
    B, N, _, D = left_fp16.shape
    # Tunable block sizes – empirically good for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,     # rows
        (N + BLOCK_N - 1) // BLOCK_N,     # cols
        B * D                             # combined batch‑hidden dimension
    )

    tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B, N, D,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2), left_fp16.stride(3),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2), right_fp16.stride(3),
        out_fp32.stride(0), out_fp32.stride(1), out_fp32.stride(2), out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (FP32, on the same device).
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # Configuration
    dim = config["dim"]            # input/output channel size
    hidden_dim = config["hidden_dim"]  # internal hidden size

    # ------------------------------------------------------------------
    # 1. First LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]  # [hidden_dim, dim]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        # mask shape: [B, N, N] → [B, N, N, 1] → float for multiplication
        mask_f = mask_tensor.to(dtype=x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate
    # out_gate will be used later (shape matches final hidden tensor)

    # ------------------------------------------------------------------
    # 5. TriMul kernel (the O(N³) part)
    # ------------------------------------------------------------------
    # Cast to FP16 before kernel launch to reduce bandwidth
    left_fp16  = left.to(dtype=torch.float16)
    right_fp16 = right.to(dtype=torch.float16)

    # Allocate output buffer (FP32 as required by subsequent ops)
    out_hidden = torch.empty_like(left_fp16, dtype=torch.float32)  # [B,N,N,hidden_dim]

    tri_mul_forward(left_fp16, right_fp16, out_hidden)

    # ------------------------------------------------------------------
    # 6. Second LayerNorm + out gating
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_hidden = out_hidden * out_gate   # element‑wise gating (broadcasted)

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]   # [dim, hidden_dim]
    out = torch.nn.functional.linear(out_hidden, to_out_weight)   # [B, N, N, dim]

    return out
