"""
TriMul (outgoing) forward kernel for AlphaFold3 style models.

The heavy part of the operation is the 4‑D tensor contraction

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are the projected and gated pairwise
representations of shape (B, N, N, H).  This is equivalent to a batched
matrix multiplication L_h @ R_hᵀ for every hidden channel `h`.  The
kernel below computes this contraction in FP16 (memory‑friendly) while
accumulating in FP32 for numerical stability.  The surrounding LayerNorm,
gates and final linear projection are performed in PyTorch.

The kernel launches a 3‑D grid:
    (num_blocks_M, num_blocks_N, B * H)

* program_id(0) – tile along the i‑axis (M dimension)
* program_id(1) – tile along the j‑axis (N dimension)
* program_id(2) – encodes (batch, hidden) = b * H + h

Each program computes a (BLOCK_M × BLOCK_N) output tile by iterating over
the reduction axis `k` in blocks of size `BLOCK_K`.  The implementation
uses standard Triton best‑practices (masking for out‑of‑bounds tiles,
asynchronous loads, and FP32 accumulation).

The full forward pass:

    1. LayerNorm over the last channel (dim)
    2. Linear projections → left/right (dim → hidden)
    3. Optional masking of pairwise entries
    4. Gating (sigmoid of linear transforms) and element‑wise multiplication
    5. Triton kernel for the O(N³·H) contraction
    6. LayerNorm over hidden dim, out‑gate multiplication
    7. Final linear (hidden → dim)

All operations run on the GPU; only the contraction is written in Triton.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: compute ``out = Σ_k left[...,k,h] * right[...,k,h]``
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)            # tile row (i)
    pid_n = tl.program_id(1)            # tile col (j)
    pid_hb = tl.program_id(2)           # combined batch & hidden index

    b = pid_hb // H                     # batch index
    h = pid_hb % H                      # hidden channel index

    # ------------------------------------------------------------------
    # Compute absolute indices of the tile
    # ------------------------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # Base offsets for the three tensors (including batch & hidden offsets)
    # ------------------------------------------------------------------
    base_l = left_ptr + b * stride_l_b + h * stride_l_h
    base_r = right_ptr + b * stride_r_b + h * stride_r_h
    base_o = out_ptr + b * stride_o_b + h * stride_o_h

    # Offsets that stay constant for a given tile
    i_offset = i[:, None] * stride_l_i          # (BLOCK_M, 1)
    j_offset = j[None, :] * stride_r_i          # (1, BLOCK_N)

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks of BLOCK_K
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = base_l + i_offset + cur_k[None, :] * stride_l_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        right_ptrs = base_r + j_offset + cur_k[:, None] * stride_r_k
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # FP16 * FP16 -> FP32 accumulation
        acc += tl.dot(left, right).to(tl.float32)

    # ------------------------------------------------------------------
    # Write back the result (store in FP16)
    # ------------------------------------------------------------------
    out_ptrs = base_o + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights      : dict of model parameters (see code)
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim] (same dtype as input)
    """
    torch.cuda.set_device("cuda:0")
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    # cast weights to the dtype we will use for the heavy compute (float16)
    proj_dtype = torch.float16
    x_h = x.to(proj_dtype)
    left = torch.nn.functional.linear(x_h, left_proj_w.to(proj_dtype))   # [B,N,N,H]
    right = torch.nn.functional.linear(x_h, right_proj_w.to(proj_dtype))

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is 0/1 float (or bool); make it same dtype as left/right
        mask_f = mask.to(proj_dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear transforms over the original dim)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, left_gate_w.to(proj_dtype))
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, right_gate_w.to(proj_dtype))
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_h, out_gate_w.to(proj_dtype))
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton contraction: out = Σ_k left[...,k,h] * right[...,k,h]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left, dtype=proj_dtype)

    # Ensure tensors are contiguous for correct strides
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Extract strides (in element units, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tunable block sizes (experimentally good for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # ------------------------------------------------------------------
    # Post‑processing (hidden→dim)
    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w.to(proj_dtype),
        bias=to_out_norm_b.to(proj_dtype),
    )

    # Apply out‑gate (same hidden dim)
    out = out * out_gate.to(proj_dtype)

    # Final linear projection hidden → dim
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w.to(proj_dtype))

    # Cast back to original dtype of the input (usually float32)
    out = out.to(inp.dtype)

    return out
