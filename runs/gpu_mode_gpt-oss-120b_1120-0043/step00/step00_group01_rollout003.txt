"""
TriMul (“outgoing”) forward pass (AlphaFold‑3) – Triton + PyTorch hybrid.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension).
2. Linear projections → left, right (shape [B,N,N,H]).
3. (optional) apply the binary mask.
4. Compute three sigmoidal gates (left, right, out) from the
   normalized input.
5. Gate left/right:   left  = left  * left_gate,
                     right = right * right_gate.
6. Core operation: for every hidden channel h,
     out_h = left_h @ right_hᵀ   (N×N matrix‑multiply).
   This is a batched GEMM with batch‑size = B·H.
   The multiplication is executed by a custom Triton kernel
   that works on FP16 inputs while accumulating in FP32.
7. Layer‑norm over the hidden dimension, multiply by out_gate,
   and a final linear projection back to the original channel size.

Only the heavy N³ matrix‑multiply (step 6) is written in Triton;
all other linear / norm / gate ops use PyTorch which already
delivers highly‑optimised kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul C = A @ Bᵀ  (A: (M,K), B: (K,N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,                     # FP16*  (batch, M, K)
    b_ptr,                     # FP16*  (batch, K, N)
    c_ptr,                     # FP32*  (batch, M, N)
    batch, M, N, K,            # int32 runtime arguments
    stride_a_batch, stride_a_m, stride_a_k,   # strides of A
    stride_b_batch, stride_b_k, stride_b_n,   # strides of B
    stride_c_batch, stride_c_m, stride_c_n,   # strides of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)          # batch id
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile col

    # coordinate in the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds for M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    eviction_policy='evict_last')   # FP16 load

        # B tile (already transposed, shape (BLOCK_K, BLOCK_N))
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    eviction_policy='evict_last')   # FP16 load

        # dot‑product (FP16 * FP16) → FP32 accumulation
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]
        - mask         : torch.Tensor [B, N, N]  (may be None)
        - weights      : dict of torch.Tensor, model parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, D] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # weight tensors (all are on the same device, no bias for Linear layers)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (D,),
                     weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right = F.linear(x, right_proj_weight)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional binary mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask[..., None].to(left.dtype)          # [B,N,N,1]
        left  = left  * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core N³ multiplication – Triton batched GEMM
    # ------------------------------------------------------------------
    #   a) reshape to (B*H, M, K) and (B*H, K, N) where M=N=K=seq_len
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    #   b) transpose right so that its first dimension is K (the summed axis)
    right_T = right_perm.permute(0, 1, 3, 2).contiguous()  # [B,H,N,N] → (K,N)

    #   c) flatten batch*hidden
    BH = B * hidden_dim
    left_flat  = left_perm.view(BH, N, N)    # (BH, M, K)
    right_flat = right_T.view(BH, N, N)     # (BH, K, N)

    #   d) FP16 for bandwidth, keep accumulation in FP32 inside the kernel
    left_fp16  = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    #   e) allocate output (FP32)
    out_fp32 = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    #   f) strides (in elements, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_fp16.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_fp16.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_fp32.stride()

    #   g) launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_matmul_kernel[grid](
        left_fp16, right_fp16,
        out_fp32,
        BH, N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 performs best with many warps
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7) Reshape back to [B,N,N,H]
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8) Output layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                # gate after norm
    out = F.linear(out, to_out_weight)  # [B,N,N,D]  (no bias)

    return out
