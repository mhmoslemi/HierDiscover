"""
TriMul “outgoing” implementation with Triton acceleration.

The heavy part of the TriMul forward pass is the tensor contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

which can be expressed as a batch of matrix multiplications
C = A @ Bᵀ where A and B are (N×N) matrices per head (head = batch * hidden_dim).

This kernel collapses the batch and hidden dimension into a single “head” axis,
loads tiles of A (i,k) and Bᵀ (k,j) with block‑wise shared memory, and accumulates the
result in fp32.  The rest of the TriMul pipeline (LayerNorms, Linear projections,
gates) is performed with PyTorch for simplicity.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimension 128 (the only values used in the tests).  Mixed‑precision
is used: the projected tensors are cast to float16 before the Triton kernel,
while accumulation and the final output stay in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_ah, stride_ai, stride_ak,   # A: head, i (row), k (col)
    stride_bh, stride_bk, stride_bj,   # Bᵀ: head, k (row), j (col)
    stride_ch, stride_ci, stride_cj,   # C: head, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for a batch of (N×N) matrices.

    A and B are stored as (head, N, N).  The kernel loops over the `k`
    dimension in blocks, loading tiles of size BLOCK_M×BLOCK_K from A and
    BLOCK_K×BLOCK_N from Bᵀ, and accumulates into a BLOCK_M×BLOCK_N tile of C.
    """
    pid_h = tl.program_id(2)          # head index  (batch * hidden_dim)
    pid_m = tl.program_id(0)          # output row block index
    pid_n = tl.program_id(1)          # output col block index

    # Offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Base pointers for the current head
    a_head = a_ptr + pid_h * stride_ah
    b_head = b_ptr + pid_h * stride_bh
    c_head = c_ptr + pid_h * stride_ch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension `k`
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # Global offsets for this k‑block
        k_offset = kk * BLOCK_K + offs_k
        # Load A tile (M × K)
        a_ptrs = a_head + offs_m[:, None] * stride_ai + k_offset[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=(offs_m < N)[:, None] & (k_offset < N)[None, :],
            other=0.0,
        )
        # Load Bᵀ tile (K × N)
        b_ptrs = b_head + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(
            b_ptrs,
            mask=(k_offset < N)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        # Accumulate the block product
        acc += tl.dot(a, b)

    # Write C tile back to memory
    c_ptrs = c_head + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(
        c_ptrs,
        acc,
        mask=(offs_m < N)[:, None] & (offs_n < N)[None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights_dict : mapping str → torch.Tensor containing all model weights
        - config_dict  : contains keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (dtype float32, same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    (x, mask, weights, config) = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    device = x.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (all are stored in FP32)
    # ------------------------------------------------------------------
    def _w(name):
        return weights[name]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, _w("left_proj.weight"))   # [B, N, N, hidden_dim]
    right = F.linear(x, _w("right_proj.weight"))  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Mask handling
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask shape: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, _w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x, _w("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x, _w("out_gate.weight")))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton kernel
    # ------------------------------------------------------------------
    # Permute to shape (B, hidden_dim, N, N) where the two trailing dims are
    # the matrix dimensions (i, k) for left and (j, k) for right.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Transpose right so that its trailing dims become (k, j)
    right_T = right_perm.transpose(2, 3).contiguous()   # [B, H, N, N] -> (k, j)

    # Cast to float16 to reduce memory bandwidth; computation stays in fp32.
    left_half = left_perm.to(torch.float16)
    right_half_T = right_T.to(torch.float16)

    # Collapse batch and head dimensions -> total_heads = B * hidden_dim
    total_heads = B * hidden_dim
    left_view = left_half.view(total_heads, N, N)
    right_view = right_half_T.view(total_heads, N, N)

    # Allocate output buffer in fp32
    out_view = torch.empty((total_heads, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Triton kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64  # rows per block
    BLOCK_N = 64  # cols per block
    BLOCK_K = 64  # reduction block size

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        total_heads,
    )

    _tri_mul_kernel[grid](
        a_ptr=left_view,
        b_ptr=right_view,
        c_ptr=out_view,
        N=N,
        stride_ah=left_view.stride(0),
        stride_ai=left_view.stride(1),
        stride_ak=left_view.stride(2),
        stride_bh=right_view.stride(0),
        stride_bk=right_view.stride(1),
        stride_bj=right_view.stride(2),
        stride_ch=out_view.stride(0),
        stride_ci=out_view.stride(1),
        stride_cj=out_view.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_view.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Final processing: LayerNorm, gating, linear projection back to dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=_w("to_out_norm.weight"),
        bias=_w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate  # element‑wise gating
    out = F.linear(out, _w("to_out.weight"))  # [B, N, N, dim]

    return out
