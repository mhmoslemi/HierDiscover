@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    **kwargs):
    pid = tl.program_id(axis=0)
    pid_j = tl.program_id(axis=1)
    pid_b = tl.program_id(axis=2)

    # compute starting offsets
    i_start = pid * BLOCK_I
    j_start = pid_j * BLOCK_J
    b = pid_b

    # create range indices
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, BLOCK_H)

    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left slice (i,k,h)
        # shape (BLOCK_I, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + b*stride_left_b + offs_i[:, None, None] * stride_left_i + offs_k[None, :, None] * stride_left_k + offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

        # Load right slice (j,k,h)
        right_ptrs = right_ptr + b*stride_right_b + offs_j[:, None, None] * stride_right_i + offs_k[None, :, None] * stride_right_k + offs_h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

        # compute dot across K for each h:
        # left: (BLOCK_I, BLOCK_K, BLOCK_H); right: (BLOCK_J, BLOCK_K, BLOCK_H)
        # We'll compute for each h via tl.dot( left[:,:,h], right[:,:,h] )
        # Use a for loop over h blocks
        for hh in range(0, BLOCK_H):
            # left slice (BLOCK_I, BLOCK_K)
            left_h = left[:, :, hh]
            right_h = right[:, :, hh]
            # left_h shape (BLOCK_I, BLOCK_K), right_h shape (BLOCK_J, BLOCK_K)
            # Compute dot product sum over K, produce (BLOCK_I, BLOCK_J)
            # tl.dot expects left (M,K), right (N,K)
            acc[:, :, hh] += tl.dot(left_h, right_h)
    # end k loop

    # Store results
    out_ptrs = out_ptr + b*stride_out_b + offs_i[:, None, None] * stride_out_i + offs_j[None, :, None] * stride_out_j + offs_h[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

    stride_right_b, stride_right_i, stride_right_k, stride_right_h,

def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left,right are half dtype
    B, N, N2, H = left.shape
    assert N == N2
    assert H == right.shape[3]
    out = torch.empty_like(left)  # half dtype
    # Strides:
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    # launch kernel
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 4  # must divide hidden dim
    grid = ( (N + BLOCK_I - 1) // BLOCK_I,
             (N + BLOCK_J - 1) // BLOCK_J,
             B )
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    return out

"""
TriMul (outgoing) fused kernel.

The module performs:
    1) LayerNorm on the input tensor (B,N,N,C).
    2) Linear projections to a hidden dimension H (no bias).
    3) Optional masking of the projected tensors.
    4) Sigmoid gating on the projections.
    5) Core O(N³·H) operation:
           out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
       This is implemented as a batched matmul per hidden slice using a
       Triton kernel (the heavy reduction is done on‑GPU, the rest in PyTorch).
    6) LayerNorm on the hidden result, gating, and final linear back to C.

All heavy loads and the N‑K reduction are performed in FP16 for bandwidth,
while the accumulation inside the kernel is FP32 for numerical stability.
The final output is returned as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: computes   out = Σ_k left[..., i, k, h] * right[..., j, k, h]
#   left : (B, N, N, H)  indexed as (b, i, k, h)
#   right: (B, N, N, H)  indexed as (b, j, k, h)
#   out  : (B, N, N, H)  indexed as (b, i, j, h)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *fp16   [B,N,N,H]
    right_ptr,           # *fp16   [B,N,N,H]
    out_ptr,             # *fp16   [B,N,N,H]
    B, N, H,             # int32
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,   # int32
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Batched pairwise multiplication‑sum over the K dimension.

    The kernel tiles the (i,j) output plane with BLOCK_I×BLOCK_J threads.
    For each hidden‑block of size BLOCK_H it accumulates a (BLOCK_I, BLOCK_J)
    matrix using tl.dot over K‑tiles of size BLOCK_K.
    """
    pid_i = tl.program_id(0)     # tile over i
    pid_j = tl.program_id(1)     # tile over j
    pid_b = tl.program_id(2)     # batch index

    # ------------------- coordinate base -------------------
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    b  = pid_b

    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of hidden blocks
    num_h_blocks = tl.cdiv(H, BLOCK_H)

    # -----------------------------------------------------------------
    # Loop over hidden blocks (h dimension)
    # -----------------------------------------------------------------
    for hb in range(0, num_h_blocks):
        h0 = hb * BLOCK_H
        offs_h = h0 + tl.arange(0, BLOCK_H)                # (BLOCK_H,)
        mask_h = offs_h < H

        # accumulator: float32 for safety
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # -----------------------------------------------------------------
        # Loop over the K dimension (the reduction axis)
        # -----------------------------------------------------------------
        for kb in range(0, tl.cdiv(N, BLOCK_K)):
            k0 = kb * BLOCK_K
            offs_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # --------------------- load left (i,k,h) -----------------
            left_ptrs = (
                left_ptr
                + b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            ).to(tl.float32)   # (BLOCK_I, BLOCK_K, BLOCK_H)

            # --------------------- load right (j,k,h) ----------------
            right_ptrs = (
                right_ptr
                + b * stride_right_b
                + offs_j[:, None, None] * stride_right_i   # stride_right_i == stride for the first N axis (j)
                + offs_k[None, :, None] * stride_right_k
                + offs_h[None, None, :] * stride_right_h
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            ).to(tl.float32)   # (BLOCK_J, BLOCK_K, BLOCK_H)

            # --------------------- dot per hidden slice ---------------
            for hh in range(0, BLOCK_H):
                # left[i,k]  shape (BLOCK_I, BLOCK_K)
                # right[j,k] shape (BLOCK_J, BLOCK_K)
                acc[:, :, hh] += tl.dot(left[:, :, hh], right[:, :, hh])

        # --------------------- store the hidden block ---------------
        out_ptrs = (
            out_ptr
            + b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc.to(tl.float16),
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )
# ----------------------------------------------------------------------


def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches ``tri_mul_kernel``.
    Both ``left`` and ``right`` must be fp16 tensors of shape [B,N,N,H].
    Returns an fp16 tensor of the same shape.
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2

    out = torch.empty_like(left)  # fp16 output

    # strides in memory order (b, i, k, h) for left/right, (b, i, j, h) for out
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tunable block sizes – chosen to keep per‑program register usage modest
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 4   # 4 * 32 * 32 = 4096 registers for the accumulator (≈16 KB)

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B,
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32) – the same shape as the input.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------ 1. LayerNorm on the last dimension (C) ------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # B,N,N,dim  (float32)

    # Use FP16 for the heavy path
    x_half = x_norm.half()

    # ------------ 2. Linear projections (no bias) --------------------
    left = F.linear(x_half, weights["left_proj.weight"].half())      # B,N,N,hidden_dim
    right = F.linear(x_half, weights["right_proj.weight"].half())    # B,N,N,hidden_dim

    # ------------ 3. Optional mask -----------------------------------
    if not nomask and mask_tensor is not None:
        # mask shape: [B,N,N]  ->  [B,N,N,1] -> fp16
        mask_half = mask_tensor.unsqueeze(-1).to(dtype=x_half.dtype)
        left = left * mask_half
        right = right * mask_half

    # ------------ 4. Gating (sigmoid) --------------------------------
    left_gate = torch.sigmoid(
        F.linear(x_half, weights["left_gate.weight"].half())
    )
    right_gate = torch.sigmoid(
        F.linear(x_half, weights["right_gate.weight"].half())
    )
    out_gate = torch.sigmoid(
        F.linear(x_half, weights["out_gate.weight"].half())
    ).float()                                 # keep as float32 for later

    left = left * left_gate
    right = right * right_gate

    # ------------ 5. Core reduction (TriMul) -------------------------
    # Compute Σ_k left[i,k,h] * right[j,k,h] using Triton
    out_hidden_fp16 = tri_mul_triton(left, right)   # fp16, shape B,N,N,hidden_dim

    # Convert to FP32 for the remaining layers
    out_hidden = out_hidden_fp16.float()

    # ------------ 6. Output LayerNorm --------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------ 7. Gating after norm --------------------------------
    out_hidden = out_hidden * out_gate  # broadcast over hidden dim

    # ------------ 8. Final linear back to dim -------------------------
    output = F.linear(out_hidden, weights["to_out.weight"])   # B,N,N,dim

    return output
