@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, D,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_D: tl.constexpr, EPS: tl.constexpr
):
    # grid: (B*N*N,) 1D grid for each (b,i,j) triple
    pid = tl.program_id(0)
    # compute indices
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # offsets for start of D dimension:
    x_ptrs = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_ptrs = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # compute mean and var
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)
    for off in range(0, D, BLOCK_D):
        cur_block = tl.minimum(BLOCK_D, D - off)
        # create a range
        d_off = tl.arange(0, BLOCK_D)
        mask = d_off < cur_block
        # load values
        x = tl.load(x_ptrs + (off + d_off) * stride_xd, mask=mask, other=0.0) # shape (BLOCK_D)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_val / D
    var = sum_sq / D - mean * mean
    # compute normalized values and write out
    # second loop to write output
    for off in range(0, D, BLOCK_D):
        cur_block = tl.minimum(BLOCK_D, D - off)
        d_off = tl.arange(0, BLOCK_D)
        mask = d_off < cur_block
        x = tl.load(x_ptrs + (off + d_off) * stride_xd, mask=mask, other=0.0)
        # normalize
        norm = (x - mean) / tl.sqrt(var + EPS)
        # apply weight and bias
        w = tl.load(weight_ptr + (off + d_off) * 1, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + (off + d_off) * 1, mask=mask, other=0.0)
        out = norm * w + b_
        tl.store(out_ptrs + (off + d_off) * stride_od, out, mask=mask)

    right_proj_weight = weights["right_proj.weight"]

    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"] # shape [dim, hidden_dim] (maybe transposed)

    # reshape to (B*hidden_dim, N, N)

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, D,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_D: tl.constexpr, EPS: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # base pointers offsets
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # Initialize accumulators
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    offset = tl.zeros([1], dtype=tl.int32)
    while offset < D:
        cur = tl.minimum(BLOCK_D, D - offset)
        d_range = tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = d_range < cur
        x = tl.load(x_base + (offset + d_range) * stride_xd, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        offset += BLOCK_D

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # Second pass: write normalized output
    offset = tl.zeros([1], dtype=tl.int32)
    while offset < D:
        cur = tl.minimum(BLOCK_D, D - offset)
        d_range = tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = d_range < cur
        x = tl.load(x_base + (offset + d_range) * stride_xd, mask=mask, other=0.0)
        norm = (x - mean) * inv_std
        w = tl.load(weight_ptr + (offset + d_range) * 1, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + (offset + d_range) * 1, mask=mask, other=0.0)
        out = norm * w + b_
        tl.store(out_base + (offset + d_range) * stride_od, out, mask=mask)
        offset += BLOCK_D

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5
    nomask = config.get('nomask', True)

    # Tensors and weight extraction
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    B, N, _, _ = input_tensor.shape

    # ---------- LayerNorm on input ----------
    x_norm = torch.empty_like(input_tensor)
    stride_in = input_tensor.stride()
    stride_out = x_norm.stride()
    grid = (B * N * N,)
    layernorm_kernel[grid](
        input_tensor, x_norm,
        norm_weight, norm_bias,
        B, N, dim,
        stride_in[0], stride_in[1], stride_in[2], stride_in[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_D=64, EPS=eps
    )
    # ---------- Linear projections ----------
    # shape: (B,N,N,hidden_dim)
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # ---------- Apply mask (if present) and gating ----------
    if not nomask:
        # mask is [B, N, N]; convert to float for multiplication
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # (B,N,N,1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ---------- Triangular multiplication (batched GEMM) ----------
    # Reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # Compute out = left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Reshape back to (B,N,N,hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- LayerNorm on out ----------
    out_norm = torch.empty_like(out)
    stride_out_in = out.stride()
    stride_out_out = out_norm.stride()
    grid2 = (B * N * N,)
    layernorm_kernel[grid2](
        out, out_norm,
        to_out_norm_weight, to_out_norm_bias,
        B, N, hidden_dim,
        stride_out_in[0], stride_out_in[1], stride_out_in[2], stride_out_in[3],
        stride_out_out[0], stride_out_out[1], stride_out_out[2], stride_out_out[3],
        BLOCK_D=64, EPS=eps
    )
    # ---------- Apply out_gate ----------
    out_norm = out_norm * out_gate

    # ---------- Final linear ----------
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # (B,N,N,dim)

    return output

"""
TriMul (outgoing) forward pass implemented with a custom Triton LayerNorm kernel.
The algorithm follows the reference PyTorch implementation:

1. Layer‑norm the input tensor over the last dimension (dim).
2. Project the normalized tensor to a hidden dimension with three linear layers:
   left_proj, right_proj and three gates (left_gate, right_gate, out_gate).
   The gates are passed through a sigmoid.
3. Optionally mask the left/right projections, then apply the gates.
4. Compute the triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
   left_d @ right_dᵀ for every hidden slice d.
5. Layer‑norm the result over the hidden dimension.
6. Multiply by the out‑gate and apply the final linear projection back to `dim`.

Only the LayerNorm step is performed with a Triton kernel; the rest
leverages fast PyTorch operations (linear, sigmoid, batched GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,                     # *float32, input [B,N,N,D]
    out_ptr,                   # *float32, output [B,N,N,D]
    weight_ptr,                # *float32, layernorm weight [D]
    bias_ptr,                  # *float32, layernorm bias   [D]
    B, N, D,                   # int32, static sizes
    stride_xb, stride_xi, stride_xj, stride_xd,   # int32 strides of x
    stride_ob, stride_oi, stride_oj, stride_od,   # int32 strides of out
    BLOCK_D: tl.constexpr,     # tile size for the channel dimension
    EPS: tl.constexpr,         # epsilon for variance
):
    pid = tl.program_id(0)                     # one program per (b,i,j) triple
    total = B * N * N
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # Base pointers to the start of the D‑dimensional slice
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    off = tl.int32(0)
    while off < D:
        cur = tl.minimum(BLOCK_D, D - off)               # size of the current block
        d_index = tl.arange(0, BLOCK_D, dtype=tl.int32)  # [0, BLOCK_D)
        mask = d_index < cur
        x = tl.load(x_base + (off + d_index) * stride_xd,
                    mask=mask, other=0.0)                # load a block of D
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
        off = off + BLOCK_D

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + EPS)

    # ---------- second pass: write normalized output ----------
    off = tl.int32(0)
    while off < D:
        cur = tl.minimum(BLOCK_D, D - off)
        d_index = tl.arange(0, BLOCK_D, dtype=tl.int32)
        mask = d_index < cur
        x = tl.load(x_base + (off + d_index) * stride_xd,
                    mask=mask, other=0.0)

        # (x - mean) / sqrt(var+eps)
        norm = (x - mean) * inv_std

        # Load weight & bias for this block
        w = tl.load(weight_ptr + (off + d_index), mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + (off + d_index), mask=mask, other=0.0)

        out_val = norm * w + b_
        tl.store(out_base + (off + d_index) * stride_od,
                 out_val, mask=mask)
        off = off + BLOCK_D


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Extract configuration
    # ------------------------------------------------------------------
    dim = config["dim"]            # input & output channel dimension
    hidden_dim = config["hidden_dim"]
    eps = 1e-5                     # LayerNorm epsilon
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Layer‑norm the input (custom Triton kernel)
    # ------------------------------------------------------------------
    x_norm = torch.empty_like(input_tensor)
    stride_in = input_tensor.stride()
    stride_out = x_norm.stride()
    grid = (B * N * N,)

    layernorm_kernel[grid](
        input_tensor, x_norm,
        weights["norm.weight"], weights["norm.bias"],
        B, N, dim,
        stride_in[0], stride_in[1], stride_in[2], stride_in[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_D=64, EPS=eps
    )

    # ------------------------------------------------------------------
    # Linear projections + gating (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Triangular multiplicative update via batched GEMM:
    #   out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #   = (left_d) @ (right_d).T   for every hidden slice d
    # ------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # batched matrix multiplication (left @ rightᵀ)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Layer‑norm over the hidden dimension (custom Triton kernel)
    # ------------------------------------------------------------------
    out_norm = torch.empty_like(out)
    stride_out_in = out.stride()
    stride_out_out = out_norm.stride()
    grid2 = (B * N * N,)

    layernorm_kernel[grid2](
        out, out_norm,
        weights["to_out_norm.weight"], weights["to_out_norm.bias"],
        B, N, hidden_dim,
        stride_out_in[0], stride_out_in[1], stride_out_in[2], stride_out_in[3],
        stride_out_out[0], stride_out_out[1], stride_out_out[2], stride_out_out[3],
        BLOCK_D=64, EPS=eps
    )

    # ------------------------------------------------------------------
    # Apply out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])   # shape (B,N,N,dim)

    return output
