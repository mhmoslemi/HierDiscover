    left_ptr, right_ptr, out_ptr,

    pid_n = tl.program_id(2)      # block col

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

def matmul_trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BATCH, M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointer offsets for each batch
    left_batch_offset = batch * M * K
    right_batch_offset = batch * N * K
    out_batch_offset = batch * M * N

    # offsets for k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + left_batch_offset + offs_m[:, None] * K + offs_k[None, :]
        a = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        # compute pointer for right transposed: for each row (k) and column (j)
        right_ptrs = right_ptr + right_batch_offset + offs_k[:, None] * 1 + offs_n[None, :] * N
        b = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot: a (M,K) dot b (K,N) => (M,N)
        acc += tl.dot(a, b)

    # write output
    out_ptrs = out_ptr + out_batch_offset + offs_m[:, None] * N + offs_n[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trmul_matmul_kernel(left_ptr, right_ptr, out_ptr,
                        batch, M, N, K,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute strides for contiguous (M,N) matrices
    stride_left_batch = M * K
    stride_right_batch = N * K
    stride_out_batch = M * N

    # Base pointers for this batch
    left_batch_ptr = left_ptr + batch_idx * stride_left_batch
    right_batch_ptr = right_ptr + batch_idx * stride_right_batch
    out_batch_ptr = out_ptr + batch_idx * stride_out_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = kk * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile (M, K)
        left_offs = offs_m[:, None] * K + offs_k[None, :]
        a = tl.load(left_batch_ptr + left_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load right tile transposed (K, N)
        # Right matrix is stored as (j, k) -> offset = j * K + k.
        # We need data as (k, j) for dot product.
        right_offs = offs_k[:, None] * 1 + offs_n[None, :] * N
        b = tl.load(right_batch_ptr + right_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # Write output
    out_offs = offs_m[:, None] * N + offs_n[None, :]
    tl.store(out_batch_ptr + out_offs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

def trmul_batched_matmul(left: torch.Tensor, right: torch.Tensor, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    """
    Batched matmul kernel implementing out_{i,j,d} = sum_k left_{i,k,d} * right_{j,k,d}
    left/right shapes: (B, N, N, H)
    Returns tensor of same shape.
    """
    B, N, _, H = left.shape
    # bring hidden dim to batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch = B * H
    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)
    out_flat = torch.empty_like(left_flat)

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trmul_matmul_kernel[grid](
        left_flat, right_flat, out_flat,
        batch, N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) operator.
    Implements the same computation as the PyTorch reference.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm over the last dimension ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=eps)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ---------- Apply mask (if any) ----------
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- Multiplicative update (triton kernel) ----------
    out = trmul_batched_matmul(left, right,
                               BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # ---------- Output layernorm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=eps)

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul “outgoing” forward pass (AlphaFold‑3 style) with a Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the last dimension (dim).
2. Linear projections → left, right (shape [B, N, N, H]).
3. Optional mask multiplication.
4. Gated projections (sigmoid‑gates) applied to left/right.
5. Batched matrix multiplication per hidden channel:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a custom Triton kernel that treats each
   (batch, hidden) pair as a separate matrix‑multiply problem.
6. Layer‑norm over the hidden dimension, gated by out_gate.
7. Final linear projection back to the original dim.

The heavy N³ work in step‑5 is performed on the GPU by Triton, while the
remaining inexpensive element‑wise ops use regular PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the multiplicative update (batched GEMM)
# ----------------------------------------------------------------------
@triton.jit
def _trmul_matmul_kernel(
    left_ptr,          # *float32  left   [batch, M, K]
    right_ptr,         # *float32  right  [batch, N, K] (stored as (j,k))
    out_ptr,           # *float32  out    [batch, M, N]
    batch, M, N, K,    # runtime integer scalars
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each batch:
        out[m, n] = Σ_k left[m, k] * right[n, k]
    (right is accessed transposed, i.e. right[n,k] == original[j=k, k=n]).
    """
    b_id   = tl.program_id(0)          # batch index
    pid_m  = tl.program_id(1)          # block row
    pid_n  = tl.program_id(2)          # block column

    # ------------------------------------------------------------------
    # offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # base pointers for this batch (contiguous layout)
    # ------------------------------------------------------------------
    # left  shape: (M, K)   strides: (K, 1)
    # right shape: (N, K)   stored as (j, k) → stride_j = K, stride_k = 1
    # out   shape: (M, N)   strides: (N, 1)
    stride_left_batch  = M * K
    stride_right_batch = N * K
    stride_out_batch   = M * N

    left_batch_ptr  = left_ptr  + b_id * stride_left_batch
    right_batch_ptr = right_ptr + b_id * stride_right_batch
    out_batch_ptr   = out_ptr   + b_id * stride_out_batch

    # ------------------------------------------------------------------
    # accumulation
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K in BLOCK_K sized tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = kk * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- left tile (M x K) -----
        # offset = i * K + k
        left_offs = offs_m[:, None] * K + offs_k[None, :]
        a = tl.load(left_batch_ptr + left_offs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- right tile (transposed) (K x N) -----
        # original layout: right[j, k]  offset = j*K + k
        # we need (k, j) → offset = k*1 + j*K
        right_offs = offs_k[:, None] * 1 + offs_n[None, :] * N
        b = tl.load(right_batch_ptr + right_offs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot‑product over the K dimension
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # write result tile
    # ------------------------------------------------------------------
    out_offs = offs_m[:, None] * N + offs_n[None, :]
    tl.store(out_batch_ptr + out_offs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel
# ----------------------------------------------------------------------
def _batched_trmul(left: torch.Tensor,
                   right: torch.Tensor,
                   BLOCK_M: int = 64,
                   BLOCK_N: int = 64,
                   BLOCK_K: int = 32) -> torch.Tensor:
    """
    left, right  : (B, N, N, H)  float32
    returns       : (B, N, N, H)  float32
    """
    B, N, _, H = left.shape
    # bring hidden dim into the batch dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    batch = B * H
    left_flat  = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)
    out_flat   = torch.empty_like(left_flat)

    grid = (batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _trmul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch, N, N, N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]
            mask         : torch.Tensor [B, N, N]  (may be None)
            weights      : dict mapping parameter names to torch.Tensors
            config       : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (skip if nomask is True)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        m = mask.to(left.dtype).unsqueeze(-1)         # [B,N,N,1]
        left  = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Multiplicative update – the N³ part – via Triton
    # ------------------------------------------------------------------
    out = _batched_trmul(left, right,
                         BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)   # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + output gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])      # [B,N,N,dim]

    return out
