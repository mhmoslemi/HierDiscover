import torch, triton, triton.language as tl
import torch.nn.functional as F

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)
    eps = 1e-5

    # Cast to float32 if not already (for simplicity)
    x = x.float()

    # LayerNorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    # use torch.nn.functional.layer_norm
    # input dims: last dimension size dim
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight) # shape bs,N,N,hidden_dim
    right = F.linear(x_norm, right_proj_weight)

    if not nomask:
        # mask shape [bs,N,N]; need unsqueeze
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # TriMul via Triton
    bs, N, _, _ = left.shape
    # allocate out_hidden
    out_hidden = torch.empty((bs, N, N, hidden_dim), device=left.device, dtype=torch.float32)

    # Define kernel launch config (block sizes)
    BLOCK_M = 64  # or 128 maybe depend on N
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             bs*hidden_dim)

    # Compute strides
    # left stride in bytes? triton expects pointer increments as ints in element count (multiply identity?), using pointer arithmetic with strides in elements (not bytes). We'll compute raw strides via left.stride().
    # left shape: [bs, N, N, hidden_dim]
    # Strides: (stride_b, stride_i, stride_k, stride_h)
    # We'll compute integer stride for each dimension.

    left_ptr = left
    right_ptr = right
    out_ptr = out_hidden

    # Launch kernel
    triton_kernel = tri_mul_kernel
    triton_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        bs, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_ptr.stride(0), out_ptr.stride(1), out_ptr.stride(2), out_ptr.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        # maybe pad value for out-of-bound loads (zero)
    )

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_hidden, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply gate
    out_norm = out_norm * out_gate

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out_norm, to_out_weight)  # shape [bs,N,N,dim]
    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for valid i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers offsets for i & j dimensions
    # Compute base pointers for this batch and head
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed shape (BLOCK_K, BLOCK_N)
        # right_T[i_row = k_offset, j_col = j_offset] (k row, j col)
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_i

        # Compute dot: left (M,K) @ right_T (K,N) -> (M,N)
        acc += tl.dot(left, right_T)

    # Store out
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor `x` (last dimension = dim).
2. Linear projections → `left` and `right` of shape [B, N, N, H] (H = hidden_dim).
3. (Optional) apply the binary mask over the pairwise dimensions.
4. Compute three sigmoid gates from the normalized input and multiply them into
   `left`, `right` and the later output.
5. **Heavy part** – a batched matrix multiplication:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is exactly a GEMM `left @ right.T` for each head `h`.  The operation is
   performed by a Triton kernel that tiles the i‑, j‑ and k‑axes (BLOCK_M,
   BLOCK_N, BLOCK_K).  The kernel works on FP16 inputs (to reduce bandwidth) and
   accumulates in FP32.
6. Layer‑norm the hidden result, apply the output gate, and a final linear
   projection back to the original dimension `dim`.

The kernel is launched with a 3‑D grid:
   (⌈N/BLOCK_M⌉, ⌈N/BLOCK_N⌉, B*H)
where each program id processes one (i‑block, j‑block) tile for a specific
(batch, head) pair.

The implementation mixes PyTorch for the cheap per‑element ops (layer‑norm,
linear, sigmoid) with a custom Triton kernel for the O(N³) reduction,
giving a large speed‑up on H100 GPUs.
"""

import math
from typing import Tuple, Dict

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    left_ptr,               # const __global float* (FP16 in practice)
    right_ptr,              # const __global float* (FP16)
    out_ptr,                # __global float* (FP32)
    B, N, H,                # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_i = tl.program_id(0)          # block row   (i dimension)
    pid_j = tl.program_id(1)          # block col   (j dimension)
    pid_bh = tl.program_id(2)         # combined batch * head

    # decode batch and head
    b = pid_bh // H
    h = pid_bh % H

    # offsets for the i and j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the given batch and head
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left[i,k] ----------------------------------------------------
        # shape (BLOCK_M, BLOCK_K)
        left_ptrs = (left_base
                     + offs_i[:, None] * stride_left_i
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- right_T[k,j] -------------------------------------------------
        # we load right as transposed so that we can use tl.dot directly.
        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = (right_base
                      + offs_k[:, None] * stride_right_k
                      + offs_j[None, :] * stride_right_i)
        right_T = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0).to(tl.float32)

        # ----- dot product ---------------------------------------------------
        acc += tl.dot(left, right_T)   # (M,K) @ (K,N) -> (M,N)

    # store result
    out_ptrs = (out_base
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # unpack arguments
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    # configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    ########################################
    # 1. LayerNorm over the last dimension
    ########################################
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    #####################################################
    # 2. Linear projections (no bias) → left / right tensors
    #####################################################
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x_norm, weights["right_proj.weight"])

    ########################################
    # 3. Optional mask
    ########################################
    if not nomask:
        # mask: [B,N,N] → broadcast to channel dim
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    ########################################
    # 4. Gates (sigmoid)
    ########################################
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    #####################################################
    # 5. Heavy part – fused Triton kernel (left @ right.T)
    #####################################################
    # Cast to fp16 for the kernel (memory‑bandwidth friendly)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             device=device,
                             dtype=torch.float32)

    # Tile sizes – tuned for H100 (works for all test N)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_hidden,
        B,
        N,
        hidden_dim,
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        out_hidden.stride(0),
        out_hidden.stride(1),
        out_hidden.stride(2),
        out_hidden.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    #####################################################
    # 6. Post‑processing: norm → output gate → final linear
    #####################################################
    out_hidden_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out_hidden_norm = out_hidden_norm * out_gate

    # final projection back to original dim
    out = F.linear(out_hidden_norm, weights["to_out.weight"])

    return out
