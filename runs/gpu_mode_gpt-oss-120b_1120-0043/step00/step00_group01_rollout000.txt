@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr):
    pid_batch = tl.program_id(2) # batch*hidden_index
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    
    # Decode batch and hidden
    bh = pid_batch
    b = bh // H
    h = bh % H
    
    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out of bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        # left index: [b, i, k, h]
        left_ptrs = left_ptr + b*stride_left_b + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k) + h*stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # right index: [b, j, k, h] but we want B_T index: [b, k, j, h]

        # accumulate

    # store
    out_ptrs = out_ptr + b*stride_out_b + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j) + h*stride_out_h

def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    # left, right shape [B,N,N,H]
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)
    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    # left stride dims correspond to (b, i, k, h)
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * H)

    trimul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    
    b = pid_bh // H
    h = pid_bh % H
    
    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left pointer (b,i,k,h)

import torch
import triton
import triton.language as tl

# Triton kernel implementing the core O(N^3) update:
# out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]   (i.e. left @ right^T per hidden dim)
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul for the TriMul operation."""
    pid_i = tl.program_id(0)      # row tile
    pid_j = tl.program_id(1)      # col tile
    pid_bh = tl.program_id(2)     # combined batch * hidden index

    # decode batch and hidden index
    b = pid_bh // H
    h = pid_bh % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of tiles along k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for tk in range(num_k_tiles):
        k_base = tk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile:   shape (M, K)  -> left[b, i, k, h]
        left_ptrs = left_ptr + b * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk + h * stride_lh
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile (transposed): shape (K, N) -> right[b, k, j, h]
        right_ptrs = right_ptr + b * stride_rb + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj + h * stride_rh
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right)   # (M,K) dot (K,N) => (M,N)

    # write back
    out_ptrs = out_ptr + b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj + h * stride_oh
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around `trimul_kernel` that computes the pairwise multiplication sum
    across the third axis for all batches and hidden dimensions.
    Input/Output shapes: [B, N, N, H]
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # block sizes: tuned for H100, 64 is a good compromise for N up to 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (in elements) of the tensors – they come from the contiguous layout
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (AlphaFold3).
    The heavy N³ contraction is executed by a custom Triton kernel.
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    # Using torch.nn.functional.layer_norm for speed and numerical stability.
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------

    # 3) Optional masking
    # ------------------------------------------------------------------

        mask_ = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ contraction – Triton kernel
    # ------------------------------------------------------------------
    # out shape: [B, N, N, hidden_dim]
    out = trimul_triton(left, right)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the O(N³) “TriMul” contraction
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# i.e. a batched matrix‑multiply left @ rightᵀ for each hidden channel h.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched mat‑mul kernel used inside TriMul.

    The program grid is (num_i_tiles, num_j_tiles, B*H) where the
    third dimension fuses the batch index and the hidden channel.
    """
    pid_i = tl.program_id(0)          # tile along the i‑axis
    pid_j = tl.program_id(1)          # tile along the j‑axis
    pid_bh = tl.program_id(2)         # fused batch * hidden index

    # --------------------------------------------------------------
    # Decode batch (b) and hidden channel (h)
    # --------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 (good precision & no overflow)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles we have to walk over
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for tk in range(num_k_tiles):
        k_base = tk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------------------------------------------------------
        # Load a tile of left: shape (M, K) -> left[b,i,k,h]
        # ---------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
            + h * stride_lh
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ---------------------------------------------------------
        # Load a tile of right transposed: shape (K, N) -> right[b,k,j,h]
        # ---------------------------------------------------------

            right_ptr
            + b * stride_rb
            + offs_k[:, None] * stride_rk
            + offs_j[None, :] * stride_rj
            + h * stride_rh
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # GEMM accumulation
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # --------------------------------------------------------------
    # Write the result back to global memory
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
        + h * stride_oh
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    return out


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module (AlphaFold3).

    * Layer‑norm → linear projections → optional mask → sigmoid gates
    * Core N³ contraction performed by `trimul_triton`
    * Output layer‑norm → gate → final linear projection
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)   # True → no masking needed

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.to(left.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³) contraction – Triton kernel
    # ------------------------------------------------------------------
    # Result has shape [B, N, N, hidden_dim]
    out = trimul_triton(left, right)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, and final projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
