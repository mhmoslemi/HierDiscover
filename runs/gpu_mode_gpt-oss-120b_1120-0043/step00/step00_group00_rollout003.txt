@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          batch, M, N, K,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # load A and B tiles
        a_ptrs = A_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = B_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

"""
custom_kernel
==============

Implements the “outgoing” Triangle‑Multiplicative‑Update (TriMul) operation
used in AlphaFold‑3 and related protein‑structure models.

The implementation follows the reference PyTorch code but accelerates the
core N³ contraction using a fused batched GEMM written in Triton.  The
contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

is equivalent to a batched matrix‑multiply for each hidden channel *d*:

    left_reshape   : (B·H, N, N)   where H = hidden_dim
    right_T_reshape: (B·H, N, N)   (right transposed on its last two axes)

    out_reshape = left_reshape @ right_T_reshape   → (B·H, N, N)

The kernel processes one batch‑channel (i.e. one slice of the (B·H) dimension)
per program instance and uses standard tiling (BLOCK_M × BLOCK_N) with an
inner K‑loop (BLOCK_K).  All arithmetic is performed in FP32 (the inputs are
FP32, the kernel loads them as such and accumulates in FP32).  The rest of the
TriMul pipeline – LayerNorms, linear projections, gating, and final projection
– is carried out with regular PyTorch functions.

The function `custom_kernel` expects a tuple
```
(input_tensor, mask, weights, config)
```
and returns the transformed tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply (A @ B) → C
#   A : (batch, M, K)
#   B : (batch, K, N)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    batch, M, N, K,                    # problem size
    stride_Ab, stride_Am, stride_Ak,   # strides for A
    stride_Bb, stride_Bk, stride_Bn,   # strides for B
    stride_Cb, stride_Cm, stride_Cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile
        a_ptrs = A_ptr + pid_batch * stride_Ab \
                 + offs_m[:, None] * stride_Am \
                 + offs_k[None, :] * stride_Ak
        b_ptrs = B_ptr + pid_batch * stride_Bb \
                 + offs_k[:, None] * stride_Bk \
                 + offs_n[None, :] * stride_Bn

        # Load tiles (FP32)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product over the K tile
        acc += tl.dot(a, b)   # FP32 accumulation

    # Store result
    c_ptrs = C_ptr + pid_batch * stride_Cb \
             + offs_m[:, None] * stride_Cm \
             + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point used by the benchmark / test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] or ``None`` (may be omitted
                         when config["nomask"] == True)
        - weights      : dict of torch.Tensor containing the model parameters
        - config       : dict with keys ``dim`` and ``hidden_dim`` (and
                         optional ``nomask``)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (dtype float32, same device as
        input_tensor).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Layer‑norm on the input (the last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]    # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]   # (hidden_dim, dim)

    left  = F.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left  = left  * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating linear layers + sigmoid
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,hidden_dim]

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core N³ contraction: fused batched GEMM via Triton
    # ------------------------------------------------------------------
    B, N_, _, H = left.shape   # sanity check: N_ == N
    assert N_ == N, "Inconsistent sequence length."

    # Prepare tensors for the batched GEMM:
    #   left  -> (B, H, N, N) -> view as (B*H, N, N) where dim0 = M, dim1 = K
    left_reshaped = left.permute(0, 3, 1, 2).contiguous()
    left_flat = left_reshaped.view(-1, N, N)          # (B*H, M=N, K=N)

    #   right transposed on its last two axes: (B, N, N, H) -> (B, H, K=N, N=j)
    right_T = right.permute(0, 3, 2, 1).contiguous()
    right_flat = right_T.view(-1, N, N)               # (B*H, K=N, N=N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)            # (B*H, N, N)

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    batch_total = left_flat.shape[0]   # B * hidden_dim

    # Strides for the three tensors (batch, row, col)
    stride_Ab, stride_Am, stride_Ak = left_flat.stride()
    stride_Bb, stride_Bk, stride_Bn = right_flat.stride()
    stride_Cb, stride_Cm, stride_Cn = out_flat.stride()

    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_total,
        N, N, N,
        stride_Ab, stride_Am, stride_Ak,
        stride_Bb, stride_Bk, stride_Bn,
        stride_Cb, stride_Cm, stride_Cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm → gate → final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)

    out = out * out_gate

    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)         # (B,N,N,dim)

    return out
