TriMul (outgoing) – Triton‑accelerated implementation

Algorithm
---------
1. Layer‑normalize the input tensor ``x`` (shape [B,N,N,D]).
2. Linear projections for the left/right tensors (to hidden dimension H) and
   three gating tensors (left_gate, right_gate, out_gate).
3. Apply the mask (if given) and the gates:
        left  = left  * left_gate
        right = right * right_gate
4. Core triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This O(N³·H) reduction is performed by a custom Triton kernel.
   Each program instance (a thread) computes a single output element
   (i,j) for a given batch and accumulates a vector of length H.
5. A second LayerNorm, element‑wise multiplication with ``out_gate`` and
   a final linear projection back to the original dimension D.
All operations are gradient‑free (forward only).

# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # const float*  [B,N,N,H]
    right_ptr,              # const float*  [B,N,N,H]
    out_ptr,                # float*        [B,N,N,H]
    B, N, H,                # tensor dimensions (int)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # strides for left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # strides for right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # strides for out
    BLOCK_K: tl.constexpr,                      # tile size along K
    HIDDEN_DIM: tl.constexpr,                   # hidden dimension (H)
    NUM_K_TILES: tl.constexpr,                  # ceil(N / BLOCK_K)
):
    """Each program instance computes out[b,i,j,:] by summing over k."""
    pid = tl.program_id(0)               # linear index over (B, N, N)

    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # guard for any stray threads (should never fire)
    valid = (b < B) & (i < N) & (j < N)

    # accumulator for the hidden dimension (float32 for precision)
    acc = tl.zeros((HIDDEN_DIM,), dtype=tl.float32)

    # range over hidden dimension
    h_range = tl.arange(0, HIDDEN_DIM)

    # ------------------------------------------------------------------
    # Loop over K in tiles of size BLOCK_K.
    # In each tile we load a (BLOCK_K, H) slice from left[i,:,:] and
    # right[j,:,:] and accumulate the dot‑product across K.
    # ------------------------------------------------------------------
    for tile in range(NUM_K_TILES):
        k_base = tile * BLOCK_K + tl.arange(0, BLOCK_K)             # (BLOCK_K,)
        mask_k = k_base < N

        # left[b,i,k,:]  -> shape (BLOCK_K, H)
        left_off = (
            b * stride_l_b +
            i * stride_l_i +
            k_base[:, None] * stride_l_k +
            h_range[None, :] * stride_l_h
        )
        left = tl.load(left_ptr + left_off, mask=mask_k[:, None], other=0.0)

        # right[b,j,k,:] -> shape (BLOCK_K, H)
        right_off = (
            b * stride_r_b +
            j * stride_r_j +
            k_base[:, None] * stride_r_k +
            h_range[None, :] * stride_r_h
        )
        right = tl.load(right_ptr + right_off, mask=mask_k[:, None], other=0.0)

        # element‑wise product and reduction over the K‑tile
        prod = left * right                     # (BLOCK_K, H)
        acc += tl.sum(prod, axis=0)             # (H,)

    # ------------------------------------------------------------------
    # Write the result back to memory.
    # ------------------------------------------------------------------
    out_off = (
        b * stride_o_b +
        i * stride_o_i +
        j * stride_o_j +
        h_range * stride_o_h
    )
    tl.store(out_ptr + out_off, acc, mask=valid)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor of shape [B, N, N, D]
        mask    – torch.Tensor of shape [B, N, N]  (or None)
        weights – dict with all linear / layer‑norm parameters
        config  – dict with ``dim`` (=D) and ``hidden_dim`` (=H)

    Returns
    -------
    torch.Tensor of shape [B, N, N, D]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Layer‑norm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (shape [B,N,N,1]) – broadcast over the hidden dim
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating tensors (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triangular multiplicative update – Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # strides are in element counts (not bytes)
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()

    BLOCK_K = 32                                            # tunable
    NUM_K_TILES = (N + BLOCK_K - 1) // BLOCK_K

    total_programs = B * N * N
    trimul_kernel[(total_programs,)](
        left, right, out,
        B, N, hidden_dim,
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
        NUM_K_TILES=NUM_K_TILES,
        num_warps=4,          # H100 has many warps – 4 is a safe default
    )

    # ------------------------------------------------------------------
    # Second LayerNorm, gating and final linear projection back to D
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
