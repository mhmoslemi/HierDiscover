def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # Determine dtype: Use input dtype (float32). We'll cast to float16 after LayerNorm to reduce cost.
    # LayerNorm
    eps = 1e-5
    # Using torch.nn.functional.layer_norm with weight,bias
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Apply mask if provided (if mask is not None)
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # Apply gates
    left = left * left_gate
    right = right * right_gate
    # Convert to half for core multiplication
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare for batched matmul
    B, N, _, _ = left.shape  # left shape [B,N,N,H]
    # Permute for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # shape [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right for matmul
    right_t = right_perm.transpose(2, 3).contiguous()  # shape [B, H, N, N] where rows=K, cols=N
    # Flatten batch*H
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_t.view(batch_h, N, N)
    # Compute out via batched matmul kernel
    out_flat = batched_matmul_triton(left_flat, right_flat)  # shape (batch_h,N,N) half
    # Reshape back to [B, H, N, N]
    out_perm = out_flat.view(B, hidden_dim, N, N)
    # Permute back to [B,N,N,H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)  # Convert back to float32 for subsequent layers

    # to_out_norm (LayerNorm) over hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply out_gate (which is currently float32)
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward pass implemented with a fused Triton batched‑matmul.
The heavy part of the computation

    out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]

is expressed as a batch of H independent N×N matrix multiplications:
    left[d] (N×N)  ×  right[d].T (N×N)  →  out[d] (N×N)

We fuse these H·B matmuls with a single Triton kernel that processes one
matrix pair per batch element (batch = B * hidden_dim).  All other
operations (LayerNorm, Linear projections, gating, final projection) are
performed with PyTorch, but the core contraction is fully GPU‑accelerated.
The kernel works in FP16 with FP32 accumulation and produces the result
in FP16, which is then cast back to FP32 for the remaining layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B.T) for many batches
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_am, stride_ak, stride_batch_a,   # A strides (batch, row, col)
    stride_bk, stride_bn, stride_batch_b,   # B strides (batch, row(K), col(N))
    stride_cm, stride_cn, stride_batch_c,   # C strides (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C = A @ B.T for many independent batches.
    A shape: (batch, M, K)
    B shape: (batch, K, N)   (already transposed from the original right tensor)
    C shape: (batch, M, N)
    """
    pid_batch = tl.program_id(2)          # batch index
    pid_m = tl.program_id(0)              # block row index
    pid_n = tl.program_id(1)              # block column index

    # Offsets inside a block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for the start of the block (including batch offset)
    a_base = a_ptr + pid_batch * stride_batch_a
    b_base = b_ptr + pid_batch * stride_batch_b
    c_base = c_ptr + pid_batch * stride_batch_c

    # Pointers to the current tile
    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Masks for out‑of‑bounds loads
    a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
    b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)   # (BM, BK)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)   # (BK, BN)
        acc += tl.dot(a, b)                                          # (BM, BN)

        # Advance pointers to next K‑block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write‑back with output mask
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    A : (batch, M, K)    fp16, contiguous
    B : (batch, K, N)    fp16, contiguous (already transposed)
    Returns C : (batch, M, N) fp16
    """
    assert A.dtype == torch.float16 and B.dtype == torch.float16
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=torch.float16, device=A.device)

    # Strides in number of elements (not bytes)
    stride_am, stride_ak, stride_batch_a = A.stride()
    stride_bk, stride_bn, stride_batch_b = B.stride()
    stride_cm, stride_cn, stride_batch_c = C.stride()

    # Tunable block sizes – 64 works well on H100 for these sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(M, BLOCK_M),   # program_id(0)
        triton.cdiv(N, BLOCK_N),   # program_id(1)
        batch,                     # program_id(2) – one batch per hidden slice
    )

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_am, stride_ak, stride_batch_a,
        stride_bk, stride_bn, stride_batch_b,
        stride_cm, stride_cn, stride_batch_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask         : [B, N, N]        (bool/int, 1 = keep, 0 = drop)
        - weights      : dict of model parameters (float32)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    out : torch.Tensor
        Processed tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5                     # same eps as nn.LayerNorm

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)        # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gate computations
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction: batch of H independent N×N matmuls
    # ------------------------------------------------------------------
    # Use half precision for the heavy compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Rearrange to (B, H, N, N) → view as (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose right so we can do A @ B.T  (right now is [B,H,N,N] → [B,H,N,N] with K‑axis first)
    right_t = right_perm.transpose(2, 3).contiguous()   # [B, H, N, N] where dim0 is K

    # Flatten batch * hidden_dim
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)           # (B*H, N, N)
    right_flat = right_t.view(batch_h, N, N)            # (B*H, N, N)

    # Perform batched matmul on the GPU via Triton
    out_flat = _batched_matmul_triton(left_flat, right_flat)  # (B*H, N, N) fp16

    # Reshape back to [B, H, N, N] → [B, N, N, H]
    out_perm = out_flat.view(B, hidden_dim, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]

    # Back to FP32 for the remaining layers
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
