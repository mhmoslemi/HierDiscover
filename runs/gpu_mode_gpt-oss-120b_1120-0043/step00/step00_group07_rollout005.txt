"""
TriMul (outgoing) implementation for AlphaFold3 style pairwise updates.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension.
2. Project the normalized tensor to a hidden dimension `h` using three linear
   layers (left, right, gates).  Apply the optional pairwise mask and element‑wise
   sigmoid gates.
3. Compute the cubic “triangle multiplicative” update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of independent matrix multiplications (one per hidden channel
   `d`).  The heavy reduction over `k` is performed by a custom Triton kernel
   that tiles the i/j dimensions (`BLOCK_M`, `BLOCK_N`) and the reduction axis
   (`BLOCK_K`).  The kernel processes a single hidden channel per program
   instance (batch * hidden_dim programs are launched, i.e. a 3‑D grid).
4. Layer‑norm the result over the hidden dimension, apply the `out_gate`
   (computed from the original normalized tensor), and finally project back to
   the original dimension with a linear layer.

The implementation fuses the O(N³·H) reduction in Triton while keeping the
lighter linear and normalization work in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: one matrix multiplication per hidden channel.
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                # float32 *
    B, N, H,                # int32 scalars
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
    Each program instance processes a single (b, h) pair and a tile
    of (i, j).  `BLOCK_M`, `BLOCK_N`, `BLOCK_K` are compile‑time tile sizes.
    """
    pid_z = tl.program_id(2)                         # combined batch & hidden index
    pid_b = pid_z // H                                # batch index
    pid_h = pid_z % H                                 # hidden channel index

    pid_i = tl.program_id(0)                          # i‑tile
    pid_j = tl.program_id(1)                          # j‑tile

    # ------------------- compute base offsets -------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers to the start of the (b, h) slice for A and B
    left_base  = left_ptr  + pid_b * stride_left_batch  + pid_h * stride_left_h
    right_base = right_ptr + pid_b * stride_right_batch + pid_h * stride_right_h
    out_base   = out_ptr   + pid_b * stride_out_batch   + pid_h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(0, num_k):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left tile  : shape (BLOCK_M, BLOCK_K)  (row‑major)
        # Load right tile : shape (BLOCK_K, BLOCK_N)  (row‑major = transposed)
        # ------------------------------------------------------------------
        a_ptrs = left_base  + offs_i[:, None] * stride_left_i \
                               + offs_k[None, :] * stride_left_k
        b_ptrs = right_base + offs_k[:, None] * stride_right_k \
                               + offs_j[None, :] * stride_right_i  # note transposition

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)  # fp16
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)  # fp16

        # Perform the matrix‑multiply for this tile (fp16→fp32 accumulation)
        acc += tl.dot(a, b)

    # ----------------------- store result -----------------------
    out_ptrs = out_base + offs_i[:, None] * stride_out_i \
                         + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Entry‑point: custom_kernel
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Args:
        data: Tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask:  torch.Tensor of shape [B, N, N]  (may be None)
            - weights: dict with the model parameters (see docstring)
            - config: dict containing at least "dim", "hidden_dim", optional "nomask"

    Returns:
        output: torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # input feature dimension
    hidden_dim = config["hidden_dim"] # projection dimension
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Helper: optional mask handling
    # ------------------------------------------------------------------
    nomask = config.get("nomask", True)
    if nomask or mask is None:
        mask_exp = 1.0
    else:
        # mask is bool/float, expand to last dim
        mask_exp = mask.unsqueeze(-1).to(dtype)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections + gates
    # ------------------------------------------------------------------
    # Linear proj: dim -> hidden_dim (no bias)
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # Apply mask (if any) BEFORE gating, exactly as in reference code
    left  = left * mask_exp
    right = right * mask_exp

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Triangle Mul via Triton (heavy O(N³·H) part)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Cast to fp16 for the kernel (GPU memory bandwidth)
    left_fp  = left.to(torch.float16)
    right_fp = right.to(torch.float16)

    # Allocate output tensor (FP32, needed for subsequent LayerNorm)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tile sizes – chosen for good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (in elements) for the three tensors
    stride_left  = left_fp.stride()   # (B, i, k, h)
    stride_right = right_fp.stride()
    stride_out   = out.stride()

    # Flatten (batch, hidden) into a single program‑id dimension (required < 3D grid)
    grid_x = (N + BLOCK_M - 1) // BLOCK_M
    grid_y = (N + BLOCK_N - 1) // BLOCK_N
    grid_z = B * hidden_dim

    # Launch kernel
    tri_mul_kernel[(grid_x, grid_y, grid_z)](
        left_fp,
        right_fp,
        out,
        B, N, hidden_dim,
        stride_left[0],  stride_left[1],  stride_left[2],  stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0],   stride_out[1],   stride_out[2],   stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4  # small number of warps works well for this matmul shape
    )

    # ------------------------------------------------------------------
    # 4. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # shape [dim, hidden_dim]

    # LayerNorm over the hidden dimension
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Apply the output gate (broadcasted over the hidden dimension)
    out = out * out_gate

    # Final linear projection back to `dim`
    output = F.linear(out, to_out_weight)   # bias is None (as in reference)

    return output
```
