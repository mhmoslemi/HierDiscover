@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute global batch idx:
    total_blocks_m = tl.cdiv(M, BLOCK_M)
    total_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (total_blocks_m * total_blocks_n)
    block_idx = pid % (total_blocks_m * total_blocks_n)
    pid_m = block_idx // total_blocks_n
    pid_n = block_idx % total_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Compute offsets for the block
    a_ptrs = a_ptr + batch_idx * batch_stride_a + (offs_m[:, None] * stride_am) + (tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_ptr + batch_idx * batch_stride_b + (tl.arange(0, BLOCK_K)[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B tiles
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (tl.arange(0, BLOCK_K)[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        # compute
        acc += tl.dot(a, b)
        # advance pointers

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    eps = 1e-5

    # LayerNorm on dim
    # Using F.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask handling
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.to(input_tensor.dtype).unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_
        right = right * mask_
    else:
        # no mask, nothing to do
        pass

    # Gates
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # Convert to half for matmul kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Permute to (B, hidden_dim, N, N) and final shapes
    # left: (B, N, N, hidden_dim) -> (B, hidden_dim, N, N)
    left_mat = left_h.permute(0, 3, 1, 2).contiguous()
    # right: (B, N, N, hidden_dim) -> (B, hidden_dim, N, N) where we need (B, hidden_dim, K, J) (k as row, j as col)
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()

    # batch dimension = B * hidden_dim
    batch = B * hidden_dim
    a = left_mat.view(batch, N, N)
    b = right_mat.view(batch, N, N)
    c = torch.empty_like(a)  # float16

    # Launch Triton kernel
    import triton
    import triton.language as tl

    @triton.jit
    def batched_matmul_kernel(
        a_ptr, b_ptr, c_ptr,
        M, N, K,
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid = tl.program_id(0)
        total_blocks_m = tl.cdiv(M, BLOCK_M)
        total_blocks_n = tl.cdiv(N, BLOCK_N)
        batch_idx = pid // (total_blocks_m * total_blocks_n)
        block_idx = pid % (total_blocks_m * total_blocks_n)
        pid_m = block_idx // total_blocks_n
        pid_n = block_idx % total_blocks_n

        # offsets within the block
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # base pointers for the current batch
        a_batch = a_ptr + batch_idx * batch_stride_a
        b_batch = b_ptr + batch_idx * batch_stride_b
        c_batch = c_ptr + batch_idx * batch_stride_c

        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension
        for k in range(0, tl.cdiv(K, BLOCK_K)):
            cur_k = k * BLOCK_K
            # Load A tile
            a = tl.load(
                a_batch + offs_m[:, None] * stride_am + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak,
                mask=(offs_m[:, None] < M) & (cur_k + tl.arange(0, BLOCK_K)[None, :] < K),
                other=0.0,
            )
            # Load B tile (note B is KxN)
            b = tl.load(
                b_batch + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn,
                mask=(cur_k + tl.arange(0, BLOCK_K)[:, None] < K) & (offs_n[None, :] < N),
                other=0.0,
            )
            acc += tl.dot(a, b)

        # Write back result
        c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
        tl.store(
            c,
            acc.to(tl.float16),
            mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
        )

    # Determine block sizes (tuned)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch * ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N),)

    batched_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=N,
        N=N,
        K=N,
        stride_am=a.stride(1),
        stride_ak=a.stride(2),
        stride_bk=b.stride(1),
        stride_bn=b.stride(2),
        stride_cm=c.stride(1),
        stride_cn=c.stride(2),
        batch_stride_a=a.stride(0),
        batch_stride_b=b.stride(0),
        batch_stride_c=c.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Gather output and finish rest of ops
    out_h = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out_fp32 = out_h.to(torch.float32)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Gate
    out = out_norm * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – fast forward implementation.

The operator receives a 4‑D pair representation `x` of shape
[batch, N, N, dim] and a (optional) mask [batch, N, N].
It performs:

1. Layer‑norm over the last dimension (dim).
2. Two linear projections (left/right) from dim → hidden_dim.
3. Optional masking of the projected tensors.
4. Three sigmoid gates (left, right, out) computed from the
   original normalized tensor.
5. Gated elementwise multiplication of the left/right tensors.
6. **Batched matrix‑multiplication**:
   for each hidden channel `h` we compute  L_h @ (R_h)ᵀ,
   i.e. out[b,i,j,h] = Σₖ left[b,i,k,h]·right[b,j,k,h].
   This O(N³) step is executed by a custom Triton kernel that
   treats (batch × hidden_dim) as the outer batch dimension.
7. Layer‑norm over the hidden dimension, gating with out_gate,
   and a final linear projection back to `dim`.

All heavy arithmetic (step 6) is performed in FP16 for bandwidth
while accumulating in FP32 to keep numerical accuracy. The rest
of the pipeline runs in FP32 using native PyTorch ops.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched (M×K)·(K×N) → (M×N) for (batch*hidden) independent matrices."""
    pid = tl.program_id(0)

    # total number of blocks per matrix dimension
    total_blocks_m = tl.cdiv(M, BLOCK_M)
    total_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode program id into batch, block‑row, block‑col
    batch_idx = pid // (total_blocks_m * total_blocks_n)
    block_idx = pid % (total_blocks_m * total_blocks_n)
    pid_m = block_idx // total_blocks_n
    pid_n = block_idx % total_blocks_n

    # block‑level offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for the current batch element
    a_batch = a_ptr + batch_idx * batch_stride_a
    b_batch = b_ptr + batch_idx * batch_stride_b
    c_batch = c_ptr + batch_idx * batch_stride_c

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction dimension K
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = kk * BLOCK_K

        # load a tile of A (M×K)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_am
            + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak,
            mask=(offs_m[:, None] < M)
                 & (cur_k + tl.arange(0, BLOCK_K)[None, :] < K),
            other=0.0,
        )
        # load a tile of B (K×N)
        b = tl.load(
            b_batch
            + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk
            + offs_n[None, :] * stride_bn,
            mask=(cur_k + tl.arange(0, BLOCK_K)[:, None] < K)
                 & (offs_n[None, :] < N),
            other=0.0,
        )
        # FP32 dot product
        acc += tl.dot(a, b)

    # write back the result tile
    c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c,
        acc,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )


def custom_kernel(data):
    """
    Forward‑only TriMul (outgoing) using a custom Triton kernel for the
    O(N³) multiplication step.
    """
    # unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = input_tensor.shape   # input shape [B,N,N,dim]

    # -----------------------------------------------------------------
    # 1) LayerNorm over the last dim (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections to hidden_dim (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    if mask is not None and not nomask:
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating scalars (sigmoid)
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # -----------------------------------------------------------------
    # 5) Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6) Batched matmul (kernel): out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    #    Cast to FP16 for bandwidth, accumulate in FP32.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    #   reshape to (B*hidden_dim, N, N) with appropriate transposes
    left_mat = left_h.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)
    right_mat = right_h.permute(0, 3, 2, 1).contiguous()        # (B, H, N, N) where K is first N

    batch = B * hidden_dim
    a = left_mat.view(batch, N, N)      # (batch, M=N, K=N)   – FP16
    b = right_mat.view(batch, N, N)     # (batch, K=N, N)     – FP16
    c = torch.empty((batch, N, N), dtype=torch.float32, device=device)  # result FP32

    # launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch * ((N + BLOCK_M - 1) // BLOCK_M) * ((N + BLOCK_N - 1) // BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=N,
        N=N,
        K=N,
        stride_am=a.stride(1),
        stride_ak=a.stride(2),
        stride_bk=b.stride(1),
        stride_bn=b.stride(2),
        stride_cm=c.stride(1),
        stride_cn=c.stride(2),
        batch_stride_a=a.stride(0),
        batch_stride_b=b.stride(0),
        batch_stride_c=c.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,hidden_dim)
    out = c.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # FP32

    # -----------------------------------------------------------------
    # 7) Output LayerNorm (over hidden_dim) + out_gate + final linear
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
