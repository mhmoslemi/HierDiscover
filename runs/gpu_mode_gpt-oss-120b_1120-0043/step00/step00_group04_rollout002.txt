```python
import torch
import triton
import triton.language as tl

@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, hidden_dim,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # compute indices
    tmp = pid
    b = tmp // (N * N)
    tmp = tmp % (N * N)
    i = tmp // N
    j = tmp % N

    # load mask scalar
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_offset)
    # mask_val is broadcasted across hidden dim

    # loop over hidden dimension
    for off in range(0, hidden_dim, BLOCK_H):
        cur_off = off + tl.arange(0, BLOCK_H)
        # mask for overflows
        hmask = cur_off < hidden_dim

        # compute linear offsets
        proj_offset = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + cur_off * stride_proj_h
        gate_offset = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + cur_off * stride_gate_h

        proj = tl.load(proj_ptr + proj_offset, mask=hmask, other=0.0).to(tl.float32)
        gate = tl.load(gate_ptr + gate_offset, mask=hmask, other=0.0).to(tl.float32)

        # sigmoid gate
        gate = 1.0 / (1.0 + tl.exp(-gate))

        # apply mask and gate
        out = proj * gate * mask_val

        tl.store(out_ptr + proj_offset, out, mask=hmask)
```

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # (We may ignore config["dim"] match with dim)
    # Compute layernorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"])
    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate_pre = torch.nn.functional.linear(x, weights["left_gate.weight"])
    right_gate_pre = torch.nn.functional.linear(x, weights["right_gate.weight"])

    # mask handling
    if nomask:
        mask = torch.ones(bs, N, N, dtype=input_tensor.dtype, device=input_tensor.device)
    else:
        mask = mask_tensor.to(input_tensor.dtype)
    # Ensure mask contiguous
    mask = mask.contiguous()
    # Prepare output tensors for fused gating
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)
    # Launch fused kernel for left
    total_positions = bs * N * N
    BLOCK_H = 32
    grid = (total_positions,)
    fused_proj_gate_mask_kernel[grid](
        left_proj,
        left_gate_pre,
        mask,
        left,
        bs,
        N,
        hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate_pre.stride(0), left_gate_pre.stride(1), left_gate_pre.stride(2), left_gate_pre.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_H=BLOCK_H,
    )
    # Launch fused kernel for right
    fused_proj_gate_mask_kernel[grid](
        right_proj,
        right_gate_pre,
        mask,
        right,
        bs,
        N,
        hidden_dim,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate_pre.stride(0), right_gate_pre.stride(1), right_gate_pre.stride(2), right_gate_pre.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_H=BLOCK_H,
    )
    # Compute out = left @ right^T per hidden dim
    # Permute to [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Matrix multiplication
    out_mat = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    # out shape [B, hidden_dim, N, N] -> permute back
    out = out_mat.permute(0, 2, 3, 1).contiguous()
    # Output layernorm
    out_norm = torch.nn.functional.layer_norm(out,
                                               (hidden_dim,),
                                               weight=weights["to_out_norm.weight"],
                                               bias=weights["to_out_norm.bias"])
    # Compute out_gate (pre sigmoid)
    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"])
    out_gate = out_gate_pre.sigmoid()
    # Apply out_gate
    out_norm = out_norm * out_gate
    # Final linear projection to original dim
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return out_final

import torch
import triton
import triton.language as tl

# Triton kernel: fused projection, gate sigmoid, mask multiplication.
@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, hidden_dim,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr):
    """Fuse linear projection, gate sigmoid and optional mask for a single (b,i,j) position.
    Each program handles one (batch, row, col) coordinate and loops over hidden_dim.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from linear pid
    tmp = pid
    b = tmp // (N * N)
    tmp = tmp % (N * N)
    i = tmp // N
    j = tmp % N

    # load scalar mask (or 1.0 if mask is all ones)
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_off)

    # iterate over hidden dimension in blocks
    for off in range(0, hidden_dim, BLOCK_H):
        idx = off + tl.arange(0, BLOCK_H)   # hidden indices
        mask_h = idx < hidden_dim            # out-of-range guard

        # offsets for proj and gate
        proj_off = b * stride_proj_b + i * stride_proj_i + j * stride_proj_j + idx * stride_proj_h
        gate_off = b * stride_gate_b + i * stride_gate_i + j * stride_gate_j + idx * stride_gate_h

        proj = tl.load(proj_ptr + proj_off, mask=mask_h, other=0.0).to(tl.float32)
        gate = tl.load(gate_ptr + gate_off, mask=mask_h, other=0.0).to(tl.float32)

        # sigmoid
        gate = 1.0 / (1.0 + tl.exp(-gate))

        # fused elementwise: proj * gate * mask
        out = proj * gate * mask_val

        tl.store(out_ptr + proj_off, out, mask=mask_h)

"""
TriMul (outgoing) forward pass – Triton‑fused elementwise part.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the feature dimension `dim`.
2. Compute four linear maps on the normalized tensor:
      * left_proj   = x @ W_left_proj   (shape [..., hidden_dim])
      * right_proj  = x @ W_right_proj
      * left_gate   = sigmoid( x @ W_left_gate )
      * right_gate  = sigmoid( x @ W_right_gate )
   The three operations above are fused in a single Triton kernel
   (projection, sigmoid‑gate, optional mask) to avoid an extra
   materialisation of the mask and to keep the elementwise work on‑GPU.
3. Apply mask (binary 0/1) – when ``nomask`` is true a tensor of ones is used.
4. Fuse the projection and gate:
        left  = left_proj  * left_gate  * mask
        right = right_proj * right_gate * mask
5. Compute the pairwise multiplicative update:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   The sum over `k` is a batched matrix‑multiply for every hidden
   channel (equivalently `out = left @ rightᵀ` with hidden_dim as a
   batch dimension).  This uses torch’s highly‑optimised cuBLAS path.
6. Layer‑norm the result over `hidden_dim`, multiply by an output gate
   (`out_gate = sigmoid(x @ W_out_gate)`) and finally project back to
   the original feature dimension (`to_out`).
7. Return a tensor of shape `[B, N, N, dim]`.

Only the elementwise projection/gate/mask step is written in Triton
(`fused_proj_gate_mask_kernel`); the heavy matrix‑multiply is performed
by torch which already provides a fast implementation on H100.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def fused_proj_gate_mask_kernel(
    proj_ptr,     # pointer to projection tensor   [B,N,N,hidden_dim]
    gate_ptr,     # pointer to pre‑sigmoid gate   [B,N,N,hidden_dim]
    mask_ptr,     # pointer to mask               [B,N,N]
    out_ptr,      # output pointer                [B,N,N,hidden_dim]
    B, N, hidden_dim,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr,
):
    """Fuse projection, sigmoid gate and mask for one (b,i,j) coordinate.

    Each program processes a single (batch, row, col) location and loops
    over the hidden dimension in blocks of size ``BLOCK_H``.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # ---- decode linear pid to (b,i,j) ---------------------------------
    tmp = pid
    b = tmp // (N * N)
    tmp = tmp % (N * N)
    i = tmp // N
    j = tmp % N

    # ---- load scalar mask (broadcasted to hidden_dim) ------------------
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_off)

    # ---- iterate over hidden dimension ---------------------------------
    for off in range(0, hidden_dim, BLOCK_H):
        hid = off + tl.arange(0, BLOCK_H)                # hidden indices
        hid_mask = hid < hidden_dim                       # guard for tail

        # offsets within projection and gate tensors
        proj_off = (
            b * stride_proj_b
            + i * stride_proj_i
            + j * stride_proj_j
            + hid * stride_proj_h
        )
        gate_off = (
            b * stride_gate_b
            + i * stride_gate_i
            + j * stride_gate_j
            + hid * stride_gate_h
        )

        # load values (float32 for arithmetic)
        proj = tl.load(proj_ptr + proj_off, mask=hid_mask, other=0.0).to(tl.float32)
        gate = tl.load(gate_ptr + gate_off, mask=hid_mask, other=0.0).to(tl.float32)

        # sigmoid gate
        gate = 1.0 / (1.0 + tl.exp(-gate))

        # fused element‑wise: proj * gate * mask
        out = proj * gate * mask_val

        # store result
        tl.store(out_ptr + proj_off, out, mask=hid_mask)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear maps (project & gate pre‑sigmoid)
    # ------------------------------------------------------------------
    # NOTE: all Linear layers have bias=False in the reference model.
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate_pre = torch.nn.functional.linear(x, weights["left_gate.weight"])
    right_gate_pre = torch.nn.functional.linear(x, weights["right_gate.weight"])

    # ------------------------------------------------------------------
    # 3. Prepare mask (float32)
    # ------------------------------------------------------------------
    if nomask:
        mask = torch.ones(bs, N, N, dtype=input_tensor.dtype, device=input_tensor.device)
    else:
        mask = mask_tensor.to(input_tensor.dtype)
    mask = mask.contiguous()  # ensure contiguous layout for Triton

    # ------------------------------------------------------------------
    # 4. Fuse projection + gate + mask using Triton
    # ------------------------------------------------------------------
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_positions = bs * N * N
    BLOCK_H = 32  # fits hidden_dim ≤ 384 comfortably

    # launch for the left branch
    fused_proj_gate_mask_kernel[(total_positions,)](
        left_proj,
        left_gate_pre,
        mask,
        left,
        bs,
        N,
        hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate_pre.stride(0), left_gate_pre.stride(1), left_gate_pre.stride(2), left_gate_pre.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_H=BLOCK_H,
    )

    # launch for the right branch
    fused_proj_gate_mask_kernel[(total_positions,)](
        right_proj,
        right_gate_pre,
        mask,
        right,
        bs,
        N,
        hidden_dim,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate_pre.stride(0), right_gate_pre.stride(1), right_gate_pre.stride(2), right_gate_pre.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 5. Core multiplicative update: out = left @ rightᵀ (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, hidden_dim, N, N] for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # batched matrix multiplication across the sequence dimension
    out_mat = torch.matmul(left_perm, right_perm.transpose(-2, -1))  # [B, hidden_dim, N, N]
    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑processing (layer‑norm, out‑gate, final linear)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"])
    out_gate = out_gate_pre.sigmoid()

    out_norm = out_norm * out_gate  # broadcast over hidden_dim

    # final projection back to original feature dimension
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out_final
