    # a: shape (batch, M, K), b: shape (batch, K, N)
    batch, M, K = a.shape
    _, _, N = b.shape
    # Ensure contiguous and maybe cast to float16 for speed
    a = a.contiguous()
    b = b.contiguous()
    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)
    # compute strides in element counts
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        batch=batch,
        M=M, N=N, K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,

        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,

    )

def custom_kernel(data):
    x, mask, weights, config = data
    # x shape [B,N,N,dim] float32
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # maybe not needed
    device = x.device

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)    # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # Mask
    if mask is not None:
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Move hidden_dim to second dimension for batched matmul
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # flatten batch and hidden
    batch = B * hidden_dim
    left_flat = left.view(batch, N, N)
    # Transpose for right
    right_T = right.transpose(-2, -1).contiguous()  # [B, H, N, N] -> [B, H, N, N] but transposed across N dimensions.
    right_flat = right_T.view(batch, N, N)

    # Compute matmul using Triton
    out_flat = batched_matmul(left_flat, right_flat)  # shape (batch, N, N)
    # Reshape to [B, H, N, N]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out_gate (apply after to_out_norm)
    out = out * out_gate

    # Final linear
    to_out_w = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_w)

    return out

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(axis=0)
    # compute block indices
    num_m_blocks = tl.cdiv(M, BLOCK_M)
    num_n_blocks = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_m_blocks * num_n_blocks)
    rem = pid % (num_m_blocks * num_n_blocks)
    block_m = rem // num_n_blocks
    block_n = rem % num_n_blocks

    # compute offsets for each
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute base pointer for a and b for given batch
    a_batch_ptr = a_ptr + batch_idx * stride_a_batch
    b_batch_ptr = b_ptr + batch_idx * stride_b_batch
    c_batch_ptr = c_ptr + batch_idx * stride_c_batch

    # compute starting offset for rows and columns
    row_offset = block_m * BLOCK_M
    col_offset = block_n * BLOCK_N

    # compute mask for out-of-bound rows/cols (used later when storing)
    row_mask = (row_offset + offs_m) < M
    col_mask = (col_offset + offs_n) < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K

        # compute pointer offsets
        a_ptrs = a_batch_ptr + (row_offset + offs_m)[:, None] * stride_a_m + (k_offset + offs_k)[None, :] * stride_a_k
        b_ptrs = b_batch_ptr + (k_offset + offs_k)[:, None] * stride_b_k + (col_offset + offs_n)[None, :] * stride_b_n

        # apply mask to load
        a_mask = row_mask[:, None] & ((k_offset + offs_k) < K)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # store results

    c_mask = row_mask[:, None] & col_mask[None, :]

def custom_kernel(data):
    """Implementation of the outgoing TriMul operation using Triton for the core
    pairwise multiplication. This function follows the reference PyTorch version
    but fuses the N^3 einsum into a batched matrix multiplication kernel written
    in Triton for high performance on H100 GPUs.

    Args:
        data: Tuple containing (input_tensor, mask_tensor, weights_dict, config_dict).

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] with the processed output.
    """
    import torch
    import triton
    import triton.language as tl

    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm over channel dimension (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # Linear projections (bias=False)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present and not nominally omitted
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Core N^3 operation -> batched matmul via Triton
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape

    # Move hidden dimension to second axis to treat each channel as a separate batch
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Flatten batch and hidden dimensions for a single batched matmul
    batch = B * hidden_dim
    left_flat = left.view(batch, N, N)
    right_T = right.transpose(-2, -1).contiguous()
    right_flat = right_T.view(batch, N, N)

    # Compute tiled matmul with Triton
    out_flat = batched_matmul(left_flat, right_flat)  # shape (batch, N, N)

    # Reshape to original layout
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # Post‑processing
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          batch, M, N, K,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_m_tiles * num_n_tiles)
    tile_idx = pid % (num_m_tiles * num_n_tiles)
    block_m = tile_idx // num_n_tiles
    block_n = tile_idx % num_n_tiles

    # Offsets
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute start indices for this tile
    m_start = block_m * BLOCK_M
    n_start = block_n * BLOCK_N

    # Masks for partial tiles
    mask_m = (m_start + offs_m) < M
    mask_n = (n_start + offs_n) < N

    # Base pointers for this batch
    a_batch = a_ptr + batch_idx * stride_a_batch
    b_batch = b_ptr + batch_idx * stride_b_batch
    c_batch = c_ptr + batch_idx * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        mask_k = (k_start + offs_k) < K

        a_ptrs = a_batch + (m_start + offs_m)[:, None] * stride_a_m + (k_start + offs_k)[None, :] * stride_a_k
        b_ptrs = b_batch + (k_start + offs_k)[:, None] * stride_b_k + (n_start + offs_n)[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Perform dot product and accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Store the result
    c_ptrs = c_batch + (m_start + offs_m)[:, None] * stride_c_m + (n_start + offs_n)[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using a custom Triton kernel.
    Both a and b are expected to be contiguous tensors of shape (B, M, K) and (B, K, N)
    respectively, with matching batch size B and inner dimension K.
    """
    assert a.is_contiguous()
    assert b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match for matmul"

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # Extract strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tile sizes - chosen to match H100 tensor‑core friendly shapes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_m_tiles = (M + BLOCK_M - 1) // BLOCK_M
    num_n_tiles = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_m_tiles * num_n_tiles,)

    batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c

def custom_kernel(data):
    """
    Triton‑accelerated implementation of the outgoing TriMul operator.
    The function follows the reference PyTorch implementation but replaces the
    O(N³) einsum with a fused batched matrix‑multiplication kernel written in
    Triton. All other linear layers, LayerNorms, and gating are executed with
    standard torch kernels.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim]
    """
    import torch

    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input layer normalisation
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply optional mask (mask = 0/1) – broadcast on hidden channel dim
    # -----------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B, N, N, 1)
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating mechanisms (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core N³ operation: einsum('... i k d, ... j k d -> ... i j d')
    #    Perform as a batched matmul per hidden channel using Triton.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = [B, N, N, hidden_dim]

    # Move hidden dimension to the second axis and flatten batch*hidden for a single
    # batched matmul.
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Flatten batch and hidden dimensions
    flat_batch = B * hidden_dim
    left_flat = left.view(flat_batch, N, N)                         # (B*H, N, N)
    right_T = right.transpose(-2, -1).contiguous()                 # (B, H, N, N) with K and N swapped
    right_flat = right_T.view(flat_batch, N, N)                     # (B*H, N, N)

    # Triton‐based batched matmul
    out_flat = batched_matmul(left_flat, right_flat)                # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output normalisation, gating, and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input tensor (`x` of shape [B, N, N, C]).
2. Compute four linear projections without bias:
   * left  = Linear(x, W_left)          → [B,N,N,H]
   * right = Linear(x, W_right)         → [B,N,N,H]
   * left_gate  = sigmoid(Linear(x, W_lgate))   → [B,N,N,H]
   * right_gate = sigmoid(Linear(x, W_rgate))   → [B,N,N,H]
   * out_gate   = sigmoid(Linear(x, W_outgate)) → [B,N,N,H]
3. Apply optional mask (broadcast on the hidden dimension) to `left` and `right`.
4. Gate the projections: `left *= left_gate`, `right *= right_gate`.
5. Core O(N³) operation:
   The einsum
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   is equivalent to a batched matrix multiplication per hidden channel:
        out_d = left_d @ right_dᵀ
   We fuse all channels (`H`) and batches (`B`) into a single batched
   mat‑mul: shape (B*H, N, N) × (B*H, N, N)ᵀ → (B*H, N, N).
   This is performed by a custom Triton kernel (`batched_matmul_kernel`).
6. Reshape the result back to [B,N,N,H].
7. Apply a second LayerNorm (`to_out_norm`), multiply by `out_gate`,
   and a final Linear projection back to the original channel size `C`.

Only the heavy N³ contraction is implemented in Triton; the remaining
operations use fast PyTorch kernels. The Triton kernel tiles the output
with BLOCK_M×BLOCK_N tiles and loops over K‑tiles, accumulating in FP32
for numerical stability.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (A @ B), where A∈(B,M,K), B∈(B,K,N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B for many independent matrix pairs using tiling."""
    pid = tl.program_id(0)

    # Number of tiles in each dimension
    num_m_tiles = tl.cdiv(M, BLOCK_M)
    num_n_tiles = tl.cdiv(N, BLOCK_N)

    # Decode a 3‑D launch grid (batch, tile_m, tile_n) from a 1‑D pid
    batch_idx = pid // (num_m_tiles * num_n_tiles)
    tile_idx  = pid %  (num_m_tiles * num_n_tiles)
    block_m   = tile_idx // num_n_tiles
    block_n   = tile_idx %  num_n_tiles

    # Offsets inside the tile
    offs_m = tl.arange(0, BLOCK_M)
    offs_n = tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Starting coordinates for this tile
    m_start = block_m * BLOCK_M
    n_start = block_n * BLOCK_N

    # Masks for edge tiles
    mask_m = (m_start + offs_m) < M
    mask_n = (n_start + offs_n) < N

    # Base pointers for the current batch
    a_batch = a_ptr + batch_idx * stride_a_batch
    b_batch = b_ptr + batch_idx * stride_b_batch
    c_batch = c_ptr + batch_idx * stride_c_batch

    # Accumulator in FP32 (more stable than FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        mask_k = (k_start + offs_k) < K

        # Pointers to the current A and B sub‑blocks
        a_ptrs = a_batch + (m_start + offs_m)[:, None] * stride_a_m \
                          + (k_start + offs_k)[None, :] * stride_a_k
        b_ptrs = b_batch + (k_start + offs_k)[:, None] * stride_b_k \
                          + (n_start + offs_n)[None, :] * stride_b_n

        # Load with masks to avoid OOB accesses
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Matrix‑multiply‑accumulate (dot returns same dtype as inputs;
        # cast to FP32 before accumulation)
        acc += tl.dot(a, b).to(tl.float32)

    # Store the result back to C
    c_ptrs = c_batch + (m_start + offs_m)[:, None] * stride_c_m \
                      + (n_start + offs_n)[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def batched_matmul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around the Triton kernel that performs batched matrix multiplication.
    a : (B, M, K) contiguous tensor
    b : (B, K, N) contiguous tensor
    Returns c : (B, M, N) contiguous tensor.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match."

    # Allocate output
    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # Strides are expressed in element units (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # Tile sizes – chosen for good occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid dimensions
    num_m_tiles = (M + BLOCK_M - 1) // BLOCK_M
    num_n_tiles = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_m_tiles * num_n_tiles,)

    # Launch the kernel
    batched_matmul_kernel[grid](
        a_ptr=a.data_ptr(),
        b_ptr=b.data_ptr(),
        c_ptr=c.data_ptr(),
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim].
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------- 1. Input LayerNorm ---------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------- 2. Linear projections -----------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------- 3. Optional mask ---------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # --------------------------- 4. Gating -------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------- 5. Core N³ contraction -------------------
    B, N, _, _ = left.shape  # left shape: [B, N, N, hidden_dim]

    # Move hidden dimension to the second axis → [B, H, N, N]
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and hidden dims to obtain a single batched matmul problem
    flat_batch = B * hidden_dim
    left_flat = left.view(flat_batch, N, N)                     # (B*H, N, N)

    # Right operand needs to be transposed on the two N‑axes (i ⇄ j)
    right_T = right.transpose(-2, -1).contiguous()              # (B, H, N, N)
    right_flat = right_T.view(flat_batch, N, N)                 # (B*H, N, N)

    # Compute the batched matrix multiplication via Triton
    out_flat = batched_matmul(left_flat, right_flat)            # (B*H, N, N)

    # Reshape back to the original 4‑D layout
    out = (
        out_flat
        .view(B, hidden_dim, N, N)          # (B, H, N, N)
        .permute(0, 2, 3, 1)                # (B, N, N, H)
        .contiguous()
    )

    # --------------------------- 6. Output LayerNorm & gating ------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
