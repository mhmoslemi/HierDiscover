def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # read config: bs, seq_len, dim, hidden_dim etc.
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5.

    B, N, _, C = input_tensor.shape

    # Norm x
    x = input_tensor
    # compute mean var per element across dim (last dimension)
    # Use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (C,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask handling
    if not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for kernel
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # Compute out via Triton
    out = triton_gemm(left_f16, right_f16)   # returns float32

    # Apply to_out_norm

    # Multiply out_gate (float32)
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, to_out_weight)  # weight shape (dim, hidden_dim)

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_task = tl.program_id(2)  # combined batch*hidden

    # compute batch idx and hidden idx
    hidden_idx = pid_task % hidden_dim
    batch_idx = pid_task // hidden_dim

    # base pointers for specific batch and hidden channel
    left_base = left_ptr + batch_idx * stride_l_b + hidden_idx * stride_l_d
    right_base = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_d
    out_base = out_ptr + batch_idx * stride_o_b + hidden_idx * stride_o_d

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # create range vectors
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # masks for i and j
    mask_i = i_range < N
    mask_j = j_range < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left tile (M x K)
        a_ptrs = left_base + i_range[:, None] * stride_l_i + k_range[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask = mask_i[:, None] & mask_k[None, :], other = 0.0, dtype=tl.float16)
        a = tl.cast(a, tl.float32)

        # right tile (K x N) (transposed)
        b_ptrs = right_base + k_range[:, None] * stride_r_i + j_range[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask = mask_k[:, None] & mask_j[None, :], other = 0.0, dtype=tl.float16)
        b = tl.cast(b, tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_base + i_range[:, None] * stride_o_i + j_range[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask = mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Fused batch*channel matrix multiplication for TriMul.
    Computes, for each batch b and hidden channel h:
        out[b,i,j,h] = sum_{k=0}^{N-1} left[b,i,k,h] * right[b,j,k,h]
    using a classic tiled matrix multiplication kernel.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_task = tl.program_id(2)  # batch * hidden dimension

    # decode batch index and hidden channel index
    hidden_idx = pid_task % hidden_dim
    batch_idx = pid_task // hidden_dim

    # pointer to the beginning of the slice (b, :, :, h)
    left_base = left_ptr + batch_idx * stride_l_b + hidden_idx * stride_l_d
    right_base = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_d
    out_base = out_ptr + batch_idx * stride_o_b + hidden_idx * stride_o_d

    # compute absolute coordinates of the current tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # masks for boundary handling
    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k-tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_start = kk * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left tile: shape (M, K)
        a_ptrs = left_base + i_range[:, None] * stride_l_i + k_range[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0, dtype=tl.float16)
        a = tl.cast(a, tl.float32)

        # right tile transposed: we need right[b, k, j, h]
        # which corresponds to original right[b, i=k, k=j, h]
        b_ptrs = right_base + k_range[:, None] * stride_r_i + j_range[None, :] * stride_r_k
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0, dtype=tl.float16)
        b = tl.cast(b, tl.float32)

        # accumulate the dot product of the two tiles
        acc += tl.dot(a, b)

    # write the result
    c_ptrs = out_base + i_range[:, None] * stride_o_i + j_range[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def triton_trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Executes the Triton kernel that computes the pairwise contraction
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
    Arguments must be half-precision tensors of shape [B,N,N,H] and be contiguous.
    Returns a float32 tensor of the same shape.
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # fetch stride information (in element units)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Tunable block sizes – chosen to fit in registers on an H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,  # one program for each (batch, hidden) pair
    )

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operator.
    Performs LayerNorm, linear projections, gating, a fused N^2×N contraction
    (implemented in Triton), a second LayerNorm, gating, and final linear projection.
    """
    import torch
    # unpack arguments
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------- LayerNorm on input -------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------- Linear projections -------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------- Mask (if present) -------------------
    if not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------- Gating -------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------- Triton contraction -------------------
    # Cast to fp16 for the kernel (saves memory and increases throughput)
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    # Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = triton_trimul_matmul(left_f16, right_f16)   # float32 tensor

    # ------------------- Second LayerNorm -------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------- Output gating and linear -------------------
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # final shape [B,N,N,dim]

    return out

"""
TriMul (outgoing) – fused Triton implementation

The heavy part of the module is the triple‑wise contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is a batch‑ and channel‑wise matrix multiplication (N×N)·(N×N)ᵀ
for every hidden channel `h`.  The remaining operations (LayerNorm,
linear projections, gating) are cheap and are performed with PyTorch.
The contraction is realised with a tiled GEMM kernel written in Triton,
operating on fp16 inputs and accumulating in fp32 for numerical
stability.  The kernel processes one (batch, hidden) pair per program
instance and writes the result as float32.
"""

import torch
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: one (batch, hidden) pair → C = A @ Bᵀ
# ------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr,          # fp16  [B,N,N,H]
    right_ptr,         # fp16  [B,N,N,H]
    out_ptr,           # fp32  [B,N,N,H]
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The kernel is a classic tiled matmul where the second operand is
    accessed transposed (Bᵀ).  Each program processes a single (batch,
    hidden) slice, yielding a tile of size BLOCK_M×BLOCK_N that is
    accumulated over K‑tiles of size BLOCK_K.
    """
    pid_i = tl.program_id(0)          # tile row (i)
    pid_j = tl.program_id(1)          # tile col (j)
    pid_task = tl.program_id(2)       # combined batch*hidden index

    # decode batch and hidden indices
    hidden_idx = pid_task % H
    batch_idx  = pid_task // H

    # base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_l_b + hidden_idx * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_h
    out_base   = out_ptr   + batch_idx * stride_o_b + hidden_idx * stride_o_h

    # coordinates of the current tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_start = kk * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ----- left tile: shape (M, K) -----
        a_ptrs = left_base + i_range[:, None] * stride_l_i + k_range[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask = mask_i[:, None] & mask_k[None, :],
                    other = 0.0,
                    dtype = tl.float16)
        a = tl.cast(a, tl.float32)

        # ----- right tile (transposed): we need right[b,k,j,h] -----
        # original layout: right[b,i,k,h] → stride_i over i, stride_k over k
        # transposed access uses k as the first index, j as the second
        b_ptrs = right_base + k_range[:, None] * stride_r_i + j_range[None, :] * stride_r_k
        b = tl.load(b_ptrs,
                    mask = mask_k[:, None] & mask_j[None, :],
                    other = 0.0,
                    dtype = tl.float16)
        b = tl.cast(b, tl.float32)

        # accumulate the product of the two tiles
        acc += tl.dot(a, b)

    # store the resulting tile
    c_ptrs = out_base + i_range[:, None] * stride_o_i + j_range[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask = mask_i[:, None] & mask_j[None, :])


def triton_trimul_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launch the Triton kernel that computes the contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Arguments:
        left, right – torch.float16 tensors of shape [B,N,N,H], contiguous.
    Returns:
        torch.float32 tensor of shape [B,N,N,H].
    """
    assert left.dtype == torch.float16 and right.dtype == torch.float16
    B, N, _, H = left.shape

    # output in float32 (accumulator type)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # strides in element units
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # tiling parameters – chosen to fit in registers on an H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows of output tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # cols of output tiles
        B * H,                          # one program per (batch, hidden) pair
    )

    trimul_gemm_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ------------------------------------------------------------
# Public entry point – the forward pass of TriMul (outgoing)
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.

    data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns a torch.Tensor of shape [bs, seq_len, seq_len, dim].
    """
    import torch

    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input (dim‑wise)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask
    # -----------------------------------------------------------------
    if not config.get("nomask", False):
        mask = mask_tensor.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid of a linear layer)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton contraction (the cubic N³ operation)
    # -----------------------------------------------------------------
    # Cast to fp16 to reduce bandwidth; the kernel accumulates in fp32.
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    # out has shape [B,N,N,hidden_dim] and dtype float32
    out = triton_trimul_matmul(left_f16, right_f16)

    # -----------------------------------------------------------------
    # 6. Second LayerNorm (hidden_dim‑wise)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 7. Output gating and final linear projection back to dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
