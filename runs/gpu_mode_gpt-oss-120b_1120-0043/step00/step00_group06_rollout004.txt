"""
TriMul (outgoing) – fast forward implementation using Triton.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} and performs:

1. Layer‑norm over the channel dimension C.
2. Two linear projections (left/right) and three gating projections (left,
   right, out) from C → H (hidden_dim). The gate values are passed through
   sigmoid().
3. The mask (if present) is applied to the left/right projected tensors.
4. Gated tensors are multiplied element‑wise:
      left  = left_proj  * left_gate
      right = right_proj * right_gate
5. The core “outgoing” update:
      out_{b,i,j,d} = Σ_{k=0}^{N-1} left_{b,i,k,d} × right_{b,j,k,d}
   which is a batched GEMM for every hidden channel d.
   This O(N³·H) reduction is fused in a Triton kernel that tiles the
   i‑, j‑, k‑ and d‑dimensions (BLOCK_I×BLOCK_J×BLOCK_K×BLOCK_D).
6. Layer‑norm over the hidden dimension, a final gating (out_gate) and a
   linear projection back to C.

Only the heavy reduction (step 5) is written in Triton; the remaining
operations use PyTorch (highly‑optimized cuBLAS / cuDNN kernels).  The
implementation supports B∈{1,2}, N up to 1024, C up to 384 and H up to 384.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel for the outgoing triangular multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def tr_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    N, H,                                       # sequence length, hidden dim
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)  # i‑tile
    pid_j = tl.program_id(1)  # j‑tile
    pid_h = tl.program_id(2)  # hidden‑tile

    # ------------------------------------------------------------------
    # Offsets for this tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    h_base = pid_h * BLOCK_H

    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_h = h_base + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator for the (i,j,h) tile
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # number of k‑tiles
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left[i,k,h]  (shape: BLOCK_I × BLOCK_K × BLOCK_H)
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # (BI, BK, BH)

        # ------------------------------------------------------------------
        # Load right[j,k,h] transposed to (k,j,h) for GEMM
        #   right[j,k,h] → we need right_T[k,j,h]
        # ------------------------------------------------------------------
        right_ptrs = right_ptr + \
            offs_k[:, None, None] * stride_right_i + \
            offs_j[None, :, None] * stride_right_k + \
            offs_h[None, None, :] * stride_right_h
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # (BK, BJ, BH)

        # ------------------------------------------------------------------
        # Perform dot product for each hidden slice
        # ------------------------------------------------------------------
        for dh in range(BLOCK_H):
            # If the hidden index is out‑of‑bounds the loaded tile is zero, so
            # the dot contributes nothing.
            left_tile = left[:, :, dh]   # (BI, BK)
            right_tile = right[:, :, dh] # (BK, BJ)
            acc[:, :, dh] += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )

# ----------------------------------------------------------------------
# Entry point called by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over channel dimension C
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (C,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections & gates (use cuBLAS via torch.nn.functional.linear)
    # ------------------------------------------------------------------
    # flatten the leading three dimensions for a single matmul
    x_flat = x.view(-1, C)                       # (B*N*N, C)

    left_proj_weight   = weights["left_proj.weight"]   # (H, C)
    right_proj_weight  = weights["right_proj.weight"]  # (H, C)

    left_gate_weight   = weights["left_gate.weight"]   # (H, C)
    right_gate_weight  = weights["right_gate.weight"]  # (H, C)
    out_gate_weight    = weights["out_gate.weight"]    # (H, C)

    left_proj = F.linear(x_flat, left_proj_weight).view(B, N, N, hidden_dim)
    right_proj = F.linear(x_flat, right_proj_weight).view(B, N, N, hidden_dim)

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) *after* projection, before gating
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype=dtype)

    mask_exp = mask_tensor.unsqueeze(-1)                               # (B,N,N,1)
    left_proj = left_proj * mask_exp
    right_proj = right_proj * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # (B,N,N,H)
    right = right_proj * right_gate       # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (outgoing) – Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)          # (B,N,N,H)

    # Block sizes – tuned for H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8   # hidden‑dim tile

    for b in range(B):
        # slice the batch dimension
        left_b  = left[b]   # (N, N, H)
        right_b = right[b]  # (N, N, H)
        out_b   = out[b]    # (N, N, H)

        # stride information (in elements, not bytes)
        stride_li = left_b.stride(0)   # i‑axis
        stride_lk = left_b.stride(1)   # k‑axis
        stride_lh = left_b.stride(2)   # h‑axis

        stride_ri = right_b.stride(0)  # j‑axis (called i in kernel for clarity)
        stride_rk = right_b.stride(1)  # k‑axis
        stride_rh = right_b.stride(2)  # h‑axis

        stride_oi = out_b.stride(0)    # i‑axis
        stride_oj = out_b.stride(1)    # j‑axis
        stride_oh = out_b.stride(2)    # h‑axis

        # launch grid
        grid_i = (N + BLOCK_I - 1) // BLOCK_I
        grid_j = (N + BLOCK_J - 1) // BLOCK_J
        grid_h = (hidden_dim + BLOCK_H - 1) // BLOCK_H

        tr_mul_outgoing_kernel[
            (grid_i, grid_j, grid_h)
        ](
            left_b.data_ptr(),
            right_b.data_ptr(),
            out_b.data_ptr(),
            N,
            hidden_dim,
            stride_li,
            stride_lk,
            stride_lh,
            stride_ri,
            stride_rk,
            stride_rh,
            stride_oi,
            stride_oj,
            stride_oh,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # (B,N,N,H)

    out = out * out_gate                               # gated output

    to_out_weight = weights["to_out.weight"]           # (C, H)
    out_flat = out.view(-1, hidden_dim)                # (B*N*N, H)
    output = F.linear(out_flat, to_out_weight)        # (B*N*N, C)
    output = output.view(B, N, N, C)                  # (B,N,N,C)

    return output
