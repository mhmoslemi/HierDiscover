    left_ptr, right_ptr, out_ptr,
    B, N, H,

   pid_n = tl.program_id(1)

def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_li, stride_lk, stride_lh,
    stride_rj, stride_rk, stride_rh,
    stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)
    pid_h = tl.program_id(2)

    # Compute coordinates
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
tri_mul_outgoing_kernel
~~~~~~~~~~~~~~~~~~~~~~~

Implementation of the *outgoing* TriMul operation from AlphaFold‑3.

Given two projected tensors ``left`` and ``right`` of shape
``[B, N, N, H]`` (H = hidden_dim) the core computation is

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

i.e. a batched matrix multiplication per hidden channel (with the second
operand transposed).  The kernel computes a block‑wise GEMM for a single
hidden slice at a time (the hidden slice index is encoded in the third
program‑id axis) and accumulates the K‑reduction in registers.  The
remaining operations (LayerNorm, linear projections, gating, final
projection) are performed in PyTorch.

The kernel works for arbitrary batch size (we launch it per batch), any
sequence length N (≤ 1024 in the tests) and hidden dimension H (≤ 384).
It uses float32 arithmetic for exactness; it can be easily switched to
float16 by changing the load/store dtypes.

"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out = left @ right.T across the K dimension
# for a single hidden channel slice.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,            # *float32
    right_ptr,           # *float32
    out_ptr,             # *float32
    N, H,                # sequence length, hidden dim (int32)
    stride_li, stride_lk, stride_lh,   # left strides (int32)
    stride_rj, stride_rk, stride_rh,   # right strides (int32)
    stride_oi, stride_oj, stride_oh,   # out strides   (int32)
    BLOCK_M: tl.constexpr,              # tile size in M (i)
    BLOCK_N: tl.constexpr,              # tile size in N (j)
    BLOCK_K: tl.constexpr,              # tile size in reduction k
):
    """Block‑wise matmul for a single hidden slice."""
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_h = tl.program_id(2)   # hidden slice

    # ------------------------------------------------------------------
    # Coordinates of the tile
    # ------------------------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    h = pid_h                                      # scalar hidden index

    # Masks for out‑of‑bounds rows / cols
    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # Accumulator for the output tile (M × N)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # K‑reduction loop
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        ks = k_base + tl.arange(0, BLOCK_K)                 # [BLOCK_K]
        mask_k = ks < N

        # --------------------------------------------------------------
        # Load a sub‑tile of ``left`` : shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        a_ptrs = left_ptr + (
            i[:, None] * stride_li +            # i dimension
            ks[None, :] * stride_lk +           # k dimension
            h * stride_lh                        # hidden dimension (scalar)
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )   # (BLOCK_M, BLOCK_K)

        # --------------------------------------------------------------
        # Load a sub‑tile of ``right`` transposed :
        # we need shape (BLOCK_K, BLOCK_N) i.e. right[k, j]
        # --------------------------------------------------------------
        b_ptrs = right_ptr + (
            ks[:, None] * stride_rk +            # k dimension first
            j[None, :] * stride_rj +            # j dimension second
            h * stride_rh                        # hidden dimension (scalar)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )   # (BLOCK_K, BLOCK_N)

        # --------------------------------------------------------------
        # Accumulate the partial product
        # --------------------------------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Apply masks for the final write (optional, mainly for edge tiles)
    # ------------------------------------------------------------------
    acc = tl.where(mask_i[:, None] & mask_j[None, :], acc, 0.0)

    # ------------------------------------------------------------------
    # Store the result tile to ``out``
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + (
        i[:, None] * stride_oi +
        j[None, :] * stride_oj +
        h * stride_oh
    )
    tl.store(out_ptrs, acc)


# ----------------------------------------------------------------------
# Public entry point – performs the full TriMul forward pass.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (float32 or bool)
        - weights      : dict of weight tensors (see comments below)
        - config       : dict containing ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    B, N, N2, D = input_tensor.shape
    assert N == N2 and D == dim, "Input shape mismatch."

    # --------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # Optional mask (if provided). Mask is broadcast over the hidden dim.
    # --------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # Gating : sigmoid(linear(x))
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Allocate output of the multiplicative update
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # same [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Prepare strides for the Triton kernel (in elements, not bytes)
    # ------------------------------------------------------------------
    # left  : (B,N,i,k,h)  -> we need strides for i,k,h
    stride_li = left.stride(1)      # stride over i dimension
    stride_lk = left.stride(2)      # stride over k dimension
    stride_lh = left.stride(3)      # stride over hidden dimension

    # right : (B,N,j,k,h)
    stride_rj = right.stride(1)     # stride over j dimension
    stride_rk = right.stride(2)     # stride over k dimension
    stride_rh = right.stride(3)     # stride over hidden dimension

    # out   : (B,N,i,j,h)
    stride_oi = out.stride(1)       # stride over i dimension
    stride_oj = out.stride(2)       # stride over j dimension
    stride_oh = out.stride(3)       # stride over hidden dimension

    # ------------------------------------------------------------------
    # Tuning parameters – these work well on an H100 for the given sizes.
    # Feel free to adjust if you target a different GPU.
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i (rows)
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j (cols)
        hidden_dim,                     # one block per hidden slice
    )

    # ------------------------------------------------------------------
    # Launch Triton kernel for each batch element independently.
    # ------------------------------------------------------------------
    for b in range(B):
        # Compute base pointers for the current batch slice
        left_ptr = left.data_ptr() + b * left.stride(0)
        right_ptr = right.data_ptr() + b * right.stride(0)
        out_ptr = out.data_ptr() + b * out.stride(0)

        tri_mul_outgoing_kernel[
            grid
        ](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            stride_li,
            stride_lk,
            stride_lh,
            stride_rj,
            stride_rk,
            stride_rh,
            stride_oi,
            stride_oj,
            stride_oh,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # --------------------------------------------------------------
    # Post‑processing: normalize, gating, final linear projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                 # element‑wise gating
    out = F.linear(out, weights["to_out.weight"])   # -> [B,N,N,dim]

    return out
