def layernorm(x, weight, bias, eps=1e-5):
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    inv = torch.rsqrt(var + eps)
    x_norm = (x - mean) * inv
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm
```

def batched_matmul_triton(A,B):
    # A shape (B, M, K)
    # B shape (B, K, N) (already transposed)
    M,N,K = A.shape[1], B.shape[2], A.shape[2]
    # ensure contiguous
    A = A.contiguous()
    B = B.contiguous()
    C = torch.empty((A.shape[0], M, N), dtype=A.dtype, device=A.device)
    # compute strides
    stride_am, stride_ak = A.stride(1), A.stride(2)
    stride_bk, stride_bn = B.stride(1), B.stride(2)
    stride_cm, stride_cn = C.stride(1), C.stride(2)
    batch_stride_a = A.stride(0)
    batch_stride_b = B.stride(0)
    batch_stride_c = C.stride(0)
    grid = ( (M + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             A.shape[0] )
    batched_matmul_kernel[grid](
         A_ptr=A, B_ptr=B, C_ptr=C,
         M=M, N=N, K=K,
         stride_am=stride_am,
         stride_ak=stride_ak,
         stride_bk=stride_bk,
         stride_bn=stride_bn,
         stride_cm=stride_cm,
         stride_cn=stride_cn,
         batch_stride_a=batch_stride_a,
         batch_stride_b=batch_stride_b,
         batch_stride_c=batch_stride_c,
         BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return C
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm x
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # x normalized
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    inv = torch.rsqrt(var + eps)
    x = (input_tensor - mean) * inv
    if norm_weight is not None:
        x = x * norm_weight
    if norm_bias is not None:
        x = x + norm_bias

    # Linear projections without bias
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Reshape for batched GEMM (multiply left @ right^T across k)
    B, N, _, _ = left.shape
    # Permute to bring hidden_dim to 2nd axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Flatten batch and hidden
    left_flat = left_perm.view(B * hidden_dim, N, N)   # [B*H, M=N, K=N]
    right_flat_T = right_perm.transpose(2, 3).contiguous().view(B * hidden_dim, N, N) # [B*H, K=N, N]

    # Compute batched matrix multiplication via Triton kernel
    out_flat = batched_matmul_triton(left_flat, right_flat_T)  # shape [B*H, N, N]

    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_out = out.mean(dim=-1, keepdim=True)
    var_out = out.var(dim=-1, unbiased=False, keepdim=True)
    inv_out = torch.rsqrt(var_out + eps)
    out_norm = (out - mean_out) * inv_out
    if to_out_norm_weight is not None:
        out_norm = out_norm * to_out_norm_weight
    if to_out_norm_bias is not None:
        out_norm = out_norm + to_out_norm_bias

    # gate
    out_norm = out_norm * out_gate

    # final linear
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return out

"""
custom_kernel - Triangular multiplicative update (outgoing version)

The kernel follows the PyTorch reference of AlphaFold3's TriMul:
  1. Layer‑norm over the last dimension.
  2. Linear projections (left/right) and three sigmoidal gates.
  3. Element‑wise mask (if present) and gating.
  4. Core O(N³⋅H) operation:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
     which is a batched matrix‑multiply L @ Rᵀ for every hidden channel d.
     We implement this heavy reduction with a custom Triton GEMM kernel
     (batch = B·H, M=N, K=N, N=N).
  5. Layer‑norm over the hidden dimension, second gate, and final linear.
All tensors stay on the GPU; only the GEMM is written in Triton.
"""

import torch
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton batched GEMM: C = A @ B   (A : [B, M, K],  B : [B, K, N])
# ---------------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                      # pointers
    M, N, K,                                   # matrix shapes
    stride_am, stride_ak,                      # A strides (M, K)
    stride_bk, stride_bn,                      # B strides (K, N)
    stride_cm, stride_cn,                      # C strides (M, N)
    batch_stride_a, batch_stride_b, batch_stride_c,  # batch stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # block along M
    pid_n = tl.program_id(1)          # block along N
    pid_b = tl.program_id(2)          # batch index (B*H)

    # -----------------------------------------------------------------------
    # Offsets for this block
    # -----------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # batch start pointers
    a_batch = a_ptr + pid_b * batch_stride_a
    b_batch = b_ptr + pid_b * batch_stride_b
    c_batch = c_ptr + pid_b * batch_stride_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # Loop over K dimension in tiles
    # -----------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < K

        a_ptrs = a_batch + offs_m[:, None] * stride_am + k_off[None, :] * stride_ak
        b_ptrs = b_batch + k_off[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # promote to float32 before dot to keep numerical stability
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # -----------------------------------------------------------------------
    # Store result
    # -----------------------------------------------------------------------
    c_ptrs = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM implemented with Triton.
    A : [B, M, K]   (float32/float16)
    B : [B, K, N]   (float32/float16)
    Returns C : [B, M, N] (same dtype as A/B)
    """
    assert A.is_contiguous()
    assert B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides (in elements, not bytes)
    stride_am, stride_ak = A.stride(1), A.stride(2)
    stride_bk, stride_bn = B.stride(1), B.stride(2)
    stride_cm, stride_cn = C.stride(1), C.stride(2)
    batch_stride_a = A.stride(0)
    batch_stride_b = B.stride(0)
    batch_stride_c = C.stride(0)

    # Tile sizes – tuned for H100, float32/float16
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )
    _batched_gemm_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

# Helper: LayerNorm (no bias fused) – pure torch (fast enough)
# ---------------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    """LayerNorm over the last dimension."""
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    inv = torch.rsqrt(var + eps)
    out = (x - mean) * inv
    if weight is not None:
        out = out * weight
    if bias is not None:
        out = out + bias
    return out


# ---------------------------------------------------------------------------
# Entry point required by the evaluator
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward‑only TriMul (outgoing) kernel.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask        : torch.Tensor, shape [B, N, N] (bool or 0/1)
        - weights     : dict of torch.Tensor containing all learned parameters
        - config      : dict with keys ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim], dtype float32 and on the same device.
    """
    # -----------------------------------------------------------------------
    # Unpack
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------------
    # 1) Input layer‑norm
    # -----------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x, weights["left_proj.weight"]
    )   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(
        x, weights["right_proj.weight"]
    )   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden dimension)
    # -----------------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # -----------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x, weights["left_gate.weight"]
    ).sigmoid()                                   # [B, N, N, hidden_dim]
    right_gate = torch.nn.functional.linear(
        x, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x, weights["out_gate.weight"]
    ).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5) Core O(N³⋅H) reduction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Treat each hidden channel as an independent batch element.
    # -----------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dimension next to batch for flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # flatten batch & hidden dim
    left_flat = left_perm.view(B * hidden_dim, N, N)    # [B*H, M=N, K=N]
    # right must be transposed on the last two dims (K,N) for matmul
    right_flat_T = right_perm.transpose(2, 3).contiguous().view(B * hidden_dim, N, N)

    # Batched GEMM: L @ Rᵀ  (R already transposed)
    out_flat = _batched_gemm_triton(left_flat, right_flat_T)  # [B*H, N, N]

    # Restore original layout: [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------------
    # 6) Second layer‑norm (over hidden dimension)
    # -----------------------------------------------------------------------
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------------
    # 7) Output gate and final linear
    # -----------------------------------------------------------------------
    out = out * out_gate                     # gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"]
    )                                        # [B, N, N, dim]

    return out
