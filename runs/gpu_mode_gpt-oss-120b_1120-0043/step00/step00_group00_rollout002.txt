@triton.jit
def trmul_output_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lr_b, stride_lr_i, stride_lr_k, stride_lr_d,
    stride_rr_b, stride_rr_j, stride_rr_k, stride_rr_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M : tl.constexpr, BLOCK_N : tl.constexpr, BLOCK_K : tl.constexpr
):
   # Program ID for each (b,d, i_block, j_block)
   pid = tl.program_id(axis=0)
   # Flatten to have total number of programs = B*D * ceil(N/BLOCK_M)*ceil(N/BLOCK_N)
   # Compute block index
   total_blocks_i = tl.cdiv(N, BLOCK_M)
   total_blocks_j = tl.cdiv(N, BLOCK_N)
   pid_bd = pid // (total_blocks_i * total_blocks_j)
   pid_ij = pid % (total_blocks_i * total_blocks_j)
   i_block = pid_ij // total_blocks_j
   j_block = pid_ij % total_blocks_j
   b = pid_bd // D
   d = pid_bd % D

   # Compute offsets
   offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
   # Offsets for k loop
   offs_k = tl.arange(0, BLOCK_K)

   # Pointers for left: left[b,i,k,d]
   left_ptrs = left_ptr + b*stride_lr_b + d*stride_lr_d + offs_i[:, None] * stride_lr_i + offs_k[None, :] * stride_lr_k
   # right: right[b,j,k,d] shape we want right transposed k->k position still k; we need right[b,j,k,d] but we need to access right[b,j,k,d] where we multiply left_i_k * right_j_k. So we load right tile of shape (BLOCK_N, BLOCK_K) for each j block.
   right_ptrs = right_ptr + b*stride_rr_b + d*stride_rr_d + offs_j[:, None] * stride_rr_j + offs_k[None, :] * stride_rr_k

   # Accumulate
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(N, BLOCK_K)):
       left = tl.load(left_ptrs, mask=(offs_i[:, None] < N) & (offs_k[None, :] < N), other=0.0)
       right = tl.load(right_ptrs, mask=(offs_j[:, None] < N) & (offs_k[None, :] < N), other=0.0)
       # compute dot across k dimension: left * right => accumulate sum over k
       # We need elementwise multiply left_i_k * right_j_k and sum across k, which in tile terms we can do:
       # left shape (BLOCK_M, BLOCK_K), right shape (BLOCK_N, BLOCK_K). Need dot product across K: sum_k left[i,k] * right[j,k]
       # This can be expressed as matmul: left @ right.T
       acc += tl.dot(left, right, trans_b=True)   # trans_b will transpose right shape (BLOCK_N, BLOCK_K) to (BLOCK_K, BLOCK_N)
       # Update pointers
       left_ptrs += BLOCK_K * stride_lr_k
       right_ptrs += BLOCK_K * stride_rr_k

   # Store
   out_ptrs = out_ptr + b*stride_out_b + d*stride_out_d + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
   tl.store(out_ptrs, acc.to(tl.float16) if ??? else ???)
   # We need final output in float32; can store directly as float32.

"""
TriMul (outgoing) implementation with a fused Triton kernel for the N³ contraction.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (over the last dimension `dim`).
2. Compute four linear projections:
   * left_proj  = x @ W_left      (shape B,N,N,H)
   * right_proj = x @ W_right
   * left_gate  = sigmoid(x @ W_left_gate)
   * right_gate = sigmoid(x @ W_right_gate)
   * out_gate   = sigmoid(x @ W_out_gate)   (used after the contraction)
   (W_* are weight matrices supplied in ``weights``.)
3. Apply the optional mask (broadcast on the hidden dimension) and the
   element‑wise gates:
        left  = left_proj  * mask * left_gate
        right = right_proj * mask * right_gate
4. **Core TriMul contraction** – compute for every batch `b`,
   output channel `h` and pair (i,j):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication over the ``k`` dimension.
   The contraction is performed by a custom Triton kernel that
   tiles the i, j and k dimensions (BLOCK_M×BLOCK_N×BLOCK_K) and
   accumulates in FP32 while the data lives in FP16.
5. Layer‑norm the contraction result over the hidden dimension `H`,
   multiply by ``out_gate`` and apply the final linear projection
   `to_out` (shape dim←H) to obtain the output tensor with shape
   ``[B, N, N, dim]`` (float32).

The Triton kernel is deliberately simple and works for any batch size
(1‑2), sequence length up to 1024 and hidden dimension up to 384.
Mixed‑precision (FP16 inside the kernel, FP32 for the rest) gives a large
speed‑up on H100 while keeping the final result in FP32 as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: computes   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trmul_einsum_kernel(
    left_ptr,               # *half   left  (B,N,N,H) – indexed as (b,i,k,h)
    right_ptr,              # *half   right (B,N,N,H) – indexed as (b,j,k,h)
    out_ptr,                # *half   out   (B,N,N,H) – indexed as (b,i,j,h)

    B, N, H,                # static problem sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # strides of left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # strides of right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # strides of out

    BLOCK_M: tl.constexpr,  # tile size for i
    BLOCK_N: tl.constexpr,  # tile size for j
    BLOCK_K: tl.constexpr   # tile size for k
):
    pid = tl.program_id(0)

    # ---- compute block indices -------------------------------------------------
    num_i_blocks = tl.cdiv(N, BLOCK_M)
    num_j_blocks = tl.cdiv(N, BLOCK_N)
    blocks_ij = num_i_blocks * num_j_blocks

    pid_bd = pid // blocks_ij            # combined batch*head index
    pid_ij = pid - pid_bd * blocks_ij    # intra‑(i,j) block index

    # decode batch and head (hidden) indices
    b = pid_bd // H
    h = pid_bd - b * H

    # decode i‑block and j‑block
    i_block = pid_ij // num_j_blocks
    j_block = pid_ij - i_block * num_j_blocks

    # ---- tile offsets ---------------------------------------------------------
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the beginning of the tiles
    left_tile_ptr  = (left_ptr
                      + b * stride_l_b
                      + h * stride_l_h
                      + offs_i[:, None] * stride_l_i
                      + offs_k[None, :] * stride_l_k)

    right_tile_ptr = (right_ptr
                      + b * stride_r_b
                      + h * stride_r_h
                      + offs_j[:, None] * stride_r_j
                      + offs_k[None, :] * stride_r_k)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over the k‑dimension ------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for _ in range(num_k_tiles):
        # out‑of‑bounds masks for the current tile
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        left = tl.load(left_tile_ptr,
                       mask=(mask_i[:, None] & mask_k[None, :]),
                       other=0.0)          # half
        right = tl.load(right_tile_ptr,
                        mask=(mask_j[:, None] & mask_k[None, :]),
                        other=0.0)         # half

        # dot: (BLOCK_M, BLOCK_K) × (BLOCK_N, BLOCK_K)ᵀ → (BLOCK_M, BLOCK_N)
        # accumulation is performed in FP32
        acc += tl.dot(left, right, trans_b=True, out_dtype=tl.float32)

        # advance pointers to the next k‑tile
        left_tile_ptr  += BLOCK_K * stride_l_k
        right_tile_ptr += BLOCK_K * stride_r_k

    # ---- write back ------------------------------------------------------------
    out_tile_ptr = (out_ptr
                    + b * stride_o_b
                    + h * stride_o_h
                    + offs_i[:, None] * stride_o_i
                    + offs_j[None, :] * stride_o_j)

    tl.store(out_tile_ptr, acc,
             mask=(mask_i[:, None] & mask_j[None, :]))


# ----------------------------------------------------------------------
# Entry point used by the testing harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (dtype = torch.float32)
    """
    with torch.no_grad():
        # ------------------------------------------------------------------
        # unpack inputs
        # ------------------------------------------------------------------
        input_tensor, mask, weights, config = data
        device = input_tensor.device
        dtype = input_tensor.dtype

        dim = config["dim"]
        hidden_dim = config["hidden_dim"]
        nomask = config.get("nomask", False)

        B, N, _, _ = input_tensor.shape

        # ------------------------------------------------------------------
        # LayerNorm on the last dimension (dim)
        # ------------------------------------------------------------------
        norm_weight = weights["norm.weight"]
        norm_bias   = weights["norm.bias"]
        x = F.layer_norm(input_tensor,
                         normalized_shape=(dim,),
                         weight=norm_weight,
                         bias=norm_bias)

        # ------------------------------------------------------------------
        # Linear projections (no bias) – use torch.nn.functional.linear which
        # works on the last dimension.
        # ------------------------------------------------------------------
        left_proj  = F.linear(x, weights["left_proj.weight"])
        right_proj = F.linear(x, weights["right_proj.weight"])

        # ------------------------------------------------------------------
        # Optional mask (broadcast on hidden dimension)
        # ------------------------------------------------------------------
        if nomask or mask is None:
            mask_broadcast = torch.ones((B, N, N, 1), device=device, dtype=x.dtype)
        else:
            # mask may be bool or float; ensure float32 and broadcast
            mask_broadcast = mask.unsqueeze(-1).to(dtype)

        left  = left_proj  * mask_broadcast
        right = right_proj * mask_broadcast

        # ------------------------------------------------------------------
        # Gates (sigmoid of linear transforms of the *normalized* input)
        # ------------------------------------------------------------------
        left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
        right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
        out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

        left  = left  * left_gate
        right = right * right_gate

        # ------------------------------------------------------------------
        # Cast to half precision before the heavy N³ contraction.
        # ------------------------------------------------------------------
        left_f16  = left.to(torch.float16).contiguous()
        right_f16 = right.to(torch.float16).contiguous()
        out_f16   = torch.empty((B, N, N, hidden_dim),
                                dtype=torch.float16,
                                device=device)

        # ------------------------------------------------------------------
        # Launch Triton kernel
        # ------------------------------------------------------------------
        # strides in number of elements (not bytes)
        stride_l = left_f16.stride()    # (b,i,k,h)
        stride_r = right_f16.stride()
        stride_o = out_f16.stride()

        # Block sizes – tuned for H100 (fits into shared memory)
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 64

        # Compute launch grid
        grid_i = (N + BLOCK_M - 1) // BLOCK_M
        grid_j = (N + BLOCK_N - 1) // BLOCK_N
        total_blocks = B * hidden_dim * grid_i * grid_j

        trmul_einsum_kernel[(total_blocks,)](
            left_f16.data_ptr(),
            right_f16.data_ptr(),
            out_f16.data_ptr(),
            B, N, hidden_dim,
            stride_l[0], stride_l[1], stride_l[2], stride_l[3],
            stride_r[0], stride_r[1], stride_r[2], stride_r[3],
            stride_o[0], stride_o[1], stride_o[2], stride_o[3],
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )
        # ------------------------------------------------------------------
        # Back to FP32 for the remaining (LayerNorm, gating, final linear)
        # ------------------------------------------------------------------
        out = out_f16.to(torch.float32)

        # to_out_norm (LayerNorm over hidden_dim)
        to_out_norm_weight = weights["to_out_norm.weight"]
        to_out_norm_bias   = weights["to_out_norm.bias"]
        out = F.layer_norm(out,
                           normalized_shape=(hidden_dim,),
                           weight=to_out_norm_weight,
                           bias=to_out_norm_bias)

        # Multiply by out_gate (already in FP32)
        out = out * out_gate

        # Final linear projection (hidden_dim -> dim)
        out = F.linear(out, weights["to_out.weight"])

        return out
```
