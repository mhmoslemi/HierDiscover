@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                   stride_r_b, stride_r_j, stride_r_k, stride_r_h,
                   stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0) # batch*hidden index
    pid_i = tl.program_id(1) # block row
    pid_j = tl.program_id(2) # block column

    # compute the batch index and hidden channel from pid_b
    total_batch = B * H
    b_h = pid_b
    b = b_h // H
    h = b_h % H

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointer offsets for left and right
    # left offset: [b, i, k, h]
    # right offset: [b, j, k, h] (need to use k dimension same as left)
    # We'll loop over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left block
        left_ptrs = left_ptr + b * stride_l_b + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k) + h * stride_l_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right block (transposed)
        right_ptrs = right_ptr + b * stride_r_b + (offs_j[:, None] * stride_r_j) + (offs_k[None, :] * stride_r_k) + h * stride_r_h
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # dot (sum over k)
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # store output
    out_ptrs = out_ptr + b * stride_o_b + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j) + h * stride_o_h
    tl.store(out_ptrs, acc.to(tl.float16) if out_dtype==tl.float16 else acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = x.device
    dtype = x.dtype

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Proj
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Compute out via Triton kernel
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Compute strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()

    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride() # note j dimension is third

    total_batch = B * hidden_dim
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (total_batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) – Triton‐accelerated implementation

Given an input tensor x ∈ ℝ^[B,N,N,C] the module performs:

1. LayerNorm over the feature dimension C.
2. Two linear projections (left/right) to a hidden dimension H.
3. Optional pair‑wise mask applied to the projected tensors.
4. Gated multiplicative update:
       left  = left  * sigmoid(Linear(x))
       right = right * sigmoid(Linear(x))
   The core operation is
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batched matrix multiplication
       out[b,:,:,h] = left[b,:,:,h] @ right[b,:,:,h].T
   for each hidden channel h.  This step is executed by
   a custom Triton kernel that treats (B·H) as the batch dimension
   and performs a tiled GEMM (BLOCK_M×BLOCK_N) with a BLOCK_K inner
   reduction.  All arithmetic is carried out in float32.

5. LayerNorm over the hidden dimension, element‑wise gating with the
   “out_gate”, and a final linear projection back to C.

The kernel works for any batch size (B ≤ 2), sequence length N (≤1024),
feature dimension C (e.g. 128/384) and hidden dimension H (e.g. 128).
Only the O(N³) multiplicative update is Triton‑fused; the remaining
operations are performed with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul update.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,                     # tensor pointers
    B, N, H,                                          # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left  strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out   strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_bh = tl.program_id(0)   # combined batch * hidden index
    pid_i  = tl.program_id(1)   # block row index
    pid_j  = tl.program_id(2)   # block column index

    # --------------------------------------------------------------
    # Resolve batch (b) and hidden channel (h) from the combined id.
    # --------------------------------------------------------------
    total = B * H
    bh = pid_bh
    b  = bh // H
    h  = bh % H

    # --------------------------------------------------------------
    # Offsets for the output tile.
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Tile over the reduction dimension k.
    # --------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k):
        k_start = kk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block  : shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr \
            + b * stride_l_b \
            + (offs_i[:, None] * stride_l_i) \
            + (offs_k[None, :] * stride_l_k) \
            + h * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right block : shape (BLOCK_K, BLOCK_N)
        # We need right as (k, j) = right[b, j, k, h]
        right_ptrs = right_ptr \
            + b * stride_r_b \
            + (offs_k[:, None] * stride_r_k) \
            + (offs_j[None, :] * stride_r_i) \
            + h * stride_r_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate the dot product.
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Write the tile back to the output tensor.
    # --------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_o_b \
        + (offs_i[:, None] * stride_o_i) \
        + (offs_j[None, :] * stride_o_j) \
        + h * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point used by the grader.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data = (x, mask, weights, config)
            x        : torch.Tensor [B, N, N, C] (float32)
            mask     : torch.Tensor [B, N, N] (float32 or bool) or None
            weights  : dict of model parameters (see docstring)
            config   : dict with keys "dim" and "hidden_dim"
    Returns:
        out : torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs.
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim).
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # --------------------------------------------------------------
    # 2) Linear projections (no bias).
    # --------------------------------------------------------------
    left  = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension).
    # --------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # Ensure mask is float32 for multiplication.
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4) Gating (sigmoid of linear transforms of the normalized input).
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Core multiplicative update via Triton kernel.
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    # Output tensor: [B, N, N, hidden_dim]
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides (in element units, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Kernel launch configuration.
    # We treat each (batch, hidden) pair as an independent GEMM.
    total_bh = B * hidden_dim
    BLOCK_M = 64    # rows per block
    BLOCK_N = 64    # cols per block
    BLOCK_K = 32    # reduction block size

    grid = (
        total_bh,                          # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,      # number of row tiles
        (N + BLOCK_N - 1) // BLOCK_N       # number of column tiles
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # --------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm, gating, and output projection.
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
