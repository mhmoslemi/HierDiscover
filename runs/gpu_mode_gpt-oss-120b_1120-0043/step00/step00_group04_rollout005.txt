@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # pointers
        a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch * batch_stride_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # load
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0).to(tl.float16)

        # compute dot product
        acc += tl.dot(a, b).to(tl.float32)

    # write out
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask)

"""
Optimized Triton implementation of the “outgoing” Triangle Multiplication (TriMul)
module used in AlphaFold‑3. The kernel computes the core O(N³) tensor contraction:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

by treating each hidden channel `d` (and batch) as an independent matrix
multiply:   left_d @ right_dᵀ.
All linear projections, gating, masking and layer‑norms are performed with PyTorch,
while the massive batched GEMM is executed by a custom Triton kernel (FP32
precision, FP32 accumulation).  This yields a correct FP32 result and is fast
enough on an H100 (target ~1 ms for the supplied configurations).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
# A: (batch, M, K)   B: (batch, K, N)  -> C: (batch, M, N)
# All tensors are float32, accumulator is float32.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # strides for A (row, col)
    stride_bk, stride_bn,          # strides for B (row(K), col(N))
    stride_cm, stride_cn,          # strides for C (row, col)
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)        # block row
    pid_n = tl.program_id(2)        # block col

    # block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Pointers for the current tile of A and B
        a_ptrs = a_ptr + batch_id * batch_stride_a \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak

        b_ptrs = b_ptr + batch_id * batch_stride_b \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Load tiles (masked for out‑of‑bounds)
        a_mask = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        b_mask = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)   # fp32
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)   # fp32

        # Accumulate the dot product
        acc += tl.dot(a, b)   # fp32→fp32

    # Write out C
    c_ptrs = c_ptr + batch_id * batch_stride_c \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn

    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def batched_gemm_fp32(A: torch.Tensor, B_T: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B_T  using the Triton kernel above.
    A : (B*H, M, K)   (float32)
    B_T : (B*H, K, N) (float32)   # B transposed already
    Returns C : (B*H, M, N) (float32)
    """
    assert A.dim() == 3 and B_T.dim() == 3
    batch, M, K = A.shape
    _, K2, N = B_T.shape
    assert K == K2

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # Strides in *elements* (not bytes)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bk = B_T.stride(1)   # stride over K dimension
    stride_bn = B_T.stride(2)   # stride over N dimension

    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    batch_stride_a = A.stride(0)
    batch_stride_b = B_T.stride(0)
    batch_stride_c = C.stride(0)

    # Block sizes – chosen to fit comfortably on H100 SMs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        A,
        B_T,
        C,
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        batch_stride_a,
        batch_stride_b,
        batch_stride_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor_or_None, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------- LayerNorm on the input --------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -------------------- Linear projections (no bias) --------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B, N, N, hidden_dim]

    # -------------------- Apply mask (if any) --------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # -------------------- Gating values --------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -------------------- Apply gates --------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------- Prepare for batched GEMM --------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden_dim, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)      # [B*H, N, N]
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(-1, N, N)    # [B*H, N, N]

    # We need right transposed (K dimension first) → shape (B*H, N, N) with K,N swapped
    right_flat_T = right_flat.permute(0, 2, 1).contiguous()                # [B*H, N, N] = Bᵀ

    # -------------------- Batched matrix multiplication --------------------
    out_flat = batched_gemm_fp32(left_flat, right_flat_T)   # [B*H, N, N] in fp32

    # Reshape back to 4‑D tensor [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # -------------------- Output layer‑norm + gating --------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # -------------------- Final linear projection back to dim --------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
