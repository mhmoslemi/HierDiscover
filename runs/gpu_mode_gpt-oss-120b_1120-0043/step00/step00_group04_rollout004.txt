@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    i_offsets = i_base + tl.arange(0, BLOCK_M)
    j_offsets = j_base + tl.arange(0, BLOCK_N)
    
    # Mask for i and j
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # Create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Compute number of K tiles
    num_k = tl.cdiv(N, BLOCK_K)

        k_offsets = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # Load left tile shape (M,K,H)
        left_ptrs = left_ptr + pid_b * stride_l_b \
                         + i_offsets[:, None, None] * stride_l_i \
                         + k_offsets[None, :, None] * stride_l_k \
                         + tl.arange(0, BLOCK_H)[None, None, :] * stride_l_h
        left = tl.load(left_ptrs, mask=(i_mask[:, None, None] & k_mask[None, :, None] & (tl.arange(0, BLOCK_H)[None, None, :] < H)), other=0.0)

        # Load right tile shape (K,N,H)

"""
TriMul “outgoing” kernel (AlphaFold3 style)

We compute
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` are the projected & gated pairwise
features of shape [B, N, N, H] (H = hidden_dim).  The heavy
N³ reduction is carried out by a custom Triton kernel that tiles the
(i, j) plane and the reduction dimension k.  The hidden dimension is
fused as an extra vector dimension (BLOCK_H) so that each Triton
thread accumulates a small (M×N) matrix for several hidden channels at
once.  All other operations (LayerNorm, Linear projections, gating,
final LayerNorm + Linear) are performed with PyTorch for simplicity.

The kernel works for any batch size ≤2, sequence length up to 1024 and
hidden dimensions that are multiples of the compile‑time BLOCK_H (the
remainder is safely masked out).  It returns a float32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the 4‑D einsum:  out[...,i,j,h] = Σ_k left[...,i,k,h] * right[...,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # *float32   left  [B, N, N, H]
    right_ptr,           # *float32   right [B, N, N, H]
    out_ptr,             # *float32   out   [B, N, N, H]
    B, N, H,             # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    # --------------------------------------------------------------
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute the TriMul einsum using tiled GEMM per hidden channel."""
    pid_b = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)   # i‑tile index
    pid_n = tl.program_id(2)   # j‑tile index

    # Offsets for the (i,j) plane
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N
    i_off = i_base + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    j_off = j_base + tl.arange(0, BLOCK_N)   # [BLOCK_N]
    h_off = tl.arange(0, BLOCK_H)            # [BLOCK_H]

    i_mask = i_off < N
    j_mask = j_off < N
    h_mask = h_off < H

    # Accumulator for a (BLOCK_M × BLOCK_N × BLOCK_H) tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_tile in range(num_k):
        k_base = k_tile * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        k_mask = k_off < N

        # ------------------------------------------------------------------
        # Load left[i,k,h]   shape (M, K, H)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_l_b
            + i_off[:, None, None] * stride_l_i
            + k_off[None, :, None] * stride_l_k
            + h_off[None, None, :] * stride_l_h
        )
        left_mask = (
            i_mask[:, None, None]
            & k_mask[None, :, None]
            & h_mask[None, None, :]
        )
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)   # (M, K, H)

        # ------------------------------------------------------------------
        # Load right[j,k,h]   shape (K, N, H)   (note the k‑first layout)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_r_b
            + k_off[:, None, None] * stride_r_k
            + j_off[None, :, None] * stride_r_i
            + h_off[None, None, :] * stride_r_h
        )
        right_mask = (
            k_mask[:, None, None]
            & j_mask[None, :, None]
            & h_mask[None, None, :]
        )
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)   # (K, N, H)

        # ------------------------------------------------------------------
        # For every hidden slice h compute a small GEMM:
        #   acc[...,h] += left[...,h] @ right[...,h]
        # ------------------------------------------------------------------
        for hh in range(0, BLOCK_H):
            # 2‑D slices of shape (M,K) and (K,N)
            left_h  = left[:, :, hh]   # (M, K)
            right_h = right[:, :, hh]  # (K, N)
            acc[:, :, hh] += tl.dot(left_h, right_h)   # (M,N)

    # ------------------------------------------------------------------
    # Store the result tile to global memory
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_o_b
        + i_off[:, None, None] * stride_o_i
        + j_off[None, :, None] * stride_o_j
        + h_off[None, None, :] * stride_o_h
    )
    out_mask = (
        i_mask[:, None, None]
        & j_mask[None, :, None]
        & h_mask[None, None, :]
    )
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Python entry point – performs the full forward pass of the TriMul module
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be dummy if `nomask` is True)
        - weights      : dict of model parameters (LayerNorm weights/bias, Linear weights, etc.)
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Projections (no bias) + optional mask
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask is float (0/1); broadcast on hidden dim
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape   # N is the pairwise sequence length

    # ------------------------------------------------------------------
    # 4. Triton kernel: compute the N³ reduction per hidden channel
    # ------------------------------------------------------------------
    out = torch.empty_like(left)   # shape [B, N, N, hidden]

    # Tiling parameters – chosen to fit typical H100 workloads
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8   # hidden dimension chunk (must be >0)

    # Compute strides (in element units)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    grid = (
        B,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 5. Final LayerNorm + gating + linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # shape [dim, hidden]

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return output
