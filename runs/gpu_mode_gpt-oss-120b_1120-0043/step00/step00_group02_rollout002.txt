def custom_kernel(data):
    """
    Custom Triton kernel implementation of the TriMul (outgoing) operation.
    This function reproduces the forward pass of the PyTorch `TriMul` module,
    but computes the expensive `einsum('... i k d, ... j k d -> ... i j d')`
    using a Triton‑based batched matrix multiplication.  All other
    operations (LayerNorm, linear projections, gating) are performed with
    standard PyTorch for clarity.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ---------- 1. LayerNorm on input ----------
    # Compute per‑position mean / var over the channel dimension `dim`
    # Shape: [B, N, N, dim]
    x = input_tensor
    # Reshape for convenience - PyTorch broadcasting works fine
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias  # (B,N,N,dim)

    # ---------- 2. Linear projections ----------
    # Linear without bias: y = x @ weight.T
    left = torch.nn.functional.linear(x_norm, left_proj_w)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_w)  # (B,N,N,hidden_dim)

    # ---------- 3. Optional mask ----------
    if config.get("nomask", False):
        mask = None
    else:
        # mask shape [B,N,N], broadcast over hidden dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ---------- 4. Gating ----------
    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Expensive O(N^3) contraction ----------
    # Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

    B, N, _, H = left.shape
    device = left.device
    # Allocate output tensor
    out = torch.empty_like(left)  # (B,N,N,hidden_dim)

    # Ensure tensors are contiguous for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Triton kernel definition (single channel per launch)
    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        N,
        stride_li, stride_lk,
        stride_ri, stride_rk,
        stride_oi, stride_oj,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        """
        Compute out[i,j] = sum_k left[i,k] * right[j,k] for one channel.
        """
        pid_m = tl.program_id(0)   # i‑tile
        pid_n = tl.program_id(1)   # j‑tile

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_m = offs_m < N
        mask_n = offs_n < N

        # Accumulator in FP32 (left/right may be FP16/FP32)
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over the reduction dimension k
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Pointers to the sub‑blocks
            # left[i,k]  (M×K)
            left_ptrs = left_ptr + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
            # right[j,k] (N×K)
            right_ptrs = right_ptr + offs_n[:, None] * stride_ri + offs_k[None, :] * stride_rk

            # Load with zero‑padding for out‑of‑bounds
            left_block = tl.load(left_ptrs,
                                 mask=mask_m[:, None] & mask_k[None, :],
                                 other=0.0).to(tl.float32)
            right_block = tl.load(right_ptrs,
                                  mask=mask_n[:, None] & mask_k[None, :],
                                  other=0.0).to(tl.float32)

            # Standard GEMM update: acc += left_block @ right_block.T
            acc += tl.dot(left_block, right_block.T)

        # Store the result back
        out_ptrs = out_ptr + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
        tl.store(out_ptrs,
                 acc,
                 mask=mask_m[:, None] & mask_n[None, :])

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Loop over batches and hidden channels
    for b in range(B):
        # Create 2‑D views for a single batch
        left_b = left[b]   # (N,N,hidden_dim)
        right_b = right[b]
        out_b = out[b]

        # Strides for the (N,N) matrix (no hidden dimension)
        stride_li = left_b.stride(0)   # stride for i dimension
        stride_lk = left_b.stride(1)   # stride for k dimension
        stride_ri = right_b.stride(0)
        stride_rk = right_b.stride(1)
        stride_oi = out_b.stride(0)
        stride_oj = out_b.stride(1)

        # Grid size
        grid = ( (N + BLOCK_M - 1) // BLOCK_M,
                 (N + BLOCK_N - 1) // BLOCK_N )

        # Launch the kernel once per hidden channel
        for h in range(H):
            # use tensor slices for the current channel
            left_chan = left_b[..., h]   # shape (N,N)
            right_chan = right_b[..., h]
            out_chan = out_b[..., h]

            triton_trimul_kernel[grid](
                left_chan,               # ptr
                right_chan,
                out_chan,
                N,
                stride_li, stride_lk,
                stride_ri, stride_rk,
                stride_oi, stride_oj,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
            )

    # ---------- 6. Output LayerNorm ----------
    # LayerNorm over hidden_dim for each (b,i,j) entry
    out_mean = out.mean(-1, keepdim=True)
    out_var = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ---------- 7. Gate & final linear ----------
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, to_out_w)  # no bias

    return output

            left_ptrs = left_ptr + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
            right_ptrs = right_ptr + offs_n[:, None] * stride_ri + offs_k[None, :] * stride_rk

            left_block = tl.load(left_ptrs,
                                 mask=mask_m[:, None] & mask_k[None, :],
                                 other=0.0).to(tl.float32)
            right_block = tl.load(right_ptrs,
                                  mask=mask_n[:, None] & mask_k[None, :],
                                  other=0.0).to(tl.float32)

            # GEMM update
            acc += tl.dot(left_block, right_block.T)

        out_ptrs = out_ptr + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
        tl.store(out_ptrs,
                 acc,
                 mask=mask_m[:, None] & mask_n[None, :])

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Loop over batch and hidden dimension
    for b in range(B):
        left_b = left[b]   # (N,N,H)
        right_b = right[b]
        out_b = out[b]

        # Strides for the (N,N) matrices (hidden dimension is collapsed)
        stride_li = left_b.stride(0)   # i stride
        stride_lk = left_b.stride(1)   # k stride
        stride_ri = right_b.stride(0)
        stride_rk = right_b.stride(1)
        stride_oi = out_b.stride(0)
        stride_oj = out_b.stride(1)

        grid = ((N + BLOCK_M - 1) // BLOCK_M,
                (N + BLOCK_N - 1) // BLOCK_N)

        for h in range(H):
            left_chan = left_b[..., h]   # (N,N)
            right_chan = right_b[..., h]
            out_chan = out_b[..., h]

            triton_trimul_kernel[grid](
                left_chan,
                right_chan,
                out_chan,
                N,
                stride_li, stride_lk,
                stride_ri, stride_rk,
                stride_oi, stride_oj,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
            )

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm
    # ----------------------------------------------------------------------
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ----------------------------------------------------------------------
    # 7. Gate and final projection
    # ----------------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    The implementation follows the reference PyTorch code:
        * LayerNorm on the input tensor (over the last dimension `dim`).
        * Linear projections (`left_proj`, `right_proj`) without bias.
        * Optional binary mask applied to the projections.
        * Three sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
        * Heavy contraction:
              out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
          This O(N³·H) term is computed with a custom Triton kernel that
          performs a batched matrix multiplication for each hidden channel.
        * LayerNorm over the hidden dimension, multiplication by `out_gate`,
          and a final linear projection back to `dim`.

    Only the O(N³) contraction is performed on the GPU via Triton; the
    surrounding linear / normalization / gating steps use standard PyTorch
    operations for clarity and ease of implementation.
    """
    import torch
    import triton
    import triton.language as tl

    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Extract model weights (all are 1‑D or 2‑D tensors, bias‑free where noted)
    # ------------------------------------------------------------------
    norm_w      = weights["norm.weight"]
    norm_b      = weights["norm.bias"]
    left_w      = weights["left_proj.weight"]
    right_w     = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w= weights["right_gate.weight"]
    out_gate_w  = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w    = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (per position over the channel dimension)
    # ------------------------------------------------------------------
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var  = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b                     # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_norm, left_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B,N,N] → [B,N,N,1] for broadcasting
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. TriMul contraction – Triton kernel
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty_like(left)                         # [B,N,N,hidden_dim]

    # Ensure contiguous tensors for pointer arithmetic
    left  = left.contiguous()
    right = right.contiguous()
    out   = out.contiguous()

    @triton.jit
    def triton_trimul_kernel(
        left_ptr, right_ptr, out_ptr,
        N,
        stride_li, stride_lk,                 # left  strides (i, k)
        stride_ri, stride_rk,                 # right strides (j, k)
        stride_oi, stride_oj,                 # out  strides (i, j)
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        """
        Compute a single‑channel matrix product:
            out[i, j] = Σ_k left[i, k] * right[j, k]
        where left/right/out are 2‑D matrices (i,k), (j,k), (i,j).
        """
        pid_m = tl.program_id(0)   # i‑tile
        pid_n = tl.program_id(1)   # j‑tile

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_m = offs_m < N
        mask_n = offs_n < N

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # reduction over k
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Pointers to the current blocks
            left_ptrs  = left_ptr  + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
            right_ptrs = right_ptr + offs_n[:, None] * stride_ri + offs_k[None, :] * stride_rk

            left_block  = tl.load(left_ptrs,
                                  mask=mask_m[:, None] & mask_k[None, :],
                                  other=0.0).to(tl.float32)
            right_block = tl.load(right_ptrs,
                                  mask=mask_n[:, None] & mask_k[None, :],
                                  other=0.0).to(tl.float32)

            # GEMM update: acc += left_block @ right_block.T
            acc += tl.dot(left_block, right_block.T)

        # Store the result
        out_ptrs = out_ptr + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
        tl.store(out_ptrs,
                 acc,
                 mask=mask_m[:, None] & mask_n[None, :])

    # Tile sizes – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Launch kernel per batch and per hidden channel
    for b in range(B):
        left_b  = left[b]   # shape (N,N,hidden_dim)
        right_b = right[b]
        out_b   = out[b]

        # Strides for the (N,N) matrix view (hidden dim collapsed)
        stride_li = left_b.stride(0)   # i‑stride = N*H
        stride_lk = left_b.stride(1)   # k‑stride = H
        stride_ri = right_b.stride(0)
        stride_rk = right_b.stride(1)
        stride_oi = out_b.stride(0)
        stride_oj = out_b.stride(1)

        grid = ((N + BLOCK_M - 1) // BLOCK_M,
                (N + BLOCK_N - 1) // BLOCK_N)

        for h in range(H):
            # Slice the current hidden channel → 2‑D matrices
            left_chan  = left_b[..., h]   # (N,N)
            right_chan = right_b[..., h]
            out_chan   = out_b[..., h]

            triton_trimul_kernel[grid](
                left_chan,
                right_chan,
                out_chan,
                N,
                stride_li, stride_lk,
                stride_ri, stride_rk,
                stride_oi, stride_oj,
                BLOCK_M=BLOCK_M,
                BLOCK_N=BLOCK_N,
                BLOCK_K=BLOCK_K,
            )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # ------------------------------------------------------------------
    # 7. Apply final gate and linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, to_out_w)   # no bias

    return output
