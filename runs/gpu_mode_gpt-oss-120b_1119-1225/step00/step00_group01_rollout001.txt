def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # layernorm over dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), norm_weight, norm_bias, eps=eps)

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if not config.get("nomask", True):
        mask_ = mask[..., None].type_as(left)
        left = left * mask_
        right = right * mask_

    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # compute out via triton
    out = tri_mul_triton(left, right) # implement function returning B,N,N,hidden

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), to_out_norm_weight, to_out_norm_bias, eps=1e-5)

    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row i
    pid_n = tl.program_id(1)  # block col j
    pid_hb = tl.program_id(2) # combined batch*head

    # Compute batch index and head (hidden dim) index
    b = pid_hb // H
    h = pid_hb % H

    # Range of indices
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute mask for i and j
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            h * stride_left_h + \
            i_offset[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k

        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            h * stride_right_h + \
            k_offset[:, None] * stride_right_k + \
            j_offset[None, :] * stride_right_j

        right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        h * stride_out_h + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

"""
TriMul (outgoing) – Triton‑accelerated implementation.

The kernel computes the core triple‑multiplicative update
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
for every batch `b` and hidden channel `h`.  The tensor
shape is [B, N, N, H] (H = hidden_dim).  The contraction is
equivalent to H independent matrix‑multiplications
`left_h @ right_h.T`.  The Triton kernel tiles the i‑ and j‑axes
(and the hidden dimension is handled by a separate program‑id axis)
and accumulates the sum over k with a block‑wise dot product.
All other steps (LayerNorm, linear projections, gating) are performed
with PyTorch for simplicity; the heavy N³ reduction is off‑loaded
to the Triton kernel.

The function `custom_kernel` follows the required signature:
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
and returns the processed tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N³ reduction: out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,       # *Pointer* to left tensor       [B, N, N, H]
    right_ptr,      # *Pointer* to right tensor      [B, N, N, H]
    out_ptr,        # *Pointer* to output tensor      [B, N, N, H]
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b,  stride_out_i,  stride_out_j,  stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Tiles:
        i (row) -> BLOCK_M
        j (col) -> BLOCK_N
        k (reduction) -> BLOCK_K (looped)
    Program ID 0 : i‑tile
    Program ID 1 : j‑tile
    Program ID 2 : combined batch‑hidden index = b*H + h
    """
    pid_m = tl.program_id(0)               # tile over i
    pid_n = tl.program_id(1)               # tile over j
    pid_hb = tl.program_id(2)              # batch * hidden channel

    # decode batch and hidden channel
    b = pid_hb // H
    h = pid_hb % H

    # ---------- index ranges ----------
    i_off = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i_off < N
    j_valid = j_off < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- reduction over k ----------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_off = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_off < N

        # left[i, k, h]  shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            h * stride_left_h + \
            i_off[:, None] * stride_left_i + \
            k_off[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=i_valid[:, None] & k_valid[None, :],
                       other=0.0,
                       dtype=tl.float32)

        # right[j, k, h]  we need it transposed to (k, j)
        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            h * stride_right_h + \
            k_off[:, None] * stride_right_k + \
            j_off[None, :] * stride_right_j
        right = tl.load(right_ptrs,
                        mask=k_valid[:, None] & j_valid[None, :],
                        other=0.0,
                        dtype=tl.float32)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        acc += tl.dot(left, right)

    # ---------- write back ----------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        h * stride_out_h + \
        i_off[:, None] * stride_out_i + \
        j_off[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=i_valid[:, None] & j_valid[None, :])


# ----------------------------------------------------------------------
# Helper that launches the above kernel
# ----------------------------------------------------------------------
def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, H]  (float32)
    returns out:   [B, N, N, H]  (float32)
    """
    assert left.shape == right.shape, "left and right must have identical shapes"
    B, N, _, H = left.shape

    # Allocate output tensor
    out = torch.empty_like(left, dtype=torch.float32)

    # Strides (in element units, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tiling parameters – chosen to fit well on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H                      # batch‑*‑hidden
    )

    _tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor: torch.Tensor [B, N, N, dim]
            mask_tensor : torch.Tensor [B, N, N] (may be ignored if nomask=True)
            weights_dict: mapping of weight names to tensors (all on CUDA)
            config_dict : dict containing at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    eps = 1e-5
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # --------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gating linear layers + sigmoid
    # --------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Core N³ contraction – Triton kernel
    # --------------------------------------------------------------
    # The kernel expects float32 tensors; left/right are already float32.
    out = _tri_mul_triton(left, right)     # shape [B, N, N, H]

    # --------------------------------------------------------------
    # Output normalization, gating and final projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight,
        bias=to_out_norm_bias, eps=eps
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dimension (no bias)
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
