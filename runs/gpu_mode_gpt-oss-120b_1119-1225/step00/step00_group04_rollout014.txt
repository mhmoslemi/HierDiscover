def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Use torch.nn.functional.layer_norm
    # Compute mean and var across last dim
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute the N^3 matmul via Triton
    B, N, _, _ = left.shape
    # allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Set block sizes
    BLOCK_M = 128  # maybe choose 64?
    BLOCK_N = 128
    BLOCK_K = 32  # small for speed

    # Compute strides for left (a) and right (b) and out (c)
    # left shape: [B,N,N,hidden_dim]
    # We treat A as left[..., :, :, d] shape (B,N,N)
    # Strides: we can compute using .stride()
    # e.g., left.stride() returns (N*N*hidden_dim, N*hidden_dim, hidden_dim, 1)
    # But if we treat left as (B,N,N,hidden_dim), then for a slice pointer offset for d we shift by d * stride_d.

@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, K,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bi, stride_bk,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program idx
    pid = tl.program_id(0)
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    num_tiles = num_m * num_n

    pid_b = pid // num_tiles
    pid_tile = pid % num_tiles
    pid_m = pid_tile // num_n
    pid_n = pid_tile % num_n

    # offsets within tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    i_mask = i < N
    j_mask = j < N

    # pointer to start of block for A and B
    a_ptrs = a_ptr + pid_b * stride_ab + i[:, None] * stride_ai + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + j[None, :] * stride_bi

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offsets each iteration:
        k_offset = k * BLOCK_K
        a_offset_ptrs = a_ptrs + k_offset * stride_ak
        b_offset_ptrs = b_ptrs + k_offset * stride_bk

        # load
        a = tl.load(a_offset_ptrs, mask=(i_mask[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K)), other=0.0)
        b = tl.load(b_offset_ptrs, mask=((k_offset + tl.arange(0, BLOCK_K)[:, None] < K) & j_mask[None, :]), other=0.0)

@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    B: tl.constexpr, M: tl.constexpr, N: tl.constexpr, K: tl.constexpr,
    stride_ab, stride_ai, stride_ak,
    stride_bb, stride_bi, stride_bk,
    stride_cb, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute M_tiling & N_tiling
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_m * num_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_n
    pid_n = tile_idx % num_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # clamp masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers (for current batch)
    a_base = a_ptr + batch_idx * stride_ab
    b_base = b_ptr + batch_idx * stride_bb
    c_base = c_ptr + batch_idx * stride_cb

    # compute pointer offsets without k (to be updated in loop)
    a_ptrs = a_base + offs_m[:, None] * stride_ai
    b_ptrs = b_base + offs_n[None, :] * stride_bi

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # current k indices
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block (M x K)
        a = tl.load(a_ptrs + offs_k[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block (K x N) (note the transposed layout)
        b = tl.load(b_ptrs + offs_k[:, None] * stride_bk,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)

    # Store
    c_ptrs = c_base + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul module ("outgoing" variant).
    ...
    """

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor,
                                            normalized_shape=(dim,),
                                            weight=norm_weight,
                                            bias=norm_bias,
                                            eps=eps)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [..., hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if provided (mask shape [B,N,N])
    if mask is not None:
        # mask maybe bool or float
        mask_unsq = mask.unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute out = sum_k left[...,i,k,:] * right[...,j,k,:] using Triton
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Get strides in elements (not bytes)
    stride_a_batch = left.stride(0)
    stride_a_i = left.stride(1)   # stride for i dimension
    stride_a_k = left.stride(2)   # stride for k dimension
    stride_d = left.stride(3)     # =1

    stride_b_batch = right.stride(0)
    stride_b_i = right.stride(1)
    stride_b_k = right.stride(2)

    stride_c_batch = out.stride(0)
    stride_c_i = out.stride(1)
    stride_c_j = out.stride(2)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid size
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * num_m * num_n
    grid = (total_tiles,)

    # Launch kernel for each hidden dimension slice
    # Using a simple loop over hidden_dim
    # The kernel processes one hidden dimension at a time.
    for d in range(hidden_dim):
        a_ptr = left.data_ptr() + d * stride_d
        b_ptr = right.data_ptr() + d * stride_d
        c_ptr = out.data_ptr() + d * stride_d

        triton_matmul_kernel[grid](
            a_ptr, b_ptr, c_ptr,
            B, N, N, N,
            stride_a_batch, stride_a_i, stride_a_k,
            stride_b_batch, stride_b_i, stride_b_k,
            stride_c_batch, stride_c_i, stride_c_j,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )

    # Now we have out [B,N,N,hidden_dim]
    # Apply to_out_norm (LayerNorm over hidden_dim)
    out_normed = torch.nn.functional.layer_norm(out,
                                                normalized_shape=(hidden_dim,),
                                                weight=to_out_norm_weight,
                                                bias=to_out_norm_bias,
                                                eps=eps)

    # Multiply by out_gate (elementwise)
    out_normed = out_normed * out_gate

    # Final linear projection to original dim
    out_final = torch.nn.functional.linear(out_normed, to_out_weight)

    return out_final

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The module implements the forward part of the TriMul operation from AlphaFold‑3.
It follows the reference PyTorch implementation:

1. Layer‑norm the input tensor.
2. Linear projections (`left_proj`, `right_proj`) and corresponding gates.
3. Optional masking of the projected tensors.
4. The core N³ operation:
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiply of each hidden‑dimension slice:
      left[..., :, :, d] @ right[..., :, :, d].T
   This is performed by a custom Triton kernel (`triton_matmul_kernel`).
5. Layer‑norm + out gate on the accumulated tensor.
6. Final linear projection back to the original `dim`.

Only the expensive N³ multiplication is done in Triton; all surrounding
operations use regular PyTorch for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def triton_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers to A, B, C
    B, M, N, K,                              # batch, rows, cols, reduction size
    stride_ab, stride_ai, stride_ak,         # strides for A  (batch, i, k)
    stride_bb, stride_bi, stride_bk,         # strides for B  (batch, j, k)
    stride_cb, stride_ci, stride_cj,         # strides for C  (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul C = A @ Bᵀ where A,B have shape (B, M, K) and (B, N, K).

    The kernel computes C[b, i, j] = Σₖ A[b, i, k] * B[b, j, k] for each hidden
    dimension slice.  It is a straightforward tiled GEMM following the pattern
    shown in the Triton matrix‑multiply examples.
    """
    pid = tl.program_id(0)

    # number of tiles in the i‑ and j‑ dimensions
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_m * num_n

    # decode program id into batch, tile‑row and tile‑col indices
    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_n          # tile row
    pid_n = tile_idx % num_n           # tile col

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)  # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)  # j indices

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the selected batch
    a_batch = a_ptr + batch_idx * stride_ab
    b_batch = b_ptr + batch_idx * stride_bb
    c_batch = c_ptr + batch_idx * stride_cb

    # pointers that will be stepped in the k‑loop (no k‑offset yet)
    a_ptrs = a_batch + offs_m[:, None] * stride_ai        # (BLOCK_M, 1)
    b_ptrs = b_batch + offs_n[None, :] * stride_bi        # (1, BLOCK_N)

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks of size BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # current k indices and validity mask
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(a_ptrs + k_offset[None, :] * stride_ak,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B block transposed: shape (BLOCK_K, BLOCK_N)
        b = tl.load(b_ptrs + k_offset[:, None] * stride_bk,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = c_batch + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask          : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights       : dict of torch.Tensors with the model weights
        - config        : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the result of the TriMul forward pass.
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------------- #
    # Layer‑norm the input
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------- #
    # Optional mask (broadcasted over the hidden dimension)
    # --------------------------------------------------------------------- #
    if mask is not None:
        # Ensure mask has the same dtype for multiplication
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------------- #
    # Gating linear layers + sigmoid
    # --------------------------------------------------------------------- #
    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # Core N³ contraction implemented with Triton
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Strides expressed in number of elements (not bytes)
    stride_a_batch = left.stride(0)   # B‑stride
    stride_a_i     = left.stride(1)   # i‑stride
    stride_a_k     = left.stride(2)   # k‑stride
    stride_d       = left.stride(3)   # hidden‑dim stride (==1 for contiguous)

    stride_b_batch = right.stride(0)
    stride_b_i     = right.stride(1)   # j‑stride (same layout as i)
    stride_b_k     = right.stride(2)

    stride_c_batch = out.stride(0)
    stride_c_i     = out.stride(1)
    stride_c_j     = out.stride(2)

    # Tunable block sizes – these work well on H100 for the sizes in the spec.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: one program per (batch, tile_i, tile_j)
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * num_m * num_n
    grid = (total_tiles,)

    # Launch one kernel per hidden‑dimension slice (d)
    for d in range(hidden_dim):
        a_ptr = left.data_ptr() + d * stride_d
        b_ptr = right.data_ptr() + d * stride_d
        c_ptr = out.data_ptr() + d * stride_d

        triton_matmul_kernel[grid](
            a_ptr,
            b_ptr,
            c_ptr,
            B,
            N,          # M  (i‑dimension)
            N,          # N  (j‑dimension)
            N,          # K  (reduction over k)
            stride_a_batch,
            stride_a_i,
            stride_a_k,
            stride_b_batch,
            stride_b_i,
            stride_b_k,
            stride_c_batch,
            stride_c_i,
            stride_c_j,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # --------------------------------------------------------------------- #
    # Post‑processing: layer‑norm, out‑gate, final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    out_normed = out_normed * out_gate

    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    final_out = torch.nn.functional.linear(out_normed, to_out_weight)  # → [B, N, N, dim]

    return final_out
