@triton.jit
def fused_proj_gate_kernel(
    x_ptr, mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    w_left_ptr, w_right_ptr,
    w_left_gate_ptr, w_right_gate_ptr,
    w_out_gate_ptr,
    B, N, dim, hidden_dim,
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_maskb, stride_maski, stride_maskj,
    stride_leftb, stride_lefti, stride_leftj, stride_lefth,
    stride_rightb, stride_righti, stride_rightj, stride_righth,
    stride_out_gateb, stride_out_gatei, stride_out_gatej, stride_out_gateh,
    stride_w_l_dim, stride_w_l_hidden,
    stride_w_r_dim, stride_w_r_hidden,
    stride_w_lg_dim, stride_w_lg_hidden,
    stride_w_rg_dim, stride_w_rg_hidden,
    stride_w_og_dim, stride_w_og_hidden,
    BLOCK_DIM: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total_pos = B * N * N
    if pid >= total_pos:
        return

    # compute b,i,j from pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # mask value
    m = 1.0
    if mask_ptr != 0:
        m = tl.load(mask_ptr + b * stride_maskb + i * stride_maski + j * stride_maskj)

    # base pointers
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj

    # pointers for outputs
    left_base = left_out_ptr + b * stride_leftb + i * stride_lefti + j * stride_leftj
    right_base = right_out_ptr + b * stride_rightb + i * stride_righti + j * stride_rightj
    out_gate_base = out_gate_out_ptr + b * stride_out_gateb + i * stride_out_gatei + j * stride_out_gatej

    # Loop over hidden dimension in tiles
    for h_offset in tl.range(0, hidden_dim, BLOCK_H):
        # Determine valid hidden size
        hid_valid = (h_offset + tl.arange(0, BLOCK_H)) < hidden_dim
        # Initialize accumulators
        acc_left = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_ogate = tl.zeros([BLOCK_H], dtype=tl.float32)

        # Loop over dim
        for d_offset in tl.range(0, dim, BLOCK_DIM):
            # valid dim mask
            dim_valid = (d_offset + tl.arange(0, BLOCK_DIM)) < dim
            # load x block
            x_idx = d_offset + tl.arange(0, BLOCK_DIM)
            x = tl.load(x_base + x_idx * stride_xd, mask=dim_valid, other=0.0)  # shape (BLOCK_DIM,)
            # reshape to (1, BLOCK_DIM)
            x_row = x[None, :]  # shape (1, BLOCK_DIM)

            # load weight tiles
            # left proj weight tile
            w_l = tl.load(w_left_ptr + (d_offset + tl.arange(0, BLOCK_DIM))[:, None] * stride_w_l_dim
                            + (h_offset + tl.arange(0, BLOCK_H))[None, :] * stride_w_l_hidden,
                          mask=dim_valid[:, None] & hid_valid[None, :], other=0.0)  # shape (BLOCK_DIM, BLOCK_H)
            # right proj
            w_r = tl.load(w_right_ptr + (d_offset + tl.arange(0, BLOCK_DIM))[:, None] * stride_w_r_dim
                            + (h_offset + tl.arange(0, BLOCK_H))[None, :] * stride_w_r_hidden,
                          mask=dim_valid[:, None] & hid_valid[None, :], other=0.0)
            # left gate
            w_lg = tl.load(w_left_gate_ptr + (d_offset + tl.arange(0, BLOCK_DIM))[:, None] * stride_w_lg_dim
                            + (h_offset + tl.arange(0, BLOCK_H))[None, :] * stride_w_lg_hidden,
                          mask=dim_valid[:, None] & hid_valid[None, :], other=0.0)
            # right gate
            w_rg = tl.load(w_right_gate_ptr + (d_offset + tl.arange(0, BLOCK_DIM))[:, None] * stride_w_rg_dim
                            + (h_offset + tl.arange(0, BLOCK_H))[None, :] * stride_w_rg_hidden,
                          mask=dim_valid[:, None] & hid_valid[None, :], other=0.0)
            # out gate
            w_og = tl.load(w_out_gate_ptr + (d_offset + tl.arange(0, BLOCK_DIM))[:, None] * stride_w_og_dim
                            + (h_offset + tl.arange(0, BLOCK_H))[None, :] * stride_w_og_hidden,
                          mask=dim_valid[:, None] & hid_valid[None, :], other=0.0)

            # dot products: (1, BLOCK_DIM) dot (BLOCK_DIM, BLOCK_H) => (1, BLOCK_H)
            # left proj
            acc_left = acc_left + tl.dot(x_row, w_l)[0]
            acc_right = acc_right + tl.dot(x_row, w_r)[0]
            acc_lgate = acc_lgate + tl.dot(x_row, w_lg)[0]
            acc_rgate = acc_rgate + tl.dot(x_row, w_rg)[0]
            acc_ogate = acc_ogate + tl.dot(x_row, w_og)[0]

        # after complete dim loop
        # apply sigmoid to gate accumulators
        left_gate = tl.sigmoid(acc_lgate)
        right_gate = tl.sigmoid(acc_rgate)
        out_gate = tl.sigmoid(acc_ogate)

        # apply mask and gating
        left_res = acc_left * left_gate * m
        right_res = acc_right * right_gate * m

        # store results
        hid_idx = h_offset + tl.arange(0, BLOCK_H)
        tl.store(left_base + hid_idx * stride_lefth, left_res, mask=hid_valid)
        tl.store(right_base + hid_idx * stride_righth, right_res, mask=hid_valid)
        tl.store(out_gate_base + hid_idx * stride_out_gateh, out_gate, mask=hid_valid)

@triton.jit
def fused_proj_gate_kernel(
    # Pointers
    x_ptr, mask_ptr,
    left_ptr, right_ptr, out_gate_ptr,
    # Weight pointers (transposed to [dim, hidden_dim])
    w_left_ptr, w_right_ptr,
    w_left_gate_ptr, w_right_gate_ptr,
    w_out_gate_ptr,
    # Metadata
    B, N, dim, hidden_dim,
    # strides
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_maskb, stride_maski, stride_maskj,
    stride_leftb, stride_lefti, stride_leftj, stride_lefth,
    stride_rightb, stride_righti, stride_rightj, stride_righth,
    stride_out_gateb, stride_out_gatei, stride_out_gatej, stride_out_gateh,
    # weight strides
    stride_wl_dim, stride_wl_hidden,
    stride_wr_dim, stride_wr_hidden,
    stride_wlg_dim, stride_wlg_hidden,
    stride_wrg_dim, stride_wrg_hidden,
    stride_wog_dim, stride_wog_hidden,
    # compile‑time constants
    BLOCK_DIM: tl.constexpr,  # tile size for input dimension
    BLOCK_H: tl.constexpr,    # tile size for hidden dimension
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Compute (b,i,j) from pid
    b = pid // (N * N)
    residual = pid % (N * N)
    i = residual // N
    j = residual % N

    # Base pointers
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    mask_val = 1.0
    # Load mask scalar if provided
    if mask_ptr != 0:
        mask_val = tl.load(mask_ptr + b * stride_maskb + i * stride_maski + j * stride_maskj)

    left_base = left_ptr + b * stride_leftb + i * stride_lefti + j * stride_leftj
    right_base = right_ptr + b * stride_rightb + i * stride_righti + j * stride_rightj
    out_gate_base = out_gate_ptr + b * stride_out_gateb + i * stride_out_gatei + j * stride_out_gatej

    # Loop over hidden dimension in tiles of BLOCK_H
    h_off = 0
    while h_off < hidden_dim:
        hid_range = h_off + tl.arange(0, BLOCK_H)
        hid_mask = hid_range < hidden_dim

        # Accumulators for this tile
        acc_left = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_ogate = tl.zeros([BLOCK_H], dtype=tl.float32)

        d_off = 0
        while d_off < dim:
            dim_range = d_off + tl.arange(0, BLOCK_DIM)
            dim_mask = dim_range < dim

            # Load a block of the normalized input vector
            x = tl.load(x_base + dim_range * stride_xd,
                        mask=dim_mask, other=0.0)          # (BLOCK_DIM,)

            # Promote to (1, BLOCK_DIM) for dot
            x_row = x[None, :]                               # (1, BLOCK_DIM)

            # Load weight tiles
            w_l = tl.load(w_left_ptr
                          + dim_range[:, None] * stride_wl_dim
                          + hid_range[None, :] * stride_wl_hidden,
                          mask=dim_mask[:, None] & hid_mask[None, :],
                          other=0.0)                         # (BLOCK_DIM, BLOCK_H)

            w_r = tl.load(w_right_ptr
                          + dim_range[:, None] * stride_wr_dim
                          + hid_range[None, :] * stride_wr_hidden,
                          mask=dim_mask[:, None] & hid_mask[None, :],
                          other=0.0)

            w_lg = tl.load(w_left_gate_ptr
                           + dim_range[:, None] * stride_wlg_dim
                           + hid_range[None, :] * stride_wlg_hidden,
                           mask=dim_mask[:, None] & hid_mask[None, :],
                           other=0.0)

            w_rg = tl.load(w_right_gate_ptr
                           + dim_range[:, None] * stride_wrg_dim
                           + hid_range[None, :] * stride_wrg_hidden,
                           mask=dim_mask[:, None] & hid_mask[None, :],
                           other=0.0)

            w_og = tl.load(w_out_gate_ptr
                           + dim_range[:, None] * stride_wog_dim
                           + hid_range[None, :] * stride_wog_hidden,
                           mask=dim_mask[:, None] & hid_mask[None, :],
                           other=0.0)

            # dot products accumulate over the BLOCK_DIM dimension
            acc_left = acc_left + tl.dot(x_row, w_l)[0]
            acc_right = acc_right + tl.dot(x_row, w_r)[0]
            acc_lgate = acc_lgate + tl.dot(x_row, w_lg)[0]
            acc_rgate = acc_rgate + tl.dot(x_row, w_rg)[0]
            acc_ogate = acc_ogate + tl.dot(x_row, w_og)[0]

            d_off += BLOCK_DIM

        # Apply sigmoid to gating values
        left_gate = tl.sigmoid(acc_lgate)
        right_gate = tl.sigmoid(acc_rgate)
        out_gate = tl.sigmoid(acc_ogate)

        # Apply mask (broadcasted) and gates to projections
        left_val = acc_left * left_gate * mask_val
        right_val = acc_right * right_gate * mask_val

        # Store results
        tl.store(left_base + hid_range * stride_lefth, left_val, mask=hid_mask)
        tl.store(right_base + hid_range * stride_righth, right_val, mask=hid_mask)
        tl.store(out_gate_base + hid_range * stride_out_gateh, out_gate, mask=hid_mask)

        h_off += BLOCK_H

def custom_kernel(data):
    """
    Triton-accelerated forward pass for the “outgoing” TriMul operator.
    The kernel fuses the first LayerNorm and the five linear projections
    (left_proj, right_proj, left_gate, right_gate, out_gate) together
    with mask application.  The O(N^2 * hidden_dim) pairwise
    multiplication is performed by a batched matmul in PyTorch,
    which is already highly optimized on GPU.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # LayerNorm on the last dimension
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Ensure mask is float32 and on device; if no mask, use ones
    if mask is None or config.get("nomask", False):
        mask_tensor = torch.ones((bs, seq_len, seq_len), dtype=torch.float32, device=device)
    else:
        mask_tensor = mask.to(dtype=torch.float32, device=device)

    # Allocate outputs for the fused kernel

    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # to_out_norm (LayerNorm)

    to_out_norm_bias = weights["to_out_norm.bias"]

        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # Multiply by out_gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_norm, to_out_weight)   # bias is None

    return output

"""
custom_kernel
==============

Triton‑accelerated forward pass for the **outgoing** TriMul operator
(AlphaFold‑3).  

The kernel fuses the first LayerNorm and the five linear projections
(`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`) together
with the optional pairwise mask.  Each CUDA thread processes a single
pairwise position ``(b,i,j)`` and computes all hidden‑dimension outputs in
tiles, re‑using the normalized input vector only once.  The O(N²·H) pairwise
multiplication (``einsum('... i k d, ... j k d -> ... i j d')``) is performed
by a batched matrix‑multiply in PyTorch, which is already highly‑optimised.
The result is passed through a second LayerNorm, gated, and finally projected
back to the original channel dimension.

The implementation respects the constraints:
* Works for B∈{1,2}, N up to 1024, dim∈{128,384}, hidden_dim∈{128,384}.
* Handles the presence or absence of a mask.
* Uses mixed‑precision accumulation (float32) inside Triton.
* Returns a float32 tensor of shape [B, N, N, dim].

"""

import torch
import triton
import triton.language as tl


@triton.jit
def fused_proj_gate_kernel(
    # --- data pointers ----------------------------------------------------
    x_ptr,               # normalized input   [B,N,N,dim]
    mask_ptr,            # mask               [B,N,N]
    left_ptr,            # left projection    [B,N,N,hidden]
    right_ptr,           # right projection   [B,N,N,hidden]
    out_gate_ptr,        # out‑gate (sigmoid) [B,N,N,hidden]

    # --- transposed weight pointers (shape dim x hidden) -----------------
    w_left_ptr,          # left_proj.weight.T
    w_right_ptr,         # right_proj.weight.T
    w_left_gate_ptr,     # left_gate.weight.T
    w_right_gate_ptr,    # right_gate.weight.T
    w_out_gate_ptr,      # out_gate.weight.T

    # --- scalar meta data ------------------------------------------------
    B, N, dim, hidden_dim,
    # --- strides (in elements) -------------------------------------------
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_maskb, stride_maski, stride_maskj,
    stride_leftb, stride_lefti, stride_leftj, stride_lefth,
    stride_rightb, stride_righti, stride_rightj, stride_righth,
    stride_out_gateb, stride_out_gatei, stride_out_gatej, stride_out_gateh,
    # --- weight strides (dim, hidden) ------------------------------------
    stride_wl_dim, stride_wl_hidden,
    stride_wr_dim, stride_wr_hidden,
    stride_wlg_dim, stride_wlg_hidden,
    stride_wrg_dim, stride_wrg_hidden,
    stride_wog_dim, stride_wog_hidden,

    # --- compile‑time constants -----------------------------------------
    BLOCK_DIM: tl.constexpr,  # tile size along the input dimension (dim)
    BLOCK_H: tl.constexpr,    # tile size along the hidden dimension (hidden)
):
    """Fused projection + gating kernel."""
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # --------------------------------------------------------------
    # map pid -> (b,i,j)
    b = pid // (N * N)
    resid = pid % (N * N)
    i = resid // N
    j = resid % N

    # base pointers for this position
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    mask_val = tl.load(mask_ptr + b * stride_maskb + i * stride_maski + j * stride_maskj)

    left_base  = left_ptr  + b * stride_leftb  + i * stride_lefti  + j * stride_leftj
    right_base = right_ptr + b * stride_rightb + i * stride_righti + j * stride_rightj
    out_gate_base = out_gate_ptr + b * stride_out_gateb + i * stride_out_gatei + j * stride_out_gatej

    # --------------------------------------------------------------
    # loop over hidden dimension in tiles of BLOCK_H
    h_off = 0
    while h_off < hidden_dim:
        hid_range = h_off + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        hid_mask   = hid_range < hidden_dim                # bool mask for last tile

        # accumulators for this tile
        acc_left  = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_right = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_lgate = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_rgate = tl.zeros([BLOCK_H], dtype=tl.float32)
        acc_ogate = tl.zeros([BLOCK_H], dtype=tl.float32)

        # ----------------------------------------------------------
        # loop over the input channel dimension
        d_off = 0
        while d_off < dim:
            dim_range = d_off + tl.arange(0, BLOCK_DIM)          # [BLOCK_DIM]
            dim_mask  = dim_range < dim

            # ----- load a tile of the normalized input vector -----
            x = tl.load(x_base + dim_range * stride_xd,
                        mask=dim_mask, other=0.0)               # (BLOCK_DIM,)

            # turn into row vector for tl.dot
            x_row = x[None, :]                                   # (1, BLOCK_DIM)

            # ----- load corresponding weight tiles (dim x hidden) -----
            # left projection
            w_l = tl.load(
                w_left_ptr
                + dim_range[:, None] * stride_wl_dim
                + hid_range[None, :] * stride_wl_hidden,
                mask=dim_mask[:, None] & hid_mask[None, :],
                other=0.0,
            )
            # right projection
            w_r = tl.load(
                w_right_ptr
                + dim_range[:, None] * stride_wr_dim
                + hid_range[None, :] * stride_wr_hidden,
                mask=dim_mask[:, None] & hid_mask[None, :],
                other=0.0,
            )
            # left gate
            w_lg = tl.load(
                w_left_gate_ptr
                + dim_range[:, None] * stride_wlg_dim
                + hid_range[None, :] * stride_wlg_hidden,
                mask=dim_mask[:, None] & hid_mask[None, :],
                other=0.0,
            )
            # right gate
            w_rg = tl.load(
                w_right_gate_ptr
                + dim_range[:, None] * stride_wrg_dim
                + hid_range[None, :] * stride_wrg_hidden,
                mask=dim_mask[:, None] & hid_mask[None, :],
                other=0.0,
            )
            # out gate
            w_og = tl.load(
                w_out_gate_ptr
                + dim_range[:, None] * stride_wog_dim
                + hid_range[None, :] * stride_wog_hidden,
                mask=dim_mask[:, None] & hid_mask[None, :],
                other=0.0,
            )

            # ----- accumulate dot‑products over the BLOCK_DIM -----------------
            acc_left  = acc_left  + tl.dot(x_row, w_l)[0]
            acc_right = acc_right + tl.dot(x_row, w_r)[0]
            acc_lgate = acc_lgate + tl.dot(x_row, w_lg)[0]
            acc_rgate = acc_rgate + tl.dot(x_row, w_rg)[0]
            acc_ogate = acc_ogate + tl.dot(x_row, w_og)[0]

            d_off += BLOCK_DIM
        # end while over dim

        # ----- apply sigmoid to gating values ----------------------------
        left_gate  = tl.sigmoid(acc_lgate)
        right_gate = tl.sigmoid(acc_rgate)
        out_gate   = tl.sigmoid(acc_ogate)

        # ----- mask (broadcasted) and store results --------------------
        left_val  = acc_left  * left_gate  * mask_val
        right_val = acc_right * right_gate * mask_val

        tl.store(left_base  + hid_range * stride_lefth,  left_val, mask=hid_mask)
        tl.store(right_base + hid_range * stride_righth, right_val, mask=hid_mask)
        tl.store(out_gate_base + hid_range * stride_out_gateh, out_gate, mask=hid_mask)

        h_off += BLOCK_H
    # end while over hidden



def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a fused Triton kernel.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * ``input_tensor`` – FloatTensor of shape [B, N, N, C].
        * ``mask_tensor`` – FloatTensor of shape [B, N, N] (or ``None``).
        * ``weights`` – dict containing the model parameters:
          ``norm.weight``, ``norm.bias``,
          ``left_proj.weight``, ``right_proj.weight``,
          ``left_gate.weight``, ``right_gate.weight``,
          ``out_gate.weight``,
          ``to_out_norm.weight``, ``to_out_norm.bias``,
          ``to_out.weight``.
        * ``config`` – dictionary with keys ``dim`` and ``hidden_dim`` (and
          optionally ``nomask``).

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] containing the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input tensor
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Prepare mask (float32).  If ``nomask`` is True or ``mask_tensor`` is None,
    #    use a tensor of ones.
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones((bs, seq_len, seq_len),
                          dtype=torch.float32, device=device)
    else:
        mask = mask_tensor.to(dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 3. Allocate outputs for the fused kernel
    left = torch.empty((bs, seq_len, seq_len, hidden_dim),
                       dtype=torch.float32, device=device)
    right = torch.empty_like(left)
    out_gate = torch.empty_like(left)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 4. Transpose weight matrices to (dim, hidden) for efficient dot‑product
    w_left      = weights["left_proj.weight"].t().contiguous()
    w_right     = weights["right_proj.weight"].t().contiguous()
    w_left_gate = weights["left_gate.weight"].t().contiguous()
    w_right_gate = weights["right_gate.weight"].t().contiguous()
    w_out_gate  = weights["out_gate.weight"].t().contiguous()

    # ------------------------------------------------------------------
    # 5. Compute strides (in elements) for all tensors
    stride_xb, stride_xi, stride_xj, stride_xd = x.stride()
    stride_maskb, stride_maski, stride_maskj = mask.stride()
    stride_leftb, stride_lefti, stride_leftj, stride_lefth = left.stride()
    stride_rightb, stride_righti, stride_rightj, stride_righth = right.stride()
    stride_out_gateb, stride_out_gatei, stride_out_gatej, stride_out_gateh = out_gate.stride()

    # weight strides (dim, hidden)
    stride_wl_dim, stride_wl_hidden   = w_left.stride()
    stride_wr_dim, stride_wr_hidden   = w_right.stride()
    stride_wlg_dim, stride_wlg_hidden = w_left_gate.stride()
    stride_wrg_dim, stride_wrg_hidden = w_right_gate.stride()
    stride_wog_dim, stride_wog_hidden = w_out_gate.stride()

    # ------------------------------------------------------------------
    # 6. Launch the Triton kernel (one program per (b,i,j) position)
    total_positions = bs * seq_len * seq_len
    grid = (total_positions,)

    fused_proj_gate_kernel[grid](
        x_ptr=x.data_ptr(),
        mask_ptr=mask.data_ptr(),
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_gate_ptr=out_gate.data_ptr(),
        w_left_ptr=w_left.data_ptr(),
        w_right_ptr=w_right.data_ptr(),
        w_left_gate_ptr=w_left_gate.data_ptr(),
        w_right_gate_ptr=w_right_gate.data_ptr(),
        w_out_gate_ptr=w_out_gate.data_ptr(),
        B=bs,
        N=seq_len,
        dim=dim,
        hidden_dim=hidden_dim,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xd=stride_xd,
        stride_maskb=stride_maskb,
        stride_maski=stride_maski,
        stride_maskj=stride_maskj,
        stride_leftb=stride_leftb,
        stride_lefti=stride_lefti,
        stride_leftj=stride_leftj,
        stride_lefth=stride_lefth,
        stride_rightb=stride_rightb,
        stride_righti=stride_righti,
        stride_rightj=stride_rightj,
        stride_righth=stride_righth,
        stride_out_gateb=stride_out_gateb,
        stride_out_gatei=stride_out_gatei,
        stride_out_gatej=stride_out_gatej,
        stride_out_gateh=stride_out_gateh,
        stride_wl_dim=stride_wl_dim,
        stride_wl_hidden=stride_wl_hidden,
        stride_wr_dim=stride_wr_dim,
        stride_wr_hidden=stride_wr_hidden,
        stride_wlg_dim=stride_wlg_dim,
        stride_wlg_hidden=stride_wlg_hidden,
        stride_wrg_dim=stride_wrg_dim,
        stride_wrg_hidden=stride_wrg_hidden,
        stride_wog_dim=stride_wog_dim,
        stride_wog_hidden=stride_wog_hidden,
        BLOCK_DIM=32,   # 32‑wide tiles for the channel axis (covers dim≤384)
        BLOCK_H=32,     # 32‑wide tiles for the hidden axis (covers hidden≤384)
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7. Pairwise multiplication (equivalent to the einsum)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as batched matmul per hidden head.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_flat  = left_perm.view(bs * hidden_dim, seq_len, seq_len)   # (B*H,N,N)
    right_flat = right_perm.view(bs * hidden_dim, seq_len, seq_len)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))      # (B*H,N,N)
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)\
                  .permute(0, 2, 3, 1)                               # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8. Second LayerNorm over the hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 9. Apply the out‑gate and the final linear projection back to dim
    out_norm = out_norm * out_gate  # gating
    to_out_weight = weights["to_out.weight"]   # shape [dim, hidden]
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # bias is None

    return output
