@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          batch_stride, M_stride, K_stride,
                          batch_stride_B, N_stride_B, K_stride_B,
                          batch_stride_C, M_stride_C, N_stride_C,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # In-range mask for M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ptrs for A and B
    a_ptrs = A_ptr + pid_batch * batch_stride + offs_m[:, None] * M_stride + tl.arange(0, BLOCK_K)[None, :] * K_stride
    b_ptrs = B_ptr + pid_batch * batch_stride_B + tl.arange(0, BLOCK_K)[:, None] * K_stride_B + offs_n[None, :] * N_stride_B

    # initialization
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load A and B blocks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # move pointers
        a_ptrs += BLOCK_K * K_stride
        b_ptrs += BLOCK_K * K_stride_B
    # write back
    c_ptrs = C_ptr + pid_batch * batch_stride_C + offs_m[:, None] * M_stride_C + offs_n[None, :] * N_stride_C
    tl.store(c_ptrs, acc.to(tl.float16) if C_ptr.dtype == tl.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = x.shape # x shape [B,N,N,dim] (dim may equal dim)
    # Ensure we have mask proper
    if mask is None:
        mask = torch.ones(bs, N, N, device=x.device, dtype=x.dtype)
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask if not nomask? config may provide nomask flag. If mask is provided, multiply.
    if mask is not None:
        # mask shape [B,N,N]; unsqueeze(-1) to broadcast over hidden_dim
        mask_ = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute core multiplication using Triton
    # left shape [B,N,N,hidden_dim]. Need to bring hidden_dim as inner batch.
    # We'll permute to [B, hidden_dim, N, N] then view to [B*hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose right on last two dims
    right_T = right_perm.transpose(2, 3).contiguous()
    B_H = bs * hidden_dim
    # reshape to 3d
    left_mat = left_perm.view(B_H, N, N)
    right_mat = right_T.view(B_H, N, N)

    # output matrix
    out_mat = torch.empty_like(left_mat, dtype=left_mat.dtype)

    # define grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B_H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # compute strides
    stride_batch = N * N
    stride_row = N
    stride_col = 1

    batched_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        stride_batch, stride_row, stride_col,
        stride_batch, stride_row, stride_col,
        stride_batch, stride_row, stride_col,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # Continue: to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

                          stride_Bb, stride_Bi, stride_Bk,
                          stride_Cb, stride_Ci, stride_Ck,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask offgrid
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer offsets base
    a_ptr = A + pid_b * stride_Ab + offs_m[:, None] * stride_Ai + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptr = B + pid_b * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bk + offs_n[None, :] * stride_Ck

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute dynamic offsets for k block
        cur_a = tl.load(a_ptr, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < K - k*BLOCK_K), other=0.0)

        acc += tl.dot(cur_a, cur_b)

        # increment pointers to next k block
        a_ptr += BLOCK_K * stride_Ak
        b_ptr += BLOCK_K * stride_Bk

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Normalize input
    x_norm = F.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)

    # Projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Mask
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Prepare tensors for kernel
    B, N, _, _ = left_h.shape  # B,N,N,H

    # Permute to (B, H, N, N)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    # Transpose right: (B, H, N, N) -> swap last two dims
    right_T = right_perm.transpose(2, 3).contiguous()

    # Flatten batch*head
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_T.view(BH, N, N)

    # Output tensor
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Strides for row-major (batch, M, K) layout
    stride_batch = N * N
    stride_row = N
    stride_col = 1

    batched_matmul_kernel[grid](
        left_mat,               # A
        right_mat,              # B (already transposed)
        out_mat,                # C
        stride_batch, stride_row, stride_col,
        stride_batch, stride_row, stride_col,
        stride_batch, stride_row, stride_col,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to [B, N, N, hidden_dim] (float32)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # to_out_norm + out_gate
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)
    out = out * out_gate

    # final projection
    out = F.linear(out, weights["to_out.weight"])

    return out
```

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bi, stride_Bk,
    stride_Cb, stride_Ci, stride_Ck,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    bbox = tl.program_id(0)  # batch * hidden_dim index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointer to start of A and B blocks for this batch
    a_ptrs = A_ptr + bbox * stride_Ab + offs_m[:, None] * stride_Ai + tl.arange(0, BLOCK_K)[None, :] * stride_Ak
    b_ptrs = B_ptr + bbox * stride_Bb + tl.arange(0, BLOCK_K)[:, None] * stride_Bi + offs_n[None, :] * stride_Bk

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute valid k mask
        valid_k = tl.arange(0, BLOCK_K) < K - k * BLOCK_K

        a = tl.load(a_ptrs, mask=mask_m[:, None] & valid_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & valid_k[:, None], other=0.0)

        acc += tl.dot(a, b)

        # advance pointers for next K block
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bi

    # store result
    c_ptrs = C_ptr + bbox * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Ck
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward implementation using Triton.

The kernel computes the expensive tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    (L_h)_{i,k} @ (R_h)_{j,k}^T   for every head d (hidden_dim)
for each sample in the batch.  The contraction is performed in
half‑precision with a custom Triton GEMM kernel (BLOCK_M/N/K = 64)
to reduce the O(N³·hidden_dim) work‑time.  All other operations
(LayerNorm, Linear projections, gating and final projection) are
handled by PyTorch.

Input:
    data = (x, mask, weights, config)
    x          : [B, N, N, dim]       (float32)
    mask       : [B, N, N] or None   (bool/float)
    weights    : dict of model tensors
    config     : dict with keys ["dim","hidden_dim","nomask"]

Output:
    out        : [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched half‑precision GEMM (A @ Bᵀ) for the TriMul contraction
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    stride_Ab, stride_Ai, stride_Ak,   # A strides: batch, row (i), col (k)
    stride_Bb, stride_Bi, stride_Bk,   # B strides: batch, row (k), col (j) – note B is already transposed
    stride_Cb, stride_Ci, stride_Ck,   # C strides: batch, row (i), col (j)
    M, N, K,                           # matrix sizes (M=N=N, K=N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ
    A, B, C are shaped (batch, M, K), (batch, K, N), (batch, M, N) respectively.
    All tensors are row‑major (C‑contiguous) and stored as fp16.
    Accumulator is fp32.
    """
    batch_id = tl.program_id(0)   # combined batch * hidden_dim
    pid_m    = tl.program_id(1)   # block row
    pid_n    = tl.program_id(2)   # block col

    # ------------------------------------------------------------------- #
    # Offsets for the current block
    # ------------------------------------------------------------------- #
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the block for A and B (k‑dimension at zero)
    a_ptrs = A_ptr + batch_id * stride_Ab \
                     + offs_m[:, None] * stride_Ai \
                     + tl.arange(0, BLOCK_K)[None, :] * stride_Ak

    b_ptrs = B_ptr + batch_id * stride_Bb \
                     + tl.arange(0, BLOCK_K)[:, None] * stride_Bi \
                     + offs_n[None, :] * stride_Bk

    # ------------------------------------------------------------------- #
    # Accumulate over K
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        # Determine which k‑elements are inside the matrix
        k_valid = tl.arange(0, BLOCK_K) < K - k * BLOCK_K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)          # fp16
        b = tl.load(b_ptrs,
                    mask=mask_n[None, :] & k_valid[:, None],
                    other=0.0)          # fp16

        acc += tl.dot(a, b)          # fp32 accumulation

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bi

    # ------------------------------------------------------------------- #
    # Store the result (convert back to fp16)
    # ------------------------------------------------------------------- #
    c_ptrs = C_ptr + batch_id * stride_Cb \
                     + offs_m[:, None] * stride_Ci \
                     + offs_n[None, :] * stride_Ck
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point expected by the test harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x        : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask     : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights  : dict of pre‑loaded torch tensors
        - config   : dict with keys ``dim``, ``hidden_dim`` and optional ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm
    # ------------------------------------------------------------------- #
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # mask shape -> [B,N,N,1] to broadcast over hidden_dim
        mask_broadcast = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid) and gated projections
    # ------------------------------------------------------------------- #
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Triton kernel for the “einsum” contraction
    #    Convert to fp16 and reshape to (B*H, N, N) – a batch of GEMMs.
    # ------------------------------------------------------------------- #
    # Cast to half‑precision to reduce memory traffic.
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape   # [B, N, N, hidden_dim]

    # (B, hidden_dim, N, N) contiguous layout
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    # transpose the last two axes of right so we can do A @ Bᵀ
    right_T = right_perm.transpose(2, 3).contiguous()

    BH = B * hidden_dim                       # total number of GEMMs
    left_mat = left_perm.view(BH, N, N)       # [BH, N, N]
    right_mat = right_T.view(BH, N, N)        # [BH, N, N]

    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # ------------------------------------------------------------------- #
    # 6. Launch Triton kernel
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,                                       # batch * hidden_dim
        (N + BLOCK_M - 1) // BLOCK_M,            # rows
        (N + BLOCK_N - 1) // BLOCK_N,            # cols
    )

    stride_batch = N * N    # stride between consecutive GEMM batches
    stride_row   = N        # row stride inside a GEMM
    stride_col   = 1        # column stride inside a GEMM

    batched_matmul_kernel[grid](
        left_mat,                     # A
        right_mat,                    # B (already transposed)
        out_mat,                      # C
        stride_batch, stride_row, stride_col,   # A strides
        stride_batch, stride_row, stride_col,   # B strides (row‑major)
        stride_batch, stride_row, stride_col,   # C strides
        N, N, N,                     # M, N, K
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # 7. Restore original layout (float32) and continue with the rest of the model
    # ------------------------------------------------------------------- #
    out = (
        out_mat
        .view(B, hidden_dim, N, N)         # [B, hidden_dim, N, N]
        .permute(0, 2, 3, 1)               # [B, N, N, hidden_dim]
        .to(torch.float32)                 # back to fp32 for the remaining ops
    )

    # ------------------------------------------------------------------- #
    # 8. Output LayerNorm, out‑gate, and final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
```
