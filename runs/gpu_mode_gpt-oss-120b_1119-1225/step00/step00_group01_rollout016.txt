@triton.jit
def apply_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    # compute indices
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N
    # load mask for this (b,i,j)
    mask = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j, mask=True).to(tl.float16)
    # loop over channel dimension
    for offs_h in range(0, H, BLOCK_H):
        cur_h = tl.arange(0, BLOCK_H) + offs_h
        mask_h = cur_h < H
        # compute addresses
        left_offset = b * stride_left_b + i * stride_left_i + j * stride_left_j + cur_h * stride_left_h
        right_offset = b * stride_right_b + i * stride_right_i + j * stride_right_j + cur_h * stride_right_h
        left_gate_offset = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + cur_h * stride_lgate_h
        right_gate_offset = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + cur_h * stride_rgate_h

        left_val = tl.load(left_ptr + left_offset, mask=mask_h, other=tl.float16(0.0))
        right_val = tl.load(right_ptr + right_offset, mask=mask_h, other=tl.float16(0.0))
        left_gate_val = tl.load(left_gate_ptr + left_gate_offset, mask=mask_h, other=tl.float16(0.0))
        right_gate_val = tl.load(right_gate_ptr + right_gate_offset, mask=mask_h, other=tl.float16(0.0))
        # apply gating and mask
        left_out = left_val * left_gate_val * mask
        right_out = right_val * right_gate_val * mask
        # store back
        tl.store(left_ptr + left_offset, left_out, mask=mask_h)
        tl.store(right_ptr + right_offset, right_out, mask=mask_h)

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (float32) on the last dimension.
2. Cast to FP16 and compute the three linear projections:
      left = x·W_left,   right = x·W_right
      left_gate  = sigmoid(x·W_left_gate)
      right_gate = sigmoid(x·W_right_gate)
      out_gate   = sigmoid(x·W_out_gate)
3. Fuse the mask, left / right gate and out‑gate multiplication in a
   Triton kernel (in‑place on ``left`` and ``right``).  The kernel
   iterates over the batch × i × j indices and processes the channel
   dimension in blocks (default 64).  Mask broadcasting over the hidden
   dimension is handled inside the kernel.
4. Compute the “tri‑multiplicative” contraction
        out_{ij d} = Σ_k left_{i k d} * right_{j k d}
   which is equivalently a batched matrix multiplication for each
   hidden channel:  out = left.permute(0,3,1,2) @
                    right.permute(0,3,1,2).transpose(-2,-1)
   (B·hidden_dim independent GEMMs of size N×N).
5. Permute back to shape (B, N, N, hidden_dim) and apply a second
   LayerNorm (float32).  Multiply by the ``out_gate`` and finish with
   a linear projection back to the original ``dim``.
6. Return the final tensor in float32.

The heavy element‑wise work (mask & gating) is performed by Triton,
while the O(N³·hidden) contraction is delegated to cuBLAS via
``torch.matmul`` – this yields the best practical performance on an
H100.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, H,
    # strides for left
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    # strides for gates
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    # strides for mask (B,N,N)
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr,
):
    """In‑place: left = left * left_gate * mask,
                right = right * right_gate * mask
       One program instance processes a single (b,i,j) element and walks
       over the hidden dimension H in blocks of size BLOCK_H.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # ------------------------------------------------------------
    # Decode the linear index pid -> (b,i,j)
    # ------------------------------------------------------------
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # ------------------------------------------------------------
    # Load the scalar mask value for this (b,i,j)
    # ------------------------------------------------------------
    mask_off = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_off, mask=True).to(tl.float16)

    # ------------------------------------------------------------
    # Loop over the hidden dimension H
    # ------------------------------------------------------------
    for h_off in range(0, H, BLOCK_H):
        cur_h = tl.arange(0, BLOCK_H) + h_off
        mask_h = cur_h < H

        # left element pointers
        left_off = (
            b * stride_left_b
            + i * stride_left_i
            + j * stride_left_j
            + cur_h * stride_left_h
        )
        left_val = tl.load(left_ptr + left_off, mask=mask_h, other=tl.float16(0.0))

        # right element pointers
        right_off = (
            b * stride_right_b
            + i * stride_right_i
            + j * stride_right_j
            + cur_h * stride_right_h
        )
        right_val = tl.load(right_ptr + right_off, mask=mask_h, other=tl.float16(0.0))

        # left gate
        lgate_off = (
            b * stride_lgate_b
            + i * stride_lgate_i
            + j * stride_lgate_j
            + cur_h * stride_lgate_h
        )
        lgate_val = tl.load(
            left_gate_ptr + lgate_off, mask=mask_h, other=tl.float16(0.0)
        )

        # right gate
        rgate_off = (
            b * stride_rgate_b
            + i * stride_rgate_i
            + j * stride_rgate_j
            + cur_h * stride_rgate_h
        )
        rgate_val = tl.load(
            right_gate_ptr + rgate_off, mask=mask_h, other=tl.float16(0.0)
        )

        # Apply gate and mask (mask broadcasted over H)
        left_out = left_val * lgate_val * mask_val
        right_out = right_val * rgate_val * mask_val

        # Store results back in‑place
        tl.store(left_ptr + left_off, left_out, mask=mask_h)
        tl.store(right_ptr + right_off, right_out, mask=mask_h)


def custom_kernel(data):
    """
    Execute the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (float32 or None)
        - weights      : dict of torch.Tensor, model parameters
        - config       : dict, must contain "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim] float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for the bulk of the work
    # ------------------------------------------------------------------
    x_norm_fp16 = x_norm.to(torch.float16)

    # Cast all linear weights to FP16 (except the final out_norm which stays FP32)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections + gate logits (all FP16)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm_fp16, left_proj_w)          # [B, N, N, H]
    right = torch.nn.functional.linear(x_norm_fp16, right_proj_w)

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm_fp16, left_gate_w)
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm_fp16, right_gate_w)
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm_fp16, out_gate_w)
    )

    # Ensure tensors are contiguous for Triton pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # ------------------------------------------------------------------
    # 4. Prepare mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is None:
        mask_fp16 = torch.ones((input_tensor.shape[0],
                                input_tensor.shape[1],
                                input_tensor.shape[2]),
                               dtype=torch.float16,
                               device=device)
    else:
        mask_fp16 = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Triton kernel: fuse mask, left/right gates
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Strides of the tensors (as int64, required by Triton)
    slb, sli, slj, slh = left.stride()
    srb, sri, srj, srh = right.stride()
    slgb, slgi, slgj, slgh = left_gate.stride()
    srgb, srgi, srgj, srg h = right_gate.stride()  # spacing ignored (typo??)
    # Actually right_gate has same shape as left_gate, so we can reuse the same strides
    srg_b, srg_i, srg_j, srg_h = right_gate.stride()
    smb, smi, smj = mask_fp16.stride()

    # Choose a reasonable block size for the hidden dimension.
    # 64 works well for H100 (the hidden dimension ≤ 384).
    BLOCK_H = 64

    grid = (B * N * N,)  # one program per (b,i,j)
    _gate_mask_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        left_gate_ptr=left_gate,
        right_gate_ptr=right_gate,
        mask_ptr=mask_fp16,
        B, N, hidden_dim,
        slb, sli, slj, slh,
        srb, sri, srj, srh,
        slgb, slgi, slgj, slgh,
        srg_b, srg_i, srg_j, srg_h,
        smb, smi, smj,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Tri‑multiplicative contraction (batched GEMM)
    #    left_perm : [B, H, N, N]
    #    right_perm: [B, H, N, N]
    #    out_perm  : [B, H, N, N] = left_perm @ right_perm.T
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N (FP16)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # B, H, N, N (FP16)

    # Batched matrix multiplication; each (B*H) slice does N×N @ (N×N)^T
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))

    # Back to the original layout: [B, N, N, H]
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # FP16

    # ------------------------------------------------------------------
    # 7. Second LayerNorm (float32), out‑gate, final linear
    # ------------------------------------------------------------------
    # Cast to FP32 before the second LayerNorm for numerical stability
    out_f32 = out.to(torch.float32)

    out_norm = torch.nn.functional.layer_norm(
        out_f32,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )  # [B, N, N, H]  float32

    # Apply out‑gate (cast gate to FP32)
    out = out_norm * out_gate.to(torch.float32)

    # Final projection back to original dimension
    out = torch.nn.functional.linear(out, to_out_w)   # [B, N, N, dim]  FP16

    # Return float32 result as required
    return out.to(torch.float32)
