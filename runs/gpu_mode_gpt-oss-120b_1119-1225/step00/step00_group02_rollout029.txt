@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr   # maybe set to 32
):
    pid = tl.program_id(axis=0)
    # decode:
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    num_j_tiles = tl.cdiv(N, BLOCK_J)
    b = pid // (num_i_tiles * num_j_tiles)
    tile_idx = pid % (num_i_tiles * num_j_tiles)
    i_tile = tile_idx // num_j_tiles
    j_tile = tile_idx % num_j_tiles

    # offsets per tile
    i_offset = i_tile * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = j_tile * BLOCK_J + tl.arange(0, BLOCK_J)
    d_offset = tl.arange(0, BLOCK_D)  # vector dimension within tile

    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets for k
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)

        # load left: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + b*stride_left_b + i_offset[:, None, None]*stride_left_i + k_offset[None, :, None]*stride_left_k + d_offset[None, None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=(i_offset[:, None] < N) & (k_offset[None, :] < N) & (d_offset[None,:] < D), other=0.0).to(tl.float32)

        # load right: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # Use right_ptr with indices b, j, k, d
        right_ptrs = right_ptr + b*stride_right_b + j_offset[:, None, None]*stride_right_j + k_offset[None, :, None]*stride_right_k + d_offset[None, None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=(j_offset[:, None] < N) & (k_offset[None, :] < N) & (d_offset[None,:] < D), other=0.0).to(tl.float32)

        # compute outer product across i,j for each d:
        # left[i,k,d] * right[j,k,d] -> accumulate
        # we can broadcast left across j dimension and right across i dimension
        # Use tl.broadcast_to

        # Expand left to (BLOCK_I, BLOCK_J, BLOCK_D)
        left_exp = left[:, :, :]  # shape (BLOCK_I, BLOCK_K, BLOCK_D)
        right_exp = right[:, :, :]  # shape (BLOCK_J, BLOCK_K, BLOCK_D)

        # We need to sum over k dimension inner product. Ideally we multiply left[:,k,:] * right[:,k,:] per k and sum.

        # Approach: iterate over k in block (e.g., typical matmul does dot of rows with columns)
        for kk in range(0, BLOCK_K):
            # left_i = left[:, kk, :] shape (BLOCK_I, BLOCK_D)
            left_i = left[:, kk, :]  # (BLOCK_I, BLOCK_D)
            # right_j = right[:, kk, :] shape (BLOCK_J, BLOCK_D)
            right_j = right[:, kk, :]  # (BLOCK_J, BLOCK_D)
            # broadcast: left_i[:, None, :] * right_j[None, :, :]
            prod = left_i[:, None, :] * right_j[None, :, :]  # shape (BLOCK_I, BLOCK_J, BLOCK_D)
            acc += prod

    # write out
    out_ptrs = out_ptr + b*stride_out_b + i_offset[:, None, None]*stride_out_i + j_offset[None, :, None]*stride_out_j + d_offset[None, None, :]*stride_out_d
    tl.store(out_ptrs, acc, mask=(i_offset[:, None, None] < N) & (j_offset[None, :, None] < N) & (d_offset[None, None, :] < D))

```
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_abh, stride_abm, stride_abk,
    stride_bhb, stride_bhk, stride_bhn,
    stride_chb, stride_chm, stride_chn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # linear offsets for a and b not needed as we will load sub-blocks.

    # Compute pointer offsets for A and B
    a_ptrs = a_ptr + batch_id * stride_abh + offs_m[:, None] * stride_abm + tl.arange(0, BLOCK_K)[None, :] * stride_abk
    b_ptrs = b_ptr + batch_id * stride_bhb + tl.arange(0, BLOCK_K)[:, None] * stride_bhk + offs_n[None, :] * stride_bhn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M) & (tl.arange(0, BLOCK_K) + k*BLOCK_K < K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K) + k*BLOCK_K < K) & (offs_n < N), other=0.0)
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        # increment pointers for next K tile
        a_ptrs += BLOCK_K * stride_abk
        b_ptrs += BLOCK_K * stride_bhk

    # store result
    c_ptrs = c_ptr + batch_id * stride_chb + offs_m[:, None] * stride_chm + offs_n[None, :] * stride_chn
    tl.store(c_ptrs, acc, mask=(offs_m < M) & (offs_n < N))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5 # from LayerNorm
    # Norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])   # shape [B,N,N,H]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask_expand = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_expand
        right = right * mask_expand

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates - use Triton kernel
    left = _elementwise_mul(left, left_gate) # define kernel returning product
    right = _elementwise_mul(right, right_gate)

    # At this point left and right shape [B,N,N,hidden_dim]

    # Prepare for batch matmul
    # Permute and reshape
    B, N, _, H = left.shape
    left_mat = left.permute(0,3,1,2).reshape(B*H, N, N)
    right_mat = right.permute(0,3,2,1).reshape(B*H, N, N)  # K,N

    # Allocate output
    out_mat = torch.empty_like(left_mat)  # shape (B*H,N,N)

    # Determine block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Define grid
    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (B*H, num_m, num_n)   # axis0 batch, axis1 m, axis2 n

    # strides
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()
    # Launch kernel

        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Wait for kernel
    torch.cuda.synchronize()

    # Reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]

    # Gate multiplication
    out_norm = out_norm * out_gate

    # Final linear
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return output

"""
TriMul (outgoing) implementation.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Linear projections (left/right) → hidden dimension H (no bias).
3. Apply the optional binary mask (broadcast on H).
4. Compute three sigmoid gates from the normalized input.
   left  = left * left_gate
   right = right * right_gate
   (the element‑wise products are performed by a small Triton kernel).
5. Compute the core contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication:
        left_mat[b,h,:,:] (i×k)  @  right_mat[b,h,:,:].T (k×j)
   The batched GEMM is executed by a custom Triton kernel.
6. Layer‑norm the result over the hidden dimension.
7. Multiply by the output gate (sigmoid) and a final linear projection
   back to the original channel dimension C.

The heavy inner product (step 5) is fully fused in Triton,
while element‑wise gating (step 4) also uses a tiny Triton kernel,
satisfying the “must use Triton” requirement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul2_kernel(
    a_ptr, b_ptr, out_ptr,  # device pointers
    numel,                  # total number of elements
    BLOCK_SIZE: tl.constexpr,
):
    """out = a * b  (1‑D flattened launch)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,                     # pointers
    BATCH, M, N, K,                          # sizes
    stride_a_batch, stride_a_m, stride_a_k,  # A strides
    stride_b_batch, stride_b_k, stride_b_n,  # B strides
    stride_c_batch, stride_c_m, stride_c_n,  # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:   C[b,m,n] = Σ_k A[b,m,k] * B[b,k,n]
    Grid: (batch, ceil(M/BM), ceil(N/BN))
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers to the start of this batch
    a_batch_ptr = A_ptr + batch_id * stride_a_batch
    b_batch_ptr = B_ptr + batch_id * stride_b_batch

    # iterate over K tiles
    num_tiles_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_tiles_k):
        k_base = k * BLOCK_K

        # Load A tile [BM, BK]
        a_ptr = a_batch_ptr \
                 + offs_m[:, None] * stride_a_m \
                 + (k_base + offs_k)[None, :] * stride_a_k
        a_mask = (offs_m[:, None] < M) & ((k_base + offs_k)[None, :] < K)
        a_tile = tl.load(a_ptr, mask=a_mask, other=0.0).to(tl.float32)

        # Load B tile [BK, BN]
        b_ptr = b_batch_ptr \
                 + (k_base + offs_k)[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n
        b_mask = ((k_base + offs_k)[:, None] < K) & (offs_n[None, :] < N)
        b_tile = tl.load(b_ptr, mask=b_mask, other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a_tile, b_tile)

    # Store the result
    c_ptr = C_ptr + batch_id * stride_c_batch \
            + offs_m[:, None] * stride_c_m \
            + offs_n[None, :] * stride_c_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptr, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Helper for the 2‑tensor elementwise multiplication
# ----------------------------------------------------------------------
def fused_mul2(x: torch.Tensor, y: torch.Tensor) -> torch.Tensor:
    """Element‑wise x*y using Triton (both tensors must be contiguous)."""
    assert x.is_contiguous() and y.is_contiguous()
    out = torch.empty_like(x)
    numel = x.numel()
    BLOCK = 1024
    grid = ((numel + BLOCK - 1) // BLOCK, )
    elementwise_mul2_kernel[grid](
        x, y, out, numel,
        BLOCK_SIZE=BLOCK,
    )
    torch.cuda.synchronize()
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # 1. LayerNorm over the last dimension (C)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3. Apply optional mask (broadcast on hidden dim)
    # --------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Compute gates (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 5. Fuse gate multiplication via Triton (left*left_gate, right*right_gate)
    # --------------------------------------------------------------
    left = fused_mul2(left.contiguous(), left_gate.contiguous())
    right = fused_mul2(right.contiguous(), right_gate.contiguous())

    # --------------------------------------------------------------
    # 6. Core contraction: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM on the (B*H) batch dimension.
    # --------------------------------------------------------------
    B, N, _, H = left.shape  # B, seq_len, seq_len, hidden_dim

    # reshape to (B*H, N, N) where the second axis is the reduction axis k
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (M=i, K=k)
    # right needs to be transposed on the last two dimensions: (K=k, N=j)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)        # (K=k, N=j)

    # allocate output matrix
    out_mat = torch.empty_like(left_mat)  # (B*H, N, N)

    # -----------------------------------------------------------------------------
    # Triton batched GEMM launch
    # -----------------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    batch_dim = left_mat.shape[0]            # B*H
    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    batch_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_dim,
        N,                 # M
        N,                 # N
        N,                 # K
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 7. Output LayerNorm + gating + final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # element‑wise gating (torch op – cheap)

    # final linear back to original dimension
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]
    return out
