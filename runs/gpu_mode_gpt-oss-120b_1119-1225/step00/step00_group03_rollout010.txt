```python
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, BH,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_rj, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_blocks_m * num_blocks_n)
    pid_rem = pid % (num_blocks_m * num_blocks_n)
    block_m = pid_rem // num_blocks_n
    block_n = pid_rem % num_blocks_n

    # offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers base for this batch
    left_base = left_ptr + batch_id * stride_lb
    right_base = right_ptr + batch_id * stride_rb
    out_base = out_ptr + batch_id * stride_ob

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block (M,K)
        a_ptrs = left_base + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

    c_ptrs = out_base + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
```

```python
def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    eps = 1e-5

    # 1. layernorm on x
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # 2. projections
    left_proj = F.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, right_proj_weight)

    # 3. mask
    nomask = config.get("nomask", False)
    if not nomask and mask is not None:
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # 4. gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # 5. apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    B, N, _, _ = left.shape  # shape B,N,N,H
    # reorder for kernel
    left_t = left.permute(0, 3, 1, 2).contiguous()   # B, H, N, N
    right_t = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_t.view(-1, N, N)
    right_flat = right_t.view(-1, N, N)

    BH = left_flat.shape[0]  # B * H

    # allocate output
    out_flat = torch.empty_like(left_flat)  # float32

    # compute strides
    stride_lb = left_flat.stride(0)
    stride_li = left_flat.stride(1)
    stride_lk = left_flat.stride(2)

    stride_rb = right_flat.stride(0)
    stride_rj = right_flat.stride(1)  # corresponds to i/j dimension
    stride_rk = right_flat.stride(2)

    stride_ob = out_flat.stride(0)
    stride_oi = out_flat.stride(1)
    stride_oj = out_flat.stride(2)

    # launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (BH * tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N),)  # but we cannot compute tl.cdiv outside kernel; use Python math.
    import math
    num_blocks_m = math.ceil(N / BLOCK_M)
    num_blocks_n = math.ceil(N / BLOCK_N)
    grid = (BH * num_blocks_m * num_blocks_n,)

    trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BH,
        stride_lb, stride_li, stride_lk,
        stride_rb, stride_rj, stride_rk,
        stride_ob, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # B,N,N,H

    # 6. to_out_norm & gating
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate

    # 7. final linear
    out = F.linear(out, to_out_weight)    # shape B,N,N,dim

    return out
```

"""
TriMul (outgoing) Triton implementation.

The module performs:
1. LayerNorm over the last channel dimension of the input tensor.
2. Four linear projections without bias (left/right and three gates).
3. Optional masking of the left/right projected tensors.
4. Element‑wise gating (sigmoid) on the projected tensors.
5. The core O(N³) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication of shape (B·H, N, N) @ (B·H, N, N)^T.
   This step is executed in a custom Triton kernel for maximum throughput.
6. LayerNorm, gating and final linear projection back to the original channel size.
All operations are fused as much as possible; only the heavy contraction is
hand‑written in Triton. The kernel works for any batch size (≤2), sequence
length up to 1024 and hidden dimension 128 (or any other size that fits in GPU
memory). The implementation uses float32 throughout (the kernel accumulates in
float32), but the code can be easily switched to half‑precision if desired.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ  (A,B shape: (B·H, N, N))
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    a_ptr,          # left  (B·H, N, N)
    b_ptr,          # right (B·H, N, N)
    c_ptr,          # out   (B·H, N, N)
    # problem size
    N,              # sequence length
    BH,             # B * hidden_dim (total batch dimension)
    # strides (in elements, not bytes)
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_k,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM where the second operand is transposed.
    Computes for each batch `b`:
        C[b, i, j] = Σ_k A[b, i, k] * B[b, j, k]
    This is exactly the `einsum('... i k d, ... j k d -> ... i j d')` used in
    the PyTorch reference.
    """
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) 2‑D block grid inside each batch
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_id   = pid // (num_blocks_m * num_blocks_n)
    block_id   = pid %  (num_blocks_m * num_blocks_n)
    block_i    = block_id // num_blocks_n
    block_j    = block_id %  num_blocks_n

    # Global offsets for the current tile
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch
    a_base = a_ptr + batch_id * stride_a_batch
    b_base = b_ptr + batch_id * stride_b_batch
    c_base = c_ptr + batch_id * stride_c_batch

    # --------------------------------------------------------------
    # 2) Accumulator for the tile
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over K dimension in blocks
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (M, K)
        a_ptrs = a_base + offs_i[:, None] * stride_a_i \
                           + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile transposed on‑the‑fly: shape (K, N)
        # Original layout of B is (i, k); we need (k, i) → use k stride first.
        b_ptrs = b_base + offs_k[:, None] * stride_b_k \
                           + offs_j[None, :] * stride_b_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate product
        acc += tl.dot(a, b)        # (M, K) @ (K, N) → (M, N)

    # --------------------------------------------------------------
    # 4) Write the result
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_c_i \
                     + offs_j[None, :] * stride_c_j
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x : torch.Tensor of shape [B, N, N, dim]
        - mask : torch.Tensor of shape [B, N, N] (may be None)
        - weights : dict of pre‑loaded weight tensors
        - config : dict with keys ``dim`` and ``hidden_dim`` (and optional ``nomask``)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as inputs)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # convenience aliases for the weight tensors
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    eps = 1e-5   # LayerNorm epsilon, matches PyTorch default

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_norm, left_proj_weight)   # [B, N, N, H]
    right_proj = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (mask is broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # ensure same dtype and broadcast on last dim
        mask = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid after a linear projection)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------
    # 5. Apply the gates
    # ------------------------------------------------------------------
    left  = left_proj * left_gate       # [B, N, N, H]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 6. Prepare tensors for the Triton kernel
    #    Layout required: (B·H, N, N) where the two inner dims are (i,k) and (j,k)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # N is already known, just reuse

    # Move hidden dim in front and make the buffer contiguous
    left_t  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse the batch and hidden dimensions
    left_flat  = left_t.view(-1, N, N)    # [BH, N, N]
    right_flat = right_t.view(-1, N, N)   # [BH, N, N]
    BH = left_flat.shape[0]               # B * hidden_dim

    # Output buffer
    out_flat = torch.empty_like(left_flat)   # float32 by default

    # ------------------------------------------------------------------
    # 7. Compute strides (in elements) for the kernel
    # ------------------------------------------------------------------
    stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_i, stride_c_k = out_flat.stride()

    # ------------------------------------------------------------------
    # 8. Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (BH * num_blocks_m * num_blocks_n,)

    trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        BH,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9. Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 10. Final LayerNorm, gate, and linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
