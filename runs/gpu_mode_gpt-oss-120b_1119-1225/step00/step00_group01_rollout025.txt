@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ACTIVATION: tl.constexpr # maybe optional
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + batch_idx * stride_am + offs_m[:, None] * stride_an + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_idx * stride_bm + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_idx * stride_cm + offs_m[:, None] * stride_cn + offs_n[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)  # maybe present.

    # layernorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,H
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask apply
    if not nomask:
        mask = mask.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Batched matmul using triton
    B, N, _, _ = left.shape
    # Flatten heads
    left_ = left.permute(0,3,1,2).contiguous()  # (B,H,N,N)
    right_ = right.permute(0,3,1,2).contiguous()

    # Reshape to (M,N,N)
    M = B * hidden_dim
    left_flat = left_.view(M, N, N)
    right_flat = right_.view(M, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)  # shape M,N,N (float32)

    # Convert to fp16 for kernel
    left_flat_f16 = left_flat.to(torch.float16)
    right_flat_f16 = right_flat.to(torch.float16)
    out_flat_f16 = torch.empty_like(left_flat_f16)

    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (M,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        left_flat_f16,
        right_flat_f16,
        out_flat_f16,
        M, N, N,
        left_flat_f16.stride(0), left_flat_f16.stride(1), left_flat_f16.stride(2),
        right_flat_f16.stride(0), right_flat_f16.stride(2), right_flat_f16.stride(1), # note swapped k and n
        out_flat_f16.stride(0), out_flat_f16.stride(1), out_flat_f16.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Convert back to float32
    out_flat = out_flat_f16.to(torch.float32)

    # Reshape back to (B,N,N,hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out = out * out_gate
    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_an, stride_ak,
    stride_bm, stride_bk, stride_bn,
    stride_cm, stride_cn, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_idx * stride_am + offs_m[:, None] * stride_an + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

"""
TriMul (outgoing) forward pass implemented with a fused
batched matrix‑multiply kernel in Triton.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (last dimension = dim).
2. Linear projections → left/right (shape [B, N, N, H]).
3. Optional mask multiplication (mask ∈ {0,1} broadcasted on H).
4. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   same normalized tensor and apply them element‑wise.
5. Contraction:
      out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   is a batched matrix‑multiply for each hidden channel.  The tensors are
   permuted to (B, H, N, N) and made contiguous, then the leading
   dimension (B·H) is flattened.  A custom Triton kernel computes
   (M,N,N) × (M,N,N)ᵀ → (M,N,N) with tile sizes (64,64,64).
6. Layer‑norm over the hidden dimension, apply the output gate,
   and a final linear mapping back to the original dim.

The kernel works on fp16 for bandwidth‑limited loads and accumulates in
fp32 for numerical stability.  The final result is cast back to float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                 # pointers
    M, N, K,                             # matrix sizes
    stride_am, stride_an, stride_ak,     # strides for A (M, N, K)
    stride_bm, stride_bk, stride_bn,     # strides for B (M, K, N) – note swapped K/N
    stride_cm, stride_cn, stride_cj,     # strides for C (M, N, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[m, i, j] = Σₖ A[m, i, k] * B[m, k, j]"""
    batch_idx = tl.program_id(0)          # M dimension (B·H)
    pid_i = tl.program_id(1)              # tile over i (rows)
    pid_j = tl.program_id(2)              # tile over j (cols)

    # Offsets for the output tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M, i, k)
        a_ptrs = a_ptr + batch_idx * stride_am \
                 + offs_i[:, None] * stride_an \
                 + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        # Load B tile (M, k, j) – note that strides are swapped for the transpose view
        b_ptrs = b_ptr + batch_idx * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_j[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        # Accumulate the dot product
        acc += tl.dot(a, b)

    # Write back C tile (M, i, j) as fp16
    c_ptrs = c_ptr + batch_idx * stride_cm \
             + offs_i[:, None] * stride_cn \
             + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor [B, N, N, dim] (float32)
            - mask:  torch.Tensor [B, N, N]   (bool or float)
            - weights: dict of model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---------- 1. Input LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ---------- 2. Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, H]

    # ---------- 3. Optional mask ----------
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)           # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4. Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Batched contraction (Trident) ----------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, H

    # Move hidden dimension next to batch and make data contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    M = B * hidden_dim                      # flattened batch * head dimension
    left_flat = left_perm.view(M, N, N)     # (M, N, N)
    right_flat = right_perm.view(M, N, N)   # (M, N, N)

    # Cast to fp16 for the kernel (memory‑bandwidth friendly)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)  # (M, N, N)

    # Tile sizes – chosen for H100 (good occupancy and shared‑memory usage)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        M,
        N,
        N,
        # strides for A (M, i, k)
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        # strides for B (M, k, j) – note the swap of last two axes
        right_fp16.stride(0),
        right_fp16.stride(2),
        right_fp16.stride(1),
        # strides for C (M, i, j)
        out_fp16.stride(0),
        out_fp16.stride(1),
        out_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to float32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # Reshape back to [B, N, N, H]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------- 6. Output LayerNorm ----------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ---------- 7. Output gate ----------
    out = out * out_gate

    # ---------- 8. Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
