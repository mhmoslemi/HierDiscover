@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    D_PER_ITER: tl.constexpr
):
    pid = tl.program_id(0)
    # compute total tiles per batch
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    batch = pid // (num_i * num_j)
    pid_rem = pid % (num_i * num_j)
    i_block = pid_rem // num_j
    j_block = pid_rem % num_j

    # offsets within block
    i = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    j = j_block * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i < N
    j_mask = j < N

    # accumulator shape (BLOCK_I, BLOCK_J, D_PER_ITER)
    # compute number of d_tiles
    num_d_tiles = D // D_PER_ITER   # D divisible by D_PER_ITER

    # For each d_tile: we will accumulate into local acc
    for d_tile in range(num_d_tiles):
        # base of hidden dimension
        d_base = d_tile * D_PER_ITER
        # allocate acc for this d_tile
        acc = tl.zeros((BLOCK_I, BLOCK_J, D_PER_ITER), dtype=tl.float32)

        # Loop over K
        for k_base in range(0, N, BLOCK_K):
            k = k_base + tl.arange(0, BLOCK_K)
            k_mask = k < N

            # Load left block for all D_PER_ITER values (broadcast)
            # Build offsets for left: shape (BLOCK_I, BLOCK_K, D_PER_ITER)
            # Use broadcasting
            i_idx = i[:, None, None]   # (BLOCK_I,1,1)
            k_idx = k[None, :, None]   # (1,BLOCK_K,1)
            d_idx = d_base + tl.arange(0, D_PER_ITER)[None, None, :]  # (1,1,D_PER_ITER)

            left_offset = (i_idx * stride_li) + (k_idx * stride_lk) + (d_idx * stride_ld) + batch * stride_lb
            left_tile = tl.load(left_ptr + left_offset, mask = i_mask[:, None, None] & k_mask[None, :, None] & (d_idx < D), other=0.0)
            # cast to fp32
            left_tile = left_tile.to(tl.float32)

            # Load right block transposed (K,J,D_PER_ITER)
            # we want shape (BLOCK_K, BLOCK_J, D_PER_ITER)
            k_idx_r = k[:, None, None]  # (BLOCK_K,1,1)
            j_idx = j[None, :, None]    # (1,BLOCK_J,1)
            right_offset = (k_idx_r * stride_rk) + (j_idx * stride_rj) + (d_idx * stride_rd) + batch * stride_rb
            right_tile = tl.load(right_ptr + right_offset, mask = (k_idx_r < N) & j_mask[None, :, None] & (d_idx < D), other=0.0)
            right_tile = right_tile.to(tl.float32)

            # Now compute dot product per hidden d:
            # Use tl.dot: left_tile is (BLOCK_I, BLOCK_K, D_PER_ITER), right_tile (BLOCK_K, BLOCK_J, D_PER_ITER)
            # We'll loop over d in compile-time D_PER_ITER
            for d_off in range(D_PER_ITER):
                lt = left_tile[:, :, d_off]   # shape (BLOCK_I, BLOCK_K)
                rt = right_tile[:, :, d_off]  # shape (BLOCK_K, BLOCK_J)
                acc[:, :, d_off] += tl.dot(lt, rt)   # shape (BLOCK_I, BLOCK_J)

        # store acc to output
        # compute out offsets: shape (BLOCK_I, BLOCK_J, D_PER_ITER)
        i_idx_out = i[:, None, None]
        j_idx_out = j[None, :, None]
        d_idx_out = d_base + tl.arange(0, D_PER_ITER)[None, None, :]
        out_offset = (i_idx_out * stride_oi) + (j_idx_out * stride_oj) + (d_idx_out * stride_od) + batch * stride_ob
        # tl.store expects pointer same shape as data.
        tl.store(out_ptr + out_offset, acc, mask = i_mask[:, None, None] & j_mask[None, :, None] & (d_idx_out < D))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if not nomask:
        mask = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask
    # gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # apply gates
    left = left * left_gate
    right = right * right_gate

    # convert to half for kernel
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_f16.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_f16.device)

    # strides
    stride_lb, stride_li, stride_lk, stride_ld = left_f16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_f16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # grid
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    D_PER_ITER = 8
    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    total_blocks = B * num_i * num_j
    grid = (total_blocks,)

    tri_mul_outgoing_kernel[grid](
        left_f16, right_f16, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
        D_PER_ITER=D_PER_ITER, D=hidden_dim,
        num_warps=4
    )

    # post-processing
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

    stride_rb, stride_rj, stride_rk, stride_rd,

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # projection
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        m = mask.to(left.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # convert to half
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_fp16.device)

    # strides
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # block sizes
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64

    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J

    grid = (B * num_i, num_j, hidden_dim)

    tri_mul_outgoing_kernel[grid](
        left_fp16, right_fp16, out,
        B, N,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K,
        num_warps=8
    )

    # post-processing
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) kernel for AlphaFold3 style pairwise updates.
The forward pass consists of:

1. LayerNorm over the channel dimension (dim).
2. Linear projections to a hidden dimension (hidden_dim).
3. Optional pairwise mask applied to the projected tensors.
4. Gating (sigmoid) applied to the projected tensors.
5. Core "triangle multiplicative" update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   Implemented as a fused Triton kernel that computes a block of the
   (i,j) output plane for a single hidden channel `d` using a tiled
   matrix‑multiply (tl.dot).  The kernel is launched with a 3‑D grid:
   (batch * ceil(N/BI), ceil(N/BJ), hidden_dim).
6. LayerNorm over the hidden dimension, multiplicative gating and a final
   linear projection back to the original channel dimension.

All heavy computations (steps 1‑4 are handled by PyTorch’s highly‑optimized
kernels; step 5 is the only part written in Triton).  The Triton kernel
operates on FP16 data and accumulates in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,                     # fp16 left tensor  [B,N,N,hidden]
    right_ptr,                    # fp16 right tensor [B,N,N,hidden]
    out_ptr,                      # fp32 output tensor [B,N,N,hidden]
    N,                            # sequence length (constexpr)
    stride_lb, stride_li, stride_lk, stride_ld,   # strides for left
    stride_rb, stride_rj, stride_rk, stride_rd,   # strides for right
    stride_ob, stride_oi, stride_oj, stride_od,   # strides for out
    BLOCK_I: tl.constexpr,        # tile size in the i‑dimension
    BLOCK_J: tl.constexpr,        # tile size in the j‑dimension
    BLOCK_K: tl.constexpr,        # tile size in the reduction (k) dimension
):
    """Compute a tile of the TriMul update for a single hidden channel.

    Each program instance is identified by three IDs:
        pid_x : combined (batch * i‑tile) index
        pid_y : j‑tile index
        pid_z : hidden channel index d
    The kernel loads a (BLOCK_I×BLOCK_K) tile from `left` and a
    (BLOCK_K×BLOCK_J) tile from `right`, performs a matrix multiplication
    (tl.dot) and accumulates the result into an (BLOCK_I×BLOCK_J) tile of
    `out`.  Boundary handling is performed with masks; the input tensors are
    assumed to be contiguous in memory.
    """
    pid_x = tl.program_id(0)  # batch * i‑tile
    pid_y = tl.program_id(1)  # j‑tile
    pid_z = tl.program_id(2)  # hidden channel d

    # --------------------------------------------------------------
    # Tile layout
    # --------------------------------------------------------------
    num_i_tiles = tl.cdiv(N, BLOCK_I)               # tiles along i
    batch      = pid_x // num_i_tiles                # which batch element
    i_tile_idx = pid_x % num_i_tiles                 # tile index in i‑dim
    j_tile_idx = pid_y                               # tile index in j‑dim
    d          = pid_z                               # hidden channel

    # absolute coordinates for this tile
    i = i_tile_idx * BLOCK_I + tl.arange(0, BLOCK_I)   # (BLOCK_I,)
    j = j_tile_idx   * BLOCK_J + tl.arange(0, BLOCK_J) # (BLOCK_J,)

    i_mask = i < N
    j_mask = j < N

    # --------------------------------------------------------------
    # Accumulator for the (i,j) tile (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        k = k_base + tl.arange(0, BLOCK_K)               # (BLOCK_K,)
        k_mask = k < N

        # ---- load left tile : shape (BLOCK_I, BLOCK_K) -----------------
        # offsets = batch*stride_lb + i*stride_li + k*stride_lk + d*stride_ld
        off_left = (batch * stride_lb
                    + i[:, None] * stride_li
                    + k[None, :] * stride_lk
                    + d * stride_ld)
        left = tl.load(left_ptr + off_left,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)                         # fp16

        # ---- load right tile (transposed) : shape (BLOCK_K, BLOCK_J) --
        # offsets = batch*stride_rb + j*stride_rj + k*stride_rk + d*stride_rd
        off_right = (batch * stride_rb
                     + j[None, :] * stride_rj
                     + k[:, None] * stride_rk
                     + d * stride_rd)
        right = tl.load(right_ptr + off_right,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0)                        # fp16

        # ---- matrix multiplication (BLOCK_I x BLOCK_K) * (BLOCK_K x BLOCK_J)
        acc += tl.dot(left, right)                        # fp32 accumulate

    # --------------------------------------------------------------
    # Write the result tile to the output tensor
    # --------------------------------------------------------------
    off_out = (batch * stride_ob
               + i[:, None] * stride_oi
               + j[None, :] * stride_oj
               + d * stride_od)
    tl.store(out_ptr + off_out,
             acc,
             mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul)
    using a fused Triton kernel for the N³ reduction.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool or 0/1) – optional
        - weights      : dict of model parameters (float32 tensors)
        - config       : dict with keys ``dim`` and ``hidden_dim`` and optional
                         ``nomask`` flag.
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs -------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # --------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------------
    # 2. Linear projections to hidden_dim
    # --------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------
    # 3. Apply optional mask (broadcast over the hidden dimension)
    # --------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to the same dtype as the projections
        m = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * m
        right = right * m

    # --------------------------------------------------------------------
    # 4. Gating (sigmoid) and element‑wise multiplication
    # --------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # 5. TriMul core: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented with Triton (FP16 tiles, FP32 accumulation)
    # --------------------------------------------------------------------
    # Convert to FP16 for the kernel (memory‑bandwidth friendly)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape
    device = left_fp16.device

    # Allocate output tensor (FP32 accumulation)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (int) – layout is (B, N, N, hidden)
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()

    # Tile sizes – chosen to fit registers/shared memory on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64

    # Grid dimensions
    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    grid = (B * num_i_tiles, num_j_tiles, hidden_dim)   # (x, y, z)

    # Launch kernel
    tri_mul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        N,                                 # constexpr sequence length
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,                       # H100 can efficiently use many warps
    )

    # --------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, gating, final linear projection
    # --------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out_norm * out_gate                     # element‑wise gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
