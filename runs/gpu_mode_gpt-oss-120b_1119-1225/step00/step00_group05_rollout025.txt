def custom_kernel(data):
    # unpack data
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # optional config has "nomask" flag?
    nomask = config.get("nomask", False) # if True means no mask

    # get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # layernorm over last dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape bs,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # mask
    if not nomask:
        # mask shape bs,N,N -> unsqueeze(-1)
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # apply gating
    left = left * left_gate
    right = right * right_gate

    # ensure contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()

    # compute out = left @ right^T for each hidden channel via Triton
    out = trmul_matmul_transpose(left, right)  # returns bs,N,N,hidden_dim

    # apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out_gate multiplication
    out = out * out_gate

    # final linear map to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape bs,N,N,dim

    return out

def trmul_out_kernel(
    A, B, C,
    M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row_T, stride_b_col_T,
    stride_c_batch, stride_c_row, stride_c_col,

"""
custom_kernel
==============

Implements the *outgoing* TriMul operator (AlphaFold‑3) with a fused
Triton kernel for the expensive N³ “tri‑multiply” part.

Algorithm
---------
1. Layer‑Norm the input tensor (float32).
2. Linear projections (left/right) and three gating linear layers.
3. Apply optional mask and element‑wise sigmoid gates.
4. Compute out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d] for every batch
   and hidden channel `d`.  This is a batched matrix multiplication
   C = L @ Rᵀ where each `d` is a separate matrix.  The heavy O(N³)
   work is performed by a custom Triton kernel that loads half‑precision
   (`float16`) tiles of L and R, accumulates in `float32` and writes the
   result back as `float32`.
5. Layer‑Norm over the hidden dimension, multiply by the output gate,
   and a final linear projection back to the original channel size.

The kernel is written for a 3‑D batched tensor (batch*hidden, M=N, K=N)
and uses block sizes 64×64×64, which fits well on an H100.
"""

import torch
import triton
import triton.language as tl
from torch.nn.functional import linear, layer_norm, sigmoid

# ----------------------------------------------------------------------
# Triton kernel: out = A @ (Bᵀ)   (A,B shape = [B, M, K] , C = [B, M, N])
# ----------------------------------------------------------------------
@triton.jit
def trmul_out_kernel(
    A_ptr, B_ptr, C_ptr,                # pointers
    M, N, K,                            # matrix dimensions (all = seq_len)
    stride_a_batch, stride_a_row, stride_a_col,   # A strides
    stride_b_batch, stride_b_row_T, stride_b_col_T,   # Bᵀ strides
    stride_c_batch, stride_c_row, stride_c_col,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # Decompose the flat program id to (batch, block_m, block_n)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_matrix = num_blocks_m * num_blocks_n

    batch_id = pid // blocks_per_matrix
    block_id = pid % blocks_per_matrix
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # --------------------------------------------------------------
    # Base pointers for this batch
    # --------------------------------------------------------------
    a_batch = batch_id * stride_a_batch
    b_batch = batch_id * stride_b_batch
    c_batch = batch_id * stride_c_batch

    # --------------------------------------------------------------
    # Accumulator (float32 for numerical stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension K
    # --------------------------------------------------------------
    num_k = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k):
        k = k_iter * BLOCK_K
        cur_k = k + offs_k

        # Load A tile (shape BLOCK_M x BLOCK_K)
        a_mask = (offs_m[:, None] < M) & (cur_k[None, :] < K)
        a_ptrs = A_ptr + a_batch + (offs_m[:, None] * stride_a_row
                                    + cur_k[None, :] * stride_a_col)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0, dtype=tl.float16)

        # Load Bᵀ tile (shape BLOCK_K x BLOCK_N)
        b_mask = (cur_k[:, None] < K) & (offs_n[None, :] < N)
        b_ptrs = B_ptr + b_batch + (cur_k[:, None] * stride_b_row_T
                                    + offs_n[None, :] * stride_b_col_T)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0, dtype=tl.float16)

        # Accumulate in float32
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # --------------------------------------------------------------
    # Write back the result tile
    # --------------------------------------------------------------
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    c_ptrs = C_ptr + c_batch + (offs_m[:, None] * stride_c_row
                                + offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc, mask=c_mask)


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def trmul_matmul_transpose(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    """
    left/right: [bs, N, N, hidden] (float16)
    Returns out: [bs, N, N, hidden] (float32) where
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    """
    bs, N, _, hidden = left_fp16.shape
    total = bs * hidden            # treat (bs, hidden) as a single batch dimension
    # reshape to [total, N, N]
    L = left_fp16.permute(0, 3, 1, 2).contiguous().view(total, N, N)
    R = right_fp16.permute(0, 3, 1, 2).contiguous().view(total, N, N)

    # output buffer (float32)
    O = torch.empty((total, N, N), dtype=torch.float32, device=L.device)

    # stride information
    stride_a_batch = L.stride(0)
    stride_a_row   = L.stride(1)    # distance between rows (M stride)
    stride_a_col   = L.stride(2)    # distance between columns (K stride)

    stride_b_batch   = R.stride(0)
    # Bᵀ: rows correspond to original columns, columns to original rows
    stride_b_row_T = R.stride(2)    # original col stride -> row stride of Bᵀ
    stride_b_col_T = R.stride(1)    # original row stride -> col stride of Bᵀ

    stride_c_batch = O.stride(0)
    stride_c_row   = O.stride(1)
    stride_c_col   = O.stride(2)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (total * num_blocks_m * num_blocks_n,)

    trmul_out_kernel[grid](
        L, R, O,
        N, N, N,
        stride_a_batch, stride_a_row, stride_a_col,
        stride_b_batch, stride_b_row_T, stride_b_col_T,
        stride_c_batch, stride_c_row, stride_c_col,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # reshape back to [bs, N, N, hidden]
    out = O.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing variant) with a fused
    Triton kernel for the N³ multiplication.
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Layer‑norm over the last (channel) dimension
    # ------------------------------------------------------------------
    x = layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = linear(x, weights["left_proj.weight"])   # [bs, N, N, hidden]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Gating linear layers + sigmoid
    # ------------------------------------------------------------------
    left_gate  = sigmoid(linear(x, weights["left_gate.weight"]))
    right_gate = sigmoid(linear(x, weights["right_gate.weight"]))
    out_gate   = sigmoid(linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Optional mask (mask shape: [bs, N, N])
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)          # [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy N³ kernel
    # ------------------------------------------------------------------
    left_fp16  = left.contiguous().to(torch.float16)
    right_fp16 = right.contiguous().to(torch.float16)

    # ------------------------------------------------------------------
    # Tri‑multiply: out = Σₖ left[...,k,:] * right[...,k,:]ᵀ   (via Triton)
    # ------------------------------------------------------------------
    out = trmul_matmul_transpose(left_fp16, right_fp16)   # [bs, N, N, hidden] (float32)

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden dimension)
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # Multiply by the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection back to original channel size
    # ------------------------------------------------------------------
    out = linear(out, weights["to_out.weight"])   # [bs, N, N, dim]

    return out
