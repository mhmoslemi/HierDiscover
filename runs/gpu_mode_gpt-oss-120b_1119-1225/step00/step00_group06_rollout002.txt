def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # optionally get other config e.g., eps
    eps = 1e-5
    
    # 1. LayerNorm on input (dim channel)
    x_norm = torch.empty_like(input_tensor)
    # define helper function to launch layernorm kernel
    layernorm(x=input_tensor, out=x_norm, weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # 2. Linear projections (no bias)
    # left = x_norm @ left_proj_weight.T (with shape [hidden_dim, dim])
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # 3. Apply mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # 4. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    # apply gates
    left = left * left_gate
    right = right * right_gate
    
    # 5. Compute out via batched matmul across hidden_dim.
    B = input_tensor.shape[0]; N = input_tensor.shape[1]
    # reshape for batched gemm
    # left: (B, N, N, H) -> (B*H, N, N)
    left_reshape = left.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    right_reshape = right.permute(0,3,1,2).reshape(B*hidden_dim, N, N)
    # matmul left * right^T
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1,2))  # (B*H, N, N)
    # reshape back
    out = out_reshape.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # [B, N, N, H]

    # 6. to_out_norm
    # compute LayerNorm over hidden_dim dim (i.e., last dim).
    out_norm = torch.empty_like(out)
    layernorm(x=out, out=out_norm, weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # 7. Apply out_gate (broadcast)
    out_norm = out_norm * out_gate

    # 8. Final linear projection to dim
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # apply first layernorm via Triton
    x_norm = layernorm_triton(input_tensor, weights["norm.weight"], weights["norm.bias"], eps=eps, block_c=64)

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask: if present and not nomask
    if mask is not None:

        left = left * mask_exp
        right = right * mask_exp

    # gates

    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))

"""
custom_kernel – Triton‑accelerated forward of the TriMul (outgoing) block.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (dim).
   Implemented with a custom Triton kernel that reduces over the channel
   axis and applies the learned weight & bias.
2. Compute five bias‑free linear projections of the normalized tensor:
   * left_proj    – shape [..., hidden_dim]
   * right_proj   – shape [..., hidden_dim]
   * left_gate    – shape [..., hidden_dim] → sigmoid
   * right_gate   – shape [..., hidden_dim] → sigmoid
   * out_gate     – shape [..., hidden_dim] → sigmoid
3. Apply optional pairwise mask (broadcast over the channel axis) and
   element‑wise gates:
   ```
   left  = left_proj  * mask * left_gate
   right = right_proj * mask * right_gate
   ```
4. Core cubic update:
   For each hidden channel `h` compute
   `out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]`.
   This is equivalent to a batched matrix multiplication
   `L_h @ R_hᵀ` for every hidden dimension.
   We fuse it by reshaping to `(B*hidden_dim, N, N)` and calling
   `torch.bmm`, which uses cuBLAS (highly optimized on H100).
5. Layer‑norm the result over the hidden dimension (again via Triton).
6. Multiply by `out_gate`, apply the final linear `to_out` (bias‑free)
   and return the tensor of shape `[B, N, N, dim]`.

Only the two LayerNorms are executed with Triton – this satisfies the
requirement of “at least part of the operations in a kernel” while the
expensive cubic contraction is handled by a highly tuned cuBLAS batched
GEMM.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm kernel (generic for any trailing dimension)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,                # pointer to input  (contiguous, shape [..., C])
    out_ptr,              # pointer to output (same shape as input)
    weight_ptr,           # pointer to weight  (shape [C])
    bias_ptr,             # pointer to bias    (shape [C])
    C,                    # number of features (int)
    eps,                  # epsilon for stability (float)
    BLOCK_C: tl.constexpr # tile size for the reduction loop
):
    pid = tl.program_id(0)                # one program per vector (slice over C)
    # Offset to the start of the current vector (contiguous in memory)
    base = pid * C

    # ------------------------------------------------------------------
    # First pass: compute mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    c = 0
    while c < C:
        offs = tl.arange(0, BLOCK_C) + c          # [0, BLOCK_C)
        mask = offs < C
        # Load a tile of the vector
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)
        c += BLOCK_C

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = tl.float32(1.0) / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized values multiplied by weight & bias
    # ------------------------------------------------------------------
    c = 0
    while c < C:
        offs = tl.arange(0, BLOCK_C) + c
        mask = offs < C
        x = tl.load(x_ptr + base + offs, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs,   mask=mask, other=0.0).to(tl.float32)

        out = (x - mean) * inv_std * w + b
        tl.store(out_ptr + base + offs, out, mask=mask)
        c += BLOCK_C


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5,
                      block_c: int = 64) -> torch.Tensor:
    """
    Triton‑based LayerNorm over the last dimension.
    Input must be contiguous and of floating point dtype.
    """
    assert x.is_contiguous(), "Input to _layernorm_triton must be contiguous"
    C = x.shape[-1]
    total_vectors = x.numel() // C
    out = torch.empty_like(x)

    grid = (total_vectors,)
    # Number of warps is a heuristic; 4 works well for small C.
    _layernorm_kernel[grid](
        x,
        out,
        weight,
        bias,
        C,
        eps,
        BLOCK_C=block_c,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block with Triton‑accelerated LayerNorms.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors containing all linear & norm params
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channels (dim)
    # ------------------------------------------------------------------
    x_norm = _layernorm_triton(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block_c=64,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (bias‑free) and gates
    # ------------------------------------------------------------------
    # Projections to hidden dimension
    left  = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Optional mask – broadcast to hidden dimension
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates (sigmoid after linear)
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Cubic update: batched GEMM over the hidden dimension
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N = input_tensor.shape[0], input_tensor.shape[1]

    # Reshape to (B*H, N, N) -> treat each hidden channel as a separate batch
    left_bh = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_bh = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiply with transposed right operand
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout [B, N, N, H]
    out = (
        out_bh.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 4️⃣ LayerNorm over hidden dimension + out‑gate
    # ------------------------------------------------------------------
    out_norm = _layernorm_triton(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block_c=64,
    )
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
