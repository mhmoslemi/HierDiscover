def tri_mul_gemm_kernel(left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hi = tl.program_id(2)  # flattened batch*hidden

    # compute batch and hidden index
    b = pid_hi // H
    h = pid_hi % H

    # compute offsets
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # mask for out of bounds
    mask_i = i < N
    mask_j = j < N

    # compute base pointers for this batch and hidden
    left_base = left_ptr + b*stride_bl + h*stride_hl
    right_base = right_ptr + b*stride_br + h*stride_hr
    out_base = out_ptr + b*stride_bo + h*stride_ho

    # load left tile
    # left has shape (N, N) indexed by i,k
    left_ptrs = left_base + i[:, None] * stride_il + k[None, :] * stride_kl
    # mask for left load
    mask_left = mask_i[:, None] & (k < N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k block offset
        k_offset = kk * BLOCK_K
        cur_k = k + k_offset
        # mask for cur_k
        mask_k = cur_k < N

        # rederive pointers for cur block
        left_ptrs = left_base + i[:, None] * stride_il + (cur_k)[None, :] * stride_kl
        # right tile (K,N) shape
        # right indexed as (j,k) -> we need as (k,j) for dot, so we use k as rows, j as columns.

    out_ptrs = out_base + i[:, None] * stride_io + j[None, :] * stride_jo

    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Determine batch and hidden index
    b = pid_z // H
    h = pid_z - b * H

    # Compute start indices for the tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # Base pointers for this (b,h) slice
    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left tile (M, K)
        left_off = left_base + i[:, None] * stride_il + cur_k[None, :] * stride_kl
        left_tile = tl.load(left_off,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # Load right tile as (K, N) by swapping the k/j axes
        # right original layout: (j, k) -> we need (k, j) for dot
        right_off = right_base + cur_k[:, None] * stride_kr + j[None, :] * stride_jr
        right_tile = tl.load(right_off,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # Accumulate partial product
        acc += tl.dot(left_tile, right_tile)

    # Store result
    out_off = out_base + i[:, None] * stride_io + j[None, :] * stride_jo
    tl.store(out_off,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def tri_mul_gemm(left: torch.Tensor, right: torch.Tensor, block_m=64, block_n=64, block_k=32):
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)  # shape (B,N,N,H)
    grid = (triton.cdiv(N, block_m),
            triton.cdiv(N, block_n),
            B * H)
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_bl=left.stride(0),
        stride_il=left.stride(1),
        stride_kl=left.stride(2),
        stride_hl=left.stride(3),
        stride_br=right.stride(0),
        stride_jr=right.stride(1),
        stride_kr=right.stride(2),
        stride_hr=right.stride(3),
        stride_bo=out.stride(0),
        stride_io=out.stride(1),
        stride_jo=out.stride(2),
        stride_ho=out.stride(3),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
        num_warps=4
    )

```python
def custom_kernel(data):
    """
    Triton-accelerated forward of the TriMul module (outgoing version).
    Performs layer‑norm, linear projections, gating, a fused batched GEMM
    (the ∑_k left[..., i, k, d] * right[..., j, k, d] contraction) via a
    custom Triton kernel, followed by output norm, gating and final linear.
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ---------------- LayerNorm on input ----------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------------- Linear projections ----------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ---------------- Mask handling ----------------
    if not config.get("nomask", True):
        # mask shape [B, N, N]; broadcast to [B,N,N,1]
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ---------------- Gating ----------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------------- Core contraction via Triton ----------------
    # Compute out = sum_k left[..., i, k, d] * right[..., j, k, d]
    out = tri_mul_gemm(left, right)

    # ---------------- Output LayerNorm ----------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ---------------- Apply output gate ----------------
    out = out * out_gate

    # ---------------- Final linear ----------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out
```

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy part of the TriMul module is the contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which is a batched GEMM over the sequence dimension (N×N) for each hidden
channel.  This kernel performs the above contraction for all batches and
all hidden channels in a single launch, using tiling (BLOCK_M×BLOCK_N×BLOCK_K)
and 32‑bit accumulation for numerical stability.  All other operations
(LayerNorm, Linear projections, gating, final Linear) are performed with
standard PyTorch kernels.

The result matches the reference PyTorch implementation while providing a
substantial speed‑up on modern GPUs (e.g. H100).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel that computes the batched GEMM:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------------------- #
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # -------------------------------------------------------------
    # program IDs (block indices)
    # -------------------------------------------------------------
    pid_i = tl.program_id(0)          # block in the i (row) dimension
    pid_j = tl.program_id(1)          # block in the j (col) dimension
    pid_z = tl.program_id(2)          # combined (batch, hidden) dimension

    # -------------------------------------------------------------
    # decode batch & hidden channel from pid_z
    # -------------------------------------------------------------
    b = pid_z // H                     # batch index
    h = pid_z - b * H                  # hidden‑channel index

    # -------------------------------------------------------------
    # tile offsets
    # -------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # loop over the reduction dimension k
    # -------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_block in range(0, num_k_blocks):
        cur_k = k_block * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left tile: shape (M, K)  -> indices (i, k)
        left_off = left_base + i[:, None] * stride_il + cur_k[None, :] * stride_kl
        left_tile = tl.load(left_off,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # right tile: we need (K, N) for a dot‑product,
        # so we load the original (j, k) layout as (k, j)
        right_off = right_base + cur_k[:, None] * stride_kr + j[None, :] * stride_jr
        right_tile = tl.load(right_off,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # accumulate product
        acc += tl.dot(left_tile, right_tile)   # (M,K) @ (K,N) -> (M,N)

    # -------------------------------------------------------------
    # write back the result
    # -------------------------------------------------------------
    out_off = out_base + i[:, None] * stride_io + j[None, :] * stride_jo
    tl.store(out_off, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Helper that launches the kernel for the entire tensor
# --------------------------------------------------------------------------- #
def _tri_mul_gemm(left: torch.Tensor, right: torch.Tensor,
                  BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    """
    left, right: [B, N, N, H]    (float32)
    returns:      out[B, N, N, H] (float32)
    """
    assert left.shape == right.shape, "left/right must share shape"
    B, N, _, H = left.shape

    # output buffer
    out = torch.empty_like(left)

    # grid dimensions
    grid = (triton.cdiv(N, BLOCK_M),    # i‑blocks
            triton.cdiv(N, BLOCK_N),    # j‑blocks
            B * H)                      # (batch, hidden) blocks

    _tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_bl=left.stride(0),
        stride_il=left.stride(1),
        stride_kl=left.stride(2),
        stride_hl=left.stride(3),
        stride_br=right.stride(0),
        stride_jr=right.stride(1),
        stride_kr=right.stride(2),
        stride_hr=right.stride(3),
        stride_bo=out.stride(0),
        stride_io=out.stride(1),
        stride_jo=out.stride(2),
        stride_ho=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough for fp32 work‑group
    )
    return out


# --------------------------------------------------------------------------- #
# Entry point used by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------- #
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask shape: [B, N, N]; ensure same dtype as left/right
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # (B, N, N, 1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid of linear transforms)
    # ------------------------------------------------------------------- #
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Core contraction using the Triton kernel
    # ------------------------------------------------------------------- #
    # Ensure tensors are contiguous for the kernel
    left  = left.contiguous()
    right = right.contiguous()
    out = _tri_mul_gemm(left, right)   # shape (B, N, N, hidden_dim)

    # ------------------------------------------------------------------- #
    # 6) Output LayerNorm
    # ------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ------------------------------------------------------------------- #
    # 7) Output gating
    # ------------------------------------------------------------------- #
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 8) Final linear projection back to `dim`
    # ------------------------------------------------------------------- #
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # (B, N, N, dim)

    return out
