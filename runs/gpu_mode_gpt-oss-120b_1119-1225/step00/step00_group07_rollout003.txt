    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    right_gate_proj = torch.nn.functional.linear(x_norm, weights["right_gate.weight"])

    mask_present = not config.get("nomask", False)

    right = apply_gate_and_mask(right_proj, right_gate_proj, mask_tensor, hidden_dim, mask_present)

    out = torch.einsum('bikd,bjkd->bijd', left, right)

    out_gate = torch.sigmoid(out_gate_proj)

    out = out_norm * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Linear projections:
   - left_proj   = x·W_left          (→ hidden_dim)
   - right_proj  = x·W_right
   - left_gate   = sigmoid(x·W_left_gate)
   - right_gate  = sigmoid(x·W_right_gate)
   - out_gate    = sigmoid(x·W_out_gate)
3. Fuse mask multiplication and sigmoid gating for *left* and *right*
   with a small Triton kernel (`apply_gate_mask_kernel`).  
   Output tensors `left` and `right` have shape [B,N,N,H].
4. Compute the TriMul contraction
      out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   using a single `torch.einsum` (equivalent to batched matmul per
   hidden channel).
5. Layer‑norm the result over the hidden dimension (`to_out_norm`).
6. Multiply by `out_gate` and linearly project back to the original
   dimension (`to_out`).

Only the mask + sigmoid gating step is performed in Triton; the rest
leverages highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,               # *float32   left/right projection
    gate_ptr,               # *float32   corresponding gate pre‑sigmoid
    mask_ptr,               # *float32   mask (optional, shape B,N,N)
    out_ptr,                # *float32   result
    total_elements: tl.int64,
    hidden_dim: tl.constexpr,
    mask_present: tl.constexpr,
    BLOCK_SIZE: tl.constexpr = 65536,
):
    """Fuse `proj * sigmoid(gate) * mask` (mask broadcast over hidden_dim)."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # load projection and gate
    proj = tl.load(proj_ptr + offs, mask=active).to(tl.float32)
    gate = tl.load(gate_ptr + offs, mask=active).to(tl.float32)

    # sigmoid(gate)
    sigmoid_gate = 1.0 / (1.0 + tl.exp(-gate))

    if mask_present:
        # each hidden entry shares the same mask element
        mask_idx = offs // hidden_dim          # integer division
        mask_val = tl.load(mask_ptr + mask_idx, mask=active).to(tl.float32)
        out = proj * sigmoid_gate * mask_val
    else:
        out = proj * sigmoid_gate

    tl.store(out_ptr + offs, out, mask=active)


def _apply_gate_and_mask(proj, gate, mask, hidden_dim, mask_present):
    """
    Helper that launches the Triton kernel to compute
        proj * sigmoid(gate) * mask (if present).
    """
    total = proj.numel()
    out = torch.empty_like(proj)

    # ensure contiguous tensors for raw pointer access
    proj_c = proj.contiguous()
    gate_c = gate.contiguous()
    if mask_present:
        mask_c = mask.contiguous()
    else:
        # dummy tensor – will never be accessed because mask_present=0
        mask_c = torch.empty(0, device=proj.device, dtype=proj.dtype)

    # launch config
    BLOCK = 65536
    grid = (triton.cdiv(total, BLOCK),)

    apply_gate_mask_kernel[grid](
        proj_c,
        gate_c,
        mask_c,
        out,
        total,
        hidden_dim,
        1 if mask_present else 0,
        BLOCK_SIZE=BLOCK,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                      # input channel dimension (D)
    hidden_dim = config["hidden_dim"]        # projection dimension (H)
    eps = 1e-5                               # LayerNorm epsilon

    # ------------------------------
    # 1. Input LayerNorm
    # ------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, D]

    # ------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate_pre = torch.nn.functional.linear(x, weights["left_gate.weight"])
    right_gate_pre = torch.nn.functional.linear(x, weights["right_gate.weight"])
    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"])

    # ------------------------------
    # 3. Apply mask + sigmoid gating (Triton)
    # ------------------------------
    nomask = config.get("nomask", False)
    mask_present = not nomask
    if mask_present:
        # ensure mask is float (0/1) and same dtype as the projections
        mask_f = mask.to(dtype=dtype)
    else:
        mask_f = None

    left = _apply_gate_and_mask(left_proj, left_gate_pre, mask_f, hidden_dim, mask_present)
    right = _apply_gate_and_mask(right_proj, right_gate_pre, mask_f, hidden_dim, mask_present)

    # ------------------------------
    # 4. Tri‑multiplicative contraction (einsum)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------
    # Using einsum which internally dispatches to a highly optimized batched GEMM.
    out = torch.einsum("bikd,bjkd->bijd", left, right)  # [B, N, N, H]

    # ------------------------------
    # 5. Output LayerNorm over hidden dimension
    # ------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    # ------------------------------
    # 6. Out‑gate (sigmoid) and final linear projection
    # ------------------------------
    out_gate = torch.sigmoid(out_gate_pre)          # [B, N, N, H]
    out_gated = out_norm * out_gate                 # element‑wise
    out_final = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return out_final
