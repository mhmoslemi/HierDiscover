import torch
import torch.nn.functional as F
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5 # typical LayerNorm eps

    # LayerNorm on input
    # mean and var across last dim
    # x normalized: (x - mean) / sqrt(var+eps)
    # then weight and bias
    # Compute in fp32 for stability
    x = input_tensor
    # compute mean, var
    mean = x.mean(-1, keepdim=True)
    var = ((x - mean) ** 2).mean(-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]
    # Linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # shape [bs, N, N, 1] broadcast
        left = left * mask_
        right = right * mask_
    # Gating
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for gemm
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # Permute for gemm
    # left: (bs, N, N, hidden) -> (bs, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right: (bs, N, N, hidden) -> we need (bs, hidden, N, N) where second N is j, but we need transposed along last two dims
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # Flatten batch*hidden
    batch_hidden = bs * hidden_dim
    M = N = seq_len
    K = seq_len
    a = left_perm.view(batch_hidden, M, K)
    b = right_perm.view(batch_hidden, K, N)
    # Triton gemm
    c = batched_gemm(a, b, block_m=64, block_n=64, block_k=32)
    # Reshape back to (bs, hidden, N, N) -> (bs, N, N, hidden)
    out = c.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1)
    # to_out_norm: LayerNorm over hidden_dim
    # Compute mean and var across hidden_dim for each (b,i,j)
    # shape [bs, seq_len, seq_len, hidden_dim]
    out_mean = out.mean(-1, keepdim=True)
    out_var = ((out - out_mean) ** 2).mean(-1, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    # out_gate gating (shape [bs, seq_len, seq_len, hidden_dim] from earlier)
    out_norm = out_norm * out_gate
    # final linear
    out_final = F.linear(out_norm, weights["to_out.weight"])
    return out_final
```

"""
TriMul (outgoing) forward pass with a custom Triton GEMM kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]).
2. Linear projections `left` and `right` (C → H) and optional mask.
3. Gating: sigmoid(linear(x)) → left_gate, right_gate, out_gate.
   left = left * left_gate ; right = right * right_gate.
4. Compute the contracted product
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   for every batch `b` and hidden channel `h`.  
   This is a batched matrix multiplication:
        A = left   reshaped to (B*H, N, K=N)
        B = right  reshaped to (B*H, K=N, N)  (k‑j transposed)
        C = A @ B   → shape (B*H, N, N)
   The heavy O(N³) work is performed by a Triton GEMM kernel that
   works on FP16/BF16 inputs, accumulates in FP32 and writes BF16.
5. Reshape the result back to [B,N,N,H] and apply a second
   Layer‑norm over the hidden dimension.
6. Multiply by `out_gate` and a final linear projection (H → C).

Only the matrix‑multiplication step is off‑loaded to Triton; all
other ops use native PyTorch (they are cheap compared to the N³‑work).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ B   (A: (batch,M,K), B: (batch,K,N))
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ----------- program id → batch, tile i, tile j -------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    bid = pid // (num_m * num_n)                 # which batch we are processing
    pid_rem = pid % (num_m * num_n)
    tile_m = pid_rem // num_n                    # tile row
    tile_n = pid_rem % num_n                     # tile col

    # ----------- offsets for this tile --------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Create a pointer to the beginning of the tile (without the K offset)
    a_base = a_ptr + bid * stride_a_batch + offs_m[:, None] * stride_a_m
    b_base = b_ptr + bid * stride_b_batch + offs_n[None, :] * stride_b_n

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load tile of A (M x K) and B (K x N)
        a = tl.load(a_base + offs_k[None, :] * stride_a_k,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_base + offs_k[:, None] * stride_b_k,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Convert to FP32 before dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Rank‑K update
        acc += tl.dot(a, b)

    # Write result back (store as BF16 to save bandwidth)
    c_ptrs = c_ptr + bid * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.bfloat16), mask=mask_m[:, None] & mask_n[None, :])


def _batched_gemm(a: torch.Tensor, b: torch.Tensor,
                  block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Batched matrix multiplication A @ B using the Triton kernel above.
    A: (BATCH, M, K)   – contiguous, dtype = bf16
    B: (BATCH, K, N)   – contiguous, dtype = bf16
    Returns C: (BATCH, M, N) dtype = bf16
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, Kb, N = b.shape
    assert K == Kb

    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # strides in element (not byte) units
    stride_a_batch, stride_a_m, stride_a_k = a.stride()
    stride_b_batch, stride_b_k, stride_b_n = b.stride()
    stride_c_batch, stride_c_m, stride_c_n = c.stride()

    # launch grid: one program per (batch, tile_i, tile_j)
    grid = (batch * ((M + block_m - 1) // block_m) *
            ((N + block_n - 1) // block_n),)

    _batched_gemm_kernel[grid](
        a, b, c,
        batch, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=block_m, BLOCK_N=block_n, BLOCK_K=block_k,
        num_warps=4,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, C] (dtype = torch.float32, on GPU)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (often 128)

    # ------------------------------------------------------------------
    # LayerNorm on input (over the last dimension C)
    # ------------------------------------------------------------------
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # Linear projections (C → H) – bias‑free
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask shape = [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).type_as(left)   # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gated projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare operands for the Triton GEMM
    # ------------------------------------------------------------------
    # use BF16 for the heavy multiplication (better dynamic range on H100)
    left = left.to(torch.bfloat16)
    right = right.to(torch.bfloat16)

    # left   : (B, N, N, H) → (B, H, N, N)  (i‑k layout)
    left_t = left.permute(0, 3, 1, 2).contiguous()
    # right  : (B, N, N, H) → (B, H, N, N) with (j‑k) → (k‑j) for GEMM
    right_t = right.permute(0, 3, 2, 1).contiguous()

    # collapse batch & hidden into a single GEMM batch dimension
    batch_h = B * H
    A = left_t.view(batch_h, N, N)          # (batch_h, M=N, K=N)
    B_ = right_t.view(batch_h, N, N)        # (batch_h, K=N, N)

    # ------------------------------------------------------------------
    # Batched matrix multiplication:  out = Σ_k left[:,i,k,:] * right[:,j,k,:]
    # ------------------------------------------------------------------
    C = _batched_gemm(A, B_, block_m=64, block_n=64, block_k=32)  # BF16
    # reshape back to (B, H, N, N) then to (B, N, N, H)
    out = C.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 2nd LayerNorm on hidden channel
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    # Gate & final linear projection (H → C)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
