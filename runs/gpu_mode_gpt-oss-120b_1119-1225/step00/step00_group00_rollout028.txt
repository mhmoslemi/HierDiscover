import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    # get program IDs
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # compute offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # integrate batch offset
    left_batch_offset = pid_b * N * N * D
    right_batch_offset = pid_b * N * N * D
    out_batch_offset = pid_b * N * N * D

    # accumulate over hidden dim blocks
    for d_start in range(0, D, BLOCK_D):
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # inner loop over K
        for k_start in range(0, N, BLOCK_K):
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left block [I, K, D]
            left_ptrs = left_ptr + left_batch_offset \
                + offs_i[:, None, None] * stride_left_i \
                + offs_k[None, :, None] * stride_left_k \
                + offs_d[None, None, :] * stride_left_d
            left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

            # Load right block [J, K, D]
            right_ptrs = right_ptr + right_batch_offset \
                + offs_j[:, None, None] * stride_right_i \
                + offs_k[None, :, None] * stride_right_k \
                + offs_d[None, None, :] * stride_right_d
            right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

            # Loop over k within block
            for kk in range(0, BLOCK_K):
                # scalar mask for kk within K block:
                k_valid = mask_k[kk]
                # If out of bounds, left/right values are zero due to mask.
                l_vec = left[:, kk, :]   # (I, D)
                r_vec = right[:, kk, :]  # (J, D)
                # Broadcast to (I,J,D)
                l_broadcast = tl.broadcast_to(l_vec, (BLOCK_I, BLOCK_J, BLOCK_D))
                r_broadcast = tl.broadcast_to(r_vec, (BLOCK_I, BLOCK_J, BLOCK_D))
                acc += l_broadcast * r_broadcast

        # store results
        out_ptrs = out_ptr + out_batch_offset \
            + offs_i[:, None, None] * stride_out_i \
            + offs_j[None, :, None] * stride_out_j \
            + offs_d[None, None, :] * stride_out_d
        tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])
```

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor.
2. Linear projections for the “left”, “right” and three gating tensors.
3. Apply the optional pairwise mask and the element‑wise sigmoid gates.
4. Compute the pairwise multiplicative update
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a Triton kernel that tiles the (i,j) plane and loops over the
   sequence dimension k.  The hidden dimension is processed in small
   blocks (default 32) to keep register pressure low.
5. Layer‑normalize the intermediate result, apply the output gate and a
   final linear projection back to the original channel dimension.

The kernel fuses the heavy O(N³·C) part, while the surrounding
layer‑norms and linear layers are performed with PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # const float*  left  [B, N, N, C]
    right_ptr,           # const float*  right [B, N, N, C]
    out_ptr,             #       float*  out   [B, N, N, C]
    N,                   # int   sequence length
    hidden_dim,          # int   total hidden dimension (runtime)
    d_offset,            # int   start offset of the hidden block
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d,
    stride_r_batch, stride_r_i, stride_r_k, stride_r_d,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Triton kernel for the core 4‑D einsum:
        out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

    The hidden dimension is processed in blocks of size BLOCK_D.
    """

    # ------------------------------------------------------------------
    # Program IDs (grid is (num_i_blocks, num_j_blocks, batch))
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)          # block index over i
    pid_j = tl.program_id(1)          # block index over j
    pid_b = tl.program_id(2)          # batch index

    # ------------------------------------------------------------------
    # Offsets for the i / j tiles
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Hidden‑dimension offsets for the current block
    # ------------------------------------------------------------------
    offs_d = d_offset + tl.arange(0, BLOCK_D)
    mask_d = offs_d < hidden_dim

    # ------------------------------------------------------------------
    # Batch base pointers
    # ------------------------------------------------------------------
    base_l = left_ptr  + pid_b * stride_l_batch
    base_r = right_ptr + pid_b * stride_r_batch
    base_o = out_ptr   + pid_b * stride_o_batch

    # accumulator for this tile (float32 for precision)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the sequence dimension k in blocks
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------- load left block [I, K, D] -----------------
        left_ptrs = base_l \
            + offs_i[:, None, None] * stride_l_i \
            + offs_k[None, :, None] * stride_l_k \
            + offs_d[None, None, :] * stride_l_d
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # -------------------- load right block [J, K, D] ----------------
        right_ptrs = base_r \
            + offs_j[:, None, None] * stride_r_i \
            + offs_k[None, :, None] * stride_r_k \
            + offs_d[None, None, :] * stride_r_d
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # -------------------- inner reduction over k --------------------
        # left : (I, K, D), right : (J, K, D)
        # for each k we compute outer product over (i, j) and accumulate.
        for kk in range(0, BLOCK_K):
            l_vec = left[:, kk, :]        # (I, D)
            r_vec = right[:, kk, :]       # (J, D)

            # Broadcast to (I,J,D) and accumulate
            l_b = tl.broadcast_to(l_vec, (BLOCK_I, BLOCK_J, BLOCK_D))
            r_b = tl.broadcast_to(r_vec, (BLOCK_I, BLOCK_J, BLOCK_D))
            acc += l_b * r_b

    # ------------------------------------------------------------------
    # Store the result for this hidden block
    # ------------------------------------------------------------------
    out_ptrs = base_o \
        + offs_i[:, None, None] * stride_o_i \
        + offs_j[None, :, None] * stride_o_j \
        + offs_d[None, None, :] * stride_o_d
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]
            - mask  : torch.Tensor [B, N, N] (may be ignored if config["nomask"])
            - weights: dict of model weight tensors
            - config : dict with keys "dim", "hidden_dim", "nomask", ...

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask is assumed to contain 0/1 values; convert to float and expand
        mask_exp = mask_tensor.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating tensors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare output buffer for the heavy O(N³·C) part
    # ------------------------------------------------------------------
    out = torch.empty_like(left)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # Triton kernel launch parameters (tunable)
    # ------------------------------------------------------------------
    BLOCK_I = 8   # tiles for i dimension
    BLOCK_J = 8   # tiles for j dimension
    BLOCK_K = 16  # tiles for reduction over k
    BLOCK_D = 32  # hidden‑dim block size (must divide register usage)

    N = input_tensor.shape[1]  # sequence length

    # strides for pointer arithmetic (in elements, not bytes)
    stride_l_batch, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_batch, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # grid: (i_blocks, j_blocks, batch)
    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        left.shape[0],  # batch size
    )

    # Loop over the hidden dimension in blocks of BLOCK_D
    for d_start in range(0, hidden_dim, BLOCK_D):
        trimul_outgoing_kernel[
            grid
        ](
            left,
            right,
            out,
            N,
            hidden_dim,
            d_start,                      # d_offset
            stride_l_batch,
            stride_l_i,
            stride_l_k,
            stride_l_d,
            stride_r_batch,
            stride_r_i,
            stride_r_k,
            stride_r_d,
            stride_o_batch,
            stride_o_i,
            stride_o_j,
            stride_o_d,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # ------------------------------------------------------------------
    # Post‑processing: to_out_norm, gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # -> [B, N, N, dim]

    return out
