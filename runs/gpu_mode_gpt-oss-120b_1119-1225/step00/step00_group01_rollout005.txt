@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute pointers
        a_ptrs = a_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # load
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0).to(tl.float32)

        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)

    # store
    c_ptrs = c_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16) if a_ptr.dtype == tl.float16 else acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # B,N,N,D
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # pytorch layernorm manually
    # shape: [B,N,N,D]
    # Compute mean and var across last dim.
    x = input_tensor
    # flatten to compute per element: mean = x.mean(-1, keepdim=True)
    # but Stage: using built-in torch.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps
    )
    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if needed
    if not nomask:
        # mask shape (B,N,N)
        mask_ = mask.unsqueeze(-1).to(x_norm.dtype)  # broadcast
        left = left * mask_
        right = right * mask_
    else:
        # might still need to ensure mask dtype replaced no ops
        pass

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # perform batched matmul using Triton
    B,N,N_dim = left.shape[0], left.shape[1], left.shape[2]
    H = hidden_dim

    # reshape for batched multiplies
    # we need left and right to be shape (B*H,N,N)
    left_mat = left.permute(0,3,1,2).contiguous().reshape(B*H, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().reshape(B*H, N, N)

    # call Triton kernel
    out_mat = triton_batched_matmul(left_mat, right_mat)  # shape (B*H,N,N)

    # reshape back
    out = out_mat.reshape(B, H, N, N).permute(0,2,3,1)  # shape (B,N,N,H)

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps
    )
    # apply out_gate
    out = out * out_gate

    # final linear projection back to dim
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
```

"""
TriMul (outgoing) forward kernel.

The heavy part of TriMul is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication
    out_d = left_d @ right_d.T   for each hidden channel d.

We implement:
  * LayerNorms, Linear projections and sigmoid gates in PyTorch (FP32).
  * The batched matmul for the N×N products in a custom Triton kernel.
    The kernel works on FP16 inputs (to halve memory traffic) and
    accumulates in FP32, producing a FP32 output.
  * The remaining LayerNorm, gating and final Linear projection are
    performed again in PyTorch.

The Triton kernel processes a 3‑D batch (batch = B * hidden_dim) and
computes C[batch, i, j] = Σ_k A[batch, i, k] * B[batch, k, j].
Block sizes are chosen to give good occupancy on an H100.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------------- #
# Triton batched matmul kernel (A @ B^T) with FP16 inputs and FP32 accumulation
# ----------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, M, N, K,                    # logical dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes C[batch, i, j] = Σ_k A[batch,i,k] * B[batch,k,j]
    where A, B, C are 3‑D tensors of shape (batch, M, K), (batch, K, N),
    (batch, M, N) respectively.
    A and B are expected to be FP16; accumulation is in FP32.
    """
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # -------------------- block offsets --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds accesses
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- main K‑loop --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointer arithmetic for A (M x K) and B (K x N)
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # load with masking, cast to FP32 for the dot product
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate dot product
        acc += tl.dot(a, b)

    # -------------------- store result --------------------
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul using the Triton kernel above.
    A: (batch, M, K)   – FP16
    B: (batch, K, N)   – FP16 (already transposed version of the original right matrix)
    Returns C: (batch, M, N) – FP32
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match"

    # Allocate output (FP32)
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Block sizes (tuned for H100, FP16 inputs)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        BATCH=batch,
        M=M,
        N=N,
        K=K,
        stride_a_batch=A.stride(0),
        stride_a_m=A.stride(1),
        stride_a_k=A.stride(2),
        stride_b_batch=B.stride(0),
        stride_b_k=B.stride(1),
        stride_b_n=B.stride(2),
        stride_c_batch=C.stride(0),
        stride_c_m=C.stride(1),
        stride_c_n=C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return C


# ----------------------------------------------------------------------------- #
# Main entry point: custom Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Compute the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]
        - mask         : torch.Tensor   [B, N, N]  (may be None)
        - weights      : dict of torch.Tensors containing all linear &
                         layer‑norm parameters
        - config       : dict with keys
            * "dim"         – input feature dimension
            * "hidden_dim"  – hidden dimension used in the projections
            * "nomask"      – optional bool, True if no mask should be applied

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (FP32)
    """
    # ------------------------------------------------------------------ #
    # Unpack inputs
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5  # LayerNorm epsilon

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (over channel dim)
    # ------------------------------------------------------------------ #
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------ #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------ #
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------ #
    # 3. Optional masking (applied before gating)
    # ------------------------------------------------------------------ #
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4. Gating sigmoid (per hidden channel)
    # ------------------------------------------------------------------ #
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 5. Batched matrix multiplication using Triton
    #    Compute out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
    #    = left_d @ right_d.T  for each hidden channel d.
    # ------------------------------------------------------------------ #
    B, N, _, H = left.shape  # H == hidden_dim

    # Convert to shape (B*H, N, N) and cast to FP16 for bandwidth savings
    left_mat = left.permute(0, 3, 1, 2).contiguous().reshape(B * H, N, N).to(torch.float16)
    right_mat = right.permute(0, 3, 1, 2).contiguous().reshape(B * H, N, N).to(torch.float16)

    # Transpose the second operand (right) to get shape (B*H, K, N) where K==N
    right_mat_t = right_mat.permute(0, 2, 1).contiguous()

    # Triton batched matmul: result is FP32
    out_mat = batched_matmul_triton(left_mat, right_mat_t)  # [B*H, N, N] FP32

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm (over hidden_dim) and final gating
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # apply output gate

    # ------------------------------------------------------------------ #
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------ #
    output = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return output
```
