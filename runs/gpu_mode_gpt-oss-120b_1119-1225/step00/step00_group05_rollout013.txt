def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape

    # LayerNorm on dim
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )
    # Projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1)  # shape [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Allocate out
    out = torch.empty((bs, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=device)

    # Triton kernel call
    # Compute strides = left.stride()
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()
    # Compute block sizes
    BLOCK_K = 64  # maybe 64
    num_k_blocks = (seq_len + BLOCK_K - 1) // BLOCK_K

    # Launch
    grid = (bs, seq_len, seq_len)
    trimul_outgoing_kernel[grid](
        left, right, out,
        bs, seq_len,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        C=hidden_dim,
        BLOCK_K=BLOCK_K,
        NUM_K_BLOCKS=num_k_blocks,
    )

    # After kernel: out shape [B,N,N,hidden_dim]
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out_norm * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])
    return output

"""
TriMul “outgoing” forward implementation.

The tensor x ∈ ℝ⁽ᴮ×ᴺ×ᴺ×C⁾ is first layer‑normalized and linearly projected to a hidden
dimension H (named `hidden_dim`).  Gating tensors are obtained with sigmoid‑gated
linear layers and the projection results are masked (if a mask is provided).

The expensive part is the “triangular multiplicative” update:

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

for all batch b, sequence positions i,j and hidden channels d.
This O(N³·H) reduction is implemented in a custom Triton kernel that
processes each (b,i,j) triple with a vector of length H in registers and
accumulates over k in blocks of size BLOCK_K.

After the reduction we apply a hidden‑dimensional LayerNorm, an output gate,
and a final linear projection back to the original feature dimension C.

Only the forward pass is required – no gradients are computed.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# Triton kernel: computes the N³ reduction for a fixed (batch,i,j) triple.
# ----------------------------------------------------------------------
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr,               # const float*   left  [B,N,N,H]
    right_ptr,              # const float*   right [B,N,N,H]
    out_ptr,                #       float*   out   [B,N,N,H]
    B, N,                  # int32 scalars
    stride_lb, stride_li, stride_lk, stride_lc,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rc,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oc,   # out strides
    C: tl.constexpr,       # hidden dimension (compile‑time)
    BLOCK_K: tl.constexpr, # tile size along the reduction axis
    NUM_K_BLOCKS: tl.constexpr, # number of K‑tiles (compile‑time)
):
    b = tl.program_id(0)     # batch index
    i = tl.program_id(1)     # first sequence index
    j = tl.program_id(2)     # second sequence index

    # ------------------------------------------------------------------
    # Register accumulator for the hidden dimension.
    # ------------------------------------------------------------------
    acc = tl.zeros([C], dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K in blocks.  The loop is fully unrolled because NUM_K_BLOCKS
    # is a compile‑time constant.
    # ------------------------------------------------------------------
    for block_k in range(NUM_K_BLOCKS):
        # Global k indices for this tile
        k = block_k * BLOCK_K + tl.arange(0, BLOCK_K)   # shape (BLOCK_K,)
        k_valid = k < N                                  # mask for tail

        # ------------------------------------------------------------------
        # Load a (BLOCK_K × C) tile from left  : left[b,i,k,:]
        # ------------------------------------------------------------------
        d = tl.arange(0, C)   # hidden dimension indices
        left_offset = (
            b * stride_lb
            + i * stride_li
            + k[:, None] * stride_lk
            + d[None, :] * stride_lc
        )
        left_tile = tl.load(left_ptr + left_offset,
                            mask=k_valid[:, None],
                            other=0.0)                     # shape (BLOCK_K, C)

        # ------------------------------------------------------------------
        # Load the corresponding tile from right : right[b,j,k,:]
        # ------------------------------------------------------------------
        right_offset = (
            b * stride_rb
            + j * stride_ri
            + k[:, None] * stride_rk
            + d[None, :] * stride_rc
        )
        right_tile = tl.load(right_ptr + right_offset,
                             mask=k_valid[:, None],
                             other=0.0)                     # shape (BLOCK_K, C)

        # ------------------------------------------------------------------
        # Element‑wise product and reduction over the K‑axis.
        # ------------------------------------------------------------------
        prod = left_tile * right_tile            # (BLOCK_K, C)
        acc += tl.sum(prod, axis=0)              # (C,)

    # ------------------------------------------------------------------
    # Write the result vector to the output tensor.
    # ------------------------------------------------------------------
    out_offset = (
        b * stride_ob
        + i * stride_oi
        + j * stride_oj
        + d * stride_oc
    )
    tl.store(out_ptr + out_offset, acc)


# ----------------------------------------------------------------------
# Public entry point expected by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, C] float32
        - mask_tensor  : [B, N, N]   (optional, may be all‑ones)
        - weights_dict : mapping of parameter names to tensors
        - config_dict  : contains 'dim' and 'hidden_dim' (H)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape          # C == config["dim"]
    hidden_dim = config["hidden_dim"]         # H (always 128 in the tests)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the input feature dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # shape (B,N,N,C)

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask is broadcast on the hidden axis)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating tensors (sigmoid‑gated linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # Apply gates to the projected tensors
    left = left * left_gate          # (B,N,N,H)
    right = right * right_gate       # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Triton kernel: compute the N³ reduction over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides in elements (not bytes) – required by the kernel.
    stride_left = left.stride()      # (B_stride, i_stride, k_stride, d_stride)
    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_K = 64                     # tile size for the reduction axis
    num_k_blocks = (N + BLOCK_K - 1) // BLOCK_K

    grid = (B, N, N)                 # one program per (b,i,j) triple
    _trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        C=hidden_dim,
        BLOCK_K=BLOCK_K,
        NUM_K_BLOCKS=num_k_blocks,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: hidden‑dim LayerNorm, output gating and final linear
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out_norm * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])
    return output
