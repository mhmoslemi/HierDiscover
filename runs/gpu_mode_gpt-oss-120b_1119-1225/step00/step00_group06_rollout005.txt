    eps = 1e-5

    # Get weight tensors as FP16

    norm_bias = weights["norm.bias"].to(torch.float16)
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # LayerNorm on input (dim)
    x_norm = triton_layer_norm(x, norm_weight, norm_bias, epsilon=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if not nomask:

        mask_f16 = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f16
        right = right * mask_f16

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # left: [B,N,N,H]; right: [B,N,N,H]
    # permute to [B,H,N,N] and transpose right
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2)   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1) # [B, H, N, N] where second dim is k, third is j

def layer_norm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    epsilon,
    N_ROWS,
    C,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    if pid >= N_ROWS:
        return

    row_start = pid * C
    # Compute mean and variance
    sum_val = tl.zeros([], tl.float32)
    sumsq_val = tl.zeros([], tl.float32)
    c_blocks = tl.cdiv(C, BLOCK_C)
    for i in range(c_blocks):
        offset = i * BLOCK_C
        offs = offset + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(inp_ptr + row_start + offs, mask=mask, other=0.0)
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f)
        sumsq_val += tl.sum(x_f * x_f)

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # Second pass: store normalized
    for i in range(c_blocks):
        offset = i * BLOCK_C
        offs = offset + tl.arange(0, BLOCK_C)
        mask = offs < C
        x = tl.load(inp_ptr + row_start + offs, mask=mask, other=0.0)
        x_f = x.to(tl.float32)
        norm = (x_f - mean) * inv_std

        w = tl.load(weight_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs, mask=mask, other=0.0).to(tl.float32)
        y = norm * w + b
        y_f16 = y.to(tl.float16)
        tl.store(out_ptr + row_start + offs, y_f16, mask=mask)

TriMul (outgoing) forward implementation with a Triton fused LayerNorm.

Algorithm
---------
1. Cast the input pair tensor `[B, N, N, C]` to FP16.
2. Apply a LayerNorm over the last dimension (`C`) using a custom

4. If a mask is present (``nomask=False``) multiply `left` and `right`
   by the broadcast mask.
5. Apply sigmoid gates to `left` and `right`, and multiply them element‑wise.
6. Compute the cubic reduction  
      out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]  
   which is equivalent to a batched matrix multiplication:
      left  → (B, H, N, N)   with axes (i, k)  
      right → (B, H, N, N)   with axes (k, j) after a transpose.  
   The multiplication is performed with `torch.bmm`, which is highly
   optimised on H100.
7. A second LayerNorm over the hidden dimension (`H`) is applied using the
   same Triton kernel.
8. Multiply the normalised tensor by the output gate, then project back
   to the original channel dimension with a final linear layer.
9. Cast the result back to FP32 and return.

Only the LayerNorms are implemented in Triton; the rest of the
computation leverages fast fused PyTorch kernels (cuBLAS), satisfying the
requirement that “at least part of the operations” be written as a
Triton kernel while keeping the overall runtime competitive.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel (computes mean/var in FP32)
# ----------------------------------------------------------------------
@triton.jit
def layer_norm_kernel(
    inp_ptr,                # *const T*   input  [rows, C]
    out_ptr,                # *mut   T*   output [rows, C]
    weight_ptr,             # *const T*   gamma  [C]
    bias_ptr,               # *const T*   beta   [C]
    epsilon,                # float32     epsilon
    N_ROWS,                 # i32         number of rows (B*N*N)
    C,                      # i32         channel dimension (constexpr)
    BLOCK_C: tl.constexpr, # i32         block size for the channel dim
):
    pid = tl.program_id(0)
    if pid >= N_ROWS:
        return

    # offset for the beginning of the current row
    row_offset = pid * C

    # ------------------------------------------------------------------
    # First pass: compute mean and variance (FP32 accumulation)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([], tl.float32)
    sumsq_val = tl.zeros([], tl.float32)
    num_c_blocks = tl.cdiv(C, BLOCK_C)

    for block in range(num_c_blocks):
        col = block * BLOCK_C + tl.arange(0, BLOCK_C)
        mask = col < C

        x = tl.load(inp_ptr + row_offset + col, mask=mask, other=0.0)  # fp16
        x_f = x.to(tl.float32)

        sum_val += tl.sum(x_f)
        sumsq_val += tl.sum(x_f * x_f)

    mean = sum_val / C
    var = sumsq_val / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # ------------------------------------------------------------------
    # Second pass: normalise, apply weight & bias and store
    # ------------------------------------------------------------------
    for block in range(num_c_blocks):
        col = block * BLOCK_C + tl.arange(0, BLOCK_C)
        mask = col < C

        x = tl.load(inp_ptr + row_offset + col, mask=mask, other=0.0)
        x_f = x.to(tl.float32)

        # normalisation
        norm = (x_f - mean) * inv_std

        # weight & bias (stored as fp16, promote to fp32 for the math)
        w = tl.load(weight_ptr + col, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + col, mask=mask, other=0.0).to(tl.float32)

        y = norm * w + b               # still fp32
        y_f16 = y.to(tl.float16)       # store as fp16 to match the rest of the pipeline

        tl.store(out_ptr + row_offset + col, y_f16, mask=mask)


def triton_layer_norm(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      epsilon: float = 1e-5) -> torch.Tensor:
    """
    Triton‑based LayerNorm for tensors of shape [B, N, N, C].

    Parameters
    ----------
    x      : torch.Tensor (float16) – input tensor.
    weight : torch.Tensor (float16) – gamma, shape [C].
    bias   : torch.Tensor (float16) – beta,  shape [C].
    epsilon: float – numerical stability constant.

    Returns
    -------
    torch.Tensor (float16) – normalised tensor with the same shape as ``x``.
    """
    assert x.is_contiguous()
    B, N, _, C = x.shape
    rows = B * N * N

    # Flatten the leading three dimensions to a 2‑D tensor [rows, C]
    x_flat = x.view(rows, C)
    out_flat = torch.empty_like(x_flat)

    # Ensure weight/bias are on the same device/dtype as x
    weight = weight.to(x.dtype)
    bias = bias.to(x.dtype)

    # Launch one program per row
    grid = (rows,)
    layer_norm_kernel[grid](
        x_flat,
        out_flat,
        weight,
        bias,
        epsilon,
        rows,
        C,
        BLOCK_C=64,          # 64 works for C up to 384
        num_warps=4,
    )
    # Reshape back to the original 4‑D layout
    return out_flat.view(B, N, N, C)


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor of shape [B, N, N, dim] (float32)
        mask    – torch.Tensor of shape [B, N, N] (float32 or bool)
        weights – dict of model parameters (float32)
        config  – dict with keys:
                  "dim"        – input channel dimension
                  "hidden_dim" – hidden dimension of the multiplicative update
                  "nomask"     – True if no mask should be applied

    Returns
    -------
    torch.Tensor – output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # Cast everything to FP16 for the heavy computation path
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------
    # Load / cast model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)

    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)

    to_out_weight = weights["to_out.weight"].to(torch.float16)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = triton_layer_norm(x, norm_weight, norm_bias, epsilon=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x_norm, left_proj_weight)      # left projection
    right = F.linear(x_norm, right_proj_weight)    # right projection

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating as in the reference)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f16
        right = right * mask_f16

    # ------------------------------------------------------------------
    # 4) Gating pathways
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cubic reduction via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Rearrange to [B, H, N, N] for a batched matmul
    left_perm = left.permute(0, 3, 1, 2)                     # (i, k)
    right_perm = right.permute(0, 3, 2, 1)                   # (k, j) after transpose

    # Collapse batch and head dimensions for a single bmm call
    left_flat = left_perm.reshape(B * H, N, N)
    right_flat = right_perm.reshape(B * H, N, N)

    out_flat = torch.bmm(left_flat, right_flat)           # (B*H, N, N)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)   # back to [B, N, N, H]

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out_norm = triton_layer_norm(out, to_out_norm_weight, to_out_norm_bias, epsilon=eps)

    # ------------------------------------------------------------------
    # 7) Output gating and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_gated = out_norm * out_gate
    # Final projection to original channel dimension `dim`
    output_fp16 = F.linear(out_gated, to_out_weight)   # shape [B, N, N, dim]

    # Cast back to float32 for the public API
    return output_fp16.to(torch.float32)
```
