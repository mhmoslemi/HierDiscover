def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # parse config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # Normalize x
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape [B,N,N], broadcast
        m = mask.to(x.dtype).unsqueeze(-1)   # shape [B,N,N,1]
        left = left * m
        right = right * m

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Compute multiplicative update via Triton
    out = trimul_triton(left_h, right_h)  # shape [B,N,N,hidden_dim], dtype float32

    # Normalize output
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Apply out gate
    out_norm = out_norm * out_gate

    # Final linear projection to original dim
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile
    pid_f = tl.program_id(2)  # batch*hidden feature index

    # decode batch and hidden channel
    b = pid_f // hidden_dim
    d = pid_f % hidden_dim

    # compute absolute indices
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # range of indices
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    # masks for out-of-range rows/cols
    mask_i = i < N
    mask_j = j < N

    # base pointers for this (b,d) slice
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base = out_ptr + b * stride_out_b + d * stride_out_d

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left_tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # right_tile transposed: shape (BLOCK_K, BLOCK_N)
        # We need right[j,k] -> transposed as (k,j)
        right_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_i
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # dot product: (M,K) x (K,N) -> (M,N)
        acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)
    # store result
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul “outgoing” implementation for AlphaFold3 style pairwise updates.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the last dimension (dim).
2. Compute two linear projections `left` and `right` (dim → hidden_dim) and
   apply an optional binary mask.
3. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from
   the normalized input and fuse them with the projected tensors.
4. The core operation is a batched matrix‑multiplication:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   i.e. for each (b,d) pair we need `out = left @ rightᵀ`.  This is
   performed by a custom Triton kernel that tiles over the (i,j) plane
   and iterates over k in blocks, loading `left` and the transposed view of
   `right`.  The kernel works in FP16 for memory bandwidth and accumulates
   in FP32 for precision.
5. Layer‑normalize the result (`out_norm`) and multiply by `out_gate`.
6. Final linear projection maps back to `dim` channels.

The function `custom_kernel` follows the same signature as required by
the test harness and returns the final tensor in float32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const * fp16
    right_ptr,         # const * fp16
    out_ptr,           # * fp32
    B, N, hidden_dim,  # scalar meta
    # strides for left   (b, i, k, d)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right  (b, i, k, d) – we will read it as (k, j)
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    # strides for out    (b, i, j, d)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    The loop is tiled over i and j (M,N) with a k‑block loop.
    """
    pid_m = tl.program_id(0)   # i‑tile index
    pid_n = tl.program_id(1)   # j‑tile index
    pid_f = tl.program_id(2)   # flattened (b, d) index

    # decode batch and hidden channel from pid_f
    b = pid_f // hidden_dim
    d = pid_f % hidden_dim

    # absolute start indices for the current tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # range vectors for the tile
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    # masks for boundary handling
    mask_i = i < N
    mask_j = j < N

    # base pointers for the (b,d) slice
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension in blocks
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float16)

        # right tile transposed: we need right[j,k] → (k,j) layout
        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + k[:, None] * stride_right_k + j[None, :] * stride_right_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float16)

        # dot: (M,K) @ (K,N) → (M,N), accumulation in fp32
        acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)

    # write back the result with masking
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched pairwise multiplicative update using Triton.

    Args:
        left, right: tensors of shape [B, N, N, H] and dtype float16.
    Returns:
        out: tensor of shape [B, N, N, H] and dtype float32.
    """
    B, N, _, hidden_dim = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Tiling parameters – chosen to fit H100 SMs while keeping register pressure low.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * hidden_dim                         # (b,d) combos
    )

    # Strides for each tensor (assumed contiguous)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # 1) LayerNorm over the last dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # 2) Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3) Optional mask (broadcasted on the channel dimension)
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # 4) Gates (sigmoid after a linear projection)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # 5) Convert to FP16 for the heavy GEMM step
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # 6) Core multiplicative update (B * hidden_dim independent GEMMs)
    out = trimul_triton(left_h, right_h)                 # fp32 result, shape [B,N,N,hidden_dim]

    # 7) Output LayerNorm over hidden_dim
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 8) Multiply by the output gate
    out_norm = out_norm * out_gate

    # 9) Final linear projection to original dimension (bias‑free)
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
