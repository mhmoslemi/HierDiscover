def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on x
    # compute mean, var across dim axis (last)
    # use torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # linear projections (no bias)
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # gating linear + sigmoid
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # elementwise combine using Triton kernel
    left = apply_gate_mask_kernel(left_proj, left_gate, mask, out_shape = left_proj.shape)
    right = apply_gate_mask_kernel(right_proj, right_gate, mask, out_shape = right_proj.shape)

    # Convert to half for batched GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape for batched matmul
    B,N,_ , H = left.shape # left shape B,N,N,H
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)

    # bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()

    # layernorm over hidden dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # out_gate broadcast and multiply
    out_gate = out_gate  # shape B,N,N,hidden_dim already
    # maybe out_gate in same dtype as out (float32). Cast as needed.
    out = out * out_gate

    # final linear: hidden_dim -> dim
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

    # Projections

    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))

    right = right.to(torch.float16)

    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

"""
TriMul (outgoing) forward pass implemented with a small Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (last dimension = dim).
2. Compute four linear projections on `x` (no bias):
      left_proj, right_proj, left_gate, right_gate, out_gate.
3. Apply sigmoid to the three gate projections.
4. Fuse the three element‑wise multiplications (left_proj*left_gate,
   right_proj*right_gate) and the optional mask in a Triton kernel.
5. Convert the gated tensors to float16 and perform the expensive
   contraction
        out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
   as a batched matrix multiplication:
        (B, H, N, N) @ (B, H, N, N).transpose(-1,-2)
   where H = hidden_dim.
6. Layer‑norm the result over the hidden dimension, multiply by the
   sigmoid‑gated out‑gate, and project back to `dim` with a final linear
   layer.
All heavy arithmetic (step 4 and the N³ contraction) is performed on the
GPU; the Triton kernel guarantees that at least one operation is
executed outside pure PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: element‑wise (proj * gate * mask)
# ----------------------------------------------------------------------
@triton.jit
def _gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,          # pointers
    B, N, H,                                         # sizes
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_SIZE: tl.constexpr,
):
    """Multiply a projection, a gate and a broadcasted mask.
    The tensors are all 4‑D [B, N, N, H] (mask lacks the H dim)."""
    total_elems = B * N * N * H
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    m = offs < total_elems                      # mask for out‑of‑range threads

    # ----- Decode linear index into (b,i,j,h) -----------------------
    b = offs // (N * N * H)
    r = offs - b * (N * N * H)
    i = r // (N * H)
    r = r - i * (N * H)
    j = r // H
    h = r - j * H

    # ----- Load values ---------------------------------------------
    proj = tl.load(
        proj_ptr
        + b * stride_proj_b
        + i * stride_proj_i
        + j * stride_proj_j
        + h * stride_proj_h,
        mask=m,
        other=0.0,
    )
    gate = tl.load(
        gate_ptr
        + b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h,
        mask=m,
        other=0.0,
    )
    # mask has no H dimension – broadcast it
    mask_val = tl.load(
        mask_ptr
        + b * stride_mask_b
        + i * stride_mask_i
        + j * stride_mask_j,
        mask=m,
        other=1.0,
    )

    out = proj * gate * mask_val

    tl.store(
        out_ptr
        + b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h,
        out,
        mask=m,
    )


def _apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based element‑wise multiplication: out = proj * gate * mask.
    `proj` and `gate` are [B, N, N, H]; `mask` is [B, N, N] (float32).
    Returns a new tensor with the same shape as ``proj``.
    """
    assert proj.shape == gate.shape
    B, N, _, H = proj.shape
    device = proj.device

    # ensure contiguous tensors – Triton works on plain contiguous memory
    proj = proj.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()
    out = torch.empty_like(proj)

    # strides are expressed in element counts (not bytes)
    s_proj = proj.stride()
    s_gate = gate.stride()
    s_out = out.stride()
    s_mask = mask.stride()

    total_elements = proj.numel()
    BLOCK = 1024
    grid = (triton.cdiv(total_elements, BLOCK),)

    _gate_mask_kernel[grid](
        proj,
        gate,
        mask,
        out,
        B,
        N,
        H,
        s_proj[0],
        s_proj[1],
        s_proj[2],
        s_proj[3],
        s_gate[0],
        s_gate[1],
        s_gate[2],
        s_gate[3],
        s_mask[0],
        s_mask[1],
        s_mask[2],
        s_out[0],
        s_out[1],
        s_out[2],
        s_out[3],
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]                # input & output channel size
    hidden_dim = config["hidden_dim"]  # projection size
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (dim‑wise)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Gate projections + sigmoid
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4. Fuse (proj * gate * mask) with a Triton kernel
    # ------------------------------------------------------------------
    # Ensure mask is float32 and broadcastable; if `nomask` we just use 1.
    if mask is None or config.get("nomask", False):
        mask = torch.ones(x.shape[:3], dtype=torch.float32, device=x.device)
    else:
        mask = mask.to(dtype=torch.float32, device=x.device)

    left = _apply_gate_mask(left_proj, left_gate, mask)   # [B,N,N,hidden_dim]
    right = _apply_gate_mask(right_proj, right_gate, mask)

    # ------------------------------------------------------------------
    # 5. Contract over the third axis (the N³ operation)
    # ------------------------------------------------------------------
    # Convert to fp16 for the heavy matmul – H100 excels at fp16.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N) for batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_flat = left_mat @ right_mat^T   (sum over k)
    out_flat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # restore original 4‑D layout: [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to fp32 for the remaining ops

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, out‑gate, and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate                         # element‑wise gating
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)           # [B,N,N,dim]

    return out
