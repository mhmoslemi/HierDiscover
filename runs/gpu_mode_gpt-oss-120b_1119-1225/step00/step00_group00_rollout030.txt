Pseudo:

```
import triton
import triton.language as tl

@triton.jit
def tri_mul_fused_kernel(
    x_ptr, mask_ptr,
    norm_weight_ptr, norm_bias_ptr,
    left_proj_w_ptr, left_gate_w_ptr,
    right_proj_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    B, N, dim, hidden_dim,
    stride_xb, stride_xn, stride_xk, stride_xc,
    stride_mb, stride_mn, stride_mk,
    stride_lob, stride_lon, stride_lok, stride_lod,
    stride_rob, stride_ron, stride_rok, stride_rod,
    stride_ogb, stride_ogn, stride_ogk, stride_ogd,
    eps: tl.constexpr,
    BLOCK_X: tl.constexpr,   # number of elements processed per block (threads) i.e., block size.
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    # decode indices
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # load mask, treat if mask_ptr==0 then mask_val=1 else load
    mask_val = 1.0
    if mask_ptr != 0:
        mask_offset = b * stride_mb + i * stride_mn + j * stride_mk
        mask_val = tl.load(mask_ptr + mask_offset)

    # load x vector
    c = tl.arange(0, dim)
    x_ptrs = x_ptr + b * stride_xb + i * stride_xn + j * stride_xk + c * stride_xc
    x = tl.load(x_ptrs, mask=c < dim, other=0.0).to(tl.float32)

    # compute mean
    sum_x = tl.sum(x)
    mean = sum_x * (1.0 / dim)
    diff = x - mean
    var = tl.sum(diff * diff) * (1.0 / dim)
    inv_std = 1.0 / tl.sqrt(var + eps)

    # normalize
    norm_w = tl.load(norm_weight_ptr + c, mask=c < dim, other=0.0)
    norm_b = tl.load(norm_bias_ptr + c, mask=c < dim, other=0.0)
    x_norm = (diff * inv_std) * norm_w + norm_b   # shape dim
    # cast to float16 for remaining ops
    x_norm_fp16 = x_norm.to(tl.float16)

    # loop over hidden dim
    d = tl.arange(0, BLOCK_D)
    for offset in range(0, hidden_dim, BLOCK_D):
        # compute actual d idx
        d_idx = offset + d
        mask_d = d_idx < hidden_dim

        # load weight slices for each matrix
        # left_proj
        left_proj_ptrs = left_proj_w_ptr + c[:, None] * hidden_dim + d_idx[None, :]
        left_proj = tl.load(left_proj_ptrs, mask=(c[:, None] < dim) & mask_d[None,:], other=0.0)  # shape (dim, BLOCK_D)
        left_proj = left_proj.to(tl.float16)

        # left_gate
        left_gate_ptrs = left_gate_w_ptr + c[:, None] * hidden_dim + d_idx[None, :]
        left_gate_raw = tl.load(left_gate_ptrs, mask=(c[:, None] < dim) & mask_d[None,:], other=0.0).to(tl.float16)
        # compute dot
        # shape (1, BLOCK_D)
        left_proj_out = tl.dot(x_norm_fp16[None, :], left_proj)  # (1,BLOCK_D)

import triton
import triton.language as tl

@triton.jit
def tri_mul_fused_kernel(
    # pointers
    x_ptr, mask_ptr,
    norm_w_ptr, norm_b_ptr,
    left_proj_w_ptr, left_gate_w_ptr,
    right_proj_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    # sizes
    B, N, dim, hidden_dim,
    # strides
    stride_xb, stride_xn, stride_xk, stride_xc,
    stride_mb, stride_mn, stride_mk,
    stride_lob, stride_lon, stride_lok, stride_lod,
    stride_rob, stride_ron, stride_rok, stride_rod,
    stride_ogb, stride_ogn, stride_ogk, stride_ogd,
    # constants
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode b,i,j
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # -----------------------------------------------------------------
    # Load input vector x[b,i,j,:] (dim)
    c = tl.arange(0, dim)
    x_ptrs = x_ptr + b * stride_xb + i * stride_xn + j * stride_xk + c * stride_xc
    x = tl.load(x_ptrs, mask=c < dim, other=0.0).to(tl.float32)

    # -----------------------------------------------------------------
    # Layer norm over dim using provided weight/bias
    mean = tl.mean(x)                      # scalar FP32
    diff = x - mean
    var = tl.mean(diff * diff)
    inv_std = 1.0 / tl.sqrt(var + eps)

    # load norm parameters (fp16)
    norm_w = tl.load(norm_w_ptr + c, mask=c < dim, other=0.0).to(tl.float32)
    norm_b = tl.load(norm_b_ptr + c, mask=c < dim, other=0.0).to(tl.float32)

    x_norm = (diff * inv_std) * norm_w + norm_b   # FP32
    x_norm_f16 = x_norm.to(tl.float16)

    # -----------------------------------------------------------------
    # Load mask (scalar) – mask has shape [B,N,N] and is broadcasted
    mask_offset = b * stride_mb + i * stride_mn + j * stride_mk
    mask_val = tl.load(mask_ptr + mask_offset)   # fp16 (or fp32)
    mask_val = mask_val.to(tl.float16)

    # -----------------------------------------------------------------
    # Loop over hidden dimension in blocks of BLOCK_D
    d_offset = 0
    while d_offset < hidden_dim:
        d = tl.arange(0, BLOCK_D) + d_offset

def custom_kernel(data):
    """Fused Triton implementation of the outgoing TriMul operator.

    The kernel performs:
        * LayerNorm over the last dimension (dim)
        * Linear projections for the left/right paths and the three gates
        * Sigmoid activation on gate values
        * Application of the pairwise mask and element‑wise gating
    The heavy pairwise reduction (einsum) is carried out with a
    batched matrix‑multiply (torch.matmul) which maps to cuBLAS on H100.
    The final LayerNorm, gating and output linear projection are done in
    PyTorch for simplicity.
    """
    import torch
    import triton
    import triton.language as tl

    # -----------------------------------------------------------------
    # unpack arguments
    input_tensor, mask, weights, config = data
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "Input must be square in sequence dimension"
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    device = input_tensor.device

    # -----------------------------------------------------------------
    # Prepare mask – if it is None or config says nomask, use ones.
    if mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)

    # Cast inputs to fp16 for the kernel (weights will also be fp16)
    x_fp16 = input_tensor.to(torch.float16)

    # Allocate fused outputs in fp16
    left = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    right = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)
    out_gate = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Obtain strides (in elements, not bytes) for all tensors
    x_stride = x_fp16.stride()
    mask_stride = mask_tensor.stride()
    left_stride = left.stride()
    right_stride = right.stride()
    out_gate_stride = out_gate.stride()

    # Cast all weight tensors to fp16 (they are bias‑free)
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # Triton launch
    total_elements = B * N * N
    # each program id handles one (b,i,j). Use a single program per element.
    grid = (total_elements,)

    # Compile‑time block size for the hidden‑dim loop.
    BLOCK_D = 32  # must divide hidden_dim? masked for remainder.

    tri_mul_fused_kernel[grid](
        x_fp16, mask_tensor,
        norm_w, norm_b,
        left_proj_w, left_gate_w,
        right_proj_w, right_gate_w,
        out_gate_w,
        left, right, out_gate,
        B, N, dim, hidden_dim,
        x_stride[0], x_stride[1], x_stride[2], x_stride[3],
        mask_stride[0], mask_stride[1], mask_stride[2],
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_gate_stride[0], out_gate_stride[1], out_gate_stride[2], out_gate_stride[3],
        eps=eps,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # Convert back to fp32 for the remaining ops (cuBLAS works in fp16 but we keep fp32 for accuracy)
    left = left.to(torch.float32)
    right = right.to(torch.float32)
    out_gate = out_gate.to(torch.float32)

    # Batched matrix multiplication: left @ right^T for each hidden channel.
    # shape transformations: (B,N,N,H) -> (B,H,N,N)
    left_perm = left.permute(0, 3, 1, 2)   # B,H,N,N
    right_perm = right.permute(0, 3, 1, 2) # B,H,N,N
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # B,H,N,N
    out = out.permute(0, 2, 3, 1)  # B,N,N,H

    # Normalization of the intermediate hidden representation
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )

    # Apply the final output gate
    out = out * out_gate

    # Final linear projection back to the original channel dimension
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # bias is None

    return out

"""
TriMul (outgoing) – Triton‑fused part

The heavy N³ reduction is expressed as a batched matrix multiply
(left @ rightᵀ) which is executed by torch.matmul (cuBLAS).  The
remaining work (LayerNorm, the five linear projections, sigmoid
gates and mask application) is fused into a single Triton kernel.
Each Triton program processes one pair‑wise element (b,i,j) and
produces the three tensors required for the subsequent matmul:

    left  : (B, N, N, H)   = (x·W_left_proj)   * sigmoid(x·W_left_gate)   * mask
    right : (B, N, N, H)   = (x·W_right_proj)  * sigmoid(x·W_right_gate)  * mask
    out_g : (B, N, N, H)   = sigmoid(x·W_out_gate)

All linear layers share the same normalized input x̂ (LayerNorm over the
last dimension).  Computations are performed in fp16 for speed, while
the final matmul and post‑processing use fp32 to preserve accuracy.
"""
import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_fused_kernel(
    # input and mask
    x_ptr, mask_ptr,
    # LayerNorm parameters
    norm_w_ptr, norm_b_ptr,
    # projection / gate weight pointers
    left_proj_w_ptr, left_gate_w_ptr,
    right_proj_w_ptr, right_gate_w_ptr,
    out_gate_w_ptr,
    # output tensors
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    # problem sizes
    B, N, dim, hidden_dim,
    # strides (in elements, not bytes)
    stride_xb, stride_xn, stride_xk, stride_xc,
    stride_mb, stride_mn, stride_mk,
    stride_lob, stride_lon, stride_lok, stride_lod,
    stride_rob, stride_ron, stride_rok, stride_rod,
    stride_ogb, stride_ogn, stride_ogk, stride_ogd,
    # compile‑time constants
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # -----------------------------------------------------------------
    # decode global indices (batch, i, j)
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # -----------------------------------------------------------------
    # load one input vector x[b,i,j,:]  (dim elements)
    c = tl.arange(0, dim)
    x_ptrs = x_ptr + b * stride_xb + i * stride_xn + j * stride_xk + c * stride_xc
    x = tl.load(x_ptrs, mask=c < dim, other=0.0).to(tl.float32)  # fp32 for stable stats

    # -----------------------------------------------------------------
    # LayerNorm over dim
    mean = tl.mean(x)                      # scalar fp32
    diff = x - mean
    var = tl.mean(diff * diff)             # scalar fp32
    inv_std = 1.0 / tl.sqrt(var + eps)

    # load LayerNorm gamma/beta (fp16 -> fp32 for the arithmetic)
    norm_w = tl.load(norm_w_ptr + c, mask=c < dim, other=0.0).to(tl.float32)
    norm_b = tl.load(norm_b_ptr + c, mask=c < dim, other=0.0).to(tl.float32)

    x_norm = (diff * inv_std) * norm_w + norm_b   # fp32
    x_norm_f16 = x_norm.to(tl.float16)           # fp16 for the GEMM‑like work

    # -----------------------------------------------------------------
    # mask is a scalar per pair (b,i,j); broadcast to hidden dim
    mask_offset = b * stride_mb + i * stride_mn + j * stride_mk
    mask_val = tl.load(mask_ptr + mask_offset).to(tl.float16)

    # -----------------------------------------------------------------
    # loop over hidden dimension in chunks of BLOCK_D
    d_offset = 0
    while d_offset < hidden_dim:
        d = tl.arange(0, BLOCK_D) + d_offset            # hidden indices
        d_valid = d < hidden_dim                         # mask for tail

        # ---------- left projection ----------
        left_w_ptrs = left_proj_w_ptr + c[:, None] * hidden_dim + d[None, :]
        left_w = tl.load(left_w_ptrs,
                         mask=(c[:, None] < dim) & d_valid[None, :],
                         other=0.0).to(tl.float16)      # (dim, BLOCK_D)
        left_proj = tl.dot(x_norm_f16[None, :], left_w)[0]   # (BLOCK_D,)

        # ---------- left gate ----------
        left_gate_w_ptrs = left_gate_w_ptr + c[:, None] * hidden_dim + d[None, :]
        left_gate_w = tl.load(left_gate_w_ptrs,
                              mask=(c[:, None] < dim) & d_valid[None, :],
                              other=0.0).to(tl.float16)
        left_gate_pre = tl.dot(x_norm_f16[None, :], left_gate_w)[0]
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_pre))

        # ---------- right projection ----------
        right_w_ptrs = right_proj_w_ptr + c[:, None] * hidden_dim + d[None, :]
        right_w = tl.load(right_w_ptrs,
                          mask=(c[:, None] < dim) & d_valid[None, :],
                          other=0.0).to(tl.float16)
        right_proj = tl.dot(x_norm_f16[None, :], right_w)[0]

        # ---------- right gate ----------
        right_gate_w_ptrs = right_gate_w_ptr + c[:, None] * hidden_dim + d[None, :]
        right_gate_w = tl.load(right_gate_w_ptrs,
                               mask=(c[:, None] < dim) & d_valid[None, :],
                               other=0.0).to(tl.float16)
        right_gate_pre = tl.dot(x_norm_f16[None, :], right_gate_w)[0]
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_pre))

        # ---------- out gate ----------
        out_gate_w_ptrs = out_gate_w_ptr + c[:, None] * hidden_dim + d[None, :]
        out_gate_w = tl.load(out_gate_w_ptrs,
                             mask=(c[:, None] < dim) & d_valid[None, :],
                             other=0.0).to(tl.float16)
        out_gate_pre = tl.dot(x_norm_f16[None, :], out_gate_w)[0]
        out_gate_val = 1.0 / (1.0 + tl.exp(-out_gate_pre))

        # ---------- apply mask and store ----------
        left_res = left_proj * left_gate * mask_val
        right_res = right_proj * right_gate * mask_val

        left_ptrs = left_out_ptr + b * stride_lob + i * stride_lon + j * stride_lok + d * stride_lod
        right_ptrs = right_out_ptr + b * stride_rob + i * stride_ron + j * stride_rok + d * stride_rod
        out_gate_ptrs = out_gate_out_ptr + b * stride_ogb + i * stride_ogn + j * stride_ogk + d * stride_ogd

        tl.store(left_ptrs, left_res, mask=d_valid)
        tl.store(right_ptrs, right_res, mask=d_valid)
        tl.store(out_gate_ptrs, out_gate_val, mask=d_valid)

        d_offset += BLOCK_D

    """

    The kernel fuses LayerNorm, the five linear projections, sigmoid

    """

    B, N, N2, dim = input_tensor.shape

    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # mask (if missing create a tensor of ones)
    if mask is None:

    else:
        mask_tensor = mask.to(torch.float16)

    x_fp16 = input_tensor.to(torch.float16)

    left_stride = left.stride()
    right_stride = right.stride()
    out_gate_stride = out_gate.stride()

    # Cast model weights to fp16 (they are stored without bias)
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # -----------------------------------------------------------------
    # launch Triton kernel – one program per (b,i,j) pair
    total_elements = B * N * N
    grid = (total_elements,)

    # hidden‑dim block size (compile‑time constant).  32 works for all
    # hidden dimensions up to 384 (the remainder is masked).
    BLOCK_D = 32

    tri_mul_fused_kernel[grid](
        # pointers
        x_fp16, mask_tensor,
        norm_w, norm_b,
        left_proj_w, left_gate_w,
        right_proj_w, right_gate_w,
        out_gate_w,
        left, right, out_gate,
        # problem sizes
        B, N, dim, hidden_dim,
        # strides
        x_stride[0], x_stride[1], x_stride[2], x_stride[3],
        mask_stride[0], mask_stride[1], mask_stride[2],
        left_stride[0], left_stride[1], left_stride[2], left_stride[3],
        right_stride[0], right_stride[1], right_stride[2], right_stride[3],
        out_gate_stride[0], out_gate_stride[1], out_gate_stride[2], out_gate_stride[3],
        # constants
        eps=eps,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # -----------------------------------------------------------------
    # Convert to fp32 for the remainder (matmul, layer‑norm, final linear)
    left = left.to(torch.float32)
    right = right.to(torch.float32)
    out_gate = out_gate.to(torch.float32)

    # Batched matmul: (B,N,N,H) -> (B,H,N,N), multiply left @ rightᵀ
    left_perm = left.permute(0, 3, 1, 2)    # B, H, N, N
    right_perm = right.permute(0, 3, 1, 2)  # B, H, N, N
    out = torch.matmul(left_perm, right_perm.transpose(-1, -2))  # B, H, N, N
    out = out.permute(0, 2, 3, 1)  # B, N, N, H

    # Hidden‑dim LayerNorm (to_out_norm)
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to original channel dimension
    to_out_w = weights["to_out.weight"]          # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)  # bias is None

    return out
