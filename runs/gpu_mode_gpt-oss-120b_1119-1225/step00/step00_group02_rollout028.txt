def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # x norm
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape B,N,N, hidden_dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if provided (nomask True case)
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # shape B,N,N,1
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates to left/right
    left = left * left_gate
    right = right * right_gate

    # Compute outgoing TriMul: out = left @ right^T across k dimension
    # Use Triton kernel
    B, N, _, _ = left.shape

    # allocate output tensor
    out = torch.empty_like(left)   # shape B,N,N,hidden_dim

    # Get strides
    # For left and right: stride (B,N,N,H) in element counts
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Block sizes choose based on N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Launch kernel
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    import triton
    import triton.language as tl

    # Call kernel
    tri_mul_gemm_kernel[grid](

        c_ptr=out,

        N=N,

def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_ab, stride_ai, stride_ak, stride_ah,
    stride_bb, stride_bi, stride_bk, stride_bh,
    stride_cb, stride_ci, stride_cj, stride_ch,

):

    pid_n = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_bb + h * stride_bh
    c_base = c_ptr + b * stride_cb + h * stride_ch

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (dim).
2. Linear projections (no bias) to a hidden dimension H:
      left  = x @ W_left   (shape [B, N, N, H])
      right = x @ W_right  (shape [B, N, N, H])
3. Apply optional pair‑wise mask (broadcast on H) and element‑wise gates
      left  = left  * sigmoid(x @ W_left_gate)  * mask
      right = right * sigmoid(x @ W_right_gate) * mask
   The output gate `out_gate` is computed later.
4. Core operation: for every batch `b` and hidden slice `h`,
   compute the matrix product `L @ Rᵀ` where
        L[b, :, :, h]  ∈ ℝ^{N×N}
        R[b, :, :, h]  ∈ ℝ^{N×N}
   The result `out[b, :, :, h]` has shape `[N, N]`.
   This N³ reduction is performed by a Triton kernel that treats each
   `(b, h)` pair as an independent batch and carries out a blocked GEMM.
5. Layer‑norm over the hidden dimension, apply the final output gate,
   and a linear projection back to the original dimension.

The Triton kernel loads tiles of `left` and the transposed view of `right`,
accumulates with `tl.dot`, and writes the result in FP32.  All other
operations stay in PyTorch for simplicity and to keep the implementation
readable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul core (L @ Rᵀ)
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,                     # *float32  left  [B,N,N,H]
    b_ptr,                     # *float32  right [B,N,N,H]
    c_ptr,                     # *float32  out   [B,N,N,H]
    B,                         # i32  batch size
    N,                         # i32  sequence length
    H,                         # i32  hidden dimension
    stride_ab, stride_ai, stride_ak, stride_ah,   # i32 strides for A (left)
    stride_bb, stride_bi, stride_bk, stride_bh,   # i32 strides for B (right)
    stride_cb, stride_ci, stride_cj, stride_ch,   # i32 strides for C (out)
    BLOCK_M: tl.constexpr,     # tile rows   (i‑dim)
    BLOCK_N: tl.constexpr,     # tile cols   (j‑dim)
    BLOCK_K: tl.constexpr,     # tile reduction (k‑dim)
):
    # ------------------------------------------------------------------- #
    # Program IDs
    # ------------------------------------------------------------------- #
    pid_m = tl.program_id(0)          # tile index over i   (rows)
    pid_n = tl.program_id(1)          # tile index over j   (cols)
    pid_bh = tl.program_id(2)         # flattened (batch, hidden)

    # De‑compose flattened (batch, hidden) index
    b = pid_bh // H                  # batch index   ∈ [0, B)
    h = pid_bh % H                   # hidden index ∈ [0, H)

    # ------------------------------------------------------------------- #
    # Tile offsets
    # ------------------------------------------------------------------- #
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # ------------------------------------------------------------------- #
    # Base pointers for this (b, h) slice
    # ------------------------------------------------------------------- #
    a_base = a_ptr + b * stride_ab + h * stride_ah
    b_base = b_ptr + b * stride_bb + h * stride_bh
    c_base = c_ptr + b * stride_cb + h * stride_ch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------- #
    for k_start in range(0, N, BLOCK_K):
        k_offset = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ---- left tile: shape (BLOCK_M, BLOCK_K) ----
        a_ptrs = a_base + i_offset[:, None] * stride_ai + k_offset[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ---- right tile (transposed view): shape (BLOCK_K, BLOCK_N) ----
        # right[j, k] -> after transpose we need (k, j)
        b_ptrs = b_base + k_offset[:, None] * stride_bk + j_offset[None, :] * stride_bi
        b = tl.load(b_ptrs,
                    mask=mask_j[None, :] & mask_k[:, None],
                    other=0.0)

        # ---- rank‑K update ----
        acc += tl.dot(a, b)   # (M,K) @ (K,N) → (M,N)

    # ------------------------------------------------------------------- #
    # Write back result
    # ------------------------------------------------------------------- #
    c_ptrs = c_base + i_offset[:, None] * stride_ci + j_offset[None, :] * stride_cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Entry point expected by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask_tensor, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim] (float32)
            mask_tensor  : torch.Tensor [B, N, N]   (bool or float) – may be None
            weights      : dict of weight tensors (see keys below)
            config       : dict with keys:
                               "dim"        – input/output feature dimension
                               "hidden_dim" – hidden projection dimension
                               "nomask"     – bool, if True the mask can be ignored
    Returns:
        output : torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # Load model parameters
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]      # [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]     # [hidden_dim, dim]

    left_gate_weight  = weights["left_gate.weight"]      # [hidden_dim, dim]
    right_gate_weight = weights["right_gate.weight"]     # [hidden_dim, dim]
    out_gate_weight   = weights["out_gate.weight"]       # [hidden_dim, dim]

    to_out_norm_weight = weights["to_out_norm.weight"]   # [hidden_dim]
    to_out_norm_bias   = weights["to_out_norm.bias"]     # [hidden_dim]
    to_out_weight      = weights["to_out.weight"]        # [dim, hidden_dim]

    # ------------------------------------------------------------------- #
    # 1. Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------- #
    # torch.nn.functional.linear expects weight of shape [out_features, in_features]
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------- #
    # 3. Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------- #
    if not cfg.get("nomask", False) and mask is not None:
        # ensure mask is float32 and broadcastable
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,H]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5. Core TriMul multiplication: out = left @ rightᵀ over the k‑dim.
    #    This is the only part executed on the GPU via Triton.
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape
    out = torch.empty_like(left)   # allocate [B, N, N, H]

    # Ensure tensors are contiguous (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Tensor strides (in element counts, not bytes)
    stride_left = left.stride()    # (B, N, N, H)
    stride_right = right.stride()
    stride_out = out.stride()

    # ------------------------------------------------------------------- #
    # Launch parameters – tile sizes chosen for good occupancy on H100
    # ------------------------------------------------------------------- #
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # reduction tile size

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * H,                                 # (batch, hidden) tiles
    )

    # Triton kernel call
    tri_mul_gemm_kernel[grid](
        a_ptr=left,
        b_ptr=right,
        c_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_ab=stride_left[0],
        stride_ai=stride_left[1],
        stride_ak=stride_left[2],
        stride_ah=stride_left[3],
        stride_bb=stride_right[0],
        stride_bi=stride_right[1],
        stride_bk=stride_right[2],
        stride_bh=stride_right[3],
        stride_cb=stride_out[0],
        stride_ci=stride_out[1],
        stride_cj=stride_out[2],
        stride_ch=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # enough warps to keep the SM busy
    )

    # ------------------------------------------------------------------- #
    # 6. Post‑processing: layer‑norm over hidden_dim, output gate, final linear
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate                     # element‑wise gating

    # Final projection back to `dim`
    output = F.linear(out, to_out_weight)    # [B, N, N, dim]

    return output
