def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Layer norm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )
    # linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask handling
    if not config.get("nomask", True):
        # mask shape (bs, N, N)
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (bs, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Convert left and right to float16 for matmul kernel
    left_f16 = left.to(torch.float16)
    right_f16 = right.permute(0, 3, 2, 1).contiguous().to(torch.float16)  # transpose k and j

    # reshape to batched matrices (batch*hidden, N, N)
    left_flat = left_f16.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)
    right_flat = right_f16.contiguous().view(bs * hidden_dim, N, N)  # already (batch, hidden, K, N)

    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # get strides
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bk, stride_bn = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), bs * hidden_dim)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_ab,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cb,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape output back to (bs, N, N, hidden)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # to float32
    out = out.to(torch.float32)

    # to_out_norm (layernorm over hidden)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # out gating
    out = out * out_gate

    # final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul “outgoing” forward pass (AlphaFold3) accelerated with a custom Triton
batched‑matrix‑multiply kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,D]).
2. Project the normalized tensor to a hidden dimension H with four linear layers:
   * left_proj, right_proj  – primary projections (no bias)
   * left_gate, right_gate, out_gate – sigmoid‑gated scalars
3. (Optional) Apply a pair‑wise mask on the two projections.
4. Multiply each projection by its gate.
5. Compute the core “TriMul” operation:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched GEMM: for every hidden slice h we multiply
   (N×N) @ (N×N)ᵀ.  The heavy work is performed by a Triton kernel that
   processes many (B·H) independent matrix multiplies in parallel.
   The intermediate multiplication is done in FP16 with FP32 accumulation.
6. Reshape the result back to [B,N,N,H] and convert to FP32.
7. Apply a second Layer‑norm over the hidden dimension, multiply by out_gate,
   and finally linearly project back to the original dimension D.

The kernel works for any batch size ≤2, sequence length up to 1024, and
hidden dimensions used in the tests (128).  It falls back to PyTorch for
all non‑compute‑bound parts, meeting the requirement of “implement at
least part of the operations in a kernel”.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (all = seq_len)
    stride_ab, stride_am, stride_ak,   # strides for A (batch, M, K)
    stride_bb, stride_bk, stride_bn,   # strides for B (batch, K, N)
    stride_cb, stride_cm, stride_cn,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched  (M×K) @ (K×N) → (M×N)  for many independent batches."""
    pid_b = tl.program_id(2)               # batch index (B * H)
    pid_m = tl.program_id(0)               # tile row
    pid_n = tl.program_id(1)               # tile column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointers to the start of the tile in A and B
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Loop over the K dimension in blocks
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)                       # (BLOCK_M, BLOCK_K)·(BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_ak             # advance K‑block
        b_ptrs += BLOCK_K * stride_bk

    # Write the result (cast back to FP16)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, D]
            - mask: torch.Tensor of shape [B, N, N] (may be dummy when nomask=True)
            - weights: dict of model weights (see code for required keys)
            - config: dict with at least 'dim' and 'hidden_dim', optional 'nomask'

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]                 # hidden dimension (e.g. 128)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (applies before gating)
    use_mask = not config.get("nomask", True)
    if use_mask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # (B,N,N,1)
        left_proj  = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear layers, no bias)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul (batched GEMM) – kernel works in FP16
    #    left : (B,N,N,H)  -> (B,H,N,N) -> (B*H,N,N)
    #    right: (B,N,N,H)  -> transpose k/j -> (B,H,N,N) -> (B*H,N,N)
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16).permute(0, 3, 2, 1).contiguous()  # (B,H,N,N)

    # reshape to (batch*H, N, N)
    left_flat  = left_f16.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)      # (B*H, M, K)
    right_flat = right_f16.view(B * H, N, N)                                      # (B*H, K, N)

    # allocate output buffer (FP16)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # strides in element units
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bk, stride_bn = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    # Block sizes – chosen to fit H100 shared‑memory limits
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # tiles along M (i dimension)
        triton.cdiv(N, BLOCK_N),   # tiles along N (j dimension)
        B * H,                     # one program per (batch * hidden) slice
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_bk, stride_bn,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H) and convert to FP32 for the remaining ops
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply the output gate
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final projection back to original dimension D
    out = F.linear(out, weights["to_out.weight"])

    return out
