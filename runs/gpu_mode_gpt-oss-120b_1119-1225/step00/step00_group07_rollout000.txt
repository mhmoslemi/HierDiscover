@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # combine batch and hidden dimension: each pid correspond to a (b, h) pair
    b = pid // H
    h = pid % H
    # i and j block indices
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        # left: [i,k]
        left_ptrs = left_ptr + b*stride_lb + h*stride_lh + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k < N), other=0.0).to(tl.float32)

        # right: [j,k]
        right_ptrs = right_ptr + b*stride_rb + h*stride_rh + offs_j[:, None] * stride_ri + offs_k[None, :] * stride_rk
        right = tl.load(right_ptrs, mask=mask_j[:, None] & (offs_k < N), other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    out_ptrs = out_ptr + b*stride_ob + h*stride_oh + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implementation.

The kernel computes the core tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right have already been projected, masked and gated.
We fuse this N³ reduction using a Triton tiled GEMM: each (b,d) pair
is processed as a separate batched matrix multiply (N×N)·(N×N)ᵀ.
The surrounding LayerNorms, linear projections and gating are
performed with PyTorch for simplicity.  The kernel works in FP16
with FP32 accumulation and returns FP16, which is later cast to
float32 for the final LayerNorm/linear steps.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ------------------------------------------------------------------
# Triton kernel for the triangular multiplicative update (outgoing)
# ------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # *mut left  [B,N,N,H] (fp16)
    right_ptr,              # *mut right [B,N,N,H] (fp16)
    out_ptr,                # *mut out   [B,N,N,H] (fp16)
    B, N, H,                # shape parameters
    stride_lb, stride_li, stride_lk, stride_lh,  # left strides
    stride_rb, stride_ri, stride_rk, stride_rh,  # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,  # out strides
    BLOCK_M: tl.constexpr,  # tile size in i
    BLOCK_N: tl.constexpr,  # tile size in j
    BLOCK_K: tl.constexpr   # tile size in k (reduction)
):
    # --------------------------------------------------------------
    # program IDs:
    #   pid0 : (batch * hidden) flattened -> one GEMM per (b, h)
    #   pid1 : block row for i dimension
    #   pid2 : block col for j dimension
    # --------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden indices from pid0
    b = pid0 // H            # batch index
    h = pid0 % H             # hidden (channel) index

    # coordinate offsets for the i and j tiles
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ---- load left tile: shape (BLOCK_M, BLOCK_K) ----
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + h * stride_lh
            + offs_i[:, None] * stride_li
            + cur_k[None, :] * stride_lk
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- load right tile: shape (BLOCK_N, BLOCK_K) ----
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + h * stride_rh
            + offs_j[:, None] * stride_ri
            + cur_k[None, :] * stride_rk
        )
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # ---- accumulate dot product over k ----
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K)·(BLOCK_N, BLOCK_K)ᵀ -> (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write back the result (store as fp16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + h * stride_oh
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_oj
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ------------------------------------------------------------------
# Entrypoint required by the grader
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N] (float32 or bool)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # LayerNorm over the last channel dimension (dim)
    # --------------------------------------------------------------
    # torch.mean / var are already very fast; we reuse them.
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    # Apply learned affine parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = x_norm * norm_weight + norm_bias

    # --------------------------------------------------------------
    # Linear projections (no bias) and gating
    # --------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]        # (hidden_dim, dim)

    left_gate_weight = weights["left_gate.weight"]          # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]        # (hidden_dim, dim)
    out_gate_weight = weights["out_gate.weight"]            # (hidden_dim, dim)

    # Project
    left = F.linear(x_norm, left_proj_weight)              # (B,N,N,hidden)
    right = F.linear(x_norm, right_proj_weight)            # (B,N,N,hidden)

    # Gating logits → sigmoid
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # --------------------------------------------------------------
    # Apply mask (if present) and element‑wise gates
    # --------------------------------------------------------------
    if mask is not None:
        # mask has shape (B,N,N); broadcast over hidden dim
        mask = mask.to(dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Triton kernel for the N³ contraction
    # --------------------------------------------------------------
    # Cast to fp16 for the kernel (saved memory & faster dot)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    # Allocate output buffer
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides for the contiguous tensors
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    B, N, _, _ = left.shape  # (B,N,N,hidden)

    # Tiling parameters – experimentally good for H100
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_ri=stride_ri,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Cast back to fp32 for the remaining post‑processing
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # Output LayerNorm + gating + final projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # Apply final gating
    out = out * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)

    return out
