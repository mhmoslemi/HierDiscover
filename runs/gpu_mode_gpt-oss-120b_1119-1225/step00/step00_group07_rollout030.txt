import torch
import triton
import triton.language as tl

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    The kernel computes the cubic pairwise update:
        out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    where left/right are linear projections of the normalized input,
    gated by sigmoid gates and optionally masked.
    The heavy O(N^3 * hidden_dim) term is executed in a fused
    Triton kernel that treats each (batch, channel) pair as an
    independent matrix‑multiply (N×N)·(N×N)^T.
    """
    import torch
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------
    # 1. LayerNorm on input x
    # --------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # --------------------------
    # 2. Linear projections and gates
    # --------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # linear projections (FP32)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # optional mask
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 3. Trilinear multiplicative update via Triton
    # -------------------------------------------------
    # Cast to half for the compute‑intensive part
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # allocate output buffer (float32 accumulation)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements)
    stride_lhs_batch = left_h.stride(0)
    stride_lhs_i = left_h.stride(1)
    stride_lhs_k = left_h.stride(2)
    stride_lhs_d = left_h.stride(3)

    stride_rhs_batch = right_h.stride(0)
    stride_rhs_j = right_h.stride(1)   # j dimension (second N)
    stride_rhs_k = right_h.stride(2)
    stride_rhs_d = right_h.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_d = out.stride(3)  # should be 1

    # Kernel definition
    @triton.jit
    def tri_mul_kernel(
        lhs_ptr, rhs_ptr, out_ptr,
        M, N, K,
        stride_lhs_batch, stride_lhs_i, stride_lhs_k, stride_lhs_d,
        stride_rhs_batch, stride_rhs_j, stride_rhs_k, stride_rhs_d,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
        hidden_dim: tl.constexpr,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid = tl.program_id(0)

        # Compute block coordinates
        total_blocks_m = tl.cdiv(M, BLOCK_M)
        total_blocks_n = tl.cdiv(N, BLOCK_N)
        blocks_per_batch = total_blocks_m * total_blocks_n

        batch_idx = pid // blocks_per_batch
        block_idx = pid % blocks_per_batch
        block_m = block_idx // total_blocks_n
        block_n = block_idx % total_blocks_n

        # Calculate indices within M and N
        offs_i = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_i = offs_i < M
        mask_j = offs_j < N

        # Decode (batch, channel) from flat batch index
        b_idx = batch_idx // hidden_dim
        d_idx = batch_idx % hidden_dim

        # Base pointers for this (b,d) slice
        lhs_base = lhs_ptr + b_idx * stride_lhs_batch + d_idx * stride_lhs_d
        rhs_base = rhs_ptr + b_idx * stride_rhs_batch + d_idx * stride_rhs_d
        out_base = out_ptr + b_idx * stride_out_batch + d_idx * stride_out_d

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over K dimension in tiles
        for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
            k_offset = k_tile * BLOCK_K
            offs_k = k_offset + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            # Load left tile [M, K]
            lhs_ptrs = lhs_base + (offs_i[:, None] * stride_lhs_i) + (offs_k[None, :] * stride_lhs_k)
            lhs = tl.load(lhs_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

            # Load right tile as [K, N] (note the swapped order)
            rhs_ptrs = rhs_base + (offs_k[:, None] * stride_rhs_k) + (offs_j[None, :] * stride_rhs_j)
            rhs = tl.load(rhs_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

            # Accumulate dot product
            acc += tl.dot(lhs, rhs)

        # Write back result
        out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # Launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    total_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = B * hidden_dim
    grid = (total_batches * total_blocks_m * total_blocks_n,)

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        N, N, N,
        stride_lhs_batch, stride_lhs_i, stride_lhs_k, stride_lhs_d,
        stride_rhs_batch, stride_rhs_j, stride_rhs_k, stride_rhs_d,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # might adjust based on size
    )

    # ------------------------------
    # 4. Post‑processing: to_out_norm, gating and final linear
    # ------------------------------
    # LayerNorm over hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate  # apply output gate

    # Final linear back to original dim
    final = torch.nn.functional.linear(out_norm, to_out_weight)  # [B, N, N, dim]

    return final
```

"""
custom_kernel: Triton‑accelerated forward pass of the TriMul (outgoing) operator.

The heavy O(N³·C_hidden) update
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed with a fused Triton kernel that treats each (batch, channel) pair as an
independent (N×N)·(N×N)ᵀ matrix multiplication.  All other operations
(LayerNorm, Linear projections, sigmoid gates, optional masking, and the final
normalisation + linear projection) are done in PyTorch.  The kernel works on
half‑precision inputs (to cut memory traffic) while accumulating in float32,
and writes the result back as float32.
"""

import torch
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask_tensor:  torch.Tensor of shape [B, N, N] or None
            - weights_dict: mapping from weight name to torch.Tensor
            - config_dict: must contain "dim" (C) and "hidden_dim" (C_hidden);
                           may contain "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # 1. unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 2. LayerNorm on the input (over last dim C)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 3. Linear projections & sigmoid gates
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]          # [hidden_dim, C]
    right_proj_weight = weights["right_proj.weight"]        # [hidden_dim, C]
    left_gate_weight = weights["left_gate.weight"]          # [hidden_dim, C]
    right_gate_weight = weights["right_gate.weight"]        # [hidden_dim, C]
    out_gate_weight = weights["out_gate.weight"]            # [hidden_dim, C]
    to_out_norm_weight = weights["to_out_norm.weight"]      # [hidden_dim]
    to_out_norm_bias = weights["to_out_norm.bias"]          # [hidden_dim]
    to_out_weight = weights["to_out.weight"]                # [C, hidden_dim]

    # linear projections (float32)
    left = torch.nn.functional.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # optional mask (if config["nomask"] is False)
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Triton kernel for the cubic accumulation
    # ------------------------------------------------------------------
    # Cast to half to reduce bandwidth; kernel accumulates in float32
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape

    # output buffer (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides in element units (as given by PyTorch)
    stride_lhs_batch = left_h.stride(0)   # N * N * hidden_dim
    stride_lhs_i = left_h.stride(1)       # N * hidden_dim
    stride_lhs_k = left_h.stride(2)       # hidden_dim
    stride_lhs_d = left_h.stride(3)       # 1

    stride_rhs_batch = right_h.stride(0)
    stride_rhs_j = right_h.stride(1)      # same as stride_lhs_i (row index j)
    stride_rhs_k = right_h.stride(2)      # same as stride_lhs_k
    stride_rhs_d = right_h.stride(3)      # 1

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_d = out.stride(3)          # should be 1

    @triton.jit
    def tri_mul_kernel(
        lhs_ptr, rhs_ptr, out_ptr,
        M, N, K,
        stride_lhs_batch, stride_lhs_i, stride_lhs_k, stride_lhs_d,
        stride_rhs_batch, stride_rhs_j, stride_rhs_k, stride_rhs_d,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
        hidden_dim: tl.constexpr,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        pid = tl.program_id(0)

        # block grid layout -------------------------------------------------
        total_blocks_m = tl.cdiv(M, BLOCK_M)
        total_blocks_n = tl.cdiv(N, BLOCK_N)
        blocks_per_batch = total_blocks_m * total_blocks_n

        batch_idx = pid // blocks_per_batch
        block_idx = pid % blocks_per_batch
        block_m = block_idx // total_blocks_n
        block_n = block_idx % total_blocks_n

        # indices inside the tile
        offs_i = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_i = offs_i < M
        mask_j = offs_j < N

        # decode original batch & channel from the flattened batch index
        b_idx = batch_idx // hidden_dim
        d_idx = batch_idx % hidden_dim

        # base pointers for this (b, d) slice
        lhs_base = lhs_ptr + b_idx * stride_lhs_batch + d_idx * stride_lhs_d
        rhs_base = rhs_ptr + b_idx * stride_rhs_batch + d_idx * stride_rhs_d
        out_base = out_ptr + b_idx * stride_out_batch + d_idx * stride_out_d

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # iterate over K dimension in tiles
        for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
            k_offset = k_tile * BLOCK_K
            offs_k = k_offset + tl.arange(0, BLOCK_K)
            mask_k = offs_k < K

            # left tile: shape (BLOCK_M, BLOCK_K)
            lhs_ptrs = lhs_base + (offs_i[:, None] * stride_lhs_i) + (offs_k[None, :] * stride_lhs_k)
            lhs = tl.load(lhs_ptrs,
                          mask=mask_i[:, None] & mask_k[None, :],
                          other=0.0)

            # right tile as (BLOCK_K, BLOCK_N) – note the swapped axes
            rhs_ptrs = rhs_base + (offs_k[:, None] * stride_rhs_k) + (offs_j[None, :] * stride_rhs_j)
            rhs = tl.load(rhs_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

            # accumulate dot product
            acc += tl.dot(lhs, rhs)

        # write back result
        out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # launch configuration
    BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 32
    total_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    total_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_batches = B * hidden_dim
    grid = (total_batches * total_blocks_m * total_blocks_n,)

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        N, N, N,
        stride_lhs_batch, stride_lhs_i, stride_lhs_k, stride_lhs_d,
        stride_rhs_batch, stride_rhs_j, stride_rhs_k, stride_rhs_d,
        stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 5. Post‑processing: LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_norm = out_norm * out_gate        # apply output gate
    final = torch.nn.functional.linear(out_norm, to_out_weight)  # [B, N, N, C]

    return final
