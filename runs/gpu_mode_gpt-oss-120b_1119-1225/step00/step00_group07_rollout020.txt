@triton.jit
def mask_gate_fusion(
    src_ptr, gate_ptr, mask_ptr, dst_ptr,
    B, N, H,
    stride_src_b0, stride_src_b1, stride_src_b2, stride_src_b3,
    stride_gate_b0, stride_gate_b1, stride_gate_b2, stride_gate_b3,
    stride_mask_b0, stride_mask_b1, stride_mask_b2,
    BLOCK_N: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    # flatten batch*seq indices? We'll map (b, i, j) to a linear program id.
    # Use grid: num_blocks = B * N * N
    total = B * N * N
    block_idx = pid
    # compute b,i,j from block_idx
    b = block_idx // (N * N)
    rem = block_idx % (N * N)
    i = rem // N
    j = rem % N

    # compute offsets for n dimension (size H) via range
    offs_h = tl.arange(0, BLOCK_H)

    # compute pointer offsets
    src = src_ptr + b * stride_src_b0 + i * stride_src_b1 + j * stride_src_b2 + offs_h * stride_src_b3
    gate = gate_ptr + b * stride_gate_b0 + i * stride_gate_b1 + j * stride_gate_b2 + offs_h * stride_gate_b3
    mask = mask_ptr + b * stride_mask_b0 + i * stride_mask_b1 + j * stride_mask_b2
    dst = dst_ptr + b * stride_src_b0 + i * stride_src_b1 + j * stride_src_b2 + offs_h * stride_src_b3

    src_val = tl.load(src, mask=offs_h < H)
    gate_val = tl.load(gate, mask=offs_h < H)
    mask_val = tl.load(mask, mask=tl.full([1], True, dtype=tl.int1)) # broadcast
    # mask is scalar per (b,i,j); we broadcast across H
    mask_val = tl.broadcast_to(mask_val, [BLOCK_H])
    out_val = src_val * gate_val * mask_val
    tl.store(dst, out_val, mask=offs_h < H)
```

    src_ptr, gate_ptr, mask_ptr, dst_ptr,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    stride_src_b, stride_src_i, stride_src_j, stride_src_h,

    stride_mask_b, stride_mask_i, stride_mask_j,

):

    stride_src_b: tl.constexpr, stride_src_i: tl.constexpr, stride_src_j: tl.constexpr, stride_src_h: tl.constexpr,
    stride_gate_b: tl.constexpr, stride_gate_i: tl.constexpr, stride_gate_j: tl.constexpr, stride_gate_h: tl.constexpr,
    stride_mask_b: tl.constexpr, stride_mask_i: tl.constexpr, stride_mask_j: tl.constexpr,

"""
custom_kernel: Triton‑accelerated forward pass of the TriMul (outgoing) module.

Algorithm
---------
1. Layer‑normalize the input tensor over the feature dimension `dim`.
2. Compute four linear maps (no bias) on the normalized tensor:
   * left  = x @ W_left          (shape [..., hidden_dim])
   * right = x @ W_right
   * left_gate  = sigmoid(x @ W_left_gate)
   * right_gate = sigmoid(x @ W_right_gate)
   * out_gate   = sigmoid(x @ W_out_gate)   (used later)
3. Fuse mask‑application and gate multiplication for `left` and `right`
   with a tiny Triton kernel:
        out = left * mask * left_gate   (and analog for right)
   The kernel works on a 3‑D grid (batch, i, j) and processes the
   hidden dimension in blocks (`BLOCK_H`).  It loads a vector of
   `hidden_dim` values, multiplies by the scalar mask and gate, and
   writes the result.
4. Cast the gated tensors to FP16 and perform the core N³ contraction.
   For each hidden channel the operation is a matrix multiplication:
        out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
   This is equivalent to a batched GEMM:
        left_fp16 (B*H, N, N)  @  right_fp16.transpose(-1,-2) → (B*H, N, N)
   The result is reshaped back to (B, N, N, H) and promoted to FP32.
5. Apply a second LayerNorm over the hidden dimension, multiply by
   `out_gate` (computed in step 2) and finish with a linear projection
   back to `dim`.
The kernel is only used for the cheap element‑wise mask/gate fusion; the
heavy N³ kernel is implemented with the highly‑optimized cuBLAS batched
matrix multiplication, which on an H100 (Tensor‑cores) easily meets the
target runtime.

Input:
    data = (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]   (float32)
        mask         : [B, N, N]        (bool/int/float) – may be omitted.
        weights      : dict of model parameters (no bias on Linear layers)
        config       : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

Output:
    Tensor of shape [B, N, N, dim] (float32)
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_fuse_kernel(
    src_ptr,               # float32*   left/right projected tensor
    gate_ptr,              # float32*   corresponding gate (sigmoid)
    mask_ptr,              # float32*   mask scalar per (b,i,j)
    dst_ptr,               # float32*   output (same shape as src)
    B, N, H,               # runtime dimensions
    stride_src_b, stride_src_i, stride_src_j, stride_src_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr  # compile‑time block size (must ≥ hidden_dim)
):
    """
    Element‑wise fusion: dst = src * gate * mask
    Grid: (B, N, N) → one program per (b,i,j) coordinate.
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # vector of hidden indices
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H            # guard for hidden_dim < BLOCK_H

    # Pointers to the start of the 1‑D hidden slice
    src_offset = (
        pid_b * stride_src_b
        + pid_i * stride_src_i
        + pid_j * stride_src_j
        + offs_h * stride_src_h
    )
    gate_offset = (
        pid_b * stride_gate_b
        + pid_i * stride_gate_i
        + pid_j * stride_gate_j
        + offs_h * stride_gate_h
    )

    src_vec = tl.load(src_ptr + src_offset, mask=mask_h, other=0.0)
    gate_vec = tl.load(gate_ptr + gate_offset, mask=mask_h, other=0.0)

    # scalar mask value for this (b,i,j)
    mask_offset = pid_b * stride_mask_b + pid_i * stride_mask_i + pid_j * stride_mask_j
    mask_val = tl.load(mask_ptr + mask_offset)

    # broadcast scalar mask over the hidden dimension
    out_vec = src_vec * gate_vec * mask_val

    # store
    tl.store(dst_ptr + src_offset, out_vec, mask=mask_h)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul operator.
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # sanity check for our Triton block size
    BLOCK_H = 128
    assert hidden_dim <= BLOCK_H, "hidden_dim exceeds BLOCK_H (128) in this implementation"

    # ------------------------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections + gate values (float32)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # (B,N,N,hidden_dim)
    right_proj = F.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Handle mask (float32)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask_fp = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)
    else:
        mask_fp = mask.to(dtype)

    # ------------------------------------------------------------------
    # 4. Fuse mask & gate (Triton kernel)
    # ------------------------------------------------------------------
    # allocate output tensors for the fused results
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # launch the kernel for left and right separately
    # grid: (batch, seq_len, seq_len)
    grid = (bs, seq_len, seq_len)

    # left
    _mask_gate_fuse_kernel[grid](
        left_proj, left_gate, mask_fp, left,
        bs, seq_len, hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask_fp.stride(0), mask_fp.stride(1), mask_fp.stride(2),
        BLOCK_H=BLOCK_H,
    )
    # right
    _mask_gate_fuse_kernel[grid](
        right_proj, right_gate, mask_fp, right,
        bs, seq_len, hidden_dim,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_fp.stride(0), mask_fp.stride(1), mask_fp.stride(2),
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 5. Core N³ contraction via batched GEMM (FP16 for speed)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape  # B, N, N, hidden_dim
    # reshape to (B*H, N, N)
    left_flat = left_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_flat = right_fp16.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication: left @ rightᵀ  -> (B*H, N, N)
    out_fp16 = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim) and promote to FP32
    out = out_fp16.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
