import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matmul of shape [B,N,N] x [B,N,N]^T
@triton.jit
def trmul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_i,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets for each tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # pointers
        left_ptrs = left_ptr \
            + b * stride_left_batch \
            + i[:, None] * stride_left_i \
            + k[None, :] * stride_left_k

        right_ptrs = right_ptr \
            + b * stride_right_batch \
            + k[:, None] * stride_right_k \
            + j[None, :] * stride_right_i

        # loads
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left_tile, right_tile)

    # store
    out_ptrs = out_ptr \
        + b * stride_out_batch \
        + i[:, None] * stride_out_i \
        + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    '''
    Triton‐accelerated implementation of the “outgoing” TriMul operator.
    It follows the reference PyTorch code:
        - Layer‑norm on the input
        - Linear projections (left/right) and three gates
        - Masking and gate fusion
        - Core N³ interaction computed by a custom Triton kernel
        - Output layer‑norm, gating and final linear projection.
    The heavy N³ term is performed with a fused batched matrix multiplication
    (left @ rightᵀ) across the hidden dimension.
    '''
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # optional mask handling
    if mask is None or config.get("nomask", False):
        mask = torch.ones(input_tensor.shape[:3], dtype=input_tensor.dtype, device=input_tensor.device)

    # ---- 1. LayerNorm on the last dim ----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---- 2. Linear projections (no bias) ----
    left_proj_w = weights["left_proj.weight"]  # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ---- 3. Gates ----
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ---- 4. Apply mask (if any) ----
    # mask shape [B, N, N]; unsqueeze last dim to broadcast over hidden dim
    mask_f = mask.unsqueeze(-1).type_as(x)
    left = left * mask_f
    right = right * mask_f

    # ---- 5. Apply element‑wise gates ----
    left = left * left_gate
    right = right * right_gate

    # ---- 6. Core N³ multiplicative update using Triton ----
    # Cast to half‑precision for the heavy tile‑wise multiplication.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # N is seq_len
    # output tensor (half precision)
    out = torch.empty_like(left, dtype=torch.float16)

    # Get strides for the 3‑D view (batch, i, k) after slicing hidden dim.
    # We will launch one kernel per hidden channel.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Loop over hidden channels – each launch works on a 3‑D slice.
    for h in range(hidden_dim):
        # slice tensors for current hidden channel
        left_h = left[..., h]    # shape (B, N, N)
        right_h = right[..., h]
        out_h = out[..., h]

        # Ensure tensors are contiguous so that strides are simple.
        left_h = left_h.contiguous()
        right_h = right_h.contiguous()
        out_h = out_h.contiguous()

        stride_left = left_h.stride()   # (stride_batch, stride_i, stride_k)
        stride_right = right_h.stride()
        stride_out = out_h.stride()

        grid = (
            B,                               # batch dimension (program_id(0))
            (N + BLOCK_M - 1) // BLOCK_M,    # i‑blocks
            (N + BLOCK_N - 1) // BLOCK_N,    # j‑blocks
        )

        trmul_matmul_kernel[grid](
            left_h,
            right_h,
            out_h,
            B,
            N,
            stride_left[0], stride_left[1], stride_left[2],
            stride_right[0], stride_right[2], stride_right[1],  # note: k‑ and i‑stride are swapped for the transposed access
            stride_out[0], stride_out[1], stride_out[2],
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # Cast back to fp32 for the rest of the pipeline
    out = out.to(torch.float32)

    # ---- 7. Output layer‑norm ----
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # ---- 8. Final gating and linear projection ----
    out = out * out_gate
    to_out_w = weights["to_out.weight"]  # shape (dim, hidden_dim)
    out = F.linear(out, to_out_w)  # (B,N,N,dim)

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass

The module follows the reference implementation in the AlphaFold3 paper:

1. Layer‑norm on the input tensor (last dimension = dim).
2. Two linear projections (left / right) and three linear gates (left, right,
   output). All linear layers are bias‑free – the provided weight tensors have
   shape (out_features, in_features).
3. Optional mask (broadcast over the hidden dimension) is applied to the
   projected tensors.
4. Element‑wise gates (sigmoid) are fused with the masked projections.
5. Core N³ multiplicative update:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is the same as a batched matrix multiplication
   left[..., d] @ right[..., d]ᵀ for each hidden channel d.
   The heavy N³ term is computed by a custom Triton kernel that tiles the
   multiplication over the i, j and k dimensions (BLOCK_M × BLOCK_N × BLOCK_K)
   and accumulates in fp32 while the inputs are stored in fp16.
6. The resulting tensor (shape [B,N,N,hidden_dim]) is layer‑normed,
   multiplied by the output‑gate and finally projected back to the original
   dimension (dim) with a linear layer.

Only the N³ interaction is performed on the GPU with Triton; all remaining
operations use standard PyTorch kernels (highly‑optimised already).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul L @ Rᵀ for a single hidden channel.
#   left  : [B, N, N]  (i, k)
#   right : [B, N, N]  (j, k) – accessed as (k, j) to realise the transpose
#   out   : [B, N, N]  (i, j)
# --------------------------------------------------------------------------- #
@triton.jit
def trmul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N,                                  # problem size
    stride_left_batch, stride_left_i, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_i,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ----- program ids -------------------------------------------------------
    b = tl.program_id(0)               # batch index
    pid_m = tl.program_id(1)           # i‑block
    pid_n = tl.program_id(2)           # j‑block

    # ----- block coordinates -------------------------------------------------
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32 (more accurate than fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- iterate over the reduction axis k ---------------------------------
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[i, k]
        left_ptrs = left_ptr \
            + b * stride_left_batch \
            + i[:, None] * stride_left_i \
            + k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # rightᵀ[k, j]  ←  right[j, k] with k as the leading dimension
        right_ptrs = right_ptr \
            + b * stride_right_batch \
            + k[:, None] * stride_right_k \
            + j[None, :] * stride_right_i
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # block‑wise dot product (M×K)·(K×N) → (M×N)
        acc += tl.dot(left_tile, right_tile)

    # ----- write back --------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_batch \
        + i[:, None] * stride_out_i \
        + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Public entry point – matches the signature required by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] or None
        - weights      : dict of torch.Tensors containing all model weights
        - config       : dict with at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # --------------------------------------------------------------------- #
    # unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------- #
    # 1) Input layer‑norm (over the last dimension = dim)
    # --------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # --------------------------------------------------------------- #
    # 2) Linear projections (bias‑free)
    # --------------------------------------------------------------- #
    left_proj_w  = weights["left_proj.weight"]   # [hidden_dim, dim]
    right_proj_w = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # --------------------------------------------------------------- #
    # 3) Gates (sigmoid)
    # --------------------------------------------------------------- #
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # --------------------------------------------------------------- #
    # 4) Optional mask – broadcast over the hidden dimension
    # --------------------------------------------------------------- #
    if mask is None or config.get("nomask", False):
        mask = torch.ones(input_tensor.shape[:3],
                          dtype=input_tensor.dtype,
                          device=input_tensor.device)
    mask = mask.unsqueeze(-1).type_as(x)   # (B,N,N,1)

    left  = left  * mask
    right = right * mask

    # --------------------------------------------------------------- #
    # 5) Apply element‑wise gates
    # --------------------------------------------------------------- #
    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- #
    # 6) Core N³ multiplicative update – Triton
    # --------------------------------------------------------------- #
    # Cast to half‑precision for the heavy compute
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape
    out_half = torch.empty_like(left, dtype=torch.float16)

    # block size – tuned for H100 (fp16 tensor cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Loop over hidden channels; each launch handles a 3‑D slice [B,N,N]
    for h in range(hidden_dim):
        left_h  = left[..., h].contiguous()   # (B,N,N)
        right_h = right[..., h].contiguous()
        out_h   = out_half[..., h].contiguous()

        # strides for the 3‑D view (batch, i, k)
        stride_l = left_h.stride()   # (stride_batch, stride_i, stride_k)
        stride_r = right_h.stride()
        stride_o = out_h.stride()

        grid = (
            B,                                      # batch dimension
            (N + BLOCK_M - 1) // BLOCK_M,           # i‑blocks
            (N + BLOCK_N - 1) // BLOCK_N,           # j‑blocks
        )

        trmul_matmul_kernel[grid](
            left_h,
            right_h,
            out_h,
            B, N,
            stride_l[0], stride_l[1], stride_l[2],
            stride_r[0], stride_r[2], stride_r[1],   # note the swap (k,i) for the transposed access
            stride_o[0], stride_o[1], stride_o[2],
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )

    # Convert back to fp32 for the remainder of the pipeline
    out = out_half.to(torch.float32)

    # --------------------------------------------------------------- #
    # 7) Output layer‑norm over the hidden dimension
    # --------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # --------------------------------------------------------------- #
    # 8) Final gating and linear projection back to `dim`
    # --------------------------------------------------------------- #
    out = out * out_gate                         # (B,N,N,hidden_dim)
    to_out_w = weights["to_out.weight"]          # [dim, hidden_dim]
    out = F.linear(out, to_out_w)                # (B,N,N,dim)

    return out
