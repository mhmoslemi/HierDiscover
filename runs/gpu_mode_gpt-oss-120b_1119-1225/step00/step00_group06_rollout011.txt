def custom_kernel(data):
    """
    Triton-accelerated TriMul outgoing implementation.
    Performs layernorm, linear projections, gated elementwise ops,
    then computes batched matrix multiplication across the sequence
    dimension using a custom Triton kernel.
    Finally applies output norm, gating and final linear projection.
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # 1. LayerNorm over last dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # 2. Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3. Apply mask if needed
    if not config.get("nomask", False):
        # mask shape [B,N,N]; expand to last dim
        mask_expanded = mask.unsqueeze(-1).to(dtype)
        left = left * mask_expanded
        right = right * mask_expanded

    # 4. Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # 5. Apply gates
    left = left * left_gate
    right = right * right_gate

    # 6. Compute batched multiplication using Triton
    B, N, _, _ = left.shape
    # Permute to bring hidden dim to batch
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim

    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    out_flat = torch.empty_like(left_flat)

    # launch triton kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # prepare strides as int64
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # stride for K (original column)
    stride_bn = right_flat.stride(1)   # stride for N (original row)

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    grid = ( ( (N + BLOCK_M - 1) // BLOCK_M ) * ( (N + BLOCK_N - 1) // BLOCK_N ), batch_hidden )
    trimul_outgoing_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4
    )

    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,hidden_dim)

    # 7. Output norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # 8. Multiply by out_gate
    out = out * out_gate

    # 9. Final linear proj to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel: Triton‑accelerated outgoing Triangular Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑normalize the input tensor over the last dimension (dim).
2. Apply three linear projections (left, right, gates) to the normalized tensor.
   * left/right are projected to `hidden_dim`.
   * left_gate, right_gate, out_gate are projected to `hidden_dim` and passed through sigmoid.
3. Optional masking of left/right (mask shape [B, N, N]).
4. Apply element‑wise gating to left and right.
5. Compute the core O(N³) contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel that treats each hidden slice as an independent
   batch and performs a batched GEMM (A @ Bᵀ) with the K‑dimension tiled.
6. Apply a second LayerNorm on the hidden dimension, multiply by `out_gate`,
   and linearly project back to the original `dim`.

The heavy contraction (step 5) is the only part implemented in Triton; the
remaining operations use PyTorch’s highly‑optimized kernels.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    A_ptr,               # left  [batch_hidden, M, K]
    B_ptr,               # right [batch_hidden, N, K]  (will be accessed as K×N)
    C_ptr,               # out   [batch_hidden, M, N]
    M, N, K,
    stride_a_batch, stride_am, stride_ak,      # A strides
    stride_b_batch, stride_bk, stride_bn,      # B strides (K then N)
    stride_c_batch, stride_cm, stride_cn,      # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ Bᵀ with batched dimensions.
    A: [batch_hidden, M, K]
    B: [batch_hidden, N, K]   (accessed as K×N)
    Output C: [batch_hidden, M, N]
    """
    pid = tl.program_id(0)          # tile index over (M,N)
    batch = tl.program_id(1)        # combined (batch * hidden_dim)

    # -------------------------------------------------------------
    # tile indices
    # -------------------------------------------------------------
    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask = mask_m[:, None] & mask_n[None, :]

    # -------------------------------------------------------------
    # pointers
    # -------------------------------------------------------------
    # base offset for the current batch slice
    a_batch_offset = batch * stride_a_batch
    b_batch_offset = batch * stride_b_batch
    c_batch_offset = batch * stride_c_batch

    # A: (BLOCK_M, BLOCK_K)
    a_ptrs = A_ptr + a_batch_offset + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)

    # B is accessed as K×N (i.e. B_T).  Load shape (BLOCK_K, BLOCK_N)
    b_ptrs = B_ptr + b_batch_offset + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # main loop over K
    # -------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                     # (M,K)·(K,N) → (M,N)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # -------------------------------------------------------------
    # store result
    # -------------------------------------------------------------
    c_ptrs = C_ptr + c_batch_offset + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor: torch.Tensor of shape [B, N, N] (may be ignored if config["nomask"] is True)
            - weights_dict: mapping of weight names to torch.Tensors
            - config_dict: contains "dim", "hidden_dim", optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape [B, N, N])
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply elementwise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 6. Triton kernel: batched GEMM over the sequence dimension
    #    Reshape to treat hidden_dim as part of the batch.
    # ------------------------------------------------------------------
    # bring hidden_dim to batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_hidden = B * hidden_dim

    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # allocate output for the contraction
    out_flat = torch.empty_like(left_flat)

    # -------------------------------------------------------------
    # stride information (all int64)
    # -------------------------------------------------------------
    stride_a_batch = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_bk = right_flat.stride(2)   # column stride → K dimension
    stride_bn = right_flat.stride(1)   # row stride    → N dimension

    stride_c_batch = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m * num_pid_n, batch_hidden)

    trimul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Output LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ------------------------------------------------------------------
    # 8. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
