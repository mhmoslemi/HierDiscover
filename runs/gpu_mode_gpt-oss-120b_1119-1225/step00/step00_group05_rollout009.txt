@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0) # row block
    pid_n = tl.program_id(1) # col block
    batch_pid = tl.program_id(2) # batch block (B*H)
    
    # Compute block start offsets:
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # No block index for k because we will loop over k
    # offset for k
    off_k = tl.arange(0, BLOCK_K)
    
    # Compute base pointer offsets
    a_ptrs = a_ptr + batch_pid * stride_a_batch + off_m[:, None] * stride_a_m + off_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_pid * stride_b_batch + off_k[:, None] * stride_b_k + off_n[None, :] * stride_b_n
    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # Loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(off_m[:, None] < M) & (off_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(off_k[:, None] < K) & (off_n[None, :] < N), other=0.0)
        # Cast to fp32
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k
    # Write result
    c_ptrs = c_ptr + batch_pid * stride_c_batch + off_m[:, None] * stride_c_m + off_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc.to(tl.float16) if c_ptr.dtype == tl.float16 else acc, mask=(off_m[:, None] < M) & (off_n[None, :] < N))

"""
custom_kernel: Fast Triton implementation of the “outgoing” TriMul operator.

Algorithm
---------
1️⃣ Layer‑norm the input tensor over the feature dimension (dim).
2️⃣ Linear projections (left_proj, right_proj) → shape [B,N,N,H].
3️⃣ (Optional) Apply the pairwise mask.
4️⃣ Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized input and multiply them into left/right/out respectively.
5️⃣ Compute the core tensor contraction:
       out[...,i,j,d] = Σ_k left[...,i,k,d] * right[...,j,k,d]
   This is a batched matrix‑multiply for each hidden channel:
       out_d = left_d @ right_dᵀ   (d = 0…H‑1).
   We fuse this with a custom Triton kernel that treats each (batch, channel)
   pair as an independent GEMM (A: M×K, B: K×N, C: M×N) using FP16
   arithmetic and FP32 accumulation.
6️⃣ Layer‑norm over the hidden dimension, multiply by out_gate.
7️⃣ Final linear projection back to the original dimension.

The kernel processes the N³ multiplication efficiently (≈128 × N³ FLOPs)
by tiling the M‑ and N‑dimensions (64 × 64) and looping over K in blocks
(32).  All heavy‑compute work is done in the Triton kernel; the remaining
operations are performed with highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication (A @ Bᵀ) for many (B*H) batches.
# A : (batch, M, K)   – left tiles  (i, k)
# B : (batch, K, N)   – right_T    (k, j)
# C : (batch, M, N)   – output     (i, j)
# --------------------------------------------------------------------------- #
@triton.jit
def _batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # 3‑D grid: (block_i, block_j, batch_channel)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (batch, channel) slice
    a_batch = a_ptr + pid_b * stride_a_batch
    b_batch = b_ptr + pid_b * stride_b_batch
    c_batch = c_ptr + pid_b * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Pointers to the current sub‑tiles of A and B
        a_ptrs = a_batch + (offs_i[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_k) + (offs_j[None, :] * stride_b_n)

        # Load with masks for OOB elements (use 0.0 as the padding value)
        a = tl.load(a_ptrs,
                    mask=(offs_i[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_j[None, :] < N),
                    other=0.0, dtype=tl.float16)

        # Cast to FP32 before dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Accumulate the block‑dot product
        acc += tl.dot(a, b)

    # Write the result back to C (still FP32)
    c_ptrs = c_batch + (offs_i[:, None] * stride_c_m) + (offs_j[None, :] * stride_c_n)
    tl.store(c_ptrs,
             acc,
             mask=(offs_i[:, None] < M) & (offs_j[None, :] < N))


# --------------------------------------------------------------------------- #
# Entry‑point: custom kernel used by the evaluator.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (x, mask, weights, config)
        - x            : torch.Tensor, shape [B, N, N, D] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool or float) or None
        - weights      : dict of torch.Tensors containing all model params
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------- #
    # 1️⃣ Unpack inputs and hyper‑parameters
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # 2️⃣ Layer‑norm over the feature dimension (dim)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape [B, N, N, D], dtype float32

    # ------------------------------------------------------------------- #
    # 3️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj_w  = weights["left_proj.weight"]      # (H, D)
    right_proj_w = weights["right_proj.weight"]     # (H, D)

    left  = F.linear(x_norm, left_proj_w)   # [B, N, N, H]  float32
    right = F.linear(x_norm, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 4️⃣ Optional masking (applied before gates)
    # ------------------------------------------------------------------- #
    if not nomask and mask is not None:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 5️⃣ Compute sigmoid gates (no bias) and apply them
    # ------------------------------------------------------------------- #
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w)) # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))   # [B, N, N, H]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 6️⃣ Core contraction via Triton: out = Σ_k left[...,i,k,:] * right[...,j,k,:]
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape

    # Cast to FP16 for the heavy GEMM
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Rearrange tensors so that each (batch, channel) pair forms a standard
    # matrix multiply:   left: (i, k)   right_T: (k, j)
    left_t  = left_fp16.permute(0, 3, 1, 2).contiguous()   # (B, H, N, K)
    right_t = right_fp16.permute(0, 3, 2, 1).contiguous()  # (B, H, K, N)

    # Merge batch and channel dimensions -> (B*H, N, N)
    left_mat  = left_t.view(B * hidden_dim, N, N)   # A : (M=N, K=N)
    right_mat = right_t.view(B * hidden_dim, N, N)  # B : (K=N, N)

    # Allocate output buffer (FP32 accumulation needed for next layer‑norm)
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)

    # Strides for the Triton kernel
    stride_a_batch = left_mat.stride(0)
    stride_a_m     = left_mat.stride(1)
    stride_a_k     = left_mat.stride(2)

    stride_b_batch = right_mat.stride(0)
    stride_b_k     = right_mat.stride(1)
    stride_b_n     = right_mat.stride(2)

    stride_c_batch = out_mat.stride(0)
    stride_c_m     = out_mat.stride(1)
    stride_c_n     = out_mat.stride(2)

    # Tile sizes – tuned for H100 FP16 matmul
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # blocks along M (i)
        triton.cdiv(N, BLOCK_N),      # blocks along N (j)
        B * hidden_dim,               # one program for each (batch, channel)
    )

    _batched_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H] and convert to FP32 for layer‑norm
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 7️⃣ Second layer‑norm, out‑gate, and final projection
    # ------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )   # [B, N, N, H]

    out = out * out_gate                     # apply the output gate

    to_out_w = weights["to_out.weight"]      # (D, H)
    out = F.linear(out, to_out_w)            # final linear → [B, N, N, D] (float32)

    return out
```
