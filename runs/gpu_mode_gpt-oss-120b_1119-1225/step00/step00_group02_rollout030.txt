    right = right * right_gate

    # compute output: left @ right_T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))   # shape [B*H,N,N]
    out = out_mat.reshape(B, H, N, N).permute(0,2,3,1).contiguous()
    # out normalization and gating and final linear projection can be fused in a Triton kernel
    out = fused_norm_gate_proj(out, out_gate, to_out_norm_weight, to_out_norm_bias,
                               to_out_weight)
    return out

"""
TriMul forward (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the last dimension.
2. Linear projection → left/right tensors (shape B×N×N×H) and their
   sigmoid‑gates (same shape).
3. Fuse mask, gate and element‑wise multiplication in a tiny Triton
   kernel (one program per (b,i,j) position).  The kernel loads a
   whole hidden‑dimensional vector (max H = 384) and writes left/out
   tensors after applying:  out = proj * gate * mask.
4. Compute the triangular update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiply L·Rᵀ for each hidden channel.
   By reshaping (B,H,N,N) → (B*H,N,N) we can reuse cuBLAS via
   torch.bmm, which is highly optimised on H100.
5. Layer‑norm the result over the hidden dimension, multiply by the
   output‑gate, and a final linear projection back to the original
   dimension.

Only the mask‑gate‑fusion step is implemented in Triton; all heavy
linear/algebra operations are delegated to PyTorch (cuBLAS) for
maximum throughput.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate and element‑wise multiplication
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,          # projected tensor   (B,N,N,H)
    gate_ptr,          # gate tensor        (B,N,N,H)
    mask_ptr,          # mask tensor        (B,N,N)
    out_ptr,           # output tensor      (B,N,N,H)
    B, N, hidden_dim,  # sizes
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b,  stride_out_i,  stride_out_j,  stride_out_h,
    BLOCK_H: tl.constexpr,
):
    """Apply ``proj * sigmoid(gate) * mask`` element‑wise.

    Each program processes one (b,i,j) entry and loops over the hidden
    dimension in a single vector of size BLOCK_H (≥ hidden_dim).  The
    mask is a scalar per (b,i,j) and is broadcast to the hidden dim.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode (b,i,j) from the linear program id
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # hidden‑dim offsets
    h_offs = tl.arange(0, BLOCK_H)

    # load scalar mask (float, 0. or 1.)
    mask = tl.load(mask_ptr + b * stride_mask_b
                             + i * stride_mask_i
                             + j * stride_mask_j)

    # load projection and gate vectors, masking out‑of‑range lanes
    proj = tl.load(proj_ptr + b * stride_proj_b
                              + i * stride_proj_i
                              + j * stride_proj_j
                              + h_offs * stride_proj_h,
                   mask=h_offs < hidden_dim, other=0.0)
    gate = tl.load(gate_ptr + b * stride_gate_b
                              + i * stride_gate_i
                              + j * stride_gate_j
                              + h_offs * stride_gate_h,
                   mask=h_offs < hidden_dim, other=0.0)

    # fused multiplication
    out = proj * gate * mask

    # store result
    tl.store(out_ptr + b * stride_out_b
                       + i * stride_out_i
                       + j * stride_out_j
                       + h_offs * stride_out_h,
             out,
             mask=h_offs < hidden_dim)


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D]
            - mask_tensor : torch.Tensor of shape [B, N, N] (optional)
            - weights     : dict of weight tensors
            - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # Layer‑norm over the feature dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias
    )  # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gate logits
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (H, D)
    right_proj_weight = weights["right_proj.weight"]  # (H, D)
    left_gate_weight  = weights["left_gate.weight"]   # (H, D)
    right_gate_weight = weights["right_gate.weight"]  # (H, D)
    out_gate_weight   = weights["out_gate.weight"]    # (H, D)

    # project
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,H]

    # gates (sigmoid activated)
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Prepare mask (float, 1.0 where allowed, 0.0 where masked)
    # ------------------------------------------------------------------
    if mask_tensor is None or mask_tensor.numel() == 0 or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=x.dtype, device=device)

    # ------------------------------------------------------------------
    # Fuse mask & gate with a Triton kernel (produces gated tensors)
    # ------------------------------------------------------------------
    # allocate output containers
    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Triton launch configuration
    total_positions = B * N * N
    grid = (total_positions,)

    # Choose a compile‑time block size that covers the maximum hidden dim
    BLOCK_H = 384  # >= max hidden_dim required by the tests

    # left
    apply_gate_mask_kernel[grid](
        left, left_gate, mask, left_out,
        B, N, hidden_dim,
        *left.stride(),
        *left_gate.stride(),
        *mask.stride(),
        *left_out.stride(),
        BLOCK_H=BLOCK_H,
    )
    # right
    apply_gate_mask_kernel[grid](
        right, right_gate, mask, right_out,
        B, N, hidden_dim,
        *right.stride(),
        *right_gate.stride(),
        *mask.stride(),
        *right_out.stride(),
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # Triangular update = batched matrix multiplication L @ Rᵀ
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for each hidden channel
    left_flat = left_out.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # (B*H, N, N)
    # transpose the inner N‑N dimensions of right (swap axes 1↔2) then flatten
    right_T = right_out.permute(0, 3, 2, 1).contiguous()                     # (B, H, K, J)
    right_flat = right_T.view(-1, N, N)                                      # (B*H, N, N)

    # batched matmul: (i,k) @ (k,j) -> (i,j)
    out_flat = torch.bmm(left_flat, right_flat)                               # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final LayerNorm, output‑gate, and projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # (D, H)

    # hidden‑dim layer‑norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    # apply output gate
    out_gated = out_norm * out_gate

    # final linear projection (bias = False in the reference)
    output = torch.nn.functional.linear(out_gated, to_out_weight)

    return output
