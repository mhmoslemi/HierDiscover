def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    ...
    # compute dtype
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # LayerNorm pos: use torch.nn.functional.layer_norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # left/right proj
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # left/right gating weights linear
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # mask
    if mask_tensor is not None:

        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Cast to half for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Prepare output tensor
    out_tensor = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Get strides
    stride_b, stride_i, stride_k, stride_h = left_half.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out_tensor.stride()

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # we can set 32 or 64.
    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

"""
Triangular Multiplicative Update (outgoing) – Triton implementation

The kernel computes the core contraction
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
for every batch `b` and hidden channel `h`.  The tensor layout is
    [B, N, N, H]  (B – batch, N – sequence length, H – hidden_dim).

The algorithm:
1. Layer‑norm the input.
2. Linear projections for left/right and three gating vectors.
3. Apply optional pairwise mask and the sigmoid gates.
4. Cast left/right to FP16 and launch a Triton kernel that treats each
   (b, h) pair as an independent GEMM of size (N×N)·(N×N)ᵀ.
   The kernel uses a 2‑D block‑tiling (BLOCK_M×BLOCK_N) and an inner
   K‑tiling (BLOCK_K).  The inner product is performed with `tl.dot`,
   which maps to Tensor‑Core ACCs (FP16×FP16→FP32).
5. Layer‑norm the hidden result, multiply by the output gate and apply
   the final linear projection back to the original channel dimension.
All heavy arithmetic (the N³ contraction) is performed on the GPU via Triton,
while the surrounding linear / norm / sigmoid ops use PyTorch’s highly‑optimized
implementations.
"""

    B, N, H,                            # problem size
    stride_b, stride_i, stride_k, stride_h,  # left/right strides
    out_stride_b, out_stride_i, out_stride_j, out_stride_h,  # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute one (batch, hidden) slice of the TriMul contraction.
    Grid dimensions:
        axis 0 – (b * H + h)          → total slices = B*H
        axis 1 – block rows over N    → i dimension
        axis 2 – block cols over N    → j dimension
    """
    # -------------------------------------------------------------
    # program IDs
    pid_bh = tl.program_id(0)   # combined batch+hidden
    pid_i  = tl.program_id(1)   # block row
    pid_j  = tl.program_id(2)   # block col

    # -------------------------------------------------------------
    # decode batch and hidden channel from pid_bh
    b = pid_bh // H
    h = pid_bh % H

    # start offsets in the i/j dimensions
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # 1‑D index vectors
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)

    # boundary masks for i/j
    mask_i = i < N
    mask_j = j < N

    # base address for this (b,h) slice
    base = b * stride_b + h * stride_h

    # accumulator in FP32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # loop over the K dimension (the contracted axis)
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- left block (M×K)
        # offsets: i * stride_i + k * stride_k
        a_ptrs = base + i[:, None] * stride_i + k[None, :] * stride_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)                     # FP16 load

        # ----- right block (K×N) – note the transposition
        # offsets: k * stride_k + j * stride_i  (j is the “column”)
        b_ptrs = base + k[:, None] * stride_k + j[None, :] * stride_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)                     # FP16 load

        # ----- accumulate
        acc += tl.dot(a, b)        # FP16·FP16 → FP32

    # -------------------------------------------------------------
    # store the result into out[b,i,j,h]
    out_ptrs = base + i[:, None] * out_stride_i + j[None, :] * out_stride_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Entry‑point implementing the outgoing TriMul module.
    Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # shapes & config
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # ---- LayerNorm on the input ------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # -----------------------------------------------------------------
    # ---- Linear projections (FP32) ---------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # ---- Gating tensors (sigmoid) ----------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # ---- Optional pairwise mask ------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left_proj  = left_proj * mask
        right_proj = right_proj * mask

    # -----------------------------------------------------------------
    # ---- Apply gates ------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # ---- Cast to FP16 for the heavy contraction --------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer for the contracted hidden tensor
    out_hidden = torch.empty((B, N, N, hidden_dim),
                             dtype=torch.float32,
                             device=device)

    # -----------------------------------------------------------------
    # ---- Strides (all tensors are contiguous with layout [B, i, k, h])
    stride_b, stride_i, stride_k, stride_h = left_fp16.stride()
    # Output strides: i‑stride = stride_i, j‑stride = stride_k (same layout)
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out_hidden.stride()

    # -----------------------------------------------------------------
    # ---- Triton launch configuration --------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_hidden,
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,  # H100 works well with 8 warps per block
    )

    # -----------------------------------------------------------------
    # ---- Post‑contraction LayerNorm, gating and final linear -------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_hidden,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=eps)

    out_norm = out_norm * out_gate          # apply output gate (FP32)

    # final projection back to `dim`
    out = F.linear(out_norm, weights["to_out.weight"])
    return out
