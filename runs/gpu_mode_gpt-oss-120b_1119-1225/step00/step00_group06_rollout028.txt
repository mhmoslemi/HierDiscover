Pseudo-code:

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_j, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch and hidden index
    h = pid_bh % H
    b = pid_bh // H

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        # Check bounds for loading left and right
        i = i0 + tl.arange(0, BLOCK_M)[:, None]  # shape (M,1)
        j = j0 + tl.arange(0, BLOCK_N)[None, :]  # shape (1,N)
        # Offsets left: b*stride_left_b + i*stride_left_i + k*stride_left_k + h
        left_off = b * stride_left_b + i * stride_left_i + k[None, :] * stride_left_k + h
        left_mask = (i < N) & (k < N)
        left = tl.load(left_ptr + left_off, mask=left_mask, other=0.0)  # shape (M, K)

        # Offsets right: b*stride_right_b + j*stride_right_j + k*stride_right_k + h
        right_off = b * stride_right_b + j * stride_right_j + k[:, None] * stride_right_k + h
        # Note: right indexing layout: we need right[b, j, k, h] (j dimension is second, k third). So stride_right_j = N*H (jump over j dimension), stride_right_k = H.
        # To use broadcasting correctly for matmul: left is (M,K), right need to be (K,N) => we need shape (K,N) for right, thus we should load right as shape (K,N). We'll set j dimension as columns.
        right_mask = (j < N) & (k < N)
        right = tl.load(right_ptr + right_off, mask=right_mask, other=0.0)  # shape (K, N)

        # compute multiplication: sum over k of left[i,k] * right[k,j]
        acc += tl.dot(left, right)   # shape (M,N)

    # write output
    # Compute out offset: b*stride_out_b + (i0 + tl.arange(0,BLOCK_M)) * stride_out_i + (j0 + tl.arange(0,BLOCK_N))[None,:] * stride_out_j + h

    out_mask_i = out_off_i < N
    out_mask_j = out_off_j < N

    tl.store(out_ptr + out_off, acc, mask=out_mask)

def tri_mul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k,
    stride_right_b, stride_right_j, stride_right_k,
    stride_out_b, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    Compute out[b,i,j,h] = sum_{k} left[b,i,k,h] * right[b,j,k,h]

    Implements batched GEMM for each (b,h) using tile based reduction.

"""
custom_kernel for the TriMul “outgoing” update (AlphaFold3).

The implementation follows the reference PyTorch module but accelerates the
core O(N³·C_hidden) operation

    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

with a Triton kernel.  The kernel treats each (batch, hidden‑channel) pair
as an independent GEMM: left[..., :, :, c] @ right[..., :, :, c]ᵀ.
All LayerNorms, linear projections, gating and the final output linear are
performed with PyTorch (they are cheap compared to the cubic reduction).
The heavy reduction is tiled (BLOCK_M × BLOCK_N) and accumulates over k
in blocks (BLOCK_K) using `tl.dot`, accumulating in FP32 for numerical
stability.  The kernel works for any batch size (1 or 2), sequence length
(≤ 1024) and hidden dimension (≤ 384) and respects an optional mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: computes the einsum reduction
# --------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,            # ptr to left  [B,N,N,H]  (float32)
    right_ptr,           # ptr to right [B,N,N,H]  (float32)
    out_ptr,             # ptr to out   [B,N,N,H]  (float32)
    B, N, H,             # problem sizes (int64)
    stride_left_b, stride_left_i, stride_left_k,   # strides for left
    stride_right_b, stride_right_j, stride_right_k,  # strides for right
    stride_out_b, stride_out_i, stride_out_j,     # strides for out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    This is equivalent to a batched GEMM per hidden channel:
        out_h = left_h @ right_hᵀ
    """
    pid_i = tl.program_id(0)               # tile over i (rows)
    pid_j = tl.program_id(1)               # tile over j (cols)
    pid_bh = tl.program_id(2)              # combined batch * hidden index

    # ---------------------------------------------------------
    # decode batch id and hidden channel
    # ---------------------------------------------------------
    b = pid_bh // H                         # batch index
    h = pid_bh % H                          # hidden channel index
    # make sure they are 64‑bit integers for pointer arithmetics
    b = tl.cast(b, tl.int64)
    h = tl.cast(h, tl.int64)

    # ---------------------------------------------------------
    # tile origin in the i/j dimensions
    # ---------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------
    # loop over the reduction dimension k in blocks
    # ---------------------------------------------------------
    k = 0
    while k < N:
        cur_k = k + tl.arange(0, BLOCK_K)                # (K,)
        k_valid = cur_k < N

        i_range = i0 + tl.arange(0, BLOCK_M)            # (M,)
        i_valid = i_range < N

        j_range = j0 + tl.arange(0, BLOCK_N)            # (N,)
        j_valid = j_range < N

        # ----- left block: shape (M, K) ---------------------------------
        # address = b*stride_left_b + i*stride_left_i + k*stride_left_k + h
        left_offset = (
            b * stride_left_b
            + (i_range[:, None] * stride_left_i)
            + (cur_k[None, :] * stride_left_k)
            + h
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=i_valid[:, None] & k_valid[None, :],
            other=0.0,
        )  # (M, K)

        # ----- right block: shape (K, N) --------------------------------
        # we need rows = k, cols = j → offset = b*SB + k*SK + j*SJ + h
        right_offset = (
            b * stride_right_b
            + (cur_k[:, None] * stride_right_k)
            + (j_range[None, :] * stride_right_j)
            + h
        )
        right = tl.load(
            right_ptr + right_offset,
            mask=k_valid[:, None] & j_valid[None, :],
            other=0.0,
        )  # (K, N)

        # accumulate the matmul
        acc += tl.dot(left, right)   # (M, N)

        k += BLOCK_K

    # ---------------------------------------------------------
    # write the result tile
    # ---------------------------------------------------------
    out_offset = (
        b * stride_out_b
        + (i_range[:, None] * stride_out_i)
        + (j_range[None, :] * stride_out_j)
        + h
    )
    tl.store(
        out_ptr + out_offset,
        acc,
        mask=i_valid[:, None] & j_valid[None, :],
    )
# --------------------------------------------------------------
# Python entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # fetch weights (all tensors are assumed to be on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]          # (H, D)
    right_proj_weight = weights["right_proj.weight"]        # (H, D)
    left_gate_weight = weights["left_gate.weight"]          # (H, D)
    right_gate_weight = weights["right_gate.weight"]        # (H, D)
    out_gate_weight = weights["out_gate.weight"]            # (H, D)
    to_out_norm_weight = weights["to_out_norm.weight"]      # (H,)
    to_out_norm_bias = weights["to_out_norm.bias"]          # (H,)
    to_out_weight = weights["to_out.weight"]                # (D, H)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, D]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # left and right have shape [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)          # (B,N,N,H)
    right = F.linear(x, right_proj_weight)        # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [B, N, N, 1]
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid on linear transforms of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate          # still [B,N,N,H]
    right = right * right_gate

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5) Heavy cubic reduction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)     # will hold [B,N,N,H] in float32

    # Block sizes – you may tune these for your hardware
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * hidden_dim,                 # batch × hidden‑channel tiles
    )

    # launch the kernel
    tri_mul_einsum_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2),
        right.stride(0), right.stride(1), right.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm over hidden_dim, gating and final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
