import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_size, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(0)  # batch index
    pid_m = tl.program_id(1)  # row block index
    pid_n = tl.program_id(2)  # col block index

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for block of A and B
    a_ptrs = A_ptr + pid_b * stride_a_batch + (offs_m[:, None] * stride_a_row + offs_k[None, :] * stride_a_col)
    b_ptrs = B_ptr + pid_b * stride_b_batch + (offs_k[:, None] * stride_b_row + offs_n[None, :] * stride_b_col)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B tiles
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K - k*BLOCK_K), other=0.0)

def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ensure mask dtype
    if mask_tensor is None:
        mask_tensor = torch.ones(input_tensor.shape[:3], dtype=input_tensor.dtype, device=input_tensor.device)
    mask = mask_tensor

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)  # shape [B,N,N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply mask and gating
    mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape [B,N,N,1]
    left = left * mask_exp * left_gate
    right = right * mask_exp * right_gate

    # Reorder for batched matmul
    B = left.shape[0]
    N = left.shape[1]  # since shape is [B,N,N,hidden_dim]
    # Actually left shape [B,N,N,hidden_dim]; permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # flatten batch and hidden
    total = B * hidden_dim
    left_mat = left_perm.view(total, N, N)
    # right needs transpose on last two dims
    right_mat = right_perm.view(total, N, N).transpose(2,1).contiguous()  # shape (total,N,N)

    # Allocate output
    out_mat = torch.empty_like(left_mat, device=left_mat.device)

    # Define kernel parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Get strides
    stride_a_batch, stride_a_i, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_mat.stride()
    stride_c_batch, stride_c_i, stride_c_k = out_mat.stride()
    # checkpoint: a_i = stride over rows, a_k = stride over columns etc.

    # Launch kernel
    grid = (total, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))
    batched_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        total, N, N, N,
        stride_a_batch, stride_a_i, stride_a_k,
        stride_b_batch, stride_b_i, stride_b_k,
        stride_c_batch, stride_c_i, stride_c_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Reshape output back
    out_perm = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1)  # shape [B,N,N,hidden_dim]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_perm, (hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)

    # apply out gate
    out = out_norm * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul (outgoing) implementation using Triton.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C).
2. Project the normalized tensor onto a hidden dimension H with two
   linear matrices (`left_proj`, `right_proj`) and compute three
   sigmoid‑gates (`left_gate`, `right_gate`, `out_gate`).
3. Apply the optional pairwise mask and the two input gates:
   left  = left_proj  * mask * left_gate
   right = right_proj * mask * right_gate
4. Fuse the O(N³) contraction
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   into a batched matrix multiplication:
      left₍b,d₎ (N×N)  @  right₍b,d₎ᵀ (N×N) → out₍b,d₎ (N×N)
   The heavy GEMM is performed by a custom Triton kernel that
   processes each (batch × hidden) pair as an independent GEMM.
5. Reshape the result back to (B, N, N, H) and apply a second
   layer‑norm (`to_out_norm`), the output gate (`out_gate`), and a
   final linear projection back to the original channel dimension C.

The kernel works in FP32 (you may cast to FP16 for extra speed) and
covers arbitrary batch sizes (B ≤ 2) and hidden dimensions.  All other
operations are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton batched matrix‑multiply kernel (A @ Bᵀ) for the N³ contraction
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_batch = tl.program_id(0)      # which (batch × hidden) pair
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block column

    # offsets within the M/N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the tile in A and B
    a_ptrs = (A_ptr + pid_batch * stride_a_batch
              + offs_m[:, None] * stride_a_row
              + offs_k[None, :] * stride_a_col)
    b_ptrs = (B_ptr + pid_batch * stride_b_batch
              + offs_k[:, None] * stride_b_row
              + offs_n[None, :] * stride_b_col)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension in tiles
    num_k = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k):
        # predicates for the current K‑tile
        cur_mask_a = mask_m[:, None] & (offs_k[None, :] < K - k * BLOCK_K)
        cur_mask_b = (offs_k[:, None] < K - k * BLOCK_K) & mask_n[None, :]

        a = tl.load(a_ptrs, mask=cur_mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=cur_mask_b, other=0.0)

        # dot product of shape (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_a_col
        b_ptrs += BLOCK_K * stride_b_row

    # write back only the in‑bounds region
    c_ptrs = (C_ptr + pid_batch * stride_c_batch
              + offs_m[:, None] * stride_c_row
              + offs_n[None, :] * stride_c_col)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns
    -------
    torch.Tensor
        Output of shape (B, N, N, C) (float32)
    """
    torch.cuda.set_device(torch.cuda.current_device())

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (all bias‑free)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # (B,N,N,H)
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3. Apply mask (if any) and input gates
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(input_tensor.shape[:3], dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1).to(x.dtype)   # (B,N,N,1)

    left = left_proj * mask * left_gate    # (B,N,N,H)
    right = right_proj * mask * right_gate

    # ------------------------------------------------------------------
    # 4. Prepare tensors for the batched GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # (B, H, N, N) -> flatten B×H dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B,H,N,N)

    total = B * hidden_dim
    left_mat = left_perm.view(total, N, N)               # (total,N,N)
    # transpose the last two dims of right so we can do A @ Bᵀ
    right_mat = right_perm.view(total, N, N).transpose(2, 1).contiguous()  # (total,N,N)

    # output buffer
    out_mat = torch.empty_like(left_mat, device=left_mat.device)

    # ------------------------------------------------------------------
    # 5. Launch Triton kernel for the O(N³) contraction
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    stride_a_batch, stride_a_i, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_mat.stride()
    stride_c_batch, stride_c_i, stride_c_k = out_mat.stride()

    grid = (total,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        total,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_c_batch,
        stride_c_i,
        stride_c_k,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Reshape back to (B,N,N,H) and continue with the remaining layers
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # second layer‑norm over the hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # apply the output gate
    out = out * out_gate

    # final linear back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
