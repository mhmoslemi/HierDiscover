@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_h = tl.program_id(2) # batch_h index (0..B*H-1)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    # mask for out of bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers for left subblock: each row i and col k
    left_ptrs = left_ptr + pid_h * stride_left_batch + offs_i[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
    # load left tile
    left = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k < N), other=0.0).to(tl.float16) # maybe cast to fp16
    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset base = k*BLOCK_K
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        # if at last block, bound checks
        mask_k = offs_k < N

        # Left pointer for current K block
        left_ptrs = left_ptr + pid_h * stride_left_batch + offs_i[:, None]*stride_left_m + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        # Right pointer: shape (K,N) need offset k row index and column j
        right_ptrs = right_ptr + pid_h * stride_right_batch + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_n
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        acc += tl.dot(left_tile, right_tile) # left tile shape (M,K), right tile (K,N)
    # Write output
    out_ptrs = out_ptr + pid_h * stride_out_batch + offs_i[:, None] * stride_out_m + offs_j[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

```python
import torch
import triton
import triton.language as tl

@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_h = tl.program_id(2)   # batch * hidden index
    pid_i = tl.program_id(0)   # row block
    pid_j = tl.program_id(1)   # column block

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left tile (M x K)
        left_tile_ptr = left_ptr + pid_h * stride_left_batch \
                         + offs_i[:, None] * stride_left_m \
                         + cur_k[None, :] * stride_left_k
        left = tl.load(left_tile_ptr,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float16)

        # Load right tile (K x N)
        right_tile_ptr = right_ptr + pid_h * stride_right_batch \
                         + cur_k[:, None] * stride_right_k \
                         + offs_j[None, :] * stride_right_n
        right = tl.load(right_tile_ptr,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float16)

        # Accumulate dot product
        acc += tl.dot(left, right)   # (M,N) in fp32

    # Write out
    out_ptrs = out_ptr + pid_h * stride_out_batch \
               + offs_i[:, None] * stride_out_m \
               + offs_j[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])
```

"""
custom_kernel
==============

Triton‑accelerated forward pass of the “outgoing” Triangle Multiplicative Update
used in AlphaFold‑3.  The heavy O(B·H·N³) operation

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

is implemented as a batched matrix multiplication (left @ rightᵀ) with a
custom Triton kernel.  All other operations (LayerNorm, Linear projections,
sigmoid gates, masking) are performed with PyTorch.  Mixed‑precision
computation (FP16 inside the kernel, FP32 for the LayerNorms and final
projection) reduces memory traffic while keeping the final result in
float32.

Input
-----
data : tuple
    (input_tensor, mask_tensor, weights_dict, config_dict)
    - input_tensor : torch.Tensor, [B, N, N, C] (float32)
    - mask_tensor  : torch.Tensor or None, [B, N, N] (bool/float)
    - weights_dict : dict of model parameters (all on the same device)
    - config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

Output
------
torch.Tensor of shape [B, N, N, dim] (float32)
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM kernel for out = left @ right.T  (FP16 → FP32 accumulation)."""
    pid_h = tl.program_id(2)                # combined batch * hidden index
    pid_i = tl.program_id(0)                # output row block
    pid_j = tl.program_id(1)                # output column block

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left block   (M × K)   -> fp16
        left_ptrs = left_ptr + pid_h * stride_left_batch \
                    + offs_i[:, None] * stride_left_m \
                    + cur_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float16)

        # Load right block  (K × N)   -> fp16
        right_ptrs = right_ptr + pid_h * stride_right_batch \
                     + cur_k[:, None] * stride_right_k \
                     + offs_j[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float16)

        # Accumulate (M × N) in FP32
        acc += tl.dot(left, right)

    # Write the result back (store as FP16)
    out_ptrs = out_ptr + pid_h * stride_out_batch \
               + offs_i[:, None] * stride_out_m \
               + offs_j[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """Forward pass of the TriMul (outgoing) module with a Triton kernel."""
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]               # input / output channel dimension C
    hidden_dim = config["hidden_dim"] # intermediate dimension H
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Helper to fetch a weight tensor (ensuring correct dtype/device)
    # ------------------------------------------------------------------
    def get(name):
        w = weights[name]
        return w.to(device).to(torch.float32)

    # ------------------------------------------------------------------
    # LayerNorm on the input (over the last dim)
    # ------------------------------------------------------------------
    norm_weight = get("norm.weight")
    norm_bias   = get("norm.bias")
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, C], float32

    # ------------------------------------------------------------------
    # Linear projections (FP32 → cast to FP16 for the kernel)
    # ------------------------------------------------------------------
    left_proj_w  = get("left_proj.weight")
    right_proj_w = get("right_proj.weight")
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Optional masking (applied before gating)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype=left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid) – also cast to FP16 so that subsequent mul stays in FP16
    # ------------------------------------------------------------------
    left_gate_w  = get("left_gate.weight")
    right_gate_w = get("right_gate.weight")
    out_gate_w   = get("out_gate.weight")

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w)).to(torch.float16)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w)).to(torch.float16)
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))  # keep FP32 for later

    left  = (left.to(torch.float16) * left_gate)   # FP16
    right = (right.to(torch.float16) * right_gate) # FP16

    # ------------------------------------------------------------------
    # Prepare tensors for the Triton GEMM:
    #   left  : [B, H, N, N]   (i, k) ordering
    #   right : [B, H, N, N]   (k, j) ordering  -> transpose axes 2&3 of original right
    # ------------------------------------------------------------------
    # (B,N,N,H) -> (B,H,N,N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    # right needs a transpose on the two N axes before permuting
    right_T    = right.permute(0, 3, 2, 1).contiguous()

    B, H, N, _ = left_perm.shape  # H == hidden_dim

    # Output placeholder (FP16)
    out_perm = torch.empty_like(left_perm)  # shape [B, H, N, N], FP16

    # ------------------------------------------------------------------
    # Compute strides (flattened batch*hidden dimension)
    # ------------------------------------------------------------------
    # For tensors of shape (B, H, N, N) the stride over the combined
    # (B*H) dimension equals stride of the second axis (H) because the
    # memory layout is [B][H][N][N] row‑major.
    stride_left_batch = left_perm.stride(1)   # N*N
    stride_left_m     = left_perm.stride(2)   # N   (row i)
    stride_left_k     = left_perm.stride(3)   # 1   (col k)

    stride_right_batch = right_T.stride(1)    # N*N
    stride_right_k     = right_T.stride(2)    # N   (row k)
    stride_right_n     = right_T.stride(3)    # 1   (col j)

    stride_out_batch = out_perm.stride(1)     # N*N
    stride_out_m     = out_perm.stride(2)     # N   (row i)
    stride_out_n     = out_perm.stride(3)     # 1   (col j)

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,      # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,      # j tiles
        B * H,                             # flattened batch*hidden
    )

    _trimul_gemm_kernel[grid](
        left_ptr=left_perm.data_ptr(),
        right_ptr=right_T.data_ptr(),
        out_ptr=out_perm.data_ptr(),
        B=B,
        H=H,
        N=N,
        stride_left_batch=stride_left_batch,
        stride_left_m=stride_left_m,
        stride_left_k=stride_left_k,
        stride_right_batch=stride_right_batch,
        stride_right_k=stride_right_k,
        stride_right_n=stride_right_n,
        stride_out_batch=stride_out_batch,
        stride_out_m=stride_out_m,
        stride_out_n=stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,            # H100 gives good occupancy with 8 warps
    )

    # ------------------------------------------------------------------
    # Reshape back to original layout and promote to FP32 for the remaining ops
    # ------------------------------------------------------------------
    # out_perm: [B, H, N, N]  →  [B, N, N, H]
    out_pre_fp16 = out_perm.permute(0, 2, 3, 1).contiguous()
    out_pre = out_pre_fp16.to(torch.float32)   # now FP32

    # ------------------------------------------------------------------
    # LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = get("to_out_norm.weight")
    to_out_norm_b = get("to_out_norm.bias")
    out_norm = torch.nn.functional.layer_norm(
        out_pre,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )  # shape [B,N,N,hidden]

    # Apply the output gate (still FP32)
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    to_out_w = get("to_out.weight")
    output = torch.nn.functional.linear(out_norm, to_out_w)  # [B,N,N,dim]

    return output
