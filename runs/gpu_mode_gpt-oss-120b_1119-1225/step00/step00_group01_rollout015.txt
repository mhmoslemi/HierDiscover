"""
TriMul (outgoing) kernel for AlphaFold3 style models.

The heavy part of the forward is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is an O(N³·D) operation (N = seq_len, D = hidden_dim).
This implementation fuses the inner‐sum using a custom Triton kernel
that performs a blocked GEMM for each hidden channel block.
All other parts (LayerNorms, Linear projections, gating) are executed
with native PyTorch ops.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# -------------------------------------------------------------------------
# Triton kernel: block‐wise matrix multiplication for each hidden channel block
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,       # const float*  (B,N,N,D) left projection
    right_ptr,      # const float*  (B,N,N,D) right projection
    out_ptr,        #       float*  (B,N,N,D) output accumulator
    B, N, D,        # runtime dimensions (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Each program instance computes a tile of shape (BLOCK_I, BLOCK_J, BLOCK_D)
    of the output tensor. The computation is
        out[i, j, d] += Σ_k left[i, k, d] * right[j, k, d]
    where `left` has layout (i, k, d) and `right` is accessed as (k, j, d)
    to enable a fast GEMM via tl.dot.
    """
    pid_b = tl.program_id(2)          # batch index
    pid_i = tl.program_id(0)          # tile over i (first pair axis)
    pid_j = tl.program_id(1)          # tile over j (second pair axis)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # -------------------------------------------------
    # index vectors and boundary masks
    # -------------------------------------------------
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    i_valid = offs_i < N
    j_valid = offs_j < N

    # Number of hidden‑dim blocks
    num_d_blocks = tl.cdiv(D, BLOCK_D)

    # -------------------------------------------------
    # iterate over hidden‑dim blocks
    # -------------------------------------------------
    for db in range(0, num_d_blocks):
        d_base = db * BLOCK_D
        # actual D‑size for the current block (may be smaller on the last block)
        offs_d = d_base + tl.arange(0, BLOCK_D)
        d_valid = offs_d < D

        # accumulator in FP32
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # -------------------------------------------------
        # iterate over K tiles
        # -------------------------------------------------
        for k_base in range(0, N, BLOCK_K):
            cur_k = tl.minimum(BLOCK_K, N - k_base)
            offs_k = k_base + tl.arange(0, BLOCK_K)
            k_valid = offs_k < N

            # -----------------------------------------------------------------
            # Load left tile: shape (BLOCK_I, cur_k, BLOCK_D)
            # Layout: (i, k, d)
            # -----------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_d[None, None, :] * stride_left_d
            )
            left = tl.load(
                left_ptrs,
                mask=i_valid[:, None, None] & k_valid[None, :, None] & d_valid[None, None, :],
                other=0.0,
            ).to(tl.float32)

            # -----------------------------------------------------------------
            # Load right tile: we load it as (k, j, d) to be ready for tl.dot
            # -----------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_j[None, :, None] * stride_right_j   # j dimension is second
                + offs_k[:, None, None] * stride_right_k   # k dimension is first
                + offs_d[None, None, :] * stride_right_d
            )
            right = tl.load(
                right_ptrs,
                mask=j_valid[None, :, None] & k_valid[:, None, None] & d_valid[None, None, :],
                other=0.0,
            ).to(tl.float32)   # shape (cur_k, BLOCK_J, BLOCK_D)

            # -----------------------------------------------------------------
            # For each hidden channel inside the block compute a GEMM:
            #   (BLOCK_I, cur_k) @ (cur_k, BLOCK_J)  -> (BLOCK_I, BLOCK_J)
            # -----------------------------------------------------------------
            for dbi in range(0, BLOCK_D):
                # guard out‑of‑range d channels
                if dbi + d_base >= D:
                    break
                # left slice (I x K)
                left_slice = left[:, :, dbi]                # (BLOCK_I, cur_k)
                # right slice (K x J) – already stored as (K, J)
                right_slice = right[:, :, dbi]              # (cur_k, BLOCK_J)
                # accumulate GEMM result
                acc[:, :, dbi] += tl.dot(left_slice, right_slice)

        # -----------------------------------------------------------------
        # Store the accumulated tile to the output tensor
        # -----------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_d[None, None, :] * stride_out_d
        )
        tl.store(
            out_ptrs,
            acc,
            mask=i_valid[:, None, None] & j_valid[None, :, None] & d_valid[None, None, :],
        )


# -------------------------------------------------------------------------
# Python entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Implements the outgoing TriMul operator.
    Input:
        data = (input_tensor, mask, weights, config)
    Output:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32 on the same device).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # unpack config & weight tensors
    # -----------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # LayerNorm for input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # Linear weight tensors (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # Output projection & its LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # Normalization of the input (LayerNorm over the channel dimension)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Linear projections (no bias) and gating
    # -----------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # Optional mask (broadcast over hidden dimension)
    if mask is not None and mask.numel() > 0:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Contract left and right via custom Triton kernel
    # -----------------------------------------------------------------
    out = torch.empty(
        (bs, N, N, hidden_dim), device=device, dtype=torch.float32
    )  # accumulator in fp32

    # Strides (in element units) for the three tensors
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Kernel launch configuration
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 32

    grid = (
        triton.cdiv(N, BLOCK_I),
        triton.cdiv(N, BLOCK_J),
        bs,
    )

    tri_mul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=bs,
        N=N,
        D=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_d=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_j=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_d=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_d=stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # -----------------------------------------------------------------
    # Post‑processing: LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out
