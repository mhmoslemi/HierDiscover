def tri_mul_batch_matmul_kernel(a_ptr, b_ptr, c_ptr,
                               batch, M, N, K,
                               stride_a_batch, stride_a_i, stride_a_k,
                               stride_b_batch, stride_b_i, stride_b_j,
                               stride_c_batch, stride_c_i, stride_c_j,
                               BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets inside block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k loops
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k range
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A: shape BLOCK_M x BLOCK_K
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_T: we need B's element (j, k) leading to B_T(k, j)
        # We'll load B block of shape BLOCK_K x BLOCK_N (k dimension by j).
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_j + offs_n[None, :] * stride_b_i
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product
        acc += tl.dot(a, b)  # default: dot axis=K dims (BLOCK_K)
        # Next block
        # (...) update pointers not needed because we compute fresh each iteration

    # Store result
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j
    # Write only valid positions
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul "outgoing" forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm `x` over the last dimension (`dim`).
2. Linear projections:
   left  = x @ W_left   (shape: [B,N,N,H])
   right = x @ W_right  (shape: [B,N,N,H])
3. Optional binary mask (broadcast over the hidden dim) zeroes out
   masked pairwise features.
4. Compute three sigmoid gates from `x`:
   left_gate  = sigmoid(x @ W_left_gate)
   right_gate = sigmoid(x @ W_right_gate)
   out_gate   = sigmoid(x @ W_out_gate)
   Apply left/right gates element‑wise.
5. Fuse the expensive contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix multiplication
      out_b,d = left_b,d @ right_b,dᵀ
   for each (batch, hidden) pair.
   We flatten `(B,H)` into a single batch dimension and launch a
   Triton kernel that computes the batched matmul in FP16 (accumulating
   in FP32 for numerical stability).
6. Reshape the result back to `[B,N,N,H]`, apply a second LayerNorm
   over `H`, multiply by `out_gate` and a final linear projection
   `to_out` (shape `[dim,H]`) to obtain the output `[B,N,N,dim]`.

The Triton kernel operates on blocks of size (BLOCK_M×BLOCK_N) with an
inner reduction over BLOCK_K.  It loads tiles of `A` (left) and of the
transpose of `B` (right) on‑the‑fly, performs `tl.dot`, accumulates in
FP32 and stores the result in FP16.  The rest of the operations (layer‑norm,
linear, sigmoid) are performed with PyTorch which is already highly
optimized.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched matmul  C = A @ Bᵀ
#   A : (batch, M, K)   (float16)
#   B : (batch, N, K)   (float16)   (will be read as Bᵀ inside the kernel)
#   C : (batch, M, N)   (float16)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aa, stride_ai, stride_ak,           # A strides: batch, row, col
    stride_bb, stride_bi, stride_bj,           # B strides: batch, row, col (original B)
    stride_cc, stride_ci, stride_cj,           # C strides: batch, row, col
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch index
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block column index

    # ------------------ block offsets ------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in BLOCK_K sized chunks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load tile of A ------------------------------------------------
        # A[pid_batch, offs_m, offs_k]
        a_ptrs = (a_ptr + pid_batch * stride_aa
                  + offs_m[:, None] * stride_ai
                  + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # ---- load tile of Bᵀ (i.e. B transposed on‑the‑fly) ------------------
        # We need B_T[k, n] = B[n, k]
        # B original layout: row stride = stride_bi, col stride = stride_bj
        b_ptrs = (b_ptr + pid_batch * stride_bb
                  + offs_k[:, None] * stride_bj   # move along original column (k) -> row of Bᵀ
                  + offs_n[None, :] * stride_bi)  # move along original row (n) -> column of Bᵀ
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ---- matrix multiply‑accumulate ------------------------------------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ------------------ write back ------------------
    c_ptrs = (c_ptr + pid_batch * stride_cc
              + offs_m[:, None] * stride_ci
              + offs_n[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated forward of TriMul (outgoing version)
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32) – TriMul forward output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]      # corresponds to `c_z` in AlphaFold‑3 notation

    # ------------------------------------------------------------------
    # Load model weights (all tensors are assumed to be on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    to_out_norm_weight = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias   = weights["to_out_norm.bias"]     # (hidden_dim,)

    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask if present (mask shape: [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is floating point and broadcastable
        mask_broadcast = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()   # [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication: out = left @ rightᵀ  (per hidden dim)
    #     Reshape to (B*hidden, N, N) → treat hidden as additional batch dim.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden dim next to batch for efficient flattening
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, hidden, N, N]

    batch_total = B * hidden_dim
    left_mat  = left_perm.view(batch_total, N, N)   # [B*H, N, N]
    right_mat = right_perm.view(batch_total, N, N)  # [B*H, N, N]

    # Cast to FP16 for the kernel (accumulate in FP32)
    left_fp16  = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)
    out_fp16   = torch.empty_like(left_fp16, dtype=torch.float16)

    # Strides for A (left), B (right) and C (output)
    stride_aa, stride_ai, stride_ak = left_fp16.stride()   # (batch, row, col)
    stride_bb, stride_bi, stride_bj = right_fp16.stride()
    stride_cc, stride_ci, stride_cj = out_fp16.stride()

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # Empirically good for H100 FP16 matmul

    grid = (
        batch_total,                                   # axis 0 : batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,                  # axis 1 : block rows
        (N + BLOCK_N - 1) // BLOCK_N,                  # axis 2 : block columns
    )

    tri_mul_batch_matmul_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_fp16,
        batch=batch_total,
        M=N,
        N=N,
        K=N,
        stride_aa=stride_aa,
        stride_ai=stride_ai,
        stride_ak=stride_ak,
        stride_bb=stride_bb,
        stride_bi=stride_bi,
        stride_bj=stride_bj,
        stride_cc=stride_cc,
        stride_ci=stride_ci,
        stride_cj=stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Restore FP32, reshape back to [B,N,N,hidden]
    # ------------------------------------------------------------------
    out = (
        out_fp16.to(torch.float32)
        .view(B, hidden_dim, N, N)                # [B, hidden, N, N]
        .permute(0, 2, 3, 1)                       # [B, N, N, hidden]
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden dimension, gated output, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias
    )
    out = out * out_gate                # element‑wise gating
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
