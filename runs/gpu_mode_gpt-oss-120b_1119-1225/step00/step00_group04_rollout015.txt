def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Offsets for layernorm, linear etc.

    # Compute normalized input
    # We have weight and bias tensors for norm: shape dim.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5 # default for LayerNorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Compute projections
    left_proj_weight = weights["left_proj.weight"] # shape hidden_dim, dim
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape [bs, seq_len, seq_len, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Compute gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()  # [bs, seq_len, seq_len, hidden_dim]

    # Apply mask if present and not nomask i.e., config.get("nomask", False) False
    if not config.get("nomask", False):
        # mask shape [bs, seq_len, seq_len]; unsqueeze
        mask_unsqueezed = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_unsqueezed
        right = right * mask_unsqueezed

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # For kernel, we could cast left and right to fp16 for speed.
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output buffer in float32
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # Kernel launch
    # Choose block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute grid dimensions:
    grid_M = (seq_len + BLOCK_M - 1) // BLOCK_M
    grid_N = (seq_len + BLOCK_N - 1) // BLOCK_N
    grid_B = bs * hidden_dim
    # Using triton.launch with grid = (grid_M, grid_N, grid_B)
    # Pass stride arguments
    a_stride_b0, a_stride_b1, a_stride_b2, a_stride_b3 = left_fp16.stride()
    b_stride_b0, b_stride_b1, b_stride_b2, b_stride_b3 = right_fp16.stride()
    c_stride_b0, c_stride_b1, c_stride_b2, c_stride_b3 = out.stride()

    trimul_kernel[grid_M, grid_N, grid_B](
        left_fp16, right_fp16, out,
        bs, seq_len, hidden_dim,
        a_stride_b0, a_stride_b1, a_stride_b2, a_stride_b3,
        b_stride_b0, b_stride_b1, b_stride_b2, b_stride_b3,
        c_stride_b0, c_stride_b1, c_stride_b2, c_stride_b3,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Now out has shape [bs, seq_len, seq_len, hidden_dim] float32.

    # Apply to_out_norm (layernorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear: to_out_weight shape dim, hidden_dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [bs, seq_len, seq_len, dim]

    return out

@triton.jit
def trimul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,  # ints
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Batched matmul for TriMul: C[b,h,i,j] = sum_k A[b,h,i,k] * B[b,h,j,k]  (i.e., A @ B^T)
    where A and B are the left and right projected tensors after gating/masking.
    Computes in FP16 for A,B and accumulates in FP32.
    """

    pid_batch = tl.program_id(axis=2)   # range 0 .. B*H-1
    b = pid_batch // H
    h = pid_batch % H

    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # Compute start offsets for m and n
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Create masks for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_blocks):
        # compute k offsets
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + \
            b * stride_a_batch + \
            h * stride_a_h + \
            offs_m[:, None] * stride_a_i + \
            offs_k[None, :] * stride_a_k

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N) where first dim = k, second = j
        # Original B indices: (b, j, k, h)
        # We want B_T[k, j] = B[b, j, k, h]
        b_ptrs = b_ptr + \
            b * stride_b_batch + \
            h * stride_b_h + \
            offs_k[:, None] * stride_b_k + \
            offs_n[None, :] * stride_b_i  # using stride_i for j dimension

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # Compute dot product accumulation
        # a: (M, K), b: (K, N)
        acc += tl.dot(a, b)

    # Write back to C
    c_ptrs = c_ptr + \
        b * stride_c_batch + \
        h * stride_c_h + \
        offs_m[:, None] * stride_c_i + \
        offs_n[None, :] * stride_c_j

    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

Note: In strides we passed: stride_c_i = stride for i dimension (the first N), stride_c_j = stride for j dimension (the second N), stride_c_h = stride for hidden_dim.

B shape (b,j,k,h). So stride_b_batch = right.stride(0); stride_b_i (j dimension) = right.stride(1). stride_b_k = right.stride(2). stride_b_h = right.stride(3). """
TriMul (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the 4‑D input (last dim = dim).
2. Linear projections → left_proj, right_proj   (dim → hidden_dim).
3. Compute three sigmoidal gates (left_gate, right_gate, out_gate)
   from the same normalized input.
4. Apply optional mask, then multiply each projection by its gate.
5. **Core contraction**: for every (batch, hidden) pair compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication L @ Rᵀ.
   This is performed by the `trimul_kernel` in FP16 (inputs) / FP32 (accum).
6. Layer‑norm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to `dim`.

Only the heavy O(N³·hidden·B) contraction is executed on the GPU
with Triton; all other ops use fast torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    B, N, H,                           # batch, seq_len, hidden_dim
    stride_a_batch, stride_a_i, stride_a_k, stride_a_h,
    stride_b_batch, stride_b_i, stride_b_k, stride_b_h,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matmul for TriMul:
        C[b,h,i,j] = Σ_k A[b,h,i,k] * B[b,h,j,k]   (A @ Bᵀ)

    A (left) : shape (B, N, N, H) → (b,i,k,h)
    B (right): shape (B, N, N, H) → (b,j,k,h)    (used transposed as k‑j)
    C (out)  : shape (B, N, N, H) → (b,i,j,h)
    All tensors are contiguous in memory (CUDA).
    """
    # ------------------------------------------------------------------
    # 1) Identify the (batch, hidden) pair processed by this program.
    # ------------------------------------------------------------------
    pid_batch = tl.program_id(axis=2)                # 0 … B*H-1
    b = pid_batch // H                               # batch index
    h = pid_batch % H                                # hidden index

    # ------------------------------------------------------------------
    # 2) 2‑D grid gives a tile of the output matrix (i‑block, j‑block).
    # ------------------------------------------------------------------
    pid_m = tl.program_id(axis=0)                    # block over i (rows)
    pid_n = tl.program_id(axis=1)                    # block over j (cols)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M) # i offsets
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N) # j offsets

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # 3) Accumulator (FP32) for the tile.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) Loop over the reduction dimension k.
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_blocks):
        # k‑axis offsets for this block
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- Load a tile of A (left) : (M, K) ----
        a_ptrs = (
            a_ptr
            + b * stride_a_batch
            + h * stride_a_h
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ---- Load a tile of B transposed (right) : (K, N) ----
        # Original layout of B is (b, j, k, h).  We need Bᵀ[k, j].
        b_ptrs = (
            b_ptr
            + b * stride_b_batch
            + h * stride_b_h
            + offs_k[:, None] * stride_b_k     # k dimension first
            + offs_n[None, :] * stride_b_i     # j dimension as column
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float16)

        # ---- Accumulate the dot product ----
        acc += tl.dot(a, b)   # a:(M,K), b:(K,N) → (M,N)

    # ------------------------------------------------------------------
    # 5) Store the resulting tile to C (out)
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + b * stride_c_batch
        + h * stride_c_h
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape (bs, seq_len, seq_len, dim)  (float32)
    """
    # ------------------------------------------------------------------
    # 0) Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcasted over the hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask: [B, N, N] → [B, N, N, 1] (same dtype as activations)
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cast to FP16 for the heavy contraction (reduces memory traffic)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7) Allocate output buffer (FP32 accumulator)
    # ------------------------------------------------------------------
    out = torch.empty_like(left_fp16, dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 8) Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (seq_len + BLOCK_M - 1) // BLOCK_M
    grid_n = (seq_len + BLOCK_N - 1) // BLOCK_N
    grid_b = bs * hidden_dim                      # one program per (batch, hidden)

    # Strides for the three tensors (all are contiguous in C order)
    a_s0, a_s1, a_s2, a_s3 = left_fp16.stride()
    b_s0, b_s1, b_s2, b_s3 = right_fp16.stride()
    c_s0, c_s1, c_s2, c_s3 = out.stride()

    # ------------------------------------------------------------------
    # 9) Launch the kernel
    # ------------------------------------------------------------------
    trimul_kernel[(grid_m, grid_n, grid_b)](
        left_fp16,
        right_fp16,
        out,
        bs,
        seq_len,
        hidden_dim,
        a_s0,
        a_s1,
        a_s2,
        a_s3,
        b_s0,
        b_s1,
        b_s2,
        b_s3,
        c_s0,
        c_s1,
        c_s2,
        c_s3,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10) Post‑processing: norm → out‑gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
