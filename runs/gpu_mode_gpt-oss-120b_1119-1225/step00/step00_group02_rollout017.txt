"""
TriMul (outgoing) forward kernel.

The heavy part of the TriMul module is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of independent matrix‑multiplications
(left[:,:, :,d] @ right[:,:, :,d].T) for each hidden dimension d.
We implement this contraction with a Triton kernel that operates on one
hidden channel (d) at a time, i.e. a batched GEMM with batch size B*D.
All other operations (LayerNorms, linear projections, gating) are kept
in PyTorch for simplicity.

The kernel works on fp16 inputs (left/right) and accumulates in fp32,
using Tensor‑Core dot products.  The result is stored as fp32.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, D,
    # strides for left  [batch, i, k, d]
    stride_al, stride_ai, stride_ak, stride_ad,
    # strides for right [batch, j, k, d] (j = first N, k = second N)
    stride_br, stride_bj, stride_bk, stride_bd,
    # strides for out   [batch, i, j, d]
    stride_co, stride_ci, stride_cj, stride_cd,
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: for each (batch, hidden) compute
       out[i,j] = Σ_k left[i,k] * right[j,k]   (right is used transposed)."""
    pid_m = tl.program_id(0)          # i‑tile
    pid_n = tl.program_id(1)          # j‑tile
    pid_bd = tl.program_id(2)         # combined batch * hidden index

    # decode batch & hidden‑channel index
    d = pid_bd % D
    b = pid_bd // D

    # tile offsets
    i_base = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # predicates for out‑of‑bounds i/j
    mask_i = i_base < N
    mask_j = j_base < N

    # offsets that are constant inside the k‑loop
    batch_off_left  = b * stride_al
    batch_off_right = b * stride_br
    batch_off_out   = b * stride_co
    d_off_left  = d * stride_ad
    d_off_right = d * stride_bd
    d_off_out   = d * stride_cd

    # accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k‑loop
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # A tile: shape (BLOCK_M, BLOCK_K) – left[i,k,d]
        a_ptr = left_ptr \
                + batch_off_left + d_off_left \
                + i_base[:, None] * stride_ai \
                + k_base[None, :] * stride_ak
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)  # fp16

        # B tile: shape (BLOCK_K, BLOCK_N) – right[j,k,d] (we need k as rows)
        b_ptr = right_ptr \
                + batch_off_right + d_off_right \
                + k_base[:, None] * stride_bk \
                + j_base[None, :] * stride_bj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)  # fp16

        # Matrix multiply for this k‑tile (fp16 → fp32 accumulation)
        acc += tl.dot(a, b).to(tl.float32)

    # write results
    c_ptr = out_ptr \
            + batch_off_out + d_off_out \
            + i_base[:, None] * stride_ci \
            + j_base[None, :] * stride_cj
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[bs, N, N, dim] (float32)
        - mask         : torch.Tensor[bs, N, N]      (float32 or bool) or None
        - weights      : dict of model parameters
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    out : torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Helper to fetch weights (they are stored as raw tensors)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (dim → hidden_dim) – no bias
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [bs,N,N,hd]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [bs,N,N,hd]

    # ------------------------------------------------------------------
    # 3) Optional pairwise mask
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.unsqueeze(-1).to(x.dtype)               # [bs,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid after a separate linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left  = left * left_gate   # [bs,N,N,hd]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to fp16 for the heavy contraction
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # output buffer for the contraction (fp32)
    out_fp32 = torch.empty((bs, N, N, hidden_dim),
                           dtype=torch.float32,
                           device=input_tensor.device)

    # ------------------------------------------------------------------
    # 6) Triton kernel launch – batched GEMM over hidden_dim
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # strides (in elements) – PyTorch gives them already in elements
    stride_al = left_fp16.stride(0)   # batch
    stride_ai = left_fp16.stride(1)   # i
    stride_ak = left_fp16.stride(2)   # k
    stride_ad = left_fp16.stride(3)   # d

    stride_br = right_fp16.stride(0)  # batch
    stride_bj = right_fp16.stride(1)  # j (first N)
    stride_bk = right_fp16.stride(2)  # k (second N)
    stride_bd = right_fp16.stride(3)  # d

    stride_co = out_fp32.stride(0)    # batch
    stride_ci = out_fp32.stride(1)    # i
    stride_cj = out_fp32.stride(2)    # j
    stride_cd = out_fp32.stride(3)    # d

    # grid: (tiles over i, tiles over j, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )

    tri_mul_kernel[grid](
        left_fp16, right_fp16, out_fp32,
        bs, N, hidden_dim,
        stride_al, stride_ai, stride_ak, stride_ad,
        stride_br, stride_bj, stride_bk, stride_bd,
        stride_co, stride_ci, stride_cj, stride_cd,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7) Post‑processing: to_out_norm, out_gate, final linear
    # ------------------------------------------------------------------
    # LayerNorm on the collapsed hidden dimension
    out = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # [bs,N,N,hd]

    # Apply the output gate (broadcasted)
    out = out * out_gate

    # Final linear projection back to dim (no bias)
    out = torch.nn.functional.linear(out, to_out_weight)  # [bs,N,N,dim]

    return out
