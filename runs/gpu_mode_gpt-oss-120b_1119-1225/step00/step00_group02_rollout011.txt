    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # Linear projections
    left_pre = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_pre = torch.nn.functional.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))

    left_pre = left_pre.contiguous()
    right_pre = right_pre.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    else:

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # shape [B*H, N, N]

    # Reshape back to [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm (LayerNorm over hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

"""
TriMul forward (outgoing) implementation.

Algorithm
---------

1. **Layer‑norm** the input tensor `x` (shape [B,N,N,C]).
2. **Project** `x` into a hidden space (size `hidden_dim`) with two linear layers
   → `left_pre`, `right_pre`.
3. Compute three **sigmoid gates** from `x`:
   `left_gate`, `right_gate`, `out_gate`.
4. Fuse the *mask* (if present) and the two projection gates with a tiny
   Triton kernel (`mask_gate_kernel`).   The kernel processes one
   `(batch, i, j)` coordinate at a time and a vector of `BLOCK_H` hidden
   channels, performing
   ``out = proj * gate * mask``.
   The result is stored in `left` and `right`.
5. The heavy ``einsum('... i k d, ... j k d -> ... i j d')`` reduces to a
   batched matrix‑multiply per hidden channel:
   for every hidden dimension `d`,
   ``out[...,d] = left[...,d] @ right[...,d].T``.
   This is carried out with a single call to `torch.bmm` on the
   reshaped tensors (leveraging cuBLAS/Tensor‑cores on H100).
6. **Layer‑norm** the intermediate result over the hidden dimension,
   multiply by `out_gate`, and apply the final linear projection back to
   the original `dim`.
7. Return the tensor of shape `[B, N, N, dim]`.

Only the mask‑gate fusion is written in Triton – the rest uses
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(
    inp_ptr, gate_ptr, mask_ptr, out_ptr,
    stride_inp_b, stride_inp_i, stride_inp_j, stride_inp_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Fuse mask, projection and gate (per‑channel multiplication).

    Each program instance works on a single (b,i,j) coordinate and processes
    up to ``BLOCK_H`` hidden channels.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # Decode (b,i,j) from the linear program id.
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    offs_h = tl.arange(0, BLOCK_H)               # hidden indices inside the block
    mask_val = tl.load(mask_ptr + b * stride_mask_b
                                 + i * stride_mask_i
                                 + j * stride_mask_j)  # scalar broadcast

    # Load vectors of length BLOCK_H (masked for out‑of‑range channels)
    inp = tl.load(
        inp_ptr + b * stride_inp_b
                 + i * stride_inp_i
                 + j * stride_inp_j
                 + offs_h * stride_inp_h,
        mask=offs_h < H, other=0.0
    )
    gate = tl.load(
        gate_ptr + b * stride_gate_b
                  + i * stride_gate_i
                  + j * stride_gate_j
                  + offs_h * stride_gate_h,
        mask=offs_h < H, other=0.0
    )

    # Element‑wise fused operation
    out = inp * gate * mask_val

    tl.store(
        out_ptr + b * stride_out_b
                 + i * stride_out_i
                 + j * stride_out_j
                 + offs_h * stride_out_h,
        out,
        mask=offs_h < H
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, C] (C == dim)
            - mask : torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict containing the model parameters
            - config : dict with keys "dim", "hidden_dim" (and optional "nomask")

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same device as input)
    """
    # ------------------------------------------------------------------ #
    # 1️⃣ unpack arguments & convenience aliases
    # ------------------------------------------------------------------ #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H
    B, N, _, C = input_tensor.shape           # C == dim

    # ------------------------------------------------------------------ #
    # 2️⃣ fetch weight tensors
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]          # shape (dim, hidden_dim)

    # ------------------------------------------------------------------ #
    # 3️⃣ Input Layer‑norm
    # ------------------------------------------------------------------ #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------ #
    # 4️⃣ Project → hidden, compute gates (sigmoid)
    # ------------------------------------------------------------------ #
    left_pre = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_pre = torch.nn.functional.linear(x, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Ensure contiguity for safe pointer arithmetic
    left_pre = left_pre.contiguous()
    right_pre = right_pre.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # ------------------------------------------------------------------ #
    # 5️⃣ Mask handling (optional)
    # ------------------------------------------------------------------ #
    if mask_tensor is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=dtype, device=device)

    mask = mask.contiguous()

    # ------------------------------------------------------------------ #
    # 6️⃣ Fuse mask + projection gate using Triton
    # ------------------------------------------------------------------ #
    left = torch.empty_like(left_pre)      # placeholder for gated left
    right = torch.empty_like(right_pre)    # placeholder for gated right

    BLOCK_H = 32                                     # compile‑time tile over hidden dim
    grid = (B * N * N,)                              # one program per (b,i,j)

    # left side
    mask_gate_kernel[grid](
        left_pre,
        left_gate,
        mask,
        left,
        left_pre.stride(0), left_pre.stride(1), left_pre.stride(2), left_pre.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        B=B, N=N, H=hidden_dim, BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # right side
    mask_gate_kernel[grid](
        right_pre,
        right_gate,
        mask,
        right,
        right_pre.stride(0), right_pre.stride(1), right_pre.stride(2), right_pre.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        B=B, N=N, H=hidden_dim, BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------ #
    # 7️⃣ Core TriMul: batched GEMM across hidden dimension
    # ------------------------------------------------------------------ #
    # Re‑order to [B, hidden, N, N] for easy batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Fuse batch and hidden into a single batch dimension
    left_mat = left_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_mat = right_perm.view(B * hidden_dim, N, N)  # [B*H, N, N]

    # Matrix multiply: left @ right.T   (each slice → [N,N])
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 8️⃣ Output Layer‑norm, gating, final projection
    # ------------------------------------------------------------------ #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # → [B,N,N,dim]

    return out
