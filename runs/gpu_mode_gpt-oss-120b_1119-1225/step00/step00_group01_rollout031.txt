```
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_d,
    stride_right_bs, stride_right_j, stride_right_k, stride_right_d,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    # program ids
    pid0 = tl.program_id(0) # index across batch*tiles_i*tiles_j
    pid1 = tl.program_id(1) # hidden tile index
    # compute batch and tile indices
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    # total per batch: num_i * num_j
    # compute b index
    batch_id = pid0 // (num_i * num_j)
    rem = pid0 % (num_i * num_j)
    tile_i = rem // num_j
    tile_j = rem % num_j

    # Hidden tile start
    d_offset = pid1 * BLOCK_D + tl.arange(0, BLOCK_D)

    # Offsets
    i_offset = tile_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = tile_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask ranges
    mask_i = i_offset < N
    mask_j = j_offset < N
    mask_d = d_offset < D

    # base pointers for left and right
    # left: shape [B,N,N,D]; access left[batch_id, i_offset, k_offset, d_offset]
    # right: shape [B,N,N,D]; access right[batch_id, j_offset, k_offset, d_offset]

    # output pointer: out[batch_id, i_offset, j_offset, d_offset]
    out_ptrs = out_ptr + batch_id * stride_out_bs + i_offset[:, None, None] * stride_out_i + j_offset[None, :, None] * stride_out_j + d_offset[None, None, :] * stride_out_d

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over k
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + batch_id * stride_left_bs + i_offset[:, None, None] * stride_left_i + k_offset[None, :, None] * stride_left_k + d_offset[None, None, :] * stride_left_d
```

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_bs, stride_left_i, stride_left_k, stride_left_d,
    stride_right_bs, stride_right_j, stride_right_k, stride_right_d,
    stride_out_bs, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    # Compute i,j offsets as described
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    batch_idx = pid0 // (num_i * num_j)
    tmp = pid0 % (num_i * num_j)
    tile_i = tmp // num_j
    tile_j = tmp % num_j

    # offsets
    offs_i = tile_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = tile_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_d = pid1 * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # base pointers for left and right
    # left[b,i,k,d]
    left_base = left_ptr + batch_idx * stride_left_bs
    right_base = right_ptr + batch_idx * stride_right_bs
    out_base = out_ptr + batch_idx * stride_out_bs

    # compute output pointer
    out_ptrs = out_base + offs_i[:, None, None] * stride_out_i + offs_j[None, :, None] * stride_out_j + offs_d[None, None, :] * stride_out_d

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left loads: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_base + offs_i[:, None, None] * stride_left_i + offs_k[None, :, None] * stride_left_k + offs_d[None, None, :] * stride_left_d
        left_val = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # right loads: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_base + offs_j[:, None, None] * stride_right_j + offs_k[None, :, None] * stride_right_k + offs_d[None, None, :] * stride_right_d
        right_val = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

        # Since left_val shape (BLOCK_I, BLOCK_K, BLOCK_D) and right_val shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # we need to sum over k dimension: for each kk, compute outer product across i and j for each d
        # Use inner loop across kk:
        for kk in range(BLOCK_K):
            # ensure that this kk is valid (mask_k[kk])
            if not mask_k[kk]:
                continue
            # left slice shape (BLOCK_I, BLOCK_D)
            l = left_val[:, kk, :]  # (BLOCK_I, BLOCK_D)
            # right slice shape (BLOCK_J, BLOCK_D)
            r = right_val[:, kk, :]  # (BLOCK_J, BLOCK_D)
            # broadcast: l[:, None, :] * r[None, :, :] => (BLOCK_I, BLOCK_J, BLOCK_D)
            acc += l[:, None, :] * r[None, :, :]
    # store result (convert to fp16)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    # extract
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    # layernorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5

    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )
    # linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape [bs,N,N]; unsqueeze to [bs,N,N,1]
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # gating
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # cast to half for reduction
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # compute out using Triton kernel
    out_half = tri_mul_outgoing_triton(left_half, right_half, hidden_dim)
    out = out_half.to(torch.float32)

    # to_out_norm
    to_out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # multiply by out_gate
    out = to_out_norm * out_gate

    # final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

    B, N, _, D = left.shape
    assert D == hidden_dim
    # ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # compute strides

    stride_right = right.stride()
    stride_out = out.stride()

    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32

    num_j = math.ceil(N / BLOCK_J)

        left, right, out,
        B, N, D,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )
    return out

import torch
import torch.nn.functional as F
import math
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid0 = tl.program_id(0)  # covers batch * i_tile * j_tile
    pid1 = tl.program_id(1)  # covers hidden tiles

    # Compute tile indices
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    batch_idx = pid0 // (num_i * num_j)
    rem = pid0 % (num_i * num_j)
    tile_i = rem // num_j
    tile_j = rem % num_j

    # Offsets within this tile
    offs_i = tile_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = tile_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_d = pid1 * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # Base pointers for this batch
    left_batch = left_ptr + batch_idx * stride_left_b
    right_batch = right_ptr + batch_idx * stride_right_b
    out_batch = out_ptr + batch_idx * stride_out_b

    # Pointer to output element
    out_ptrs = out_batch + offs_i[:, None, None] * stride_out_i \
                         + offs_j[None, :, None] * stride_out_j \
                         + offs_d[None, None, :] * stride_out_d

    # Accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left slice: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_batch + offs_i[:, None, None] * stride_left_i \
                                 + offs_k[None, :, None] * stride_left_k \
                                 + offs_d[None, None, :] * stride_left_d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0).to(tl.float32)

        # Load right slice: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_batch + offs_j[:, None, None] * stride_right_j \
                                   + offs_k[None, :, None] * stride_right_k \
                                   + offs_d[None, None, :] * stride_right_d
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                        other=0.0).to(tl.float32)

        # Compute contribution for each k within this tile
        for kk in range(BLOCK_K):
            # l: (BLOCK_I, BLOCK_D)
            l = left[:, kk, :]   # (i, d)
            # r: (BLOCK_J, BLOCK_D)
            r = right[:, kk, :]  # (j, d)
            # Broadcast and accumulate: (i, j, d)
            acc += l[:, None, :] * r[None, :, :]

    # Store the result
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

def tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Compute the outgoing TriMul contraction using a Triton kernel.
    left, right: tensors of shape [B, N, N, D] with D == hidden_dim.
    Returns a tensor of the same shape and dtype (float32).
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, D = left.shape
    assert D == hidden_dim

    # Cast to half for memory bandwidth reduction (kernel accumulates in fp32)
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    out = torch.empty_like(left_half, dtype=torch.float32)  # final result in fp32

    # Strides in element units
    stride_left = left_half.stride()
    stride_right = right_half.stride()
    stride_out = out.stride()

    # Block sizes (tuned for H100; can be adjusted)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32

    # Grid dimensions
    num_i = math.ceil(N / BLOCK_I)
    num_j = math.ceil(N / BLOCK_J)
    grid0 = B * num_i * num_j
    grid1 = math.ceil(hidden_dim / BLOCK_D)

    tri_mul_outgoing_kernel[grid0, grid1](
        left_half, right_half, out,
        B, N, D,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )
    return out

    """
    TriMul (outgoing) forward implementation using Triton for the heavy
    contraction. The steps follow the reference PyTorch implementation:
    1. LayerNorm on input.
    2. Linear projections to hidden_dim (left/right) and separate gates.
    3. Optional mask applied to left/right before gating.
    4. Gate multiplication.
    5. Contraction: out_{i,j,d} = sum_k left_{i,k,d} * right_{j,k,d}
       (implemented with a Triton kernel).
    6. LayerNorm over hidden dimension, out‑gate multiplication, final linear.
    """
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # LayerNorm on the input (affine)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # Linear projections to hidden_dim (no bias)
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask before gating if required
    if not nomask:
        # mask shape [B, N, N]; unsqueeze to broadcast over hidden dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates (sigmoid)
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # Prepare tensors for the Triton kernel (ensure contiguity)
    left = left.contiguous()
    right = right.contiguous()

    # Contraction using Triton
    out = tri_mul_outgoing_triton(left, right, hidden_dim)  # [B,N,N,hidden_dim] fp32

    # Output LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Multiply by out gate (broadcasted over hidden dim)
    out = out * out_gate

    # Final linear mapping back to the original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – fast Triton implementation of the heavy N³ hidden‑dim
contraction.

Algorithm
---------
1. Layer‑norm the input tensor (affine).
2. Linear projections to a hidden dimension (left/right) and three gating
   linear layers (left‑gate, right‑gate, out‑gate).  Gates are passed through a
   sigmoid.
3. If a mask is supplied, apply it to the projected tensors before gating.
4. Fuse gating into the projected tensors.
5. Compute the core contraction
      out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   using a custom Triton kernel.  The kernel tiles the (i,j) plane and the
   hidden channel dimension, loading blocks of the left/right tensors,
   performing a broadcasted outer‑product across the k‑dimension and
   accumulating in fp32.  Input tensors are passed as fp16 to reduce bandwidth.
6. Apply a hidden‑dim layer‑norm, multiply by the out‑gate and project back to
   the original dimension with a final linear layer.

The kernel handles arbitrary batch size, sequence length and hidden dimension,
including tail‑cases via masking.
"""
import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # *fp16  [B,N,N,D]
    right_ptr,         # *fp16  [B,N,N,D]
    out_ptr,           # *fp32  [B,N,N,D]

    B, N, D,                         # compile‑time constants
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Compute out_{i,j,d} = Σₖ left_{i,k,d} * right_{j,k,d}."""
    pid0 = tl.program_id(0)   # (batch * i_tile * j_tile)
    pid1 = tl.program_id(1)   # hidden‑dim tile

    # ------------ tile indices ---------------------------------
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)

    batch_idx = pid0 // (num_i * num_j)
    rem = pid0 % (num_i * num_j)
    tile_i = rem // num_j
    tile_j = rem % num_j

    # ------------ offsets within the tile ----------------------
    offs_i = tile_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = tile_j * BLOCK_J + tl.arange(0, BLOCK_J)
    offs_d = pid1 * BLOCK_D + tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # ------------ base pointers for this batch -----------------
    left_batch  = left_ptr  + batch_idx * stride_left_b
    right_batch = right_ptr + batch_idx * stride_right_b
    out_batch   = out_ptr   + batch_idx * stride_out_b

    # ------------ output pointers -------------------------------
    out_ptrs = out_batch \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + offs_d[None, None, :] * stride_out_d

    # ------------ accumulator (fp32) ---------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------ loop over k ----------------------------------
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_batch \
            + offs_i[:, None, None] * stride_left_i \
            + offs_k[None, :, None] * stride_left_k \
            + offs_d[None, None, :] * stride_left_d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] &
                            mask_k[None, :, None] &
                            mask_d[None, None, :],
                       other=0.0).to(tl.float32)

        # right: (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_batch \
            + offs_j[:, None, None] * stride_right_j \
            + offs_k[None, :, None] * stride_right_k \
            + offs_d[None, None, :] * stride_right_d
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] &
                             mask_k[None, :, None] &
                             mask_d[None, None, :],
                        other=0.0).to(tl.float32)

        # broadcasted outer‑product over the k‑axis
        for kk in range(BLOCK_K):
            # left slice (i,d) and right slice (j,d) for current k
            l = left[:, kk, :]          # (BLOCK_I, BLOCK_D)
            r = right[:, kk, :]         # (BLOCK_J, BLOCK_D)
            # accumulate l_i_d * r_j_d into acc[i,j,d]
            acc += l[:, None, :] * r[None, :, :]

    # ------------ store result ---------------------------------
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None, None] &
                  mask_j[None, :, None] &
                  mask_d[None, None, :])


def tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor, hidden_dim: int) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left / right : [B, N, N, D] (float32)
    Returns out  : [B, N, N, D] (float32)
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, D = left.shape
    assert D == hidden_dim

    # cast to fp16 for bandwidth, kernel accumulates in fp32
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    out = torch.empty_like(left_h, dtype=torch.float32)

    # strides in element units
    s_left = left_h.stride()
    s_right = right_h.stride()
    s_out = out.stride()

    # tile sizes (tuned for H100)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32

    # launch grid
    num_i = math.ceil(N / BLOCK_I)
    num_j = math.ceil(N / BLOCK_J)
    grid0 = B * num_i * num_j          # batch * i‑tiles * j‑tiles
    grid1 = math.ceil(hidden_dim / BLOCK_D)

    tri_mul_outgoing_kernel[grid0, grid1](
        left_h, right_h, out,
        B, N, D,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_right[0], s_right[1], s_right[2], s_right[3],
        s_out[0], s_out[1], s_out[2], s_out[3],
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    The heavy N³ contraction is performed by a Triton kernel,
    all other operations use standard PyTorch functions.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ---------------------------------------------------------
    # 1) Input layer‑norm (affine)
    # ---------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ---------------------------------------------------------
    # 2) Linear projections to hidden dim
    # ---------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ---------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] -> broadcast
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ---------------------------------------------------------
    # 4) Gating (sigmoid)
    # ---------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------
    # 5) Core contraction via Triton
    # ---------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = tri_mul_outgoing_triton(left, right, hidden_dim)   # [B,N,N,hidden_dim] fp32

    # ---------------------------------------------------------
    # 6) Output layer‑norm, out‑gate, final linear
    # ---------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
