    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rk, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Derive batch and hidden index
    batch_idx = pid_z // H
    hidden_idx = pid_z % H

    # Offsets within N dimension
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Bounds mask
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension (reduction)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offset for K block
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        a_ptr = left_ptr + batch_idx * stride_lb + hidden_idx * stride_lh \
                + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(a_ptr, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0)

        # Load b tile from right (transposed): shape (BLOCK_K, BLOCK_N)
        # right_ptr is transposed view (B,K,J,H)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------- LayerNorm -------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias
    )
    
    # ------- Projections -------
    # Linear: output = x @ weight.T
    left_proj = torch.nn.functional.linear(x, left_proj_weight)  # (B,N,N,hidden)
    right_proj = torch.nn.functional.linear(x, right_proj_weight)
    
    # ------- Gates -------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    
    # ------- Mask handling -------
    if not nomask and mask is not None:
        # Convert bool to float and broadcast
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # (B,N,N,1)
    else:
        mask_f = torch.ones_like(x[..., :1])  # shape (B,N,N,1)
    
    # Apply mask and gates
    left = left_proj * mask_f * left_gate
    right = right_proj * mask_f * right_gate
    
    # Ensure contiguity for kernel
    left = left.contiguous()
    right = right.contiguous()
    
    # ------- Triple product (einsum) using Triton -------
    out = trimul_out_triton(left, right)  # shape (B,N,N,hidden_dim)
    
    # ------- Output LayerNorm -------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias
    )
    
    # Multiply by out_gate
    out = out * out_gate  # broadcast over hidden dim
    
    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_weight)
    
    return out

Implements the forward pass of AlphaFold3 TriMul using a fused Triton kernel for the
pairwise reduction. The kernel computes:
out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where left/right are linear and gated projections of the input. The remaining
operations (LayerNorm, linear layers, sigmoid gates) are performed by PyTorch
for simplicity.

    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rk, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs correspond to tile coordinates (i, j) and (batch, hidden) index.
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Decompose the third dimension into batch and hidden-channel.
    batch_idx = pid_z // H
    hidden_idx = pid_z % H

    # Offsets within the N×N matrix.
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns.
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the block result.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k.
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------------------------------
        # Load a tile of left: shape (BLOCK_M, BLOCK_K)
        # left[b, i, k, h]   -> strides: (lb, li, lk, lh)
        # -------------------------------------------------
        a_ptr = left_ptr + batch_idx * stride_lb \
                + hidden_idx * stride_lh \
                + offs_i[:, None] * stride_li \
                + offs_k[None, :] * stride_lk
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # -------------------------------------------------
        # Load a tile of right (already transposed): shape (BLOCK_K, BLOCK_N)
        # right_t[b, k, j, h]   -> strides: (rb, rk, rj, rh)
        # -------------------------------------------------
        b_ptr = right_ptr + batch_idx * stride_rb \
                + hidden_idx * stride_rh \
                + offs_k[:, None] * stride_rk \
                + offs_j[None, :] * stride_rj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Multiply‑accumulate.
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # -------------------------------------------------
    # Store the block result.
    # out[b, i, j, h]   -> strides: (ob, oi, oj, oh)
    # -------------------------------------------------
    c_ptr = out_ptr + batch_idx * stride_ob \
            + hidden_idx * stride_oh \
            + offs_i[:, None] * stride_oi \
            + offs_j[None, :] * stride_oj
    tl.store(c_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def trimul_out_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the 4‑D einsum ``... i k d, ... j k d -> ... i j d`` using a
    custom Triton kernel. ``left`` and ``right`` must have shape
    (B, N, N, H) and be contiguous in memory. The function returns a tensor
    of the same shape and dtype.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    # Allocate output.
    out = torch.empty_like(left)

    # Transpose the second axis of ``right`` so that the reduction becomes a
    # regular matrix multiply: right_t[b, k, j, h] = right[b, j, k, h].
    right_t = right.permute(0, 2, 1, 3).contiguous()

    # Extract strides (element‑wise, i.e. in number of elements, not bytes).
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rk, stride_rj, stride_rh = right_t.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tunable block sizes – these work well on H100 for the problem sizes
    # required by the tests.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H
    )
    trimul_out_kernel[grid](
        left,
        right_t,
        out,
        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rk, stride_rj, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor of shape (B, N, N, dim)
        - mask         : torch.Tensor of shape (B, N, N) (may be ``None``)
        - weights      : dict of pre‑loaded model parameters
        - config       : dict containing ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, dim)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------- Load weights --------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------- LayerNorm on input --------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5
    )

    # -------------------- Linear projections --------------------
    # shape after projection: (B, N, N, hidden_dim)
    left_proj = torch.nn.functional.linear(x, left_proj_weight)   # no bias
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # -------------------- Gates (sigmoid) --------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # -------------------- Mask handling --------------------
    if not nomask and mask is not None:
        # mask is (B, N, N) – expand to broadcast over the hidden dimension.
        mask_exp = mask.to(x.dtype).unsqueeze(-1)   # (B, N, N, 1)
    else:
        # No mask – treat as all‑ones.
        mask_exp = torch.ones_like(x[..., :1])

    # Apply mask and respective gates.
    left = left_proj * mask_exp * left_gate
    right = right_proj * mask_exp * right_gate

    # Ensure continuity for the Triton kernel.
    left = left.contiguous()
    right = right.contiguous()

    # -------------------- Core TriMul contraction (Triton) --------------------
    out = trimul_out_triton(left, right)   # (B, N, N, hidden_dim)

    # -------------------- Output LayerNorm --------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5
    )

    # Multiply by the output gate.
    out = out * out_gate

    # -------------------- Final linear projection back to original dim --------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # (B, N, N, dim)

    return out

"""
TriMul (outgoing) forward pass – a Triton‑accelerated implementation.

The module follows the reference PyTorch version from AlphaFold3.  The
computation can be written as

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where ``left`` and ``right`` are linear projections of the layer‑normalised
input, gated by a sigmoid, and optionally masked.  The reduction over ``k``
is an O(N³) operation.  All cheap operations (LayerNorm, Linear, Sigmoid,
masking) are performed with PyTorch; the heavy triple‑product is carried
out by a custom Triton kernel that treats each ``(batch, hidden)`` pair as a
separate matrix multiplication:

    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ

The kernel tiles the output (i,j) in blocks (BLOCK_M × BLOCK_N) and loops
over the reduction axis k in blocks (BLOCK_K).  It works for the test
configurations (B∈{1,2}, N∈{128,256,512,1024}, hidden_dim∈{128,384}) on an
H100 GPU.

The final output is passed through a second LayerNorm, multiplied by an
output gate, and projected back to the original channel dimension.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rk, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute  out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]   (H = hidden).

    The kernel tiles the (i,j) plane and loops over k.  The third program
    dimension (pid_z) encodes the combined (batch, hidden) index.
    """
    pid_i = tl.program_id(0)               # tile row   i
    pid_j = tl.program_id(1)               # tile col   j
    pid_z = tl.program_id(2)               # (batch, hidden) index

    # Decompose pid_z into batch index and hidden‑channel index
    batch_idx  = pid_z // H
    hidden_idx = pid_z %  H

    # Offsets inside the N‑dimensional matrices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------------------------------------------------------
        # Load a tile from left:   left[b, i, k, h]
        # Strides: (lb, li, lk, lh)
        # -----------------------------------------------------------------
        a_ptr = left_ptr + batch_idx * stride_lb \
                + hidden_idx * stride_lh \
                + offs_i[:, None] * stride_li \
                + offs_k[None, :] * stride_lk
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # -----------------------------------------------------------------
        # Load a tile from right (already transposed):
        # right_t[b, k, j, h]  (shape (BLOCK_K, BLOCK_N))
        # Strides: (rb, rk, rj, rh)
        # -----------------------------------------------------------------
        b_ptr = right_ptr + batch_idx * stride_rb \
                + hidden_idx * stride_rh \
                + offs_k[:, None] * stride_rk \
                + offs_j[None, :] * stride_rj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Multiply‑accumulate for this K‑block
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------
    # Store the computed tile into out[b, i, j, h]
    # Strides: (ob, oi, oj, oh)
    # -----------------------------------------------------------------
    c_ptr = out_ptr + batch_idx * stride_ob \
            + hidden_idx * stride_oh \
            + offs_i[:, None] * stride_oi \
            + offs_j[None, :] * stride_oj
    tl.store(c_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def trimul_out_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the 4‑D einsum ``... i k d, ... j k d -> ... i j d`` using the
    Triton kernel above.  ``left`` and ``right`` must be contiguous tensors
    of shape (B, N, N, H).  The returned tensor has the same shape and dtype.
    """
    assert left.shape == right.shape, "Left and right must share shape"
    B, N, _, H = left.shape

    # Allocate output
    out = torch.empty_like(left)

    # Transpose the second axis of ``right`` so the reduction becomes a
    # standard matrix multiplication: right_t[b, k, j, h] = right[b, j, k, h].
    right_t = right.permute(0, 2, 1, 3).contiguous()

    # Extract element‑wise strides (not byte strides)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rk, stride_rj, stride_rh = right_t.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tunable tile sizes – these values give good occupancy on H100 for the
    # required problem sizes.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (tiles over i, tiles over j, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H

        left,
        right_t,

        B, N, H,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rk, stride_rj, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # 4 warps works well for the tested shapes
    )
    return out

    Forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor of shape (B, N, N, dim)
        - mask         : torch.Tensor of shape (B, N, N) (may be None)
        - weights      : dict of model parameters (see code)
        - config       : dict containing at least ``dim``, ``hidden_dim`` and
                         ``nomask`` (bool, True → no mask)

    Returns
    -------
    torch.Tensor
        Output tensor of shape (B, N, N, dim)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Load model parameters
    # --------------------------------------------------------------
    norm_weight         = weights["norm.weight"]
    norm_bias           = weights["norm.bias"]
    left_proj_weight    = weights["left_proj.weight"]
    right_proj_weight   = weights["right_proj.weight"]
    left_gate_weight    = weights["left_gate.weight"]
    right_gate_weight   = weights["right_gate.weight"]
    out_gate_weight     = weights["out_gate.weight"]
    to_out_norm_weight  = weights["to_out_norm.weight"]
    to_out_norm_bias    = weights["to_out_norm.bias"]
    to_out_weight       = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5
    )

    # --------------------------------------------------------------
    # 2) Linear projections (dim → hidden_dim)
    # --------------------------------------------------------------
    # Shape after linear: (B, N, N, hidden_dim)
    left_proj  = torch.nn.functional.linear(x, left_proj_weight)   # bias=False
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3) Gates (sigmoid of another linear projection)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # --------------------------------------------------------------
    # 4) Mask handling – broadcast over the hidden dimension
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: (B, N, N) → (B, N, N, 1) and cast to input dtype
        mask_exp = mask.to(x.dtype).unsqueeze(-1)
    else:
        mask_exp = torch.ones_like(x[..., :1])   # all ones

    # Apply mask and gates (element‑wise)
    left  = left_proj * mask_exp * left_gate
    right = right_proj * mask_exp * right_gate

    # Ensure tensors are contiguous for the Triton kernel
    left  = left.contiguous()
    right = right.contiguous()

    # --------------------------------------------------------------
    # 5) Core Triple‑product (einsum) – Triton acceleration
    # --------------------------------------------------------------
    out = trimul_out_triton(left, right)   # (B, N, N, hidden_dim)

    # --------------------------------------------------------------
    # 6) Output LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5
    )

    # --------------------------------------------------------------
    # 7) Apply the output gate
    # --------------------------------------------------------------
    out = out * out_gate   # broadcast over hidden_dim

    # --------------------------------------------------------------
    # 8) Final linear projection back to original dim (hidden → dim)
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # bias=False

    return out
