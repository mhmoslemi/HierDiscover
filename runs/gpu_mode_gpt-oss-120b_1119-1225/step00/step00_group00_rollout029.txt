"""
TriMul (outgoing) – Triton implementation of the heavy pairwise reduction.

The operator works on a 4‑D pair representation `x` of shape [B, N, N, D].
It consists of:

1. Layer‑norm over the feature dimension D.
2. Linear projections → left/right tensors of shape [B, N, N, H].
3. Optional pair mask applied to both tensors.
4. Gated elementwise multiplication (sigmoid gates on the original norm).
5. Core “tri‑mul”:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   (the “outgoing” variant: both tensors sum over the second axis `k`).
   This is a massive N³ reduction that we accelerate with a custom Triton
   kernel.  The kernel tiles the i‑, j‑ and k‑axes and processes a small
   hidden‑block (H_BLOCK) at a time.  For each hidden slice it performs a
   standard GEMM:   (M×K) @ (K×N)ᵀ, yielding an (M×N) partial result which
   is accumulated over all K‑tiles.
6. Layer‑norm on the hidden dimension, gated by `out_gate`,
   and a final linear projection back to D.

The kernel computes the N³ reduction while respecting padding (masking
out‑of‑bounds i, j, k and hidden indices).  All other lightweight
operations are performed with PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                    # const float *
    right_ptr,                   # const float *
    out_ptr,                     # float *
    B, N, H,                     # problem size (int32)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,   # int64
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_m = tl.program_id(0)            # block row (i)
    pid_n = tl.program_id(1)            # block col (j)
    pid_z = tl.program_id(2)            # combined batch & hidden block

    # ---- block start indices -------------------------------------------------
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # ---- decode batch and hidden block ---------------------------------------
    n_hblocks = tl.cdiv(H, BLOCK_H)                 # hidden blocks per batch
    b = pid_z // n_hblocks                          # batch index
    h_block = pid_z % n_hblocks
    h_start = h_block * BLOCK_H                     # hidden offset for this block

    # ---- intra‑block offsets --------------------------------------------------
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    # ---- out‑of‑bounds masks --------------------------------------------------
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # ---- accumulator: one hidden vector per output element --------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N, BLOCK_H], dtype=tl.float32)

    # ---- loop over the reduction dimension k ---------------------------------
    for k_start in range(0, N, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile : shape (M, K, H)
        a_ptrs = left_ptr \
                 + b * stride_left_b \
                 + offs_i[:, None, None] * stride_left_i \
                 + offs_k[None, :, None] * stride_left_k \
                 + offs_h[None, None, :] * stride_left_h
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # right tile : shape (N, K, H)
        b_ptrs = right_ptr \
                 + b * stride_right_b \
                 + offs_j[:, None, None] * stride_right_i \
                 + offs_k[None, :, None] * stride_right_k \
                 + offs_h[None, None, :] * stride_right_h
        b = tl.load(
            b_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )

        # per‑hidden GEMM: (M×K) @ (K×N)ᵀ → (M×N)
        for hh in range(BLOCK_H):
            # slices have shape (M, K) and (N, K)
            a_slice = a[:, :, hh]          # (M, K)
            b_slice = b[:, :, hh]          # (N, K)
            # `trans_b=True` treats the second operand as (K, N)
            acc[:, :, hh] += tl.dot(a_slice, b_slice, trans_b=True)

    # ---- write back -----------------------------------------------------------
    for hh in range(BLOCK_H):
        # store only if this hidden index is within the real hidden dimension
        store_mask = mask_i[:, None] & mask_j[None, :] & mask_h[hh]
        c_ptrs = out_ptr \
                 + b * stride_out_b \
                 + offs_i[:, None] * stride_out_i \
                 + offs_j[None, :] * stride_out_j \
                 + (h_start + hh) * stride_out_h
        tl.store(c_ptrs, acc[:, :, hh], mask=store_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Args:
        data: tuple (x, mask, weights, config)
            x      – torch.Tensor, [B, N, N, D] (float32)
            mask   – torch.Tensor, [B, N, N] (bool or 0/1) or None
            weights – dict of all linear/ln weights as torch.Tensors
            config – dict with `dim` (D) and `hidden_dim` (H) and optional `nomask`

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # -------------------------------------------------------------------------
    # unpack arguments
    # -------------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------------------
    # load model parameters
    # -------------------------------------------------------------------------
    norm_w, norm_b = weights["norm.weight"], weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w, to_out_norm_b = weights["to_out_norm.weight"], weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # -------------------------------------------------------------------------
    # 1) layer‑norm over the feature dimension
    # -------------------------------------------------------------------------
    x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b)

    # -------------------------------------------------------------------------
    # 2) linear projections (bias‑free)
    # -------------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_w)   # [B, N, N, H]
    right = F.linear(x_norm, right_proj_w)  # [B, N, N, H]

    # -------------------------------------------------------------------------
    # 3) optional pair mask
    # -------------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------------
    # 4) gated elementwise multiplication
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5) core outgoing TriMul (Trition kernel)
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B, N, N, H]

    # strides in elements (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Triton block sizes – tuned for H100, FP32
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 4   # must divide hidden_dim or be padded (mask handles remainder)

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_h = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (grid_m, grid_n, B * grid_h)

    trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left[0],
        stride_left_i=stride_left[1],
        stride_left_k=stride_left[2],
        stride_left_h=stride_left[3],
        stride_right_b=stride_right[0],
        stride_right_i=stride_right[1],
        stride_right_k=stride_right[2],
        stride_right_h=stride_right[3],
        stride_out_b=stride_out[0],
        stride_out_i=stride_out[1],
        stride_out_j=stride_out[2],
        stride_out_h=stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # -------------------------------------------------------------------------
    # 6) post‑processing: layer‑norm over hidden dim, gate, final projection
    # -------------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)
    out = out * out_gate
    out = F.linear(out, to_out_w)

    return out
