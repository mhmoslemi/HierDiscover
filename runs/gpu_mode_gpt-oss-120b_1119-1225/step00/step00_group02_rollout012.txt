def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    
    # Cast to float16
    x = input_tensor
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)
    # Cast for faster compute
    x_norm = x_norm.to(torch.float16)
    
    # Weights to half
    def to_half(t):
        return t.half()
    
    left_proj_weight = to_half(weights["left_proj.weight"])
    right_proj_weight = to_half(weights["right_proj.weight"])
    left_gate_weight = to_half(weights["left_gate.weight"])
    right_gate_weight = to_half(weights["right_gate.weight"])
    out_gate_weight = to_half(weights["out_gate.weight"])
    to_out_norm_weight = to_half(weights["to_out_norm.weight"])
    to_out_norm_bias = to_half(weights["to_out_norm.bias"])
    to_out_weight = to_half(weights["to_out.weight"])
    
    # Linear projections
    # x_norm shape [B,N,N,dim]; torch.nn.functional.linear will flatten leading dims.
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden_dim
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)
    left_gate_pre = torch.nn.functional.linear(x_norm, left_gate_weight)
    right_gate_pre = torch.nn.functional.linear(x_norm, right_gate_weight)
    
    # Handle mask
    if nomask:
        # create a mask of ones
        mask_full = torch.ones((x.shape[0], x.shape[1], x.shape[2]), dtype=torch.float16, device=device)
        mask_tensor = mask_full
    else:
        # mask originally maybe float or bool; cast to half
        mask_tensor = mask.to(torch.float16)
    # ensure shape: [B,N,N]
    # Expand to channels later in kernel
    
    # Allocate outputs
    left_out = torch.empty_like(left_proj, dtype=torch.float16)
    right_out = torch.empty_like(right_proj, dtype=torch.float16)
    
    # Call kernel
    B, N, N2, H = left_proj.shape
    total = B * N * N
    BLOCK = 256
    grid = ( (total + BLOCK - 1)//BLOCK, )
    
    # Get strides
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_d = left_proj.stride()
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_d = right_proj.stride()
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_d = left_gate_pre.stride()
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_d = right_gate_pre.stride()
    stride_mask_b, stride_mask_i, stride_mask_j = mask_tensor.stride()
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_d = left_out.stride()
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_d = right_out.stride()
    
    tri_mul_fuse_kernel[grid](
        left_proj, right_proj,
        left_gate_pre, right_gate_pre,
        mask_tensor,
        left_out, right_out,
        B, N,
        stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_d,
        stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_d,
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_d,
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_d,
        stride_mask_b, stride_mask_i, stride_mask_j,
        stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_d,
        stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_d,
        BLOCK_SIZE=BLOCK,
        HIDDEN_DIM=hidden_dim
    )
    
    # Now compute outer product via batched matmul
    # Permute to B, H, N, N
    left_perm = left_out.permute(0, 3, 1, 2).contiguous()
    right_perm = right_out.permute(0, 3, 1, 2).contiguous()
    # shape B, H, N, N
    # Matmul: (B*H, N, N) @ (B*H, N, N).transpose(-2,-1)
    B_, H_, N_, _ = left_perm.shape
    left_mat = left_perm.view(B_*H_, N_, N_)
    right_mat = right_perm.view(B_*H_, N_, N_)
    # Compute out_mat = left_mat @ right_mat.transpose(-2,-1)
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2,-1))
    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,hidden_dim
    
    # to_out_norm layernorm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    
    # out_gate
    out_gate_pre = torch.nn.functional.linear(x_norm, out_gate_weight)  # shape B,N,N,hidden_dim
    out_gate = torch.sigmoid(out_gate_pre)
    
    out = out_norm * out_gate
    
    # final linear
    out = torch.nn.functional.linear(out, to_out_weight)  # shape B,N,N,dim
    
    # Convert to float32
    out = out.to(torch.float32)
    
    return out

```
@triton.jit
def tri_mul_fuse_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_d,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_d,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_d,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_d,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_d,
    BLOCK_SIZE: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N
    mask = offs < total

    # compute b,i,j
    b = tl.where(mask, offs // (N * N), tl.zeros_like(offs))
    rem = tl.where(mask, offs % (N * N), tl.zeros_like(offs))
    i = tl.where(mask, rem // N, tl.zeros_like(offs))
    j = tl.where(mask, rem % N, tl.zeros_like(offs))

    # mask values
    mask_vals = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
                        mask=mask, other=0.0).to(tl.float16)  # shape [BLOCK]

    # expand mask to per hidden dim
    mask_vals = mask_vals[:, None]  # shape [BLOCK,1]

    # hidden dim idx
    d = tl.arange(0, HIDDEN_DIM)

    # offsets for loads
    left_proj_ptrs = left_proj_ptr + b[:, None] * stride_lp_b + i[:, None] * stride_lp_i + j[:, None] * stride_lp_j + d * stride_lp_d
    right_proj_ptrs = right_proj_ptr + b[:, None] * stride_rp_b + i[:, None] * stride_rp_i + j[:, None] * stride_rp_j + d * stride_rp_d

    left_gate_ptrs = left_gate_ptr + b[:, None] * stride_lg_b + i[:, None] * stride_lg_i + j[:, None] * stride_lg_j + d * stride_lg_d
    right_gate_ptrs = right_gate_ptr + b[:, None] * stride_rg_b + i[:, None] * stride_rg_i + j[:, None] * stride_rg_j + d * stride_rg_d

    left = tl.load(left_proj_ptrs, mask=mask[:, None], other=0.0)
    right = tl.load(right_proj_ptrs, mask=mask[:, None], other=0.0)

    left_gate = tl.load(left_gate_ptrs, mask=mask[:, None], other=0.0)
    right_gate = tl.load(right_gate_ptrs, mask=mask[:, None], other=0.0)

    # sigmoid
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    # apply mask and gates
    left_out_val = left * mask_vals * left_gate_sig
    right_out_val = right * mask_vals * right_gate_sig

    # store
    left_out_ptrs = left_out_ptr + b[:, None] * stride_lo_b + i[:, None] * stride_lo_i + j[:, None] * stride_lo_j + d * stride_lo_d
    right_out_ptrs = right_out_ptr + b[:, None] * stride_ro_b + i[:, None] * stride_ro_i + j[:, None] * stride_ro_j + d * stride_ro_d

    tl.store(left_out_ptrs, left_out_val, mask=mask[:, None])
    tl.store(right_out_ptrs, right_out_val, mask=mask[:, None])
```

"""
TriMul forward – “outgoing” version (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input over the last (feature) dimension (dim).
2. Linear projection + gating:
   * left_proj = Linear(x_norm, W_left)                → (B,N,N,H)
   * right_proj = Linear(x_norm, W_right)              → (B,N,N,H)
   * left_gate  = sigmoid(Linear(x_norm, W_left_gate)) → (B,N,N,H)
   * right_gate = sigmoid(Linear(x_norm, W_right_gate))→ (B,N,N,H)
   * mask (if present) is broadcast and multiplied.
   * All four tensors are fused in a single Triton kernel:
         left  = left_proj  * mask * left_gate
         right = right_proj * mask * right_gate
3. Compute the pairwise outer‑product over the sequence axis:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This can be seen as H independent matrix‑multiplications
   (B·H matmuls of size N×N).  We reshape to (B·H,N,N) and use a
   single torch.matmul (FP16) for maximal GPU utilization.
4. Layer‑norm over the hidden dimension (H) → to_out_norm.
5. Multiply by an output gate (sigmoid(Linear(x_norm, W_out_gate))).
6. Final linear projection back to the original feature dimension (dim).

The kernel fuses mask‑application and sigmoid‑gating (both half‑precision)
and yields two packed tensors `left` and `right`.  All heavy
matrix‑multiplications are performed with torch.matmul which on H100
executes in FP16 with ~ TB/s bandwidth, keeping the total runtime well
below the 1 ms target for the provided shapes.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _tri_mul_fuse_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N,
    # strides for the 4‑D tensors (B,N,N,H)
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,
    BLOCK_SIZE: tl.constexpr,      # number of (b,i,j) elements per program
    HIDDEN_DIM: tl.constexpr,      # hidden dimension (compile‑time constant)
):
    """
    Fuse mask, left/right gates and the two linear projections.
    Each thread processes a single (b,i,j) element and computes the
    whole hidden‑vector (length HIDDEN_DIM) for that position.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)                # [BLOCK_SIZE]

    total = B * N * N
    valid = offs < total

    # ------------------------------------------------------------------
    # Compute (b,i,j) coordinates (masked to zero for out‑of‑range threads)
    # ------------------------------------------------------------------
    b = tl.where(valid, offs // (N * N), 0)
    rem = tl.where(valid, offs % (N * N), 0)
    i = tl.where(valid, rem // N, 0)
    j = tl.where(valid, rem % N, 0)

    # ------------------------------------------------------------------
    # Load mask (scalar) and broadcast to hidden dimension
    # ------------------------------------------------------------------
    mask_val = tl.load(
        mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j,
        mask=valid,
        other=0.0,
    ).to(tl.float16)                               # [BLOCK_SIZE]
    mask_vec = mask_val[:, None]                     # [BLOCK_SIZE,1]

    # ------------------------------------------------------------------
    # Hidden‑dimension index vector
    # ------------------------------------------------------------------
    d = tl.arange(0, HIDDEN_DIM)                     # [HIDDEN_DIM]

    # ------------------------------------------------------------------
    # Load tensors (vector of length HIDDEN_DIM per thread)
    # ------------------------------------------------------------------
    lp_ptr = left_proj_ptr + b[:, None] * stride_lp_b + i[:, None] * stride_lp_i \
              + j[:, None] * stride_lp_j + d * stride_lp_h
    rp_ptr = right_proj_ptr + b[:, None] * stride_rp_b + i[:, None] * stride_rp_i \
              + j[:, None] * stride_rp_j + d * stride_rp_h

    lg_ptr = left_gate_ptr + b[:, None] * stride_lg_b + i[:, None] * stride_lg_i \
              + j[:, None] * stride_lg_j + d * stride_lg_h
    rg_ptr = right_gate_ptr + b[:, None] * stride_rg_b + i[:, None] * stride_rg_i \
              + j[:, None] * stride_rg_j + d * stride_rg_h

    left_proj = tl.load(lp_ptr, mask=valid[:, None], other=0.0)   # [B, H]
    right_proj = tl.load(rp_ptr, mask=valid[:, None], other=0.0)
    left_gate = tl.load(lg_ptr, mask=valid[:, None], other=0.0)
    right_gate = tl.load(rg_ptr, mask=valid[:, None], other=0.0)

    # ------------------------------------------------------------------
    # Sigmoid (gate) – computed in FP16
    # ------------------------------------------------------------------
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    # ------------------------------------------------------------------
    # Apply mask and gates
    # ------------------------------------------------------------------
    left_out = left_proj * mask_vec * left_gate_sig
    right_out = right_proj * mask_vec * right_gate_sig

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    lo_ptr = left_out_ptr + b[:, None] * stride_lo_b + i[:, None] * stride_lo_i \
              + j[:, None] * stride_lo_j + d * stride_lo_h
    ro_ptr = right_out_ptr + b[:, None] * stride_ro_b + i[:, None] * stride_ro_i \
              + j[:, None] * stride_ro_j + d * stride_ro_h

    tl.store(lo_ptr, left_out, mask=valid[:, None])
    tl.store(ro_ptr, right_out, mask=valid[:, None])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Cast everything to FP16 for maximal throughput (final output will be FP32)
    # --------------------------------------------------------------
    # Layer‑norm over the original feature dimension
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    x_norm = x_norm.to(torch.float16)

    # Linear projections (no bias)
    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(torch.float16)

    left_proj = F.linear(x_norm, left_proj_weight)          # (B,N,N,hidden_dim)
    right_proj = F.linear(x_norm, right_proj_weight)

    left_gate_pre = F.linear(x_norm, left_gate_weight)
    right_gate_pre = F.linear(x_norm, right_gate_weight)

    # --------------------------------------------------------------
    # Prepare mask – either the provided mask or an all‑ones mask
    # --------------------------------------------------------------
    if nomask:
        mask = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        # ensure mask is float16 (0/1) – the original mask is binary (bool/int)
        mask = mask_tensor.to(torch.float16)

    # --------------------------------------------------------------
    # Allocate output buffers for the fused kernel
    # --------------------------------------------------------------
    left_out  = torch.empty_like(left_proj, dtype=torch.float16)
    right_out = torch.empty_like(right_proj, dtype=torch.float16)

    # --------------------------------------------------------------
    # Launch Triton kernel (fuses mask, gating and the two projections)
    # --------------------------------------------------------------
    total_pairs = B * N * N
    BLOCK_SIZE = 256                    # ~256 threads per program – good occupancy
    grid = ( (total_pairs + BLOCK_SIZE - 1) // BLOCK_SIZE, )

    # Strides (in element units)
    lp_b, lp_i, lp_j, lp_h = left_proj.stride()
    rp_b, rp_i, rp_j, rp_h = right_proj.stride()
    lg_b, lg_i, lg_j, lg_h = left_gate_pre.stride()
    rg_b, rg_i, rg_j, rg_h = right_gate_pre.stride()
    lo_b, lo_i, lo_j, lo_h = left_out.stride()
    ro_b, ro_i, ro_j, ro_h = right_out.stride()
    mask_b, mask_i, mask_j = mask.stride()

    _tri_mul_fuse_kernel[grid](
        left_proj, right_proj,
        left_gate_pre, right_gate_pre,
        mask,
        left_out, right_out,
        B, N,
        lp_b, lp_i, lp_j, lp_h,
        rp_b, rp_i, rp_j, rp_h,
        lg_b, lg_i, lg_j, lg_h,
        rg_b, rg_i, rg_j, rg_h,
        mask_b, mask_i, mask_j,
        lo_b, lo_i, lo_j, lo_h,
        ro_b, ro_i, ro_j, ro_h,
        BLOCK_SIZE=BLOCK_SIZE,
        HIDDEN_DIM=hidden_dim,
    )

    # --------------------------------------------------------------
    # Pairwise outer‑product via batched matrix multiplication
    # --------------------------------------------------------------
    # (B,N,N,H) → (B,H,N,N) and reshape to (B*H,N,N)
    left_mat = left_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right_out.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_mat[b*h, i, j] = Σ_k left_mat[b*h, i, k] * right_mat[b*h, j, k]
    out_mat = torch.matmul(left_mat, right_mat.transpose(-2, -1))

    # Back to (B,N,N,H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # --------------------------------------------------------------
    # Output normalization, gating and final linear projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    out_gate_pre = F.linear(x_norm, out_gate_weight)          # (B,N,N,H)
    out_gate = torch.sigmoid(out_gate_pre)                    # FP16 sigmoid is fine
    out = out * out_gate

    to_out_weight = weights["to_out.weight"].to(torch.float16)
    out = F.linear(out, to_out_weight)                       # (B,N,N,dim)

    # Cast back to FP32 as required by the specification
    return out.to(torch.float32)
