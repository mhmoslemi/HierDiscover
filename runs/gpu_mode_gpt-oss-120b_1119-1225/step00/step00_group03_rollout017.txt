    out = out_half.to(torch.float32)

    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Multiply by out_gate and final linear

    out = out * out_gate
    # final linear: out_proj from hidden to dim (no bias)
    out = torch.nn.functional.linear(out, to_out_weight)

"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm over the last dimension of the pair representation.
2. Linear projections (left/right) and gating layers (all without bias).
3. Apply optional mask and element‑wise gates.
4. Compute the core operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is exactly a batched matrix multiplication
       out = left @ rightᵀ
   for every hidden channel `d`.  The heavy O(N³) work is performed by a
   fused Triton kernel that treats each (batch, hidden) pair as an
   independent matrix‑multiply task.
5. Layer‑norm over the hidden dimension, element‑wise out‑gate,
   and a final linear projection back to the original `dim`.

The Triton kernel implements a standard GEMM (A·B) with fp16 inputs,
fp32 accumulation and fp16 output.  It is launched with a 3‑D grid:
   (tile_m, tile_n, batch*hidden).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (M×K)·(K×N) → M×N
    stride_am, stride_ak, stride_an,   # a: batch, row, col strides
    stride_bm, stride_bk, stride_bn,   # b: batch, row, col strides
    stride_cm, stride_ck, stride_cn,   # c: batch, row, col strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ B.
    A and B are fp16, accumulation in fp32, result stored as fp16.
    Shapes:
        A : (batch, M, K)
        B : (batch, K, N)
        C : (batch, M, N)
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_z = tl.program_id(2)          # batch * hidden

    # -------------------------------------------------------------
    # Tile indices
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [N]
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of this batch (z‑dimension)
    a_batch = a_ptr + pid_z * stride_am
    b_batch = b_ptr + pid_z * stride_bm

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension in BLOCK_K sized chunks
    # -------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + (offs_m[:, None] * stride_ak) + (offs_k[None, :] * stride_an)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile: (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Dot product (uses TensorCores when possible)
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # Write back the result (fp16)
    # -------------------------------------------------------------
    c_ptrs = c_ptr + pid_z * stride_cm + (offs_m[:, None] * stride_ck) + (offs_n[None, :] * stride_cn)
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle Multiplicative Update (TriMul).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # Weight tensors (all provided as float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)       # (out_features=hidden_dim, in_features=dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul (batched GEMM) – implemented in Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape                     # B=batch, N=seq_len, H=hidden_dim
    # reshape to (B, H, N, N) and make contiguous for simple strides
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)
    # transpose the last two axes of `right` to obtain rightᵀ for the GEMM
    right_perm_T = right_perm.permute(0, 1, 3, 2).contiguous()  # (B, H, N, N)

    # flatten batch & hidden into a single batch dimension
    left_flat = left_perm.view(-1, N, N)            # (B*H, N, N)
    right_flat = right_perm_T.view(-1, N, N)        # (B*H, N, N)

    # cast to fp16 for the kernel (fp32 accumulation inside)
    left_half = left_flat.to(torch.float16)
    right_half = right_flat.to(torch.float16)

    # output buffer (fp16)
    out_half = torch.empty_like(left_half, dtype=torch.float16)

    # -----------------------------------------------------------------------------
    # Kernel launch configuration
    # -----------------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),          # tiles in M dimension
        triton.cdiv(N, BLOCK_N),          # tiles in N dimension
        left_half.shape[0]                # batch * hidden
    )

    _tri_mul_matmul_kernel[grid](
        left_half,
        right_half,
        out_half,
        N, N, N,                         # M, N, K
        left_half.stride(0), left_half.stride(1), left_half.stride(2),
        right_half.stride(0), right_half.stride(1), right_half.stride(2),
        out_half.stride(0), out_half.stride(1), out_half.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original layout: [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_half.to(torch.float32)                 # back to fp32 for the rest
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ------------------------------------------------------------------
    # 7. Apply out‑gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                      # element‑wise gating
    out = F.linear(out, to_out_weight)        # (out_features=dim, in_features=hidden_dim)

    return out
