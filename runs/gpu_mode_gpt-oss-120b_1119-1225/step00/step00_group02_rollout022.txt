"""
TriMul outgoing kernel implementation (alphaFold3 style).

The operation processes a 4‑D tensor x ∈ ℝ^{B×N×N×C}
1. Layer‑norm over the channel dimension C.
2. Two linear projections (left/right) → hidden_dim H and learned gates (sigmoid).
3. Apply mask (if present) and element‑wise gates.
4. Compute the “outgoing” pairwise update:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched matrix multiplication of shape (B·H, N, N) × (B·H, N, N)^T.
   The heavy N³ work is performed by a Triton kernel that loads half‑precision
   values and accumulates in fp32.
5. Layer‑norm over H, multiplicative out‑gate, final linear projection back
   to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel for the batched multiplication A @ Bᵀ
# A, B: (BH, N, N)  (row‑major, contiguous)
# C: (BH, N, N) = Σ_k A[...,i,k] * B[...,j,k]
# ------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_batch = tl.program_id(axis=2)        # BH index
    pid_m = tl.program_id(axis=0)            # block row (i)
    pid_n = tl.program_id(axis=1)            # block col (j)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------------------------------------------------
        # Load a block of A (shape M×K) – contiguous in k
        # ----------------------------------------------------
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ----------------------------------------------------
        # Load a block of B transposed (shape K×N)
        # B is stored as (j, k), we need (k, j) → exchange axes.
        # ----------------------------------------------------
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Accumulate the dot product over the K dimension
        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)
    # ----------------------------------------------------
    # Write back the result
    # ----------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    Forward pass of the outgoing TriMul module.
    `data` is a tuple (input, mask, weights, config).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm over the last channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B, N, N, dim], dtype=float32

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gate values
    # ------------------------------------------------------------------
    # Projections to hidden_dim
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Optional mask (broadcast to hidden_dim)
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)            # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Gates (sigmoid)
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply gates
    # ------------------------------------------------------------------
    # Cast to half‑precision before the heavy N³ work (reduces memory traffic)
    left = (left_proj * left_gate).to(torch.float16).contiguous()
    right = (right_proj * right_gate).to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 4. Triton kernel: batched A @ Bᵀ, where
    #    A = left, B = right, result shape [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    # Prepare tensors for the kernel: flatten batch & hidden dimension
    # into a single “batch” axis of size BH = B * hidden_dim.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N]
    right_flat = right_perm.view(BH, N, N)

    # Output buffer (FP32 accumulation)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Tile sizes – chosen to fit well on H100 for fp16 inputs.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          BH )
    )

    # Strides (torch gives them as python ints)
    stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_i, stride_c_j = out_flat.stride()

    # Launch kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Reshape the kernel output back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate   # multiplicative gating
    # Final projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul outgoing kernel (AlphaFold3 style).

The forward pass performs:
1. Layer‑norm over the last channel (dim).
2. Two bias‑free linear projections (left/right) and three sigmoid gates.
3. Optional masking of the projections.
4. A heavy N³ update:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched matrix multiplication A @ Bᵀ where
   A,B ∈ ℝ^{(B·H)×N×N} (H = hidden_dim). The computation is carried out
   by a custom Triton kernel that works in fp16 (half) on the inputs and
   accumulates in fp32 for numerical stability.
5. A second Layer‑norm over H, multiplicative out‑gate, and a final
   linear projection back to the original channel dimension.

Only the N³ core is implemented in Triton; all other operations use
standard PyTorch which is cheap compared to the matrix multiply.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: batched A @ Bᵀ  (A and B are half‑precision)
# ------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs (grid = (num_M_blocks, num_N_blocks, BH))
    # ------------------------------------------------------------------
    pid_batch = tl.program_id(axis=2)   # index over B·H
    pid_m = tl.program_id(axis=0)       # block‑row (i)
    pid_n = tl.program_id(axis=1)       # block‑col (j)

    # ------------------------------------------------------------------
    # Offsets for the current M×N block
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis k
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- Load a block of A (M×K) ---------------------------------
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ----------- Load a block of Bᵀ (K×N) -------------------------------
        # B is stored as (j,k); we need (k,j) → exchange the axes.
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ----------- Accumulate ------------------------------------------------
        acc += tl.dot(a, b)   # (M,K)·(K,N) → (M,N)  (fp32 accumulation)
    # ----------------------------------------------------------------------
    # Write the result back to C
    # ----------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ------------------------------------------------------------
# Public entry point used by the grader
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N] (optional)
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys "dim", "hidden_dim", "nomask" ...

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. First LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    # Optional masking (applied before gates for consistency with the reference)
    if (not nomask) and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)          # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Gates (sigmoid)
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3. Apply gates – cast to half‑precision for the heavy compute
    # ------------------------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16).contiguous()
    right = (right_proj * right_gate).to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 4. Triton kernel: batched A @ Bᵀ
    # ------------------------------------------------------------------
    # Rearrange to (B, H, N, N) → flatten B·H into one batch dimension.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)   # [BH, N, N], fp16
    right_flat = right_perm.view(BH, N, N)

    # Output buffer (fp32 accumulation)
    out_flat = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Tile sizes – tuned for H100 tensor‑cores (fp16)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (num_M_blocks, num_N_blocks, BH)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    # Strides for the three tensors
    stride_a_batch, stride_a_i, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_i, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_i, stride_c_j = out_flat.stride()

    # Launch the kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_a_batch,
        stride_a_i,
        stride_a_k,
        stride_b_batch,
        stride_b_i,
        stride_b_k,
        stride_c_batch,
        stride_c_i,
        stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                       # multiplicative gating
    out = F.linear(out, weights["to_out.weight"])  # back to `dim`

    return out
```
