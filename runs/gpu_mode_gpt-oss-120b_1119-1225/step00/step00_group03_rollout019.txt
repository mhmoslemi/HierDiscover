import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)   # i tile
    pid_n = tl.program_id(1)   # j tile
    pid_z = tl.program_id(2)   # batch*hidden

    # compute batch id and hidden dim
    b = pid_z // hidden_dim
    d = pid_z % hidden_dim

    # offset within N
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # create masks for i and j
    mask_i = i < N
    mask_j = j < N

    # pointers to start of current batch and channel for left and right
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d

    # pointer iteration for k dimension
    a_ptrs = left_base + i[:, None] * stride_left_i + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_base + tl.arange(0, BLOCK_K)[:, None] * stride_right_k + j[None, :] * stride_right_i

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for _ in range(0, tl.cdiv(N, BLOCK_K)):
        # load with mask
        a = tl.load(a_ptrs, mask=mask_i[:, None] & (tl.arange(0, BLOCK_K)[None, :] < N), other=0.0)

def custom_kernel(data):
    inp, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    B, N, _, _ = inp.shape
    device = inp.device

    # LayerNorm
    x = torch.nn.functional.layer_norm(inp, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(inp.dtype)  # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Now compute truncated multiplication via Triton
    # Ensure contiguity
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim], dtype float32

    # obtain strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # sync maybe not needed

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass

The routine follows the PyTorch reference implementation:

1. Layer‑norm on the last channel (dim).
2. Two linear projections (left / right) from dim → hidden_dim.
3. Optional pairwise mask is applied to the projected tensors.
4. Three gating linear layers (left‑gate, right‑gate, out‑gate) are
   computed from the normalized input, passed through a sigmoid and
   multiplied back onto the projections.
5. The heavy N³ reduction
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is performed by a custom Triton kernel.  The kernel treats each
   (batch, hidden‑channel) pair as an independent GEMM of shape
   (N × N) and tiles the i‑ and j‑axes with BLOCK_M × BLOCK_N tiles,
   accumulating over K in blocks of size BLOCK_K.  Inputs are stored
   in fp16 to reduce memory traffic, accumulation is in fp32 and the
   final output is fp32.
6. The result is layer‑normed, multiplied by the out‑gate and passed
   through the final linear projection back to dim.

Only the N³ reduction is Triton‑accelerated – all other ops use
high‑performance PyTorch kernels.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: one GEMM per (batch, hidden_channel) pair
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ------------------------------- program ids -------------------------------
    pid_i = tl.program_id(0)                     # tile index for i (rows)
    pid_j = tl.program_id(1)                     # tile index for j (cols)
    pid_z = tl.program_id(2)                     # combined (batch, hidden)

    # decode batch index and hidden dimension from pid_z
    b = pid_z // hidden_dim
    d = pid_z % hidden_dim

    # ------------------------------- tile coordinates -------------------------------
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)  # i coordinates within N
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)  # j coordinates within N

    mask_i = i_base < N
    mask_j = j_base < N

    # base pointers for the selected batch and channel
    left_base = left_ptr \
        + b * stride_left_b \
        + d * stride_left_d
    right_base = right_ptr \
        + b * stride_right_b \
        + d * stride_right_d

    # pointers that will be advanced over K
    a_ptrs = left_base \
        + i_base[:, None] * stride_left_i \
        + tl.arange(0, BLOCK_K)[None, :] * stride_left_k
    b_ptrs = right_base \
        + tl.arange(0, BLOCK_K)[:, None] * stride_right_k \
        + j_base[None, :] * stride_right_i

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K (the sequence length)
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        k_valid = (tl.arange(0, BLOCK_K) + k_start) < N

        # mask for valid (i,k) and (k,j) entries
        mask_a = mask_i[:, None] & k_valid[None, :]
        mask_b = k_valid[:, None] & mask_j[None, :]

        # load tiles (fp16) and cast to fp32 for the dot
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16).to(tl.float32)

        # matrix multiplication of the tile
        acc += tl.dot(a, b)

        # advance pointers to the next K‑block
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # ----------------------------------------------------------------------
    # store result
    # ----------------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + d * stride_out_d \
        + i_base[:, None] * stride_out_i \
        + j_base[None, :] * stride_out_j

    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments:
        data: tuple (input, mask, weights, config)
            - input  : torch.Tensor    [bs, N, N, dim]
            - mask   : torch.Tensor or None  [bs, N, N] (optional)
            - weights: dict of torch.Tensor with the model parameters
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = x.shape
    device = x.device
    dtype = x.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last channel
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional pairwise mask
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask is [B,N,N]; broadcast on the channel dimension
        mask = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating linear layers + sigmoid
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy N³ reduction using Triton (compute in fp16, accumulate fp32)
    # ------------------------------------------------------------------
    # Cast projections to half‑precision to reduce memory traffic
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (PyTorch uses row‑major layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # tile sizes – tuned for H100 (adjustable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        bs * hidden_dim,                       # (batch, channel) dimension
    )

    trimul_kernel[grid](
        left, right, out,
        bs, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # a modest number of warps works well for 64×64 tiles
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: to_out_norm, out_gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # gating
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
