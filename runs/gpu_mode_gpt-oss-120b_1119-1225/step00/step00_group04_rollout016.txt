@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_k, stride_right_j, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    batch = pid_bh // H
    hidden = pid_bh % H

    # compute start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    # offsets for mask
    off_i = i_start + tl.arange(0, BLOCK_M)
    off_j = j_start + tl.arange(0, BLOCK_N)

    # masks to handle boundary
    mask_i = off_i < N
    mask_j = off_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        off_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # load A (left)
        a_ptrs = left_ptr \
            + batch * stride_left_batch \
            + hidden * stride_left_h \
            + off_i[:, None] * stride_left_i \
            + off_k[None, :] * stride_left_k

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # a is float16 or whatever; convert to float32
        a = a.to(tl.float32)

        # load B (right) transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr \
            + batch * stride_right_batch \
            + hidden * stride_right_h \
            + off_k[:, None] * stride_right_k \
            + off_j[None, :] * stride_right_j

        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # store result
    out_ptrs = out_ptr \
        + batch * stride_out_batch \
        + hidden * stride_out_h \
        + off_i[:, None] * stride_out_i \
        + off_j[None, :] * stride_out_j

    # Cast acc to original dtype (maybe float16). We'll store as float32. 
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated core contraction.

Algorithm
---------

1. Layer‑norm the input tensor `x` over its last dimension (`dim`).
2. Linear projections (no bias) produce left and right pair‑wise tensors
      left  = x @ W_left   (shape [B,N,N,H])
      right = x @ W_right  (shape [B,N,N,H])
   Cast the projections to float16 to save memory and bandwidth.
3. (Optional) apply the pair‑wise mask, i.e. `left *= mask` and `right *= mask`.
4. Compute three sigmoidal gates from `x`:
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
   The first two gates are cast to the projection dtype and fused into the
   projected tensors. The output gate is kept in float32.
5. **Core contraction** – compute for every batch `b`, hidden channel `h`:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply `left[..., :, :] @
   right[..., :, :].T` per hidden channel.  It is implemented as a Triton
   kernel (`tri_mul_kernel`) that performs a standard GEMM with the
   `right` operand read transposed (k‑dimension first, j‑dimension second).
   The kernel is tiled with `BLOCK_M × BLOCK_N` threads (≤ 1024 threads per
   block) and loops over `k` in `BLOCK_K` sized tiles, accumulating in
   float32.
6. Apply a second layer‑norm over the hidden dimension, multiply by the
   output gate, and finally project back to the original dimension with
   `to_out` linear layer.

The fusion of the N³ contraction into a single Triton kernel yields the
largest performance benefit over a naïve PyTorch `einsum`.  All other
operations remain in PyTorch to keep the implementation simple and
correct.

The function `custom_kernel` follows the required signature:
    (input, mask, weights, config) -> output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const * half
    right_ptr,         # const * half (will be read transposed)
    out_ptr,           # * float
    B, N, H,           # problem size
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_k, stride_right_j, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    # -----------------------------
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton implementation of the core TriMul contraction.
    Each program instance computes a (BLOCK_M × BLOCK_N) tile of the output
    matrix for a single combination of (batch, hidden) given by
    `pid_bh = tl.program_id(2)`.
    """
    pid_i = tl.program_id(0)          # tile index over the i‑dimension
    pid_j = tl.program_id(1)          # tile index over the j‑dimension
    pid_bh = tl.program_id(2)         # combined batch & hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden channel from pid_bh
    # ------------------------------------------------------------------
    batch = pid_bh // H
    hidden = pid_bh % H

    # ------------------------------------------------------------------
    # Compute the start offsets for this tile
    # ------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k‑dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- Load left (shape: M × K) -------
        a_ptrs = (
            left_ptr
            + batch * stride_left_batch
            + hidden * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        a = a.to(tl.float32)                     # promote to fp32 for accumulation

        # ------- Load right transposed (shape: K × N) -------
        # right[b, j, k, h]  -> we need right_T[k, j] = right[b, j, k, h]
        b_ptrs = (
            right_ptr
            + batch * stride_right_batch
            + hidden * stride_right_h
            + offs_k[:, None] * stride_right_k    # k dimension first (row)
            + offs_j[None, :] * stride_right_j    # j dimension second (col)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        b = b.to(tl.float32)

        # ------- Accumulate the block product -------
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # ------------------------------------------------------------------
    # Write the result back to global memory
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_out_batch
        + hidden * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )
    # end of kernel


# ----------------------------------------------------------------------
# Entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config) where
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] (optional, may be all‑ones)
            - weights      : dict of torch.Tensors matching the module's parameters
            - config       : dict containing at least 'dim', 'hidden_dim', and 'nomask'
    Returns
    -------
    torch.Tensor : shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)        # True -> ignore mask

    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the `dim` channel (last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) -> cast to float16 to save bandwidth
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"]).to(torch.float16)   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Optional mask multiplication (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # ensure mask dtype matches the projections
        mask_f = mask.to(left.dtype).unsqueeze(-1)    # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Compute gating tensors (keep them in float32, then cast)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])).to(left.dtype)
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])).to(right.dtype)
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))        # float32

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core contraction via Triton kernel
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), device=device, dtype=torch.float32)

    # Strides (in elements, not bytes)
    stride_left_batch = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_h = left.stride(3)

    stride_right_batch = right.stride(0)
    stride_right_k = right.stride(2)      # k‑dimension (the third axis)
    stride_right_j = right.stride(1)      # j‑dimension (second axis)
    stride_right_h = right.stride(3)

    stride_out_batch = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Tunable tile sizes – 32×32 threads per block (1024 total)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,           # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,           # j‑tiles
        B * hidden_dim,                         # batch × hidden
    )

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_batch=stride_left_batch,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_batch=stride_right_batch,
        stride_right_k=stride_right_k,
        stride_right_j=stride_right_j,
        stride_right_h=stride_right_h,
        stride_out_batch=stride_out_batch,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ensure kernel completion before using `out`
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 6) Post‑process: second LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final shape [B,N,N,dim]

    return out
