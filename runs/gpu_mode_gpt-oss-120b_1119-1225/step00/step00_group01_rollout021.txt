@triton.jit
def trmul_kernel(A, B, C,
                 M, N, K,
                 stride_am, stride_ai, stride_ak,
                 stride_bm, stride_bi, stride_bk,
                 stride_cm, stride_ci, stride_cj,
                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)  # batch
    pid_i = tl.program_id(1)  # tile row
    pid_j = tl.program_id(2)  # tile col

    # compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base offsets for A and B
    a_batch_off = pid * stride_am
    b_batch_off = pid * stride_bm
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A + a_batch_off + offs_i[:, None] * stride_ai + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load B tile transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = B + b_batch_off + offs_j[:, None] * stride_bi + cur_k[None, :] * stride_bk
        # Since we want B[j,k] (original B shape)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b, out_dtype=tl.float32)
    # Store result
    # compute offset in C: linear offset: row i * N + col j
    # C is shaped (M,N,N)
    # stride across rows = N (since contiguous)
    # use stride_ci = N, stride_cj = 1 perhaps
    c_batch_off = pid * stride_cm
    # compute linear offsets for each element
    c_ptrs = C + c_batch_off + (offs_i[:, None] * stride_ci) + offs_j[None, :]
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
Triangular Multiplicative Update (outgoing) – Triton‑accelerated core.

The heavy part of the TriMul forward pass is the contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
For each batch * hidden channel we have a matrix multiplication:
    out₍b,d₎ = left₍b,d₎ @ right₍b,d₎ᵀ   (N×N) @ (N×N)ᵀ → (N×N).

A custom Triton kernel performs the batched mat‑mul
    (M = B·hidden,  A shape (M,N,N),  B shape (M,N,N))
using a tiled algorithm (BLOCK_M×BLOCK_N×BLOCK_K).  All other
operations (LayerNorm, linear projections, gating, masking) are
expressed with PyTorch for simplicity and correctness.

The kernel works with fp32 inputs (the same dtype as the reference
implementation) and accumulates in fp32, guaranteeing numerical
equivalence.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiply with transpose on the second
# operand (exactly the einsum used in TriMul).
# ----------------------------------------------------------------------
@triton.jit
def trmul_kernel(
    A_ptr,               # left  matrix (M,N,K)   – stride  (batch,row,col)
    B_ptr,               # right matrix (M,N,K)   – stride  (batch,row,col)
    C_ptr,               # output (M,N,N)         – stride  (batch,row,col)
    M, N, K,             # logical dimensions
    stride_am, stride_ai, stride_ak,   # A strides
    stride_bm, stride_bi, stride_bk,   # B strides
    stride_cm, stride_ci, stride_cj,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)          # batch index   0 <= pid < M
    pid_i = tl.program_id(1)        # tile row index
    pid_j = tl.program_id(2)        # tile col index

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets for this batch element
    a_batch_off = pid * stride_am
    b_batch_off = pid * stride_bm
    c_batch_off = pid * stride_cm

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main K‑loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = cur_k < K

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + a_batch_off \
                 + offs_i[:, None] * stride_ai \
                 + cur_k[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # B tile (transposed): we need Bᵀ[k, j] = B[j, k]
        # Load as shape (BLOCK_K, BLOCK_N)
        b_ptrs = B_ptr + b_batch_off \
                 + cur_k[:, None] * stride_bi \
                 + offs_j[None, :] * stride_bk
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate product over K
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # Store the result tile
    # --------------------------------------------------------------
    c_ptrs = C_ptr + c_batch_off \
             + offs_i[:, None] * stride_ci \
             + offs_j[None, :] * stride_cj
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point – reproduces the forward method of the reference
# TriMul module while delegating the O(N³·hidden) contraction to Triton.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the outgoing TriMul operation using a fused Triton kernel
    for the core three‑dimensional contraction.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [bs, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [bs, N, N] or None
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor of shape [bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper to fetch parameters (weights are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → shape [bs, N, N, hidden]
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # (bs,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask is bool/float, expand to last dim
        mask_exp = mask.unsqueeze(-1).to(x.dtype)           # [bs,N,N,1]
        left  = left  * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    # Rearrange tensors to (batch, hidden, N, N) → collapse batch*hidden
    left  = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()

    M = bs * hidden_dim                 # number of independent matmuls
    left_flat  = left.view(M, N, N)     # (M, N, N)
    right_flat = right.view(M, N, N)    # (M, N, N)

    # Output buffer (FP32 as required by the model)
    out_flat = torch.empty((M, N, N), dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_M = 64   # rows per tile
    BLOCK_N = 64   # cols per tile
    BLOCK_K = 32   # reduction tile

    grid = (M,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N, N,                               # M, N (output rows), K (reduction)
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),   # A strides
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),# B strides
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),     # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,    # reasonable for fp32 on H100
    )

    # Reshape back to the original layout: (bs, N, N, hidden)
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm → gating → final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
