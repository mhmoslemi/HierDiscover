```python
import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C = A @ B.T where A and B are (BATCH, M, K) and (BATCH, N, K) respectively."""
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the batch
    a_batch_ptr = A_ptr + batch_idx * stride_a_batch
    b_batch_ptr = B_ptr + batch_idx * stride_b_batch
    c_batch_ptr = C_ptr + batch_idx * stride_c_batch

    # Masks for out-of-bound rows / cols
    mask_m = offs_m[:, None] < M
    mask_n = offs_n[None, :] < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Load A tile (M x K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_row) + ((k_offset + offs_k)[None, :] * stride_a_col)
        # Load B tile transposed (K x N)  (original B is (N, K))
        b_ptrs = b_batch_ptr + ((k_offset + offs_k)[:, None] * stride_b_col) + (offs_n[None, :] * stride_b_row)

        # Compute per‑tile masks
        mask_a = mask_m & ((k_offset + offs_k)[None, :] < K)
        mask_b = ((k_offset + offs_k)[:, None] < K) & mask_n

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        acc += tl.dot(a, b)

    # Store the result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    mask_c = mask_m & mask_n
    tl.store(c_ptrs, acc, mask=mask_c)

def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    The heavy 3‑D tensor multiplication (∑_k left[...,i,k,d] * right[...,j,k,d]) is
    executed with a fused Triton kernel for high performance.
    """
    # unpack arguments
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # LayerNorm on the input (dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # Linear projections (dim → hidden_dim)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [bs, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Mask (optional)
    # -----------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones(bs, N, N, device=device, dtype=dtype)
    else:
        mask = mask_tensor.to(dtype)
    mask = mask.unsqueeze(-1)   # [bs, N, N, 1]
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # Gates (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Triton kernel for the O(N³) multiplicative update:
    #    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
    # -----------------------------------------------------------------
    # Rearrange to (batch*hidden_dim, N, N) for a single matmul per channel.
    left = left.permute(0, 3, 1, 2).contiguous()   # [bs, hidden_dim, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    B = bs * hidden_dim
    left_flat = left.view(B, N, N)
    right_flat = right.view(B, N, N)

    # Allocate output
    out_flat = torch.empty((B, N, N), dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B, N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [bs, N, N, hidden_dim]
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # Output layer normalization (hidden_dim) and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B.T for a batch of matrices.

    A : (BATCH, M, K)
    B : (BATCH, N, K)   # note the K‑dimension is the last one
    C : (BATCH, M, N)

    The kernel is a straightforward tiled mat‑mul using `tl.dot`.
    """
    batch_idx = tl.program_id(0)          # which (batch, channel) we are processing
    pid_m = tl.program_id(1)              # tile row index
    pid_n = tl.program_id(2)              # tile column index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_batch_ptr = A_ptr + batch_idx * stride_a_batch
    b_batch_ptr = B_ptr + batch_idx * stride_b_batch
    c_batch_ptr = C_ptr + batch_idx * stride_c_batch

    # masks for out‑of‑bounds rows / cols
    mask_m = offs_m[:, None] < M
    mask_n = offs_n[None, :] < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for cur_k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = cur_k * BLOCK_K

        # A tile (M × K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_row) \
                           + ((k_offset + offs_k)[None, :] * stride_a_col)

        # B tile transposed → shape (K × N)
        # original B is (N, K), we need (K, N) for the dot
        b_ptrs = b_batch_ptr + ((k_offset + offs_k)[:, None] * stride_b_col) \
                           + (offs_n[None, :] * stride_b_row)

        # per‑tile masks
        mask_a = mask_m & ((k_offset + offs_k)[None, :] < K)
        mask_b = ((k_offset + offs_k)[:, None] < K) & mask_n

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)   # accumulate in fp32

    # store the result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_c_row) + (offs_n[None, :] * stride_c_col)
    mask_c = mask_m & mask_n
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the Triton “outgoing” TriMul operator.
    The heavy ∑ₖ left[...,i,k,d]·right[...,j,k,d] reduction is performed
    by the `tri_mul_kernel` defined above.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [bs, N, N, dim]

    # -----------------------------------------------------------------
    # Linear projections (dim -> hidden_dim)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [bs, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Optional mask (broadcast on hidden_dim)
    # -----------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones(bs, N, N, device=device, dtype=dtype)
    else:
        mask = mask_tensor.to(dtype)
    mask = mask.unsqueeze(-1)   # [bs, N, N, 1]
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Triton kernel: out[...,i,j,d] = Σₖ left[...,i,k,d] * right[...,j,k,d]
    # -----------------------------------------------------------------
    # Move the hidden dimension to the batch axis so we can treat each
    # (batch, hidden_dim) pair as an independent matrix multiplication.
    left = left.permute(0, 3, 1, 2).contiguous()   # [bs, hidden_dim, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    batch_channels = bs * hidden_dim
    left_flat = left.view(batch_channels, N, N)    # (BCH, M, K)
    right_flat = right.view(batch_channels, N, N)  # (BCH, N, K)

    # Output buffer (float32)
    out_flat = torch.empty((batch_channels, N, N), dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        batch_channels,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_channels, N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [bs, N, N, hidden_dim]
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # Output LayerNorm (hidden_dim) + final linear projection (hidden_dim -> dim)
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate                     # element‑wise gated output
    out = torch.nn.functional.linear(out, to_out_weight)   # -> [bs, N, N, dim]

    return out
