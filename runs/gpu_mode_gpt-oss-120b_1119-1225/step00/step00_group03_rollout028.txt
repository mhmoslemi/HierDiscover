@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    stride_a_m, stride_a_k,
    stride_b_k, stride_b_n,
    stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr,
):
    pid = tl.program_id(0)  # batch * hidden_id
    # compute block offset in M,N dimension.
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # compute starting indices
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute pointer offsets base for this batch
    a_batch_ptr = a_ptr + pid * batch_stride_a
    b_batch_ptr = b_ptr + pid * batch_stride_b
    c_batch_ptr = c_ptr + pid * batch_stride_c

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute K offset
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load A tile (M,K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_a_m) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)

        # Load B tile (K,N)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k) + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        # dot in FP32
        acc += tl.dot(a, b)

        # advance pointers not needed because we recompute each iteration.

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_batch_a, stride_batch_b, stride_batch_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # batch*hidden dimension
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # offset calculations
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask inside M,N
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers for this batch slice
    a_batch = A_ptr + pid * stride_batch_a
    b_batch = B_ptr + pid * stride_batch_b
    c_batch = C_ptr + pid * stride_batch_c

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = a_batch + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile
        b_ptrs = b_batch + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
custom_kernel  –  TriMul “outgoing” forward pass (AlphaFold3)

Algorithm
---------
1. Layer‑normalize the input tensor x  (..., dim) using the provided LN weights.
2. Project x to the hidden dimension (hidden_dim) with four linear layers:
     * left_proj, right_proj   – linear projections   (bias‑free)
     * left_gate, right_gate   – gated with sigmoid
   Multiply the projections by their gates.
3. Optionally apply a pairwise mask (broadcasted on the hidden dim).
4. Fuse the contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for each (batch, hidden) pair:
        (N×N) @ (N×N)^T  → (N×N)
   The heavy N³ work is performed by a custom Triton kernel that
   computes many independent GEMMs (B·hidden) in parallel.
5. Apply the output LayerNorm, the output gate (sigmoid), and a final
   linear layer back to the original dimensionality (dim).

All tensors are kept on the GPU; the kernel runs in FP32 (the
intermediate accumulation is in float32).  The rest of the pipeline
uses torch.nn.functional calls for high‑throughput linear layers and
layer‑norms.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    stride_batch_a, stride_batch_b, stride_batch_c,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes many independent GEMMs:
        C[b,h] = A[b,h] @ B[b,h]   with A shape (M,K), B shape (K,N)
    The batch index (b*h) is encoded in program_id(0).
    """
    pid = tl.program_id(0)               # index = batch * hidden
    pid_m = tl.program_id(1)             # block row index (i)
    pid_n = tl.program_id(2)             # block col index (j)

    # Starting offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the beginning of the slice for this pid
    a_ptr = A_ptr + pid * stride_batch_a
    b_ptr = B_ptr + pid * stride_batch_b
    c_ptr = C_ptr + pid * stride_batch_c

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M × K)
        a_tile_ptrs = a_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_tile_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # Load B tile (K × N)
        b_tile_ptrs = b_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_tile_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float32)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)

    # Write back the result
    c_tile_ptrs = c_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_tile_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections + gates (flattened for fast GEMM)
    # ------------------------------------------------------------------
    x_flat = x.view(-1, dim)                         # (B*N*N, dim)

    # left branch
    left_proj = torch.nn.functional.linear(x_flat, left_proj_weight)   # (B*N*N, hidden)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_weight))
    left = (left_proj * left_gate).view(B, N, N, hidden_dim)

    # right branch
    right_proj = torch.nn.functional.linear(x_flat, right_proj_weight)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_weight))
    right = (right_proj * right_gate).view(B, N, N, hidden_dim)

    # out gate (used later)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_flat, out_gate_weight)
    ).view(B, N, N, hidden_dim)

    # Free temporary flattened tensor
    del x_flat

    # ------------------------------------------------------------------
    # 3) Optional mask application
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask shape -> [B, N, N, 1] broadcasting over hidden dim
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Batched GEMM (TriMul contraction) using Triton
    #    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    #   Rearrange tensors to (B, hidden, N, N) where the inner dims are
    #   (i,k) for left and (k,j) for right.
    left_A = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)   -> A(i,k)
    right_B = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)   -> B(k,j)
    del left, right

    # Output tensor in the same layout (B, H, N, N)
    out_perm = torch.empty_like(left_A)

    # Strides (in number of elements)
    stride_batch_a = left_A.stride(1)   # N*N
    stride_batch_b = right_B.stride(1)  # N*N
    stride_batch_c = out_perm.stride(1)  # N*N

    stride_am = left_A.stride(2)   # row stride   = N
    stride_ak = left_A.stride(3)   # col stride   = 1
    stride_bk = right_B.stride(2)  # K‑row stride = N
    stride_bn = right_B.stride(3)  # N‑col stride = 1
    stride_cm = out_perm.stride(2)  # row stride   = N
    stride_cn = out_perm.stride(3)  # col stride   = 1

    # Block sizes – tuned for H100 FP32 GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Launch configuration: (batch*hidden, ceil(N/BM), ceil(N/BN))
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_A,
        right_B,
        out_perm,
        stride_batch_a,
        stride_batch_b,
        stride_batch_c,
        stride_am,
        stride_ak,
        stride_bk,
        stride_bn,
        stride_cm,
        stride_cn,
        N,                 # M
        N,                 # N
        N,                 # K
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to (B, N, N, hidden)
    out = out_perm.permute(0, 2, 3, 1).contiguous()
    del out_perm, left_A, right_B

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate

    # Final linear: hidden → dim
    out = torch.nn.functional.linear(
        out.view(-1, hidden_dim), to_out_weight
    ).view(B, N, N, dim)

    return out
```
