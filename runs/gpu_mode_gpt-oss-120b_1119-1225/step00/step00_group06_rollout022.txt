def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   pid_m = tl.program_id(0) # i block
   pid_n = tl.program_id(1) # j block
   pid_hb = tl.program_id(2) # combined batch and hidden

   b = pid_hb // H
   h = pid_hb % H

   # offsets:
   offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

   # mask
   mask_i = offs_i < N
   mask_j = offs_j < N

   # Compute batch offset
   batch_offset = b * stride_l_batch

   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(N, BLOCK_K)):
       k_offset = k * BLOCK_K
       offs_k = k_offset + tl.arange(0, BLOCK_K)
       mask_k = offs_k < N

       # Load left tile: shape (BLOCK_M, BLOCK_K) for hidden slice h.
       left_ptrs = left_ptr + batch_offset + (offs_i[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k) + h * stride_l_h
       left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
                   stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
                   stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                   BLOCK_H: tl.constexpr):
    pid_i = tl.program_id(0) # i block
    pid_j = tl.program_id(1) # j block
    pid_hb = tl.program_id(2) # combined batch + hidden block

    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_hb // num_h_blocks
    hb = pid_hb % num_h_blocks
    h_base = hb * BLOCK_H

    # Offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)  # will be added inside loop
    offs_h = h_base + tl.arange(0, BLOCK_H)

    # Masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # Compute batch offset
    batch_off = b * stride_l_batch

    # Output pointer base
    out_ptrs_base = out_ptr + batch_off + (offs_i[:, None, None] * stride_o_i) + (offs_j[None, :, None] * stride_o_j) + (offs_h[None, None, :] * stride_o_h)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        cur_k = k_base + offs_k
        mask_k = cur_k < N

        # Left tile shape (BLOCK_M, BLOCK_K, BLOCK_H)
        left_ptrs = left_ptr + batch_off \
            + (offs_i[:, None, None] * stride_l_i) \
            + (cur_k[None, :, None] * stride_l_k) \
            + (offs_h[None, None, :] * stride_l_h)
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

        # Right tile shape (BLOCK_N, BLOCK_K, BLOCK_H)
        right_ptrs = right_ptr + batch_off \
            + (offs_j[:, None, None] * stride_r_j) \
            + (cur_k[None, :, None] * stride_r_k) \
            + (offs_h[None, None, :] * stride_r_h)
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

        # Compute dot per hidden slice
        # right shape (BLOCK_N, BLOCK_K, BLOCK_H). We need (BLOCK_K, BLOCK_N) per hidden slice.
        # Transpose k and n dims.
        # avail: tl.trans which swaps axes. We'll compute right_T = tl.trans(right, (1,0,2)) -> shape (BLOCK_K, BLOCK_N, BLOCK_H).
        right_T = tl.trans(right, (1, 0, 2))

        # For each hidden slice, accumulate
        for h in range(0, BLOCK_H):
            # Check if hidden channel is within range
            if mask_h[h] == False:
                continue
            a = left[:, :, h]   # (BLOCK_M, BLOCK_K)
            b_mat = right_T[:, :, h]  # (BLOCK_K, BLOCK_N)
            acc[:, :, h] += tl.dot(a, b_mat)
    # Store result
    # apply mask for i and j
    mask_ij = mask_i[:, None] & mask_j[None, :]
    store_mask = mask_ij[:, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs_base, acc, mask=store_mask)

"""
custom_kernel for the TriMul “outgoing” update (AlphaFold3).

Algorithm
---------
1. Layer‑Norm the input tensor `x` over the last dimension using the supplied
   `norm.weight` / `norm.bias`.
2. Linear projections (no bias):
      left  = x @ left_proj.weight   (→ hidden_dim)
      right = x @ right_proj.weight  (→ hidden_dim)
3. Apply the optional pairwise mask (shape [B,N,N]) by broadcasting on the
   hidden dimension.
4. Gating: compute three sigmoid‑gates from the normalized input:
      left_gate  = sigmoid(x @ left_gate.weight)
      right_gate = sigmoid(x @ right_gate.weight)
      out_gate   = sigmoid(x @ out_gate.weight)
   Multiply the projected tensors with their respective gates.
5. Core Tri‑Multiplicative update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix multiplication `left[...,h] @ right[...,h].T`
   for every hidden channel `h`.  It is implemented as a fused Triton kernel
   (blocked GEMM) that processes a tile of the `i`‑axis (BM), a tile of the
   `j`‑axis (BN) and a tile of the hidden dimension (BH).  The kernel works in
   FP16 for the loads (to reduce traffic) while accumulating in FP32.
6. Layer‑Norm the result over the hidden dimension with
   `to_out_norm.{weight,bias}`.
7. Apply the output gate (`out_gate`) and a final linear projection
   (`to_out.weight`) back to the original feature dimension.
8. Return the float‑32 output tensor of shape [B, N, N, dim].

The kernel launches a 3‑D grid:
    (ceil(N/BM), ceil(N/BN), B * ceil(hidden_dim/BH))
and is tuned for H100 GPUs (BLOCK_M=64, BLOCK_N=64, BLOCK_K=64,
BLOCK_H=4, 4 warps per CTA).
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the tri‑multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                # fp16 tensor [B,N,N,H]
    right_ptr,               # fp16 tensor [B,N,N,H]
    out_ptr,                 # fp32 tensor [B,N,N,H]
    B, N, H,                 # int: batch, seq_len, hidden_dim
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_j, stride_r_k, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)          # block on the i‑axis
    pid_j = tl.program_id(1)          # block on the j‑axis
    pid_hb = tl.program_id(2)         # combined (batch, hidden‑block)

    # --------------------------------------------------------------
    # Decode batch index and hidden‑block index
    # --------------------------------------------------------------
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_hb // num_h_blocks               # batch index
    hb = pid_hb % num_h_blocks                # hidden‑block index
    h_base = hb * BLOCK_H                     # first hidden channel in this block

    # --------------------------------------------------------------
    # Offsets for the i, j and k dimensions
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = h_base + tl.arange(0, BLOCK_H)

    # Masks for out‑of‑bounds positions
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # --------------------------------------------------------------
    # Base pointers (including batch offset)
    # --------------------------------------------------------------
    batch_off_l = b * stride_l_batch
    batch_off_r = b * stride_r_batch
    batch_off_o = b * stride_o_batch

    # Output pointer base (will be offset later per i/j/h)
    out_ptr_base = out_ptr + batch_off_o \
        + (offs_i[:, None, None] * stride_o_i) \
        + (offs_j[None, :, None] * stride_o_j) \
        + (offs_h[None, None, :] * stride_o_h)

    # --------------------------------------------------------------
    # Accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile -------------------------------------------------
        left_ptrs = left_ptr + batch_off_l \
            + (offs_i[:, None, None] * stride_l_i) \
            + (offs_k[None, :, None] * stride_l_k) \
            + (offs_h[None, None, :] * stride_l_h)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                       other=0.0,
                       dtype=tl.float16).to(tl.float32)      # (BM, BK, BH)

        # ----- load right tile (j,k) -----------------------------------------
        right_ptrs = right_ptr + batch_off_r \
            + (offs_j[:, None, None] * stride_r_j) \
            + (offs_k[None, :, None] * stride_r_k) \
            + (offs_h[None, None, :] * stride_r_h)
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                        other=0.0,
                        dtype=tl.float16).to(tl.float32)     # (BN, BK, BH)

        # ----- per‑hidden‑channel dot product ---------------------------------
        # right is (BN, BK, BH). For each h we need (BK, BN) for the dot.
        for h_off in range(0, BLOCK_H):
            a = left[:, :, h_off]                 # (BM, BK)
            b_mat = right[:, :, h_off]            # (BN, BK)
            b_T = tl.trans(b_mat)                  # (BK, BN)
            acc[:, :, h_off] += tl.dot(a, b_T)    # (BM, BN)

    # --------------------------------------------------------------
    # Store the result (apply i/j/h masks)
    # --------------------------------------------------------------
    mask_ij = mask_i[:, None] & mask_j[None, :]
    store_mask = mask_ij[:, :, None] & mask_h[None, None, :]
    tl.store(out_ptr_base, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Helper: LayerNorm (no running stats, inference only)
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5):
    # x: [..., C]; weight & bias: [C]
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm


# ----------------------------------------------------------------------
# Entry‑point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]):
    """
    Forward pass of the “outgoing” TriMul module using a fused Triton kernel for
    the O(N³) pairwise update.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    # config contains "dim" (input feature dim) and "hidden_dim" (projection dim)
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]               # = H in the code below
    eps = 1e-5

    # ------------------------------------------------------------------
    # Retrieve model weights (all are 2‑D tensors: out_features × in_features,
    # except LayerNorm weights which are 1‑D)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]             # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # Normalisation of the input
    # ------------------------------------------------------------------
    x_norm = _layer_norm(x, norm_weight, norm_bias, eps)   # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    # Using torch.nn.functional.linear for explicit weight usage
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional pairwise mask (mask may be all‑ones when nomask=True)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    mask_exp = mask.unsqueeze(-1).type_as(x)   # shape [B,N,N,1]
    left = left * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: compute out = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    B, N_seq, _, _ = left.shape
    # Cast to half‑precision for the kernel (reduces bandwidth)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (FP32 accumulation)
    out = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides (in elements, not bytes)
    stride_l = left_fp16.stride()   # (B, i, k, h)
    stride_r = right_fp16.stride()
    stride_o = out.stride()

    # Blocking parameters (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 4      # hidden‑dimension block size

    # Grid dimensions
    grid_i = (N_seq + BLOCK_M - 1) // BLOCK_M
    grid_j = (N_seq + BLOCK_N - 1) // BLOCK_N
    num_h_blocks = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid_hb = B * num_h_blocks

    # Launch the Triton kernel
    tri_mul_kernel[(grid_i, grid_j, grid_hb)](
        left_fp16,
        right_fp16,
        out,
        B,
        N_seq,
        hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, output gate, final linear
    # ------------------------------------------------------------------
    out_norm = _layer_norm(out, to_out_norm_weight, to_out_norm_bias, eps)
    out_norm = out_norm * out_gate                     # apply output gate
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # → [B,N,N,dim]

    return output
