@triton.jit
def matmul_triton_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # define masks for out of bounds
    m_mask = offs_m < M
    n_mask = offs_n < N
    
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # loop over k
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        k_mask = cur_k < K
        # load A
        a_ptrs = a_ptr + pid_batch * stride_a_batch + (offs_m[:, None] * stride_am) + (cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=m_mask[:, None] & k_mask[None, :], other=0.0)
        # load B (transposed)
        b_ptrs = b_ptr + pid_batch * stride_b_batch + (cur_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & n_mask[None, :], other=0.0)
        # dot product
        acc += tl.dot(a, b) # but dot expects inner dims: a shape (BLOCK_M, BLOCK_K), b shape (BLOCK_K, BLOCK_N)
    # store C
    c_ptrs = c_ptr + pid_batch * stride_c_batch + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=m_mask[:, None] & n_mask[None, :])

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype
    # normalise
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # layernorm over last dim of input (dim)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )
    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))
    # apply gates
    left = left * left_gate
    right = right * right_gate
    # apply mask if present
    if mask is not None and mask.numel() > 0:
        mask = mask.to(dtype=dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask
    # cast to half for compute
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    # permute to (B, hidden_dim, N, N)
    left = left.permute(0,3,1,2).contiguous()
    right = right.permute(0,3,1,2).contiguous()

    # flatten batch dimension for kernel
    left_mat = left.view(B*H, N, N)

                  M, N, K,
                  stride_am, stride_ak, stride_ab,
                  stride_bk, stride_bn, stride_bb,
                  stride_cm, stride_cn, stride_cb,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

"""
TriMul outgoing kernel (AlphaFold3 style) – forward only.

Algorithm
---------
1. Layer‑norm the input over the last channel dimension (dim).
2. Compute two linear projections (left/right) and three gating projections
   (left_gate, right_gate, out_gate) from the normalized tensor.
3. Apply sigmoid on the gating projections and element‑wise multiply:
       left  = left  * left_gate  * mask
       right = right * right_gate * mask
   (mask is optional, broadcasted on the hidden dimension).
4. Fuse the heavy O(N³·hidden_dim) computation with a custom Triton GEMM:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication of shape (batch*hidden_dim,
   N, N) = left @ rightᵀ.  The kernel works on FP16 inputs and accumulates
   in FP32 for numerical stability.
5. Reshape/reorder the result back to [B, N, N, hidden_dim].
6. Apply a second LayerNorm (hidden_dim), multiply by out_gate,
   and finish with a linear projection back to the original channel size.

The Triton part only covers the N³ term; all other operations are done in
PyTorch for clarity and to keep the implementation compact.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for C = A @ Bᵀ   (A: (M,K), B: (N,K) -> C: (M,N))
# Supports a batched dimension (program id 0) and works on FP16 inputs.
# ----------------------------------------------------------------------
@triton.jit
def matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix dimensions
    stride_am, stride_ak, stride_ab,  # A strides: (row, col, batch)
    stride_bk, stride_bn, stride_bb,  # B strides (treated as transposed)
    stride_cm, stride_cn, stride_cb,  # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)   # batch * hidden dim
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block col

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, K, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_ab
            + (offs_m[:, None] * stride_am)
            + (cur_k[None, :] * stride_ak)
        )
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load Bᵀ block: we treat B as transposed, so indices are (k, n)
        # Shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_bb
            + (cur_k[:, None] * stride_bk)      # k‑dimension stride (original column)
            + (offs_n[None, :] * stride_bn)     # n‑dimension stride (original row)
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Dot product (FP16) → cast to FP32 before accumulation
        acc += tl.cast(tl.dot(a, b), tl.float32)

    # Write out the result
    c_ptrs = (
        c_ptr
        + pid_batch * stride_cb
        + (offs_m[:, None] * stride_cm)
        + (offs_n[None, :] * stride_cn)
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Execute the TriMul outgoing operation using a hybrid PyTorch–Triton pipeline.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of torch.Tensors with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config       : dict with entries "dim", "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over channel dimension (`dim`)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections + gating
    # ------------------------------------------------------------------
    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])   # [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Gating logits → sigmoid
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    # Apply gates and (optional) mask
    if mask is not None:
        # ensure mask broadcasts over the hidden dimension
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * left_gate * mask_f
        right = right * right_gate * mask_f
    else:
        left = left * left_gate
        right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Heavy N³ computation – fused batched matmul via Triton
    #    Compute out = left @ rightᵀ for each hidden slice.
    # ------------------------------------------------------------------
    # Cast to FP16 for the kernel (speed / memory)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to (batch, hidden_dim, N, N) → then flatten batch*hidden
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape
    batch_h = B * H

    # (batch_h, N, N) matrices
    left_mat = left_perm.view(batch_h, N, N)
    right_mat = right_perm.view(batch_h, N, N)

    # Output container (FP32)
    out_mat = torch.empty_like(left_mat, dtype=torch.float32)

    # Strides (in elements) for the three tensors
    # A (left_mat) : (batch, M, K)
    stride_am = left_mat.stride(1)   # stride over M (row) = N
    stride_ak = left_mat.stride(2)   # stride over K (col) = 1
    stride_ab = left_mat.stride(0)   # stride over batch       = N * N

    # B (right_mat) is accessed as transposed:
    #   we need Bᵀ[k, n] = right_mat[n, k]
    stride_bk = right_mat.stride(2)  # original column stride = 1  (K‑dim for transposed)
    stride_bn = right_mat.stride(1)  # original row stride    = N  (N‑dim for transposed)
    stride_bb = right_mat.stride(0)  # batch stride          = N * N

    # C (out_mat) : (batch, M, N)
    stride_cm = out_mat.stride(1)    # row stride = N
    stride_cn = out_mat.stride(2)    # col stride = 1
    stride_cb = out_mat.stride(0)    # batch stride = N * N

    # Tile sizes – tuned for H100 fp16 matmul, but work for all configs
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch_h, num_blocks_M, num_blocks_N)
    grid = (
        batch_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    matmul_transpose_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N,               # M
        N,               # N
        N,               # K
        stride_am,
        stride_ak,
        stride_ab,
        stride_bk,
        stride_bn,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Second layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    output = torch.nn.functional.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return output
```
