import triton
import triton.language as tl

@triton.jit
def trimul_out_kernel(left_ptr, right_ptr, out_ptr,
                      B, N, H,
                      stride_l_b, stride_l_i, stride_l_k, stride_l_h,
                      stride_r_b, stride_r_j, stride_r_k, stride_r_h,
                      stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # compute batch and hidden index from program id 0
    pid = tl.program_id(0)
    total_h = B * H
    b = pid // H
    h = pid % H

    # i and j tile indices
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute row and column offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute offsets for left: (i,k)
        left_ptrs = left_ptr + b * stride_l_b + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k + h * stride_l_h
        # compute mask for left
        left_mask = mask_i[:, None] & mask_k[None, :]
        # load left tile
        a = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)  # cast to float32 for dot

        # compute offsets for right: (k, j) . Need shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b * stride_r_b + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j + h * stride_r_h
        right_mask = mask_k[:, None] & mask_j[None, :]
        b_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b_tile)  # a shape (BLOCK_M, BLOCK_K), b_tile shape (BLOCK_K, BLOCK_N)

    # store result
    out_ptrs = out_ptr + b * stride_o_b + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j + h * stride_o_h
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

"""
TriMul (outgoing) – Triton‑accelerated core matrix multiply.

The kernel computes, for each batch b and hidden channel h,

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where `left` and `right` have already been projected, masked and gated.
The reduction over the sequence dimension k is performed with a tiled
matrix‑multiply (BLOCK_M × BLOCK_N output tiles, BLOCK_K reduction tiles)
fully in Triton, while the surrounding LayerNorms, linear projections,
gates and the final output projection are done in PyTorch.  The kernel
operates on fp16 inputs and accumulates in fp32, returning a fp32
tensor of shape (B, N, N, H).

The implementation follows the layout of a contiguous tensor
[B, N, N, H] (last dimension is the fastest).  Strides are passed to the
kernel so that it works for any batch size, sequence length or hidden
dimension.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: tiled matmul (i,k) × (j,k)ᵀ → (i,j) for a single (b,h)
# ----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,           # *f16  [B,N,N,H]
    right_ptr,          # *f16  [B,N,N,H]
    out_ptr,            # *f32  [B,N,N,H]

    B: tl.int64,
    N: tl.int64,
    H: tl.int64,

    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    One program instance computes a (BLOCK_M × BLOCK_N) tile of the output
    for a specific (batch, hidden) pair.
    """
    # ------------------------------------------------------------------
    # 1) Identify batch b and hidden channel h from program id 0
    # ------------------------------------------------------------------
    pid = tl.program_id(0)               # 0 .. B*H-1
    b = pid // H                          # batch index
    h = pid % H                           # hidden channel index

    # ------------------------------------------------------------------
    # 2) Tile coordinates in the (i,j) plane (program id 1 & 2)
    # ------------------------------------------------------------------
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # 3) Accumulator (fp32) for the reduction over k
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) Loop over k in chunks of BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- left: shape (BLOCK_M, BLOCK_K) ---------------------------------
        left_offset = (
            b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        left_mask = mask_i[:, None] & mask_k[None, :]
        a = tl.load(left_ptr + left_offset,
                    mask=left_mask,
                    other=0.0,
                    dtype=tl.float16)

        # ---- right: shape (BLOCK_K, BLOCK_N) --------------------------------
        # we load (k, j) so that the second dimension is j → matches tl.dot
        right_offset = (
            b * stride_r_b
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
            + h * stride_r_h
        )
        right_mask = mask_k[:, None] & mask_j[None, :]
        b_tile = tl.load(right_ptr + right_offset,
                         mask=right_mask,
                         other=0.0,
                         dtype=tl.float16)

        # ---- accumulate -------------------------------------------------------
        acc += tl.dot(a, b_tile)   # a: (M,K), b_tile: (K,N) → (M,N)

    # ------------------------------------------------------------------
    # 5) Write back the result (fp32)
    # ------------------------------------------------------------------
    out_offset = (
        b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptr + out_offset, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (may be None)
        - weights      : dict of torch.Tensor containing all linear / LN weights
        - config       : dict with keys ["dim", "hidden_dim"]

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]   # called `hidden_dim` in the reference

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    # flatten for a single linear call
    x_flat = x.view(-1, dim)                     # (B*N*N, dim)

    # projections
    left_proj = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right_proj = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # gates (sigmoid)
    left_gate = F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    right_gate = F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()
    out_gate = F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim).sigmoid()

    # ------------------------------------------------------------------
    # 3) Apply mask (if any) and the gating
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape [B,N,N] → broadcast to the hidden dimension
        m = mask.unsqueeze(-1).to(left_proj.dtype)   # float32 -> broadcast
    else:
        # no mask means all‑ones
        m = torch.ones_like(left_proj, dtype=left_proj.dtype)

    left = left_proj * m * left_gate
    right = right_proj * m * right_gate

    # Cast to half for the Triton kernel (memory bandwidth)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Triton core: compute Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute strides (in elements) for the three tensors
    stride_l = left.stride()     # (B,N,N,hidden)
    stride_r = right.stride()
    stride_o = out.stride()

    grid = (
        B * hidden_dim,                     # program id 0  (batch * hidden)
        (N + BLOCK_M - 1) // BLOCK_M,       # program id 1  (i tiles)
        (N + BLOCK_N - 1) // BLOCK_N,       # program id 2  (j tiles)
    )

    trimul_out_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l[0],
        stride_l[1],
        stride_l[2],
        stride_l[3],
        stride_r[0],
        stride_r[1],
        stride_r[2],
        stride_r[3],
        stride_o[0],
        stride_o[1],
        stride_o[2],
        stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=3,
    )

    # ------------------------------------------------------------------
    # 5) Post‑processing: to_out_norm → gate → final linear
    # ------------------------------------------------------------------
    # LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )   # still [B,N,N,hidden_dim] float32

    # Multiply by out_gate (already sigmoid‑ed)
    out = out * out_gate

    # Final projection back to `dim`
    out_flat = out.view(-1, hidden_dim)
    out = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, dim)

    return out
```

This implementation:

* Performs the heavy N³ reduction (`einsum('... i k d, ... j k d -> ... i j d')`) using a custom Triton kernel that tiles the `(i,j)` output plane and reduces over `k`.  
* Keeps the rest of the model (LayerNorms, linear projections, sigmoid gates) in PyTorch for simplicity and correctness.  
* Supports both masked and un‑masked inputs, arbitrary batch sizes (1‑2), sequence lengths up to 1024, and hidden dimensions up to 128.  
* Returns a float‑32 tensor of shape `[B, N, N, dim]` as required.
