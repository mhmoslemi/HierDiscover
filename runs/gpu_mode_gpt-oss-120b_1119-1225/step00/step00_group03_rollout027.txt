@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # get program ids
    pid_m = tl.program_id(0)  # block row index
    pid_n = tl.program_id(1)  # block col index
    pid_d = tl.program_id(2)  # batch*hidden index

    # compute batch and hidden indices
    b = pid_d // H
    h = pid_d % H

    # compute start pointer offsets
    # base pointer includes batch offset and hidden offset
    # compute sizes
    stride_b = N * N * H  # stride for batch
    stride_i = N * H
    stride_k = H
    stride_h = 1

    # pointer to start for given b and h
    left_base = left_ptr + b * stride_b + h * stride_h
    right_base = right_ptr + b * stride_b + h * stride_h
    out_base = out_ptr + b * stride_b + h * stride_h

    # compute block start indices in M & N dims
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

left_base = left_ptr + offset_batch + offset_hidden
right_base = right_ptr + offset_batch + offset_hidden
out_base = out_ptr + offset_batch + offset_hidden

stride_k = H

j_start = pid_n * BLOCK_N

offs_j = j_start + tl.arange(0, BLOCK_N)

mask_i = offs_i < N
mask_j = offs_j < N

acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col
    pid_h = tl.program_id(2)   # (batch, hidden) pair

    # decode batch and hidden index
    b = pid_h // H
    h = pid_h % H

    # strides in elements (contiguous layout)
    stride_b = N * N * H
    stride_i = N * H
    stride_k = H
    stride_h = 1

    # base pointer for this (b, h)
    left_base = left_ptr + b * stride_b + h * stride_h
    right_base = right_ptr + b * stride_b + h * stride_h
    out_base = out_ptr + b * stride_b + h * stride_h

    # start offsets for this tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: rows i, cols k
        left_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # right transposed: rows k, cols j
        right_ptrs = right_base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float16)

        # perform matmul tile (M x K) @ (K x N) -> (M x N), accumulate in FP32
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # write output tile
    out_ptrs = out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – fast Triton implementation
================================================

The TriMul module from AlphaFold‑3 performs the following operations
on a tensor `x` of shape [B, N, N, D]:

    1. LayerNorm over the last dimension (D).
    2. Linear projections `left = W_left  @ x` and
       `right = W_right @ x`  (producing a hidden dimension H).
    3. Optional binary mask M ∈ {0,1}^{B,N,N} applied to both projections.
    4. Gated projections:
           left  = left  * sigmoid(W_left_gate  @ x)
           right = right * sigmoid(W_right_gate @ x)
    5. **Core heavy kernel** – for every batch element b and hidden channel h
       compute the matrix product

           out[b, :, :, h] = left[b, :, :, h] @ right[b, :, :, h].T

       which is the same as the einsum
           `'... i k d, ... j k d -> ... i j d'`.

       This step is implemented in Triton.  The kernel treats each (b,h)
       pair as an independent matrix‑multiply (size N×N) and tiles the
       computation with configurable block sizes (M, N, K).  Input tensors
       are cast to float16 to exploit Tensor‑Core matmul, while the
       accumulator is float32 for full precision.  The kernel writes the
       result in FP32.

    6. LayerNorm over the hidden dimension H, element‑wise gating with
       `out_gate = sigmoid(W_out_gate @ x)`, and final linear projection
       back to the original dimension D.

The surrounding LayerNorms / Linear layers are executed with regular
PyTorch (they are cheap compared to the N³ reduction).  The implementation
supports both masked and un‑masked inputs and works for the full range of
shapes required by the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # const float*  (B,N,N,H)  – half‑precision data
    right_ptr,              # const float*  (B,N,N,H)  – half‑precision data
    out_ptr,                #       float*  (B,N,N,H)  – result (FP32)
    B, N, H,                # runtime sizes
    BLOCK_M: tl.constexpr,  # tile size along the output row dimension i
    BLOCK_N: tl.constexpr,  # tile size along the output col dimension j
    BLOCK_K: tl.constexpr,  # tile size for the reduction over k
):
    """Batched matmul `out = left @ right.T` for every hidden channel."""
    pid_m = tl.program_id(0)          # block row index (i)
    pid_n = tl.program_id(1)          # block col index (j)
    pid_h = tl.program_id(2)          # combined batch‑hidden index

    # decode batch and hidden channel from the 3‑rd program id
    b = pid_h // H
    h = pid_h % H

    # contiguous layout strides (in elements, not bytes)
    stride_b = N * N * H                # jump to next batch
    stride_i = N * H                    # jump to next i‑row
    stride_k = H                        # jump to next k‑col
    stride_h = 1                        # hidden dimension is innermost

    # base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + b * stride_b + h * stride_h
    right_base = right_ptr + b * stride_b + h * stride_h
    out_base   = out_ptr   + b * stride_b + h * stride_h

    # start indices of the current tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K) – rows i, cols k
        left_ptrs = left_base + offs_i[:, None] * stride_i + offs_k[None, :] * stride_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N) – rows k, cols j
        right_ptrs = right_base + offs_k[:, None] * stride_k + offs_j[None, :] * stride_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # matrix multiply tile (M×K)·(K×N) → (M×N), accumulate in FP32
        acc += tl.dot(left, right, out_dtype=tl.float32)

    # write the tile to global memory
    out_ptrs = out_base + offs_i[:, None] * stride_i + offs_j[None, :] * stride_k
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])

    out_gate_weight       = weights["out_gate.weight"]
    to_out_norm_weight    = weights["to_out_norm.weight"]
    to_out_norm_bias      = weights["to_out_norm.bias"]
    to_out_weight         = weights["to_out.weight"]

    # ----------------------------------------------------------------------
    # 1. LayerNorm over the input channel dimension D
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # 2. Linear projections to hidden dimension H
    left  = F.linear(x, left_proj_weight)    # (B,N,N,H)
    right = F.linear(x, right_proj_weight)

    # 3. Optional binary mask M ∈ {0,1}
    if not config.get("nomask", False):
        mask_f = mask.to(dtype).unsqueeze(-1)        # (B,N,N,1)
        left  = left * mask_f
        right = right * mask_f

    # 4. Gating (sigmoid of linear projections)
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Triton kernel – compute the summed outer‑product over k
    left  = left.contiguous()
    right = right.contiguous()

    # cast to FP16 to make use of Tensor‑Core matmul
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # output buffer (FP32)
    out = torch.empty_like(left_h, dtype=torch.float32)

    # tiling parameters – chosen to fit H100 shared‑memory budget
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (seq_len + BLOCK_M - 1) // BLOCK_M,   # blocks along M (i)
        (seq_len + BLOCK_N - 1) // BLOCK_N,   # blocks along N (j)
        batch_size * hidden_dim               # one program for each (b, h)
    )

    tri_mul_kernel[grid](
        left_h, right_h, out,
        batch_size, seq_len, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ----------------------------------------------------------------------
    # 6. LayerNorm over hidden dimension, gating, final linear projection
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                     # element‑wise gating
    out = F.linear(out, to_out_weight)       # back to original dimension D

    return out
```
