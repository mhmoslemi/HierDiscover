def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # compute flag for mask
    if mask_tensor is None:
        mask = None
    else:
        mask = mask_tensor
    # layernorm on input
    # compute mean and var per (b,i,j)
    # Using torch.nn.functional.layer_norm:
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask
    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # Compute out via TriMul kernel
    B, N, _, _ = left.shape
    # allocate out hidden tensor
    out_hidden = torch.empty((B, N, N, hidden_dim), device=input_tensor.device, dtype=torch.float32)
    # Determine strides
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out_hidden.stride()
    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B * hidden_dim )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l[0],
        stride_l_i=stride_l[1],
        stride_l_k=stride_l[2],
        stride_l_h=stride_l[3],
        stride_r_b=stride_r[0],
        stride_r_i=stride_r[1],
        stride_r_k=stride_r[2],
        stride_r_h=stride_r[3],
        stride_o_b=stride_o[0],
        stride_o_i=stride_o[1],
        stride_o_j=stride_o[2],
        stride_o_h=stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # now out_hidden shape B,N,N,H
    # to_out_norm
    out_normed = torch.nn.functional.layer_norm(out_hidden, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # multiply with out_gate
    out_normed = out_normed * out_gate
    # final projection to original dim
    out = torch.nn.functional.linear(out_normed, weights["to_out.weight"])

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)
    # compute which batch and hidden channel
    b = pid_z // H
    h = pid_z % H
    # starting offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # base pointers
    left_base = left_ptr + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    # accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        # compute masks
        mask_i = i_offset < N
        mask_j = j_offset < N
        mask_k = k_offset < N
        # pointers for left and right.
        # left: [i,k]
        left_ptrs = left_base + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        # right: we want right[j,k] -> but we need transposed for dot. 
        # We'll load right with shape (BLOCK_K, BLOCK_N): index by k then j.
        right_ptrs = right_base + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_i
        # load values
        left_vals = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right_vals = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

    # store output
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h + i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j
    mask_i = i_offset < N
    mask_j = j_offset < N

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # x_norm = layernorm
    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # left, right projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])
    # mask

        # mask shape [B,N,N], unsqueeze

        right = right * mask
    # gates

    left = left * left_gate

TriMul forward (outgoing) implementation.

The module performs:
1. LayerNorm over the last channel dimension.
2. Two linear projections (left/right) and three gating projections.
3. Optional masking of the projected tensors.
4. Element‑wise gating of left/right.
5. A triangular multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a blocked matrix‑multiply per hidden channel using a
   custom Triton kernel (left @ rightᵀ).
6. LayerNorm over the hidden channel, gating, and final linear projection.

Only the reduction step (step 5) is executed in Triton; all other operations
use PyTorch for simplicity. The kernel processes each (batch, hidden) pair
independently, using a 2‑D tiling over the (i, j) output dimensions and a
blocked reduction over k.
"""

import triton.language as tl

# Triton kernel for the core triangular multiplicative update.
# ----------------------------------------------------------------------
@triton.jit

    left_ptr,               # const float*  left  [B, N, N, H]
    right_ptr,              # const float*  right [B, N, N, H]
    out_ptr,                # float*        out   [B, N, N, H]

    N: tl.int32,            # sequence length
    H: tl.int32,            # hidden dimension

    stride_l_b: tl.int64,
    stride_l_i: tl.int64,
    stride_l_k: tl.int64,
    stride_l_h: tl.int64,
    # strides for right (B, j, k, h) – we will read as (k, j) for dot product
    stride_r_b: tl.int64,
    stride_r_i: tl.int64,
    stride_r_k: tl.int64,
    stride_r_h: tl.int64,
    # strides for out (B, i, j, h)
    stride_o_b: tl.int64,
    stride_o_i: tl.int64,
    stride_o_j: tl.int64,
    stride_o_h: tl.int64,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1. Program IDs: (i_tile, j_tile, batch*hidden)
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_z = tl.program_id(2)          # combined batch & hidden index

    # decode batch and hidden channel
    b = pid_z // H
    h = pid_z % H

    # ------------------------------------------------------------------
    # 2. Compute the absolute indices for the tile
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for the given batch and hidden channel
    left_base  = left_ptr  + b * stride_l_b + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3. Loop over K dimension in blocks
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------- load left[i, k]  (shape BLOCK_M x BLOCK_K) ----------
        left_ptrs = (left_base
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # ------- load right[j, k] transposed to (BLOCK_K x BLOCK_N) ----
        # We need right[k, j] for the dot product.
        right_ptrs = (right_base
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_i)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # ------- block‑wise dot product over K -------------------------
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # 4. Write the result back to global memory
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + h * stride_o_h
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0. unpack arguments
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]               # input channel dimension (C)
    hidden_dim = config["hidden_dim"] # hidden dimension (H)

    # --------------------------------------------------------------
    # 1. LayerNorm over the input channels
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------
    # 2. Linear projections (left / right) – no bias
    # --------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B,N,N,H]

    # --------------------------------------------------------------
    # 3. Optional mask (mask shape [B,N,N] or None)
    # --------------------------------------------------------------
    if mask is not None:
        # ensure mask is float and broadcastable
        mask = mask.to(x_norm.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating projections
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Triangular multiplicative update (core O(N³) reduction)
    #    Implemented with a blocked Triton GEMM per hidden channel.
    # --------------------------------------------------------------
    B, N, _, _ = left.shape   # left.shape == (B, N, N, H)

    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # Allocate output for the hidden representation
    out_hidden = torch.empty(
        (B, N, N, hidden_dim), dtype=left.dtype, device=left.device
    )

    # Strides (int64) – note PyTorch uses element strides
    stride_l = left.stride()
    stride_r = right.stride()
    stride_o = out_hidden.stride()

    # Tile sizes – chosen to give good occupancy on an H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,    # program IDs for i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,    # program IDs for j‑tiles
        B * hidden_dim,                  # program IDs for (batch, hidden) pairs
    )

    _trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l[0],
        stride_l_i=stride_l[1],
        stride_l_k=stride_l[2],
        stride_l_h=stride_l[3],
        stride_r_b=stride_r[0],
        stride_r_i=stride_r[1],
        stride_r_k=stride_r[2],
        stride_r_h=stride_r[3],
        stride_o_b=stride_o[0],
        stride_o_i=stride_o[1],
        stride_o_j=stride_o[2],
        stride_o_h=stride_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6. Second LayerNorm over hidden channels
    # --------------------------------------------------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 7. Apply out‑gate and final linear projection back to dim
    # --------------------------------------------------------------
    out_hidden = out_hidden * out_gate
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
