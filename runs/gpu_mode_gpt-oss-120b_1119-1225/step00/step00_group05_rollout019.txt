    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    pid_i = tl.program_id(0)  # block i index
    pid_j = tl.program_id(1)  # block j index

    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    # masks for OOB
    mask_i = i_offset < N
    mask_j = j_offset < N

        # accumulator per hidden slice shape (BLOCK_I, BLOCK_J, cur_BLOCK_H)

        # But we can store for each hidden channel individually with loop: acc_h = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # We'll loop over hidden channels within the block just like h_sub loop.
        for h_sub in range(cur_BLOCK_H):

                mask_k = k_offset < N

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # optional: ensure contiguous
    input_tensor = input_tensor.contiguous()
    mask_tensor = mask_tensor.contiguous()
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # convert mask to same dtype, unsqueeze
    mask = mask_tensor
    if mask is None or mask.numel() == 0 or config.get("nomask", False):
        mask = torch.ones((B,N,N), dtype=input_tensor.dtype, device=input_tensor.device)
    mask = mask.unsqueeze(-1)  # shape B,N,N,1

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask
    left = left * mask
    right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Compute pairwise sum via Triton
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Get strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Kernel launch params
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * hidden_dim,
    )
    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # Post-process
    # LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

"""tri_mul_outgoing kernel (AlphaFold3 TriMul)

Implements the forward pass of the “outgoing” TriMul operation:

    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]

where
    left  = Linear(x)   * mask * sigmoid(Linear(x))
    right = Linear(x)   * mask * sigmoid(Linear(x))

The heavy O(N³·H) contraction (N = sequence length, H = hidden_dim)
is performed by a custom Triton kernel that computes a batched matrix
multiply for each hidden channel (B × H independent thread blocks).
All other linear / LayerNorm / gating operations are done with PyTorch.

The kernel works for any batch size (1‑2), any sequence length up to 1024,
and hidden dimensions up to 128 (the maximum used in the tests).  The
output is always float‑32.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel for the core contraction:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,            # *float32  left  [B,N,N,H]
    right_ptr,           # *float32  right [B,N,N,H]
    out_ptr,             # *float32  out   [B,N,N,H]
    B, N, H,             # runtime dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # --------------------- compile‑time tile sizes ---------------------
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_i = tl.program_id(0)          # tile along i (rows)
    pid_j = tl.program_id(1)          # tile along j (cols)
    pid_z = tl.program_id(2)          # combined (batch, hidden)

    # decode batch index and hidden‑channel index
    b = pid_z // H
    h = pid_z % H

    # offsets for the i/j tiles
    i_offsets = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offsets = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # masks for out‑of‑bounds rows/cols
    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # accumulator for the (i,j) tile (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in BLOCK_K chunks
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_offsets = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # ---- load left block: shape (BLOCK_I, BLOCK_K) ----------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_offsets[:, None] * stride_left_i
            + k_offsets[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- load right block transposed:
        #      we want a (BLOCK_K, BLOCK_J) view so that tl.dot(left, right_T)
        #      yields the desired Σ_k left[i,k] * right[j,k] ----------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + k_offsets[:, None] * stride_right_k
            + j_offsets[None, :] * stride_right_j
            + h * stride_right_h
        )
        right_T = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- accumulate the dot product -------------------------------------------------
        acc += tl.dot(left, right_T)      # (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_offsets[:, None] * stride_out_i
        + j_offsets[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# --------------------------------------------------------------
# Python entry point ------------------------------------------------
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            - input:   Tensor[B, N, N, D] (float32)
            - mask:    Tensor[B, N, N]   (bool/float, may be all‑ones)
            - weights: dict of model parameters
            - config:  dict containing `dim` and `hidden_dim`
    Returns:
        Tensor[B, N, N, D] = result of the TriMul forward
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    input_tensor = input_tensor.contiguous()
    mask_tensor = mask_tensor.contiguous()

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]           # always 128 in the tests

    # ------------------------------------------------------------------
    # Optional mask handling (nomask → all‑ones)
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask_tensor.numel() == 0:
        mask = torch.ones((B, N, N), dtype=input_tensor.dtype, device=input_tensor.device)
    else:
        mask = mask_tensor.to(input_tensor.dtype)
    mask = mask.unsqueeze(-1)                   # [B, N, N, 1]

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim = last axis)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply mask (if any)
    # ------------------------------------------------------------------
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of separate linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core contraction via Triton
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # strides (in elements, not bytes)
    stride_left = left.stride()    # (B, i, k, h)
    stride_right = right.stride()
    stride_out = out.stride()

    # tile sizes – chosen to fit H100 shared memory and occupancy
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,            # i dimension
        (N + BLOCK_J - 1) // BLOCK_J,            # j dimension
        B * hidden_dim,                          # batch × hidden channel
    )

    trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
