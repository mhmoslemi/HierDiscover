def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; bs = input_tensor.shape[0]; N = input_tensor.shape[1]

    # LayerNorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]; 
    # epsilon 1e-5 typical
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    else:
        # mask may be None? Provided as something else maybe all ones
        pass

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare for kernel: flatten batch and hidden dim
    # left: shape (B, N, N, hidden_dim) -> permute (B, hidden_dim, N, N) -> reshape (B*hidden_dim, N, N)

"""
TriMul (outgoing) – fast Triton implementation of the heavy contraction.

The module computes the following operations (matching the reference PyTorch code):
1. Layer‑norm on the input tensor (last dimension = dim).
2. Linear projections to a hidden dimension `hidden_dim`:
        left  = x @ left_proj.weight.T
        right = x @ right_proj.weight.T
3. Optional pair‑wise mask multiplication.
4. Gated activations:
        left  = left  * sigmoid(x @ left_gate.weight.T)
        right = right * sigmoid(x @ right_gate.weight.T)
5. **Core contraction** – compute for each batch and hidden channel :
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply `A @ Bᵀ` where each hidden channel is
   treated as an independent batch.  The contraction is performed in half‑precision
   with a custom Triton kernel (`trmul_matmul_kernel`) that fuses the K‑reduction
   (dot‑product) and accumulation.
6. Layer‑norm over the hidden dimension, multiplied by an output gate,
   and a final linear projection back to `dim`.

Only the O(N³·hidden_dim) contraction is off‑loaded to Triton; all other
operations (LayerNorms, linear layers, gating) use standard PyTorch calls.
The kernel works for any batch size (1–2), sequence length (≤1024) and hidden
dimension (≤384) and returns a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for `out[..., i, j, d] = Σ_k left[..., i, k, d] *
#                                                            right[..., j, k, d]`
# ----------------------------------------------------------------------
@triton.jit
def trmul_matmul_kernel(
    a_ptr,                     # left  matrix pointer   (batch, M, K) – contiguous
    b_ptr,                     # right matrix pointer   (batch, N, K) – contiguous
    c_ptr,                     # output pointer         (batch, M, N)
    M, N, K,                   # matrix dimensions (int64 scalars)
    stride_am, stride_ak,      # strides for A (M and K dimensions)
    stride_bk, stride_bn,      # strides for B (K and N dimensions)
    stride_cm, stride_cn,      # strides for C (M and N dimensions)
    BLOCK_M: tl.constexpr,     # tile size in M dimension
    BLOCK_N: tl.constexpr,     # tile size in N dimension
    BLOCK_K: tl.constexpr,     # tile size in K dimension
):
    """
    Compute C = A @ Bᵀ for many independent batches.
    A: (batch, M, K)   (float16)
    B: (batch, N, K)   (float16) – we read it as (K, N) to get the transpose.
    C: (batch, M, N)   (float16)
    """
    # -------------------------------------------------------------
    # Program IDs – one 3‑D grid: (block_m, block_n, batch)
    # -------------------------------------------------------------
    pid_m = tl.program_id(0)          # block index in M dimension
    pid_n = tl.program_id(1)          # block index in N dimension
    pid_b = tl.program_id(2)          # batch index ( = original batch * hidden_dim )

    # -------------------------------------------------------------
    # Offsets for the current block
    # -------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    # Batch‑level offsets in memory
    a_batch_offset = pid_b * M * K
    b_batch_offset = pid_b * N * K
    c_batch_offset = pid_b * M * N

    # Base pointers that already contain the M / N offsets.
    a_base = a_ptr + a_batch_offset + offs_m[:, None] * stride_am
    b_base = b_ptr + b_batch_offset + offs_n[None, :] * stride_bn

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # -------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_iter * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)          # [BLOCK_K]

        # Load A[M, K] tile
        a = tl.load(
            a_base + offs_k[None, :] * stride_ak,
            mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
            other=0.0,
        )
        # Load Bᵀ[K, N] tile (B is stored as (N, K); we read it transposed)
        b = tl.load(
            b_base + offs_k[:, None] * stride_bk,
            mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        # Accumulate dot product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) => (BLOCK_M, BLOCK_N)

    # -------------------------------------------------------------
    # Write the result
    # -------------------------------------------------------------
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(offs_m < M)[:, None] & (offs_n < N)[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel for the TriMul contraction
# ----------------------------------------------------------------------
def _trmul_batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right : torch.float16 tensors of shape (B*hidden_dim, N, N)
    Returns torch.float16 tensor of shape (B*hidden_dim, N, N) containing
    left @ rightᵀ for each (batch*hidden_dim) entry.
    """
    assert left.is_contiguous() and right.is_contiguous()
    batch_flat, N, K = left.shape
    assert K == N, "TriMul expects square pairwise tensors."

    out = torch.empty((batch_flat, N, N), device=left.device, dtype=torch.float16)

    # Block sizes – tuned for H100 (FP16 matmul)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N
        batch_flat,                     # one block per (batch * hidden_dim)
    )

    trmul_matmul_kernel[grid](
        left,
        right,
        out,
        N,                # M
        N,                # N
        N,                # K
        stride_am=left.stride(1),      # stride for M dimension in A
        stride_ak=left.stride(2),      # stride for K dimension in A
        stride_bk=right.stride(2),     # stride for K dimension in B (transposed view)
        stride_bn=right.stride(1),     # stride for N dimension in B (transposed view)
        stride_cm=out.stride(1),       # stride for M dimension in C
        stride_cn=out.stride(2),       # stride for N dimension in C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Arguments:
        data – tuple (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA).
    """
    # -------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------
    # Load model weights
    # -------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # -------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (dim)
    # -------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, dim]

    # -------------------------------------------------------------
    # 2️⃣ Linear projections to hidden_dim
    # -------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # -------------------------------------------------------------
    # 3️⃣ Optional masking
    # -------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] for broadcasting
        m = mask.to(x.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # -------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of a linear projection of the *original* x)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5️⃣ Heavy contraction – performed in half‑precision on GPU
    # -------------------------------------------------------------
    # Cast to FP16 for speed
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape                                     # batch size, seq length
    batch_flat = B * hidden_dim                                   # combined batch index

    # Rearrange to (batch_flat, N, N) where each hidden channel is a separate batch
    # left: (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch_flat, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch_flat, N, N)

    # Triton kernel: out = left_flat @ right_flatᵀ   (per hidden channel)
    contracted = _trmul_batched_matmul(left_flat, right_flat)     # [B*H, N, N], fp16

    # Reshape back to (B, N, N, hidden_dim)
    contracted = (
        contracted.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )   # [B, N, N, hidden_dim]

    # -------------------------------------------------------------
    # 6️⃣ LayerNorm over hidden_dim, apply output gate
    # -------------------------------------------------------------
    out_pre = F.layer_norm(
        contracted,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out_pre = out_pre * out_gate

    # -------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # -------------------------------------------------------------
    out = F.linear(out_pre, to_out_w)  # [B, N, N, dim]

    return out
