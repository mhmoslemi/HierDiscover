@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # Compute starting offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load A and B tiles
        a_ptr = A_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptr = B_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # Use mask to avoid out-of-bounds
        a = tl.load(a_ptr, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptr, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)

    # Store result
    c_ptr = C_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptr, acc.to(tl.float16) if False else acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])
    # Actually final output should be float32; we will store as float32.

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program id
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load a and b tiles
        a_ptrs = a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0, dtype=tl.float16)

        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def batched_gemm_kernel(

    M, N, K,

    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision GEMM (A @ Bᵀ) for the TriMul core.
# Input tensors have shape (batch*hidden, N, N) and are stored in
# BF16/FP16.  The accumulator is FP32 to avoid overflow.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # 0‑th axis = (batch * hidden) dimension, 1‑st = rows, 2‑nd = cols
    batch_id = tl.program_id(0)
    pid_m    = tl.program_id(1)
    pid_n    = tl.program_id(2)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load A tile (shape BLOCK_M × BLOCK_K)
        a_ptrs = a_ptr + batch_id * stride_a_batch \
                           + offs_m[:, None] * stride_a_m \
                           + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Load B tile (shape BLOCK_K × BLOCK_N).  B is already transposed
        # (rows = k, cols = n).
        b_ptrs = b_ptr + batch_id * stride_b_batch \
                           + offs_k[:, None] * stride_b_k \
                           + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Rank‑K update
        acc += tl.dot(a, b)          # FP32 accumulation

    # Store the result tile
    c_ptrs = c_ptr + batch_id * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    1. Layer‑norm the [B, N, N, C] input.
    2. Linear projections (left/right) and sigmoid gates (left/right/out).
    3. Apply mask (if present) and gate multiplicatively.
    4. Compute the heavy 3‑D contraction:
           out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       using a batched half‑precision GEMM kernel (left @ rightᵀ).
    5. Layer‑norm the intermediate tensor, apply out‑gate, and a final
       linear projection back to the original channel dimension.
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5

    # --------------------------------------------------------------
    # Load weights
    # --------------------------------------------------------------
    norm_weight        = weights["norm.weight"]
    norm_bias          = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Optional pairwise mask
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N] -> broadcast to hidden_dim
        m = mask.unsqueeze(-1).type_as(x)    # same dtype as x (float32)
        left  = left  * m
        right = right * m

    # --------------------------------------------------------------
    # 4. Gates (sigmoid activations)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Cast to half‑precision for the cubic contraction
    # --------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Bring the hidden dimension to the batch axis
    # left_perm shape:  [B, hidden_dim, N, N]
    # right_perm shape: [B, hidden_dim, N, N] where the last two axes are (k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose k ↔ j

    B, H, N, _ = left_perm.shape

    # Collapse batch and hidden dimensions → independent GEMMs
    left_flat  = left_perm.view(-1, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N)
    out_flat   = torch.empty_like(left_flat, dtype=torch.float16)   # (B*H, N, N)

    # --------------------------------------------------------------
    # 6. Triton batched GEMM: out = left @ rightᵀ
    # --------------------------------------------------------------
    BLOCK_M = 128 if N >= 128 else 64
    BLOCK_N = 128 if N >= 128 else 64
    BLOCK_K = 32

    batch_dim = left_flat.shape[0]                     # B * hidden_dim
    grid = (batch_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Strides in element units
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                      # M, N, K (all equal to sequence length)
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 7. Reshape back to [B, N, N, hidden_dim] and post‑process
    # --------------------------------------------------------------
    out = out_flat.to(torch.float32)                              # FP16 → FP32
    out = out.view(B, H, N, N).permute(0, 2, 3, 1)                # [B, N, N, hidden_dim]

    # Post‑processing LayerNorm + out‑gate
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                                          # broadcast over hidden dim

    # Final linear projection back to original channel dimension
    out = F.linear(out, to_out_weight)                            # [B, N, N, dim]
