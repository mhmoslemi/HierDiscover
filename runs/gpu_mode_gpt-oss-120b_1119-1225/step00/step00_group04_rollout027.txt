@triton.jit
def trmul_kernel(left_ptr, right_ptr, out_ptr,
                 B, N, H,
                 stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                 stride_right_b, stride_right_j, stride_right_k, stride_right_h,
                 stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                 BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, VEC: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over hidden dim tiles
    for h_tile in range(0, tl.cdiv(H, VEC)):
        h_start = h_tile * VEC
        offs_h = h_start + tl.arange(0, VEC)
        mask_h = offs_h < H

        # Accumulator init
        acc = tl.zeros((BLOCK_M, BLOCK_N, VEC), dtype=tl.float32)

        # Loop over K dimension tiles
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            k_start = k_tile * BLOCK_K
            offs_k = k_start + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left tile (M, K, V)
            left_ptrs = left_ptr + pid_b * stride_left_b \
                        + offs_i[:, None, None] * stride_left_i \
                        + offs_k[None, :, None] * stride_left_k \
                        + offs_h[None, None, :] * stride_left_h
            left = tl.load(left_ptrs,
                           mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                           other=0.0)

            # Load right tile (N, K, V)
            right_ptrs = right_ptr + pid_b * stride_right_b \
                        + offs_j[:, None, None] * stride_right_j \
                        + offs_k[None, :, None] * stride_right_k \
                        + offs_h[None, None, :] * stride_right_h
            right = tl.load(right_ptrs,
                            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                            other=0.0)

            # Reduce over K dimension (inner loop over kk)
            for kk in range(0, BLOCK_K):
                # Note: When nk <= N, mask_k ensures that out-of-range k tiles produce zeros.
                left_kk = left[:, kk, :]   # (M, V)
                right_kk = right[:, kk, :] # (N, V)
                # broadcast multiply and accumulate
                acc += left_kk[:, None, :] * right_kk[None, :, :]

        # Store result for this hidden tile
        out_ptrs = out_ptr + pid_b * stride_out_b \
                   + offs_i[:, None, None] * stride_out_i \
                   + offs_j[None, :, None] * stride_out_j \
                   + offs_h[None, None, :] * stride_out_h
        tl.store(out_ptrs,
                 acc,
                 mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
custom_kernel
==============

Implements the *outgoing* TriMul operation from AlphaFold‑3.

Algorithm
---------
1. **Layer‑norm** over the last channel dimension of the input.
2. **Linear projections** (`left_proj`, `right_proj`) and three gating
   linear layers (`left_gate`, `right_gate`, `out_gate`).
3. Optional pair‑wise mask multiplication.
4. Apply the three sigmoid gates to the projected tensors.
5. **Heavy part – O(N³) contraction**:
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`
   is performed by a custom Triton kernel.  
   The kernel processes the batch, the `(i,j)` plane in 2‑D blocks
   (`BLOCK_M`×`BLOCK_N`) and the hidden dimension in tiles of size `VEC`.
   For each tile it iterates over `k`‑tiles, loads the corresponding
   sub‑matrices (shape *M×K×V* and *N×K×V*) in FP16, and accumulates the
   outer‑product over `k` into a FP32 accumulator.
6. **Layer‑norm** over the hidden dimension of the contraction result,
   multiply by the `out_gate`, and finally a linear projection back to
   the original channel dimension.

The kernel works for any batch size (`B=1,2`), sequence length up to 1024
and hidden dimensions up to 384.  It returns a `float32` tensor of shape
`[B, N, N, dim]`.  The implementation is deliberately simple – correctness
is prioritized – while still off‑loading the cubic work to the GPU via
Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the cubic contraction:
#   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def _trmul_kernel(
    left_ptr,               # const * left  (B,N,N,H)   fp16
    right_ptr,              # const * right (B,N,N,H)   fp16
    out_ptr,                # * out (B,N,N,H)           fp16 (stored as fp16, later cast)
    B, N, H,                # scalar sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # rows of the i‑dimension tile
    BLOCK_N: tl.constexpr,  # cols of the j‑dimension tile
    BLOCK_K: tl.constexpr,  # k‑dimension tile
    VEC: tl.constexpr,      # hidden‑dim vector width (tile)
):
    pid_b = tl.program_id(2)            # batch axis
    pid_i = tl.program_id(0)            # i‑tile axis
    pid_j = tl.program_id(1)            # j‑tile axis

    # ------------------------------------------------------------------
    # Offsets for the ij‑plane
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    offs_i = i0 + tl.arange(0, BLOCK_M)   # (M)
    offs_j = j0 + tl.arange(0, BLOCK_N)   # (N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Loop over hidden dimension in tiles of size VEC
    # ------------------------------------------------------------------
    for h_tile in range(0, tl.cdiv(H, VEC)):
        h0 = h_tile * VEC
        offs_h = h0 + tl.arange(0, VEC)          # (V)
        mask_h = offs_h < H

        # Accumulator for the current hidden‑tile
        acc = tl.zeros((BLOCK_M, BLOCK_N, VEC), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over the k‑dimension (pair‑wise aggregation)
        # ------------------------------------------------------------------
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            k0 = k_tile * BLOCK_K
            offs_k = k0 + tl.arange(0, BLOCK_K)   # (K)
            mask_k = offs_k < N

            # ------------------------------------------------------------------
            # Load a tile of left  : shape (M, K, V)
            # ------------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )   # fp16

            # ------------------------------------------------------------------
            # Load a tile of right : shape (N, K, V)
            # ------------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + offs_j[:, None, None] * stride_right_j
                + offs_k[None, :, None] * stride_right_k
                + offs_h[None, None, :] * stride_right_h
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )   # fp16

            # ------------------------------------------------------------------
            # Reduce over the K‑tile.
            # For each kk we compute an outer‑product across (i,j) for every
            # hidden channel in the tile.
            # ------------------------------------------------------------------
            for kk in range(0, BLOCK_K):
                # slice shape (M, V) and (N, V)
                left_kk = left[:, kk, :]          # (M, V)
                right_kk = right[:, kk, :]        # (N, V)

                # Broadcast & accumulate:  (M,1,V) * (1,N,V) → (M,N,V)
                acc += left_kk[:, None, :] * right_kk[None, :, :]

        # ------------------------------------------------------------------
        # Store the completed hidden‑tile
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Entry point – fully‑featured forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]    (bool/float) or None
        - weights      : dict of torch.Tensor containing all model weights
        - config       : dict with keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (mask is broadcast on the channel axis)
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask = mask.unsqueeze(-1).to(x.dtype)                # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating scalars (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy O(N³) contraction – Triton kernel
    # ------------------------------------------------------------------
    # Cast to half for higher throughput, the kernel accumulates in FP32
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output buffer in half‑precision
    out_h = torch.empty_like(left_h, dtype=torch.float16, device=left_h.device)

    # Block sizes – chosen to fit nicely into an H100 SM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    VEC = 32                     # hidden‑dim tile (must divide hidden_dim or be masked)

    # Grid: (i‑tiles, j‑tiles, batch)
    grid = (
        triton.cdiv(x.shape[1], BLOCK_M),
        triton.cdiv(x.shape[1], BLOCK_N),
        x.shape[0],
    )

    _trmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        x.shape[0],
        x.shape[1],
        hidden_dim,
        left_h.stride(0), left_h.stride(1), left_h.stride(2), left_h.stride(3),
        right_h.stride(0), right_h.stride(1), right_h.stride(2), right_h.stride(3),
        out_h.stride(0), out_h.stride(1), out_h.stride(2), out_h.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        VEC=VEC,
        num_warps=4,
    )

    # Back to float32 for the remaining lightweight ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Final Layer‑norm, gate, and linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
