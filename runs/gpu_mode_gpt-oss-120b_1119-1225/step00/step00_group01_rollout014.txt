"""
TriMul (outgoing) forward pass implementation.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Project X to a hidden dimension H (left/right) and compute three
   gating projections (left‑gate, right‑gate, out‑gate) with a sigmoid.
3. Fuse mask (if provided) and gating via a custom Triton kernel:
      left_out  = left  * mask * left_gate
      right_out = right * mask * right_gate
4. Compute the “tri‑multiplicative” update:
      out[b,i,j,:] = Σ_k left_out[b,i,k,:] ⋅ right_out[b,j,k,:]
   (a batch of H matrix‑multiplications, performed with torch.bmm).
5. Apply a second Layer‑norm on the hidden dimension, multiply by out‑gate,
   and project back to the original channel dimension C.
All heavy element‑wise work (step 3) is executed in Triton; the large
matrix‑multiplications use cuBLAS via torch.bmm for maximal performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def mask_gate_kernel(
    # pointers
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    out_left_ptr, out_right_ptr,
    # dimensions
    M, H,
    # strides (in elements)
    stride_left_m, stride_left_h,
    stride_right_m, stride_right_h,
    stride_gate_m, stride_gate_h,
    stride_mask_m,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Fuse mask multiplication and gating for left/right tensors."""
    pid_m = tl.program_id(0)
    pid_h = tl.program_id(1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # Load mask (broadcast over H)
    mask = tl.load(mask_ptr + offs_m * stride_mask_m,
                   mask=offs_m < M,
                   other=1.0)          # default 1.0 → no effect when out‑of‑bounds

    # ----- left side -----
    left = tl.load(
        left_ptr + offs_m[:, None] * stride_left_m + offs_h[None, :] * stride_left_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0
    )
    left_gate = tl.load(
        left_gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0
    )
    out_left = left * mask[:, None] * left_gate
    tl.store(
        out_left_ptr + offs_m[:, None] * stride_left_m + offs_h[None, :] * stride_left_h,
        out_left,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H)
    )

    # ----- right side -----
    right = tl.load(
        right_ptr + offs_m[:, None] * stride_right_m + offs_h[None, :] * stride_right_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0
    )
    right_gate = tl.load(
        right_gate_ptr + offs_m[:, None] * stride_gate_m + offs_h[None, :] * stride_gate_h,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H),
        other=0.0
    )
    out_right = right * mask[:, None] * right_gate
    tl.store(
        out_right_ptr + offs_m[:, None] * stride_right_m + offs_h[None, :] * stride_right_h,
        out_right,
        mask=(offs_m[:, None] < M) & (offs_h[None, :] < H)
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float)
        - weights_dict : dict of model weights (see skeleton)
        - config_dict  : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ---------------------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Ensure tensors are on the same device and contiguous
    x = input_tensor.contiguous()
    B, N, _, C = x.shape
    assert C == dim, f"Expected input dim {dim}, got {C}"

    # ---------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the channel dimension
    # ---------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Flatten the pairwise dimensions to a massive batch for linear ops
    M = B * N * N                         # total number of (i,j) positions
    x_flat = x.view(M, dim).contiguous()  # shape (M, C)

    # ---------------------------------------------------------------------
    # 2️⃣ Linear projections + gating (all via torch.nn.functional.linear)
    # ---------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    # Projections
    left = F.linear(x_flat, left_proj_weight)          # (M, H)
    right = F.linear(x_flat, right_proj_weight)        # (M, H)

    # Gating logits -> sigmoid
    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_weight))

    # ---------------------------------------------------------------------
    # 3️⃣ Fuse mask & gating via Triton (produces masked & gated tensors)
    # ---------------------------------------------------------------------
    if nomask:
        mask_flat = torch.ones(M, dtype=x.dtype, device=x.device)
    else:
        mask_flat = mask_tensor.view(M).to(dtype=x.dtype)

    # Allocate output buffers for the fused results
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Triton kernel launch configuration
    BLOCK_M = 1024   # rows per program
    BLOCK_H = 32     # hidden channels per program
    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(hidden_dim, BLOCK_H))

    mask_gate_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask_flat.data_ptr(),
        out_left_ptr=left_fused.data_ptr(),
        out_right_ptr=right_fused.data_ptr(),
        M=M,
        H=hidden_dim,
        stride_left_m=left.stride(0),
        stride_left_h=left.stride(1),
        stride_right_m=right.stride(0),
        stride_right_h=right.stride(1),
        stride_gate_m=left_gate.stride(0),
        stride_gate_h=left_gate.stride(1),
        stride_mask_m=mask_flat.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )

    # ---------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update  (batched GEMM over the sequence dimension)
    # ---------------------------------------------------------------------
    # Reshape back to [B, N, N, H]
    left_fused = left_fused.view(B, N, N, hidden_dim)
    right_fused = right_fused.view(B, N, N, hidden_dim)

    # Permute to (B, H, N, N) and merge batch‑and‑head for a large batched GEMM
    left_perm = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Compute Σ_k left[...,i,k] * right[...,j,k] for each head
    out_perm = torch.bmm(left_perm, right_perm.transpose(1, 2))
    out = out_perm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ---------------------------------------------------------------------
    # 5️⃣ Output LayerNorm, out‑gate, final projection
    # ---------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Reshape out_gate to [B, N, N, H] for element‑wise scaling
    out_gate = out_gate.view(B, N, N, hidden_dim)
    out = out * out_gate

    # Final linear projection back to dim channels
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # (B,N,N,dim)

    # Ensure final dtype is float32 as required
    return out.float()
