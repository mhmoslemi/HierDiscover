def layer_norm(x, weight, bias, eps=1e-5):
    # x shape [..., dim]; weight shape [dim]; bias shape [dim]
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    inv = torch.rsqrt(var + eps)
    x_norm = (x - mean) * inv
    if weight is not None:
        x_norm = x_norm * weight + bias
    return x_norm

"""
TriMul (outgoing) forward pass implemented with Triton.

The core of the TriMul operator is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched matrix multiplication for each hidden
dimension `d`.  This kernel fuses the reduction over `k` using
Triton's `tl.dot`, accumulates in FP32 (the H100 has fast FP16→FP32
accumulators) and stores the result in FP32.  All other linear
projections, gating, and LayerNorms are performed with PyTorch
operations (they are cheap compared to the O(N³·H) contraction).

The kernel computes one GEMM per (batch, hidden) pair:
    L = left[b,:, :, h]      # (N, N)
    Rᵀ = right[b,:, :, h].transpose(0,1)   # (N, N)
    out[b,:, :, h] = L @ Rᵀ
A 2‑D tiling (BLOCK_M × BLOCK_N) with a K‑tile (BLOCK_K) is used.
Masking (if present) is applied before the kernel, and gating is fused
into the `left`/`right` tensors prior to the Triton call.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the einsum reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_einsum_kernel(
    left_ptr,        # const float* left  [B, N, N, H] (FP16)
    right_ptr,       # const float* right [B, N, N, H] (FP16)
    out_ptr,         # float* out          [B, N, N, H] (FP32)
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b,   stride_out_i,   stride_out_j,   stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h] """
    pid     = tl.program_id(axis=0)   # combined batch & hidden index
    pid_m   = tl.program_id(axis=1)   # tile over i
    pid_n   = tl.program_id(axis=2)   # tile over j

    # ------------------------------------------------------------------
    # Decode batch and hidden indices
    # ------------------------------------------------------------------
    b = pid // H                      # batch index
    h = pid % H                       # hidden/channel index

    # ------------------------------------------------------------------
    # Offsets for the i and j tiles
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks
    # ------------------------------------------------------------------
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load left block   [BM, BK]  (i × k)
        # --------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + h * stride_left_h
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)               # FP16

        # --------------------------------------------------------------
        # Load right block transposed  [BK, BN]  (k × j)
        # --------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + h * stride_right_h
            + offs_k[:, None] * stride_right_k   # rows = k
            + offs_j[None, :] * stride_right_i   # cols = j
        )
        right_t = tl.load(right_ptrs,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)               # FP16

        # --------------------------------------------------------------
        # Multiply‑accumulate using half‑precision dot, accumulate in FP32
        # --------------------------------------------------------------
        acc += tl.dot(left.to(tl.float16), right_t.to(tl.float16))

    # ------------------------------------------------------------------
    # Store the result (FP32)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def _trimul_einsum(left_fp16: torch.Tensor, right_fp16: torch.Tensor,
                   BLOCK_M=64, BLOCK_N=64, BLOCK_K=64) -> torch.Tensor:
    """
    left_fp16/right_fp16 : [B, N, N, H] (torch.float16)
    returns out : [B, N, N, H] (torch.float32)
    """
    assert left_fp16.stride() == right_fp16.stride(), "left/right must have identical layout"
    B, N, _, H = left_fp16.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left_fp16.device)

    # Strides in element units
    lsb, lsi, lsk, lsh = left_fp16.stride()
    rsb, rsi, rsk, rsh = right_fp16.stride()
    osb, osi, osj, osh = out.stride()

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_einsum_kernel[grid](
        left_fp16.data_ptr(),
        right_fp16.data_ptr(),
        out.data_ptr(),
        B, N, H,
        lsb, lsi, lsk, lsh,
        rsb, rsi, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Entry point as required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) implemented with a Triton kernel for the
    high‑cost O(N³·H) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask          : torch.Tensor of shape [B, N, N] (float32) or None
        - weights       : dict of weight tensors (all float32)
        - config        : dict containing 'dim', 'hidden_dim' and optional 'nomask'

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight        = weights["norm.weight"]
    norm_bias          = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (FP32)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Gating signals (FP32, sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Apply mask if present (mask shape [B,N,N] -> broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_factor = 1.0
    else:
        mask_factor = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]

    # ------------------------------------------------------------------
    # Fuse projection, mask, and gate (still FP32)
    # ------------------------------------------------------------------
    left  = left_proj * mask_factor * left_gate   # [B,N,N,H]
    right = right_proj * mask_factor * right_gate

    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy contraction
    # ------------------------------------------------------------------
    left_hp  = left.to(torch.float16)
    right_hp = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Triton contraction: out_fp32 [B,N,N,H]
    # ------------------------------------------------------------------
    out = _trimul_einsum(left_hp, right_hp,
                         BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                      # gating after norm
    out = F.linear(out, to_out_weight)        # [B,N,N,dim]

    return out
