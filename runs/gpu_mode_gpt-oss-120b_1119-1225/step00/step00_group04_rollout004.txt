custom_kernel: Triton‑accelerated TriMul (outgoing) forward pass.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension).
2. Project the normalized tensor to a hidden dimension with two linear layers
   (left_proj / right_proj).  Apply an optional pairwise mask.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized input and multiply them into the projected tensors.
4. The core “tri‑multiplicative” update is
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
   For each (b,d) this is a matrix multiplication `L_d @ R_d.T`.  A
   Triton GEMM kernel tiles the (i,j) plane with BLOCK_M×BLOCK_N tiles and
   accumulates over K‑tiles of size BLOCK_K using fp16 arithmetic (the
   accumulator is fp32 for numerical stability).
5. Apply a second layer‑norm on the result, multiply by the output gate and
   finish with a linear projection back to the original feature dimension.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

        # load with masking (outside‑range entries are zero)
        left_tile  = tl.load(left_ptrs,
                             mask=(mask_i[:, None] & mask_k[None, :]),
                             other=0.0)           # fp16
        right_tile = tl.load(right_ptrs,
                             mask=(mask_j[:, None] & mask_k[None, :]),
                             other=0.0)           # fp16

        # GEMM on the tile: (M×K) @ (K×N)  ==  (M×N)
        # right_tile needs to be transposed so that its inner dimension aligns with K
        acc += tl.dot(left_tile, tl.trans(right_tile)).to(tl.float32)

    # write back the result
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=(mask_i[:, None] & mask_j[None, :]))


# ----------------------------------------------------------------------
# entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module using a Triton GEMM kernel.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    # shapes
    B, N, N2, dim = input_tensor.shape
    assert N == N2, "input must be square on the pairwise dimensions"
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Layer‑norm on the input (last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half‑precision for the heavy GEMM and allocate output
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=input_tensor.device)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32                                            # good for fp16 on H100

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # stride information (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_h.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # launch kernel
    _trimul_gemm_kernel[grid](
        left_h,
        right_h,
        out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_i, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Second layer‑norm, output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)

    out_norm = out_norm * out_gate                              # apply out gate

    final_output = F.linear(out_norm, weights["to_out.weight"]) # [B,N,N,dim]

    return final_output
