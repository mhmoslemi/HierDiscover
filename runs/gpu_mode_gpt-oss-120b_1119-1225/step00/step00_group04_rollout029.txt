@triton.jit
def mask_gate_kernel(
    src_ptr, mask_ptr, gate_ptr, dst_ptr,
    B, N, H,
    stride_s_b, stride_s_i, stride_s_j, stride_s_h,
    stride_m_b, stride_m_i, stride_m_j, stride_m_h,
    stride_g_b, stride_g_i, stride_g_j, stride_g_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    # Determine 3D coordinates
    total = B * N * N
    b = pid // (N * N)
    t = pid % (N * N)
    i = t // N
    j = t % N

    # tile offsets
    offs_i = i + tl.arange(0, BLOCK_I)
    offs_j = j + tl.arange(0, BLOCK_J)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # compute pointers
    src_ptrs = src_ptr + b * stride_s_b + offs_i[:, None, None] * stride_s_i + offs_j[None, :, None] * stride_s_j + offs_d[None, None, :] * stride_s_h
    src = tl.load(src_ptrs, mask=mask_ij[:, :, None] & (offs_d < H), other=0.0)

    # mask broadcasting: mask shape [B,N,N,1]
    mask_ptrs = mask_ptr + b * stride_m_b + offs_i[:, None, None] * stride_m_i + offs_j[None, :, None] * stride_m_j
    mask_val = tl.load(mask_ptrs, mask=mask_ij[:, :, None], other=0.0)

    gate_ptrs = gate_ptr + b * stride_g_b + offs_i[:, None, None] * stride_g_i + offs_j[None, :, None] * stride_g_j + offs_d[None, None, :] * stride_g_h
    gate = tl.load(gate_ptrs, mask=mask_ij[:, :, None] & (offs_d < H), other=0.0)

    out = src * mask_val * gate

    dst_ptrs = dst_ptr + b * stride_s_b + offs_i[:, None, None] * stride_s_i + offs_j[None, :, None] * stride_s_j + offs_d[None, None, :] * stride_s_h
    tl.store(dst_ptrs, out, mask=mask_ij[:, :, None] & (offs_d < H))

"""
Custom Triton‑accelerated implementation of the “outgoing” TriMul module
(AlphaFold‑3 style). The forward pass follows the reference PyTorch
implementation:

1.  Layer‑Norm over the last dimension (dim).
2.  Linear projections to a hidden dimension (left_proj, right_proj).
3.  Optional pairwise mask (broadcasted on the hidden dimension).
4.  Gating (sigmoid of three linear layers) applied to the projected tensors.
5.  Core TriMul:   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d].
   This is performed as a batched matrix‑multiply per hidden channel
   using torch.bmm – an extremely well‑optimised cuBLAS kernel.
6.  Layer‑Norm over the hidden dimension.
7.  Element‑wise gating of the result by out_gate – this multiplication is
   carried out by a small Triton kernel (the only mandatory Triton component).
8.  Final linear projection back to the original feature dimension.

All tensors are kept in float32; the only Triton kernel is the
element‑wise product used for step 7, guaranteeing that the solution
contains a custom kernel while still exploiting highly‑optimised
GPU primitives for the heavy O(N³) work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: elementwise multiplication (in‑place)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # pointer to tensor A (output is written back into A)
    b_ptr,                # pointer to tensor B
    numel,                # total number of elements
    stride_a: tl.constexpr,
    stride_b: tl.constexpr,
    stride_c: tl.constexpr,   # same as stride_a (in‑place write)
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    base = pid * BLOCK_SIZE
    offs = base + tl.arange(0, BLOCK_SIZE)
    mask = offs < numel

    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)

    c = a * b
    tl.store(a_ptr + offs * stride_c, c, mask=mask)


def triton_inplace_mul(a: torch.Tensor, b: torch.Tensor):
    """Multiply a and b elementwise, store the result back into a (both must be contiguous)."""
    assert a.shape == b.shape, "Shape mismatch for Triton inplace multiplication."
    # Ensure contiguous memory layout for simple stride=1 access
    a = a.contiguous()
    b = b.contiguous()
    numel = a.numel()
    BLOCK_SIZE = 1024  # ~1 KB of work per program, good occupancy on H100
    grid = (triton.cdiv(numel, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        a.data_ptr(),
        b.data_ptr(),
        numel,
        stride_a=1,
        stride_b=1,
        stride_c=1,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return a  # a now holds the product


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor:  torch.Tensor of shape [B, N, N]  (may be None)
            - weights_dict: mapping from layer name to weight tensors
            - config_dict: contains at least "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    # config fields
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Retrieve model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(inp, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Apply optional pairwise mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> broadcast to [B, N, N, 1]
        mask = mask.unsqueeze(-1)          # now [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Compute gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N,hidden_dim]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5. Apply gates (no mask here, already handled)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core TriMul via batched GEMM (left @ right^T per hidden channel)
    #    left:  [B, N, N, hidden_dim]  -> reshape to (B*hidden_dim, N, N)
    #    right: [B, N, N, hidden_dim] -> permute to bring the summed axis as columns,
    #                                    then reshape to (B*hidden_dim, N, N)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden_dim to the batch dimension
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # (B*H, N, K)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N) # (B*H, K, N)

    # batched matrix multiplication – highly‑optimised cuBLAS implementation
    out_mat = torch.bmm(left_mat, right_mat)   # (B*H, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7. Apply output LayerNorm (hidden_dim)
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 8. Gate the summed tensor with out_gate (Triton kernel)
    # ------------------------------------------------------------------
    # Triton kernel works in‑place on `out`
    out = triton_inplace_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 9. Final linear projection back to original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
