Define kernel:

```
@triton.jit
def batched_gemm_kernel(A_ptr, B_ptr, C_ptr,
                        M, N, K,
                        stride_am, stride_ak, stride_ak2,
                        stride_bk, stride_bn, stride_bn2,
                        stride_cm, stride_cn, stride_cn2,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointers for this batch
    a_ptrs = A_ptr + pid_batch * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = B_ptr + pid_batch * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs, mask= (offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask= (offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Store result
    c_ptrs = C_ptr + pid_batch * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn

@triton.jit
def batched_gemm_kernel(A, B, C,
                        M, N, K,
                        stride_am, stride_ak, stride_ah,
                        stride_bk, stride_bn, stride_bh,
                        stride_cm, stride_cn, stride_ch,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # compute pointer to batch start
    a_batch_ptr = A + pid_batch * stride_ah
    b_batch_ptr = B + pid_batch * stride_bh
    c_batch_ptr = C + pid_batch * stride_ch

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute mask for out-of-bounds for m and n
    mask_m = offs_m < M
    mask_n = offs_n < N

    # load pointers for A and B at each k iteration
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, K, BLOCK_K):
        # load a block
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K)[None, :], other=0.0)
        # load b block
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b)

        # advance pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated implementation of the “outgoing” Triangle Multiplicative
Update (TriMul) from AlphaFold‑3.

Algorithm
---------
1. Layer‑norm over the channel dimension `dim`.
2. Linear projections (left, right) + elementwise mask (if present).
3. Gated projections:
   left  = left  * sigmoid(linear(x_norm, left_gate_W))
   right = right * sigmoid(linear(x_norm, right_gate_W))
4. Core operation:
   out[i,j] = Σ_k left[i,k] • right[j,k]   (for every batch and hidden‑dim)
   This is a batched GEMM: for each hidden slice h,
   out_h = left_h @ right_hᵀ .
   Implemented with a custom Triton kernel that processes many (batch·hidden)
   matrix‑multiplications in parallel (fp16 → fp32 accumulation).
5. Layer‑norm over the hidden dimension.
6. Output gating: out = out * sigmoid(linear(x_norm, out_gate_W)).
7. Final linear projection back to `dim`.
All heavy compute (step 4) runs on the GPU via Triton; the surrounding
operations use PyTorch (half‑precision) for simplicity.

The kernel is tuned for H100 (BLOCK_M/N=64, BLOCK_K=32) and works for
seq_len up to 1024, hidden_dim = 128.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Batched GEMM kernel: C = A @ B   where A.shape = [B, M, K],
#                                 B.shape = [B, K, N],
#                                 C.shape = [B, M, N]
# A, B, C are half‑precision, accumulation is float‑32.
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A, B, C,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(axis=2)          # batch*hidden index
    pid_m = tl.program_id(axis=0)              # tile row
    pid_n = tl.program_id(axis=1)              # tile col

    # pointers to the start of this batch
    a_batch_ptr = A + pid_batch * stride_ah
    b_batch_ptr = B + pid_batch * stride_bh
    c_batch_ptr = C + pid_batch * stride_ch

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for out‑of‑bounds rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for the first K‑tile
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k + offs_k                                    # actual K indices
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                                 # (BM, BK) half
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                                 # (BK, BN) half

        # dot returns half; promote to fp32 before accumulation
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance to next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back (cast to half)
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, tl.cast(acc, tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------------------
# Entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]  (or None)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype_fp16 = torch.float16
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. LayerNorm over `dim`
    # -----------------------------------------------------------------
    with torch.no_grad():
        x = torch.nn.functional.layer_norm(
            input_tensor,
            normalized_shape=(dim,),
            weight=weights["norm.weight"],
            bias=weights["norm.bias"],
            eps=eps,
        )
        # cast to half for the heavy ops
        x_h = x.to(dtype_fp16)

        # -----------------------------------------------------------------
        # 2. Linear projections (left/right)
        # -----------------------------------------------------------------
        left = torch.nn.functional.linear(
            x_h, weights["left_proj.weight"].to(dtype_fp16), bias=None
        )  # [B,N,N,hidden_dim]
        right = torch.nn.functional.linear(
            x_h, weights["right_proj.weight"].to(dtype_fp16), bias=None
        )  # [B,N,N,hidden_dim]

        # -----------------------------------------------------------------
        # 3. Apply mask (if any)
        # -----------------------------------------------------------------
        if not nomask and mask is not None:
            # mask: [B, N, N] -> [B, N, N, 1] broadcast
            mask_h = mask.to(dtype_fp16).unsqueeze(-1)
            left = left * mask_h
            right = right * mask_h

        # -----------------------------------------------------------------
        # 4. Gating
        # -----------------------------------------------------------------
        left_gate = torch.sigmoid(
            torch.nn.functional.linear(
                x_h, weights["left_gate.weight"].to(dtype_fp16), bias=None
            )
        )
        right_gate = torch.sigmoid(
            torch.nn.functional.linear(
                x_h, weights["right_gate.weight"].to(dtype_fp16), bias=None
            )
        )
        out_gate = torch.sigmoid(
            torch.nn.functional.linear(
                x_h, weights["out_gate.weight"].to(dtype_fp16), bias=None
            )
        )
        left = left * left_gate
        right = right * right_gate

        # -----------------------------------------------------------------
        # 5. Core TriMul via batched GEMM
        #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
        # -----------------------------------------------------------------
        B, N, _, _ = left.shape

        # bring hidden dim to front to create a batch of independent GEMMs
        # shape -> [B, H, N, N]
        left_perm = left.permute(0, 3, 1, 2).contiguous()
        right_perm = right.permute(0, 3, 1, 2).contiguous()

        # flatten batch*hidden for the kernel
        batch_h = B * hidden_dim
        A = left_perm.view(batch_h, N, N).to(dtype_fp16)          # [BH, M, K]
        B_mat = right_perm.transpose(-2, -1).contiguous().view(batch_h, N, N).to(dtype_fp16)  # [BH, K, N]

        # output placeholder (half)
        C = torch.empty_like(A, dtype=dtype_fp16)

        # strides (int64) required by Triton
        stride_am, stride_ak, stride_ah = A.stride()
        stride_bk, stride_bn, stride_bh = B_mat.stride()
        stride_cm, stride_cn, stride_ch = C.stride()

        # choose block sizes (tuned for H100)
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 32

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            batch_h,
        )

        batched_gemm_kernel[grid](
            A,
            B_mat,
            C,
            N,
            N,
            N,
            stride_am,
            stride_ak,
            stride_ah,
            stride_bk,
            stride_bn,
            stride_bh,
            stride_cm,
            stride_cn,
            stride_ch,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

        # reshape back to [B, N, N, hidden_dim] (float16)
        out = (
            C.view(B, hidden_dim, N, N)
            .permute(0, 2, 3, 1)
            .contiguous()
        )  # [B, N, N, hidden_dim]

        # -----------------------------------------------------------------
        # 6. Output LayerNorm (hidden_dim) and gating
        # -----------------------------------------------------------------
        out = torch.nn.functional.layer_norm(
            out,
            normalized_shape=(hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
            eps=eps,
        )
        out = out * out_gate  # elementwise gating (both half)

        # -----------------------------------------------------------------
        # 7. Final linear projection back to `dim`
        # -----------------------------------------------------------------
        out = torch.nn.functional.linear(
            out, weights["to_out.weight"].to(dtype_fp16), bias=None
        )  # [B, N, N, dim]

        # Cast result to float32 as expected by downstream code
        return out.to(torch.float32)
