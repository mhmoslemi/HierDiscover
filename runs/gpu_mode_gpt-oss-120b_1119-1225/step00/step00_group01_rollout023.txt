@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_l_b, stride_l_i, stride_l_j, stride_l_h,
                   stride_r_b, stride_r_i, stride_r_j, stride_r_h,
                   stride_o_b, stride_o_i, stride_o_j, stride_o_h,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden indices
    total_H = H
    batch_idx = pid_batch // H
    hid = pid_batch % H

    # offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute pointer base addresses for this batch and hidden
    left_base = left_ptr + batch_idx * stride_l_b + hid * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_b + hid * stride_r_h
    out_base = out_ptr + batch_idx * stride_o_b + hid * stride_o_h

    # Initialize accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # maybe FP32 accumulator

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k tile start
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for valid rows/columns
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # load left block
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_j
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)  # Cast to FP32 if needed

        # load right block transposed (k, j)
        right_ptrs = right_base + offs_k[:, None] * stride_r_j + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = right.to(tl.float32)

        # dot product accumulate
        acc += tl.dot(left, right)  # (M,K) dot (K,N) -> (M,N)

    # write out
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

```python
"""TriMul custom kernel

Implements the "outgoing" variant of the TriMul operator used in AlphaFold3.
The kernel fuses the expensive O(N^3) pairwise multiplication using a
batched matrix multiplication implemented in Triton. All other operations
(LayerNorms, linear projections, gating) are performed with PyTorch for
simplicity.

The kernel computes:
    out[b,i,j,h] = \sum_{k=0}^{N-1} left[b,i,k,h] * right[b,j,k,h]
for every batch b and hidden channel h, where left/right are obtained
from linear projections of the normalized input and are masked by the
provided mask.
"""

import torch
import triton
import triton.language as tl

# Triton kernel performing batched matmul
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *fp32
    right_ptr,              # *fp32
    out_ptr,                # *fp32
    B, N, H,                # i32
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ right^T  (batch * hidden_dim, N, N)
    left   shape: [B, N, N, H]   (i,k)
    right  shape: [B, N, N, H]   (j,k)
    out    shape: [B, N, N, H]   (i,j)
    The kernel is launched with a 3‑D grid:
        grid = (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid_batch_h = tl.program_id(0)   # combined batch * hidden
    pid_i = tl.program_id(1)         # block row
    pid_j = tl.program_id(2)         # block column

    # decode batch and hidden indices
    hidden = H
    batch_idx = pid_batch_h // hidden
    hid = pid_batch_h % hidden

    # compute offsets for i and j dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers that already include batch and hidden offset
    left_base = left_ptr + batch_idx * stride_l_b + hid * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_b + hid * stride_r_h
    out_base = out_ptr + batch_idx * stride_o_b + hid * stride_o_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------------------------------------------------------
        # Load a tile of left: shape (BLOCK_M, BLOCK_K) => (i, k)
        # left[i,k] uses stride_l_i for i, stride_l_j for k
        # -----------------------------------------------------------------
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_j
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        left = left.to(tl.float32)

        # -----------------------------------------------------------------
        # Load a tile of right transposed: shape (BLOCK_K, BLOCK_N) => (k, j)
        # right[j,k] has stride_r_i for j (first N) and stride_r_j for k (second N)
        # We want right^T[k, j] = right[j, k]
        # -----------------------------------------------------------------
        right_ptrs = right_base + offs_k[:, None] * stride_r_j + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)
        right = right.to(tl.float32)

        # Accumulate the dot product over the K dimension
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # -----------------------------------------------------------------
    # Write the result
    # -----------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        * input_tensor: torch.Tensor of shape [B, N, N, dim]
        * mask        : torch.Tensor of shape [B, N, N] (float or bool)
        * weights     : dict of torch.Tensors containing all model weights
        * config      : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] containing the output of the
        TriMul block.
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    # weight and bias have shape (dim,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply mask (if present)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype)
        mask = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Compute gating values (sigmoid of linear projections of x)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply the gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton kernel: the expensive N^3 contraction
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Allocate output tensor for the contraction
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # Extract strides (in elements, not bytes) for the four‑dimensional tensors
    stride_l_b, stride_l_i, stride_l_j, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_j, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Triton grid configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_j,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_j,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""TriMul outgoing implementation.

The function `custom_kernel` computes the forward pass of the “outgoing”
TriMul block used in AlphaFold‑3.  The heavy O(N³) contraction

    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

is performed by a custom Triton kernel that fuses a batched matrix
multiplication for each hidden channel `h`.  All other operations
(LayerNorm, linear projections, sigmoid gates, final projection) are
implemented with PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,   # *fp32
    right_ptr,  # *fp32
    out_ptr,    # *fp32
    B, N, H,    # i32 – batch, sequence length, hidden dim
    stride_l_b, stride_l_i, stride_l_j, stride_l_h,
    stride_r_b, stride_r_i, stride_r_j, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out = left @ rightᵀ for each (batch, hidden) pair.
    left  – shape [B, N, N, H]  (indexed i,k)
    right – shape [B, N, N, H]  (indexed j,k)
    out   – shape [B, N, N, H]  (indexed i,j)
    The kernel is launched with a 3‑D grid:
        grid = (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    """
    pid_bhid = tl.program_id(0)   # combined batch * hidden index
    pid_i = tl.program_id(1)      # block row (i)
    pid_j = tl.program_id(2)      # block column (j)

    # ---------------------------------------------------------------
    # Decode batch and hidden indices
    # ---------------------------------------------------------------
    hidden = H
    batch_idx = pid_bhid // hidden
    hid = pid_bhid % hidden

    # ---------------------------------------------------------------
    # Offsets for the output tile (i,j)
    # ---------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---------------------------------------------------------------
    # Base pointers that already include batch and hidden offset
    # ---------------------------------------------------------------
    left_base = left_ptr + batch_idx * stride_l_b + hid * stride_l_h
    right_base = right_ptr + batch_idx * stride_r_b + hid * stride_r_h
    out_base = out_ptr + batch_idx * stride_o_b + hid * stride_o_h

    # ---------------------------------------------------------------
    # Accumulator for the (BLOCK_M, BLOCK_N) output tile
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over the reduction dimension k
    # ---------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]  (M × K) -------------------------------------------------
        left_ptrs = left_base + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_j
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        left = left.to(tl.float32)

        # right[j, k] transposed to (K × N) ------------------------------------
        # right[j, k] is stored with stride_r_i for the j‑axis and stride_r_j for k‑axis.
        # We load it as a (K, N) tile = (k, j) so that a dot product works.
        right_ptrs = right_base + offs_k[:, None] * stride_r_j + offs_j[None, :] * stride_r_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)
        right = right.to(tl.float32)

        # Accumulate the matrix product for this k‑tile
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # ---------------------------------------------------------------
    # Write the result tile back to memory
    # ---------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[:, None])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensors containing model parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the TriMul output.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply mask if provided (mask is broadcast over the hidden dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4. Compute sigmoid gates (still using the normalized input)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply the gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity for Triton (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # 5. Triton kernel – the O(N³) contraction
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=left.dtype,
                      device=left.device)

    # Extract element‑wise strides (PyTorch returns strides in terms of elements)
    s_l_b, s_l_i, s_l_j, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_j, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Triton block sizes – chosen to fit well in shared memory on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑tiles
    )

    # Launch the kernel
    tri_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_j,
        s_l_h,
        s_r_b,
        s_r_i,
        s_r_j,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6. Post‑processing: LayerNorm, out‑gate and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
