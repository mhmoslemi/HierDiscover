def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    epsilon = 1e-5

    # unpack weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Layernorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=epsilon)

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # (B,N,N,H)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).type_as(left)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for tri-mul
    B, N, _, _ = left.shape
    # Permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Combine batch and hidden dims
    BH = B * hidden_dim
    left_reshape = left_perm.view(BH, N, N)
    right_reshape = right_perm.view(BH, N, N)

    # Allocate output tensor
    out_reshape = torch.empty_like(left_reshape)

    # Strides
    stride_l_batch, stride_l_m, stride_l_n = left_reshape.stride()
    stride_r_batch, stride_r_m, stride_r_n = right_reshape.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_reshape.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)

    tr_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_n,
        stride_r_batch,
        stride_r_m,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

"""
TriMul forward (outgoing) implementation with a custom Triton kernel.

The operation is:
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

where:
  * `left`  = Linear(x) * mask * sigmoid(Linear(x))          (hidden_dim)
  * `right` = Linear(x) * mask * sigmoid(Linear(x))          (hidden_dim)

The heavy N³ reduction is performed by a fused Triton kernel that
computes a batched matrix multiplication for every hidden channel.
The kernel works on the reshaped tensors of shape (B*H, N, N) and
produces the same‑shaped output.  The rest of the (light‑weight)
operations – layer‑norms, gates and final linear – are kept in PyTorch.

The kernel tiles the M (i) and N (j) dimensions with BLOCK_M×BLOCK_N
tiles and loops over K in BLOCK_K steps, using `tl.dot` to accumulate
partial products.  It handles arbitrary sequence lengths via masking
and works for both masked / un‑masked inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tr_mul_kernel(
    left_ptr,            # const *float32
    right_ptr,           # const *float32
    out_ptr,             # *float32
    seq_len,             # i32  (N)
    stride_l_batch,      # i64
    stride_l_m,          # i64
    stride_l_k,          # i64
    stride_r_batch,      # i64
    stride_r_m,          # i64
    stride_r_k,          # i64
    stride_o_batch,      # i64
    stride_o_m,          # i64
    stride_o_n,          # i64
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute  out[bh, i, j] = Σₖ left[bh,i,k] * right[bh,j,k]  (tri‑multiplicative).

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to the flattened tensors
        with shape (BH, N, N) where BH = batch * hidden_dim.
    seq_len : sequence length N.
    stride_* : strides for the three‑dimensional tensors in elements.
    BLOCK_* : compile‑time tile sizes.
    """
    pid_bh = tl.program_id(2)          # batch * hidden_dim index
    pid_m = tl.program_id(0)           # i‑tile
    pid_n = tl.program_id(1)           # j‑tile

    # --------------------------------------------------------------
    # 1) Compute per‑tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # (BLOCK_N,)

    mask_m = offs_m < seq_len
    mask_n = offs_n < seq_len

    # --------------------------------------------------------------
    # 2) Initialize accumulator (float32 for numerical stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over the reduction dimension K
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(seq_len, BLOCK_K)
    for k in range(num_k_blocks):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
        mask_k = offs_k < seq_len

        # ----------- Load A = left[bh, i, k] -------------
        a_ptrs = (
            left_ptr
            + pid_bh * stride_l_batch
            + offs_m[:, None] * stride_l_m
            + offs_k[None, :] * stride_l_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # (BLOCK_M, BLOCK_K)

        # ----------- Load Bᵀ = right[bh, j, k] -------------
        # We load the transposed view: B_T[k, j] = right[bh, j, k]
        b_ptrs = (
            right_ptr
            + pid_bh * stride_r_batch
            + offs_n[None, :] * stride_r_m   # j * stride_row
            + offs_k[:, None] * stride_r_k   # k * stride_col (col stride = 1)
        )
        b = tl.load(
            b_ptrs,
            mask=mask_n[None, :] & mask_k[:, None],
            other=0.0,
        )  # (BLOCK_K, BLOCK_N)

        # ----------- Accumulate the dot product -------------
        acc = acc + tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # --------------------------------------------------------------
    # 4) Write the result back
    # --------------------------------------------------------------
    c_ptrs = (
        out_ptr
        + pid_bh * stride_o_batch
        + offs_m[:, None] * stride_o_m
        + offs_n[None, :] * stride_o_n
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] (bool/float) or None
            - weights      : dict of model parameters (see code)
            - config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weights (all expected to be on the same device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float32 and broadcastable
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul: out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    We fold the batch & hidden dimensions to a single dimension
    #    and launch a Triton kernel that performs a batched matmul.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # (B, hidden_dim, N, N) -> contiguous and then view as (BH, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    BH = B * hidden_dim
    left_reshape = left_perm.view(BH, N, N)   # [BH, N, N]
    right_reshape = right_perm.view(BH, N, N)

    out_reshape = torch.empty_like(left_reshape)   # [BH, N, N]

    # Strides in element units
    stride_l_batch, stride_l_m, stride_l_k = left_reshape.stride()
    stride_r_batch, stride_r_m, stride_r_k = right_reshape.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_reshape.stride()

    # Tile sizes – chosen to fit H100's SM resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        BH,                        # batch × hidden channels
    )

    tr_mul_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_m,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, hidden_dim)
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Post‑processing: layer‑norm over hidden_dim, multiply by out_gate,
    #    final linear back to `dim`.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )                         # [B, N, N, hidden_dim]

    out = out * out_gate       # broadcast over hidden_dim

    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
