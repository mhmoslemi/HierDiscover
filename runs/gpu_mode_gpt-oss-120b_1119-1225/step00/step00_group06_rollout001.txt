    stride_rbh, stride_rj, stride_rk,

    pid_n = tl.program_id(1)  # j block
    pid_z = tl.program_id(2)  # batch*hidden

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    j_valid = offs_n < N

    base_o = out_ptr + pid_z * stride_obh

"""
TriMul “outgoing” implementation (forward only) with a custom Triton kernel.

The heavy part of TriMul is the 4‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B, N, N, H] (H = hidden_dim).
For each (b,d) pair this is an N×N matrix multiplication
A = left[b,:,:,d]      (i,k)
B = right[b,:,:,d]ᵀ   (k,j)

We implement this contraction with a fused GEMM kernel in Triton.
The kernel works on a flattened batch dimension B*H, i.e. it computes a
large number (≤ 256) independent N×N matrix products in parallel.

All other operations (LayerNorm, Linear projections, sigmoids and the final
normalisation/linear) are performed with standard PyTorch ops.  The kernel
loads tiles of size (BLOCK_M, BLOCK_K) and (BLOCK_N, BLOCK_K), accumulates in
float32 and writes the result back in float16 (later cast to float32 for the
remaining pipeline).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # const float16 *
    right_ptr,         # const float16 *
    out_ptr,           # float16 *
    N,                 # int32  (sequence length)
    # strides for the flattened (batch*head) tensors [BH, N, N]
    stride_lbh, stride_li, stride_lk,
    stride_rbh, stride_rj, stride_rk,
    stride_obh, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_m = tl.program_id(0)  # i‑tile
    pid_n = tl.program_id(1)  # j‑tile
    pid_z = tl.program_id(2)  # flattened batch*head index (BH)

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = offs_m < N
    j_valid = offs_n < N

    # base pointers for the current (b,h) slice
    base_l = left_ptr + pid_z * stride_lbh
    base_r = right_ptr + pid_z * stride_rbh
    base_o = out_ptr + pid_z * stride_obh

    # accumulator (float32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        k_valid = offs_k < N

        # ----- A tile : shape (BLOCK_M, BLOCK_K) --------------------
        a_ptr = base_l + (offs_m[:, None] * stride_li) + (offs_k[None, :] * stride_lk)
        a = tl.load(a_ptr,
                    mask=i_valid[:, None] & k_valid[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # ----- B tile : shape (BLOCK_N, BLOCK_K)  (right matrix, not transposed)
        # right is stored as (j, k) → we load it as (row=j, col=k)
        b_ptr = base_r + (offs_n[:, None] * stride_rj) + (offs_k[None, :] * stride_rk)
        b = tl.load(b_ptr,
                    mask=j_valid[:, None] & k_valid[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # dot: (M,K) @ (N,K)ᵀ  → (M,N)  (tl.dot does the correct reduction)
        acc += tl.dot(a, b)

    # ----- write‑back --------------------------------------------------
    o_ptr = base_o + (offs_m[:, None] * stride_oi) + (offs_n[None, :] * stride_oj)
    tl.store(o_ptr,
             acc.to(tl.float16),
             mask=i_valid[:, None] & j_valid[None, :])


# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel
# ----------------------------------------------------------------------
def trimul_contract(left: torch.Tensor, right: torch.Tensor,
                    block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Perform the TriMul contraction (the 4‑D einsum) using the Triton kernel.
    `left` and `right` have shape [B, N, N, H] (dtype float16 or float32).
    Returns a tensor of the same shape (float32).
    """
    # Cast to half‑precision for bandwidth (kernel works with fp16)
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, H = left.shape
    # (B, H, N, N) where the last two axes are the matrix dimensions
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # note the transpose (k, j)

    # flatten batch*head → (BH, N, N)
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    BH = left_flat.shape[0]                    # = B * H
    grid = ( (N + block_m - 1) // block_m,    # blocks along i
             (N + block_n - 1) // block_n,    # blocks along j
             BH )                             # one program for each (b,h)

    # strides for a contiguous (BH, N, N) tensor
    stride_lbh = N * N
    stride_li = N
    stride_lk = 1

    stride_rbh = N * N
    stride_rj = N
    stride_rk = 1

    stride_obh = N * N
    stride_oi = N
    stride_oj = 1

    trimul_outgoing_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_lbh, stride_li, stride_lk,
        stride_rbh, stride_rj, stride_rk,
        stride_obh, stride_oi, stride_oj,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )

    # reshape back to original layout: (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out.to(torch.float32)


# ----------------------------------------------------------------------
# Main entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]            # input feature dimension C
    hidden_dim = config["hidden_dim"]  # projection dimension H
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Initial LayerNorm over the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape [B, N, N]) – broadcasted over H
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask may be bool or float; cast to the same dtype as x
        mask_f = mask.to(dtype=x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating signals (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core tensor contraction via the Triton kernel
    # ------------------------------------------------------------------
    out = trimul_contract(left, right,
                          block_m=64, block_n=64, block_k=32)   # shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm + out‑gate + final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
