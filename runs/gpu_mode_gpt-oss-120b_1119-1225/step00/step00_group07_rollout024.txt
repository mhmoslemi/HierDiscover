@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers
    a_ptrs = a_ptr + batch_id * stride_ah + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bh + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Load a and b tiles
    a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
    b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tile of A and B
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)
        # Increment pointers for next k tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
        offs_k += BLOCK_K

    # Write back
    c_ptrs = c_ptr + batch_id * stride_ch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_m[:, None] & mask_n[None, :])

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block offsets.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Masks for valid rows/cols.
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch.
    a_batch_ptr = a_ptr + batch_id * stride_ah
    b_batch_ptr = b_ptr + batch_id * stride_bh

    # Initialize accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offset for the current tile on the K axis.
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Masks for K dimension.
        mask_k = k_offset < K

        # Load A tile [M, K] and B tile [K, N].
        a = tl.load(
            a_batch_ptr + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch_ptr + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Write back to C.
    c_batch_ptr = c_ptr + batch_id * stride_ch
    c = c_batch_ptr + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def custom_kernel(data):
    """
    Triton implementation of the “outgoing” TriMul operator.
    It follows the PyTorch reference:
    1. Layer‑norm over the last dimension.
    2. Linear projections (left/right) and gating.
    3. Optional mask applied before gating.
    4. Core O(N³) operation computed as a batched matrix multiplication
       using a custom Triton kernel.
    5. Post‑norm, out‑gate, and final linear projection.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm over the channel dimension
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask shape [B, N, N] -> [B, N, N, 1]
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape  # (B,N,N,hidden_dim)

    # Transpose right on the second and third axes so that the contraction becomes a matmul.
    # Shape after permute: (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    # Merge batch and hidden dimensions for the kernel.
    batch_hidden = left_perm.shape[0] * left_perm.shape[1]  # B * hidden_dim
    A = left_perm.view(batch_hidden, N, N)
    B_mat = right_perm.view(batch_hidden, N, N)

    # Output buffer
    C = torch.empty_like(A, dtype=torch.float32)

    # Choose tile sizes – they work well for the given workloads on H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Launch the kernel
    grid = (batch_hidden,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        N, N, N,
        A.stride(1),  # stride_am (row stride)
        A.stride(2),  # stride_ak (col stride)
        A.stride(0),  # stride_ah (batch stride)
        B_mat.stride(1),  # stride_bk (K stride)
        B_mat.stride(2),  # stride_bn (N stride)
        B_mat.stride(0),  # stride_bh (batch stride)
        C.stride(1),  # stride_cm
        C.stride(2),  # stride_cn
        C.stride(0),  # stride_ch
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, hidden_dim, N, N) and permute to (B, N, N, hidden_dim)
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # Post processing: layer norm, out gate, final linear projection
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for this block.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch.
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh

    # Accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)

    # Write back.
    c_batch = c_ptr + batch_id * stride_ch
    c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass accelerated with Triton.

The kernel implements the O(N³) contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
by treating each hidden channel as an independent matrix multiplication:
    left   → (B, D, N, N)   (i‑k)
    right  → (B, D, N, N)   (j‑k) → transpose on (k, j) → (B, D, N, N) (k‑j)
    out    → batched matmul   left @ rightᵀ  → (B, D, N, N)

All other operations (LayerNorm, Linear projections, gating, mask handling) are
performed with PyTorch; the expensive matmul is executed by a custom Triton
kernel that fuses the K‑loop and uses fp32 accumulation while supporting
float16 inputs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                # pointers
    M, N, K,                            # matrix sizes (all = seq_len)
    stride_am, stride_ak, stride_ah,    # A: row‑stride, col‑stride, batch‑stride
    stride_bk, stride_bn, stride_bh,    # B: row‑stride (K), col‑stride (N), batch‑stride
    stride_cm, stride_cn, stride_ch,    # C: row‑stride, col‑stride, batch‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B  (A shape = [batch, M, K], B shape = [batch, K, N])"""
    batch_id = tl.program_id(0)            # which (B×D) matrix
    pid_m = tl.program_id(1)               # block tile row
    pid_n = tl.program_id(2)               # block tile col

    # Tile start indices.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch.
    a_batch = a_ptr + batch_id * stride_ah
    b_batch = b_ptr + batch_id * stride_bh

    # Accumulator in float32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension tiles.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load A tile [M, K] and B tile [K, N].
        a = tl.load(
            a_batch + offs_m[:, None] * stride_am + k_offset[None, :] * stride_ak,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        b = tl.load(
            b_batch + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bn,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)   # fp32 accumulation

    # Write the result.
    c_batch = c_ptr + batch_id * stride_ch
    c = c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]   (torch.Tensor on CUDA)
            - mask_tensor : [B, N, N]        (torch.Bool or float, may be all‑ones)
            - weights_dict: mapping of weight names to tensors (see skeleton)
            - config_dict : must contain "dim", "hidden_dim", and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Input layer‑norm (over the last dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating (sigmoid) and application
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape   # batch size, sequence length

    # -----------------------------------------------------------------
    # 5. Core contraction via batched GEMM (Triton)
    # -----------------------------------------------------------------
    #   left  : (B, N, N, D) → (B, D, N, N)  (i‑k)
    #   right : (B, N, N, D) → (B, D, N, N) with axes (k‑j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()               # (B, D, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()              # (B, D, N, N)

    # Merge batch and channel dimensions for the kernel.
    batch_hidden = left_perm.shape[0] * left_perm.shape[1]           # B * hidden_dim
    A = left_perm.view(batch_hidden, N, N)                           # [BH, N, N]
    B_mat = right_perm.view(batch_hidden, N, N)                      # [BH, N, N]

    # Output buffer (float32 accumulation output)
    C = torch.empty_like(A, dtype=torch.float32)

    # Tile sizes (tuned for H100; works well for the tested N up to 1024)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        A,
        B_mat,
        C,
        N, N, N,                           # M, N, K
        A.stride(1), A.stride(2), A.stride(0),   # a strides: row, col, batch
        B_mat.stride(1), B_mat.stride(2), B_mat.stride(0),   # b strides
        C.stride(1), C.stride(2), C.stride(0),   # c strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back: [B, D, N, N] → [B, N, N, D]
    out = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 6. Post‑norm, out‑gate, final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
