        acc += tl.dot(a, b)

    tl.store(o_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implementation for AlphaFold3‑style models.

The heavy O(N³·C_hidden) reduction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
is performed with a custom Triton kernel.  All other
operations (LayerNorms, Linear projections, gating, masking)
are executed with PyTorch, which keeps the code simple while
still allowing the core tensor contraction to be fully fused
and run on Tensor‑cores of an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    L_ptr, R_ptr, O_ptr,
    B, N, H,                     # batch, seq_len, hidden_dim
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
    stride_r_batch, stride_r_k, stride_r_j, stride_r_h,
    stride_o_batch, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication for the TriMul contraction.

    For each (b, h) pair we compute:
        out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    where `right` is accessed transposed (k ↔ j) so that a standard
    GEMM kernel can be reused.
    """

    # ----- program ids -------------------------------------------------------
    pid = tl.program_id(0)                     # linear id = b * H + h
    b = pid // H                               # batch index
    h = pid - b * H                            # hidden channel index

    pid_i = tl.program_id(1)                   # block row (i)
    pid_j = tl.program_id(2)                   # block col (j)

    # ----- offsets in each dimension ----------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # d‑offset (hidden dimension is contiguous)
    off_h_l = h * stride_l_h
    off_h_r = h * stride_r_h
    off_h_o = h * stride_o_h

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction axis k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ----------- load a tile from left (A) -----------------------------
        a_ptrs = (
            L_ptr
            + b * stride_l_batch
            + offs_i[:, None] * stride_l_i
            + cur_k[None, :] * stride_l_k
            + off_h_l
        )
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)                     # promote for accumulation

        # ----------- load a tile from right (B) (transposed) -------------
        # right is accessed as B[k, j] i.e. (row = k, col = j)
        b_ptrs = (
            R_ptr
            + b * stride_r_batch
            + cur_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
            + off_h_r
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # ----------- matrix multiply and accumulate -----------------------
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # ----------- write result -------------------------------------------------
    o_ptrs = (
        O_ptr
        + b * stride_o_batch
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + off_h_o
    )
    tl.store(o_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (no gradients).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C_in]
        - mask         : torch.Tensor, shape [B, N, N]  (may be None)
        - weights      : dict of torch.Tensor containing all learnable parameters
        - config       : dict with keys "dim" (C_in) and "hidden_dim" (C_hidden)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C_in] – the TriMul output.
    """
    # ------------------------------------------------------------------ unpack
    input_tensor, mask, weights, config = data
    B, N, _, C_in = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    C_out = config["dim"]       # same as C_in in the reference implementation

    # ----------------------------------------------------- LayerNorm (input)
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (C_in,), weight=norm_weight, bias=norm_bias)

    # -------------------------------------------------------- Linear proj.
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, C_in)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, C_in)

    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # -------------------------------------------------------------- Mask
    if mask is not None:
        # mask is float (0/1) or bool; ensure same dtype as left/right
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------ Gates
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # shape [B,N,N,hidden_dim]

    left  = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------- Tri‑Mul kernel
    # output buffer before the final LayerNorm + linear
    out = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # Strides (in elements) for the tensors – they are contiguous on GPU
    stride_l_batch, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_batch, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_batch, stride_o_i, stride_o_k, stride_o_h = out.stride()

    # For the right operand we need the transposed layout (k ↔ j)
    #   row  = k  (original third dim) -> stride_r_k
    #   col  = j  (original second dim) -> stride_r_i
    #   hidden dim stride stays stride_r_h
    # The kernel expects these in the order (batch, row, col, hidden)
    #   where row = k, col = j.
    # We therefore pass stride_r_k as the “row‑stride’’ and stride_r_i as the
    # “col‑stride’’.
    stride_r_row = stride_r_k
    stride_r_col = stride_r_i

    # ------------------------------------------------------------------- launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    # each (b, h) pair gets its own program ID on the first axis
    total_programs = B * hidden_dim
    grid = (total_programs, num_blocks_m, num_blocks_n)

    trimul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_l_batch, stride_l_i, stride_l_k, stride_l_h,
        stride_r_batch, stride_r_row, stride_r_col, stride_r_h,
        stride_o_batch, stride_o_i, stride_o_k, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # sensible default for 64×64 tiles on H100
    )

    # ------------------------------------------------------ post‑processing
    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # gating
    out = out * out_gate

    # final linear projection back to C_in dimensions
    to_out_weight = weights["to_out.weight"]   # (C_in, hidden_dim)
    out = F.linear(out, to_out_weight)        # [B, N, N, C_in]

    return out
```
