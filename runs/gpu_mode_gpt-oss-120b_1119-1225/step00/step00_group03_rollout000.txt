def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
    # weight tensors shape: weight for linear layers etc.

    # Activation:
    eps = 1e-5
    # LayerNorm on x
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Compute left proj & right proj
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # out_features=hidden_dim
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask if needed
    if not nomask:
        m = mask.unsqueeze(-1).to(left.dtype)  # shape B,N,N,1
        left = left * m
        right = right * m

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Cast left/right to fp16
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Prepare for Triton kernel
    B, N, _, _ = left.shape
    batch_mul = B * hidden_dim

    # Permute to (B, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # shape [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch and hidden
    left = left.view(batch_mul, N, N)
    right = right.view(batch_mul, N, N)

    # Output allocate
    out = torch.empty_like(left, dtype=torch.float16) # shape same.

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_mul)

    # Use pointers
    # compute strides for left: left.stride()
    stride_am, stride_ai, stride_ak = left.stride()
    # For right: stride for batch, j, k
    stride_bm, stride_bj, stride_bk = right.stride()
    # For output: same as left
    stride_cm, stride_ci, stride_cj = out.stride()

    triton_tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bj, stride_bk,
        stride_cm, stride_ci, stride_cj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    # Reshape out back to (B,N,N,hidden)
    out = out.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

"""
TriMul “outgoing” forward pass implemented with a fused Triton kernel.

The heavy step of TriMul is the cubic‑complex Einstein sum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be written as a batched matrix multiplication
    out[b,:, :, d] = left_proj[b,:, :, d] @ right_proj[b,:, :, d].T .
For every (batch, hidden) pair we therefore need a GEMM of shape
(N, N) × (N, N)ᵀ.  This kernel performs all those GEMMs in half‑precision
(FP16) with accumulation in FP32, handling arbitrary N (up to 1024) and
any batch × hidden product.  The surrounding LayerNorms, linear
projections and gating are done with PyTorch which is cheap compared to
the O(N³·hidden) work.

The kernel works on three dimensions:
    pid_m : block of rows (i)
    pid_n : block of columns (j)
    pid_b : batch * hidden index
and uses classic tiling (BLOCK_M × BLOCK_N tiles, iterating over K in
BLOCK_K steps).  Boundary handling is performed with masks.
"""

import torch
import triton
import triton.language as tl
from torch.nn.functional import layer_norm, linear


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM to compute the trianglular multiplicative
# update (left @ rightᵀ) for all batch*hidden slices.
# ----------------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    a_ptr,                     # const * half  (left)
    b_ptr,                     # const * half  (right)
    c_ptr,                     #       * half  (output)
    N,                         # i32  matrix dimension (seq_len)
    stride_am, stride_ai, stride_ak,   # i64 strides for a (batch, i, k)
    stride_bm, stride_bj, stride_bk,   # i64 strides for b (batch, j, k)
    stride_cm, stride_ci, stride_cj,   # i64 strides for c (batch, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(axis=2)          # batch * hidden
    pid_m = tl.program_id(axis=0)          # block row
    pid_n = tl.program_id(axis=1)          # block column

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over K dimension (the contracted axis)
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        cur_k = k * BLOCK_K + offs_k
        mask_k = cur_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_b * stride_am
            + offs_m[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=(mask_m[:, None] & mask_k[None, :]),
            other=0.0,
        )

        # Load B tile transposed: we need B_T[k, j] = B[j, k]
        # Original B layout: (batch, j, k) → strides (stride_bm, stride_bj, stride_bk)
        # B_T uses stride_bj for the K‑dimension (row) and stride_bk for the N‑dimension (col).
        b_ptrs = (
            b_ptr
            + pid_b * stride_bm
            + cur_k[:, None] * stride_bj   # row offset uses original j‑stride
            + offs_n[None, :] * stride_bk  # col offset uses original k‑stride
        )
        b = tl.load(
            b_ptrs,
            mask=(mask_k[:, None] & mask_n[None, :]),
            other=0.0,
        )

        # FP16 dot, accumulation in FP32
        acc += tl.dot(a, b)

    # Write back the result tile
    c_ptrs = (
        c_ptr
        + pid_b * stride_cm
        + offs_m[:, None] * stride_ci
        + offs_n[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=(mask_m[:, None] & mask_n[None, :]),
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]   (float32)
        - mask         : torch.Tensor[B, N, N]       (bool or 0/1)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys:
            * "dim"        – input channel dimension
            * "hidden_dim" – hidden dimension used in the projections
            * "nomask"     – True → ignore mask, False → apply mask

    Returns
    -------
    torch.Tensor[B, N, N, dim]
        Result of the TriMul forward computation (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    (inp, mask, weights, config) = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input
    # ------------------------------------------------------------------
    eps = 1e-5
    x = layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  FP32

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = linear(x, weights["left_proj.weight"])    # [B, N, N, hidden]
    right = linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1]
        m = mask.to(x.dtype).unsqueeze(-1)
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast projections to FP16 for the cubic GEMM
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # Rearrange to (B, hidden, N, N) → flatten batch*hidden for kernel launch
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()
    batch_hidden = B * hidden
    left = left.view(batch_hidden, N, N)
    right = right.view(batch_hidden, N, N)

    # Output buffer (FP16)
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides (in elements, not bytes) for the three tensors
    stride_am, stride_ai, stride_ak = left.stride()     # (batch, i, k)
    stride_bm, stride_bj, stride_bk = right.stride()   # (batch, j, k)
    stride_cm, stride_ci, stride_cj = out.stride()     # (batch, i, j)

    # ------------------------------------------------------------------
    # Triton GEMM launch
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch_hidden,
    )
    triton_tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original layout (float32)
    # ------------------------------------------------------------------
    out = out.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # bring back to FP32 for the rest

    # ------------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                 # element‑wise gating
    out = linear(out, weights["to_out.weight"])   # back to [B, N, N, dim]

    return out
