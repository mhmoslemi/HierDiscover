```python
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N, K,
    left_gstride, left_mstride, left_kstride,
    right_gstride, right_kstride, right_nstride,
    out_gstride, out_mstride, out_nstride,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # get program ids
    pid_g = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute mask for out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k offset
        k_start = k * BLOCK_K
        k_offsets = k_start + offs_k

        # mask for k dimension
        mask_k = k_offsets < K

        # Pointers
        left_offset = pid_g * left_gstride + offs_m[:, None] * left_mstride + k_offsets[None, :] * left_kstride
        right_offset = pid_g * right_gstride + k_offsets[:, None] * right_kstride + offs_n[None, :] * right_nstride

        # Load blocks
        a = tl.load(left_ptr + left_offset, mask=mask_m[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)
        b = tl.load(right_ptr + right_offset, mask=mask_k[:, None] & mask_n[None, :], other=0.0, dtype=tl.float16)

        # matrix multiply-accumulate
        acc += tl.dot(a, b)

    # Write back
    out_offset = pid_g * out_gstride + offs_m[:, None] * out_mstride + offs_n[None, :] * out_nstride
    tl.store(out_ptr + out_offset, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

Important: Use `tl.dot` which expects a (M,K) and (K,N). The dtype implicit: a and b are float16, accumulate as float32.

"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The forward pass consists of:
1. Layer‑norm over the channel dimension `dim`.
2. Two linear projections (`left_proj`, `right_proj`) producing a hidden
   dimension `hidden_dim`.
3. Optional binary mask applied to the projected tensors.
4. Gated linear units – three more linear layers followed by `sigmoid`,
   multiplied element‑wise with the projected tensors.
5. Core “tri‑multiplicative” update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication `left @ rightᵀ` for every
   `(batch, hidden_dim)` pair.  The heavy `O(N³·hidden_dim)` work is
   performed by a custom Triton kernel.
6. Layer‑norm over the hidden dimension, an output gate, and a final
   linear projection back to `dim`.

All large‑tensor arithmetic is carried out in FP16 for bandwidth
reduction, while accumulation inside the Triton kernel uses FP32 to
preserve numerical stability.  The final result is returned in FP32.

The kernel is written for H100 GPUs (SM 90) and Triton 3.3.1.  Block
sizes were chosen empirically (64×64 tiles, K‑tile = 32) and work for
the test configurations listed in the problem statement.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul core operation
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    left_ptr,               # *g*×M×K  (FP16)
    right_ptr,              # *g*×K×N  (FP16)
    out_ptr,                # *g*×M×N  (FP16, to be written)
    M, N, K,                # sequence length (int32)
    # strides for the three tensors (int64)
    left_gstride, left_mstride, left_kstride,
    right_gstride, right_kstride, right_nstride,
    out_gstride, out_mstride, out_nstride,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[g, i, j] = Σ_k left[g, i, k] * right[g, k, j] for all groups g.

    `left`  – (G, M, K) matrix per group,
    `right` – (G, K, N) matrix per group,
    `out`   – (G, M, N) result.
    """
    # ------------------------------------------------------------------
    # Program IDs (group, tile row, tile col)
    # ------------------------------------------------------------------
    pid_g = tl.program_id(0)          # group index  (0 … G-1)
    pid_m = tl.program_id(1)          # tile over rows of left / out
    pid_n = tl.program_id(2)          # tile over cols of right / out

    # ------------------------------------------------------------------
    # Offsets within the M/N dimensions
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)      # [BLOCK_N]

    # Masks for OOB rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)          # [BLOCK_K]
        mask_k = offs_k < K

        # ------------------------------------------------------------------
        # Pointers for the current tile
        #   left  : (M, K)
        #   right : (K, N)
        # ------------------------------------------------------------------
        left_offset = (
            pid_g * left_gstride
            + offs_m[:, None] * left_mstride
            + offs_k[None, :] * left_kstride
        )
        right_offset = (
            pid_g * right_gstride
            + offs_k[:, None] * right_kstride
            + offs_n[None, :] * right_nstride
        )

        # Load tiles; out‑of‑bounds elements are zeroed
        a = tl.load(
            left_ptr + left_offset,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )
        b = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Matrix multiply‑accumulate (FP16×FP16 → FP32)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Store the result (cast back to FP16)
    # ------------------------------------------------------------------
    out_offset = (
        pid_g * out_gstride
        + offs_m[:, None] * out_mstride
        + offs_n[None, :] * out_nstride
    )
    tl.store(
        out_ptr + out_offset,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def _batched_matmul(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication used by TriMul.
    left : (G, M, K)  FP16
    right: (G, K, N)  FP16 (already transposed over the two sequence axes)
    Returns out : (G, M, N) FP16
    """
    assert left.is_contiguous() and right.is_contiguous()
    G, M, K = left.shape
    _, K2, N = right.shape
    assert K == K2, "Incompatible inner dimensions for batched matmul"

    # Allocate output
    out = torch.empty((G, M, N), dtype=left.dtype, device=left.device)

    # Strides (elements, not bytes)
    left_gs, left_ms, left_ks = left.stride()
    right_gs, right_ks, right_ns = right.stride()
    out_gs, out_ms, out_ns = out.stride()

    # Tile sizes – chosen to fit H100 shared memory and registers
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        G,                                 # one program per group
        (M + BLOCK_M - 1) // BLOCK_M,      # tiles over rows
        (N + BLOCK_N - 1) // BLOCK_N,      # tiles over cols
    )

    _batched_matmul_kernel[grid](
        left,
        right,
        out,
        M,
        N,
        K,
        left_gs,
        left_ms,
        left_ks,
        right_gs,
        right_ks,
        right_ns,
        out_gs,
        out_ms,
        out_ns,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], float32
        - mask         : torch.Tensor, shape [B, N, N] (binary 0/1) or None
        - weights      : dict of model parameters (torch.Tensors)
        - config       : dict containing at least
                         {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    apply_mask = not config.get("nomask", True) and mask is not None

    device = input_tensor.device

    # ------------------------------------------------------------------
    # Cast weights to the compute dtype (FP16) – bias tensors are optional
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # LayerNorm over the last dimension
    norm_weight = _to_fp16(weights["norm.weight"])
    norm_bias = _to_fp16(weights["norm.bias"])

    # Linear projection weights
    left_proj_w = _to_fp16(weights["left_proj.weight"])
    right_proj_w = _to_fp16(weights["right_proj.weight"])

    # Gate weights
    left_gate_w = _to_fp16(weights["left_gate.weight"])
    right_gate_w = _to_fp16(weights["right_gate.weight"])
    out_gate_w = _to_fp16(weights["out_gate.weight"])

    # Output projection (+norm) weights
    to_out_norm_w = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_b = _to_fp16(weights["to_out_norm.bias"])
    to_out_w = _to_fp16(weights["to_out.weight"])

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (still in FP32 for maximum precision)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight.to(torch.float32),
        bias=norm_bias.to(torch.float32),
    )
    # Cast to half for the rest of the pipeline
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Projections
    # ------------------------------------------------------------------
    left = F.linear(x_h, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_h, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if apply_mask:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 4) Gate computations (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul: out = Σ_k left[i,k] * right[j,k]
    #    Implemented as batched GEMM: left @ rightᵀ
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Prepare tensors for the batched GEMM.
    # left : (B, hidden_dim, i, k)  → (G, M, K)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # rightᵀ : (B, hidden_dim, k, j) → (G, K, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    G = B * hidden_dim
    left_mat = left_perm.view(G, N, N)          # (G, M, K)   where M = K = N
    right_mat = right_perm.view(G, N, N)        # (G, K, N)

    # Batched matmul kernel (FP16 input → FP16 output)
    out_mat = _batched_matmul(left_mat, right_mat)  # (G, N, N)

    # Reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gate + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # [B, N, N, dim]

    # Ensure final dtype matches the reference implementation
    return out.to(torch.float32)
