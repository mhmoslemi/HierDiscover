```python
import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(lhs_ptr, rhs_ptr, out_ptr,
                   M, N, K,
                   stride_lhs_batch, stride_lhs_m, stride_lhs_k,
                   stride_rhs_batch, stride_rhs_k, stride_rhs_n,
                   stride_out_batch, stride_out_m, stride_out_n,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_m = tl.program_id(0)  # block row index (i)
    pid_n = tl.program_id(1)  # block col index (j)

    # Compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Mask offsets within dimensions
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = lhs_ptr + pid_b * stride_lhs_batch + offs_m[:, None] * stride_lhs_m + offs_k[None, :] * stride_lhs_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile: shape (BLOCK_K, BLOCK_N) (since rhs is K x N)
        b_ptrs = rhs_ptr + pid_b * stride_rhs_batch + offs_k[:, None] * stride_rhs_k + offs_n[None, :] * stride_rhs_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Dot product
        acc += tl.dot(a, b)

        # Advance pointers (not needed due to recomputed each iter)
    # Write back result
    c_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(c_ptrs, acc.to(tl.float16) if acc.dtype == tl.float16 else acc, mask=mask_m[:, None] & mask_n[None, :])
```

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out

"""
TriMul (outgoing) forward implementation.

The module works on a 4‑D tensor X of shape [B, N, N, C] and performs

1. LayerNorm over the channel dimension C.
2. Two bias‑free linear projections (left / right) to a hidden dimension H.
3. Optional masking of the projections.
4. Three gating projections (left_gate, right_gate, out_gate) followed by a
   sigmoid and element‑wise multiplication.
5. The core TriMul operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batch of matrix‑multiplications (one per hidden channel).
   This O(N³·H) step is executed by a custom Triton kernel that uses
   tensor‑core‑friendly FP16 loads and accumulates in FP32.
6. LayerNorm over the hidden dimension, multiplication by out_gate and a final
   linear projection back to the original channel dimension C.

Only the expensive N³ contraction is done in Triton; all other ops use
high‑performance PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel for the batch matrix multiplication required by TriMul.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    lhs_ptr,          # pointer to left tensor  (batch, M, K)   – fp16
    rhs_ptr,          # pointer to right tensor (batch, K, N)   – fp16
    out_ptr,          # pointer to output      (batch, M, N)   – fp16
    BATCH, M, N, K,   # matrix sizes
    stride_lhs_batch, stride_lhs_m, stride_lhs_k,
    stride_rhs_batch, stride_rhs_k, stride_rhs_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul:  out[b, i, j] = Σₖ lhs[b, i, k] * rhs[b, k, j]"""
    pid_batch = tl.program_id(2)          # which (batch * hidden) matrix
    pid_m = tl.program_id(0)              # tile row
    pid_n = tl.program_id(1)              # tile column

    # ------------------------------------------------------------------
    # Compute offsets for the current tile.
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a tile of LHS: (BLOCK_M, BLOCK_K)
        a_ptrs = lhs_ptr + \
                 pid_batch * stride_lhs_batch + \
                 offs_m[:, None] * stride_lhs_m + \
                 offs_k[None, :] * stride_lhs_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load a tile of RHS: (BLOCK_K, BLOCK_N)
        b_ptrs = rhs_ptr + \
                 pid_batch * stride_rhs_batch + \
                 offs_k[:, None] * stride_rhs_k + \
                 offs_n[None, :] * stride_rhs_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate the dot product.
        acc += tl.dot(a, b)

    # Write the result back.
    c_ptrs = out_ptr + \
             pid_batch * stride_out_batch + \
             offs_m[:, None] * stride_out_m + \
             offs_n[None, :] * stride_out_n
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                  # C
    hidden_dim = config["hidden_dim"]    # H
    eps = 1e-5

    # --------------------------------------------------------------
    # Helper to fetch a weight tensor (already on device)
    # --------------------------------------------------------------
    def _w(name):
        return weights[name].to(device)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the input channel dimension.
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=_w("norm.weight"),
        bias=_w("norm.bias"),
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, _w("left_proj.weight"))   # [B, N, N, H]
    right = F.linear(x_norm, _w("right_proj.weight"))  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask_tensor.to(dtype).unsqueeze(-1)    # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gates (sigmoid of bias‑free linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, _w("left_gate.weight")))
    right_gate = torch.sigmoid(F.linear(x_norm, _w("right_gate.weight")))
    out_gate = torch.sigmoid(F.linear(x_norm, _w("out_gate.weight")))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul contraction – performed with a fused Triton kernel.
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) so that the contraction becomes a batched matmul:
    #   left_perm  : (b, d, i, k)
    #   right_T    : (b, d, k, j)  <-- transpose of the last two axes of `right`
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_T = right.permute(0, 3, 2, 1).contiguous()          # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_flat = B * H                                             # treat each (b,d) as a separate batch

    # Cast to FP16 for the kernel (FP16 -> FP32 accumulation inside the kernel)
    lhs_fp16 = left_perm.view(batch_flat, N, N).to(torch.float16)
    rhs_fp16 = right_T.view(batch_flat, N, N).to(torch.float16)
    out_fp16 = torch.empty_like(lhs_fp16)

    # Strides after the view – they are identical for lhs, rhs and out
    stride_lhs_batch, stride_lhs_m, stride_lhs_k = lhs_fp16.stride()
    stride_rhs_batch, stride_rhs_k, stride_rhs_n = rhs_fp16.stride()
    stride_out_batch, stride_out_m, stride_out_n = out_fp16.stride()

    # Block size choice – balanced for H100 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # rows (i)
        (N + BLOCK_N - 1) // BLOCK_N,          # cols (j)
        batch_flat,                            # independent (b, d) matrix
    )

    _tri_mul_kernel[grid](
        lhs_fp16,
        rhs_fp16,
        out_fp16,
        batch_flat,
        N,
        N,
        N,
        stride_lhs_batch,
        stride_lhs_m,
        stride_lhs_k,
        stride_rhs_batch,
        stride_rhs_k,
        stride_rhs_n,
        stride_out_batch,
        stride_out_m,
        stride_out_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32 and restore the original layout
    out = out_fp16.to(dtype).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: hidden‑dim layer‑norm, out_gate, final projection.
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=_w("to_out_norm.weight"),
        bias=_w("to_out_norm.bias"),
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, _w("to_out.weight"))   # back to [B, N, N, C]

    return out
```
