def custom_kernel(data):
    input_tensor, mask, weights, config = data
    
    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    
    # Retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Compute layernorm over dim
    x = input_tensor
    # Compute mean/var over last dimension
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias
    
    # Linear projections (float32)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # -> [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    
    # Apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_exp
        right = right * mask_exp
    
    # Compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate
    
    # Cast left/right to float16 for faster GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    
    # Permute to bring hidden dimension before N,N
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]   (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, hidden, N, N]   (k,j)
    
    total_batch = B * hidden_dim
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)
    
    # allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)  # [B*hidden, N, N]
    
    # Compute strides
    def get_strides(t):
        return [s * t.element_size() for s in t.stride()]
    
    stride_a_batch, stride_a_m, stride_a_k = get_strides(left_flat)
    stride_b_batch, stride_b_k, stride_b_n = get_strides(right_flat)
    stride_c_batch, stride_c_m, stride_c_n = get_strides(out_flat)
    
    # launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    import math
    total_tiles_m = math.ceil(N / BLOCK_M)
    total_tiles_n = math.ceil(N / BLOCK_N)
    total_programs = total_batch * total_tiles_m * total_tiles_n
    
    batched_gemm_kernel[(total_programs,)](
        left_flat,
        right_flat,
        out_flat,
        total_batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    
    # Reshape output back to [B, N, N, hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    
    # to_out_norm (layernorm over hidden_dim)
    out = (out - out.mean(dim=-1, keepdim=True)) / torch.sqrt(out.var(dim=-1, unbiased=False, keepdim=True) + eps)
    out = out * to_out_norm_weight + to_out_norm_bias
    
    # out_gate multiplication
    out = out * out_gate
    
    # final linear to dim
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

        mask_k = offs_k < K

        # load A tile
        a = tl.load(a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        # load B tile (row = k, col = n)
        b = tl.load(b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot: result is float32
        acc += tl.dot(a, b)

    # write out
    c = acc
    tl.store(c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n,
             c,
             mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    Implements:
        x_norm = LayerNorm(x)
        left = Linear(x_norm, left_proj_weight) * mask
        right = Linear(x_norm, right_proj_weight) * mask
        left_gate = sigmoid(Linear(x_norm, left_gate_weight))
        right_gate = sigmoid(Linear(x_norm, right_gate_weight))
        out_gate = sigmoid(Linear(x_norm, out_gate_weight))
        left = left * left_gate
        right = right * right_gate
        out = batched_gemm(left, right)   # sum over k dimension via matrix multiplication
        out = LayerNorm(out, to_out_norm_weight, to_out_norm_bias)
        out = out * out_gate
        output = Linear(out, to_out_weight)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    import math
    input_tensor, mask, weights, config = data

    # Config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # Weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm over the input feature dimension
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if available
    if mask is not None and mask.numel() > 0 and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for faster GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # rearrange for batched GEMM: left(i,k), right(k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N] -> (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, hidden, N, N] -> (k,j)

    total_batch = B * hidden_dim

    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    # Output tensor for GEMM result
    out_flat = torch.empty(total_batch, N, N, dtype=torch.float32, device=left.device)

    # Strides in elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_tiles_m = math.ceil(N / BLOCK_M)
    total_tiles_n = math.ceil(N / BLOCK_N)
    total_programs = total_batch * total_tiles_m * total_tiles_n

    batched_gemm_kernel[(total_programs,)](
        left_flat,
        right_flat,
        out_flat,
        total_batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B,N,N,hidden]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                                         bias=to_out_norm_bias, eps=eps)

    # Apply output gate (broadcast)
    out = out * out_gate

    # Final linear projection back to dim
    output = torch.nn.functional.linear(out, to_out_weight)

    return output

"""
TriMul (outgoing) forward pass with a Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑normalize the input tensor over the last dimension (dim).
2. Linear projections to a hidden dimension (left_proj, right_proj) – no bias.
3. (Optional) Apply the pairwise mask (broadcast to hidden dim).
4. Compute three gating vectors with sigmoid (left_gate, right_gate, out_gate).
5. Multiply left/right by their respective gates.
6. Cast left/right to fp16 and reshape:
   * left  – [B, N, N, H] → [B*H, N, N]   (rows = i, cols = k)
   * right – [B, N, N, H] → [B*H, N, N]   (rows = k, cols = j) by swapping the
     last two axes before flattening.
7. Run a Triton kernel that performs a batched matrix multiply
   out[b,i,j] = Σ_k left[b,i,k] * right[b,k,j]   →   O(B·H·N³) work.
8. Reshape the result back to [B, N, N, H].
9. Layer‑normalize over the hidden dimension, apply out_gate, and project back
   to the original feature dimension with a final linear layer.
The heavy O(N³·H) contraction is fully fused in the Triton kernel.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ B   (A: M×K, B: K×N) for many batches.
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,               # *float16   (batch, M, K)
    b_ptr,               # *float16   (batch, K, N)
    c_ptr,               # *float32   (batch, M, N)
    batch,               # i32 total number of batches   (B*H)
    M, N, K,             # i32 matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Tile grid per batch
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_tiles_m * num_tiles_n

    batch_idx = pid // tiles_per_batch
    tile_idx = pid % tiles_per_batch
    pid_m = tile_idx // num_tiles_n
    pid_n = tile_idx % num_tiles_n

    # Offsets within the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the current batch
    a_batch = a_ptr + batch_idx * stride_a_batch
    b_batch = b_ptr + batch_idx * stride_b_batch
    c_batch = c_ptr + batch_idx * stride_c_batch

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M×K)
        a = tl.load(
            a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K×N) – note B is stored as (row = k, col = n)
        b = tl.load(
            b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Dot product (tensor cores accumulate in fp32)
        acc += tl.dot(a, b)

    # Write back the result tile
    c = acc
    tl.store(
        c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n,
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Triton‑enabled TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] or None
        - weights      : dict of model weights (see code)
        - config       : dict with keys "dim", "hidden_dim", optional "eps", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Config and weights
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm over the feature dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating vectors (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to fp16 for the heavy O(N³·H) contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 7. Rearrange for batched GEMM: left(i,k) , right(k,j)
    #    - left_perm  : [B, H, N, N] with axes (i,k)
    #    - right_perm : [B, H, N, N] with axes (k,j)
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    total_batch = B * hidden_dim  # B * H
    left_flat = left_perm.view(total_batch, N, N)   # [B*H, N, N]
    right_flat = right_perm.view(total_batch, N, N) # [B*H, N, N]

    # Output tensor for the GEMM result (float32)
    out_flat = torch.empty(
        total_batch, N, N, dtype=torch.float32, device=left.device
    )

    # ------------------------------------------------------------------
    # 8. Strides (in elements, not bytes) for Triton indexing
    # ------------------------------------------------------------------
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # ------------------------------------------------------------------
    # 9. Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    tiles_m = math.ceil(N / BLOCK_M)
    tiles_n = math.ceil(N / BLOCK_N)
    total_programs = total_batch * tiles_m * tiles_n

    batched_gemm_kernel[
        (total_programs,)
    ](
        left_flat,
        right_flat,
        out_flat,
        total_batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Reshape back to [B, N, N, hidden]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 11. Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 12. Apply output gate and final projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    output = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return output
