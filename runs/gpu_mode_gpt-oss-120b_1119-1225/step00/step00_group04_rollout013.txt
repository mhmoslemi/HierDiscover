"""
TriMul (outgoing) forward pass accelerated with Triton.

The implementation follows the reference PyTorch TriMul module:
1. Layer‑norm on the input.
2. Two linear projections (left / right) + three linear gates.
3. Optional pairwise mask applied to the projected tensors.
4. Gating (sigmoid) and casting to FP16.
5. Core contraction  out[i,j,d] = Σₖ left[i,k,d] * right[j,k,d]  performed by a
   custom Triton kernel (i‑ and j‑tiles, K‑reduction, D‑blocked loop).
6. Layer‑norm on the contracted tensor, element‑wise multiplication with
   the output‑gate and a final linear projection back to the original dimension.

Only the heavy O(N³·D) contraction is executed on the GPU via Triton; all
other (mostly linear) operations are done with PyTorch which keeps the code
short and easy to verify. The kernel works for any batch size (≤2), any
sequence length (≤1024) and hidden dimensions that are multiples of the
chosen D‑tile size (fallback masked for the remainder).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel for the 3‑tensor contraction:
#   out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, k, j, d]
# left : [B, N, N, D]  (i,k)
# right: [B, N, N, D]  (k,j)  – note the swapped indices for the reduction.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, D,                               # problem sizes
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides (b,i,k,d)
    stride_rb, stride_ri, stride_rj, stride_rd,   # right strides (b,k,j,d) – we use ri for the original i‑stride
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides (b,i,j,d)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(2)          # batch dimension
    pid_i = tl.program_id(0)          # block of i
    pid_j = tl.program_id(1)          # block of j

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    offs_d = tl.arange(0, BLOCK_D)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # Base pointers for the given batch
    left_base  = left_ptr  + pid_b * stride_lb
    right_base = right_ptr + pid_b * stride_rb
    out_base   = out_ptr   + pid_b * stride_ob

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over the K dimension
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k
        mask_k = cur_k < N

        # left[b, i, k, d]   -> shape (BLOCK_M, BLOCK_K, BLOCK_D)
        left_offset = (
            left_base
            + offs_i[:, None, None] * stride_li
            + cur_k[None, :, None] * stride_lk
            + offs_d[None, None, :] * stride_ld
        )
        left_tile = tl.load(
            left_offset,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # right[b, k, j, d]  (notice the swapped order) -> shape (BLOCK_K, BLOCK_N, BLOCK_D)
        right_offset = (
            right_base
            + cur_k[:, None, None] * stride_ri
            + offs_j[None, :, None] * stride_rj
            + offs_d[None, None, :] * stride_rd
        )
        right_tile = tl.load(
            right_offset,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ------------------------------------------------------------------
        # For each channel slice d perform a matmul: (M x K) * (K x N)
        # ------------------------------------------------------------------
        for d in range(0, BLOCK_D):
            # skip entirely out‑of‑range d‑slices (helps the compiler)
            if not mask_d[d]:
                continue
            left_mat  = left_tile[:, :, d]   # (BLOCK_M, BLOCK_K)
            right_mat = right_tile[:, :, d]  # (BLOCK_K, BLOCK_N)
            acc[:, :, d] += tl.dot(left_mat, right_mat, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write result back to global memory
    # ------------------------------------------------------------------
    out_offset = (
        out_base
        + offs_i[:, None, None] * stride_oi
        + offs_j[None, :, None] * stride_oj
        + offs_d[None, None, :] * stride_od
    )
    tl.store(
        out_offset,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
    )


# ----------------------------------------------------------------------
# Entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (x, mask, weights, config)
            x      – input tensor of shape [B, N, N, dim] (float32)
            mask   – optional mask tensor of shape [B, N, N] (bool/float)
            weights – dictionary with the model parameters
            config  – dict containing at least:
                      * "dim"         – input channel dimension
                      * "hidden_dim"  – inner hidden dimension
                      * "nomask"      – bool, true if mask should be ignored

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Layer‑norm on the input (dim‑last)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    eps = 1e-5
    x_norm = F.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B,N,N,dim], float32

    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim, no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask may be bool – cast to the same dtype as x_norm
        mask = mask.to(x_norm.dtype)
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating networks (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast the tensors that enter the heavy contraction to FP16
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # Allocate output buffer (FP32) for the contraction
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # ------------------------------------------------------------------
    # Prepare strides for Triton (convert to Python ints)
    # ------------------------------------------------------------------
    left_strides = left.stride()   # (b,i,k,d)
    right_strides = right.stride()
    out_strides = out.stride()

    # left strides
    stride_lb, stride_li, stride_lk, stride_ld = (
        left_strides[0],
        left_strides[1],
        left_strides[2],
        left_strides[3],
    )
    # right strides – we need (b, k, j, d) ordering inside the kernel
    stride_rb, stride_ri, stride_rj, stride_rd = (
        right_strides[0],
        right_strides[1],   # original i‑stride – used for the k‑index in the kernel
        right_strides[2],   # original j‑stride – used for the j‑index
        right_strides[3],
    )
    # out strides
    stride_ob, stride_oi, stride_oj, stride_od = (
        out_strides[0],
        out_strides[1],
        out_strides[2],
        out_strides[3],
    )

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 16   # must divide hidden_dim; remainder handled by masking

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B,
    )

    # ------------------------------------------------------------------
    # launch kernel
    # ------------------------------------------------------------------
    trimul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_ri,
        stride_rj,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    # (torch automatically synchronises on the next CPU‑side operation)

    # ------------------------------------------------------------------
    # Post‑contraction processing (LayerNorm, out‑gate, final Linear)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # Normalize the contracted tensor
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Apply the output gate (shape matches out_norm)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    result = F.linear(out_norm, to_out_weight)   # -> [B,N,N,dim]

    return result
