"""
custom_kernel for the *outgoing* TriMul operation (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Linear projections (no bias):
      left  = x @ W_left_proj   ( → hidden_dim )
      right = x @ W_right_proj  ( → hidden_dim )
3. Optional pair‑wise mask (shape [B,N,N]) is applied to the
   projected tensors before gating.
4. Compute element‑wise gates (sigmoid of linear layers) and
   multiply them into the projected tensors.
   The gated tensors are cast to FP16 to save bandwidth.
5. Compute the cubic “TriMul” contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
      out_batch = left_batch @ right_batchᵀ
   where each batch element corresponds to a (b,d) pair.
   This step is highly parallelised with a custom Triton kernel.
6. Perform the output LayerNorm, multiply by the “out‑gate”
   (sigmoid) and finally project back to the original channel
   dimension with a bias‑less linear layer.
7. Return a float‑32 tensor of shape [B, N, N, C].

Only the cubic contraction (step 5) is executed on the GPU via
Triton; the remaining operations use fast torch primitives.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def tri_mul_kernel(
    a_ptr,                 # left  (FP16)  [batch, N, N]
    b_ptr,                 # right (FP16)  [batch, N, N]
    c_ptr,                 # out   (FP32)  [batch, N, N]
    N,                     # sequence length (int)
    stride_a_batch,        # stride for batch dimension in A
    stride_a_i,            # stride for the row (i)  in A
    stride_a_k,            # stride for the col (k)  in A
    stride_b_batch,        # stride for batch dimension in B
    stride_b_row,          # stride for the row index (k) in Bᵀ
    stride_b_col,          # stride for the col index (j) in Bᵀ
    stride_c_batch,        # stride for batch dimension in C
    stride_c_i,            # stride for the row (i)  in C
    stride_c_j,            # stride for the col (j)  in C
    BLOCK_M: tl.constexpr,  # tile size in M (i) direction
    BLOCK_N: tl.constexpr,  # tile size in N (j) direction
    BLOCK_K: tl.constexpr,  # tile size in K (k) direction
):
    """
    Batched GEMM: for each batch element `b`
        C[b] = A[b] @ B[b]ᵀ
    A/B are FP16, accumulation is FP32.  The kernel works for any
    square N up to the limits of the GPU.
    """
    pid_batch = tl.program_id(axis=0)   # batch index (B * hidden_dim)
    pid_m = tl.program_id(axis=1)       # tile row index
    pid_n = tl.program_id(axis=2)       # tile col index

    # Coordinate ranges of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers to the start of the current batch
    a_batch = a_ptr + pid_batch * stride_a_batch
    b_batch = b_ptr + pid_batch * stride_b_batch
    c_batch = c_ptr + pid_batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # ----- Load A block (M x K) -----
        a_ptrs = a_batch + (offs_m[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        mask_a = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16).to(tl.float32)

        # ----- Load Bᵀ block (K x N) -----
        # Bᵀ is accessed as (k, j) → we use stride_b_row for k and stride_b_col for j
        b_ptrs = b_batch + (offs_k[:, None] * stride_b_row) + (offs_n[None, :] * stride_b_col)
        mask_b = (offs_k[:, None] < N) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16).to(tl.float32)

        # ----- Outer product & accumulate -----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ----- Write result back -----
    c_ptrs = c_batch + (offs_m[:, None] * stride_c_i) + (offs_n[None, :] * stride_c_j)
    mask_c = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel
    for the O(N³) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # -------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------
    x, mask, weights, config = data
    # Shapes / config
    bs, N, _, C = x.shape
    hidden = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------
    # 1. Input layer‑norm
    # -------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # shape [B, N, N, C]  (float32)

    # -------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right_proj = F.linear(x, weights["right_proj.weight"])        # [B,N,N,hidden]

    # -------------------------------------------------
    # 3. Optional mask (applied before gating)
    # -------------------------------------------------
    if mask is not None and not config.get("nomask", True):
        # mask: [B, N, N] → broadcast over hidden dim
        mask = mask.to(left_proj.dtype).unsqueeze(-1)            # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # -------------------------------------------------
    # 4. Gates (sigmoid of linear layers)
    # -------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,hidden]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B,N,N,hidden]

    # -------------------------------------------------
    # 5. Apply gates and cast to FP16 for the heavy contraction
    # -------------------------------------------------
    left = (left_proj * left_gate).to(torch.float16)   # [B,N,N,hidden]  FP16
    right = (right_proj * right_gate).to(torch.float16)  # FP16

    # Free unused tensors early to keep memory low
    del left_proj, right_proj, left_gate, right_gate

    # -------------------------------------------------
    # 6. Prepare tensors for the batched GEMM
    #    We need (batch, N, N) with batch = B * hidden
    # -------------------------------------------------
    # bring hidden dim to the second axis → (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten (B, hidden) → batch dimension
    left_batch = left_perm.view(-1, N, N)   # shape [B*hidden, N, N]   FP16
    right_batch = right_perm.view(-1, N, N) # FP16

    # output buffer (FP32)
    out_batch = torch.empty_like(left_batch, dtype=torch.float32)

    batch_dim = left_batch.shape[0]

    # -------------------------------------------------
    # 7. Launch the Triton kernel
    # -------------------------------------------------
    # Block sizes – can be tuned; they fit well on an H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Strides (in elements, not bytes)
    stride_a_batch = left_batch.stride(0)   # N*N
    stride_a_i = left_batch.stride(1)       # N
    stride_a_k = left_batch.stride(2)       # 1

    stride_b_batch = right_batch.stride(0)  # N*N
    stride_b_i = right_batch.stride(1)      # N  (row stride)
    stride_b_k = right_batch.stride(2)      # 1  (col stride)

    # For Bᵀ we exchange the role of i/k
    stride_b_row = stride_b_k               # 1
    stride_b_col = stride_b_i               # N

    stride_c_batch = out_batch.stride(0)    # N*N
    stride_c_i = out_batch.stride(1)        # N
    stride_c_j = out_batch.stride(2)        # 1

    tri_mul_kernel[grid](
        a_ptr=left_batch,
        b_ptr=right_batch,
        c_ptr=out_batch,
        N=N,
        stride_a_batch=stride_a_batch,
        stride_a_i=stride_a_i,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_row=stride_b_row,
        stride_b_col=stride_b_col,
        stride_c_batch=stride_c_batch,
        stride_c_i=stride_c_i,
        stride_c_j=stride_c_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # typical for FP16 matmul on H100
    )
    # -------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden]
    # -------------------------------------------------
    out = out_batch.view(bs, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # float32

    # -------------------------------------------------
    # 9. Output LayerNorm, gating, final projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )   # [B,N,N,hidden]

    out = out * out_gate   # element‑wise gating (float32)

    # final linear projection back to `dim` channels
    output = F.linear(out, weights["to_out.weight"])   # bias is False in the original module

    return output
```
