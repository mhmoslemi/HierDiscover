def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6  # use typical.

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    # Note: linear without bias; implement via torch.nn.functional.linear
    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if there is mask (mask shape (B,N,N) maybe bool or float)
    if mask is not None and mask.numel() > 0:
        # mask[..., None] multiply each hidden
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Compute out using Triton kernel
    out = tri_mul_kernel(left, right, BLOCK_I=..., BLOCK_J=..., BLOCK_K=..., BLOCK_H=...)  # returns shape (B,N,N,hidden_dim)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # multiply with out_gate
    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # shape (B,N,N,dim)

    return out
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    eps = 1e-6

    # --- LayerNorm ---
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # --- Projections ---
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --- Mask ---
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)  # broadcast hidden dimension
        left = left * mask_f
        right = right * mask_f

    # --- Gates ---
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Convert left/right to half for Triton
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # --- Triton contraction ---
    out_h = tri_mul_triton(left_h, right_h, BLOCK_I=32, BLOCK_J=32, BLOCK_K=64, BLOCK_H=4)

    # Back to float32
    out = out_h.to(torch.float32)

    # --- to_out_norm and gating ---
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_i_l, stride_j_l, stride_h_l,
    stride_i_r, stride_j_r, stride_h_r,
    stride_i_o, stride_j_o, stride_h_o,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_i = tl.program_id(axis=0)
    pid_j = tl.program_id(axis=1)
    pid_h = tl.program_id(axis=2)

    # offsets
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h_base = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    # create mask for valid positions
    mask_i = i_base < N
    mask_j = j_base < N
    mask_h = h_base < H

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    for k_idx in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_idx * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # compute pointers
        left_offset = i_base[:, None, None] * stride_i_l + k_base[None, :, None] * stride_j_l + h_base[None, None, :] * stride_h_l
        left_tile = tl.load(left_ptr + left_offset,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0)

        right_offset = j_base[:, None, None] * stride_i_r + k_base[None, :, None] * stride_j_r + h_base[None, None, :] * stride_h_r
        right_tile = tl.load(right_ptr + right_offset,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0)

        # compute dot per hidden
        for hh in range(BLOCK_H):
            # extract slice for hidden dimension
            lhs = left_tile[:, :, hh]   # (BLOCK_I, BLOCK_K)
            rhs = right_tile[:, :, hh]  # (BLOCK_J, BLOCK_K)
            # compute dot: (BLOCK_I, BLOCK_J) = lhs (I,K) * rhs^T (K,J)
            dot = tl.dot(lhs, rhs, trans_b=True, out_dtype=tl.float32)
            acc[:, :, hh] += dot

    # store result
    out_offset = i_base[:, None, None] * stride_i_o + j_base[None, :, None] * stride_j_o + h_base[None, None, :] * stride_h_o
    tl.store(out_ptr + out_offset,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The kernel evaluates the contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which corresponds to the einsum
    "... i k d, ... j k d -> ... i j d".
The tensor dimensions are:
    B – batch (≤2)
    N – sequence length (≤1024)
    H – hidden dimension (config["hidden_dim"], a multiple of BLOCK_H)
The implementation:
  * LayerNorm, linear projections, masking and gating are performed in PyTorch.
  * The heavy N³ reduction is carried out by a custom Triton kernel that
    tiles the (i,j) output plane and the hidden dimension.
    For each tile we load a BLOCK_I×BLOCK_K×BLOCK_H slice of `left`
    and a BLOCK_J×BLOCK_K×BLOCK_H slice of `right`, then for every hidden
    sub‑block we compute a small matrix multiplication with `tl.dot`,
    accumulating in FP32.  The kernel works for any N up to 1024 and
    hidden dimensions that are multiples of BLOCK_H (the block size can be
    tuned, the defaults work well on an H100).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel for the N³ reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # ptr to left slice   [N, N, H] (fp16)
    right_ptr,           # ptr to right slice  [N, N, H] (fp16)
    out_ptr,             # ptr to output slice [N, N, H] (fp16)
    N, H,                # scalar sizes
    stride_i_l, stride_j_l, stride_h_l,   # strides for left  (i,k,h)
    stride_i_r, stride_j_r, stride_h_r,   # strides for right (j,k,h)
    stride_i_o, stride_j_o, stride_h_o,   # strides for out   (i,j,h)
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute a tile of the TriMul contraction.
    Each program instance produces a (BLOCK_I, BLOCK_J, BLOCK_H) sub‑tensor.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # ------------------- index ranges -------------------
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    h_base = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = i_base < N
    mask_j = j_base < N
    mask_h = h_base < H

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------- reduction over k -------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_base < N

        # ----- load left[i,k,h] -----
        left_offset = (i_base[:, None, None] * stride_i_l
                     + k_base[None, :, None] * stride_j_l
                     + h_base[None, None, :] * stride_h_l)
        left_tile = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0
        )   # (BLOCK_I, BLOCK_K, BLOCK_H), fp16

        # ----- load right[j,k,h] -----
        right_offset = (j_base[:, None, None] * stride_i_r
                      + k_base[None, :, None] * stride_j_r
                      + h_base[None, None, :] * stride_h_r)
        right_tile = tl.load(
            right_ptr + right_offset,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0
        )   # (BLOCK_J, BLOCK_K, BLOCK_H), fp16

        # ----- inner loop over hidden sub‑block -----
        for hh in range(BLOCK_H):
            # (BLOCK_I, BLOCK_K) x (BLOCK_J, BLOCK_K)ᵀ → (BLOCK_I, BLOCK_J)
            lhs = left_tile[:, :, hh]          # (BLOCK_I, BLOCK_K)
            rhs = right_tile[:, :, hh]         # (BLOCK_J, BLOCK_K)
            # Compute dot product; accumulate in FP32
            dot = tl.dot(lhs, rhs, trans_b=True, out_dtype=tl.float32)
            acc[:, :, hh] += dot

    # ------------------- store result -------------------
    out_offset = (i_base[:, None, None] * stride_i_o
                + j_base[None, :, None] * stride_j_o
                + h_base[None, None, :] * stride_h_o)
    tl.store(
        out_ptr + out_offset,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel for each batch element
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                   BLOCK_I=32, BLOCK_J=32, BLOCK_K=64, BLOCK_H=4) -> torch.Tensor:
    """
    left, right : torch tensors of shape [B, N, N, H] (dtype float16)
    Returns out tensor of same shape and dtype.
    """
    B, N, N2, H = left.shape
    assert N == N2, "input must be square in the sequence dimensions"
    out = torch.empty_like(left)  # same dtype (fp16)

    for b in range(B):
        # Work on a contiguous sub‑tensor for the current batch
        l = left[b].contiguous()
        r = right[b].contiguous()
        o = out[b].contiguous()

        # strides in element counts (not bytes)
        stride_i_l, stride_j_l, stride_h_l = l.stride()
        stride_i_r, stride_j_r, stride_h_r = r.stride()
        stride_i_o, stride_j_o, stride_h_o = o.stride()

        grid = (triton.cdiv(N, BLOCK_I),
                triton.cdiv(N, BLOCK_J),
                triton.cdiv(H, BLOCK_H))

        # Launch with a moderate number of warps (8 works well on H100)
        tri_mul_kernel[grid](
            l, r, o,
            N, H,
            stride_i_l, stride_j_l, stride_h_l,
            stride_i_r, stride_j_r, stride_h_r,
            stride_i_o, stride_j_o, stride_h_o,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
            num_warps=8,
        )
    return out


# ----------------------------------------------------------------------
# Entry point required by the task
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] float32

    # ---------- Linear projections ----------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------- Mask ----------
    if mask is not None:
        # mask shape [B,N,N] -> broadcast over hidden dim
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # ---------- Gating ----------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- Cast to half for the heavy contraction ----------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # ---------- Triton contraction (N³ reduction) ----------
    out_h = tri_mul_triton(left_h, right_h,
                           BLOCK_I=32, BLOCK_J=32, BLOCK_K=64, BLOCK_H=4)

    # Back to float32 for the remaining (lighter) ops
    out = out_h.to(torch.float32)

    # ---------- Output layer norm ----------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ---------- Apply output gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out = F.linear(out, weights["to_out.weight"])

    return out
