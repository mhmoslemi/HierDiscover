def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', False)

    # Weight tensors
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # ----- LayerNorm -----
    x_norm = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Projections (half precision) -----
    # project to hidden_dim
    left_proj = F.linear(x_norm, left_proj_weight).to(torch.float16)
    right_proj = F.linear(x_norm, right_proj_weight).to(torch.float16)

    # ----- Gates (half) -----
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight)).to(torch.float16)

    # ----- Apply mask if present -----
    if not nomask:
        # mask shape [B,N,N] -> broadcast to [B,N,N,1]
        mask_f = mask[..., None].to(left_proj.dtype)  # half
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ----- Fuse projection with gates via Triton kernel -----
    left = fused_mul(left_proj, left_gate)   # returns half tensor
    right = fused_mul(right_proj, right_gate)

    # ----- Batched matmul across hidden dimension -----
    B, N, _, _ = left.shape
    # reshape to [B*hidden_dim, N, N]
    left_mat = left.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0,3,1,2).contiguous().view(B * hidden_dim, N, N)

    # out_mat = left_mat @ right_mat^T (size [B*hidden_dim,N,N])
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))
    # reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()  # half

    # ----- to_out_norm (half) -----
    # Cast layernorm weights to half
    to_out_norm_weight_h = to_out_norm_weight.to(out.dtype)
    to_out_norm_bias_h = to_out_norm_bias.to(out.dtype)
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight_h, bias=to_out_norm_bias_h)

    # ----- Multiply by out_gate via kernel -----
    out = fused_mul(out_norm, out_gate)

    # ----- Final linear projection to dim -----
    # Cast to_out_weight to same dtype
    to_out_weight_h = to_out_weight.to(out.dtype)
    out = F.linear(out, to_out_weight_h).to(torch.float32)  # final output float32

    return out

"""
custom_kernel: Highly‑optimized forward pass of the TriMul “outgoing” update.

Algorithm
---------
1. Layer‑norm the input tensor (float32).
2. Linear projections (float32) → cast to float16.
3. Compute three sigmoid gates in float32 → cast to float16.
4. Optional mask (float16) is applied to left/right projections.
5. Fuse projection×gate using a tiny Triton kernel (element‑wise multiplication).
6. Batched matrix multiplication across the hidden dimension:
   for every (batch, hidden) pair we compute  left @ rightᵀ  (half‑precision).
   This is done with torch.bmm which uses cuBLAS‑Tensor‑Cores on H100.
7. Layer‑norm the result (float32) and cast back to float16.
8. Fuse the output gate with the normalized tensor (Triton kernel).
9. Final linear projection back to the original channel dimension (half‑precision)
   and cast the final result to float32.

Only the element‑wise fusions (steps 5 & 8) are written in Triton; the heavy
matrix‑multiplication is delegated to torch.bmm for maximal performance.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernels for element‑wise multiplication (fused mul)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    out_ptr, a_ptr, b_ptr, n_elements,
    BLOCK_SIZE: tl.constexpr
):
    """out = a * b   (1‑D flattened tensors)"""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def fused_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiply two tensors of identical shape using Triton.
    Both tensors must be on the same CUDA device and have the same dtype.
    Returns a new tensor with the same dtype.
    """
    assert a.shape == b.shape, "Shape mismatch for fused_mul"
    a_c = a.contiguous()
    b_c = b.contiguous()
    out = torch.empty_like(a_c)
    total = a_c.numel()
    BLOCK = 1024  # 1 K elements per program; good trade‑off for H100
    grid = ((total + BLOCK - 1) // BLOCK,)
    elementwise_mul_kernel[grid](
        out, a_c, b_c, total, BLOCK_SIZE=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” update.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Extract weights (all are on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (float32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim] (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (float32) → cast to float16
    # ------------------------------------------------------------------
    left_proj = F.linear(x_norm, left_proj_weight).to(torch.float16)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x_norm, right_proj_weight).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Gates (computed in fp32 for stability, then cast to fp16)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight)).to(torch.float16)
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight)).to(torch.float16)
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight)).to(torch.float16)

    # ------------------------------------------------------------------
    # 4. Optional mask application (broadcasted)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] → [B, N, N, 1] → broadcast over hidden_dim
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # 5. Fuse projection × gate using Triton (half‑precision)
    # ------------------------------------------------------------------
    left = fused_mul(left_proj, left_gate)    # [B,N,N,hidden_dim] (fp16)
    right = fused_mul(right_proj, right_gate)

    # ------------------------------------------------------------------
    # 6. Batched matrix multiplication across the hidden dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    We reshape to (B*hidden_dim, N, N) and use torch.bmm.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # reshape to (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # matmul with transposed right (half‑precision, Tensor‑Core friendly)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*hidden_dim, N, N] fp16

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # ------------------------------------------------------------------
    # 7. Final LayerNorm (performed in float32 for numerical stability)
    # ------------------------------------------------------------------
    out_fp32 = out.float()
    out_norm_fp32 = F.layer_norm(
        out_fp32,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out_norm = out_norm_fp32.to(torch.float16)  # back to half

    # ------------------------------------------------------------------
    # 8. Apply the output gate (Triton fused multiplication)
    # ------------------------------------------------------------------
    out_gated = fused_mul(out_norm, out_gate)  # fp16

    # ------------------------------------------------------------------
    # 9. Final linear projection back to `dim` and cast to float32
    # ------------------------------------------------------------------
    output = F.linear(out_gated, to_out_weight.to(torch.float16))  # fp16
    output = output.float()  # ensure output dtype matches reference

    return output
