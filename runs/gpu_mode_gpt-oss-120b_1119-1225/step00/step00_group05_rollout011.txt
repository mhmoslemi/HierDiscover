@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_aba, stride_abi, stride_ak, # a_batch, a_m, a_k
    stride_bba, stride_bbk, stride_bn, # b_batch, b_k, b_n
    stride_cca, stride_cci, stride_ccn, # c_batch, c_m, c_n
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
   # we will use program_id for batch, block_m, block_n?
   batch_id = tl.program_id(0)
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute offsets
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # compute block mask for M,N bounds
   mask_m = offs_m < M
   mask_n = offs_n < N

   # pointer offsets for each iteration of k
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   for k in range(0, tl.cdiv(K, BLOCK_K)):
       # compute offsets for k tile
       k_base = k * BLOCK_K
       offs_k = k_base + tl.arange(0, BLOCK_K)
       mask_k = offs_k < K

       # load A tile
       a_ptrs = a_ptr + batch_id * stride_aba + offs_m[:, None] * stride_abi + offs_k[None, :] * stride_ak
       a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

       # load B tile but B is (K,N) so we need B(k,n)
       b_ptrs = b_ptr + batch_id * stride_bba + offs_k[:, None] * stride_bbk + offs_n[None, :] * stride_bn
       b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

       # accumulate
       acc += tl.dot(a, b) # this computes (M,K) dot (K,N) = (M,N)
       # Move pointers for next block automatically via offsets

   # write out
   c_ptrs = c_ptr + batch_id * stride_cca + offs_m[:, None] * stride_cci + offs_n[None, :] * stride_ccn
   tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask handling
    if mask is not None and not config.get("nomask", False):
        mask_ = mask.to(left.dtype).unsqueeze(-1) # shape (B,N,N,1)
        left = left * mask_
        right = right * mask_
    # else no mask multiplication needed

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B,H,N,N) where dims (B,H,K,J) but same shape

    right_flat = right_perm.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)

    BLOCK_N = 128
    BLOCK_K = 32

        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    # But we need ordering: Triton supports 3D grid: (grid0, grid1, grid2). Let's define grid=(BH, num_i_tiles, num_j_tiles)
    # call kernel with args

        BH, N, N, N,
        a_batch_stride, a_m_stride, a_k_stride,
        b_batch_stride, b_k_stride, b_n_stride,
        c_batch_stride, c_m_stride, c_n_stride,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous() # shape (B,N,N,hidden_dim)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    # Apply out_gate (already has shape (B,N,N,hidden_dim))
    out = out * out_gate

    # Final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_aba, stride_abi, stride_ak,
    stride_bba, stride_bbk, stride_bn,
    stride_cca, stride_cci, stride_ccn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):

    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A
        a_ptrs = a_ptr + batch * stride_aba + offs_m[:, None] * stride_abi + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B
        b_ptrs = b_ptr + batch * stride_bba + offs_k[:, None] * stride_bbk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write back C
    c_ptrs = c_ptr + batch * stride_cca + offs_m[:, None] * stride_cci + offs_n[None, :] * stride_ccn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

TriMul forward implementation (outgoing version).

The heavy pairwise multiplicative update
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched matrix multiplication using a custom Triton
kernel (left @ rightᵀ) for every hidden dimension `d`.  All surrounding
operations (LayerNorm, linear projections, gating, final projection) are
implemented with PyTorch functional calls.  The kernel works for arbitrary
batch size, sequence length and hidden dimension and requires only float32
inputs (the kernel can be easily switched to fp16 if desired).

The expected input tuple is:
    (input_tensor, mask_tensor, weights_dict, config_dict)

where `weights_dict` contains the exact tensors used in the reference
implementation (LayerNorm weights/biases, Linear weights, etc.) and
`config_dict` supplies `dim` and `hidden_dim`.

import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched GEMM computing out = left @ rightᵀ
# left  : (BATCH, M, K)   (here M = N, K = N)
# right : (BATCH, K, N)   (right is already transposed)
# out   : (BATCH, M, N)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    BATCH, M, N, K,                    # dimensions
    stride_aba, stride_abi, stride_ak, # a: (batch, M, K) strides
    stride_bba, stride_bbk, stride_bn, # b: (batch, K, N) strides
    stride_cca, stride_cci, stride_ccn, # c: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_n = offs_n < N

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

            a_ptr

            + offs_m[:, None] * stride_abi
            + offs_k[None, :] * stride_ak

            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + batch_id * stride_bba
            + offs_k[:, None] * stride_bbk
            + offs_n[None, :] * stride_bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # accumulate product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write back
    c_ptrs = (
        c_ptr
        + batch_id * stride_cca
        + offs_m[:, None] * stride_cci
        + offs_n[None, :] * stride_ccn
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point used by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1. LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]

    # --------------------------------------------------------------------- #
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])    # [B, N, N, hidden_dim]

    # --------------------------------------------------------------------- #
    # 3. Optional mask (broadcasted on hidden dimension)
    # --------------------------------------------------------------------- #
    if (mask_tensor is not None) and (not nomask):
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------------- #
    # 4. Gating (sigmoid activations of linear layers on the *original* x)
    # --------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 5. TriMul core: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM: for each (b, d) we compute
    #    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape
    # reshape to (B, hidden_dim, N, N) and make them contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N)  i‑k
    right_perm = right.permute(0, 3, 2, 1).contiguous()       # (B, H, N, N)  k‑j (already transposed)

    # combine batch and hidden dimension → [B*H, N, N]
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    out_flat = torch.empty_like(left_flat)

    # strides for the three tensors (batch, M, K) and (batch, K, N)
    a_batch_stride, a_m_stride, a_k_stride = left_flat.stride()
    b_batch_stride, b_k_stride, b_n_stride = right_flat.stride()
    c_batch_stride, c_m_stride, c_n_stride = out_flat.stride()

    # block sizes (tuned for H100, fp32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # launch grid  (batch*head, num_i_tiles, num_j_tiles)
    grid = (
        BH,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BH,
        N,
        N,
        N,
        a_batch_stride,
        a_m_stride,
        a_k_stride,
        b_batch_stride,
        b_k_stride,
        b_n_stride,
        c_batch_stride,
        c_m_stride,
        c_n_stride,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 6. Output LayerNorm over hidden dimension
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------------- #
    # 7. Apply final output gate and linear projection back to dim
    # --------------------------------------------------------------------- #
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
