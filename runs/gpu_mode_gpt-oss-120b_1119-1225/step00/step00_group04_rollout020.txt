```
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bk, stride_Bj,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute block start coordinates
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = offs_i < M
    mask_j = offs_j < N

    # pointer to first block of A and B for this batch
    a_batch_ptr = a_ptr + batch_idx * stride_Ab
    b_batch_ptr = b_ptr + batch_idx * stride_Bb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        # compute addresses
        a_ptrs = a_batch_ptr + (offs_i[:, None] * stride_Ai) + (offs_k[None, :] * stride_Ak)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_Bk) + (offs_j[None, :] * stride_Bj)
        # load a and b with masks
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        acc += tl.dot(a, b)

    # write back
    c_batch_ptr = c_ptr + batch_idx * stride_Cb
    c_ptrs = c_batch_ptr + (offs_i[:, None] * stride_Ci) + (offs_j[None, :] * stride_Cj)
    # store only valid entries
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # as in nn.LayerNorm

    # LayerNorm on input (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not config.get("nomask", True):
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute flops using batched GEMM
    # Prepare A and B for batched GEMM

"""
TriMul (outgoing) forward implementation with a Triton‐accelerated
batched matrix multiplication.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the last dimension `dim`.
2. Linear projections produce `left` and `right` tensors of shape
   [B, N, N, H] (H = hidden_dim).
3. (Optional) multiply by a pairwise mask.
4. Compute three sigmoid gates from the normalized input and apply them to
   `left`, `right` and the eventual output.
5. The core operation is an Einstein sum
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batched GEMM for each hidden channel:
       out = (left_permuted) @ (right_permuted.T)
   This is performed by the Triton kernel `batched_gemm_kernel`,
   processing all (B·H) batches in parallel.
6. Apply a second LayerNorm (`to_out_norm`), multiply by `out_gate`,
   and finish with a linear layer `to_out` that maps back to `dim`.

The heavy N³ work (step 5) is therefore executed on the GPU with a
custom Triton kernel, while the remaining element‑wise operations are
handled in PyTorch.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_Ab, stride_Ai, stride_Ak,
    stride_Bb, stride_Bk, stride_Bj,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM for shape (batch, M, K) @ (batch, K, N) -> (batch, M, N)."""
    batch_idx = tl.program_id(0)          # batch dimension (B * hidden)
    pid_m = tl.program_id(1)              # tile row
    pid_n = tl.program_id(2)              # tile column

    # Global offsets of the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current tile of A and B
        a_ptrs = a_ptr + batch_idx * stride_Ab \
                 + offs_i[:, None] * stride_Ai \
                 + offs_k[None, :] * stride_Ak
        b_ptrs = b_ptr + batch_idx * stride_Bb \
                 + offs_k[:, None] * stride_Bk \
                 + offs_j[None, :] * stride_Bj

        # Load with masks (zero‑pad out‑of‑range elements)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Rank‑K update
        acc += tl.dot(a, b)

    # Write the result back to C
    c_ptrs = c_ptr + batch_idx * stride_Cb \
             + offs_i[:, None] * stride_Ci \
             + offs_j[None, :] * stride_Cj
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    TriMul outgoing forward pass.
    Input `data` is a tuple:
        (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim].
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    eps = 1e-5  # LayerNorm epsilon, same default as nn.LayerNorm

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4) Compute gates and apply them
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul: batched GEMM via Triton
    #    left  : (B, N, N, H)  -> (B, H, N, N) where axes are (i, k)
    #    right : (B, N, N, H)  -> (B, H, N, N) then transpose to (B, H, k, j)
    #    out   : (B, H, N, N)  -> (B, N, N, H)
    # ------------------------------------------------------------------
    # permute to bring hidden dim next to batch for a clean batched view
    left_perm = left.permute(0, 3, 1, 2).contiguous()               # (B, H, i, k)
    right_perm = right.permute(0, 3, 1, 2).contiguous()             # (B, H, j, k)
    # transpose the last two dims of right so that it is (k, j)
    right_t = right_perm.transpose(2, 3).contiguous()               # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    total_batch = B * H

    # flatten batch & hidden into a single GEMM batch dimension
    A = left_perm.reshape(total_batch, N, N)   # (B*H, i, k)
    B_mat = right_t.reshape(total_batch, N, N)  # (B*H, k, j)
    C = torch.empty_like(A)                    # (B*H, i, j)

    # Choose tile sizes (tuned for H100 FP32)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        A,
        B_mat,
        C,
        N,               # M
        N,               # N
        N,               # K
        A.stride(0),
        A.stride(1),
        A.stride(2),
        B_mat.stride(0),
        B_mat.stride(1),
        B_mat.stride(2),
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = C.reshape(B, H, N, N).permute(0, 2, 3, 1)   # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    output = F.linear(out, weights["to_out.weight"])  # (B, N, N, dim)

    return output
