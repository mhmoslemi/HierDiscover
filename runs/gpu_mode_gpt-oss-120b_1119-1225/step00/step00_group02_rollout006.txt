@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lj, stride_lh,
    stride_rb, stride_ri, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0) # combined batch*hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid // H
    h = pid % H

    # compute offsets...
    off_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds.
    mask_i = off_i < N
    mask_j = off_j < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + (b * stride_lb + off_i[:,None] * stride_li + k_offset[None,:] * stride_lj + h * stride_lh)
        # Wait stride indices: left dimension order: (b,i,j,h). So strides: stride_lb over b, stride_li over i, stride_lj over j, stride_lh over h.
        # Offsets: i*stride_li + k*stride_lj + h*stride_lh

        left = tl.load(left_ptrs, mask=mask_i[:,None] & mask_k[None,:], other=0.0)

        # Load right: we need right[b, j, k, h] (note j index goes to column of output; we have right pointer using i/j swapped) Right shape (b, N, N, h). For out[i,j] we need right[b, j, k, h] i.e., second dimension index = j, third = k.
        # So right offset: i index = j (off_j), j index = k_offset.
        right_ptrs = right_ptr + (b * stride_rb + off_j[:,None] * stride_ri + k_offset[None,:] * stride_rj + h * stride_rh)
        # Wait stride_ri is stride over second dimension (maybe i?). Actually original shape right: (b, N, N, h) -> dimensions: b (0), i (1), j (2), h (3). We need right[b, j, k, h]; i dimension replaced by j, j dimension replaced by k. So stride_i gets used for off_j, stride_j for k_offset.

        right = tl.load(right_ptrs, mask=mask_j[:,None] & mask_k[None,:], other=0.0)

        # Multiply-accumulate
        acc += tl.dot(left, right)

    # Store results
    out_ptrs = out_ptr + (b * stride_ob + off_i[:,None] * stride_oi + off_j[None,:] * stride_oj + h * stride_oh)
    tl.store(out_ptrs, acc, mask=mask_i[:,None] & mask_j[None,:])

def custom_kernel(data):
    # unpack
    x, mask, weights, config = data
    # shape
    B, N, _, dim = x.shape
    hidden = config["hidden_dim"]
    # Ensure contiguous
    x = x.contiguous()
    # LayerNorm (norm weight,bias)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Compute mean, var along last dim
    # Use torch.mean and torch.var for efficiency
    eps = 1e-5
    # Compute mean and var
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections:
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden, dim)
    right_proj_weight = weights["right_proj.weight"]
    # Compute left = x_norm @ left_proj_weight.T
    left = torch.matmul(x_norm, left_proj_weight.T)  # shape (B,N,N,hidden)
    right = torch.matmul(x_norm, right_proj_weight.T)

    # Apply mask if needed
    if not config.get("nomask", True):
        # mask shape (B,N,N) -> unsqueeze(-1)
        mask = mask.unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask
        right = right * mask

    # Gates:
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.matmul(x_norm, left_gate_weight.T))
    right_gate = torch.sigmoid(torch.matmul(x_norm, right_gate_weight.T))
    out_gate = torch.sigmoid(torch.matmul(x_norm, out_gate_weight.T))

    left = left * left_gate
    right = right * right_gate

    # Now compute out = left @ right_T using Triton
    # Ensure left,right contiguous
    left = left.contiguous()
    right = right.contiguous()
    # Permute to (B,N,N,hidden) we already have that.
    # We'll compute out using Triton custom kernel.
    out = torch.empty_like(left)  # same shape
    # Get strides
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()
    # Pass into kernel:
    grid = (B * hidden, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))
    tri_mul_matmul_kernel[grid](
        left_ptr = left,
        right_ptr = right,
        out_ptr = out,
        B = B,
        N = N,
        H = hidden,
        stride_lb = stride_left[0],
        stride_li = stride_left[1],
        stride_lj = stride_left[2],
        stride_lh = stride_left[3],
        stride_rb = stride_right[0],
        stride_ri = stride_right[1],
        stride_rj = stride_right[2],
        stride_rh = stride_right[3],
        stride_ob = stride_out[0],
        stride_oi = stride_out[1],
        stride_oj = stride_out[2],
        stride_oh = stride_out[3],
        BLOCK_M = BLOCK_M,
        BLOCK_N = BLOCK_N,
        BLOCK_K = BLOCK_K,
    )
    # out shape (B,N,N,hidden)

    # to_out_norm (LayerNorm over hidden dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Compute mean,var over hidden dim (last)
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_normed = (out - out_mean) / torch.sqrt(out_var + eps)
    out_normed = out_normed * to_out_norm_weight + to_out_norm_bias

    # Multiply by out_gate
    out_normed = out_normed * out_gate

    # Final linear to_out weight shape (dim, hidden)
    to_out_weight = weights["to_out.weight"]
    # compute for each output: (B,N,N,dim) = out_normed @ to_out_weight.T
    out_final = torch.matmul(out_normed, to_out_weight.T)
    # Return out_final
    return out_final

def custom_kernel(data):
    """Custom forward pass of TriMul using Triton for the batched matrix multiplication.

    The kernel computes:
        out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}

    where left,right are linear projections of the normalized input and
    are elementwise masked and gated. The heavy matrix multiplication
    is performed by a Triton kernel that processes each (batch, hidden)
    slice as a separate GEMM.
    """
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # LayerNorm on dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # Projections
    left = torch.matmul(x, weights["left_proj.weight"].t())
    right = torch.matmul(x, weights["right_proj.weight"].t())

    # Optional mask
    if not config.get("nomask", True):
        mask = mask_tensor.unsqueeze(-1)  # (B,N,N,1)
        left = left * mask
        right = right * mask

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.matmul(x, weights["left_gate.weight"].t()))
    right_gate = torch.sigmoid(torch.matmul(x, weights["right_gate.weight"].t()))
    out_gate = torch.sigmoid(torch.matmul(x, weights["out_gate.weight"].t()))

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous for kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # (B,N,N,hidden)

    # Triton kernel launch
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    grid = (
        B * hidden,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    triton_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # LayerNorm on hidden dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection to dim
    out = torch.matmul(out, weights["to_out.weight"].t())

    return out

@triton.jit
def triton_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lj, stride_lh,
    stride_rb, stride_ri, stride_rj, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # combined batch * hidden
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Derive batch and hidden indices
    b = pid // H
    h = pid - b * H

    # Offsets for the tile in i and j dimensions
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A (left) tile: shape (M, K)
        a_ptrs = left_ptr + \
                 b * stride_lb + \
                 offs_i[:, None] * stride_li + \
                 offs_k[None, :] * stride_lj + \
                 h * stride_lh
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)

        # Load B (right) tile transposed: shape (K, N)
        b_ptrs = right_ptr + \
                 b * stride_rb + \
                 offs_k[:, None] * stride_ri + \
                 offs_j[None, :] * stride_rj + \
                 h * stride_rh
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        b = b.to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write back results
    out_ptrs = out_ptr + \
               b * stride_ob + \
               offs_i[:, None] * stride_oi + \
               offs_j[None, :] * stride_oj + \
               h * stride_oh
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul forward (outgoing version) with a Triton kernel.

The kernel performs the heavy O(N³) reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
where `left` and `right` are the linearly projected, masked and gated
representations of the normalized input.  The reduction is expressed as a
batched matrix multiplication (one GEMM per (batch, hidden) slice) and is
implemented in Triton for maximal fusion.  All other operations (layer‑norm,
linear projections, sigmoid gates, final linear) are carried out with
standard PyTorch primitives.

The implementation works for any batch size ∈ {1,2}, sequence length N
up to 1024, input dimension `dim` up to 384 and hidden dimension
`hidden_dim` (called c_z in the paper) up to 384.  It also correctly
handles the optional boolean mask.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides for left  (b, i, j, h)
    stride_lb, stride_li, stride_lj, stride_lh,
    # strides for right (b, i, j, h)
    stride_rb, stride_ri, stride_rj, stride_rh,
    # strides for out   (b, i, j, h)
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul eosummation.

    Each program instance (pid) is responsible for a single (batch, hidden)
    matrix pair A = left[b, :, :, h] (N×N) and B = right[b, :, :, h] (N×N).
    The product needed is  A @ Bᵀ, i.e. out[b, i, j, h] = Σ_k A[i, k] * B[j, k].
    """
    pid = tl.program_id(0)      # encodes (batch, hidden)
    pid_m = tl.program_id(1)    # tile row
    pid_n = tl.program_id(2)    # tile col

    # ----------------------------------------------
    # decode batch and hidden index from pid
    # ----------------------------------------------
    b = pid // H
    h = pid - b * H               # equivalent to pid % H

    # ----------------------------------------------
    # tile offsets in the i and j dimensions
    # ----------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----------------------------------------------
    # loop over k tiles
    # ----------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----------- load A = left[b, i, k, h]  (M×K) ----------
        a_ptrs = left_ptr + \
                 b * stride_lb + \
                 offs_i[:, None] * stride_li + \
                 offs_k[None, :] * stride_lj + \
                 h * stride_lh
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        a = a.to(tl.float32)

        # ----------- load Bᵀ = right[b, j, k, h] (K×N) ----------
        # note the order of indices: first dimension = k, second = j
        b_ptrs = right_ptr + \
                 b * stride_rb + \
                 offs_k[:, None] * stride_ri + \
                 offs_j[None, :] * stride_rj + \
                 h * stride_rh
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)
        b = b.to(tl.float32)

        # ----------- update accumulator ----------
        acc += tl.dot(a, b)   # (M×K)·(K×N) → (M×N)

    # ----------------------------------------------
    # write back the result
    # ----------------------------------------------
    out_ptrs = out_ptr + \
               b * stride_ob + \
               offs_i[:, None] * stride_oi + \
               offs_j[None, :] * stride_oj + \
               h * stride_oh
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    B, N, _, dim = input_tensor.shape
    hidden = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) LayerNorm over the input feature dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) to hidden dimension
    #    left = x @ W_leftᵀ, right = x @ W_rightᵀ
    # ------------------------------------------------------------------
    left = torch.matmul(x, weights["left_proj.weight"].t())
    right = torch.matmul(x, weights["right_proj.weight"].t())

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden axis)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.matmul(x, weights["left_gate.weight"].t()))
    right_gate = torch.sigmoid(torch.matmul(x, weights["right_gate.weight"].t()))
    out_gate = torch.sigmoid(torch.matmul(x, weights["out_gate.weight"].t()))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched matrix multiplication via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape (B, N, N, hidden)

    # tile sizes – tuned for H100 (you may adjust if needed)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # strides for contiguous tensors (b, i, j, h)
    stride_left = left.stride()   # (s_b, s_i, s_j, s_h)
    stride_right = right.stride()
    stride_out = out.stride()

    grid = (
        B * hidden,                                 # axis 0 : (batch, hidden)
        (N + BLOCK_M - 1) // BLOCK_M,               # axis 1 : tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,               # axis 2 : tiles over j
    )

    triton_trimul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Multiply by the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.matmul(out, weights["to_out.weight"].t())

    return out
