@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
   # program IDs
   pid_b = tl.program_id(2)
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)

   # offsets
   i_offset = pid_i * BLOCK_M
   j_offset = pid_j * BLOCK_N

   # compute masks
   offs_i = i_offset + tl.arange(0, BLOCK_M)
   offs_j = j_offset + tl.arange(0, BLOCK_N)

   mask_i = offs_i < N
   mask_j = offs_j < N

   # Loop over hidden dimension blocks
   for d_offset in range(0, H, BLOCK_D):
       # allocate accumulator
       acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

       # Load offsets for hidden dim
       offs_d = d_offset + tl.arange(0, BLOCK_D)

       # Loop over K tiles
       for k_offset in range(0, N, BLOCK_K):
           # k indices
           offs_k = k_offset + tl.arange(0, BLOCK_K)
           mask_k = offs_k < N

           # left pointers: shape (BM, BK, BD)
           left_ptrs = left_ptr + \
               pid_b * stride_left_b + \
               offs_i[:, None, None] * stride_left_i + \
               offs_k[None, :, None] * stride_left_k + \
               offs_d[None, None, :] * stride_left_h

           # Load left
           left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

           # right pointers: shape (BN, BK, BD)
           right_ptrs = right_ptr + \
               pid_b * stride_right_b + \
               offs_j[:, None, None] * stride_right_i + \
               offs_k[None, :, None] * stride_right_k + \
               offs_d[None, None, :] * stride_right_h

           right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0)

           # Compute broadcast multiplication and reduce over k
           # We can do for loop over kk (K within tile):
           for kk in range(0, BLOCK_K):
               k_valid = mask_k[kk]
               if not k_valid:
                   continue
               left_vec = left[:, kk, :]   # (BM, BD)
               right_vec = right[:, kk, :] # (BN, BD)
               # broadcast and accumulate
               acc += left_vec[:, None, :] * right_vec[None, :, :]  # (BM, BN, BD)

       # Store to out
       out_ptrs = out_ptr + \
           pid_b * stride_out_b + \
           offs_i[:, None, None] * stride_out_i + \
           offs_j[None, :, None] * stride_out_j + \
           offs_d[None, None, :] * stride_out_h

       tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & (offs_d < H))

"""
TriMul (outgoing) implementation for AlphaFold3 / protein models.
---------------------------------------------------------------
We compute the heavy “einsum”:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The tensor shapes are:
    left, right : [B, N, N, H]  (H = hidden_dim)
    out         : [B, N, N, H]

We apply the LayerNorms, linear projections and gating in PyTorch and
off‑load the O(N³·H) reduction to a Triton kernel.  The kernel tiles the
(i, j) output plane (BLOCK_M × BLOCK_N) and the hidden dimension (BLOCK_D).
Inside each tile we accumulate over k using a blocked loop (BLOCK_K).  All
loads are masked to handle arbitrary sequence lengths and hidden‑dim sizes.
The kernel works for any B∈{1,2}, N up to 1024 and hidden_dim up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel that computes out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,        # pointers
    B, N, H,                             # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(2)     # batch dimension
    pid_i = tl.program_id(0)     # i‑tile
    pid_j = tl.program_id(1)     # j‑tile

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    i_idx = i_base + tl.arange(0, BLOCK_M)    # (BM,)
    j_idx = j_base + tl.arange(0, BLOCK_N)    # (BN,)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension blocks
    # ------------------------------------------------------------------
    d_offset = 0
    while d_offset < H:
        d_idx = d_offset + tl.arange(0, BLOCK_D)      # (BD,)
        mask_d = d_idx < H

        # accumulator for this (i,j) tile and hidden block
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # --------------------------------------------------------------
        # Reduction over k (blocked)
        # --------------------------------------------------------------
        k_offset = 0
        while k_offset < N:
            k_idx = k_offset + tl.arange(0, BLOCK_K)  # (BK,)
            mask_k = k_idx < N

            # ------------------------------------------------------------------
            # Load left[i,k,d]  -> shape (BM, BK, BD)
            # ------------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + i_idx[:, None, None] * stride_left_i
                + k_idx[None, :, None] * stride_left_k
                + d_idx[None, None, :] * stride_left_h
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )  # (BM,BK,BD)

            # ------------------------------------------------------------------
            # Load right[j,k,d] -> shape (BN, BK, BD)
            # ------------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + j_idx[:, None, None] * stride_right_i   # j lives on the same stride as i
                + k_idx[None, :, None] * stride_right_k
                + d_idx[None, None, :] * stride_right_h
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )  # (BN,BK,BD)

            # ------------------------------------------------------------------
            # Accumulate: Σ_k left[i,k,d] * right[j,k,d]
            # Broadcast over i and j for each k.
            # ------------------------------------------------------------------
            for kk in range(0, BLOCK_K):
                # zero‑masked k‑rows are already zero, no extra check needed
                l = left[:, kk, :]          # (BM,BD)
                r = right[:, kk, :]         # (BN,BD)
                # broadcast to (BM,BN,BD) and accumulate
                acc += l[:, None, :] * r[None, :, :]   # (BM,BN,BD)

            k_offset += BLOCK_K

        # ------------------------------------------------------------------
        # Store the completed block into the output tensor
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + i_idx[:, None, None] * stride_out_i
            + j_idx[None, :, None] * stride_out_j
            + d_idx[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
        )
        d_offset += BLOCK_D


def _launch_trimul(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor):
    """
    Helper to launch the Triton kernel with sensible tile sizes.
    """
    # Expected shapes: [B, N, N, H]
    B, N, _, H = left.shape

    # Strides in element units (not bytes)
    sL = left.stride()
    sR = right.stride()
    sO = out.stride()

    # Tile sizes – chosen to balance register pressure and occupancy on H100
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32
    BLOCK_D = 16

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B,                              # batch
    )

    _trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        sL[0],
        sL[1],
        sL[2],
        sL[3],
        sR[0],
        sR[1],
        sR[2],
        sR[3],
        sO[0],
        sO[1],
        sO[2],
        sO[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )


# ----------------------------------------------------------------------
# Public entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.

    Args:
        data: Tuple (input, mask, weights, config)
            - input:   [B, N, N, dim] tensor
            - mask:    [B, N, N] mask (may be all‑ones)
            - weights: dict with model parameters
            - config : dict with 'dim' and 'hidden_dim' (and optionally 'nomask')

    Returns:
        Tensor of shape [B, N, N, dim]
    """
    # unpack
    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = inp.device
    dtype = inp.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (over last dimension = dim)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(inp, normalized_shape=(dim,), weight=norm_w, bias=norm_b)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1]
        mask_b = mask.unsqueeze(-1).type_as(x)
        left = left * mask_b
        right = right * mask_b

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy multiplication via Triton -> out_tmp [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    out_tmp = torch.empty_like(left)
    _launch_trimul(left, right, out_tmp)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out_tmp,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    # ------------------------------------------------------------------
    # 7. Apply output gate and final projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)

    return out
