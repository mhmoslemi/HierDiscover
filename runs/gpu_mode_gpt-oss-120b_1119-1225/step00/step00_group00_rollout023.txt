"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]).
2. Linear projections (no bias):
       left  = x @ left_proj_weight.T      # [B,N,N,H]
       right = x @ right_proj_weight.T
3. Apply optional mask (broadcast on the hidden dimension).
4. Compute three sigmoid gates from the normalized input:
       left_gate  = sigmoid(x @ left_gate_weight.T)
       right_gate = sigmoid(x @ right_gate_weight.T)
       out_gate   = sigmoid(x @ out_gate_weight.T)
   Multiply the gates into the projected tensors.
5. Compute the core O(N³·H) operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiplication per hidden channel.
   It is implemented with a Triton kernel that tiles the i & j dimensions
   (BLOCK_I×BLOCK_J) and reduces over k in blocks (BLOCK_K).  Each Triton
   program processes a single (batch, hidden‑channel) pair, making the
   memory accesses regular and allowing a Tensor‑Core dot product.
6. Layer‑norm the intermediate result over the hidden dimension,
   multiply by `out_gate`, and finally map back to the original channel
   dimension with a linear layer.
The kernel runs on H100 GPUs (Triton 3.3.1) and uses mixed‑precision
(half‑precision loads are cast to FP32 for accumulation).  The rest of the
pipeline is performed in PyTorch for convenience.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the O(N³·H) reduction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_out_kernel(
    left_ptr, right_ptr, out_ptr,                # pointers
    B, N, H,                                     # dimensions
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Grid: (B*H, ceil(N/BLOCK_I), ceil(N/BLOCK_J))
    """
    pid0 = tl.program_id(0)      # combined (batch, hidden)
    pid_i = tl.program_id(1)      # tile over i
    pid_j = tl.program_id(2)      # tile over j

    # ----- decode batch and hidden index --------------------------------
    total_h = H
    b = pid0 // total_h          # batch index
    h = pid0 % total_h           # hidden index (0 <= h < H)

    # ----- start offsets for the i/j tiles --------------------------------
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # coordinate vectors
    i = i_start + tl.arange(0, BLOCK_I)
    j = j_start + tl.arange(0, BLOCK_J)

    # bounds checking
    mask_i = i < N
    mask_j = j < N

    # convert coordinates to byte offsets
    i_off = i * stride_l_i            # left: i dimension
    j_off = j * stride_r_i            # right: first seq dimension (treated as j)

    # base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b  + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b  + h * stride_o_h

    # accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----- reduction over k ------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        k_off = k * stride_l_k    # stride for the k dimension (same for left/right)

        # left tile: shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_base + i_off[:, None] + k_off[None, :]
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float32)

        # right tile transposed: shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_base + k_off[:, None] + j_off[None, :]
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float32)

        # accumulate dot product (BLOCK_I x BLOCK_J)
        acc += tl.dot(left, right)

    # ----- write out the result -------------------------------------------
    out_ptrs = out_base + i_off[:, None] + j_off[None, :]
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, C] (C == config["dim"])
            - mask  : torch.Tensor [B, N, N]   (may be None)
            - weights: dict of torch.Tensors containing all linear / norm weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor with shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # extract weight tensors (all are stored as 2‑D or 1‑D tensors)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)      # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)     # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(x.dtype)           # ensure same dtype
        mask_f = mask_f.unsqueeze(-1)       # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid) and element‑wise modulation
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³·H) reduction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    H = hidden_dim

    # Ensure contiguous tensors for predictable strides
    left  = left.contiguous()
    right = right.contiguous()

    # Output tensor that will hold the intermediate result (float32)
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # Block sizes – tuned for H100 (FP32).  They comfortably cover all test sizes.
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64

    # Compute grid dimensions: (B*H, ceil(N/BLOCK_I), ceil(N/BLOCK_J))
    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid_0 = B * H
    grid = (grid_0, grid_i, grid_j)

    # Launch kernel
    tri_mul_out_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        H=H,
        stride_l_b=left.stride(0), stride_l_i=left.stride(1),
        stride_l_k=left.stride(2), stride_l_h=left.stride(3),
        stride_r_b=right.stride(0), stride_r_i=right.stride(1),
        stride_r_k=right.stride(2), stride_r_h=right.stride(3),
        stride_o_b=out.stride(0), stride_o_i=out.stride(1),
        stride_o_j=out.stride(2), stride_o_h=out.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: Layer‑norm → out_gate → final linear
    # ------------------------------------------------------------------
    # Normalise over the hidden dimension
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Apply the output gate (broadcasted over i,j)
    out = out * out_gate

    # Final linear projection back to original channel size
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
