@triton.jit
def tri_mul_kernel(
    left_ptr, # pointer to left
    right_ptr, # pointer to right transposed (so we treat right as B_T)
    out_ptr,
    G, M, N, K,  # M=N, K=N
    stride_left_g, stride_left_m, stride_left_k,
    stride_right_g, stride_right_k, stride_right_n,  # note swapped
    stride_out_g, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_g = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers with offsets
    a_ptrs = left_ptr + pid_g * stride_left_g + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
    b_ptrs = right_ptr + pid_g * stride_right_g + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n

    # Load tiles
    a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
    b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # actual offsets for this block:
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        # ignore if beyond K, but we handle mask

        a_ptrs = left_ptr + pid_g * stride_left_g + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        b_ptrs = right_ptr + pid_g * stride_right_g + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n

        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # dot does (M,K) dot (K,N) => (M,N)

    # Write back result
    out_ptrs = out_ptr + pid_g * stride_out_g + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    # strides
    stride_left_g, stride_left_m, stride_left_k,
    stride_right_g, stride_right_k, stride_right_n,
    stride_out_g, stride_out_m, stride_out_n,
    G, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_g = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute start indices for this block
    m_start = pid_m * BLOCK_M
    n_start = pid_n * BLOCK_N

    # Create range vectors with possible out-of-bounds
    offs_m = m_start + tl.arange(0, BLOCK_M)
    offs_n = n_start + tl.arange(0, BLOCK_N)

    # Mask for boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # mask for K
        mask_k = offs_k < K

        # Load A tile
        a_ptrs = left_ptr + pid_g * stride_left_g + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (right transposed)
        b_ptrs = right_ptr + pid_g * stride_right_g + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Write result
    out_ptrs = out_ptr + pid_g * stride_out_g + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul outgoing kernel (AlphaFold3)

This implementation follows the reference PyTorch code but fuses the
O(N³) reduction into a custom Triton kernel.  The kernel computes for each
batch `b` and hidden channel `h`

    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

which is exactly a batched matrix multiplication of `left` with the
(transposed) `right`.  All other operations (LayerNorm, linear projections,
gates, masking) are performed in PyTorch using half‑precision arithmetic.
The final output is returned in float32.

The Triton kernel works on tensors reshaped to
[G, N, N] where G = batch * hidden_dim, i.e. each GEMM is independent.
Block sizes (64,64,64) give good occupancy on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    # Pointers
    left_ptr,            # [G, N, N]   (left projection)
    right_ptr,           # [G, N, N]   (right projection transposed)
    out_ptr,             # [G, N, N]   (output)
    # Strides (in elements)
    stride_left_g, stride_left_m, stride_left_k,
    stride_right_g, stride_right_k, stride_right_n,
    stride_out_g, stride_out_m, stride_out_n,
    # Sizes
    G, N, K,             # G = B*hidden_dim, N = seq_len, K = N (square)
    # Compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_g = tl.program_id(0)                # which (batch, hidden) pair
    pid_m = tl.program_id(1)                # row block index
    pid_n = tl.program_id(2)                # column block index

    # Block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for tk in range(0, num_k_tiles):
        k_start = tk * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ----- Load a tile from left -----
        a_ptrs = (
            left_ptr
            + pid_g * stride_left_g
            + offs_m[:, None] * stride_left_m
            + offs_k[None, :] * stride_left_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Load a tile from right (already transposed) -----
        # right_ptr has strides (g, k, n) because we passed the transposed view
        b_ptrs = (
            right_ptr
            + pid_g * stride_right_g
            + offs_k[:, None] * stride_right_k
            + offs_n[None, :] * stride_right_n
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- Compute partial product -----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ----- Write back result (FP16) -----
    out_ptrs = (
        out_ptr
        + pid_g * stride_out_g
        + offs_m[:, None] * stride_out_m
        + offs_n[None, :] * stride_out_n
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = torch.float16          # use half‑precision for the heavy ops
    eps = 1e-5

    # ------------------------------------------------------------------
    # 0. Extract dimensions
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # 1. Cast inputs and weights to fp16 where possible
    # ------------------------------------------------------------------
    x = input_tensor.to(dtype)

    # Linear and LayerNorm weights
    norm_weight = weights["norm.weight"].to(dtype)
    norm_bias   = weights["norm.bias"].to(dtype)

    left_proj_weight  = weights["left_proj.weight"].to(dtype)   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"].to(dtype)

    left_gate_weight  = weights["left_gate.weight"].to(dtype)
    right_gate_weight = weights["right_gate.weight"].to(dtype)
    out_gate_weight   = weights["out_gate.weight"].to(dtype)

    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(dtype)
    to_out_weight      = weights["to_out.weight"].to(dtype)     # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 2. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # ------------------------------------------------------------------
    # 5. Optional mask (if a mask tensor is supplied)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # Convert mask to FP16 and broadcast over the channel dimension
        mask_f = mask.to(dtype).unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 6. Apply gating
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 7. Compute the O(N³) reduction using a Triton kernel
    #    left  : [B, N, N, hidden_dim]  →  (B, hidden_dim, N, N)
    #    right : same, but we need its transpose on the last two dims
    # ------------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) and make them contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose the inner pair of dimensions for the right operand
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()  # shape still [B, H, N, N]

    # Collapse batch and hidden dimension → G = B * H
    G = B * hidden_dim
    left_flat = left_perm.view(G, N, N)
    right_flat = right_t.view(G, N, N)
    out_flat = torch.empty_like(left_flat, dtype=dtype, device=device)

    # Strides (in elements)
    stride_left_g, stride_left_m, stride_left_k = left_flat.stride()
    stride_right_g, stride_right_k, stride_right_n = right_flat.stride()
    stride_out_g, stride_out_m, stride_out_n = out_flat.stride()

    # Block sizes – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Kernel launch configuration
    grid = (
        G,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        stride_left_g,
        stride_left_m,
        stride_left_k,
        stride_right_g,
        stride_right_k,
        stride_right_n,
        stride_out_g,
        stride_out_m,
        stride_out_n,
        G,
        N,
        N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 8. Post‑processing (LayerNorm, gating, final linear)
    # ------------------------------------------------------------------
    # Convert to FP32 for the final LayerNorm and linear (more stable)
    out = out.to(torch.float32)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight.to(torch.float32),
        bias=to_out_norm_bias.to(torch.float32),
        eps=eps,
    )

    # Apply the output gate (already half‑precision -> cast)
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32

    # Final linear projection back to `dim`
    out = F.linear(out, to_out_weight.to(torch.float32))

    return out
