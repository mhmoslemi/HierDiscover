```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # Cast to float16 for speed? But weight maybe float32. We can do half computation by converting to float16 for matmul. Then back to float32 for final output.
    # However final output must be float32.
    # We'll keep everything in float16 for matrix multiplication but convert left,right to float16 before kernel. Use torch.float16 if input dtype is float32/16.

    x = input_tensor
    # LayerNorm (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Compute mean and var across last dim dim, per all other dims.
    # shape: (B,N,N,dim)
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections: left_proj, right_proj
    # left_proj_weight shape (hidden_dim, dim)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # dims: last dim
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # shape (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # gating: left_gate, etc.
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute tri mul using Triton
    # Convert to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape  # B,N,N,H

    # Permute to (B,H,N,N)
    left_perm = left_h.permute(0,3,1,2).contiguous()
    right_perm = right_h.permute(0,3,1,2).contiguous()

    # Flatten batch*head
    left_flat = left_perm.view(-1, N, N)  # shape (Z,N,N)
    right_flat = right_perm.view(-1, N, N)

    # allocate out
    out_flat = torch.empty_like(left_flat)

    # Extract strides
    stride_az = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bz = right_flat.stride(0)
    stride_bm = right_flat.stride(1)
    stride_bk = right_flat.stride(2)

    stride_cz = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # Define block size constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N, left_flat.shape[0] )

    tri_mul_kernel[grid](
        left_flat, right_flat, out_flat,
        N, N, N,
        stride_az, stride_am, stride_ak,
        stride_bz, stride_bm, stride_bk,
        stride_cz, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # After kernel, reshape back
    out_perm = out_flat.view(B, hidden_dim, N, N)
    out = out_perm.permute(0,2,3,1).contiguous()  # shape (B,N,N,hidden_dim)

    # Out_norm + out_gate + final linear
    out_norm_weight = weights["to_out_norm.weight"]
    out_norm_bias = weights["to_out_norm.bias"]
    # LayerNorm over hidden_dim
    # shape (B,N,N,H)
    mean_out = out.mean(-1, keepdim=True)
    var_out = out.var(-1, unbiased=False, keepdim=True)
    out_normed = (out - mean_out) / torch.sqrt(var_out + eps)
    out_normed = out_normed * out_norm_weight + out_norm_bias
    out_normed = out_normed * out_gate  # out_gate shape B,N,N,hidden_dim

    # final linear
    to_out_weight = weights["to_out.weight"]  # shape (dim, hidden_dim)
    out_final = torch.nn.functional.linear(out_normed, to_out_weight)  # shape (B,N,N,dim)

    # Ensure output dtype float32
    return out_final.to(torch.float32)
```

"""
TriMul (outgoing) custom kernel implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dim (dim).
2. Linear projections `left_proj`, `right_proj` (dim → hidden_dim) are applied.
3. Optional binary mask (broadcasted on the hidden dim) is multiplied to the
   projected tensors.
4. Gated linear layers (`left_gate`, `right_gate`, `out_gate`) are computed
   from the normalized input and applied with a sigmoid.
5. The expensive pair‑wise multiplication
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is evaluated as a batched matrix multiplication
      out_h = left_h @ right_hᵀ
   for every hidden dimension `d`.  This is fused into a single Triton kernel
   where each head (hidden_dim) is treated as an independent batch.
   The kernel works on FP16 data and accumulates in FP32 for numerical safety.
6. Layer‑norm over the hidden dimension, multiplication by `out_gate`,
   and a final linear projection back to `dim`.
All operations are performed without autograd (no gradients needed).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core N³ accumulation:   C = A @ Bᵀ
# A,B : (Z , M , K)  (Z = batch*hidden_dim, M = N, K = N)
# C   : (Z , M , N)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,                      # matrix extents (int)
    stride_az, stride_am, stride_ak,   # A strides
    stride_bz, stride_bm, stride_bk,   # B strides
    stride_cz, stride_cm, stride_cn,   # C strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2)               # index for (batch * hidden_dim)
    pid_m = tl.program_id(0)               # row block
    pid_n = tl.program_id(1)               # col block

    # --------------------------------------------------------------
    # Block offsets (in element indices, not bytes)
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Bounds checks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k_tiles = tl.cdiv(K, BLOCK_K)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # --------------------------------------------------------------
    for k_tile in range(num_k_tiles):
        cur_k = k_tile * BLOCK_K + offs_k          # absolute k indices
        mask_k = cur_k < K

        # Load A tile  (BLOCK_M x BLOCK_K)
        a_ptrs = (
            a_ptr
            + pid_z * stride_az
            + offs_m[:, None] * stride_am
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B tile  (BLOCK_N x BLOCK_K)
        b_ptrs = (
            b_ptr
            + pid_z * stride_bz
            + offs_n[:, None] * stride_bm
            + cur_k[None, :] * stride_bk
        )
        b = tl.load(b_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0)

        # Dot: (M,K) x (K,N) → (M,N).  B is stored as (N,K) → use trans_b=True.
        acc += tl.dot(a, b, trans_b=True).to(tl.float32)

    # --------------------------------------------------------------
    # Write result back (convert accumulator to the half‑precision output)
    # --------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_z * stride_cz
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_ptrs, acc.to(a_ptr.dtype), mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim]
        mask         : torch.Tensor [B, N, N]  (may be all‑ones)
        weights      : dict of model parameters (all on GPU)
        config       : dict containing at least "dim", "hidden_dim"
    Returns
    -------
    torch.Tensor [B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    use_mask = not config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B,N,N,dim]  (float32)

    # ------------------------------------------------------------------
    # 2) Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on hidden_dim)
    # ------------------------------------------------------------------
    if use_mask:
        # mask is bool / float; bring to same dtype as left/right
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gated linear layers (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul (batched matmul over the sequence dimension)
    #    Compute   out[h] = left_h @ right_hᵀ  for each hidden head h.
    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy N³ kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape        # H == hidden_dim

    # Change layout to (B, H, N, N) → (Z, N, N) where Z = B*H
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    Z = B * H
    left_flat = left_perm.view(Z, N, N)
    right_flat = right_perm.view(Z, N, N)

    # Allocate output buffer (FP16)
    out_flat = torch.empty_like(left_flat)

    # Strides needed by the Triton kernel
    stride_az, stride_am, stride_ak = left_flat.stride()
    stride_bz, stride_bm, stride_bk = right_flat.stride()
    stride_cz, stride_cm, stride_cn = out_flat.stride()

    # Choose tile sizes (tuned for H100, FP16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        Z,
    )

    # Launch kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bm,
        stride_bk,
        stride_cz,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim) and convert to float32
    out_perm = out_flat.view(B, H, N, N)                # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, N, N, H)
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # broadcasted over the hidden dim

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])  # (B,N,N,dim)

    return out
