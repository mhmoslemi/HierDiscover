@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_ob, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
    EPS: tl.constexpr = 0.0
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute base offsets
    i_base = pid_m * BLOCK_M
    j_base = pid_n * BLOCK_N

    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # compute hidden dimension loop
    for d_offset in range(0, hidden_dim, BLOCK_D):
        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # loop over K
        for k_tile in range(0, (N + BLOCK_K - 1) // BLOCK_K):
            k_base = k_tile * BLOCK_K
            k = k_base + tl.arange(0, BLOCK_K)
            mask_k = k < N

            # compute pointer for left: left[batch_id, i, k, d_offset:d_offset+BLOCK_D]
            # using broadcasting:
            # i axis: shape (BLOCK_M, 1, 1)
            # k axis: shape (1, BLOCK_K, 1)
            # d axis: shape (1, 1, BLOCK_D)
            d_range = tl.arange(0, BLOCK_D)

            left_ptrs = left_ptr + batch_id * stride_bl \
                + i[:, None, None] * stride_il \
                + k[None, :, None] * stride_kl \
                + (d_offset + d_range)[None, None, :] * stride_dl
            # load
            left_tile = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

            # right: shape (BLOCK_K, BLOCK_N, BLOCK_D) because we want (k, j)
            right_ptrs = right_ptr + batch_id * stride_br \
                + j[None, :, None] * stride_jr \
                + k[:, None, None] * stride_kr \
                + (d_offset + d_range)[None, None, :] * stride_dr
            right_tile = tl.load(right_ptrs, mask=mask_j[None, :, None] & mask_k[:, None, None], other=0.0)

            # Now compute dot for each d:
            # Loop over d
            for d in range(BLOCK_D):
                a = left_tile[:, :, d]   # (M, K)
                b = right_tile[:, :, d]  # (K, N)
                # using tl.dot
                acc[:, :, d] += tl.dot(a, b)

        # after K loop, store acc to output
        out_ptrs = out_ptr + batch_id * stride_ob \
            + i[:, None, None] * stride_io \
            + j[None, :, None] * stride_jo \
            + (d_offset + dl_range)[None, None, :] * stride_do
        # but need dl_range defined

        tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None])

```
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_ob, stride_io, stride_jo, stride_do,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of slices for hidden dim
    num_d_slices = tl.cdiv(hidden_dim, BLOCK_D)
    for ds in range(0, num_d_slices):
        d_off = ds * BLOCK_D
        # mask for d dimension
        d_range = tl.arange(0, BLOCK_D)
        d_valid = (d_off + d_range) < hidden_dim

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # loop over K dimension
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for kt in range(0, num_k_tiles):
            k_off = kt * BLOCK_K
            offs_k = k_off + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # Load left tile (M, K, D)
            left_ptrs = left_ptr + batch_id * stride_bl \
                + offs_i[:, None, None] * stride_il \
                + offs_k[None, :, None] * stride_kl \
                + (d_off + d_range)[None, None, :] * stride_dl
            left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0)

            # Load right tile (K, N, D) where N corresponds to j indices
            right_ptrs = right_ptr + batch_id * stride_br \
                + offs_j[None, :, None] * stride_jr \
                + offs_k[:, None, None] * stride_kr \
                + (d_off + d_range)[None, None, :] * stride_dr
            right = tl.load(right_ptrs, mask=mask_j[None, :, None] & mask_k[:, None, None], other=0.0)

            # Compute dot for each d
            for d in range(BLOCK_D):
                # TODO: maybe mask d_valid?
                a = left[:, :, d]   # (M, K)
                b = right[:, :, d]  # (K, N)
                acc[:, :, d] += tl.dot(a, b)

        # Store accumulator
        out_ptrs = out_ptr + batch_id * stride_ob \
            + offs_i[:, None, None] * stride_io \
            + offs_j[None, :, None] * stride_jo \
            + (d_off + d_range)[None, None, :] * stride_do
        tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & d_valid[None, None, :])
```

    right = right * right_gate

    N = seq_len

    stride_br, stride_jr, stride_kr, stride_dr = right.stride()

    BLOCK_K = 64

TriMul “outgoing” kernel (AlphaFold3 style) – fast GPU contraction.

The heavy part of TriMul is the 4‑D tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]          (d = hidden)

which is a batched MatMul for every hidden channel d.
We implement that contraction with a Triton kernel that tiles the (i,j)
output plane (BLOCK_M×BLOCK_N) and the reduction over k (BLOCK_K).
The hidden dimension is split into small slices (BLOCK_D) that fit into
registers; each thread accumulates a vector of length BLOCK_D for a
single (i,j) pair and calls `tl.dot` on the slice for every k‑tile.
All surrounding operations (LayerNorm, Linear projections, gates, final
LayerNorm/Linear) are performed in PyTorch, but the O(N³·hidden) core
is computed in the custom Triton kernel.

The kernel works for any batch size (≤2), any sequence length N (≤1024)
and hidden_dim that is a multiple of BLOCK_D (we use BLOCK_D=8, hidden_dim
=128 in the provided tests).  It also respects an optional mask tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shapes / sizes
    B, N, hidden_dim,
    # strides (in elements) for the three tensors
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_jr, stride_kr, stride_dr,
    stride_ob, stride_io, stride_jo, stride_do,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Triton implementation of the TriMul contraction.

    Arguments are pointers and strides of the input tensors.
    Each program (block) computes a tile of size BLOCK_M×BLOCK_N over the
    (i,j) plane for a fixed batch index.  The hidden dimension is processed
    in slices of length BLOCK_D.
    """
    batch_id = tl.program_id(0)          # batch dimension
    pid_m = tl.program_id(1)             # tile row (i)
    pid_n = tl.program_id(2)             # tile col (j)

    # ------------------- index preparation -------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [M]
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [N]
    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of hidden‑dim slices
    num_d_slices = tl.cdiv(hidden_dim, BLOCK_D)

    # ------------------- hidden‑dim loop -------------------
    for ds in range(0, num_d_slices):
        # start offset of the current hidden slice
        d_off = ds * BLOCK_D
        # vector of hidden indices inside the slice
        d_idx = d_off + tl.arange(0, BLOCK_D)
        # mask for the tail slice (when hidden_dim % BLOCK_D != 0)
        mask_d = d_idx < hidden_dim

        # accumulator for this slice: shape (BLOCK_M, BLOCK_N, BLOCK_D)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ------------------- reduction over k -------------------
        num_k_tiles = tl.cdiv(N, BLOCK_K)
        for kt in range(0, num_k_tiles):
            k_off = kt * BLOCK_K
            offs_k = k_off + tl.arange(0, BLOCK_K)      # [K]
            mask_k = offs_k < N

            # ----- load left tile (M, K, D) -----
            # pointer = base + i*stride_il + k*stride_kl + d*stride_dl
            left_ptrs = (
                left_ptr
                + batch_id * stride_bl
                + offs_i[:, None, None] * stride_il
                + offs_k[None, :, None] * stride_kl
                + d_idx[None, None, :] * stride_dl
            )
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None],
                other=0.0,
            )   # (M, K, D)

            # ----- load right tile (K, N, D) -----
            # we need right[j, k, d] → reinterpret as (K, N, D) = (k, j, d)
            right_ptrs = (
                right_ptr
                + batch_id * stride_br
                + offs_j[None, :, None] * stride_jr   # j dimension
                + offs_k[:, None, None] * stride_kr   # k dimension (first)
                + d_idx[None, None, :] * stride_dr
            )
            right_tile = tl.load(
                right_ptrs,
                mask=mask_j[None, :, None] & mask_k[:, None, None],
                other=0.0,
            )   # (K, N, D)

            # ----- compute dot product for every d in the slice -----
            for d in range(BLOCK_D):
                a = left_tile[:, :, d]   # (M, K)
                b = right_tile[:, :, d]  # (K, N)
                acc[:, :, d] += tl.dot(a, b)   # (M, N)

        # ------------------- write back the slice -------------------
        out_ptrs = (
            out_ptr
            + batch_id * stride_ob
            + offs_i[:, None, None] * stride_io
            + offs_j[None, :, None] * stride_jo
            + d_idx[None, None, :] * stride_do
        )
        store_mask = (
            mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        )
        tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Python entry point – performs the full TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------- unpack -------------------
    input_tensor, mask_tensor, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------- optional mask handling -------------------
    # `mask_tensor` may be None when config["nomask"] is True.
    if mask_tensor is None or config.get("nomask", True):
        mask = torch.ones((bs, N, N), dtype=dtype, device=device)
    else:
        mask = mask_tensor.to(dtype)

    # ------------------- LayerNorm on the input -------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------- Linear projections -------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)    # (bs, N, N, hidden)
    right = F.linear(x, right_proj_weight)

    # ------------------- apply mask (broadcast over hidden) -------------------
    mask_expanded = mask.unsqueeze(-1)                     # (bs, N, N, 1)
    left = left * mask_expanded
    right = right * mask_expanded

    # ------------------- Gating (sigmoid) -------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    #   Triton contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    # ------------------------------------------------------------------
    # allocate output buffer for the contraction
    out = torch.empty((bs, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides (in elements) – PyTorch gives them in element units
    stride_bl, stride_il, stride_kl, stride_dl = left.stride()
    stride_br, stride_jr, stride_kr, stride_dr = right.stride()
    stride_ob, stride_io, stride_jo, stride_do = out.stride()

    # tile sizes – chosen to fit into registers on H100
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64
    BLOCK_D = 8   # hidden_dim is a multiple of 8 in all test cases

    grid = (
        bs,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    trimul_outgoing_kernel[grid](
        left, right, out,
        bs, N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_dl,
        stride_br, stride_jr, stride_kr, stride_dr,
        stride_ob, stride_io, stride_jo, stride_do,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    #   post‑processing (LayerNorm → gate → final linear)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply the output gate
    out_norm = out_norm * out_gate

    # final projection back to the original channel dimension
    to_out_weight = weights["to_out.weight"]
    final = F.linear(out_norm, to_out_weight)   # (bs, N, N, dim)

    return final
