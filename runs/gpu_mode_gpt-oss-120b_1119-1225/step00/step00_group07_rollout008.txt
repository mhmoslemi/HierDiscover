"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

The kernel implements the O(N³) update
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
by treating each hidden dimension d (and batch b) as an independent matrix
multiplication:
    out_{b,d} = left_{b,d} @ right_{b,d}ᵀ
where left_{b,d} and right_{b,d} are N×N matrices.  The heavy N³ work is
performed by a custom batched GEMM kernel written in Triton, while the
remaining linear / LayerNorm / gating operations use fast PyTorch primitives.
The implementation supports optional masking, arbitrary batch size (1‑2),
sequence length up to 1024, hidden dimensions up to 384 and works on H100
with mixed‑precision friendly block sizes.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton batched matrix‑multiply kernel (C = A @ Bᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr,                         # A: [B*H, M, K]
    b_ptr,                         # B: [B*H, N, K]  (will be read as Bᵀ)
    c_ptr,                         # C: [B*H, M, N]
    M, N, K,                       # static sizes (int)
    stride_am, stride_ak, stride_aj,   # strides of A  (batch, row, col)
    stride_bm, stride_bk, stride_bj,   # strides of B  (batch, col‑k, row‑j)
    stride_cm, stride_cn,               # strides of C  (row‑i, col‑j)
    stride_cb,                          # batch stride of C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for many independent (batch*hidden) matrices.
    Grid: (num_blocks_M, num_blocks_N, batch_hidden)
    """
    pid_batch = tl.program_id(2)          # which (batch, hidden) pair
    pid_m = tl.program_id(0)              # block row
    pid_n = tl.program_id(1)              # block col

    # ---- block offsets ----
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- iterate over K dimension in tiles ----
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers to A and Bᵀ tiles
        a_ptrs = a_ptr + pid_batch * stride_am \
                 + offs_m[:, None] * stride_ak \
                 + offs_k[None, :] * stride_aj
        b_ptrs = b_ptr + pid_batch * stride_bm \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bj

        # load with masks (masked elements are zero)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate dot product
        acc += tl.dot(a, b)

    # ---- write back result ----
    c_ptrs = c_ptr + pid_batch * stride_cb \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor of shape [B, N, N, dim]
            - mask : torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of model parameters (LayerNorm, Linear weights)
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on same device as input)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dim (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask is broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        # ensure mask is float for multiplication
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating values
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as many independent GEMMs using a Triton kernel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B*hidden, N, N) for GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # output buffer
    out_mat = torch.empty_like(left_mat)

    # strides in element units (not bytes)
    stride_am = left_mat.stride(0)   # batch stride (N*N)
    stride_ak = left_mat.stride(1)   # row‑stride (N)
    stride_aj = left_mat.stride(2)   # col‑stride (1)

    stride_bm = right_mat.stride(0)  # batch stride (N*N)
    # For Bᵀ we need stride over the k‑dimension (original column) and over
    # the j‑dimension (original row).  With layout (batch, N, K):
    stride_bk = right_mat.stride(2)  # stride of K (col) → 1
    stride_bj = right_mat.stride(1)  # stride of N (row) → K (=N)

    stride_cm = out_mat.stride(1)    # row‑stride (N)
    stride_cn = out_mat.stride(2)    # col‑stride (1)
    stride_cb = out_mat.stride(0)    # batch stride (N*N)

    # Choose block sizes – tuned for H100 and N∈[256,1024]
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # launch grid: (blocks_M, blocks_N, batch_hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        N, N, N,                     # M, N, K
        stride_am, stride_ak, stride_aj,
        stride_bm, stride_bk, stride_bj,
        stride_cm, stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
