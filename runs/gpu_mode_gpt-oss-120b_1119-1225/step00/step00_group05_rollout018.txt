import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Compute masks
        mask_a = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        mask_b = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # Compute pointers
        a_ptrs = a_ptr + batch_id * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + batch_id * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_id * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]

def custom_kernel(data):
    x, mask, weights, config = data
    # extract dims
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    # get weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input
    # Note: using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if provided
    if not config.get("nomask", True) and mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gating
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul using Triton
    # permute to (B, hidden_dim, N, N) and flatten batch+hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*hidden_dim, N, N)
    B_h = B * hidden_dim
    left = left.view(B_h, N, N)
    right = right.view(B_h, N, N)
    # transpose right for matmul
    right_T = right.transpose(1, 2).contiguous()  # shape (B_h, N, N)

    # Cast to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right_T.to(torch.float16)
    # allocate output
    out_fp16 = torch.empty_like(left_fp16)   # shape (B_h, N, N) dtype half

    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B_h, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)
    batched_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N, N, N,
        left_fp16.stride(0), left_fp16.stride(1), left_fp16.stride(2),
        right_fp16.stride(0), right_fp16.stride(1), right_fp16.stride(2),
        out_fp16.stride(0), out_fp16.stride(1), out_fp16.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # reshape back
    out = out_fp16.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape B,N,N,hidden_dim

    # Output layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Multiply by out_gate (shape B,N,N,hidden_dim)
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, to_out_weight)  # shape B,N,N,dim

    return out
```

"""
Outgoing TriMul (AlphaFold3) implementation.

The forward pass consists of:
1. LayerNorm over the input channel dimension.
2. Two linear projections (left/right) and three gating projections.
3. Elementwise mask (if provided) and gating.
4. Core contraction:
     out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
   This N×N×N operation is executed as a batched half‑precision GEMM
   using a custom Triton kernel (one batch per (batch, hidden_dim) pair).
5. LayerNorm over the hidden dimension, gating, and final linear projection
   back to the original channel size.

All heavy compute (step 4) runs in FP16 on the GPU with 32‑bit accumulation.
The rest of the ops are performed in PyTorch for simplicity.
The final output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,            # left  (B_h, M, K)  FP16
    b_ptr,            # right (B_h, K, N)  FP16 (already transposed)
    c_ptr,            # out   (B_h, M, N)  FP16
    M, N, K,          # tensor dimensions (int)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (A: [B_h, M, K], B: [B_h, K, N])"""
    batch_id = tl.program_id(0)          # which (batch, hidden) pair
    pid_m   = tl.program_id(1)          # output block row index
    pid_n   = tl.program_id(2)          # output block col index

    # ------------------------------------------------------------------
    # Compute the start offsets of the block we are responsible for.
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 for better precision.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in tiles of size BLOCK_K.
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads.
        mask_a = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        mask_b = (offs_k < K)[:, None] & (offs_n < N)[None, :]

        # Pointers to the current tiles of A and B.
        a_ptrs = (a_ptr
                  + batch_id * stride_a_batch
                  + offs_m[:, None] * stride_a_m
                  + offs_k[None, :] * stride_a_k)
        b_ptrs = (b_ptr
                  + batch_id * stride_b_batch
                  + offs_k[:, None] * stride_b_k
                  + offs_n[None, :] * stride_b_n)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        # Tensor‑core dot (FP16×FP16 → FP32 accumulation)
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result back to C.
    # ------------------------------------------------------------------
    c_ptrs = (c_ptr
              + batch_id * stride_c_batch
              + offs_m[:, None] * stride_c_m
              + offs_n[None, :] * stride_c_n)
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, config = data
    # x: [B, N, N, dim]   (float32, on CUDA)
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Load model weights
    # --------------------------------------------------------------
    norm_weight           = weights["norm.weight"]
    norm_bias             = weights["norm.bias"]
    left_proj_weight      = weights["left_proj.weight"]
    right_proj_weight     = weights["right_proj.weight"]
    left_gate_weight      = weights["left_gate.weight"]
    right_gate_weight     = weights["right_gate.weight"]
    out_gate_weight       = weights["out_gate.weight"]
    to_out_norm_weight    = weights["to_out_norm.weight"]
    to_out_norm_bias      = weights["to_out_norm.bias"]
    to_out_weight         = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3. Optional mask
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4. Gating (linear + sigmoid)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core contraction – batched GEMM in Triton
    # --------------------------------------------------------------
    # Rearrange tensors to (B, hidden_dim, N, N) and flatten batch*hidden.
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()
    B_h = B * hidden_dim
    left  = left.view(B_h, N, N)            # (B_h, M=N, K=N)
    right = right.view(B_h, N, N)           # (B_h, K=N, N)

    # Transpose right so that we can perform A @ B (no extra transpose inside kernel)
    right_t = right.transpose(1, 2).contiguous()   # (B_h, K=N, N)

    # Cast to FP16 for the kernel (Tensor Cores)
    left_f16   = left.to(torch.float16)
    right_f16  = right_t.to(torch.float16)
    out_f16    = torch.empty_like(left_f16)   # (B_h, N, N), dtype=fp16

    # Block sizes – tuned for H100 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B_h,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    _batched_matmul_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        N, N, N,                                   # M, N, K
        left_f16.stride(0), left_f16.stride(1), left_f16.stride(2),   # A strides
        right_f16.stride(0), right_f16.stride(1), right_f16.stride(2), # B strides
        out_f16.stride(0), out_f16.stride(1), out_f16.stride(2),       # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to FP32 and reshape to original layout
    out = out_f16.to(torch.float32)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 6. Output LayerNorm, gating, final projection
    # --------------------------------------------------------------
    out = F.layer_norm(
        out, (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # final projection to dim, no bias

    return out
