@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # coordinate indices
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)

    # mask bounds
    mask_i = i < N
    mask_j = j < N
    mask_k = k < N

    # base pointers for left and right (excluding k)
    left_base = left_ptr + b * stride_left_b + h * stride_left_h + i[:, None] * stride_left_i
    right_base = right_ptr + b * stride_right_b + h * stride_right_h + j[None, :] * stride_right_j

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # number of k tiles
    num_tiles = tl.cdiv(N, BLOCK_K)

    for tile_idx in range(num_tiles):
        k_offset = tile_idx * BLOCK_K
        cur_k = k + k_offset
        cur_mask_k = cur_k < N

        # load tiles
        left_tile = tl.load(left_base + cur_k[None, :] * stride_left_k,
                            mask=mask_i[:, None] & cur_mask_k[None, :],
                            other=0.0).to(tl.float32)
        right_tile = tl.load(right_base + cur_k[:, None] * stride_right_k,
                            mask=cur_mask_k[:, None] & mask_j[None, :],
                            other=0.0).to(tl.float32)

        acc += tl.dot(left_tile, right_tile)

    # store results
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)

"""
TriMul “outgoing” forward implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension `dim`.
2. Project the normalized tensor to a hidden size `hidden_dim` with two linear layers:
   `left = x @ W_left.T`, `right = x @ W_right.T`.
3. Apply an optional pairwise mask (shape [B,N,N]) and element‑wise gates
   (`left_gate`, `right_gate`, `out_gate`) obtained by sigmoid‑activated linear layers.
4. Core operation – a reduction over the third axis (`k`) of the 4‑D tensor:
   ```
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   ```
   This is exactly a batched matrix‑multiply for each hidden channel `h`.
   A custom Triton kernel performs this N³ reduction efficiently on H100,
   loading `left` as (i,k) tiles and `right` as (k,j) tiles, accumulating
   in FP32 while the data lives in FP16.
5. Apply a second Layer‑Norm over `hidden_dim`, multiply by `out_gate`,
   and finish with a linear projection back to the original `dim`.

The kernel fuses the heavy `N³` reduction, dramatically reducing runtime
compared with a pure‑Python/PyTorch implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def _trimul_kernel(
    left_ptr,                # half*  [B,N,N,H]
    right_ptr,               # half*  [B,N,N,H]
    out_ptr,                 # float* [B,N,N,H]
    B, N, H,                 # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,   # tile size in i‑dimension
    BLOCK_J: tl.constexpr,   # tile size in j‑dimension
    BLOCK_K: tl.constexpr,   # tile size in reduction k‑dimension
):
    """
    Perform:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for all batches `b` and hidden channels `h`.
    Each program instance handles a tile (i,j) of the output for a single
    (b,h) pair.
    """
    pid_i = tl.program_id(0)          # tile index on i‑axis
    pid_j = tl.program_id(1)          # tile index on j‑axis
    pid_bh = tl.program_id(2)         # combined batch‑hidden index

    # decode batch and hidden channel
    b = pid_bh // H
    h = pid_bh % H

    # global indices for the current tile
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    k = tl.arange(0, BLOCK_K)

    mask_i = i < N
    mask_j = j < N
    mask_k = k < N

    # Base pointers (without the reduction dimension)
    left_base  = left_ptr \
                 + b * stride_left_b \
                 + h * stride_left_h \
                 + i[:, None] * stride_left_i          # shape (BLOCK_I, 1)

    right_base = right_ptr \
                 + b * stride_right_b \
                 + h * stride_right_h \
                 + j[None, :] * stride_right_j        # shape (1, BLOCK_J)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    num_tiles = tl.cdiv(N, BLOCK_K)

    for tile_k in range(num_tiles):
        k_off = tile_k * BLOCK_K
        cur_k = k + k_off
        cur_mask_k = cur_k < N

        # left[i,k] tile  (BLOCK_I x BLOCK_K)
        left_ptrs = left_base + cur_k[None, :] * stride_left_k
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & cur_mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # right[k,j] tile (BLOCK_K x BLOCK_J) – note the transposed layout
        right_ptrs = right_base + cur_k[:, None] * stride_right_k
        right_tile = tl.load(
            right_ptrs,
            mask=cur_mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # accumulate the dot product over k
        acc += tl.dot(left_tile, right_tile)   # (BLOCK_I, BLOCK_J)

    # store the result back to global memory (FP32)
    out_ptrs = out_ptr \
               + b * stride_out_b \
               + h * stride_out_h \
               + i[:, None] * stride_out_i \
               + j[None, :] * stride_out_j

    mask_out = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask: [B, N, N] (bool or float) or None
            - weights: dict of model parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    batch, seq_len, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Convenience aliases for weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional pairwise mask
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # (B,N,N,1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid‑activated linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ reduction – Triton kernel
    # ------------------------------------------------------------------
    # Cast to FP16 for bandwidth‑limited kernel, keep a FP32 output buffer.
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out = torch.empty((batch, seq_len, seq_len, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    sL_b, sL_i, sL_k, sL_h = left_fp16.stride()
    sR_b, sR_j, sR_k, sR_h = right_fp16.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # Tunable block sizes – empirically good for H100, seq_len up to 1024
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        (seq_len + BLOCK_I - 1) // BLOCK_I,
        (seq_len + BLOCK_J - 1) // BLOCK_J,
        batch * hidden_dim,
    )

    _trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        batch,
        seq_len,
        hidden_dim,
        sL_b,
        sL_i,
        sL_k,
        sL_h,
        sR_b,
        sR_j,
        sR_k,
        sR_h,
        sO_b,
        sO_i,
        sO_j,
        sO_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # good default for H100
    )

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 7) Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    output = F.linear(out, to_out_weight)

    return output
