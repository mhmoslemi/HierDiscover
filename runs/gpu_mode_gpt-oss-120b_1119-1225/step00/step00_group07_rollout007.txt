def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_id = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # compute global offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_n = offs_n < N

"""
TriMul (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections (left/right) and their gating linear layers.
   Apply an optional binary mask and element‑wise sigmoid gates.
3. The core O(N³) operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This can be written as a batched matrix multiply:
      out_d = left_d @ right_dᵀ  for each hidden dimension d.
   We flatten the batch * hidden dimension into a single batch axis and
   launch a Triton kernel that computes many independent GEMMs
   (A @ Bᵀ) in parallel.
4. Layer‑norm the result over the hidden dimension, apply the output gate,
   and a final linear projection back to the original channel size.

The Triton kernel processes a batch of GEMMs with the usual tile‑based
blocking (BLOCK_M × BLOCK_N × BLOCK_K).  It works for any batch size
≤ 256 (B∈{1,2}, hidden_dim=128) and sequence lengths up to 1024.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ Bᵀ) where each batch element is
# (M × K) @ (K × N) -> (M × N)
# ----------------------------------------------------------------------
@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix sizes (int32 scalars)
    stride_am, stride_ak, stride_ab,   # strides for A (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # strides for B (batch, K, N) – B is already transposed
    stride_cm, stride_cn, stride_cb,   # strides for C (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_id = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # --------------------------------------------------------------
    # Compute the tile offsets.
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles.
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # K‑tile coordinates
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # Pointers for the current tiles
        a_ptrs = (a_ptr
                  + batch_id * stride_ab
                  + (offs_m[:, None] * stride_am)
                  + (offs_k[None, :] * stride_ak))
        b_ptrs = (b_ptr
                  + batch_id * stride_bb
                  + (offs_k[:, None] * stride_bk)
                  + (offs_n[None, :] * stride_bn))

        # Load tiles (mask out‑of‑bounds with zeros)
        a = tl.load(
            a_ptrs,
            mask=(offs_k < K)[:, None] & mask_m[:, None],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Multiply‑accumulate
        acc += tl.dot(a, b)  # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write back the result.
    # --------------------------------------------------------------
    c_ptrs = (c_ptr
              + batch_id * stride_cb
              + (offs_m[:, None] * stride_cm)
              + (offs_n[None, :] * stride_cn))
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C]
        - mask         : torch.Tensor [B, N, N] (may be None)
        - weights      : dict of model weights (see function body)
        - config       : dict containing "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments.
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # Load weights.
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input layer‑norm.
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # 2) Linear projections (no bias) and optional masking.
    # --------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)   # (hidden_dim, dim) weight
    right = F.linear(x, right_proj_weight)

    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_f = mask.to(dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 3) Gating (sigmoid of linear layer) and apply gates.
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4) Core O(N³) “tri‑multiplicative” update.
    #    Compute batched GEMM: left_d @ right_dᵀ for each hidden dimension.
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden_dim

    # Rearrange to (batch, hidden, N, N) and flatten batch*hidden.
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_flat = left.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_flat = right.view(B * hidden_dim, N, N)  # [B*H, N, N]

    # Cast to fp16 for the GEMM (reduce bandwidth, keep FP32 accumulator).
    left_fp = left_flat.to(torch.float16)
    right_fp = right_flat.to(torch.float16)

    # Transpose right so we can use the standard A @ Bᵀ formulation.
    right_fp_T = right_fp.transpose(-2, -1).contiguous()  # [B*H, N, N] (K, N)

    # Output buffer.
    out_fp = torch.empty_like(left_fp, dtype=torch.float16)

    # --------------------------------------------------------------
    # Launch Triton kernel.
    # --------------------------------------------------------------
    batch = left_fp.shape[0]         # B * hidden_dim
    M = N
    K = N
    # Block size – tuned for H100 (the defaults work well).
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        batch,
    )

    batch_matmul_kernel[grid](
        a_ptr=left_fp,
        b_ptr=right_fp_T,
        c_ptr=out_fp,
        M=M,
        N=N,
        K=K,
        stride_am=left_fp.stride(1),
        stride_ak=left_fp.stride(2),
        stride_ab=left_fp.stride(0),
        stride_bk=right_fp_T.stride(1),
        stride_bn=right_fp_T.stride(2),
        stride_bb=right_fp_T.stride(0),
        stride_cm=out_fp.stride(1),
        stride_cn=out_fp.stride(2),
        stride_cb=out_fp.stride(0),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # Synchronise to make sure kernel has finished.
    torch.cuda.synchronize()

    # Cast back to FP32 for the remaining (lightweight) ops.
    out = out_fp.to(torch.float32)

    # Restore original layout: [B, N, N, hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5) Output layer‑norm, gating, final linear projection.
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate                           # broadcast over hidden_dim
    out = F.linear(out, to_out_weight)             # (dim, hidden_dim) weight

    return out
```
