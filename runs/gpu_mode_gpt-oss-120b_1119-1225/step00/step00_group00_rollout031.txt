@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):

    # compute program ids
    pid = tl.program_id(axis=0)
    # compute number of blocks
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    total_per_batch = num_m * num_n
    batch_idx = pid // total_per_batch
    block_idx = pid % total_per_batch
    pid_m = block_idx // num_n
    pid_n = block_idx % num_n

    # base offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # iterate over hidden dim
    for h_start in range(0, H, BLOCK_H):
        cur_H = tl.minimum(BLOCK_H, H - h_start)
        offs_h = tl.arange(0, BLOCK_H) + h_start

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # load left tile (M,K,H)
            left_ptrs = left_ptr + batch_idx * stride_lb \
                + offs_m[:, None, None] * stride_li \
                + offs_k[None, :, None] * stride_lk \
                + offs_h[None, None, :] * stride_lh
            left = tl.load(left_ptrs, mask=mask_m[:, None, None] & mask_k[None, :, None] & (offs_h < H), other=0.0).to(tl.float32)

            # load right tile transposed (K,N,H)
            right_ptrs = right_ptr + batch_idx * stride_rb \
                + offs_k[:, None, None] * stride_rk \
                + offs_n[None, :, None] * stride_rj \
                + offs_h[None, None, :] * stride_rh
            right_T = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_n[None, :, None] & (offs_h < H), other=0.0).to(tl.float32)

            # accumulate using dot per hidden sub block
            # For each h in cur_H
            for hh in range(0, cur_H):
                # left slice: (M,K)
                L = left[:, :, hh]
                # right_T slice: (K,N)
                R = right_T[:, :, hh]
                acc[:, :, hh] += tl.dot(L, R)

        # store results
        out_ptrs = out_ptr + batch_idx * stride_ob \
            + offs_m[:, None, None] * stride_oi \
            + offs_n[None, :, None] * stride_oj \
            + offs_h[None, None, :] * stride_oh
        tl.store(out_ptrs, acc, mask=mask_m[:, None, None] & mask_n[None, :, None] & (offs_h < H))

import torch
import triton
import triton.language as tl

def _trimul_kernel(left, right):
    # left,right: shape [B,N,N,H] dtype float16 (or float32 but will be loaded as half for compute)
    B, N, _, H = left.shape
    assert N == left.shape[2] and H == right.shape[3]
    # allocate output float32
    out = torch.empty_like(left, dtype=torch.float32)
    # compute strides
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()
    # define block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_H = 32

    # compute grid: one block per (batch, tile_i, tile_j)
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * grid_m * grid_n

    trimul_outgoing_kernel[total_blocks](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B, N=N, H=H,
        stride_lb=stride_lb, stride_li=stride_li, stride_lk=stride_lk, stride_lh=stride_lh,
        stride_rb=stride_rb, stride_rj=stride_rj, stride_rk=stride_rk, stride_rh=stride_rh,
        stride_ob=stride_ob, stride_oi=stride_oi, stride_oj=stride_oj, stride_oh=stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    return out

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # compute program IDs
    pid = tl.program_id(0)
    # number of tiles
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m * num_n
    batch = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_n
    pid_n = block_idx % num_n

    # compute start offsets for i and j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # loop over hidden dimension in chunks
    h_offset = 0
    while h_offset < H:
        offs_h = tl.arange(0, BLOCK_H) + h_offset
        mask_h = offs_h < H

        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # iterate over k dimension
        for k_start in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k_start * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # load left tile (M,K,H)
            left_ptrs = left_ptr \
                + batch * stride_lb \
                + offs_m[:, None, None] * stride_li \
                + offs_k[None, :, None] * stride_lk \
                + offs_h[None, None, :] * stride_lh
            left = tl.load(left_ptrs,
                mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0)
            left = left.to(tl.float32)

            # load right tile transposed (K,N,H)
            right_ptrs = right_ptr \
                + batch * stride_rb \
                + offs_k[:, None, None] * stride_rk \
                + offs_n[None, :, None] * stride_rj \
                + offs_h[None, None, :] * stride_rh
            right = tl.load(right_ptrs,
                mask=mask_k[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :],
                other=0.0)
            right = right.to(tl.float32)

            # accumulate via dot product per hidden slice
            for hh in range(0, BLOCK_H):
                # early exit if this hidden slice out-of-bounds
                # (the mask_h already ensures values are zero for OOB)
                a = left[:, :, hh]   # (M, K)
                b = right[:, :, hh]  # (K, N)
                acc[:, :, hh] += tl.dot(a, b)

        # store the accumulated result for this hidden chunk
        out_ptrs = out_ptr \
            + batch * stride_ob \
            + offs_m[:, None, None] * stride_oi \
            + offs_n[None, :, None] * stride_oj \
            + offs_h[None, None, :] * stride_oh
        tl.store(out_ptrs,
            acc,
            mask=mask_m[:, None, None] & mask_n[None, :, None] & mask_h[None, None, :])
        h_offset += BLOCK_H

    # end kernel

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply mask if provided
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # Apply elementwise gating
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # Core multiplication
    out = _trimul_kernel(left_f16, right_f16)  # shape [B,N,N,hidden_dim] float32

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # gating
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) – AlphaFold3 core operator.

Given a tensor x of shape [B, N, N, D] the forward pass performs
    1) LayerNorm over the last dimension (D).
    2) Linear projections left_proj, right_proj, and three gates
       (left_gate, right_gate, out_gate).  The projections and gates are
       linear layers without bias, followed by a sigmoid on the gates.
    3) Optional masking of the projected tensors.
    4) Element‑wise gating:  left = left * left_gate,
                           right = right * right_gate.
    5) Tri‑multiplicative update:
          out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
       for every hidden channel h.  This is a batch of independent
       matrix‑multiplications (one per hidden dimension).  The heavy
       computation is carried out by a custom Triton kernel that tiles the
       i‑, j‑, k‑ and hidden‑dimensions and uses `tl.dot` for each hidden
       slice.
    6) LayerNorm over the hidden dimension, element‑wise multiplication
       with out_gate, and a final linear projection back to D.

The implementation uses PyTorch for the cheap linear layers and a
hand‑written Triton kernel for step 5.  The kernel works for any batch
size (B ≤ 2) and sequence length up to 1024, with hidden dimensions up
to 384.  All arithmetic is performed in float32 (the kernel accumulates
in float32 while loading the projected tensors as float16 for reduced
memory bandwidth).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched Tri‑Multiplicative update implemented in Triton.

    Arguments
    ---------
    left, right : torch.Tensor
        Tensors of shape [B, N, N, H] and dtype torch.float16.
        ``left`` corresponds to (b, i, k, h) and ``right`` to (b, j, k, h).

    Returns
    -------
    out : torch.Tensor
        Tensor of shape [B, N, N, H] (float32) where
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
    """
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float32)

    # Strides for pointer arithmetic (int)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – chosen to give enough parallelism while fitting
    # registers on an H100.
    BLOCK_M = 64   # tile along i
    BLOCK_N = 64   # tile along j
    BLOCK_K = 32   # tile along k
    BLOCK_H = 32   # vectorised over hidden channels

    # Grid: one program per (batch, tile_i, tile_j)
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * grid_m * grid_n

    trimul_outgoing_kernel[total_blocks](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_lb=stride_lb,
        stride_li=stride_li,
        stride_lk=stride_lk,
        stride_lh=stride_lh,
        stride_rb=stride_rb,
        stride_rj=stride_rj,
        stride_rk=stride_rk,
        stride_rh=stride_rh,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oh=stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Core Triton kernel for the Tri‑Multiplicative update."""
    pid = tl.program_id(0)

    # ----- decode program id into (batch, tile_i, tile_j) -----
    num_m = tl.cdiv(N, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_m * num_n

    batch = pid // blocks_per_batch
    tile_idx = pid % blocks_per_batch
    pid_m = tile_idx // num_n
    pid_n = tile_idx % num_n

    # ----- start indices for the i and j tiles -----
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----- loop over hidden dimension in chunks of BLOCK_H -----
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for hb in range(num_h_blocks):
        h_start = hb * BLOCK_H
        offs_h = tl.arange(0, BLOCK_H) + h_start
        mask_h = offs_h < H

        # accumulator for this hidden chunk (float32)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # ----- reduction over the k dimension -----
        num_k_blocks = tl.cdiv(N, BLOCK_K)
        for kb in range(num_k_blocks):
            k_start = kb * BLOCK_K
            offs_k = tl.arange(0, BLOCK_K) + k_start
            mask_k = offs_k < N

            # ----- load left tile: shape (M, K, H) -----
            left_ptrs = (
                left_ptr
                + batch * stride_lb
                + offs_i[:, None, None] * stride_li
                + offs_k[None, :, None] * stride_lk
                + offs_h[None, None, :] * stride_lh
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )
            left = left.to(tl.float32)

            # ----- load right tile transposed in the k dimension:
            #        we need right[j, k, h] -> (K, N, H) layout -----
            right_ptrs = (
                right_ptr
                + batch * stride_rb
                + offs_k[:, None, None] * stride_rk
                + offs_j[None, :, None] * stride_rj
                + offs_h[None, None, :] * stride_rh
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )
            right = right.to(tl.float32)

            # ----- accumulate per hidden slice with a dot product -----
            for hh in range(0, BLOCK_H):
                # Each thread holds a scalar of the output tile (i,j,hh)
                # tl.dot uses the thread’s i‑lane from `left` and j‑lane from `right`.
                acc[:, :, hh] += tl.dot(left[:, :, hh], right[:, :, hh])

        # ----- write out the completed hidden chunk -----
        out_ptrs = (
            out_ptr
            + batch * stride_ob
            + offs_i[:, None, None] * stride_oi
            + offs_j[None, :, None] * stride_oj
            + offs_h[None, None, :] * stride_oh
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


def custom_kernel(data):
    """
    Entry point matching the required signature:
        data = (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [B, N, N, dim] – the forward pass of the
        outgoing TriMul operator.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # ------------------- configuration -------------------
    dim = config["dim"]            # input / output feature dimension
    hidden_dim = config["hidden_dim"]  # hidden dimension used inside the module
    B, N, _, D = input_tensor.shape
    assert D == dim, f"input last dim ({D}) != config dim ({dim})"

    # ------------------- LayerNorm -------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------- Linear projections (no bias) -------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------- Gating (sigmoid) -------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------- Optional mask -------------------
    if mask is not None and mask.numel() > 0:
        # mask shape [B,N,N]; broadcast on the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------- Apply gates -------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------- Cast to fp16 for the Triton kernel -------------------
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # ------------------- Tri‑multiplicative update (core Triton kernel) -------------------
    out_hidden = _tri_mul_triton(left_f16, right_f16)      # [B,N,N,hidden_dim] float32

    # ------------------- Output LayerNorm + gating -------------------
    out_hidden = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_hidden = out_hidden * out_gate                      # element‑wise gate

    # ------------------- Final linear back to dim -------------------
    out = F.linear(out_hidden, weights["to_out.weight"])

    return out
