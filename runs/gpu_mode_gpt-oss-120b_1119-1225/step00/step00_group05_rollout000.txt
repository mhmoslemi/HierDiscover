@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bh = tl.program_id(0) # index over batch*head dimension
    pid_i = tl.program_id(1) # block row
    pid_j = tl.program_id(2) # block col

    # Derive batch and hidden indices
    b = pid_bh // H # integer division
    h = pid_bh % H

    # Starting indices for i and j
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Create mask for out-of-bounds i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension (seq_len)
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k_start and offsets
        k_start = k * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)

        # mask for k
        mask_k = k_offset < N

        # Compute pointers for left and right loads
        # left: shape (i,k)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_offset[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k + \
            h * stride_left_h
        # right: shape (j,k)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            j_offset[:, None] * stride_right_j + \
            k_offset[None, :] * stride_right_k + \
            h * stride_right_h

        # Load
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right_tile = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        
        # Compute dot and accumulate
        acc += tl.dot(left_tile, right_tile)  # result shape (BLOCK_M, BLOCK_N)
    
    # Write to out
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j + \
        h * stride_out_h

    # Store with mask
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Input LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj_weight = weights["left_proj.weight"] # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present: we have mask shape [B,N,N]; may be None? According to config "nomask" maybe true; but mask is provided. We'll just use mask. mask shape [B,N,N]; unsqueeze(-1)
    if mask is not None:
        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask
    # else no mask

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare output tensor
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left_fp16.stride()
    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right_fp16.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32  # typical

    grid = (B * hidden_dim, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))

    trimul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, hidden_dim,
        left_stride_b, left_stride_i, left_stride_k, left_stride_h,
        right_stride_b, right_stride_j, right_stride_k, right_stride_h,
        out_stride_b, out_stride_i, out_stride_j, out_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Out norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate and final linear
    out_gate = out_gate  # shape [B,N,N,hidden_dim]
    out_norm = out_norm * out_gate
    to_out_weight = weights["to_out.weight"]   # shape [dim, hidden_dim]
    output = torch.nn.functional.linear(out_norm, to_out_weight)

    return output

"""
custom_kernel: Triton‑accelerated implementation of the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,D]) using provided
   weight/bias (PyTorch – cheap compared to the cubic work).

2. Linear projections (no bias) → left, right  (shape [B,N,N,H]).
   Apply optional mask (broadcasted on the last dim).

3. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized input and multiply them into left/right and later into the
   output.

4. Core Cubic Operation:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply for every (b,h) pair:
   left_slice  = left[b,:,:,h]  (N×N)   – rows=i, cols=k
   right_slice = right[b,:,:,h] (N×N)   – rows=j, cols=k
   out_slice   = left_slice @ right_sliceᵀ  (i‑j matrix)
   Implemented with a Triton kernel that tiles the i‑ and j‑axes
   (BLOCK_M×BLOCK_N) and reduces over k in blocks (BLOCK_K).  The kernel
   processes all B·H slices in parallel (program_id(0) = b*H + h).

5. Layer‑norm the result across the hidden dimension (H) and apply the
   out_gate.

6. Final linear projection back to the original dimension D.

The kernel uses fp16 arithmetic for the heavy multiply‑accumulate while
accumulating in fp32 to preserve numerical quality.  The final output
tensor is float32.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr,          # const __fp16 *
    right_ptr,         # const __fp16 *
    out_ptr,           # float *
    B, N, H,           # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using tiled matrix multiplication with reduction over k.
    """
    # --------------------------------------------------------------
    # 1) Identify the slice (batch, hidden) we are computing.
    # --------------------------------------------------------------
    pid_bh = tl.program_id(0)     # 0 … B*H-1
    pid_i  = tl.program_id(1)     # block row index
    pid_j  = tl.program_id(2)     # block column index

    b = pid_bh // H               # batch index
    h = pid_bh % H                # hidden‑dim index

    # --------------------------------------------------------------
    # 2) Tile offsets for the i‑ and j‑axes.
    # --------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < N
    mask_j = j_offset < N

    # --------------------------------------------------------------
    # 3) Accumulator (fp32).
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 4) Loop over k in BLOCK_K sized tiles.
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for tk in range(0, num_k_tiles):
        k_offset = tk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Pointers for the current k‑tile.
        # left: shape (M, K)  indexed by (i, k)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            i_offset[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k + \
            h * stride_left_h

        # right: shape (N, K) indexed by (j, k)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            j_offset[:, None] * stride_right_j + \
            k_offset[None, :] * stride_right_k + \
            h * stride_right_h

        # Load tiles (fp16) with zero padding outside the valid range.
        left_tile  = tl.load(left_ptrs,
                             mask=mask_i[:, None] & mask_k[None, :],
                             other=0.0)
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0)

        # Convert to fp32 for accumulation.
        left_tile  = left_tile.to(tl.float32)
        right_tile = right_tile.to(tl.float32)

        # Dot‑product over k → (M, N) contribution.
        acc += tl.dot(left_tile, right_tile)

    # --------------------------------------------------------------
    # 5) Write the accumulated tile to global memory.
    # --------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j + \
        h * stride_out_h

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator with a Triton‑accelerated
    cubic update.  The function receives a tuple (input, mask, weights,
    config) and returns the processed tensor of shape [B, N, N, D].

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output with shape (batch, seq_len, seq_len, dim) and dtype float32.
    """
    # Unpack arguments.
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm:   x = LayerNorm(x)  (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → left, right.
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"] # (hidden_dim, dim)

    # torch.nn.functional.linear does: out = input @ weight.T
    left = F.linear(x, left_proj_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dimension).
    # ------------------------------------------------------------------
    # mask shape: (B, N, N)  ; may be omitted when config["nomask"] is True.
    if not config.get("nomask", False) and mask is not None:
        # Convert mask to same dtype as left/right and add a trailing dim.
        mask = mask.to(dtype).unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating signals (sigmoid of linear projections of x).
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare data for the Triton kernel (convert to fp16 for speed).
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, H = left_fp16.shape   # note: H == hidden_dim

    # Output buffer (fp32) – will hold the result of the cubic update.
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes) for 4‑D tensors.
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # ------------------------------------------------------------------
    # 6) Triton launch configuration.
    # ------------------------------------------------------------------
    BLOCK_M = 128   # rows per block
    BLOCK_N = 128   # cols per block
    BLOCK_K = 32    # reduction tile size

    grid = (
        B * H,                                 # program_id(0) : batch * hidden
        math.ceil(N / BLOCK_M),                # program_id(1) : i‑tile
        math.ceil(N / BLOCK_N)                 # program_id(2) : j‑tile
    )

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Post‑processing: layer‑norm over hidden dim, out‑gate, final linear.
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # Apply the output gate (broadcasted on the hidden dim).
    out = out * out_gate

    # Final linear projection back to the original dimension.
    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    output = F.linear(out, to_out_w)      # (B,N,N,dim)

    return output
