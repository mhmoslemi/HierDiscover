import torch
import triton
import triton.language as tl

@triton.jit
def outgoing_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Grid axes
    pid_m = tl.program_id(0)  # i-block
    pid_n = tl.program_id(1)  # j-block
    pid_z = tl.program_id(2)  # combined batch and hidden

    # decode batch and hidden index
    batch_idx = pid_z // H
    hid_idx = pid_z % H

    # Offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_m_mask = offs_m < N
    offs_n_mask = offs_n < N

    # Compute base pointers for this batch+hidden slice
    left_base = left_ptr + batch_idx * stride_left_b + hid_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_b + hid_idx * stride_right_h
    out_base = out_ptr + batch_idx * stride_out_b + hid_idx * stride_out_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        k_mask = offs_k < N

        # Load a tile of left: (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs, mask=offs_m_mask[:, None] & k_mask[None, :], other=0.0)

        # Load a tile of right but transposed: (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_i)
        b = tl.load(b_ptrs, mask=k_mask[:, None] & offs_n_mask[None, :], other=0.0)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # Store the result tile (apply bounds mask)
    c_ptrs = out_base + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, acc, mask=offs_m_mask[:, None] & offs_n_mask[None, :])

def trimul_einsum_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute ``out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]``
    using a Triton kernel. ``left`` and ``right`` must have shape
    [B, N, N, H] and be contiguous.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in number of elements
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Tile sizes – chosen to fit comfortably into registers on H100.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )

    outgoing_trimul_kernel[grid](
        left,
        right,
        out,
        s_left_b,
        s_left_i,
        s_left_k,
        s_left_h,
        s_right_b,
        s_right_i,
        s_right_k,
        s_right_h,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out

def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    This implementation mirrors the reference PyTorch version but off‑loads the
    cubic ``einsum`` to a custom Triton kernel for speed.
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # LayerNorm over the last dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional mask – treat missing mask as all‑ones
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gating (sigmoid)
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core cubic update – Triton kernel
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = trimul_einsum_outgoing(left, right)          # shape [B,N,N,hidden_dim]

    # Output layernorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply the output gate
    out = out * out_gate

    # Final projection back to ``dim``
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
custom_kernel – Outgoing TriMul forward pass.
The heavy cubic einsum (shape [B,N,N,H] → [B,N,N,H]) is off‑loaded to a
custom Triton kernel that computes
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
for every batch b and hidden channel d.  All other operations (LayerNorm,
linear projections, gating, mask handling, final projection) are performed
with native PyTorch calls.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def outgoing_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute the outgoing TriMul update using a GEMM‑style Triton kernel.

    left, right, out have shape [B, N, N, H] and are stored in row‑major
    order (contiguous).  The kernel loops over the reduction axis `k` and
    accumulates the product left[i,k,d] * right[j,k,d] into out[i,j,d].
    """
    pid_m = tl.program_id(0)          # tile index for i
    pid_n = tl.program_id(1)          # tile index for j
    pid_z = tl.program_id(2)          # combined batch * hidden index

    # decode batch and hidden‑channel from pid_z
    batch_idx = pid_z // H
    hid_idx   = pid_z %  H

    # row / column offsets for this tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    right_base = right_ptr + batch_idx * stride_right_b + hid_idx * stride_right_h
    out_base   = out_ptr   + batch_idx * stride_out_b   + hid_idx * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension `k`
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # right tile (transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_i)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # accumulate dot‑product along k
        acc += tl.dot(a, b)

    # write the result tile
    c_ptrs = out_base + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def trimul_einsum_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches ``outgoing_trimul_kernel``.
    ``left`` and ``right`` must be contiguous tensors of shape [B, N, N, H].
    Returns a tensor of the same shape containing the summed‑over‑k product.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in element units (not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # Tile sizes – chosen to fit comfortably on an H100.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        B * H,                     # batch × hidden
    )

    outgoing_trimul_kernel[grid](
        left,
        right,
        out,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        N,
        H,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Outgoing TriMul forward pass.
    Args:
        data: tuple (input_tensor, mask, weights, config)
            input_tensor – [B, N, N, dim] float32
            mask         – [B, N, N] (bool or 0/1) or None
            weights      – dict with the model parameters
            config       – dict containing at least 'dim' and 'hidden_dim'
    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm over `dim`
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask – broadcast over the hidden dimension
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)       # [B,N,N,1]
        left  = left  * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) – element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core cubic update (the O(N³) part) – Triton kernel
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    out = trimul_einsum_outgoing(left, right)   # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 6) Output layer‑norm, output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
