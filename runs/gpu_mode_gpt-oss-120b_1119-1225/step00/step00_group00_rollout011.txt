def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Cast to half
    x = input_tensor.to(torch.float16)

    # LayerNorm
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # Ensure mask
    if mask is None:
        mask = torch.ones(bs, N, N, dtype=torch.float16, device=device)
    else:
        mask = mask.to(torch.float16)

    # Apply mask and gates via Triton kernel
    # We'll not allocate new buffers, but keep left_proj and right_proj as the output.
    # So we need to pass pointers.
    total_elements = bs * N * N * hidden_dim
    BLOCK_SIZE = 1024
    # define kernel
    import triton
    import triton.language as tl

    @triton.jit
    def gate_mask_kernel(
        left_ptr, left_gate_ptr,
        right_ptr, right_gate_ptr,
        mask_ptr,
        B, N, H, total_elems,
        stride_l_ptr_b, stride_l_ptr_i, stride_l_ptr_j, stride_l_ptr_h,
        stride_lg_ptr_b, stride_lg_ptr_i, stride_lg_ptr_j, stride_lg_ptr_h,
        stride_r_ptr_b, stride_r_ptr_i, stride_r_ptr_j, stride_r_ptr_h,
        stride_rg_ptr_b, stride_rg_ptr_i, stride_rg_ptr_j, stride_rg_ptr_h,
        stride_mask_b, stride_mask_i, stride_mask_j,
        BLOCK_SIZE: tl.constexpr
    ):
        pid = tl.program_id(0)
        block_start = pid * BLOCK_SIZE
        offs = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offs < total_elems
        if not tl.any(mask):
            return

        # convert linear index to (b,i,j,h)
        idx = offs
        h = idx % H
        idx = idx // H
        j = idx % N
        idx = idx // N
        i = idx % N
        b = idx // N

        # compute offsets for each pointer
        left_offset = b*stride_l_ptr_b + i*stride_l_ptr_i + j*stride_l_ptr_j + h*stride_l_ptr_h
        left_gate_offset = b*stride_lg_ptr_b + i*stride_lg_ptr_i + j*stride_lg_ptr_j + h*stride_lg_ptr_h
        right_offset = b*stride_r_ptr_b + i*stride_r_ptr_i + j*stride_r_ptr_j + h*stride_r_ptr_h
        right_gate_offset = b*stride_rg_ptr_b + i*stride_rg_ptr_i + j*stride_rg_ptr_j + h*stride_rg_ptr_h

        mask_offset = b*stride_mask_b + i*stride_mask_i + j*stride_mask_j

        left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
        left_gate_val = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=0.0)
        right_val = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
        right_gate_val = tl.load(right_gate_ptr + right_gate_offset, mask=mask, other=0.0)
        m = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

        # compute
        left_out = left_val * m * left_gate_val
        right_out = right_val * m * right_gate_val

        # store
        tl.store(left_ptr + left_offset, left_out, mask=mask)
        tl.store(right_ptr + right_offset, right_out, mask=mask)

    # launch kernel
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    gate_mask_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask,
        bs, N, hidden_dim, total_elements,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE
    )

    # After gating, left_proj and right_proj hold masked and gated values.

    # Compute tri-mul via batched GEMM
    left_t = left_proj.permute(0,3,1,2)  # (B, H, N, N)
    right_t = right_proj.permute(0,3,1,2)
    out = torch.matmul(left_t, right_t.transpose(-1, -2))  # (B, H, N, N)
    out = out.permute(0,2,3,1)  # (B, N, N, H)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias = weights["to_out_norm.bias"].to(torch.float16)
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # multiply by out_gate
    out_norm = out_norm * out_gate

    # final linear
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"].to(torch.float16))

    # cast back to float32
    return final.to(torch.float32)

"""
TriMul "outgoing" kernel (AlphaFold3) – fast forward pass.

Algorithm
---------

1.  Layer‑norm the input tensor `x` over the last dimension using the supplied
    `norm.weight` / `norm.bias`.
2.  Four linear projections (no bias) are computed from `x`:
        * left   = x @ left_proj.weight
        * right  = x @ right_proj.weight
        * left_gate  = sigmoid(x @ left_gate.weight)
        * right_gate = sigmoid(x @ right_gate.weight)
        * out_gate   = sigmoid(x @ out_gate.weight)
3.  A fused Triton kernel applies the pairwise mask (if any) *and* the
    per‑element gates:
        left  = left  * mask * left_gate
        right = right * mask * right_gate
4.  The core “TriMul” operation is a batched matrix multiplication per hidden
    channel:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    This is realised as a batch of `hidden_dim` independent GEMMs:
        left   : (B, H, N, N)
        right  : (B, H, N, N) → transpose last two axes
        out    = torch.matmul(left, right.transpose(-1, -2))
5.  Apply a second LayerNorm (`to_out_norm`) on the hidden dimension,
    multiply by `out_gate` and finally project back to the original `dim`
    using `to_out.weight`.
6.  The kernel works entirely in FP16 for speed and casts the final result
    back to FP32.

The Triton kernel only performs the element‑wise mask & gate fusion,
which is the part that Torch‑compile cannot fuse efficiently.
"""

import torch
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple of
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:        torch.Tensor of shape [B, N, N] (float32 or bool) or None
            - weights:     dict of model weights (all tensors are on CUDA)
            - config:      dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # 1. unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------ #
    # 2. FP16 conversion (half‑precision gives ~3× speed on H100)
    # ------------------------------------------------------------------ #
    x = input_tensor.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 3. First LayerNorm (over `dim`)
    # ------------------------------------------------------------------ #
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias
    )  # (B,N,N,dim), half

    # ------------------------------------------------------------------ #
    # 4. Linear projections & gates (no bias)
    # ------------------------------------------------------------------ #
    left_proj  = torch.nn.functional.linear(x, weights["left_proj.weight"].to(torch.float16))
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"].to(torch.float16))

    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"].to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"].to(torch.float16)).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"].to(torch.float16)).sigmoid()

    # ------------------------------------------------------------------ #
    # 5. Prepare mask – if no mask is supplied we use a tensor of ones
    # ------------------------------------------------------------------ #
    if mask is None:
        mask = torch.ones(bs, N, N, dtype=torch.float16, device=device)
    else:
        mask = mask.to(torch.float16)

    # ------------------------------------------------------------------ #
    # 6. Triton kernel: fused mask * gate application
    # ------------------------------------------------------------------ #
    BLOCK_SIZE = 1024  # tunable; 1024 works well on H100

    @triton.jit
    def _gate_mask_fuse_kernel(
        left_ptr,           # *float16  – left projection (in‑out)
        left_gate_ptr,      # *float16  – left gate
        right_ptr,          # *float16  – right projection (in‑out)
        right_gate_ptr,     # *float16  – right gate
        mask_ptr,           # *float16  – mask (B,N,N)
        B, N, H, TOTAL_ELEM,
        stride_l_b, stride_l_i, stride_l_j, stride_l_h,
        stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
        stride_r_b, stride_r_i, stride_r_j, stride_r_h,
        stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
        stride_m_b, stride_m_i, stride_m_j,
        BLOCK_SIZE: tl.constexpr
    ):
        pid = tl.program_id(0)
        block_start = pid * BLOCK_SIZE
        offs = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offs < TOTAL_ELEM
        if not tl.any(mask):
            return

        # ---- linear index → (b,i,j,h) ----
        idx = offs
        h = idx % H
        idx = idx // H
        j = idx % N
        idx = idx // N
        i = idx % N
        b = idx // N

        # ---- compute memory offsets ----
        left_off   = b*stride_l_b   + i*stride_l_i   + j*stride_l_j   + h*stride_l_h
        leftg_off  = b*stride_lg_b  + i*stride_lg_i  + j*stride_lg_j  + h*stride_lg_h
        right_off  = b*stride_r_b   + i*stride_r_i   + j*stride_r_j   + h*stride_r_h
        rightg_off = b*stride_rg_b  + i*stride_rg_i  + j*stride_rg_j  + h*stride_rg_h
        mask_off   = b*stride_m_b   + i*stride_m_i   + j*stride_m_j

        # ---- load values (masked) ----
        l_val   = tl.load(left_ptr   + left_off,   mask=mask, other=0.0)
        lg_val  = tl.load(left_gate_ptr + leftg_off, mask=mask, other=0.0)
        r_val   = tl.load(right_ptr  + right_off,  mask=mask, other=0.0)
        rg_val  = tl.load(right_gate_ptr + rightg_off, mask=mask, other=0.0)
        m_val   = tl.load(mask_ptr   + mask_off,   mask=mask, other=0.0)

        # ---- apply mask and gates ----
        l_out = l_val * m_val * lg_val
        r_out = r_val * m_val * rg_val

        # ---- write back (in‑place) ----
        tl.store(left_ptr  + left_off,   l_out, mask=mask)
        tl.store(right_ptr + right_off,  r_out, mask=mask)

    total_elements = bs * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # launch
    _gate_mask_fuse_kernel[grid](
        left_proj, left_gate,
        right_proj, right_gate,
        mask,
        bs, N, hidden_dim, total_elements,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE
    )

    # ------------------------------------------------------------------ #
    # 7. Core TriMul: batch of GEMMs (left @ right^T) per hidden channel
    # ------------------------------------------------------------------ #
    # Permute to (B, H, N, N) – a contiguous layout is slightly faster
    left_t  = left_proj.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right_proj.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T → (B*H, N, N)
    out_hidden = torch.matmul(left_t, right_t.transpose(-1, -2))   # (B, H, N, N)

    # Back to (B, N, N, H)
    out = out_hidden.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 8. Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------ #
    to_out_norm_weight = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(torch.float16)
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )  # (B,N,N,H)

    # ------------------------------------------------------------------ #
    # 9. Apply out_gate and final projection back to `dim`
    # ------------------------------------------------------------------ #
    out = out * out_gate
    final = torch.nn.functional.linear(out, weights["to_out.weight"].to(torch.float16))

    # ------------------------------------------------------------------ #
    # 10. Cast back to FP32 (the public API expects float32)
    # ------------------------------------------------------------------ #
    return final.to(torch.float32)
