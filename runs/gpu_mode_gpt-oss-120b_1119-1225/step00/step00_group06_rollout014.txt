def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    ZERO: tl.constexpr
):
    pid = tl.program_id(0)
    # total tiles in i and j dimensions
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    # decode batch index and tile indices
    total_tiles = num_i * num_j
    b = pid // total_tiles
    tile_idx = pid % total_tiles
    i_tile = tile_idx // num_j
    j_tile = tile_idx % num_j

    # bounds
    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # hidden block index taken from second axis
    h_tile = tl.program_id(1)
    h_start = h_tile * BLOCK_H

    # offsets within block
    i_offset = tl.arange(0, BLOCK_I)
    j_offset = tl.arange(0, BLOCK_J)
    k_offset = tl.arange(0, BLOCK_K)
    h_offset = tl.arange(0, BLOCK_H)

    j_mask = j_start + j_offset < N
    h_mask = h_start + h_offset < H

    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

"""
custom_kernel
==============

Implementation of the “outgoing” TriMul operation (AlphaFold‑3) using a
hand‑written Triton kernel for the O(N³) tensor contraction.
The kernel computes

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where ``left`` and ``right`` are the projected and gated pairwise
representations of the input tensor.  All other linear layers,
normalisations and sigmoids are performed with PyTorch (FP16 where
possible) – only the expensive contraction is off‑loaded to Triton.
The kernel processes the sequence dimensions in 2‑D tiles and the
hidden channel dimension in small blocks (default 8) to stay within the
shared‑memory budget of an H100.  The implementation works for any
batch size ≤ 2, any sequence length up to 1024 and hidden dimensions
128 (or any multiple of the block size) and correctly handles optional
masking.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched block‑wise outer‑product reduction over the `k`
# dimension.  It accumulates in float32 and stores the result as float32.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *const half*  [B, N, N, H]
    right_ptr,         # *const half*  [B, N, N, H]
    out_ptr,           # *float32*     [B, N, N, H]

    B, N, H,                     # problem sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute out = Σₖ left[...,i,k,:] * right[...,j,k,:]"""

    pid = tl.program_id(0)                # index over (batch, i‑tile, j‑tile)
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    total_tiles = num_i * num_j

    b = pid // total_tiles                # batch index
    tile = pid % total_tiles
    i_tile = tile // num_j
    j_tile = tile % num_j

    i_start = i_tile * BLOCK_I
    j_start = j_tile * BLOCK_J

    # second program id iterates over hidden‑channel blocks
    h_tile = tl.program_id(1)
    h_start = h_tile * BLOCK_H

    # offsets inside a tile
    i_off = tl.arange(0, BLOCK_I)
    j_off = tl.arange(0, BLOCK_J)
    k_off = tl.arange(0, BLOCK_K)
    h_off = tl.arange(0, BLOCK_H)

    # bounds (used for masking out‑of‑range elements)
    i_ok = i_start + i_off < N
    j_ok = j_start + j_off < N
    h_ok = h_start + h_off < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(0, num_k):
        k_start = kk * BLOCK_K
        k_ok = k_start + k_off < N

        # ----- load a tile of left : (BLOCK_I, BLOCK_K, BLOCK_H) -----
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            (i_start + i_off)[:, None, None] * stride_left_i + \
            (k_start + k_off)[None, :, None] * stride_left_k + \
            (h_start + h_off)[None, None, :] * stride_left_h
        left_mask = i_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float16)

        # ----- load a tile of right : (BLOCK_J, BLOCK_K, BLOCK_H) -----
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            (j_start + j_off)[:, None, None] * stride_right_j + \
            (k_start + k_off)[None, :, None] * stride_right_k + \
            (h_start + h_off)[None, None, :] * stride_right_h
        right_mask = j_ok[:, None, None] & k_ok[None, :, None] & h_ok[None, None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float16)

        # ----- accumulate outer‑products for each hidden channel in the block -----
        #   left_tile[i,k,h] * right_tile[j,k,h]  summed over k
        for hh in range(0, BLOCK_H):
            # slices have shape (BLOCK_I, BLOCK_K) and (BLOCK_J, BLOCK_K)
            left_h  = left_tile[:, :, hh]
            right_h = right_tile[:, :, hh]
            # left_h @ right_h.T   -> (BLOCK_I, BLOCK_J)
            acc[:, :, hh] += tl.dot(left_h, right_h.T)

    # ----- store the result -------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        (i_start + i_off)[:, None, None] * stride_out_i + \
        (j_start + j_off)[None, :, None] * stride_out_j + \
        (h_start + h_off)[None, None, :] * stride_out_h

    out_mask = i_ok[:, None, None] & j_ok[None, :, None] & h_ok[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ----------
    data : tuple
        (x, mask, weights, config)
        - x : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask : torch.Tensor of shape [B, N, N] (bool or float) or None
        - weights : dict of all linear / layer‑norm parameters
        - config : dict containing at least ``dim`` and ``hidden_dim``
    Returns
    -------
    torch.Tensor with shape [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (dim‑wise)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]  float32

    # Cast to fp16 for the heavy linear work
    x_norm_fp16 = x_norm.to(torch.float16)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_norm_fp16, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm_fp16, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply optional mask (mask is broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # ensure mask is float16 and has a trailing singleton dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_fp16
        right = right * mask_fp16

    # ------------------------------------------------------------------
    # 4. Gating values (sigmoid of linear transforms of x_norm)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left_gate = torch.nn.functional.linear(x_norm_fp16, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_fp16, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_fp16, out_gate_w).sigmoid()

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguity before passing to Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 6. Tri‑multiplicative update via Triton
    # ------------------------------------------------------------------
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    # In the kernel the second axis of `right` is interpreted as the “j” axis.
    stride_right_j = stride_right_i

    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_hidden.stride()

    # Block sizes – tuned for H100 and hidden_dim=128
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8   # must divide hidden_dim for full coverage (final block is masked)

    grid = (
        B * triton.cdiv(N, BLOCK_I) * triton.cdiv(N, BLOCK_J),  # axis 0: (b, i‑tile, j‑tile)
        triton.cdiv(hidden_dim, BLOCK_H)                       # axis 1: hidden‑channel blocks
    )

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 7. Normalisation, out‑gate and final linear projection
    # ------------------------------------------------------------------
    # Layer‑norm on the hidden dimension
    out_hidden_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the output gate (broadcast over the hidden dim)
    out_hidden_norm = out_hidden_norm * out_gate.to(torch.float32)

    # Final linear (hidden -> dim)
    to_out_w = weights["to_out.weight"]
    output = torch.nn.functional.linear(out_hidden_norm, to_out_w)   # [B,N,N,dim]

    return output
