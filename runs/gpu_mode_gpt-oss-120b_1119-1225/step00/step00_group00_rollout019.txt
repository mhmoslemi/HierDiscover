def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    device = input_tensor.device

    # LayerNorm on input: (norm.weight: dim), norm.bias: dim
    eps=1e-5
    # compute mean and variance over last dim
    # Using torch.nn.functional.layer_norm with normalized_shape=dim
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])
    # left_gate, right_gate, out_gate
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['out_gate.weight']))

    # optional mask
    if mask is not None:
        mask = mask.to(input_tensor.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask
    else:
        # mask all ones assumption
        pass

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Compute trimul via Triton kernel
    B, N, N2, _ = left.shape
    assert N == N2

    # Output shape B,N,N,hidden_dim
    out = torch.empty_like(left)  # same shape and dtype

    # get strides (in elements)
    stride_left_b = left.stride(0)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_left_d = left.stride(3)

    stride_right_b = right.stride(0)
    stride_right_i = right.stride(1)  # this is j dimension
    stride_right_k = right.stride(2)
    stride_right_d = right.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_d = out.stride(3)

    # Set BLOCK sizes (could tune)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid size
    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * hidden_dim * num_blocks_i * num_blocks_j

    # launch kernel
    trimul_kernel[(total_programs,)](
        left, right, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # num warps per kernel
        # maybe enable any config
    )

    # After kernel: out shape [B,N,N,hidden_dim]
    # Apply output norm & gate
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),

@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  batch, seqlen, hidden_dim,
                  stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
                  stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
                  stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)

    # compute number of blocks
    num_blocks_i = tl.cdiv(seqlen, BLOCK_M)
    num_blocks_j = tl.cdiv(seqlen, BLOCK_N)

    # total programs per batch = hidden_dim * num_blocks_i * num_blocks_j
    progs_per_batch = hidden_dim * num_blocks_i * num_blocks_j

    batch_id = pid // progs_per_batch
    prog_rem = pid % progs_per_batch
    d_id = prog_rem // (num_blocks_i * num_blocks_j)
    prog_rem2 = prog_rem % (num_blocks_i * num_blocks_j)
    block_i = prog_rem2 // num_blocks_j
    block_j = prog_rem2 % num_blocks_j

    # offsets for i and j
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_i = i_offset < seqlen
    mask_j = j_offset < seqlen

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    num_blocks_k = tl.cdiv(seqlen, BLOCK_K)
    for kb in range(0, num_blocks_k):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < seqlen

        # pointers for A (left)
        a_ptrs = left_ptr + \
            batch_id * stride_left_batch + \
            d_id * stride_left_d + \
            i_offset[:, None] * stride_left_i + \
            k_offset[None, :] * stride_left_k

        # pointers for B (right) transposed: shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_ptr + \
            batch_id * stride_right_batch + \
            d_id * stride_right_d + \
            k_offset[:, None] * stride_right_k + \
            j_offset[None, :] * stride_right_i

        acc += tl.dot(a, b)

        batch_id * stride_out_batch + \
        d_id * stride_out_d + \
        i_offset[:, None] * stride_out_i + \
        j_offset[None, :] * stride_out_j

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy part of the TriMul operator is the triple‑wise product
```
out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
```
which can be written as a batched matrix multiplication:
    out[b,:, :, d] = left[b,:, :, d] @ right[b,:, :, d].T
The kernel below computes this operation for every batch `b` and
hidden channel `d` using a blocked GEMM (BLOCK_M×BLOCK_N tiles and
BLOCK_K reduction).  All other steps (LayerNorm, linear projections,
gates, mask handling) are performed in PyTorch for simplicity.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# -----------------------------------------------------------------------
# Triton kernel: batched matmul with transpose on the right operand.
# Computes `out = left @ right^T` for each (batch, hidden_dim) slice.
# -----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const *float32
    right_ptr,         # const *float32
    out_ptr,           #       *float32
    batch, seqlen, hidden_dim,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_d,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_d,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)

    # ---------------------------------------------------------------
    # Decode program id into (batch, hidden, block_i, block_j)
    # ---------------------------------------------------------------
    num_blocks_i = tl.cdiv(seqlen, BLOCK_M)
    num_blocks_j = tl.cdiv(seqlen, BLOCK_N)

    progs_per_batch = hidden_dim * num_blocks_i * num_blocks_j

    batch_id = pid // progs_per_batch
    prog_rem = pid % progs_per_batch
    d_id = prog_rem // (num_blocks_i * num_blocks_j)
    prog_rem2 = prog_rem % (num_blocks_i * num_blocks_j)
    block_i = prog_rem2 // num_blocks_j
    block_j = prog_rem2 % num_blocks_j

    # ---------------------------------------------------------------
    # Offsets for the output tile (i, j)
    # ---------------------------------------------------------------
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_offset < seqlen
    mask_j = j_offset < seqlen

    # accumulator for a BLOCK_M x BLOCK_N tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension (reduction)
    # ---------------------------------------------------------------
    num_blocks_k = tl.cdiv(seqlen, BLOCK_K)
    for kb in range(0, num_blocks_k):
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < seqlen

        # ----- left tile (M x K) -------------------------------------------------
        a_ptrs = left_ptr \
            + batch_id * stride_left_batch \
            + d_id * stride_left_d \
            + i_offset[:, None] * stride_left_i \
            + k_offset[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ----- right tile (K x N)  (the right operand is transposed) ------------
        b_ptrs = right_ptr \
            + batch_id * stride_right_batch \
            + d_id * stride_right_d \
            + k_offset[:, None] * stride_right_k \
            + j_offset[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ----- accumulate ---------------------------------------------------------
        acc += tl.dot(a, b)   # (M,K) @ (K,N) -> (M,N)

    # ---------------------------------------------------------------
    # Write the result tile back to memory
    # ---------------------------------------------------------------
    c_ptrs = out_ptr \
        + batch_id * stride_out_batch \
        + d_id * stride_out_d \
        + i_offset[:, None] * stride_out_i \
        + j_offset[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# -----------------------------------------------------------------------
# Public entry point
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (or None)
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------------------
    # LayerNorm on the input (over last dimension)
    # -------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # -------------------------------------------------------------------
    # Linear projections (no bias)
    # -------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------
    # Optional mask (broadcasted on the hidden dimension)
    # -------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_ = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -------------------------------------------------------------------
    # Gating values (sigmoid)
    # -------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # Triton kernel: compute out = left @ right^T per hidden channel
    # -------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)  # [B, N, N, hidden_dim]

    # Strides (in element units)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Tunable block sizes (empirically good for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * hidden_dim * num_blocks_i * num_blocks_j

    # Launch Triton kernel
    trimul_kernel[(total_programs,)](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_d,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_d,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,           # good default for H100
    )

    # -------------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # -------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
