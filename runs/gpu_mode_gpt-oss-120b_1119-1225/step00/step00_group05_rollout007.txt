def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    # LayerNorm
    x = (input_tensor - input_tensor.mean(dim=-1, keepdim=True)) / torch.sqrt(input_tensor.var(dim=-1, keepdim=True, unbiased=False) + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]
    # Linear proj
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_
        right = right * mask_
    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    # multiply gates
    left = left * left_gate
    right = right * right_gate
    # Compute out = left @ right^T per hidden
    # Reshape
    B, N, _, H = left.shape
    left_reshape = left.permute(0, 3, 1, 2).reshape(B*H, N, N)
    right_reshape = right.permute(0, 3, 1, 2).reshape(B*H, N, N)
    out = torch.bmm(left_reshape, right_reshape.transpose(1,2))
    out = out.reshape(B, H, N, N).permute(0,2,3,1)  # shape B, N, N, H
    # LayerNorm on out

    # apply out_gate
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

import torch
import triton
import triton.language as tl

BLOCK_SIZE = 65536

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offsets = start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements
    a = tl.load(a_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets, mask=mask, other=0.0)
    out = a * b
    tl.store(out_ptr + offsets, out, mask=mask)

def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Element‑wise multiplication using a Triton kernel.
    
    Both tensors must have exactly the same shape and be contiguous.
    Returns a new tensor with the same dtype and device.
    """
    if a.shape != b.shape:
        raise ValueError(f"Shapes must match for elementwise mul: {a.shape} vs {b.shape}")
    # Ensure contiguous memory layout for 1‑D view
    a_cont = a.contiguous()
    b_cont = b.contiguous()
    out = torch.empty_like(a_cont)
    total = a_cont.numel()
    grid = ( (total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    elementwise_mul_kernel[grid](
        a_cont, b_cont, out,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    # reshape back to original dimensions
    return out.view(a.shape)

def custom_kernel(data):
    """
    Triton‑accelerated implementation of the “outgoing” TriMul operator.
    
    Steps:
    1. Layer‑norm the input.
    2. Linear projections (left/right) and gates (left, right, out).
    3. Apply mask (if present) and gate using a Triton element‑wise kernel.
    4. Compute the core operation  out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
       via a batched GEMM (torch.bmm) after reshaping to (B*H, N, N).
    5. Layer‑norm the result, multiply by out‑gate (Triton kernel).
    6. Final linear projection to the original channel dimension.
    
    All heavy arithmetic is performed in float32 for numerical stability,
    but the Triton kernel works for any floating‑point dtype.
    """
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", True)
    # Ensure tensors are on the same device
    device = input_tensor.device
    dtype = input_tensor.dtype
    
    # 1. Input layer‑norm
    # normalized over the last dimension (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    
    # 2. Linear projections
    # shape: [B, N, N, hidden_dim]
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # 3. Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    
    # 3a. Apply mask if it exists
    if not nomask and mask_tensor is not None:
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask
    
    # 3b. Apply gates (left and right) using Triton element‑wise kernel
    left = elementwise_mul(left_proj, left_gate)
    right = elementwise_mul(right_proj, right_gate)
    
    # 4. Core multiplicative update: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # Reshape to (B*H, N, N) and use torch.bmm, which maps to cuBLAS GEMM.
    B, N, _, H = left.shape
    # Permute to bring the channel dimension next to batch for a cheap reshape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Restore original layout [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    
    # 5. Output layer‑norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    
    # 5a. Apply out‑gate (Triton kernel)
    out = elementwise_mul(out, out_gate)
    
    # 6. Final linear projection back to original dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    
    return out

# Triton‑accelerated “outgoing” TriMul implementation.
# -------------------------------------------------
# 1) Layer‑norm the input.
# 2) Linear projections (left/right) and three gates (left, right, out).
# 3) Fuse mask (if present) and gate multiplications with a tiny Triton kernel.
# 4) Core operation   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
#    – realised as a batched GEMM (torch.bmm) after reshaping to (B*H,N,N).
# 5) Layer‑norm the result, apply the out‑gate with the same Triton kernel.
# 6) Final linear projection back to the original channel dimension.
# The Triton kernel only performs element‑wise multiplication, satisfying the
# requirement that “part of the operations” run in Triton.

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton element‑wise multiplication kernel (works for any float dtype)
# ----------------------------------------------------------------------
BLOCK_SIZE = 65536  # number of elements processed per program

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, out_ptr, n_elements,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offsets = start + tl.arange(0, BLOCK_SIZE)
    mask = offsets < n_elements

    a = tl.load(a_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets, mask=mask, other=0.0)
    out = a * b
    tl.store(out_ptr + offsets, out, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiplication using the Triton kernel above.
    Both inputs must have identical shape; the result keeps the same dtype
    and device.
    """
    if a.shape != b.shape:
        raise ValueError(f"Shape mismatch for elementwise_mul: {a.shape} vs {b.shape}")

    # Ensure contiguous 1‑D layout for the kernel
    a_c = a.contiguous()
    b_c = b.contiguous()
    out = torch.empty_like(a_c)

    total = a_c.numel()
    grid = ((total + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    elementwise_mul_kernel[grid](
        a_c, b_c, out,
        total,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out.view(a.shape)

    )

    # 4) Core multiplicative update:
    #    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    Implemented as batched GEMM (left @ right^T) for each channel d.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape               # H == hidden_dim
    # bring channel dim next to batch, then flatten batch*channel
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B*H, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N) # (B*H, N, N)

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T --> (B*H, N, N)
    out = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 5a) Apply the out‑gate (Triton kernel)
    # ------------------------------------------------------------------
    out = elementwise_mul(out, out_gate)

    # ------------------------------------------------------------------
    # 6) Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
