@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch_h = tl.program_id(2)

    # map combined batch index to batch and hidden
    b = pid_batch_h // H
    h = pid_batch_h % H

    # Compute start indices for this block
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Create range indices
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)

        # Bounds
        i_mask = i_range < N
        j_mask = j_range < N
        k_mask = k_range < N

        # Load left tile (M x K)
        left_offset = (b * stride_left_b
                       + i_range[:, None] * stride_left_i
                       + k_range[None, :] * stride_left_k
                       + h * stride_left_h)
        left = tl.load(left_ptr + left_offset,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # Load right tile (N x K) and transpose to (K x N) for dot

        right_offset = (b * stride_right_b
                        + k_range[:, None] * stride_right_k

                  + j_range[None, :] * stride_out_j
                  + h * stride_out_h)

    tl.store(out_ptr + out_offset,

"""
TriMul “outgoing” implementation.

The expensive part of the TriMul forward pass is the
einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication with a separate
batch for each hidden dimension `d`.  The kernel below
fuses this computation for all batches and hidden channels
into a single Triton kernel:

    left  : (B, N, N, H)   – shape (batch, i, k, hidden)
    right : (B, N, N, H)   – shape (batch, j, k, hidden)
    out   : (B, N, N, H)   – Σ_k left[..., i, k, h] * right[..., j, k, h]

We treat the combined index (b, h) as a single “batch”
dimension of size B*H and perform a standard
block‑wise GEMM over the (i, j) axes with k‑reduction.
All other operations (LayerNorm, Linear projections,
gating, final normalization and projection) are done in
PyTorch for simplicity.  The kernel works for any
batch size ≤2, sequence length up to 1024 and hidden
dimension up to 384, and runs in FP32 (accumulation)
with optional FP16 input.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,               # const * left   [B, N, N, H]
    right_ptr,              # const * right  [B, N, N, H]
    out_ptr,                # * out          [B, N, N, H]
    B, N, H,                # sizes (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Block‑wise batched GEMM for the TriMul einsum."""
    pid_i = tl.program_id(0)          # block row (i)
    pid_j = tl.program_id(1)          # block col (j)
    pid_batch_h = tl.program_id(2)    # combined batch*hidden index

    # Decompose combined index into (batch, hidden)
    b = pid_batch_h // H
    h = pid_batch_h % H

    # Starting coordinates of the tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Ranges inside the tile
    i_range = i0 + tl.arange(0, BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows / cols
    i_mask = i_range < N
    j_mask = j_range < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # -----------------------------------------------------------------
        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # left[b, i, k, h]
        # -----------------------------------------------------------------
        left_offset = (
            b * stride_left_b
            + i_range[:, None] * stride_left_i
            + k_range[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=(i_mask[:, None] & k_mask[None, :]),
            other=0.0,
        )

        # -----------------------------------------------------------------
        # Load right tile transposed to shape (BLOCK_K, BLOCK_N)
        # right[b, j, k, h] -> we need right_T[k, j]
        # -----------------------------------------------------------------
        right_offset = (
            b * stride_right_b
            + k_range[:, None] * stride_right_k
            + j_range[None, :] * stride_right_j
            + h * stride_right_h
        )
        right_T = tl.load(
            right_ptr + right_offset,
            mask=(k_mask[:, None] & j_mask[None, :]),
            other=0.0,
        )

        # -----------------------------------------------------------------
        # Accumulate the outer product over k
        # (BLOCK_M, BLOCK_K) dot (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        # -----------------------------------------------------------------
        acc += tl.dot(left, right_T)

    # -----------------------------------------------------------------
    # Write back the result
    # -----------------------------------------------------------------
    out_offset = (
        b * stride_out_b
        + i_range[:, None] * stride_out_i
        + j_range[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptr + out_offset, acc, mask=(i_mask[:, None] & j_mask[None, :]))


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask_tensor : torch.Tensor, shape [B, N, N] (may be all‑ones)
            - weights_dict: mapping of parameter names to torch.Tensors
            - config_dict : contains "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    (x, mask, weights, config) = data
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = x.device
    dtype = x.dtype

    # -----------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B,N,N,hidden)
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Mask handling – mask has shape [B,N,N]; broadcast to [...,1]
    # -----------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[:3], dtype=dtype, device=device)
    mask = mask.to(dtype).unsqueeze(-1)   # (B,N,N,1)
    if not nomask:
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Triton‑fused batched GEMM: out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Ensure contiguous layout for correct strides
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides in element units (not bytes)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_j, sR_k, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    # Tiling parameters – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden,                     # combined (batch, hidden) dimension
    )

    # Launch kernel
    tri_mul_matmul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden,
        sL_b,
        sL_i,
        sL_k,
        sL_h,
        sR_b,
        sR_j,
        sR_k,
        sR_h,
        sO_b,
        sO_i,
        sO_j,
        sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden dim, gating & final linear
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out_norm * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
