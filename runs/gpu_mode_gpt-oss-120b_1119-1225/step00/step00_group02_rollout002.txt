"""
custom_kernel: High‑performance TriMul “outgoing” update.

Algorithm
---------
1. Layer‑norm over the feature dimension (dim) – `x_norm`.
2. Linear projections:   left  = x_norm @ W_left  (no bias)  
                          right = x_norm @ W_right (no bias)
3. Optional mask (broadcast on the feature axis) is applied to left/right.
4. Gating:  left_gate  = sigmoid(x_norm @ W_left_gate)  → left  *= left_gate
           right_gate = sigmoid(x_norm @ W_right_gate) → right *= right_gate
           out_gate   = sigmoid(x_norm @ W_out_gate)   (used later)
5. Compute the “outgoing” pairwise product:
        out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
   This is a batched matrix‑multiply for every hidden channel `d`.
   We fuse all B·hidden_dim matmuls into a single Triton kernel
   (`batched_gemm_kernel`).  The kernel uses half‑precision storage,
   float32 accumulation and 2‑D tiling (BLOCK_M×BLOCK_N) with a
   K‑tiling loop.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.

The kernel covers the O(N³) core of TriMul, while the surrounding
operations are handled by PyTorch (they are cheap compared to the
pairwise multiplication).  All tensors remain on the GPU; the final
output is returned in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_gemm_kernel(
    a_ptr,            # *Pointer* to left matrices   (M×K)
    b_ptr,            # *Pointer* to right matrices  (K×N) – already transposed
    c_ptr,            # *Pointer* to output matrices  (M×N)
    batch_stride_a,   # stride to next matrix in A
    batch_stride_b,   # stride to next matrix in B
    batch_stride_c,   # stride to next matrix in C
    M, N, K,          # matrix dimensions (all equal to seq_len)
    stride_am, stride_ak,   # strides inside A
    stride_bk, stride_bn,   # strides inside B
    stride_cm, stride_cn,   # strides inside C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM kernel (A @ B) for many independent (M×K)*(K×N) products.

    The grid is laid out as (batch_id, block_i, block_j) flattened into a
    1‑D program id.  Each program computes a BLOCK_M×BLOCK_N tile of the
    output matrix for a single batch (i.e. a single (b, hidden) pair).
    """
    pid = tl.program_id(0)

    # number of blocks covering one matrix
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets within the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # base pointers for this batch
    a_batch = a_ptr + batch_idx * batch_stride_a
    b_batch = b_ptr + batch_idx * batch_stride_b
    c_batch = c_ptr + batch_idx * batch_stride_c

    # pointers to the start of the tile
    a_ptrs = a_batch + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    # accumulator (float32 for precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K tiles
    for _k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (tl.arange(0, BLOCK_K) < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) x (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        # advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # store result with proper mask for edge tiles
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    mask_store = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Args:
        data: tuple (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Cast to half for the heavy path (speed) – keep a flag to return fp32
    # ------------------------------------------------------------------
    use_fp16 = input_tensor.dtype != torch.float16
    if use_fp16:
        x = input_tensor.to(torch.float16)
    else:
        x = input_tensor

    # ------------------------------------------------------------------
    # LayerNorm over the feature dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(x.dtype)
    norm_bias   = weights["norm.bias"].to(x.dtype)
    x_norm = F.layer_norm(x, normalized_shape=(dim,), weight=norm_weight,
                          bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(x.dtype)   # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"].to(x.dtype)  # (hidden_dim, dim)

    left  = F.linear(x_norm, left_proj_w)   # (B, N, N, hidden_dim)
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # Optional mask (broadcast on the last feature axis)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: (B, N, N) → (B, N, N, 1)
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating projections + sigmoid
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(x.dtype)
    right_gate_w = weights["right_gate.weight"].to(x.dtype)
    out_gate_w   = weights["out_gate.weight"].to(x.dtype)

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Prepare inputs for the batched GEMM kernel
    #   left  : (B, N, N, hidden) → (B, hidden, N, N)  (i,k)
    #   right : (B, N, N, hidden) → (B, hidden, N, N) (j,k)
    #   we need right_T : (B, hidden, K=N, N) where K dimension matches
    #   the K‑dimension of left, i.e. right transposed on its last two axes.
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)  (i,k)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden, N, N)  (j,k)
    right_t    = right_perm.transpose(-2, -1).contiguous()  # (B, hidden, K=N, N)

    B, H, N, _ = left_perm.shape
    total_batches = B * H

    # Flatten the first two dims → (total_batches, N, N)
    left_flat  = left_perm.view(total_batches, N, N)
    right_flat = right_t.view(total_batches, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)   # same dtype (half)

    # ------------------------------------------------------------------
    # Triton kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute strides in elements (not bytes) – Triton expects element strides
    stride_am = left_flat.stride(1)   # row stride (M dimension)
    stride_ak = left_flat.stride(2)   # col stride (K dimension)
    stride_bk = right_flat.stride(1)  # K‑dimension stride in B
    stride_bn = right_flat.stride(2)  # N‑dimension stride in B
    stride_cm = out_flat.stride(1)    # row stride in C
    stride_cn = out_flat.stride(2)    # col stride in C
    batch_stride_a = left_flat.stride(0)
    batch_stride_b = right_flat.stride(0)
    batch_stride_c = out_flat.stride(0)

    # Determine grid size: one program per (batch, tile_m, tile_n)
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = total_batches * num_blocks_m * num_blocks_n

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # tuned for H100
        num_stages=3,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, out_gate and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(x.dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(x.dtype)

    out = F.layer_norm(out, normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    out = out * out_gate                     # broadcast on hidden dim
    to_out_w = weights["to_out.weight"].to(x.dtype)   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)           # (B, N, N, dim)

    # Cast back to float32 for the user (the reference implementation returns fp32)
    if use_fp16:
        out = out.to(torch.float32)

    return out
