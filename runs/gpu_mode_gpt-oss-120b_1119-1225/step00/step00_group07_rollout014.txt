    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Ensure x dtype float32
    x = x.to(torch.float32)

    # LayerNorm
    eps = 1e-5
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    inv = torch.rsqrt(var + eps)
    x_norm = (x - mean) * inv
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

        mask_f = mask.to(x.dtype).unsqueeze(-1)

        right = right * mask_f

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    right = right * right_gate

    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Reshape for batched GEMM
    B, N, _, _ = x.shape
    # Permute to (B, hidden, N, N)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*hidden, N, N)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) forward pass – highly‑optimized Triton implementation.

The operator works on a 4‑D pairwise tensor `x` of shape [B, N, N, C] and
computes

    left  = Linear(x) * mask * sigmoid(Linear(x))
    right = Linear(x) * mask * sigmoid(Linear(x))
    out   = Σₖ left[..., i, k, d] * right[..., j, k, d]          (einsum)
    out   = LayerNorm(out) * sigmoid(Linear(x))
    out   = Linear(out)                                          → [B,N,N,C]

The expensive Einstein sum (a batched matrix multiplication over the third
axis) is carried out by a custom Triton kernel that processes `B*hidden`
matrices of size N×N in FP16 while accumulating in FP32.  All other
operations (layer‑norm, linear layers, gating) are performed with PyTorch
but the heavy N³ work is fully fused in the kernel.

The kernel layout:
    * program_id(0) → tile of the output rows   (i‑axis)
    * program_id(1) → tile of the output cols   (j‑axis)
    * program_id(2) → batch index                (b * hidden)

Each tile computes a `BLOCK_M × BLOCK_N` sub‑matrix of the result by
streaming over K in blocks of size `BLOCK_K`.  The kernel uses
`tl.dot` (matrix‑multiply on registers) and stores the FP32 result
directly into the output tensor.

The public entry point `custom_kernel` follows the signature described
in the problem statement:
    data = (x, mask, weights, config)
and returns the processed tensor of shape [B, N, N, C].
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply (A @ Bᵀ) used for the einsum step
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = N)
    stride_am, stride_ak,              # A strides: (row, col)
    stride_bk, stride_bn,              # B strides (after transposition)
    stride_cm, stride_cn,              # C strides
    batch_stride_a, batch_stride_b, batch_stride_c,  # stride to next batch matrix
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)  # tile row
    pid_n = tl.program_id(1)  # tile col
    pid_b = tl.program_id(2)  # batch (b * hidden)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the selected batch element
    a_base = A_ptr + pid_b * batch_stride_a
    b_base = B_ptr + pid_b * batch_stride_b

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for k_tile in range(num_k_tiles):
        # K‑range for this tile
        k_offset = k_tile * BLOCK_K
        k_range = k_offset + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds in K
        mask_k = k_range < K

        # A: [BLOCK_M, BLOCK_K]  (row‑major)
        a_ptrs = a_base \
                 + offs_m[:, None] * stride_am \
                 + k_range[None, :] * stride_ak
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # B (already transposed): [BLOCK_K, BLOCK_N]
        b_ptrs = b_base \
                 + k_range[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Matrix multiplication on registers
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # Write back the result (FP32)
    # ------------------------------------------------------------------
    c_base = C_ptr + pid_b * batch_stride_c
    c_ptrs = c_base \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper to launch the kernel
# ----------------------------------------------------------------------
def launch_tri_mul(A, B_T, C,
                   block_m=64, block_n=64, block_k=64):
    """
    A : torch.float16 tensor of shape (B_mat, M, K)
    B_T: torch.float16 tensor of shape (B_mat, K, N)  (already transposed)
    C : torch.float32 tensor of shape (B_mat, M, N)  (output)
    """
    B_mat, M, K = A.shape
    _, K2, N = B_T.shape
    assert K == K2, "Inner dimensions must match"

    # strides in number of elements (not bytes)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)
    stride_bk = B_T.stride(1)
    stride_bn = B_T.stride(2)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    batch_stride_a = A.stride(0)
    batch_stride_b = B_T.stride(0)
    batch_stride_c = C.stride(0)

    grid = (
        (M + block_m - 1) // block_m,
        (N + block_n - 1) // block_n,
        B_mat,
    )
    tri_mul_kernel[grid](
        A_ptr=A,
        B_ptr=B_T,
        C_ptr=C,
        M=M,
        N=N,
        K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        batch_stride_a=batch_stride_a,
        batch_stride_b=batch_stride_b,
        batch_stride_c=batch_stride_c,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return C


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (x, mask, weights, config)
        - x      : torch.Tensor, shape [B, N, N, C] (float32)
        - mask   : torch.Tensor, shape [B, N, N]   (bool or float)
        - weights: dict of model parameters (float32 tensors)
        - config : dict containing at least:
            * "dim"         – C (input / output channel size)
            * "hidden_dim" – hidden dimension used inside the block
            * "nomask"      – bool, if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – the module output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]          # C
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input layer‑norm: nn.LayerNorm(dim) → weight/bias from `weights`
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),                       # normalize over last axis only
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → shape [..., hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional masking (mask broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is either bool or numeric → cast to same dtype as left/right
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul – compute Σₖ left[...,i,k,d] * right[...,j,k,d]
    #    Implemented as a batched GEMM (left @ rightᵀ) per hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    # Cast to FP16 for memory bandwidth, keep higher precision for the final output
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to (B, hidden, N, N) -> contiguous for the batch GEMM
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    # Collapse batch and hidden into a single GEMM batch dimension
    batch_mat = B * hidden_dim
    left_gemm = left_perm.view(batch_mat, N, N)                # [B*H, N, N]
    right_gemm = right_perm.view(batch_mat, N, N)              # [B*H, N, N]

    # Transpose the right operand on‑the‑fly by swapping its last‑two strides.
    # Creating a view avoids a memory copy.
    right_T = right_gemm.transpose(1, 2).contiguous()          # [B*H, N, N] (K×N)

    # Output buffer (FP32 as required by downstream LayerNorm)
    out_gemm = torch.empty_like(left_gemm, dtype=torch.float32)  # [B*H, N, N]

    # Launch the Triton kernel
    launch_tri_mul(
        A=left_gemm,
        B_T=right_T,
        C=out_gemm,
        block_m=64,
        block_n=64,
        block_k=64,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_gemm.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Apply the output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
