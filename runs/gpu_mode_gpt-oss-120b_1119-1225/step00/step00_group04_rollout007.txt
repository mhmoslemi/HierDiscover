Define kernel:

@triton.jit
def apply_mask_and_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    # compute flattened index
    off = pid * BLOCK + tl.arange(0, BLOCK)
    total = B * N * N * H
    mask = off < total
    # decode indices
    idx = tl.where(mask, off, 0)
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    proj = tl.load(proj_ptr + b*stride_proj_b + i*stride_proj_i + j*stride_proj_j + h*stride_proj_h, mask=mask)
    gate = tl.load(gate_ptr + b*stride_gate_b + i*stride_gate_i + j*stride_gate_j + h*stride_gate_h, mask=mask)

def custom_kernel(data):
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Normalize input
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Gates
    left_gate_raw = torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    right_gate_raw = torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    out_gate_raw = torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    left_gate = left_gate_raw.sigmoid()
    right_gate = right_gate_raw.sigmoid()
    out_gate = out_gate_raw.sigmoid()

    # Optional mask
    if mask is None or config.get("nomask", False):
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=device)
    else:
        # ensure mask float
        mask = mask.to(dtype=x.dtype)

    # Apply mask and gate via Triton kernel
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Precompute strides
    B, N, _, _ = x.shape
    stride_left_proj = left_proj.stride()
    stride_left_gate = left_gate.stride()
    stride_mask = mask.stride()
    stride_left_out = left.stride()
    stride_right_proj = right_proj.stride()
    stride_right_gate = right_gate.stride()
    stride_right_out = right.stride()

    total = B * N * N * hidden_dim
    BLOCK = 32768  # adjust

    grid = lambda meta: (triton.cdiv(total, BLOCK),)

    fuse_proj_gate_mask_kernel[grid](
        left_proj, left_gate, mask, left,
        B, N, hidden_dim,
        *stride_left_proj,
        *stride_left_gate,
        *stride_mask,
        *stride_left_out,
        BLOCK=BLOCK
    )
    fuse_proj_gate_mask_kernel[grid](
        right_proj, right_gate, mask, right,
        B, N, hidden_dim,
        *stride_right_proj,
        *stride_right_gate,
        *stride_mask,
        *stride_right_out,
        BLOCK=BLOCK
    )

    # Pairwise multiplicative update via batched matmul
    # reshape to (B*hidden_dim, N, N)
    left_ = left.permute(0, 3, 1, 2).contiguous()
    right_ = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_.view(B * hidden_dim, N, N)

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the feature dimension `dim`.
2. Linear projections → `left_proj`, `right_proj` (shape B×N×N×H).
3. Compute three sigmoid‑gates (`left_gate`, `right_gate`, `out_gate`) with
   linear layers (no bias) on the same normalized tensor.
4. Fuse the three element‑wise multiplies with the optional mask using a
   small Triton kernel:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
   (mask is broadcast over the hidden dimension H).
5. Pairwise multiplicative update:
       out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
   This is a batched matrix multiplication for each hidden channel.
   By reshaping to (B*H, N, N) we can reuse cuBLAS `torch.bmm`.
6. Layer‑norm the result over the hidden dimension `H`,
   apply `out_gate`, and project back to the original feature dimension
   with a final linear layer.
7. Return a tensor of shape (B, N, N, dim).

The only non‑trivial Triton part is step 4 – a single 1‑D kernel that
applies mask, gate and projection in one fused pass.  All heavy
linear and matrix‑multiply work stays on cuBLAS which gives the best
performance on an H100.
"""

import math
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused mask & gate multiplication.
# ----------------------------------------------------------------------
@triton.jit
def _fused_gate_mask_kernel(
    proj_ptr,                # projected values   (B,N,N,H)
    gate_ptr,                # sigmoid gates      (B,N,N,H)
    mask_ptr,                # binary/float mask (B,N,N)
    out_ptr,                 # output buffer      (B,N,N,H)

    B, N, H,                 # tensor sizes
    # strides for proj (4) and gate (4)
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    # strides for mask (3)
    stride_mask_b, stride_mask_i, stride_mask_j,
    # strides for output (4)
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK: tl.constexpr,    # number of threads per program
):
    """Each thread computes one element of the output tensor."""
    pid = tl.program_id(0)
    index = pid * BLOCK + tl.arange(0, BLOCK)                # linear index
    total = B * N * N * H
    mask = index < total

    # --------------------------------------------------------------
    # decode linear index into (b,i,j,h)
    # --------------------------------------------------------------
    b = index // (N * N * H)
    rem = index % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # --------------------------------------------------------------
    # load values (masked loads keep undefined values out of bounds)
    # --------------------------------------------------------------
    proj = tl.load(
        proj_ptr
        + b * stride_proj_b
        + i * stride_proj_i
        + j * stride_proj_j
        + h * stride_proj_h,
        mask=mask,
        other=0.0,
    )
    gate = tl.load(
        gate_ptr
        + b * stride_gate_b
        + i * stride_gate_i
        + j * stride_gate_j
        + h * stride_gate_h,
        mask=mask,
        other=0.0,
    )
    mask_val = tl.load(
        mask_ptr
        + b * stride_mask_b
        + i * stride_mask_i
        + j * stride_mask_j,
        mask=mask,
        other=0.0,
    )
    # apply fusion: projection * gate * mask
    out = proj * gate * mask_val

    # --------------------------------------------------------------
    # store result
    # --------------------------------------------------------------
    tl.store(
        out_ptr
        + b * stride_out_b
        + i * stride_out_i
        + j * stride_out_j
        + h * stride_out_h,
        out,
        mask=mask,
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor of shape [B, N, N] (may be None)
        - weights_dict : mapping from weight name to torch.Tensor
        - config_dict  : contains 'dim' and 'hidden_dim' (and optional 'nomask')

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the TriMul output.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, _ = x.shape  # x is [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (over the feature dim)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape (B,N,N,dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # left_proj, right_proj : (B,N,N,hidden_dim)
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()  # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 4. Optional mask handling
    # ------------------------------------------------------------------
    if mask is None or nomask:
        # all‑ones mask – cheap way to keep the kernel happy
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # 5. Fuse projection, gate and mask with Triton
    # ------------------------------------------------------------------
    # allocate output buffers
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # strides (in element counts, not bytes)
    s_proj_b, s_proj_i, s_proj_j, s_proj_h = left_proj.stride()
    s_gate_b, s_gate_i, s_gate_j, s_gate_h = left_gate.stride()
    s_mask_b, s_mask_i, s_mask_j = mask.stride()
    s_left_b, s_left_i, s_left_j, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_j, s_right_h = right.stride()

    total_elements = B * N * N * hidden_dim
    BLOCK = 32768  # tuneable – powers of two work well on H100
    grid = (triton.cdiv(total_elements, BLOCK),)

    # left = left_proj * left_gate * mask
    _fused_gate_mask_kernel[grid](
        left_proj,
        left_gate,
        mask,
        left,
        B,
        N,
        hidden_dim,
        s_proj_b,
        s_proj_i,
        s_proj_j,
        s_proj_h,
        s_gate_b,
        s_gate_i,
        s_gate_j,
        s_gate_h,
        s_mask_b,
        s_mask_i,
        s_mask_j,
        s_left_b,
        s_left_i,
        s_left_j,
        s_left_h,
        BLOCK=BLOCK,
    )
    # right = right_proj * right_gate * mask
    _fused_gate_mask_kernel[grid](
        right_proj,
        right_gate,
        mask,
        right,
        B,
        N,
        hidden_dim,
        *right_proj.stride(),
        *right_gate.stride(),
        s_mask_b,
        s_mask_i,
        s_mask_j,
        *right.stride(),
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # 6. Pairwise multiplicative update (batched matmul over axis k)
    #    out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) → treat each hidden channel as a separate batch
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # batched matrix multiply: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape (B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7. Output layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # apply out_gate (broadcast over hidden dim)
    out = out * out_gate
    # final linear to original dimension (no bias)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
