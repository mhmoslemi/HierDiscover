"""
TriMul forward (outgoing) kernel.

The heavy part of the TriMul operation is the tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` have shape [B, N, N, D] (D == hidden_dim).
For each batch and each hidden dimension we need a matrix multiplication
between an (N × N) matrix and the transpose of another (N × N) matrix.
The implementation below fuses this computation into a single Triton
kernel that processes one (B, hidden) pair at a time, using block‑wise
loading and `tl.dot` accumulation.  The kernel works in FP16 for the
inputs (to reduce memory traffic) and accumulates in FP32 for numerical
stability.  All surrounding LayerNorm, gating, and linear projections are
performed with PyTorch, keeping the solution simple while still
demonstrating a custom Triton kernel for the dominant O(N³·D) work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the tri‑multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,        # *fp16  left tensor  [B,N,N,D]
    right_ptr,       # *fp16  right tensor [B,N,N,D]
    out_ptr,         # *fp16  output tensor [B,N,N,D]
    B, N, D,         # batch, sequence length, hidden dimension
    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

    left, right, out are stored as contiguous (B, N, N, D) tensors.
    The kernel iterates over k in tiles of size BLOCK_K and uses
    tl.dot to accumulate a (BLOCK_M × BLOCK_N) tile of the result.
    """

    pid = tl.program_id(0)

    # total number of independent (B, D) matrix multiplies
    total_md = B * D
    # compute which (b,d) we are processing and the tile indices i/j
    md_idx = pid // (tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    tile_i = (pid // tl.cdiv(N, BLOCK_N)) % tl.cdiv(N, BLOCK_M)
    tile_j = pid % tl.cdiv(N, BLOCK_N)

    # decode batch and hidden indices
    b = md_idx // D
    d = md_idx % D

    # Global i/j coordinates of the tile's top‑left corner
    i0 = tile_i * BLOCK_M
    j0 = tile_j * BLOCK_N

    # Range indices inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Predicate for out‑of‑bounds rows/cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_iter * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile: shape (BLOCK_M, BLOCK_K) -----
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k \
            + d * stride_left_d
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # fp16 → fp32

        # ----- load right tile: shape (BLOCK_N, BLOCK_K) -----
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + offs_j[:, None] * stride_right_j \
            + offs_k[None, :] * stride_right_k \
            + d * stride_right_d
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0).to(tl.float32)  # fp16 → fp32

        # ----- matrix‑multiply‑accumulate -----
        acc += tl.dot(left_tile, right_tile)   # (M,K)·(N,K) → (M,N)

    # ----- write the result tile back -----
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j \
        + d * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def batched_tri_mul(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    """
    left/right:   [B, N, N, D]   (torch.float16, contiguous)
    returns out: [B, N, N, D]   (torch.float16)
    """
    B, N, _, D = left_fp16.shape
    out_fp16 = torch.empty_like(left_fp16)

    # tile sizes – empirically good on H100 for the target shapes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # number of (B,D) matrix multiplications
    total_md = B * D
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = total_md * grid_i * grid_j

    # strides (all int64)
    s_lb, s_li, s_lk, s_ld = left_fp16.stride()
    s_rb, s_rj, s_rk, s_rd = right_fp16.stride()
    s_ob, s_oi, s_oj, s_od = out_fp16.stride()

    tri_mul_kernel[total_programs](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        D,
        s_lb,
        s_li,
        s_lk,
        s_ld,
        s_rb,
        s_rj,
        s_rk,
        s_rd,
        s_ob,
        s_oi,
        s_oj,
        s_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # enough warps for the tile size
    )
    return out_fp16

    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Apply mask (if any)
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)
    # reshape to broadcast over the channel dimension
    mask_broadcast = mask.unsqueeze(-1)                 # [B,N,N,1]
    left = left * mask_broadcast
    right = right * mask_broadcast

    # ------------------------------------------------------------------
    # 4. Gating networks (sigmoid on projected features)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul contraction – fused Triton kernel
    # ------------------------------------------------------------------
    # Cast to FP16 to reduce memory traffic; the kernel accumulates in FP32
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    out_fp16 = batched_tri_mul(left_fp16, right_fp16)   # [B,N,N,hidden_dim] (fp16)

    # Return to FP32 for the remaining FC layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm (over hidden_dim) and gating
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])      # [B,N,N,dim]

    return out
```
