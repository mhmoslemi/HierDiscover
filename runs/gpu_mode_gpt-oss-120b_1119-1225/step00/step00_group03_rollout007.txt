def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    pid_n = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute i and j offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # out-of-bound masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    b = pid_bh // H

    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

        acc += tl.dot(a, b)

    # store results

    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul outward (AlphaFold3) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑normalize the input `x` (shape [B,N,N,D]).
2. Linear projections `left = x·W_left`, `right = x·W_right` (to H hidden dim).
3. Optional pair mask is applied to the projected tensors.
4. Compute three sigmoid gates from the original normalized tensor:
   `left_gate`, `right_gate`, `out_gate`.
5. Apply the first two gates element‑wise: `left *= left_gate`,
   `right *= right_gate`.
6. Compute the core N³ update:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply `left @ rightᵀ` for each hidden
   dimension `h`.  The operation is performed by a custom Triton kernel
   that tiles the M (i) and N (j) axes (BLOCK_M×BLOCK_N) and iterates
   over K (k) in BLOCK_K steps, accumulating in float32.
7. Layer‑normalize the intermediate result over the hidden dimension,
   multiply by the `out_gate` sigmoid, and finally project back to the
   original feature dimension with a linear layer.
8. The final tensor has shape `[B, N, N, D]` and dtype float32.

The Triton kernel works on a combined `(batch, hidden)` axis to keep the
implementation simple while still achieving high occupancy on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trmul_kernel(
    left_ptr,            # *float32
    right_ptr,           # *float32
    out_ptr,             # *float32
    B, N, H,             # scalar ints
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ rightᵀ for each (b, h) pair,
    where left, right: [B, N, N, H] (contiguous).
    """
    pid_m = tl.program_id(0)      # block index over M (i)
    pid_n = tl.program_id(1)      # block index over N (j)
    pid_bh = tl.program_id(2)     # combined batch‑hidden index

    # Determine i / j offsets for this block
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # De‑compose (batch, hidden) index
    b = pid_bh // H
    h = pid_bh %  H

    # Base pointers for the current (b, h)
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the summed axis)
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        start_k = k * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # right is stored as [b, j, k, h]; we need element (k, j)
        b_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # Accumulate dot product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Store the result
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] (dtype float16/32)
            - mask: torch.Tensor [B, N, N] (optional, 0/1)
            - weights: dict of model weights as torch.Tensors (already on device)
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all tensors are assumed to be on the same device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the feature dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections on the original normed x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply the first two gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous layout for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # Cast to float32 for the kernel (accumulates in FP32)
    left = left.to(torch.float32)
    right = right.to(torch.float32)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5. Core TriMul: out = left @ rightᵀ  (batched matrix multiply per hidden dim)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=device)

    # Compute strides (these are integers, not tensors)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tuning parameters – chosen to fit H100 shared‑memory limits comfortably
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over i
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over j
        B * hidden_dim                 # combined batch‑hidden dimension
    )

    _trmul_kernel[grid](
        left,
        right,
        out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_j, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑process: LayerNorm → out_gate → final linear projection
    # ------------------------------------------------------------------
    # Layer‑norm over the hidden dimension
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to the original feature dimension
    out = F.linear(out, to_out_weight)  # shape [B, N, N, dim]

    # Ensure final dtype is float32 as required
    return out.to(torch.float32)
