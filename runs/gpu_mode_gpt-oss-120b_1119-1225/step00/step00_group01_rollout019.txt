@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    M, N, K,
    stride_a_m, stride_a_k,
    stride_b_k, stride_b_n,
    stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    pid_remaining = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid_remaining // num_blocks_n
    pid_n = pid_remaining % num_blocks_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute pointers for each batch
    a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch * batch_stride_b + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, K, BLOCK_K):
        # Load tiles
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)
        # Update pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k
    # Write output
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n

def custom_kernel(data):
    """TriMul forward pass using fused LayerNorm, Linear projections and a custom Triton
    batched GEMM kernel for the O(N^3 * hidden_dim) core operation.

    Only the forward pass is implemented (no gradients).
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-6)
    nomask = config.get("nomask", True)

    # LayerNorm on dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask handling
    if not nomask:
        # mask shape [B, N, N]
        mask_ = mask.unsqueeze(-1).to(x.dtype)   # broadcast to last dim
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Core multiplication: out = einsum('... i k d, ... j k d -> ... i j d', left, right)
    # Rearrange to batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    batch = B * H

    left_flat = left_perm.view(batch, N, N)
    right_flat = right_perm.view(batch, N, N)

    out_flat = torch.empty_like(left_flat)

    # Compute strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Block sizes (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m * num_blocks_n,)

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch_stride_a=stride_a_batch,
        batch_stride_b=stride_b_batch,
        batch_stride_c=stride_c_batch,
        M=N,
        N=N,
        K=N,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout
    out_perm = out_flat.view(B, H, N, N)           # (B, H, i, j)
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # (B, i, j, H)

    # LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Apply out gate
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_stride_a, batch_stride_b, batch_stride_c,
    M, N, K,
    stride_a_m, stride_a_k,
    stride_b_k, stride_b_n,
    stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM kernel for shape (B, M, K) @ (B, K, N) -> (B, M, N)
    where B = batch size (here = batch*hidden) and M=N=K=seq_len.
    The kernel is written for float32 inputs."""
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # batch id and block indices
    batch = pid // (num_blocks_m * num_blocks_n)
    pid_remaining = pid % (num_blocks_m * num_blocks_n)
    pid_m = pid_remaining // num_blocks_n
    pid_n = pid_remaining % num_blocks_n

    # block coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for this batch
    a_ptrs = a_ptr + batch * batch_stride_a + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch * batch_stride_b + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # accumulation buffer
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c_ptrs = c_ptr + batch * batch_stride_c + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

    """TriMul forward pass: layer‑norm → linear projections → gated mask → 
    batched GEMM (implemented with a custom Triton kernel) → layer‑norm → 
    output gating → final linear.  Only the forward pass is needed.
    
    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights      : dict of torch.Tensors with keys matching the model
        - config       : dict containing ``dim`` and ``hidden_dim`` (and optional ``eps`` and ``nomask``)
    
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-6)
    nomask = config.get("nomask", True)

    # -------- LayerNorm on the channel dimension --------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------- Linear projections (no bias) --------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------- Optional mask (broadcast over the hidden dim) --------
    if not nomask:
        # mask is shape [B, N, N]; we need [B, N, N, 1] to broadcast
        mask_ = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # -------- Gating (sigmoids) --------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -------- Core multiplicative update (einsum) ----------
    # Rearrange tensors so the operation becomes a batched matrix multiply:
    #   left  : (B, N, N, H) -> (B, H, i, k)
    #   right : (B, N, N, H) -> (B, H, k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()           # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, H, k, j)

    B, H, N, _ = left_perm.shape
    batch = B * H

    left_flat = left_perm.view(batch, N, N)   # (B*H, M, K)
    right_flat = right_perm.view(batch, N, N) # (B*H, K, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Strides (in elements) for the flattened tensors
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Block sizes – tuned for H100 (float32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch * num_blocks_m * num_blocks_n,)

        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch_stride_a=stride_a_batch,
        batch_stride_b=stride_b_batch,
        batch_stride_c=stride_c_batch,
        M=N,
        N=N,
        K=N,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, H)
    out_perm = out_flat.view(B, H, N, N)               # (B, H, i, j)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, i, j, H)

    # -------- LayerNorm on hidden dimension --------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original channel dimension
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

# Triton‑based implementation of the “outgoing” TriMul operator (forward only).
# The heavy O(N³·hidden) part (einsum) is realised as a batched GEMM kernel.
# All other operations (LayerNorm, Linear, Sigmoid, masking) are performed with
# PyTorch for simplicity.  The final tensor shape is [B, N, N, dim] (float32).

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Batched GEMM kernel
# -------------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch_stride_a, batch_stride_b, batch_stride_c,  # batch strides
    M, N, K,                                 # matrix sizes (all = seq_len)
    stride_a_m, stride_a_k,                  # stride of A (batch, M, K)
    stride_b_k, stride_b_n,                  # stride of B (batch, K, N)
    stride_c_m, stride_c_n,                  # stride of C (batch, M, N)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute (batch, M, K) @ (batch, K, N) -> (batch, M, N) for many batches.
    All tensors are assumed to be contiguous in row‑major order.
    """
    pid = tl.program_id(0)

    # number of blocks per matrix dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # decode program id into (batch, block_m, block_n)
    batch = pid // (num_blocks_m * num_blocks_n)
    pid_rem = pid % (num_blocks_m * num_blocks_n)
    pid_m   = pid_rem // num_blocks_n
    pid_n   = pid_rem % num_blocks_n

    # block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # pointers for this batch
    a_ptrs = a_ptr + batch * batch_stride_a \
                  + offs_m[:, None] * stride_a_m \
                  + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch * batch_stride_b \
                  + offs_k[:, None] * stride_b_k \
                  + offs_n[None, :] * stride_b_n

    # accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K steps
    for _ in range(0, K, BLOCK_K):
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)
        acc += tl.dot(a, b)          # accumulation in FP32
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c_ptrs = c_ptr + batch * batch_stride_c \
                  + offs_m[:, None] * stride_c_m \
                  + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

# Public entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing variant).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]  (float32)
        - mask         : torch.Tensor of shape [B, N, N]   (optional)
        - weights      : dict of torch.Tensors with keys:
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config       : dict containing at least:
            "dim" (C), "hidden_dim" (H), optional "eps", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # unpack
    inp, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = config.get("eps", 1e-6)
    nomask       = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_ = mask.to(x.dtype).unsqueeze(-1)
        left  = left * mask_
        right = right * mask_

    # 4. Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core multiplicative update (einsum) => batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    # Re‑order to (batch, hidden, i, k) and (batch, hidden, k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, k, j)

    B, H, N, _ = left_perm.shape

    left_flat  = left_perm.view(batch_total, N, N)   # (B*H, M, K)
    right_flat = right_perm.view(batch_total, N, N)  # (B*H, K, N)

    # allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # strides (in elements)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # tune block sizes for an H100 (FP32)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (batch_total * num_blocks_m * num_blocks_n,)

    batched_gemm_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_flat,
        c_ptr=out_flat,
        batch_stride_a=stride_a_batch,
        batch_stride_b=stride_b_batch,
        batch_stride_c=stride_c_batch,
        M=N,
        N=N,
        K=N,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to original layout
    out_perm = out_flat.view(B, H, N, N)               # (B, H, i, j)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, i, j, H)

    # -----------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm, output gate and final linear projection
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
