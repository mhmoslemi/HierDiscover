def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']; hidden_dim = config['hidden_dim']; nomask = config.get('nomask', False)

    # using torch's functional layers
    # norm
    # Use torch.nn.functional.layer_norm
    eps = 1e-5 # typical
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'], eps=eps)

    # linear projections
    left = torch.nn.functional.linear(x_norm, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x_norm, weights['right_proj.weight'])

    # mask multiplication if any
    if not config.get('nomask', True):
        # mask shape (B,N,N) -> (B,N,N,1)
        mask_unsq = mask.unsqueeze(-1)
        left = left * mask_unsq
        right = right * mask_unsq

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    B,N,_,_ = left.shape  # want B,N,N,hidden_dim

    # reshape for batched matmul
    left_reshape = left.view(B*hidden_dim, N, N)
    right_reshape = right.view(B*hidden_dim, N, N)
    # Note: we need out shape (B, N, N, hidden_dim). We'll compute as out_reshape then reshape.

    # allocate out
    out_reshape = torch.empty_like(left_reshape) # shape B*H,N,N

    # call triton kernel
    grid = ...
    triton_einsum_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B*hidden_dim, N,
        left_reshape.stride(0), left_reshape.stride(1), left_reshape.stride(2),
        right_reshape.stride(0), right_reshape.stride(1), right_reshape.stride(2),
        out_reshape.stride(0), out_reshape.stride(1), out_reshape.stride(2),
        BLOCK_M=64, BLOCK_N=64, BLOCK_K=32
    )

    # reshape back to (B,N,N,hidden_dim)
    out = out_reshape.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # to_out_norm (layernorm)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias'], eps=eps)

    # gating
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights['to_out.weight'])

    return out

@triton.jit
def batched_gram_kernel(
    left_ptr, right_ptr, out_ptr,
    B_H: tl.int32, N: tl.int32,
    stride_l_batch: tl.int32, stride_l_i: tl.int32, stride_l_k: tl.int32,
    stride_r_batch: tl.int32, stride_r_j: tl.int32, stride_r_k: tl.int32,
    stride_o_batch: tl.int32, stride_o_i: tl.int32, stride_o_j: tl.int32,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute mask for out-of-bound
    mask_m = offs_m < N
    mask_n = offs_n < N
    mask_k = offs_k < N

    # pointers offset for left (B_H,N,N)
    a_ptrs = left_ptr + pid_batch * stride_l_batch + offs_m[:, None] * stride_l_i + (offs_k[None, :] + 0) * stride_l_k
    # For right (Transposed)
    b_ptrs = right_ptr + pid_batch * stride_r_batch + (offs_k[:, None]) * stride_r_k + offs_n[None, :] * stride_r_j

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute per block offset
        cur_k = k * BLOCK_K
        # recompute k offsets for each iteration
        cur_k_offs = cur_k + offs_k
        mask_k = cur_k_offs < N

        # pointers for this k block
        a = tl.load(a_ptrs + cur_k * stride_l_k,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptrs + cur_k * stride_r_k,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)

    # store result
    c_ptrs = out_ptr + pid_batch * stride_o_batch + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=weights["norm.weight"],
                                       bias=weights["norm.bias"],
                                       eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask
    if not nomask:
        mask_unsq = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    B, N, _, _ = left.shape  # left shape (B,N,N,hidden_dim)
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # reshape to (B*hidden_dim, N, N)
    left_reshape = left_perm.view(B * hidden_dim, N, N)
    right_reshape = right_perm.view(B * hidden_dim, N, N)

    out_reshape = torch.empty_like(left_reshape)  # same dtype (float32)

    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # strides
    stride_l_batch = left_reshape.stride(0)
    stride_l_i = left_reshape.stride(1)
    stride_l_k = left_reshape.stride(2)

    stride_r_batch = right_reshape.stride(0)
    stride_r_j = right_reshape.stride(1)  # dimension for 'j' (rows) after transpose
    stride_r_k = right_reshape.stride(2)

    stride_o_batch = out_reshape.stride(0)
    stride_o_i = out_reshape.stride(1)
    stride_o_j = out_reshape.stride(2)

    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    batched_gram_kernel[grid](
        left_reshape,
        right_reshape,
        out_reshape,
        B * hidden_dim,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_j, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # reshape back
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(out,
                                         (hidden_dim,),
                                         weight=weights["to_out_norm.weight"],
                                         bias=weights["to_out_norm.bias"],
                                         eps=eps)

    # gating
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm over the channel dimension `dim`.
2. Linear projections (`left_proj`, `right_proj`) to the hidden dimension.
3. Optional pairwise mask multiplication.
4. Gated versions of the projections (`left_gate`, `right_gate`).
5. Core O(N³) update:
       out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   The heavy reduction over `k` is performed by a batched GEMM‑style Triton
   kernel (`batched_gram_kernel`).  The tensors are arranged as
   (B·H, N, N) where `H = hidden_dim`; each program instance computes a
   TILE_M × TILE_N block of the Gram matrix for one (batch, hidden) pair.
   The kernel works on FP16 inputs (to halve memory traffic and arithmetic
   cost) while accumulating in FP32.
6. Layer‑norm over the hidden dimension, gated by `out_gate`.
7. Final linear projection back to `dim`.

The implementation mixes high‑level torch ops (LayerNorm, Linear,
sigmoid) with the Triton kernel for the O(N³) work, giving a fast
GPU‑only forward pass suitable for the test configurations.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched Gram matrix (Σₖ left[i,k] * right[j,k])
# --------------------------------------------------------------
@triton.jit
def batched_gram_kernel(
    left_ptr,                 # fp16 pointer, shape (B·H, N, N)
    right_ptr,                # fp16 pointer, shape (B·H, N, N)
    out_ptr,                  # fp32 pointer, shape (B·H, N, N)
    B_H,                      # total number of (batch, hidden) slices
    N,                        # sequence length
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_j, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute a batched Gram matrix using tiled matrix multiplication.

    For each slice `b` in 0..B_H‑1 we compute
        out[b, i, j] = Σₖ left[b, i, k] * right[b, j, k]
    with tiling in the (i, j) output plane.
    """
    pid_batch = tl.program_id(0)          # which (batch, hidden) slice
    pid_m     = tl.program_id(1)          # tile row index
    pid_n     = tl.program_id(2)          # tile col index

    # ------------------------------------------------------------------
    # Offsets for the current tile in the output matrix (i,j)
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [i0, i1, ...]
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [j0, j1, ...]
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers for left[b,i,k]
        a_ptrs = (left_ptr
                  + pid_batch * stride_l_batch
                  + offs_i[:, None] * stride_l_i
                  + offs_k[None, :] * stride_l_k)

        # Pointers for right[b,j,k] -> we need (k,j) layout for dot
        b_ptrs = (right_ptr
                  + pid_batch * stride_r_batch
                  + offs_k[:, None] * stride_r_k     # row = k
                  + offs_j[None, :] * stride_r_j)    # col = j

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # fp16 -> auto‑promoted
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        # accumulate in FP32 for accuracy
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    c_ptrs = (out_ptr
              + pid_batch * stride_o_batch
              + offs_i[:, None] * stride_o_i
              + offs_j[None, :] * stride_o_j)

    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Entry point expected by the evaluator
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (float32 or bool)
        - weights      : dict mapping string names to torch.Tensor
        - config       : dict with keys "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]          # input channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    nomask     = config.get("nomask", True)

    eps = 1e-5  # LayerNorm epsilon (matches PyTorch default)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(dtype)  # (B,N,N,1)
        left  = left * mask_unsq
        right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear maps)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) update via Triton kernel
    #    - reshape to (B*H, N, N) and cast to fp16 for the kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # original batch, seq_len, seq_len, hidden_dim

    # bring hidden dimension next to batch and make tensors contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # flatten batch & hidden into one dimension
    left_fp16  = left_perm.view(B * hidden_dim, N, N).to(torch.float16)
    right_fp16 = right_perm.view(B * hidden_dim, N, N).to(torch.float16)

    # output buffer (FP32, because we accumulate in FP32)
    out_fp32 = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_l_batch = left_fp16.stride(0)
    stride_l_i     = left_fp16.stride(1)
    stride_l_k     = left_fp16.stride(2)

    stride_r_batch = right_fp16.stride(0)
    stride_r_j     = right_fp16.stride(1)   # column (j) stride
    stride_r_k     = right_fp16.stride(2)   # row   (k) stride

    stride_o_batch = out_fp32.stride(0)
    stride_o_i     = out_fp32.stride(1)
    stride_o_j     = out_fp32.stride(2)

    grid = (
        B * hidden_dim,                 # program id 0  : batch * hidden slice
        (N + BLOCK_M - 1) // BLOCK_M,  # program id 1  : tile row
        (N + BLOCK_N - 1) // BLOCK_N,  # program id 2  : tile col
    )

    batched_gram_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        B * hidden_dim,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_j, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_fp32.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output layer‑norm + gating
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
