def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # Linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    # Apply mask and gates via triton kernel
    B,N,_,_ = left.shape  # left shape (B,N,N,H)
    # casts
    mask_f = mask.to(left.dtype)  # ensure float
    # allocate new tensors
    left_fused = torch.empty_like(left)
    right_fused = torch.empty_like(right)

    # Launch kernel
    grid = lambda meta: (triton.cdiv(B*N*N*hidden_dim, meta['BLOCK_SIZE']),)

    fuse_proj_gate_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask_f,
        left_fused, right_fused,
        B, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_f.stride(0), mask_f.stride(1), mask_f.stride(2),
        # maybe pass strides for output too
        left_fused.stride(0), left_fused.stride(1), left_fused.stride(2), left_fused.stride(3),
        right_fused.stride(0), right_fused.stride(1), right_fused.stride(2), right_fused.stride(3),
        BLOCK_SIZE=256  # depends 
    )

    # Now compute einsum via batched matmul
    # reshape to (B*hidden_dim, N, N)
    left_mat = left_fused.permute(0,3,1,2).reshape(B*hidden_dim, N, N)

@triton.jit
def fuse_proj_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_left_out_b, stride_left_out_i, stride_left_out_j, stride_left_out_h,
    stride_right_out_b, stride_right_out_i, stride_right_out_j, stride_right_out_h,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offset = pid * BLOCK_SIZE
    range = tl.arange(0, BLOCK_SIZE)
    index = offset + range

    total = B * N * N * H
    mask = index < total
    if not mask.any():
        return

    # compute multi-dim indices
    tmp = index
    b = tmp // (N * N * H)
    tmp = tmp % (N * N * H)
    i = tmp // (N * H)
    tmp = tmp % (N * H)
    j = tmp // H
    h = tmp % H

    # compute pointers
    left_offset = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h

    left_gate_offset = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    right_gate_offset = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h

    hidden_dim = config['hidden_dim']

        (dim,),
        weight=weights['norm.weight'],
        bias=weights['norm.bias']
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    # optional mask handling
    if mask is None or config.get('nomask', False):
        mask_f = torch.ones_like(input_tensor[..., 0])
    else:
        # convert boolean mask to float if needed
        mask_f = mask.to(x.dtype)

    # Fuse mask and gates into left/right via Triton kernel (in-place)
    B, N, _, _ = left.shape
    H = hidden_dim
    # ensure mask shape matches B,N,N

    mask_f = mask_f.contiguous()

    stride_left = left.stride()
    stride_right = right.stride()
    stride_lgate = left_gate.stride()
    stride_rgate = right_gate.stride()
    stride_mask = mask_f.stride()

        stride_rgate[0], stride_rgate[1], stride_rgate[2], stride_rgate[3],
        stride_mask[0], stride_mask[1], stride_mask[2],

"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‑norm the 4‑D input   [B, N, N, C].
2. Four linear projections without bias:
        left, right                 : (B,N,N,hidden)
        left_gate, right_gate, out_gate : (B,N,N,hidden)   (followed by sigmoid)
3. Fuse mask (if any) and the two gates into ``left`` and ``right``.
   This step is executed by a small Triton kernel that performs an
   in‑place element‑wise multiplication:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
4. Compute the core “tri‑multiplicative” term
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   The contraction over `k` is a batched matrix multiplication.
   By reshaping we obtain (B·hidden, N, N) × (B·hidden, N, N)ᵀ via
   torch.bmm, which is heavily optimized on H100.
5. Apply a second LayerNorm, multiply by the ``out_gate`` and a final
   linear projection back to the original channel dimension C.

Only the element‑wise fusion (step 3) is written in Triton; the heavy
linear layers and the batched GEMM use PyTorch’s highly‑optimized kernels.
The implementation works for any batch size (1‑2), sequence length up to
1024, and hidden dimensions 128 (or any user‑provided value).
"""

import triton.language as tl

# Triton kernel: in‑place fusion of mask and gate tensors into left/right
# ----------------------------------------------------------------------------- #
@triton.jit
def _fuse_mask_gate_kernel(
    left_ptr,        # [B, N, N, H]  (in‑place)
    right_ptr,       # [B, N, N, H]  (in‑place)
    left_gate_ptr,   # [B, N, N, H]
    right_gate_ptr,  # [B, N, N, H]
    mask_ptr,        # [B, N, N]
    # meta‑data
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # 1‑D index → (b,i,j,h)
    idx = offs
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute flat offsets for each tensor
    left_offset   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j   + h * stride_left_h
    right_offset  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j  + h * stride_right_h
    lgate_offset  = b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j  + h * stride_lgate_h
    rgate_offset  = b * stride_rgate_b  + i * stride_rgate_i  + j * stride_rgate_j  + h * stride_rgate_h
    mask_offset   = b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j

    # load values (out‑of‑bounds reads return 0 / 1)
    left_val   = tl.load(left_ptr   + left_offset,   mask=mask, other=0.0)
    right_val  = tl.load(right_ptr  + right_offset,  mask=mask, other=0.0)

        B, N, H,
        sl[0], sl[1], sl[2], sl[3],
        sr[0], sr[1], sr[2], sr[3],
        slg[0], slg[1], slg[2], slg[3],
        srg[0], srg[1], srg[2], srg[3],
        sm[0], sm[1], sm[2],
        BLOCK_SIZE=BLOCK
    )
    # after the kernel `left` and `right` already contain the masked & gated values

    # --------------------------------------------------------------------- #
    # 4️⃣ Core TriMul contraction: Σ_k left[b,i,k,d] * right[b,j,k,d]
    # --------------------------------------------------------------------- #
    # reshape (B, N, N, H) → (B*H, N, N) for batched GEMM
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # (B*H, i, k)
    # right needs its (k, j) ordering → transpose the last two dims before flatten
    right_mat = right.permute(0, 3, 2, 1).reshape(B * H, N, N)        # (B*H, k, j)

    # batched matrix multiplication (highly tuned on H100)
    out_mat = torch.bmm(left_mat, right_mat)                         # (B*H, i, j)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B, N, N, H)

    # --------------------------------------------------------------------- #
    # 5️⃣ Output LayerNorm, gating, final projection back to C
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
