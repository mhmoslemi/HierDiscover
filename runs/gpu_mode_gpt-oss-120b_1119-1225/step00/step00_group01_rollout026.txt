def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    device = input_tensor.device

    # unpack weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # LayerNorm input
    # x_norm = (input - mean) / sqrt(var+eps) * weight + bias
    eps = 1e-5
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, normalized_shape=dim, weight=norm_weight, bias=norm_bias, eps=eps)

    # Convert to half for throughput
    x_norm_h = x_norm.half()

    # Linear projections to hidden dim
    left = torch.nn.functional.linear(x_norm_h, left_proj_weight.half())
    right = torch.nn.functional.linear(x_norm_h, right_proj_weight.half())
    left_gate = torch.nn.functional.linear(x_norm_h, left_gate_weight.half()).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_h, right_gate_weight.half()).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_h, out_gate_weight.half()).sigmoid()

    # Apply mask if present
    if mask is not None:
        # mask shape [B,N,N] maybe bool or float
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]

        out_f, normalized_shape=hidden_dim,
        weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out gate
    out_norm = out_norm * out_gate.to(torch.float32)

    # Final linear projection to dim
    output = torch.nn.functional.linear(out_norm, to_out_weight)  # weight shape [dim, hidden_dim]
    return output

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0) # combined batch and hidden
    pid1 = tl.program_id(1) # i block
    pid2 = tl.program_id(2) # j block

    # Compute indices for batch and hidden dimension
    b = pid0 // D
    d = pid0 % D
    # Offsets
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # Prepare indexes
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_ij = mask_i[:, None] & mask_j[None, :]

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Base ptr offsets for left and right and out for specific batch and hidden
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base = out_ptr + b * stride_out_b + d * stride_out_d

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute dynamic offsets for this k block
        cur_k = k * BLOCK_K
        offs_k_cur = cur_k + offs_k
        mask_k = offs_k_cur < N

        # Load A block [M,K]
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k_cur[None, :] * stride_left_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B block [K,N] (note transposed order)
        b_ptrs = right_base + offs_k_cur[:, None] * stride_right_k + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)  # (M,K)@(K,N) => (M,N)

    # Store result
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_ij)

import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the core TriMul step.
# Computes, for every batch b and hidden channel d:
#   out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
# i.e. a series of N×N matrix products left[..., :, d] @ right[..., :, d].T.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, D,                                    # sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0)          # combined batch * hidden channel
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    # ------------------------------------------------------------------
    # Decode batch index (b) and hidden channel index (d)
    # ------------------------------------------------------------------
    b = pid0 // D
    d = pid0 % D

    # ------------------------------------------------------------------
    # Starting coordinates of the tile
    # ------------------------------------------------------------------
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # ------------------------------------------------------------------
    # Offsets within the current tile
    # ------------------------------------------------------------------
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ------------------------------------------------------------------
    # Boundary masks (so we never read/write out‑of‑bounds)
    # ------------------------------------------------------------------
    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator in FP32 (higher‑precision reduction)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Base pointers for the current (b, d) slice
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + d * stride_left_d
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + d * stride_out_d

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for cur_k_block in range(0, tl.cdiv(N, BLOCK_K)):

        cur_k_offs = cur_k + offs_k
        mask_k = cur_k_offs < N

                 + offs_j[None, :] * stride_right_j

                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

def tri_mul_kernel_launcher(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches `tri_mul_kernel`.  Both `left` and `right`
    must be contiguous tensors of shape [B, N, N, H] with dtype torch.float16.
    Returns a tensor of the same shape/dtype containing the batched products.
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2

    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)                     # FP16 output

    # Strides in number of elements (not bytes)
    s_lb, s_li, s_lk, s_ld = left.stride()
    s_rb, s_rj, s_rk, s_rd = right.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # Tile sizes – tuned for H100 (fits nicely into a warp and uses Tensor Cores)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                     # program id 0  -> batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,              # program id 1  -> i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,              # program id 2  -> j‑tiles
    )

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        s_lb, s_li, s_lk, s_ld,
        s_rb, s_rj, s_rk, s_rd,
        s_ob, s_oi, s_oj, s_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the *outgoing* TriMul module
    (AlphaFold3).  The function receives a tuple
        (input_tensor, mask, weights, config)
    and returns the processed tensor of shape [B, N, N, dim].

    Steps performed:
        1. Layer‑norm the input.
        2. FP16 linear projections (left, right) and three gating projections.
        3. Optional mask application.
        4. Element‑wise gating of left/right.
        5. Batched matrix multiplication via a custom Triton kernel
           (the O(N³) core of TriMul).
        6. FP32 Layer‑norm on the hidden dimension, output gating,
           and final linear projection back to `dim`.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------
    # Unpack model parameters (all are stored as torch.float32 tensors)
    # --------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the feature dimension `dim`
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=dim,
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2. Convert to FP16 for the heavy linear work
    # --------------------------------------------------------------
    x_fp16 = x.half()

    # 3. Linear projections (no bias)
    left  = F.linear(x_fp16, left_proj_weight.half())
    right = F.linear(x_fp16, right_proj_weight.half())

    # 4. Gating projections + sigmoid
    left_gate  = F.linear(x_fp16, left_gate_weight.half()).sigmoid()
    right_gate = F.linear(x_fp16, right_gate_weight.half()).sigmoid()
    out_gate   = F.linear(x_fp16, out_gate_weight.half()).sigmoid()

    # --------------------------------------------------------------
    # 5. Optional mask (mask shape: [B, N, N])
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_fp16
        right = right * mask_fp16

    # --------------------------------------------------------------
    # 6. Apply element‑wise gates (in‑place to save memory)
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 7. Core O(N³) multiplication via Triton (outputs FP16)
    # --------------------------------------------------------------
    hidden = tri_mul_kernel_launcher(left, right)   # shape [B, N, N, hidden_dim], fp16

    # --------------------------------------------------------------
    # 8. Cast back to FP32 for the second LayerNorm and final projection
    # --------------------------------------------------------------
    hidden_fp32 = hidden.to(torch.float32)

    # Second LayerNorm over the hidden dimension
    hidden_norm = F.layer_norm(
        hidden_fp32,
        normalized_shape=hidden_dim,
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # Output gating (broadcasted over the hidden dimension)
    hidden_norm = hidden_norm * out_gate.to(torch.float32)

    # Final projection back to the original `dim`
    output = F.linear(hidden_norm, to_out_weight)   # shape [B, N, N, dim]

    return output
