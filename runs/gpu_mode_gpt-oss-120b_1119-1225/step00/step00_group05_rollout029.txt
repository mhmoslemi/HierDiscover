"""
TriMul (outgoing) custom implementation.

Algorithm
---------
1. Input LayerNorm:   x = LayerNorm(x)                         (dim)
2. Linear projections (no bias):
       left  = x @ W_left   -> (B,N,N,hidden)
       right = x @ W_right  -> (B,N,N,hidden)
3. Optional mask multiplication (broadcast to hidden dim).
4. Gating (sigmoid of linear):
       left  *= sigmoid(x @ W_left_gate)
       right *= sigmoid(x @ W_right_gate)
5. Triangular multiplicative update:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This contraction is performed by a Triton kernel that iterates over the
   sequence dimension `k` in blocks.  The kernel processes a hidden‑slice
   of size BLOCK_D (default 32) to stay within register budget.
6. LayerNorm on the hidden dimension.
7. Output gate:  out *= sigmoid(x @ W_out_gate)
8. Final linear projection back to `dim`.

Only the heavy O(N³·hidden) contraction (step 5) is written in Triton;
all other ops use PyTorch which already provides highly‑optimized
implementations.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim, hidden_offset,
    stride_l0, stride_l1, stride_l2, stride_l3,
    stride_r0, stride_r1, stride_r2, stride_r3,
    stride_o0, stride_o1, stride_o2, stride_o3,
    BLOCK_D: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] for a slice of
    the hidden dimension.

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : i64
        Base pointers of the three tensors.
    B, N, hidden_dim, hidden_offset : i32
        Batch size, sequence length, total hidden dimension and the start
        index of the hidden slice processed by this launch.
    stride_* : i32
        Strides of the tensors in the order (batch, i, k, d).
    BLOCK_D, BLOCK_K : constexpr i32
        Tile size for the hidden slice (must divide hidden_dim in calls) and
        the tile size for the reduction dimension `k`.
    """
    batch_id = tl.program_id(0)      # 0‑th axis : batch
    i = tl.program_id(1)             # 1‑st axis : i (row)
    j = tl.program_id(2)             # 2‑nd axis : j (col)

    # guard against out‑of‑bounds launches
    if (batch_id >= B) | (i >= N) | (j >= N):
        return

    # hidden indices for this slice
    d_range = hidden_offset + tl.arange(0, BLOCK_D)
    d_mask = d_range < hidden_dim        # mask for the tail slice

    # accumulator for the hidden slice (float32 for numeric stability)
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    # number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        # k‑block start
        k_base = k_iter * BLOCK_K
        k_off = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_off < N                 # valid k positions

        # -----------------------------------------------------------------
        # Load left[b,i,k,d]   shape (BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr \
            + batch_id * stride_l0 \
            + i * stride_l1 \
            + k_off[:, None] * stride_l2 \
            + d_range[None, :] * stride_l3
        left = tl.load(left_ptrs,
                       mask=k_mask[:, None] & d_mask[None, :],
                       other=0.0).to(tl.float32)

        # Load right[b,j,k,d]  shape (BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr \
            + batch_id * stride_r0 \
            + j * stride_r1 \
            + k_off[:, None] * stride_r2 \
            + d_range[None, :] * stride_r3
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & d_mask[None, :],
                        other=0.0).to(tl.float32)

        # -----------------------------------------------------------------
        # Element‑wise product and reduction over the k‑axis
        acc += tl.sum(left * right, axis=0)   # result: (BLOCK_D,)

    # -----------------------------------------------------------------
    # Store the slice of the output
    out_ptrs = out_ptr \
        + batch_id * stride_o0 \
        + i * stride_o1 \
        + j * stride_o2 \
        + d_range * stride_o3
    tl.store(out_ptrs, acc, mask=d_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module implemented with a
    Triton kernel for the core contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be all‑ones)
        - weights      : dict of named parameters (see description)
        - config       : dict containing `dim`, `hidden_dim`, and optionally
                         `nomask` flag.

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim].
    """
    import math

    # -----------------------------------------------------------------
    # Unpack arguments
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (mask is [B,N,N]; expand to hidden dim later)
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(dtype=dtype)
        mask = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid over a linear projection)
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous memory for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()

    # -----------------------------------------------------------------
    # 5. Triangular multiplicative update via Triton
    out = torch.empty_like(left)   # shape (B,N,N,hidden_dim)

    # Kernel launch configuration
    BLOCK_D = 32   # hidden slice size (must be a compile‑time constant)
    BLOCK_K = 32   # reduction tile along k

    # Strides for the three tensors (batch, i, k, d)
    stride_l0, stride_l1, stride_l2, stride_l3 = left.stride()
    stride_r0, stride_r1, stride_r2, stride_r3 = right.stride()
    stride_o0, stride_o1, stride_o2, stride_o3 = out.stride()

    # Number of hidden slices
    num_hidden_slices = (hidden_dim + BLOCK_D - 1) // BLOCK_D

    # Grid: one program per (batch, i, j)
    grid = (B, N, N)

    for slice_idx in range(num_hidden_slices):
        hidden_offset = slice_idx * BLOCK_D
        # Launch the kernel for this hidden slice
        _trimul_einsum_kernel[grid](
            left.data_ptr(),
            right.data_ptr(),
            out.data_ptr(),
            B, N, hidden_dim, hidden_offset,
            stride_l0, stride_l1, stride_l2, stride_l3,
            stride_r0, stride_r1, stride_r2, stride_r3,
            stride_o0, stride_o1, stride_o2, stride_o3,
            BLOCK_D=BLOCK_D,
            BLOCK_K=BLOCK_K,
        )
        # Synchronize is not required – kernel launch is asynchronous and
        # subsequent PyTorch ops will insert the necessary CUDA stream barriers.

    # -----------------------------------------------------------------
    # 6. Hidden‑dim LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # 7. Output gate
    out = out * out_gate

    # 8. Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
