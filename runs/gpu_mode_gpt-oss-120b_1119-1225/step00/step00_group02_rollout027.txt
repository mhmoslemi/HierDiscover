@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(0)  # batch*hidden index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute actual batch and hidden index
    batch_idx = pid_z // H
    hidden_idx = pid_z % H

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointer base for given batch and hidden
    left_ptrs = left_ptr + batch_idx * stride_left_b + hidden_idx * stride_left_h + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    right_ptrs = right_ptr + batch_idx * stride_right_b + hidden_idx * stride_right_h + offs_n[:, None] * stride_right_j + offs_k[None, :] * stride_right_k

    # note: left ptr index: (i,k) location, shape (M,K)
    # right ptr index: (j,k) location but we need right transpose across k dimension.
    # Our right shape is (b, j, k, h). We want multiplication left[i,k] * right[j,k] (k across). This is left[i, k] * right[j, k] thus both have same k dimension.
    # So we can treat right as (j, k) as is, not transposed.

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # load blocks
        a = tl.load(left_ptrs, mask=(offs_m[:, None] < N) & (offs_k[None, :] < N), other=0.0)
        b = tl.load(right_ptrs, mask=(offs_n[:, None] < N) & (offs_k[None, :] < N), other=0.0)
        # a shape (BLOCK_M, BLOCK_K), b shape (BLOCK_N, BLOCK_K)

        # Compute a @ b.T
        acc += tl.dot(a, b, trans_b=True)  # if supported? Actually tl.dot has a trans_b argument? Not sure.

        # Increment pointers for next k block
        left_ptrs += BLOCK_K * stride_left_k
        right_ptrs += BLOCK_K * stride_right_k

    out_ptrs = out_ptr + batch_idx * stride_out_b + hidden_idx * stride_out_h + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j

def custom_kernel(data):
    x, mask, weights, config = data
    # x shape (B,N,N,dim)
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    
    # For mask handling:
    if config.get('nomask', False):
        mask = None
    else:
        # mask shape (B,N,N)
        mask = mask[..., None]  # shape (B,N,N,1)
    
    # LayerNorm on x
    # using weight and bias
    norm_w = weights['norm.weight']
    norm_b = weights['norm.bias']
    eps = 1e-5
    # compute normalized x
    x_norm = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_w, bias=norm_b, eps=eps
    )
    
    # Perform linear projections
    left_proj_w = weights['left_proj.weight']  # shape (hidden_dim, dim)
    right_proj_w = weights['right_proj.weight']
    
    # torch.nn.functional.linear expects weight shape (out_features, in_features)
    left = torch.nn.functional.linear(x_norm, left_proj_w)  # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_w)
    
    if mask is not None:
        left = left * mask
        right = right * mask
    
    # Gates
    left_gate_w = weights['left_gate.weight']
    right_gate_w = weights['right_gate.weight']
    out_gate_w = weights['out_gate.weight']
    
    left_gate = torch.nn.functional.linear(x_norm, left_gate_w).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_w).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Convert left and right to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    
    # Allocate output buffer
    out_h = torch.empty_like(left_h)  # shape (B,N,N,hidden_dim)
    
    # Compute strides for kernel
    # left_h has strides: (stride_b, stride_i, stride_k, stride_h)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_h.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_h.stride()
    
    # Determine block sizes heuristically
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    
    # grid dims: (B*hidden_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        x.shape[0] * hidden_dim,
        (x.shape[1] + BLOCK_M - 1) // BLOCK_M,
        (x.shape[1] + BLOCK_N - 1) // BLOCK_N,
    )
    
    tri_mul_kernel[grid](
        left_h, right_h, out_h,
        x.shape[0], x.shape[1], hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    
    # Convert result back to float32
    out = out_h.to(torch.float32)
    
    # to_out_norm
    to_out_norm_w = weights['to_out_norm.weight']
    to_out_norm_b = weights['to_out_norm.bias']
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps
    )
    
    # Multiply by out_gate
    out = out * out_gate
    
    # Final linear to output dim
    to_out_w = weights['to_out.weight']  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)
    
    return out

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(0)  # batch * hidden index
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch = pid_z // H
    hid = pid_z % H

    # base pointers for this batch/hidden
    left_base = left_ptr + batch * stride_l_b + hid * stride_l_h
    right_base = right_ptr + batch * stride_r_b + hid * stride_r_h
    out_base = out_ptr + batch * stride_o_b + hid * stride_o_h

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # bounds
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)

        # load
        a = tl.load(left_base + offs_m[:, None] * stride_l_i + offs_k[None, :] * stride_l_k,
                    mask=a_mask, other=0.0)
        b = tl.load(right_base + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_j,
                    mask=b_mask, other=0.0)

        # matmul tile
        acc += tl.dot(a, b)  # a: (M,K), b: (K,N)

    # write result as fp16
    c = acc.to(tl.float16)
    out_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(out_base + offs_m[:, None] * stride_o_i + offs_n[None, :] * stride_o_j,
             c, mask=out_mask)

TriMul forward – “outgoing” update (AlphaFold3)

The kernel implements the cubic contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
where `left` and `right` are the projected (and gated) pairwise
representations.  The computation is a batched matrix‑multiply
for every hidden channel.  We fuse the three‑dimensional GEMM using
a Triton kernel that:
* treats each (batch, hidden) pair as an independent GEMM,
* operates on FP16 values (Tensor‑Core friendly) while accumulating in FP32,
* writes the result back as FP16 (later up‑cast to FP32 for the remaining
  LayerNorm / gating / output linear steps).

All other cheap operations (LayerNorm, linear projections, sigmoids)
are performed with PyTorch.  The overall pipeline matches the reference
PyTorch implementation while using a custom Triton kernel for the
heaviest O(N³·C) part.
"""

# Triton kernel: batched GEMM for the triangular multiplicative update.
# ----------------------------------------------------------------------

    left_ptr,                     # const __half*  (B,N,N,H)  – projected & gated left
    right_ptr,                    # const __half*  (B,N,N,H)  – projected & gated right
    out_ptr,                      # __half*        (B,N,N,H)  – result of the contraction
    B, N, H,                      # tensor dimensions (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # strides for left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # strides for right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # strides for out
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for every hidden channel `h`.  Each (b,h) pair is mapped to a
    separate program (pid_z).  Within each program we tile the `i`
    and `j` dimensions (pid_m, pid_n) and accumulate over `k`.
    """
    pid_z = tl.program_id(0)          # combined batch * hidden index
    pid_m = tl.program_id(1)          # tile over i
    pid_n = tl.program_id(2)          # tile over j

    # --------------------------------------------------------------
    # 1) Decode batch and hidden indices
    # --------------------------------------------------------------
    batch = pid_z // H                # 0 .. B-1
    hid   = pid_z %  H                # 0 .. H-1

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_l_b + hid * stride_l_h
    right_base = right_ptr + batch * stride_r_b + hid * stride_r_h
    out_base   = out_ptr   + batch * stride_o_b + hid * stride_o_h

    # --------------------------------------------------------------
    # 2) Tile offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32 (Tensor‑Core accumulation)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m[:, None] < N) & (offs_k[None, :] < N)   # (M, K)
        b_mask = (offs_k[:, None] < N) & (offs_n[None, :] < N)   # (K, N)

        # Load A tile: shape (M, K)
        a = tl.load(
            left_base
            + offs_m[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k,
            mask=a_mask,
            other=0.0,
        )   # FP16 --> will be promoted to FP32 for the dot

        # Load B tile: we need right[j,k] --> load as (K, N) = (k, j)
        b = tl.load(
            right_base
            + offs_k[:, None] * stride_r_k
            + offs_n[None, :] * stride_r_j,
            mask=b_mask,
            other=0.0,
        )   # FP16

        # GEMM tile: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(a, b)   # accumulator stays FP32

    # --------------------------------------------------------------
    # 4) Write back the result (store as FP16)
    # --------------------------------------------------------------
    out_mask = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    c = acc.to(tl.float16)
    tl.store(
        out_base
        + offs_m[:, None] * stride_o_i
        + offs_n[None, :] * stride_o_j,
        c,
        mask=out_mask,
    )

# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul update using a fused Triton kernel
    for the O(N³·C) contraction.

    Args:
        data: Tuple (input_tensor, mask_tensor, weight_dict, config_dict)
              - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
              - mask_tensor : torch.Tensor of shape [B, N, N] (float32/bool) or None
              - weight_dict : mapping of parameter names to torch.Tensor
              - config_dict : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0) unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm (LayerNorm over the last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    eps = 1e-5
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (float32 → later cast to fp16)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional masking (mask shape [B,N,N] → broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear transforms of the *normed* input)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the Triton kernel
    # ------------------------------------------------------------------
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output buffer (FP16)
    out_h = torch.empty_like(left_h)   # shape [B,N,N,hidden_dim], dtype=fp16

    # ------------------------------------------------------------------
    # 6) Prepare stride arguments (int64)
    # ------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_h.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_h.stride()

    # ------------------------------------------------------------------
    # 7) Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (batch*hidden, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    grid = (
        x.shape[0] * hidden_dim,
        (x.shape[1] + BLOCK_M - 1) // BLOCK_M,
        (x.shape[1] + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left_h, right_h, out_h,
        x.shape[0], x.shape[1], hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the remaining operations
    out = out_h.to(torch.float32)   # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 8) Post‑multiplicative update: Norm → gate → final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]

    out = F.layer_norm(
        out, (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # Apply the output gate (element‑wise)
    out = out * out_gate

    # Final projection back to `dim`
    to_out_weight = weights["to_out.weight"]       # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)            # [B,N,N,dim]

    return out
