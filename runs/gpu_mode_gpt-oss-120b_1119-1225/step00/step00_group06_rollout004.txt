@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    num_blocks_d = tl.cdiv(D, BLOCK_D)
    total_per_batch = num_blocks_m * num_blocks_n * num_blocks_d
    b = pid // total_per_batch
    pid_rest = pid % total_per_batch
    block_d = pid_rest // (num_blocks_m * num_blocks_n)
    pid_rem = pid_rest % (num_blocks_m * num_blocks_n)
    block_m = pid_rem // num_blocks_n
    block_n = pid_rem % num_blocks_n

    # Compute actual indices
    i = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    d = block_d * BLOCK_D + tl.arange(0, BLOCK_D)

    # Create masks for out-of-range
    mask_i = i < N
    mask_j = j < N
    mask_d = d < D

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_range = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left slice: shape (M, K, D)
        left_ptrs = left_ptr + b * stride_left_b \
            + i[:, None, None] * stride_left_i \
            + k_range[None, :, None] * stride_left_k \
            + d[None, None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0, dtype=tl.float16)

        # Load right slice as (K, N, D)
        right_ptrs = right_ptr + b * stride_right_b \
            + k_range[:, None, None] * stride_right_k \
            + j[None, :, None] * stride_right_j \
            + d[None, None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :], other=0.0, dtype=tl.float16)

        # Compute per d
        for di in range(0, BLOCK_D):
            left_slice = left[:, :, di]   # (M, K)
            right_slice = right[:, :, di] # (K, N)
            # cast to float32 for dot
            prod = tl.dot(left_slice.to(tl.float32), right_slice.to(tl.float32))
            acc[:, :, di] = acc[:, :, di] + prod
    # Store the result
    out_ptrs = out_ptr + b * stride_out_b \
        + i[:, None, None] * stride_out_i \
        + j[None, :, None] * stride_out_j \
        + d[None, None, :] * stride_out_d
    mask_store = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_store)
```

"""
TriMul (outgoing) implementation with a Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = `dim`).
2. Linear projections (no bias):
   * left  = x @ W_left   -> [B, N, N, H]
   * right = x @ W_right  -> [B, N, N, H]
3. Optional mask (shape [B,N,N]) is applied to the projected tensors.
4. Gated projections:
   * left  *= sigmoid(x @ W_left_gate )
   * right *= sigmoid(x @ W_right_gate)
5. Core contraction (the N³ work) is performed by a Triton kernel:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   (i.e. a batch of H independent matrix‑multiplications:
    left_h @ right_hᵀ )
   The kernel loads `left` in (M×K) tiles and `right` in (K×N) tiles,
   casts the half‑precision loads to fp32, executes `tl.dot` for each
   hidden‑channel slice and accumulates the result in fp32.
6. Layer‑norm on the hidden dimension (`to_out_norm`), gating by
   `out_gate = sigmoid(x @ W_out_gate)`, and final linear projection
   back to `dim`.

The heavy O(B·N³·H) work is thus executed on the GPU in a fused
kernel, while the surrounding linear / norm / sigmoid operations are
performed with regular PyTorch (which is cheap in comparison).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr,            # const* half
    right_ptr,           # const* half
    out_ptr,             # * float
    B, N, H,             # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # Compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Batched N×N matrix‑multiplication for every hidden channel.

    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    left  : (B, N, N, H)   indexed as (b, i, k, h)
    right : (B, N, N, H)   indexed as (b, j, k, h)
    """
    pid = tl.program_id(0)

    # ---- program layout ----------------------------------------------------
    # 1‑D grid: each program computes one (b, h_tile, i_tile, j_tile)
    num_i = tl.cdiv(N, BLOCK_M)
    num_j = tl.cdiv(N, BLOCK_N)
    num_h = tl.cdiv(H, BLOCK_H)

    programs_per_batch = num_i * num_j * num_h
    b = pid // programs_per_batch
    pid_rem = pid % programs_per_batch

    h_tile = pid_rem // (num_i * num_j)
    pid_rem2 = pid_rem % (num_i * num_j)
    i_tile = pid_rem2 // num_j
    j_tile = pid_rem2 % num_j

    # ---- tile coordinates --------------------------------------------------
    i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    h = h_tile * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ---- reduction over k --------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_range = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # left[b, i, k, h] -> (M, K, H)
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i[:, None, None] * stride_left_i
            + k_range[None, :, None] * stride_left_k
            + h[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # right[b, j, k, h] required as (K, N, H)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + k_range[:, None, None] * stride_right_k
            + j[None, :, None] * stride_right_i   # `i` stride indexes the second axis (=j here)
            + h[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ---- compute dot for each hidden slice --------------------------------
        for dh in range(0, BLOCK_H):
            # (M, K)  *  (K, N)  -> (M, N)
            a = left[:, :, dh].to(tl.float32)
            bmat = right[:, :, dh].to(tl.float32)
            prod = tl.dot(a, bmat)          # default: (M,K) @ (K,N)
            acc[:, :, dh] = acc[:, :, dh] + prod

    # ---- write back ---------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i[:, None, None] * stride_out_i
        + j[None, :, None] * stride_out_j
        + h[None, None, :] * stride_out_h
    )
    mask_store = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_store)


def custom_kernel(data):
    """
    TriMul forward (outgoing) – wrapper that prepares tensors,
    launches the Triton kernel for the N³ contraction,
    and applies the remaining layer‑norm / gating / linear steps.
    """
    # Unpack inputs -----------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # called H in the kernel

    # -------------------------------------------------------------------------
    # 1. Input layer‑norm (dim‑wise)
    # -------------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, dim]

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,H]

    # -------------------------------------------------------------------------
    # 3. Optional masking (mask shape [B,N,N])
    # -------------------------------------------------------------------------
    if mask is not None:
        # mask may be bool; convert to float for multiplication
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections on the *original* normed x)
    # -------------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Core contraction using Triton
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to fp16 for bandwidth‑limited loading, keep accumulator fp32
    left_fp16 = left.contiguous().to(torch.float16)
    right_fp16 = right.contiguous().to(torch.float16)

    # Output buffer
    out = torch.empty((B, N, N, hidden_dim), device=left.device, dtype=torch.float32)

    # Strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tile sizes – chosen to fit H100 shared memory while keeping enough parallelism
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8   # number of hidden channels per program

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_h = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    total_programs = B * num_i * num_j * num_h

    grid = (total_programs,)

    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # -------------------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim layer‑norm, gating, final linear
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate   # gating after hidden‑norm

    # Final projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
