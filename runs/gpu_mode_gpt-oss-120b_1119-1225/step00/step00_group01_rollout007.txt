"""
custom_kernel:  Outgoing TriMul (Triangular Multiplicative Update) implementation.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ (B,N,N,dim) using the provided
   `norm.weight` / `norm.bias`.
2. Linear projections (no bias) to a hidden dimension H:
      left  = x @ left_proj.weight.T
      right = x @ right_proj.weight.T
3. Compute three sigmoid gates from the same normalized x:
      left_gate  = σ(x @ left_gate.weight.T)
      right_gate = σ(x @ right_gate.weight.T)
      out_gate   = σ(x @ out_gate.weight.T)
4. Fuse mask (if present) and the two per‑position gates onto *left*
   and *right* in‑place with a tiny Triton kernel – this removes an
   extra memory pass and keeps the computation on‑device.
5. Compute the core “tri‑multiplicative” update:
      out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
   which is exactly a batch of matrix‑multiplications
      out_d = left_d @ right_dᵀ .
   We reshape to (B·H, N, N) and use a half‑precision torch.bmm
   (cuBLAS) for peak performance, then cast the result back to fp32.
6. Apply a second LayerNorm (`to_out_norm`) on the hidden dimension,
   multiply by `out_gate`, and finally a linear projection back to `dim`
   (no bias) using `to_out.weight`.
7. Return the tensor of shape (B,N,N,dim).

The Triton kernel only handles the mask‑gate fusion; everything else
uses highly‑optimised PyTorch/CuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: in‑place mask + gate fusion
# --------------------------------------------------------------------------- #
@triton.jit
def apply_mask_gate_kernel(
    left_ptr,               # float32 * (B,N,N,H)
    right_ptr,              # float32 * (B,N,N,H)
    mask_ptr,               # float32 * (B,N,N)   (flattened)
    left_gate_ptr,          # float32 * (B,N,N,H)
    right_gate_ptr,         # float32 * (B,N,N,H)
    total_elements,         # int32   (B*N*N*H)
    hidden_dim,             # int32   = H
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place:
         left  = left  * mask * left_gate
         right = right * mask * right_gate
    mask is broadcast over the hidden dimension.
    """
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)            # [0, BLOCK_SIZE)
    valid = offs < total_elements

    # Index of the corresponding mask element (mask repeats every `hidden_dim` entries)
    mask_index = offs // hidden_dim

    # Load
    left      = tl.load(left_ptr      + offs,          mask=valid, other=0.0)
    right     = tl.load(right_ptr     + offs,          mask=valid, other=0.0)
    mask_val  = tl.load(mask_ptr      + mask_index,   mask=valid, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs,          mask=valid, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs,          mask=valid, other=0.0)

    # Fuse
    left  = left  * mask_val * left_gate
    right = right * mask_val * right_gate

    # Store back in‑place
    tl.store(left_ptr  + offs, left,  mask=valid)
    tl.store(right_ptr + offs, right, mask=valid)


# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor : torch.Tensor[B, N, N, dim] (float32, on GPU)
              - mask_tensor  : torch.Tensor[B, N, N] or None
              - weights_dict : dict of torch.Tensor containing all model weights
              - config_dict  : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on GPU)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    x, mask, weights, config = data
    device = x.device
    dtype  = x.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # ------------------------------------------------------------------- #
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # ------------------------------------------------------------------- #
    # Linear projections (no bias): shape -> (B,N,N,hidden_dim)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, weights["left_proj.weight"])

    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------- #
    # Apply mask + gates in‑place using Triton (if mask is present)
    # ------------------------------------------------------------------- #
    B, N, N2, _ = left.shape
    assert N == N2, "Expected square pairwise tensor"

    if mask is None or nomask:
        # No mask: just multiply by the two per‑position gates
        left  = left  * left_gate
        right = right * right_gate
    else:
        # Ensure mask is contiguous and flatten it for the kernel
        mask = mask.to(dtype).contiguous()
        mask_flat = mask.view(-1)                     # shape (B*N*N,)

        # Triton launch configuration
        total_elems = left.numel()                    # B * N * N * H
        BLOCK = 1024
        grid = lambda meta: (triton.cdiv(total_elems, meta["BLOCK_SIZE"]),)

        # In‑place fusion kernel
        apply_mask_gate_kernel[grid](
            left, right,
            mask_flat,
            left_gate, right_gate,
            total_elems,
            hidden_dim,
            BLOCK_SIZE=BLOCK,
        )
        # At this point left/right already carry the mask and gates

    # ------------------------------------------------------------------- #
    # Core TriMul: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # This is a batched matrix‑multiply over the hidden dimension.
    # ------------------------------------------------------------------- #
    # Permute to (B, H, N, N) → (B*H, N, N) and use fp16 bmm for speed
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Cast to half‑precision for the heavy matmul
    left_fp16  = left_perm.to(torch.float16)
    right_fp16 = right_perm.to(torch.float16)

    # Reshape to (B*H, N, N)
    BH = B * hidden_dim
    left_mat  = left_fp16.reshape(BH, N, N)
    right_mat = right_fp16.reshape(BH, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N)^T  → (B*H, N, N)
    out_fp16 = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout (B, N, N, H) and cast back to fp32
    out = out_fp16.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------- #
    # Post‑processing: LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
```
