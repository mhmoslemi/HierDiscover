def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)
   pid_b = tl.program_id(2)
   
   i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
   j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
   d_base = tl.arange(0, BLOCK_D)
   
   # mask out-of-bounds
   i_mask = i_base < N
   j_mask = j_base < N
   d_mask = d_base < D

   # broadcast masks to shapes needed later
   # allocate accumulator
   acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

   # iterate over K dimension
   for k_tile in range(tl.cdiv(N, BLOCK_K)):
       k_base = k_tile * BLOCK_K
       k_range = tl.arange(0, BLOCK_K)
       k_mask = (k_base + k_range) < N

       # load left tile
       left_ptrs = left_ptr + \
           pid_b * stride_left_b + \
           i_base[:, None, None] * stride_left_i + \
           (k_base + k_range)[None, :, None] * stride_left_k + \
           d_base[None, None, :] * stride_left_d
       left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

           pid_b * stride_right_b + \
           j_base[:, None, None] * stride_right_j + \

           d_base[None, None, :] * stride_right_d

       # Loop over k within tile
       for kk in range(0, BLOCK_K):
           # Check if this kk is valid; if not, skip (values will be zero anyway)
           # left slice shape (BLOCK_I, BLOCK_D)
           left_slice = left[:, kk, :]  # shape (BLOCK_I, BLOCK_D)
           right_slice = right[:, kk, :] # shape (BLOCK_J, BLOCK_D)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]"""
    pid_i = tl.program_id(0)  # block index i
    pid_j = tl.program_id(1)  # block index j
    pid_b = tl.program_id(2)  # batch index

    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    d_base = tl.arange(0, BLOCK_D)

    i_valid = i_base < N
    j_valid = j_base < N
    d_valid = d_base < D

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over k dimension in tiles
    k = 0
    while k < N:
        # Build k indices for this tile
        k_range = tl.arange(0, BLOCK_K)
        k_idx = k + k_range                     # shape (BLOCK_K,)
        k_valid = k_idx < N

        # Pointers for the current tile
        # left: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + \
            pid_b * stride_left_b + \
            i_base[:, None, None] * stride_left_i + \
            k_idx[None, :, None] * stride_left_k + \
            d_base[None, None, :] * stride_left_d

        left = tl.load(left_ptrs,
                       mask=i_valid[:, None, None] & k_valid[None, :, None] & d_valid[None, None, :],
                       other=0.0)

        # right: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + \
            pid_b * stride_right_b + \
            j_base[:, None, None] * stride_right_j + \
            k_idx[None, :, None] * stride_right_k + \
            d_base[None, None, :] * stride_right_d

        right = tl.load(right_ptrs,
                        mask=j_valid[:, None, None] & k_valid[None, :, None] & d_valid[None, None, :],
                        other=0.0)

        # accumulate outer products for each k in the tile
        # Unroll the inner k loop (static)
        for kk in range(0, BLOCK_K):
            # left_slice: (BLOCK_I, BLOCK_D)
            left_slice = left[:, kk, :]   # shape (I, D)
            # right_slice: (BLOCK_J, BLOCK_D)
            right_slice = right[:, kk, :] # shape (J, D)

            # broadcast to (I, J, D) and accumulate
            # left_slice[:, None, :] has shape (I,1,D)
            # right_slice[None, :, :] has shape (1,J,D)
            acc += left_slice[:, None, :] * right_slice[None, :, :]

        # advance k
        k += BLOCK_K

    # write results
    out_ptrs = out_ptr + \
        pid_b * stride_out_b + \
        i_base[:, None, None] * stride_out_i + \
        j_base[None, :, None] * stride_out_j + \
        d_base[None, None, :] * stride_out_d

    tl.store(out_ptrs,
             acc,
             mask=i_valid[:, None, None] & j_valid[None, :, None] & d_valid[None, None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.
    This function reproduces the PyTorch reference using a fused
    Triton kernel for the heavy ``einsum``:
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    The surrounding LayerNorm and Linear projections are performed in
    ordinary PyTorch for simplicity.
    """
    # Unpack the tuple
    input_tensor, mask_tensor, weights, config = data
    # config fields
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # Extract weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 16  # must divide hidden_dim for best performance, otherwise we mask the tail

    # Number of blocks along each dimension
    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    # Loop over the hidden dimension in chunks of BLOCK_D
    for d_offset in range(0, hidden_dim, BLOCK_D):
        cur_D = min(BLOCK_D, hidden_dim - d_offset)
        # Slice tensors to the current D‑block
        left_slice = left[..., d_offset:d_offset+cur_D]
        right_slice = right[..., d_offset:d_offset+cur_D]
        out_slice = out[..., d_offset:d_offset+cur_D]

        # Launch the Triton kernel
        grid = (num_i, num_j, B)
        tri_mul_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            B, N, cur_D,
            stride_left_b, stride_left_i, stride_left_k, stride_left_d,
            stride_right_b, stride_right_j, stride_right_k, stride_right_d,
            stride_out_b, stride_out_i, stride_out_j, stride_out_d,
            BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
            num_warps=8
        )

    # Post‑processing: LayerNorm, gating, final linear
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # shape [B, N, N, dim]

    return out

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    The kernel iterates over the K‑dimension in tiles and accumulates
    outer‑products for each d‑channel.
    """
    pid_i = tl.program_id(0)      # block index for i
    pid_j = tl.program_id(1)      # block index for j
    pid_b = tl.program_id(2)      # batch index

    # Base indices for this block
    i_base = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_base = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    d_base = tl.arange(0, BLOCK_D)

    i_mask = i_base < N
    j_mask = j_base < N
    d_mask = d_base < D

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # -----------------------------------------------------------------
    k = 0
    while k < N:
        k_range = tl.arange(0, BLOCK_K)
        k_idx   = k + k_range                     # absolute k indices
        k_mask  = k_idx < N

        # -----------------------------------------------------------------
        # Load a tile of left:   shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # -----------------------------------------------------------------
        left_ptrs = left_ptr + \
            pid_b * stride_left_b + \
            i_base[:, None, None] * stride_left_i + \
            k_idx[None, :, None] * stride_left_k + \
            d_base[None, None, :] * stride_left_d

        left = tl.load(
            left_ptrs,
            mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        )

        # -----------------------------------------------------------------
        # Load a tile of right:  shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # -----------------------------------------------------------------
        right_ptrs = right_ptr + \
            pid_b * stride_right_b + \
            j_base[:, None, None] * stride_right_j + \
            k_idx[None, :, None] * stride_right_k + \
            d_base[None, None, :] * stride_right_d

        right = tl.load(
            right_ptrs,
            mask=j_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
            other=0.0,
        )

        # -----------------------------------------------------------------
        # Accumulate outer‑products for each k in the tile
        # -----------------------------------------------------------------
        for kk in range(0, BLOCK_K):
            # left_slice : (BLOCK_I, BLOCK_D)
            # right_slice: (BLOCK_J, BLOCK_D)
            left_slice  = left[:, kk, :]          # (I, D)
            right_slice = right[:, kk, :]         # (J, D)

            # Broadcast to (I, J, D) -> outer product over i and j for each d
            acc += left_slice[:, None, :] * right_slice[None, :, :]

        k += BLOCK_K

    # -----------------------------------------------------------------
    # Store the result
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + \
        pid_b * stride_out_b + \
        i_base[:, None, None] * stride_out_i + \
        j_base[None, :, None] * stride_out_j + \
        d_base[None, None, :] * stride_out_d

    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :],
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.
    The heavy “einsum” is computed by a custom Triton kernel; the
    surrounding LayerNorms, Linear projections and gating follow the
    reference PyTorch implementation.
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # Load model weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # LayerNorm over the feature dimension
    # --------------------------------------------------------------
    x = input_tensor.contiguous()
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------
    # Linear projections (bias‑free)
    # --------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # Optional mask
    # --------------------------------------------------------------
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # Compute gates (sigmoids of linear projections)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Triton kernel for the core multiplicative update
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # Strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Block sizes (tuned for H100, can be adjusted)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 16                     # must divide hidden_dim or be masked for the tail

    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J

    # Loop over the hidden dimension in chunks of BLOCK_D
    for d_offset in range(0, hidden_dim, BLOCK_D):
        cur_D = min(BLOCK_D, hidden_dim - d_offset)

        # Views that start at the current D‑slice (same strides)
        left_slice = left[..., d_offset:d_offset + cur_D]
        right_slice = right[..., d_offset:d_offset + cur_D]
        out_slice = out[..., d_offset:d_offset + cur_D]

        grid = (num_i, num_j, B)
        tri_mul_kernel[grid](
            left_slice,
            right_slice,
            out_slice,
            B, N, cur_D,
            stride_left_b, stride_left_i, stride_left_k, stride_left_d,
            stride_right_b, stride_right_j, stride_right_k, stride_right_d,
            stride_out_b, stride_out_i, stride_out_j, stride_out_d,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
            num_warps=8,
        )

    # --------------------------------------------------------------
    # Post‑processing: normalisation, output gating and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
