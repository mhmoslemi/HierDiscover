    hidden_dim = config["hidden_dim"]

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    right_proj_weight = weights["right_proj.weight"]

        left = left * mask
        right = right * mask

    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()

    right = right * right_gate

"""
custom_kernel for the “outgoing” TriMul operator (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) → left, right ∈ ℝ^{B×N×N×H}.
3. (Optional) mask the two projections.
4. Compute three gates (sigmoid of linear projections of x) and
   multiply left/right by their respective gates.
5. The core O(N³) reduction:
        out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]
   This is exactly a batched matrix‑multiplication:
        left reshaped to A ∈ ℝ^{(B·H)×N×N},   right transposed to B ∈ ℝ^{(B·H)×N×N}
        out_flat = A @ Bᵀ  (batch‑wise GEMM).
   A custom Triton kernel performs this batched GEMM in FP16
   (accumulating in FP32) to exploit tensor cores on an H100.
6. Reshape the result back to (B,N,N,H) and cast to FP32.
7. Layer‑norm over the hidden dimension H, multiply by the
   output‑gate, and apply a final linear projection to the original
   channel dimension (dim).

Only the O(N³) reduction is done in Triton; the remaining
operations are expressed with highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (B, M, K)   – row‑major
#   B : (B, K, N)   – row‑major (already transposed w.r.t. original tensor)
#   C : (B, M, N)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    batch, M, N, K,                           # dimensions
    stride_ab, stride_am, stride_ak,          # strides of A (batch, row, col)
    stride_bb, stride_bk, stride_bn,          # strides of B (batch, row, col)
    stride_cb, stride_cm, stride_cn,          # strides of C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(axis=2)   # batch dimension
    pid_m = tl.program_id(axis=0)   # row block
    pid_n = tl.program_id(axis=1)   # col block

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the block in A and B
    a_ptrs = a_ptr + pid_b * stride_ab \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K tiles
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load with bounds checking
        a = tl.load(a_ptrs,
                    mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
                    other=0.0)

        # Convert to fp16 for the dot; accumulate in fp32
        a = a.to(tl.float16)
        b = b.to(tl.float16)
        acc += tl.dot(a, b)

        # Advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back the result (store as FP16, later cast to FP32 in Python)
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + pid_b * stride_cb \
                     + offs_m[:, None] * stride_cm \
                     + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             c,
             mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Perform a batched GEMM A @ Bᵀ where a.shape = (B, M, K) and b.shape = (B, K, N).
    Both inputs are expected to be contiguous FP16 tensors.
    Returns a FP16 tensor of shape (B, M, N).
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, "Inner dimensions must match"

    # Allocate output
    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)

    # Tunable block sizes – chosen to give good occupancy on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N), batch)

    _batched_matmul_kernel[grid](
        a, b, c,
        batch, M, N, K,
        a.stride(0), a.stride(1), a.stride(2),
        b.stride(0), b.stride(1), b.stride(2),
        c.stride(0), c.stride(1), c.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (optional, may be None)
        - weights      : dict of torch.Tensors containing all model weights
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the last dim (C)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    # Compute mean & variance over the channel dimension
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias                       # shape [B,N,N,C]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)

    left  = F.linear(x, left_proj_weight)   # [B,N,N,H]
    right = F.linear(x, right_proj_weight)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask is [B,N,N]; broadcast on the hidden dimension
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³) reduction using a fused Triton GEMM
    #    left  : [B,N,N,H] → (B, H, N, N) → reshape → (B*H, N, N)
    #    right : [B,N,N,H] → (B, H, N, N) with (k,i) order → reshape → (B*H, N, N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Permute & reshape for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k,j)

    # Collapse batch and hidden dimensions
    left_flat  = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N)  # (B*H, N, N) where dim1 = K, dim2 = J

    # Cast to FP16 for Tensor‑Core acceleration
    left_flat_h  = left_flat.to(torch.float16)
    right_flat_h = right_flat.to(torch.float16)

    # Batched matmul C = A @ Bᵀ  (B*H, N, N) @ (B*H, N, N).transpose(-2,-1)
    # Since right_flat already has the required (K,J) orientation, we can feed it directly.
    out_flat_h = _batched_matmul_triton(left_flat_h, right_flat_h)  # FP16 result

    # Cast back to FP32 and restore original shape
    out = out_flat_h.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1)  # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) * torch.rsqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ------------------------------------------------------------------
    # 7) Apply output gate and final linear projection to original dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    to_out_weight = weights["to_out.weight"]   # (dim, hidden_dim)
    out_final = F.linear(out_norm, to_out_weight)  # [B,N,N,dim]

    return out_final
