@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    b = pid_bh // H
    h = pid_bh % H

    # compute offsets
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Create masks for out-of-range indices
    mask_i = i < N
    mask_j = j < N

    # Accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_offset = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # pointers for left [i, k]
        left_ptrs = left_ptr + b*stride_left_b + i[:, None] * stride_left_i + k_offset[None, :] * stride_left_k + h * stride_left_h
        left_vals = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # pointers for right_transposed [k, j]
        right_ptrs = right_ptr + b*stride_right_b + k_offset[:, None] * stride_right_k + j[None, :] * stride_right_j + h * stride_right_h
        right_vals = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # matrix multiplication accumulation
        acc += tl.dot(left_vals, right_vals)   # left_vals (BLOCK_I, BLOCK_K), right_vals (BLOCK_K, BLOCK_J)

    # store result
    out_ptrs = out_ptr + b*stride_out_b + i[:, None] * stride_out_i + j[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul custom kernel (AlphaFold3 “outgoing” variant).

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension (dim).
2. Project the normalized tensor to a hidden dimension `h` with two
   linear layers (`left_proj` and `right_proj`).
3. Apply an optional (binary) mask and multiplicative gates
   (`left_gate`, `right_gate`, `out_gate`).
4. Compute the core “triangle‑multiplicative” update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply per hidden channel:
   for each (b,h) we compute L(b,h) @ R(b,h).T.
   The operation is fused in a Triton kernel that tiles the i‑ and
   j‑dimensions (BLOCK_I×BLOCK_J) and iterates over k in BLOCK_K.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   `out_gate`, and finally project back to the original feature dimension
   with a linear layer (`to_out`).

Only the heavy O(N³·h) multiplication is performed in Triton; all other
operations use PyTorch functions for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the triangle‑multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # const pointer to left tensor   [B,N,N,H]
    right_ptr,              # const pointer to right tensor  [B,N,N,H]
    out_ptr,                #   pointer to output tensor     [B,N,N,H]
    B, N, H,                # shape parameters
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)                # tile index over i (row)
    pid_j = tl.program_id(1)                # tile index over j (col)
    pid_bh = tl.program_id(2)               # combined batch * hidden index

    # decode batch and hidden channel from the 3rd dimension
    b = pid_bh // H
    h = pid_bh % H

    # row / column offsets for this tile
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i < N
    mask_j = j < N

    # accumulator for the tile
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension in blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k_blocks):
        k_base = k_iter * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ----- load left[b, i, k, h] -----
        left_off = (
            b * stride_left_b +
            i[:, None] * stride_left_i +
            k[None, :] * stride_left_k +
            h * stride_left_h
        )
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )  # (BLOCK_I, BLOCK_K)  float32

        # ----- load right transposed: right[b, k, j, h] -----
        right_off = (
            b * stride_right_b +
            k[:, None] * stride_right_k +
            j[None, :] * stride_right_j +
            h * stride_right_h
        )
        right_t = tl.load(
            right_ptr + right_off,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )  # (BLOCK_K, BLOCK_J)  float32

        # matrix multiplication for this k‑block
        acc += tl.dot(left, right_t)   # (BLOCK_I, BLOCK_J)

    # ----- store the result tile -----
    out_off = (
        b * stride_out_b +
        i[:, None] * stride_out_i +
        j[None, :] * stride_out_j +
        h * stride_out_h
    )
    tl.store(
        out_ptr + out_off,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward pass (outgoing version) using a fused Triton kernel
    for the heavy O(N³·h) multiplication.

    Parameters
    ----------
    data: tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        * mask_tensor  : torch.Tensor of shape [B, N, N] (float32 or bool)
        * weights_dict : dict of torch.Tensor model parameters
        * config_dict  : dict containing at least "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Helper: LayerNorm over last dimension
    # ------------------------------------------------------------------
    def _layer_norm(x, weight, bias, eps=1e-5):
        # x: [..., C]   weight/bias: (C,)
        return F.layer_norm(x, x.shape[-1:], weight=weight, bias=bias, eps=eps)

    # ------------------------------------------------------------------
    # 1. Input normalization
    # ------------------------------------------------------------------
    x = _layer_norm(input_tensor, norm_weight, norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3. Mask handling
    # ------------------------------------------------------------------
    if mask is None:
        # No masking => use ones
        mask = torch.ones_like(x[..., 0], dtype=dtype, device=device)
    # Ensure mask is float (same dtype as projected tensors)
    mask = mask.to(dtype).unsqueeze(-1)    # shape [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triangular multiplicative update via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B,N,N,hidden_dim
    H = hidden_dim

    # Allocate output tensor
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Prepare strides (in element units, not bytes)
    # left, right, out are contiguous in memory (default torch layout)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tiling parameters – can be tuned; these work well on H100 for the target sizes
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    # Grid dimensions
    grid_i = triton.cdiv(N, BLOCK_I)
    grid_j = triton.cdiv(N, BLOCK_J)
    grid_bh = B * H   # each (batch, hidden) pair gets its own program instance

    # Launch kernel
    trimul_kernel[
        (grid_i, grid_j, grid_bh)
    ](
        left, right, out,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_I, BLOCK_J, BLOCK_K,
        num_warps=4  # a modest number of warps; H100 handles this well
    )

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = _layer_norm(out, to_out_norm_weight, to_out_norm_bias)

    # ------------------------------------------------------------------
    # 7. Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate   # broadcasting over hidden dim

    # ------------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B,N,N,dim)

    return out
