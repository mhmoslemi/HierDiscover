@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    EPS: tl.constexpr = 1e-12):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute the start indices.
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Create masks for out-of-bounds.
    offset_i = i0 + tl.arange(0, BLOCK_M)
    offset_j = j0 + tl.arange(0, BLOCK_N)

    # Loop over hidden blocks.
    for h0 in range(0, H, BLOCK_H):
        # Initialize accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over k dimension.
        for k in range(0, N, BLOCK_K):
            # Offsets
            offset_k = k + tl.arange(0, BLOCK_K)

            # Load left: shape (BLOCK_M, BLOCK_K)
            left_ptrs = left_ptr + pid_b * stride_left_b \
                + offset_i[:, None] * stride_left_i \
                + offset_k[None, :] * stride_left_k \
                + h0 * stride_left_h

            # Load right: shape (BLOCK_N, BLOCK_K)
            right_ptrs = right_ptr + pid_b * stride_right_b \
                + offset_j[:, None] * stride_right_i \
                + offset_k[None, :] * stride_right_k \
                + h0 * stride_right_h

            left = tl.load(left_ptrs, mask=(offset_i < N)[:, None] & (offset_k < N)[None, :], other=0.0).to(tl.float32)
            right = tl.load(right_ptrs, mask=(offset_j < N)[:, None] & (offset_k < N)[None, :], other=0.0).to(tl.float32)

            # Compute dot: left (M x K) dot right^T (N x K) => (M,N)
            acc += tl.dot(left, right, trans_b=True)  # right transposed

        # Store accumulator into out
        out_ptrs = out_ptr + pid_b * stride_out_b \
            + offset_i[:, None] * stride_out_i \
            + offset_j[None, :] * stride_out_j \
            + h0 * stride_out_h

        tl.store(out_ptrs, acc.to(tl.float16) if BLOCK_H>0 else acc, mask=(offset_i < N)[:, None] & (offset_j < N)[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # x_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    # Apply mask if present:
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to float16 for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output
    out_h = torch.empty_like(left_h)  # shape [B,N,N,hidden_dim], dtype float16
    # Compute grid dims
    B, N, _, H = left_h.shape
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 32  # hidden block size
    # Compute grid sizes: 
    grid = (B, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)
    # Actually we need kernel ids: (B, num_i_tiles, num_j_tiles)
    # We'll pass other arguments.

    trimul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        H,
        left_h.stride(0), left_h.stride(1), left_h.stride(2), left_h.stride(3),
        right_h.stride(0), right_h.stride(1), right_h.stride(2), right_h.stride(3),
        out_h.stride(0), out_h.stride(1), out_h.stride(2), out_h.stride(3),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
    )
    # Convert out to float32 for further ops.
    out = out_h.to(torch.float32)
    # LayerNorm on hidden dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # Multiply by out_gate
    out = out * out_gate
    # Final linear to dim

def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)  # combined batch and channel
    pid_m = tl.program_id(1)  # tile in M dimension (i)
    pid_n = tl.program_id(2)  # tile in N dimension (j)

    # compute b and d
    total_hd = B * H
    b = pid // H
    d = pid % H

    # compute start coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        # compute pointers + offsets for left and right
        left_ptrs = left_ptr + b*stride_left_b + d*stride_left_h + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        right_ptrs = right_ptr + b*stride_right_b + d*stride_right_h + offs_n[:, None] * stride_right_i + offs_k[None, :] * stride_right_k

        # load tiles
        a = tl.load(left_ptrs, mask=(offs_m < N)[:, None] & (offs_k < N)[None, :], other=0.0).to(tl.float32)
        b_ = tl.load(right_ptrs, mask=(offs_n < N)[:, None] & (offs_k < N)[None, :], other=0.0).to(tl.float32)

        # dot: a (M,K) @ b_.T (N,K) (since we need sum_k a * b)
        acc += tl.dot(a, b_, trans_b=True)
    # Store result
    out_ptrs = out_ptr + b*stride_out_b + d*stride_out_h + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=(offs_m < N)[:, None] & (offs_n < N)[None, :])

"""
TriMul (outgoing) fused implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]).
2. Linear projections (`left_proj`, `right_proj`) and gating
   (`left_gate`, `right_gate`, `out_gate`) are performed with
   `torch.nn.functional.linear`.  Gates are passed through `sigmoid`.
3. Optional pair‑wise mask is applied and the gates are multiplied in‑place:
       left  = left_proj  * left_gate  * mask
       right = right_proj * right_gate * mask
   The resulting tensors have shape [B,N,N,H] (H = hidden_dim).
4. The costly Einstein‑sum
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is executed by a custom Triton kernel.  The kernel treats each
   (batch, hidden) pair as an independent matrix‑multiply:
       out_{b,h} = left_{b,h} @ right_{b,h}ᵀ
   where each matrix is N×N.  This yields an output of shape
   [B,N,N,H] in fp16 (accumulated in fp32 for precision).
5. The result is cast back to fp32, layer‑normed (`to_out_norm`),
   multiplied by the output gate and finally projected back to the
   original channel dimension `C` with `to_out`.

Only the N³ reduction (the einsum) is performed in Triton; all
linear / activation / final normalisation steps use highly‑optimised
PyTorch kernels.

The implementation works for any batch size ≤2, sequence length up to
1024, and hidden dimensions 128‑384.  It handles the presence or
absence of a mask.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_einsum_kernel(
    left_ptr,          # const * fp16
    right_ptr,         # const * fp16
    out_ptr,           # * fp16
    B, N, H,           # int32 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM kernel for the TriMul einsum.

    One program instance processes a tile of size BLOCK_M × BLOCK_N
    for a single (batch, hidden) pair.  The reduction over the
    sequence dimension K is performed in a loop.
    """
    pid_hd = tl.program_id(0)   # combined batch * hidden index
    pid_i  = tl.program_id(1)   # tile over the i‑axis
    pid_j  = tl.program_id(2)   # tile over the j‑axis

    # --------------------------------------------------------------
    # Decode the combined index into batch (b) and hidden channel (h)
    # --------------------------------------------------------------
    b = pid_hd // H                # batch index 0 .. B‑1
    h = pid_hd - b * H             # hidden index 0 .. H‑1

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialise the accumulator in fp32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the summation axis of the einsum)
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Compute the base pointer for this (b, h) pair
        left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
        right_base = right_ptr + b * stride_right_b + h * stride_right_h

        # Load tiles of shape (BLOCK_M, BLOCK_K) and (BLOCK_N, BLOCK_K)
        #   left[i,k]  and  right[j,k]
        left_ptrs  = left_base  + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        right_ptrs = right_base + offs_j[:, None] * stride_right_i + offs_k[None, :] * stride_right_k

        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)        # (M, K)
        b_ = tl.load(right_ptrs,
                    mask=mask_j[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)        # (N, K)

        # Accumulate: left @ rightᵀ  -> (M, N)
        acc += tl.dot(a, b_, trans_b=True)

    # --------------------------------------------------------------
    # Store the result (cast back to fp16)
    # --------------------------------------------------------------
    out_base = out_ptr + b * stride_out_b + h * stride_out_h
    out_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype torch.float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections + gates (all in fp32)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,H]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask shape: [B, N, N]  ->  [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(input_tensor.dtype)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Apply gates
    left = left_proj * left_gate
    right = right_proj * right_gate

    # -------------------------------------------------------------
    # 4) Triton kernel: einsum reduction over the sequence dimension
    # -------------------------------------------------------------
    # Cast to fp16 for bandwidth‑limited GEMM
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    B, N, _, H = left_h.shape

    out_h = torch.empty_like(left_h)   # same shape, fp16

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Number of tiles
    num_i_tiles = (N + BLOCK_M - 1) // BLOCK_M
    num_j_tiles = (N + BLOCK_N - 1) // BLOCK_N
    total_hd = B * H                     # combined batch‑hidden programs

    grid = (total_hd, num_i_tiles, num_j_tiles)

    _trimul_einsum_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, H,
        left_h.stride(0), left_h.stride(1), left_h.stride(2), left_h.stride(3),
        right_h.stride(0), right_h.stride(1), right_h.stride(2), right_h.stride(3),
        out_h.stride(0), out_h.stride(1), out_h.stride(2), out_h.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # modest warp count; good for H100
    )

    # Cast back to fp32 for the remaining ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 5) Final normalisation, output gate and projection back to C
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
