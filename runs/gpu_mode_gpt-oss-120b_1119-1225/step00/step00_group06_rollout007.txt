    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get('nomask', True)
    # Possibly mask is provided; it's a tensor shape [B,N,N] (maybe bool or float). We'll treat as float.

    # 1. layernorm
    norm_weight = weights['norm.weight']; norm_bias = weights['norm.bias']
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias)

    # 2. linear projections
    left_proj_weight = weights['left_proj.weight'] # shape [hidden_dim, dim]
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']

    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    if not nomask:

        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # 3. matmul using Triton
    # Reshape to [B*hidden_dim, N, N]

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    block_i = tl.program_id(1)
    block_j = tl.program_id(2)

    # compute offsets for i and j positions
    i_base = block_i * BLOCK_M
    j_base = block_j * BLOCK_N

    # Define ranges
    range_i = i_base + tl.arange(0, BLOCK_M)
    range_j = j_base + tl.arange(0, BLOCK_N)

    # mask for out-of-bounds
    mask_i = range_i < N
    mask_j = range_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, N, BLOCK_K):
        # compute k offsets
        k_base = k
        range_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = range_k < N

        # Load tile from left: shape (BLOCK_M, BLOCK_K)
        # left_ptr + batch_idx*stride_l_batch + (range_i[:,None] * stride_l_i) + (range_k[None,:] * stride_l_k)
        left_offset = batch_idx * stride_l_batch + range_i[:, None] * stride_l_i + range_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load tile from right transposed: shape (BLOCK_K, BLOCK_N)
        # Want right[k, j] element: original right [i, j]; transposed means we need (k, j). Use strides accordingly:
        right_offset = batch_idx * stride_r_batch + range_k[:, None] * stride_r_i + range_j[None, :] * stride_r_k

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    nomask = config.get('nomask', True)

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])

    # linear projections
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    if not nomask:
        # mask shape [B,N,N] dtype float32 (or bool)
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # broadcast to hidden_dim
        left = left * mask_f
        right = right * mask_f

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # tri mul: left @ right^T per hidden channel
    # Prepare for Triton
    B, N, _, _ = x.shape
    # permute and reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch and hidden dim
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    # allocate output
    out_flat = torch.empty_like(left_flat)

    # call Triton kernel
    tri_mul_matmul(left_flat, right_flat, out_flat, N)

    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=weights['to_out_norm.weight'], bias=weights['to_out_norm.bias']
    )

    # apply out gate
    out = out * out_gate

    # final linear
    out = torch.nn.functional.linear(out, weights['to_out.weight'])

    return out
```

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_i, stride_r_k,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute global row and col offsets
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out of bounds
    mask_i = i < N
    mask_j = j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, N, BLOCK_K):
        k_end = tl.minimum(k + BLOCK_K, N)
        cur_k = k_end - k
        k_range = k + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = k_range < N

        # left tile (M, K)
        left_offset = batch_idx * stride_l_batch + i[:, None] * stride_l_i + k_range[None, :] * stride_l_k
        left_tile = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right tile transposed: (K, N) i.e., we need right[k, j] elements
        # right offset: row = k_range, col = j
        right_offset = batch_idx * stride_r_batch + k_range[:, None] * stride_r_i + j[None, :] * stride_r_k
        right_tile = tl.load(right_ptr + right_offset, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product accumulation
        acc += tl.dot(left_tile, right_tile)

    # store result
    out_offset = batch_idx * stride_o_batch + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptr + out_offset, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward kernel.

Algorithm
---------
1. Layer‑Norm the input tensor `x` on the last dimension.
2. Linear projections (no bias) → `left` and `right` of shape
   [B, N, N, H] where H = hidden_dim.
3. Optional pair‑wise mask (broadcast over H) applied to `left` and `right`.
4. Compute three sigmoid gates from the normalized input and apply
   them to `left`, `right` and the later output.
5. The core operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication `left[...,h] @ right[...,h].T`
   for every hidden channel `h`.  This is implemented with a custom
   Triton kernel that processes all hidden channels as an extra batch
   dimension and uses FP16 for the heavy inner‑product while accumulating
   in FP32 for numerical stability.
6. Layer‑Norm the intermediate result on the hidden dimension,
   multiply with the output gate and apply the final linear projection
   back to the original feature dimension.

The kernel works for any batch size (1 or 2), sequence length up to 1024
and hidden dimensions used in the tests.  All tensor arguments are
expected to be on the CUDA device.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM `C = A @ Bᵀ` where A,B are FP16 and C is FP32.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_fp16_kernel(
    a_ptr,                     # FP16 left matrix   [BH, N, N]
    b_ptr,                     # FP16 right matrix  [BH, N, N]
    c_ptr,                     # FP32 output       [BH, N, N]
    N,                         # sequence length
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_i, stride_b_k,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)        # combined batch * hidden channel
    pid_m = tl.program_id(1)           # block row of output
    pid_n = tl.program_id(2)           # block col of output

    # Global row / column indices
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        k_range = k + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # ---- load left tile  (M x K) ----
        a_offset = (
            batch_idx * stride_a_batch
            + i[:, None] * stride_a_i
            + k_range[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptr + a_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # ---- load right tile transposed (K x N) ----
        # we need Bᵀ → element (k, j) of original B
        b_offset = (
            batch_idx * stride_b_batch
            + k_range[:, None] * stride_b_i   # row = k
            + j[None, :] * stride_b_k         # col = j
        )
        b = tl.load(
            b_ptr + b_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float16,
        )

        # Multiply‑accumulate (convert dot result to FP32)
        acc += tl.cast(tl.dot(a, b), tl.float32)

    # ---- write out the tile ----
    c_offset = (
        batch_idx * stride_c_batch
        + i[:, None] * stride_c_i
        + j[None, :] * stride_c_j
    )
    tl.store(
        c_ptr + c_offset,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def _tri_mul_fp16(A: torch.Tensor, B: torch.Tensor, C: torch.Tensor, N: int):
    """
    Batched GEMM using the Triton kernel above.
    A, B : FP16 tensors of shape [BH, N, N] (contiguous)
    C    : FP32 tensor of the same shape to hold the result.
    """
    assert A.is_contiguous() and B.is_contiguous()
    BH = A.shape[0]

    # Block sizes – empirically good for H100 FP16 matmul
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        BH,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    stride_a = A.stride()
    stride_b = B.stride()
    stride_c = C.stride()

    _tri_mul_fp16_kernel[grid](
        A,
        B,
        C,
        N,
        stride_a[0],
        stride_a[1],
        stride_a[2],
        stride_b[0],
        stride_b[1],
        stride_b[2],
        stride_c[0],
        stride_c[1],
        stride_c[2],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


# ----------------------------------------------------------------------
# Main entry point expected by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (mask Tensor: [B, N, N])
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # broadcast mask over hidden dimension
        mask_f = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: for each hidden channel compute left @ rightᵀ
    # ------------------------------------------------------------------
    # Rearrange to treat hidden dimension as part of the batch:
    #   [B, N, N, H] -> [B, H, N, N] -> [BH, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    BH = bs * hidden_dim
    left_flat = left_perm.view(BH, seq_len, seq_len)     # FP32
    right_flat = right_perm.view(BH, seq_len, seq_len)   # FP32

    # Convert to FP16 for the heavy multiply‑accumulate
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Output buffer (FP32)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Triton GEMM
    _tri_mul_fp16(left_fp16, right_fp16, out_flat, seq_len)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate                     # apply output gate
    out = F.linear(out, weights["to_out.weight"])  # final projection → [B,N,N,dim]

    return out
