@triton.jit
def trimul_matmul_kernel(a_ptr, b_ptr, c_ptr,
                         M, N, K,
                         stride_am, stride_ak,
                         stride_bk, stride_bn,
                         stride_cm, stride_cn,
                         stride_batch_a, stride_batch_b, stride_batch_c,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask to handle boundary
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Compute offset for current block of K
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers
        a_ptrs = a_ptr + batch_idx * stride_batch_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + batch_idx * stride_batch_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0) # shape (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0) # shape (BLOCK_K, BLOCK_N)

        # Cast to fp32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_idx * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
import math

@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul kernel for TriMul.
    Compute for each batch m:
      C[m,i,j] = sum_k A[m,i,k] * B[m,k,j]
    where A has shape (M, N, K) and B has shape (M, K, N).
    """
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)  # block row
    pid_n = tl.program_id(2)  # block col

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out-of-range rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M block, i, k)
        a_ptrs = a_ptr + batch_idx * stride_batch_a + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (M block, k, j)
        b_ptrs = b_ptr + batch_idx * stride_batch_b + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = c_ptr + batch_idx * stride_batch_c + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operator.
    Performs:
      1) LayerNorm on input.
      2) Linear projections (left/right) and gating.
      3) Pairwise multiplication over the sequence dimension via a Triton kernel.
      4) Output LayerNorm, gating, and final linear projection.
    Arguments:
      data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
      Tensor of shape [bs, seq_len, seq_len, dim].
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input (last dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        # mask shape (B,N,N) -> (B,N,N,1)
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Gating values from the normalized input (x)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Core TriMul multiplication using Triton
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim

    # Rearrange tensors so that the hidden dimension becomes part of the batch
    # left: (B, H, N, N) where inner dim N is the 'k' index
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) -> (m,i,k)
    # right: (B, H, N, N) but we need (B, H, K=N, J=N) i.e. rows=k, cols=j
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, K, J)

    M = B * hidden_dim  # total batched matrices
    # Flatten the leading two dimensions into one batch dimension
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)
    out_flat = torch.empty_like(left_flat)  # (M, N, N)

    # Compute strides (in elements)
    stride_a_batch = left_flat.stride(0)
    stride_a_row = left_flat.stride(1)   # stride for i (row)
    stride_a_col = left_flat.stride(2)   # stride for k (col)

    stride_b_batch = right_flat.stride(0)
    stride_b_row = right_flat.stride(1)   # stride for k (row)
    stride_b_col = right_flat.stride(2)   # stride for j (col)

    stride_c_batch = out_flat.stride(0)
    stride_c_row = out_flat.stride(1)
    stride_c_col = out_flat.stride(2)

    # Select reasonable block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        N,
        stride_a_row,
        stride_a_col,
        stride_b_row,
        stride_b_col,
        stride_c_row,
        stride_c_col,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to original dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # bias is None

    return out

"""
TriMul (outgoing) – Triton‑accelerated core multiplication.

Algorithm
---------
1. Layer‑norm the input tensor (last dim) using the provided γ/β.
2. Project the normalized tensor to a hidden dimension `H` with two linear layers:
   left = x·W_left;   right = x·W_right.
3. Apply an optional boolean mask (broadcasted on the hidden dim) and
   element‑wise gating (sigmoid of three additional linear layers).
4. The heavy part:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden channel.
   We fuse the hidden channel into the batch dimension (M = B·H) and launch a
   custom Triton kernel that computes
        C[m,i,j] = Σ_k A[m,i,k] * B[m,k,j]
   where A and B are the permuted left/right tensors.
5. Layer‑norm the result across the hidden dimension, apply the output gate,
   and finally project back to the original `dim` with a linear layer.

The kernel follows the classic block‑matmul pattern (M×N×K) and works for any
sequence length up to 1024 on an H100.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,          # A: (batch, row=i, col=k)
    stride_bk, stride_bn,          # B: (batch, row=k, col=j)
    stride_cm, stride_cn,          # C: (batch, row=i, col=j)
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication:
        C[m, i, j] = Σ_k A[m, i, k] * B[m, k, j]

    A shape (M, N, K), B shape (M, K, N), C shape (M, N, N).
    """
    batch_idx = tl.program_id(0)          # which (batch * hidden) matrix
    pid_m = tl.program_id(1)              # tile row index
    pid_n = tl.program_id(2)              # tile col index

    # Tile offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks

        mask_k = offs_k < K

            a_ptr
            + batch_idx * stride_batch_a
            + offs_i[:, None] * stride_am
            + offs_k[None, :] * stride_ak

            b_ptr

            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bn

    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B, N, N, H)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Mask application (if a mask is supplied)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)                # (B, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating – sigmoid(linear(x))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul multiplication (batched GEMM) via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape                     # (B, N, N, H)

    # Bring hidden dimension into the batch dimension
    # left_perm: (B, H, N, N) where axes are (i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right_perm: (B, H, N, N) but we need (k, j) ordering → permute(0,3,2,1)
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    M = B * hidden_dim                          # total number of independent matmuls

    # Flatten leading batch/hidden dimensions → shape (M, N, N)
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)
    out_flat = torch.empty_like(left_flat)      # (M, N, N)

    # ------------------------------------------------------------------
    # Compute strides (in elements) for Triton
    # ------------------------------------------------------------------
    stride_a_batch = left_flat.stride(0)
    stride_a_row = left_flat.stride(1)   # stride for i
    stride_a_col = left_flat.stride(2)   # stride for k

    stride_b_batch = right_flat.stride(0)
    stride_b_row = right_flat.stride(1)  # stride for k
    stride_b_col = right_flat.stride(2)  # stride for j

    stride_c_batch = out_flat.stride(0)
    stride_c_row = out_flat.stride(1)    # stride for i
    stride_c_col = out_flat.stride(2)    # stride for j

    # ------------------------------------------------------------------
    # Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        M,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trimul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        N,                                 # K == N
        stride_a_row,
        stride_a_col,
        stride_b_row,
        stride_b_col,
        stride_c_row,
        stride_c_col,
        stride_a_batch,
        stride_b_batch,
        stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm, output gate, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, dim)

    return out
```
