@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bzl, stride_bzi, stride_bzk, stride_bzh,
    stride_brz, stride_brj, stride_brk, stride_brh,
    stride_outb, stride_outi, stride_outj, stride_outh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    EPS: tl.constexpr=1e-5):
    # Compute block indices
    pid_i = tl.program_id(axis=0)
    pid_j = tl.program_id(axis=1)
    pid_hb = tl.program_id(axis=2)  # corresponds to batch and hidden tile

    # number of hidden tiles
    num_h_tiles = tl.cdiv(H, BLOCK_H)
    b = pid_hb // num_h_tiles
    h_tile = pid_hb % num_h_tiles

    # offset for i, j
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # offsets within tile
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)

    # mask for i, j bounds
    mask_i = i_range < N
    mask_j = j_range < N

    # offsets for hidden within tile
    h_start = h_tile * BLOCK_H
    h_range = h_start + tl.arange(0, BLOCK_H)
    mask_h = h_range < H

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k_start = kk * BLOCK_K
        k_range = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # Load left block: shape (BLOCK_M, BLOCK_K, BLOCK_H) -> we will load per hidden channel inside loop
        # We'll do for each h in tile separately
        # Compute pointer for left: base = left_ptr + b*stride_bzl
        # left offset: i_range * stride_bzi + k_range * stride_bzk + h_offset * stride_bzh
        # use broadcasting:
        # left_ptrs = left_ptr + b*stride_bzl + i_range[:, None] * stride_bzi + k_range[None, :] * stride_bzk + h_offset * stride_bzh

        # Load right block similarly but shape (BLOCK_K, BLOCK_N, BLOCK_H)

        # compute for each h in tile
        for h_idx in range(0, BLOCK_H):
            if not mask_h[h_idx]:
                continue
            h_off = h_range[h_idx]

            # left pointer
            left_ptrs = left_ptr + b*stride_bzl + i_range[:, None] * stride_bzi + k_range[None, :] * stride_bzk + h_off * stride_bzh
            # right pointer
            right_ptrs = right_ptr + b*stride_brz + k_range[:, None] * stride_brk + j_range[None, :] * stride_brj + h_off * stride_brh

            # Load blocks with mask
            left_block = tl.load(left_ptrs,
                                 mask=mask_i[:, None] & mask_k[None, :],
                                 other=0.0)
            # right block load shape (BLOCK_K, BLOCK_N)
            right_block = tl.load(right_ptrs,
                                  mask=mask_k[:, None] & mask_j[None, :],
                                  other=0.0)

            # Convert to float32
            left_block = left_block.to(tl.float32)
            right_block = right_block.to(tl.float32)

            # Compute dot product: left (M,K) dot right (K,N) => (M,N)
            prod = tl.dot(left_block, right_block)  # shape (BLOCK_M, BLOCK_N)

            # Accumulate
            acc[:, :, h_idx] += prod

    # Write out results to out_ptr
    out_ptrs = out_ptr + b*stride_outb + i_range[:, None, None] * stride_outi + j_range[None, :, None] * stride_outj + h_range[None, None, :] * stride_outh
    # Write with mask
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

"""
TriMul "outgoing" forward pass (AlphaFold3).

The kernel fuses the O(N³·C_hidden) tensor contraction
    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
using a custom Triton kernel.  The rest of the
TriMul module (LayerNorms, Linear projections, sigmoid gates)
is performed with PyTorch for simplicity.

The kernel works on a padded hidden dimension (multiple of 32)
to guarantee safe memory accesses and processes a BLOCK_H slice
of the hidden channels per thread‑block.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: computes the einsum contraction for a block of (i, j, h)
# ---------------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    B, N, H,                                 # problem size
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # -----------------------------------------------------------------------
    # program IDs identify a tile (i, j) and a (batch, hidden‑tile) pair.
    # -----------------------------------------------------------------------
    pid_i = tl.program_id(axis=0)          # tile row   (i)
    pid_j = tl.program_id(axis=1)          # tile col   (j)
    pid_hb = tl.program_id(axis=2)         # combined batch / hidden‑tile

    num_h_tiles = tl.cdiv(H, BLOCK_H)
    b = pid_hb // num_h_tiles              # batch index
    h_tile = pid_hb % num_h_tiles           # hidden‑tile index

    # -----------------------------------------------------------------------
    # Tile offsets
    # -----------------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    h_start = h_tile * BLOCK_H

    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    h = h_start + tl.arange(0, BLOCK_H)

    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulator: (M, N, H_tile)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # Loop over the reduction dimension k.
    # -----------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k_start = kk * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # --------------------------------------------------------------------
        # Loop over hidden channels inside the tile.
        # We load a (M, K) matrix from left and a (K, N) matrix from right
        # for each hidden slice and accumulate a matrix product.
        # --------------------------------------------------------------------
        for hi in range(0, BLOCK_H):
            # pointers for the current hidden slice --------------------------------
            # left[b, i, k, hi]
            left_ptrs = (
                left_ptr
                + b * stride_l_b
                + (i[:, None] * stride_l_i)
                + (k[None, :] * stride_l_k)
                + (h_start + hi) * stride_l_h
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k[None, :],
                other=0.0,
            ).to(tl.float32)          # (M, K)

            # right[b, j, k, hi]  --> we need (K, N)
            right_ptrs = (
                right_ptr
                + b * stride_r_b
                + (k[:, None] * stride_r_k)
                + (j[None, :] * stride_r_i)   # stride over the second dimension (j) is the same as stride_i
                + (h_start + hi) * stride_r_h
            )
            right = tl.load(
                right_ptrs,
                mask=mask_k[:, None] & mask_j[None, :],
                other=0.0,
            ).to(tl.float32)           # (K, N)

            # matrix multiply for this hidden slice
            prod = tl.dot(left, right)    # (M, N)
            acc[:, :, hi] += prod

    # -----------------------------------------------------------------------
    # Write back the result (respecting i/j/h masks)
    # -----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + (i[:, None, None] * stride_o_i)
        + (j[None, :, None] * stride_o_j)
        + (h[None, None, :] * stride_o_h)
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ---------------------------------------------------------------------------
# Helper that launches the Triton kernel (pads hidden dim to a multiple of 32)
# ---------------------------------------------------------------------------
def _trimul_einsum_triton(left: torch.Tensor, right: torch.Tensor, hidden_dim: int):
    """
    left/right : torch.Tensor of shape (B, N, N, hidden_dim)  (float16)
    Returns    : torch.Tensor of shape (B, N, N, hidden_dim)  (float32)
    """
    B, N, _, H = left.shape
    assert H == hidden_dim, "left/right hidden dim mismatch with supplied hidden_dim"

    BLOCK_H = 32
    H_pad = ((hidden_dim + BLOCK_H - 1) // BLOCK_H) * BLOCK_H
    if H_pad != hidden_dim:
        pad = H_pad - hidden_dim
        left = F.pad(left, (0, pad))   # pad hidden dimension (last dim)
        right = F.pad(right, (0, pad))

    # output tensor (float32)
    out = torch.empty((B, N, N, H_pad), dtype=torch.float32, device=left.device)

    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,                # i tiles
        (N + BLOCK_N - 1) // BLOCK_N,                # j tiles
        B * ((H_pad + BLOCK_H - 1) // BLOCK_H),      # (batch, hidden‑tile) tiles
    )

    _trimul_einsum_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H_pad,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    # slice back to the original hidden dimension
    if H_pad != hidden_dim:
        out = out[..., :hidden_dim]
    return out


# ---------------------------------------------------------------------------
# Entry point for the custom kernel
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) implemented with a Triton kernel for the
    core pairwise contraction.  Arguments:

    data = (input_tensor, mask, weights_dict, config_dict)

    Returns:
        output tensor of shape (bs, seq_len, seq_len, dim) (float32)
    """
    # -----------------------------------------------------------------------
    # unpack inputs
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------------
    # fetch the weight tensors (all are 1‑D or 2‑D tensors on the same device)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------------
    # 1) Input LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------------
    # mean / var over dim
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)
    x = (input_tensor - mean) * inv_std
    # apply learned weight / bias
    x = x * norm_weight + norm_bias

    # -----------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------------
    # 3) Optional mask (mask shape: B x N x N)
    # -----------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------------
    # 4) Gating terms (sigmoid on linear projections of x)
    # -----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5) Cast to FP16 and run the Triton kernel for the contraction
    # -----------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # core contraction: out (B,N,N,hidden_dim) in float32
    out = _trimul_einsum_triton(left_fp16, right_fp16, hidden_dim)

    # -----------------------------------------------------------------------
    # 6) Output LayerNorm over hidden_dim
    # -----------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    inv_std_o = torch.rsqrt(var_o + eps)
    out_norm = (out - mean_o) * inv_std_o
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # -----------------------------------------------------------------------
    # 7) Apply final sigmoid gating and final linear projection back to dim
    # -----------------------------------------------------------------------
    out_norm = out_norm * out_gate
    output = F.linear(out_norm, to_out_weight)   # (B, N, N, dim)

    return output
```
