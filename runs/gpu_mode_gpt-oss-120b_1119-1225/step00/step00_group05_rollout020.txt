):

    # compute number of tiles
    num_tiles_m = tl.cdiv(N, BLOCK_M)

    h = bh % H

    i_start = tile_i * BLOCK_M
    j_start = tile_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)

import torch
import triton
import triton.language as tl

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    eps = 1e-5

    # read dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # note mask may be None if nomask
    # We'll use mask if mask_tensor is not None else ones
    # Ensure mask dtype matches input
    if mask_tensor is None:
        mask_tensor = torch.ones(input_tensor.shape[:3], dtype=dtype, device=device)
    else:
        mask_tensor = mask_tensor.to(dtype)

    # LayerNorm on last dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask (if needed)
    if not config.get("nomask", False):
        mask_exp = mask_tensor.unsqueeze(-1)
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Core TriMul operation implemented in Triton
    out = trimul_forward(left, right)  # defined later

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out gate
    out = out * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input over the last dimension.
2. Project the normalized tensor to a hidden dimension with four linear layers:
   - left_proj / right_proj   : value tensors
   - left_gate / right_gate   : sigmoid gates for the value tensors
   - out_gate                 : sigmoid gate applied after the multiplication.
3. Apply an optional pairwise mask (broadcast over the hidden dimension).
4. Element‑wise gate the value tensors.
5. Core update:
       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is a batched matrix multiplication of shape (N×N) × (N×N)ᵀ for each
   batch and hidden‑channel.  The heavy N³ work is performed by a Triton kernel
   that tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N) and streams over k in
   BLOCK_K sized chunks, accumulating in float32.
6. Apply a second LayerNorm over the hidden dimension, multiply by out_gate,
   and finish with a linear projection back to the original dimension.

Only the N³ multiplication (step 5) is executed on the GPU via Triton; all
other operations use PyTorch which already provides highly‑optimized kernels.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel for the N³ “tri‑multiplicative” update.
# -------------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,          # const float*  (B, N, N, H)
    right_ptr,         # const float*  (B, N, N, H)
    out_ptr,           # float*        (B, N, N, H)
    B, N, H,           # meta‑parameters
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_jr, stride_kr, stride_hr,   # right strides
    stride_bo, stride_io, stride_jo, stride_ho,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape (B,N,N,H) and out has the same shape.
    The kernel tiles the i‑ and j‑dimensions (M×N blocks) and streams
    over k in BLOCK_K chunks, accumulating in float32.
    """
    pid = tl.program_id(0)

    # Number of tiles along the i and j dimensions
    num_tiles_m = tl.cdiv(N, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)
    tiles_per_bh = num_tiles_m * num_tiles_n

    # Decode batch, hidden‑channel and tile coordinates
    bh  = pid // tiles_per_bh                     # combined batch‑hidden index
    tile_id = pid % tiles_per_bh
    tile_i = tile_id // num_tiles_n
    tile_j = tile_id % num_tiles_n

    b = bh // H                                   # batch index
    h = bh % H                                    # hidden‑channel index

    # Starting offsets for the i‑ and j‑tiles
    i_start = tile_i * BLOCK_M
    j_start = tile_j * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointer for the current (b, h) slice
    base_left  = b * stride_bl + h * stride_hl
    base_right = b * stride_br + h * stride_hr
    base_out   = b * stride_bo + h * stride_ho

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kk * BLOCK_K
        cur_k   = k_start + tl.arange(0, BLOCK_K)
        mask_k  = cur_k < N

        # left[i,k]   -> (M, K)
        left_ptrs = base_left + offs_i[:, None] * stride_il + cur_k[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[j,k] needs to be loaded as (K, N) for a dot product
        right_ptrs = base_right + cur_k[:, None] * stride_kr + offs_j[None, :] * stride_jr
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)

        # Accumulate dot(left, right)  → (M, N)
        acc += tl.dot(left, right)

    # Write the result back
    out_ptrs = base_out + offs_i[:, None] * stride_io + offs_j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Helper that launches the Triton kernel
# -------------------------------------------------------------------------
def _trimul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left / right: (B, N, N, H) tensors (float32 or float16, contiguous)
    returns out of identical shape.
    """
    B, N, _, H = left.shape
    # Allocate output tensor
    out = torch.empty_like(left)

    # Ensure contiguous memory for pointer arithmetic
    left  = left.contiguous()
    right = right.contiguous()
    out   = out.contiguous()

    # Strides (in elements) of the three tensors
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # Tunable tile sizes (empirically good for H100)
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 64

    # Grid: one program per (batch, hidden, tile_i, tile_j)
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles = B * H * grid_m * grid_n

    _trimul_kernel[total_tiles](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_jr,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# -------------------------------------------------------------------------
# Entry point required by the evaluator
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype
    eps = 1e-5

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Optional mask handling (mask shape: [B, N, N])
    # -----------------------------------------------------------------
    if mask_tensor is None:
        mask_tensor = torch.ones(input_tensor.shape[:3],
                                 dtype=dtype,
                                 device=device)
    else:
        mask_tensor = mask_tensor.to(dtype)

    # -----------------------------------------------------------------
    # LayerNorm over the last dimension
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Apply mask if required
    # -----------------------------------------------------------------
    if not nomask:
        mask_exp = mask_tensor.unsqueeze(-1)          # [B,N,N,1]
        left_proj  = left_proj  * mask_exp
        right_proj = right_proj * mask_exp

    # -----------------------------------------------------------------
    # Gating tensors (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate   = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # -----------------------------------------------------------------
    # Apply gates to the projected tensors
    # -----------------------------------------------------------------
    left  = left_proj  * left_gate
    right = right_proj * right_gate

    # -----------------------------------------------------------------
    # Core N³ multiplication via Triton
    # -----------------------------------------------------------------
    out = _trimul_forward(left, right)   # shape [B, N, N, H]

    # -----------------------------------------------------------------
    # Output LayerNorm over hidden dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # Apply the final gate
    # -----------------------------------------------------------------
    out = out * out_gate

    # -----------------------------------------------------------------
    # Final linear projection back to the model dimension
    # -----------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
