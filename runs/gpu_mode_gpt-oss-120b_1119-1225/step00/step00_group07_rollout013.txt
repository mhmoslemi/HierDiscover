def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # Retrieve weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input x
    x_norm = (x - x.mean(dim=-1, keepdim=True)) * torch.rsqrt(x.var(dim=-1, unbiased=False, keepdim=True) + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections and gates (float16 for speed)
    x_norm_f16 = x_norm.to(torch.float16)

    left = torch.nn.functional.linear(x_norm_f16, left_proj_weight.to(torch.float16))   # shape [B,N,N,H] fp16
    right = torch.nn.functional.linear(x_norm_f16, right_proj_weight.to(torch.float16))

    # Mask handling
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)  # broadcast
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x_norm_f16, left_gate_weight.to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm_f16, right_gate_weight.to(torch.float16)).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm_f16, out_gate_weight.to(torch.float16)).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Prepare for matmul kernel: permute to (B, H, N, N)
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # transpose second dim

    # Perform batched matmul via Triton kernel
    out_perm = torch.empty_like(left_perm)  # same shape (B, H, N, N)
    # Compute strides
    stride_l_batch = left_perm.stride(0)
    stride_l_m = left_perm.stride(1)
    stride_l_k = left_perm.stride(2)
    stride_r_batch = right_perm.stride(0)
    stride_r_k = right_perm.stride(1)
    stride_r_n = right_perm.stride(2)
    stride_o_batch = out_perm.stride(0)
    stride_o_m = out_perm.stride(1)
    stride_o_n = out_perm.stride(2)

    # Launch kernel
    trimul_mm_kernel[(B*hidden_dim, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))](
        left_perm, right_perm, out_perm,
        B, N, hidden_dim,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=64, BLOCK_N=64, BLOCK_K=32,
        num_stages=3, num_warps=4
    )
    # After kernel, out_perm shape (B,H,N,N). Permute back
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,H] float16
    out = out.to(torch.float32)

    # To_out_norm (LayerNorm over hidden)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate and linear to dim
    out_gate = out_gate.to(torch.float32)
    out = out * out_gate  # elementwise
    out = torch.nn.functional.linear(out, to_out_weight)  # maps hidden -> dim

    return out

@triton.jit
def trimul_mm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)  # index over (B*H)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch index and hidden index to compute per batch
    # Since we passed combined batch dimension = B * H, we treat each as separate logical batch
    # So stride calculations already incorporate per-batch offset
    # We'll compute offsets for left, right, out based on pid_batch

    # Determine start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # load a mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Compute pointer offsets for A and B
    # A (left): shape (B*H, N, N) with dimensions (batch, m, k)
    # B (right): shape (B*H, N, N) with dimensions (batch, k, n)
    # Use index linear offsets: base + pid_batch*stride_batch + ...
    a_ptrs = left_ptr + pid_batch * stride_l_batch + (offs_m[:, None] * stride_l_m) + (tl.arange(0, BLOCK_K)[None, :] * stride_l_k)
    b_ptrs = right_ptr + pid_batch * stride_r_batch + (tl.arange(0, BLOCK_K)[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offsets for this block of k
        k_offset = k * BLOCK_K
        a_ptrs = left_ptr + pid_batch * stride_l_batch + ((offs_m[:, None] * stride_l_m) + ( (k_offset + tl.arange(0, BLOCK_K)) * stride_l_k))
        b_ptrs = right_ptr + pid_batch * stride_r_batch + (( (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n))

        # Load A and B blocks; we need to mask out-of-range k.
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & (k_offset + tl.arange(0, BLOCK_K) < N)), other=0.0)
        b = tl.load(b_ptrs, mask=((k_offset + tl.arange(0, BLOCK_K)[:, None] < N) & mask_n[None, :]), other=0.0)

        # dot product
        acc += tl.dot(a, b)

    # Write back to C
    c_ptrs = out_ptr + pid_batch * stride_o_batch + (offs_m[:, None] * stride_o_m) + (offs_n[None, :] * stride_o_n)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul forward (outgoing) implementation.

The module consists of:
  1. LayerNorm on the input tensor (last dim = dim).
  2. Linear projections (left/right) and gating (left/right/out) without bias.
  3. Optional masking of the projected tensors.
  4. A fused batched matrix multiplication performed in Triton:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     after reshaping we have many independent GEMMs of shape (N,N)×(N,N) for each
     hidden channel d.
  5. LayerNorm over the hidden dimension, gating, and final linear projection
     back to “dim”.

Only the O(N³·hidden) multiplication is off‑loaded to a Triton kernel; all
other ops use native PyTorch (which already calls highly‑optimized cuBLAS /
cuDNN kernels). Mixed‑precision (FP16) is used for the heavy GEMMs while the
final result is kept in FP32.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM C = A @ B   (A : (B*H, M, K), B : (B*H, K, N))
#   - A and B are fp16, accumulation in fp32, result stored as fp16.
#   - The batch dimension is (batch * hidden_dim) – each hidden channel of each
#     batch is processed independently.
# ----------------------------------------------------------------------------- #
@triton.jit
def _trimul_matmul_kernel(
    a_ptr,          # *fp16   [BH, M, K]
    b_ptr,          # *fp16   [BH, K, N]
    c_ptr,          # *fp16   [BH, M, N]
    M, N, K,        # matrix sizes (all = seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)                     # 0 .. BH-1
    pid_m = tl.program_id(1)                         # row tile
    pid_n = tl.program_id(2)                         # col tile

    # -------------------- offsets & masks --------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------- loop over K --------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # k indices for this tile
        k_sel = k_offset + tl.arange(0, BLOCK_K)

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
               + (offs_m[:, None] * stride_a_m) \
               + (k_sel[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (k_sel[None, :] < K)),
                    other=0.0)

        # B tile: shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
               + (k_sel[:, None] * stride_b_k) \
               + (offs_n[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=((k_sel[:, None] < K) & mask_n[None, :]),
                    other=0.0)

        # fused dot product
        acc += tl.dot(a, b)

    # -------------------- write back --------------------
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
            + (offs_m[:, None] * stride_c_m) \
            + (offs_n[None, :] * stride_c_n)
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------------- #
# Public entry point used by the evaluator
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights_dict : mapping of parameter names to tensors
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # unpack
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nominal_mask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # weight aliases (all are float32)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    eps = 1e-5

    # --------------------------------------------------------------------- #
    # 1) Input layer‑norm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    # mean & var are computed in FP32 for stability
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias   # broadcast over trailing dim

    # --------------------------------------------------------------------- #
    # 2) Linear projections + gates (cast to fp16 for speed)
    # --------------------------------------------------------------------- #
    x_fp16 = x_norm.to(torch.float16)

    left  = torch.nn.functional.linear(x_fp16, left_proj_weight.to(torch.float16))   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_fp16, right_proj_weight.to(torch.float16))

    # optional mask – broadcast to last dim
    if not nominal_mask:
        mask_broadcast = mask_tensor.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left  * mask_broadcast
        right = right * mask_broadcast

    # gates (sigmoid, still fp16)
    left_gate  = torch.nn.functional.linear(x_fp16, left_gate_weight.to(torch.float16)).sigmoid()
    right_gate = torch.nn.functional.linear(x_fp16, right_gate_weight.to(torch.float16)).sigmoid()
    out_gate   = torch.nn.functional.linear(x_fp16, out_gate_weight.to(torch.float16)).sigmoid()

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 3) Batched GEMM: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Rearrange tensors to (B*hidden, N, N) and launch Triton kernel.
    # --------------------------------------------------------------------- #
    B, N, _, _ = left.shape

    # (B, hidden, N, N) → flatten batch‑hidden dimension → (BH, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(-1, N, N)   # A
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(-1, N, N)   # B^T already

    BH = left_mat.shape[0]          # = B * hidden_dim
    out_mat = torch.empty_like(left_mat)   # fp16 output

    # strides (int in python)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # tile sizes – tuned for H100 fp16 tensor‑core kernels
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (BH,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _trimul_matmul_kernel[grid](
        left_mat, right_mat, out_mat,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,  # H100 can handle many warps per block
        num_stages=4
    )

    # reshape back to original layout: (B, N, N, hidden)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # LayerNorm & final linear expect FP32

    # --------------------------------------------------------------------- #
    # 4) Output LayerNorm over hidden dimension
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps
    )

    # --------------------------------------------------------------------- #
    # 5) Output gate + final projection back to dim
    # --------------------------------------------------------------------- #
    out = out * out_gate.to(torch.float32)
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
