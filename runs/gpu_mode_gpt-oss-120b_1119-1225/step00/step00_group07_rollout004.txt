def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Move everything to same dtype/device (should be already)
    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm over last dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask application
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # reshape to (B, H, N, N) for kernel
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # allocate output
    out = torch.empty_like(left)

    # define block sizes (tuned for H100)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute strides
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out.stride()

    # launch kernel
    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), B * hidden_dim)
    tr_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_i,
        stride_r_k,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to (B,N,N,H)
    out = out.permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul custom kernel for AlphaFold3 outgoing update.

Algorithm:
1. Layer‑norm the input.
2. Linear projections (left/right) and gated projections.
3. Rearrange to (batch, hidden, N, N) and compute, for each hidden channel, the
   matrix multiplication  out_h = left_h @ right_h^T.  This dense N×N×N operation
   is the only O(N³) part and is implemented with a fused Triton kernel that
   performs batched GEMM using tiling (BLOCK_M × BLOCK_N tiles, BLOCK_K inner
   reduction).  Inputs are cast to FP16 to reduce memory traffic; accumulation
   is performed in FP32.
4. The result is reshaped back, layer‑normed, gated with `out_gate`, and a final
   linear projection back to the original channel dimension.
"""

import torch
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: batched GEMM of shape (B, H, N, N) @ (B, H, N, N)^T
# ------------------------------------------------------------
@triton.jit
def tr_mul_kernel(
    left_ptr,                 # const * half
    right_ptr,                # const * half
    out_ptr,                  # * half
    B, N, H,                  # int32
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Computes out[b, h, i, j] = sum_k left[b, h, i, k] * right[b, h, j, k] """
    pid_batch = tl.program_id(2)
    pid_m = tl.program_id(0)   # block row idx (i)
    pid_n = tl.program_id(1)   # block col idx (j)

    # decode batch + head index
    b = pid_batch // H
    h = pid_batch % H

    # Tile offsets
    i0 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i0 < N
    j_mask = j0 < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k0 < N

        # Load left block: shape (M, K)
        left_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + i0[:, None] * stride_l_i + k0[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0)
        left = left.to(tl.float32)

        # Load right block transposed: shape (K, N)
        # We want right_T[k, j] = right[b, h, j, k]
        right_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + k0[:, None] * stride_r_k + j0[None, :] * stride_r_i
        right_T = tl.load(right_ptrs,
                          mask=(k_mask[:, None] & j_mask[None, :]),
                          other=0.0)
        right_T = right_T.to(tl.float32)

        # Perform block GEMM
        acc += tl.dot(left, right_T)   # (M,K)@(K,N) -> (M,N)

    # Write back result
    out_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + i0[:, None] * stride_o_i + j0[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=(i_mask[:, None] & j_mask[None, :]))


def custom_kernel(data):
    """
    Forward pass of the TriMul "outgoing" update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, dim]
        - mask: torch.Tensor of shape [B, N, N] (may be None)
        - weights: dict of model parameters
        - config: dict containing 'dim', 'hidden_dim', 'nomask' (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Ensure tensors are on the same device and dtype
    device = input_tensor.device
    dtype = input_tensor.dtype

    # 1) LayerNorm over the last channel dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim]

    # 2) Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # 3) Optional mask (broadcast over hidden dim)
    if not nomask and mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # 4) Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # 5) Prepare tensors for Triton kernel
    B, N, _, _ = left.shape
    # shape to (B, hidden_dim, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # Cast to FP16 to reduce memory bandwidth for the kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)
    out_half = torch.empty_like(left_half)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Compute strides (in elements, not bytes)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_half.stride()
    stride_r_b, stride_r_h, stride_r_i, stride_r_k = right_half.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_half.stride()

    # Grid: (num_blocks_m, num_blocks_n, batch*head)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    tr_mul_kernel[grid](
        left_half,
        right_half,
        out_half,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_h,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_h,
        stride_r_i,
        stride_r_k,
        stride_o_b,
        stride_o_h,
        stride_o_i,
        stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 and reshape to original layout
    out = out_half.to(torch.float32).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # 6) Output layer norm and gating
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # 7) Final linear projection back to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul custom kernel for the “outgoing” update (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor over its channel dimension.
2. Compute left/right linear projections (no bias) and apply an optional
   pairwise mask.
3. Compute three sigmoid gates (left, right, out) and gate the projected
   tensors.
4. Rearrange tensors to shape (B, H, N, N) where H = hidden_dim.
   The core operation is:
       out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
   i.e. a batched matrix multiplication left_h @ right_hᵀ for each head.
   This O(N³) step is implemented with a fused Triton kernel that
   performs tiled GEMM using FP16 loads and FP32 accumulation.
5. Reshape back, apply a second layer‑norm, point‑wise multiply by the
   out‑gate, and linearly project back to the original channel dimension.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum
# ----------------------------------------------------------------------
@triton.jit
def tr_mul_kernel(
    left_ptr,          # const * fp16
    right_ptr,         # const * fp16
    out_ptr,           # * fp16
    B, N, H,           # int32 scalars
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_i, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
    The kernel processes a tile of (BLOCK_M, BLOCK_N) for each
    (batch, head) pair and loops over K in tiles of size BLOCK_K.
    """
    pid_batch = tl.program_id(2)          # index over B*H
    pid_m = tl.program_id(0)              # block row (i)
    pid_n = tl.program_id(1)              # block col (j)

    # Decode batch and head indices
    b = pid_batch // H

    i0 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i0 < N
    j_mask = j0 < N

    # Accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension in tiles

        k_mask = k0 < N

                     + k0[None, :] * stride_l_k)

                       other=0.0)

    # 7. Convert back to FP32 and reshape to original layout
    # ------------------------------------------------------------------
    out = out_half.to(torch.float32).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8. Output layer‑norm, out‑gate, and final linear projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
```
