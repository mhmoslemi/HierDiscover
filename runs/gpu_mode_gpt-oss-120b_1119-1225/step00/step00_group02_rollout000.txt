    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):

  pid_i = tl.program_id(0)
  pid_j = tl.program_id(1)

  # compute b and h from pid_b

"""
custom_kernel
==============

Implements the *outgoing* TriMul operator from AlphaFold3.

Algorithm
---------

1. **Layer‑norm on the input** (`x` of shape [B, N, N, C]).
2. **Linear projections** of `x` to the hidden dimension `H`:
   * left = Linear(x, left_proj_weight)
   * right = Linear(x, right_proj_weight)
3. **Gating**: sigmoid‑gated linear layers (`left_gate`, `right_gate`,
   `out_gate`) applied to the *pre‑norm* tensor `x`.
4. **Mask** (`mask` of shape [B,N,N]) is broadcast and applied to the
   projected tensors (if supplied).
5. **Core triangulation** – for each batch `b` and hidden channel `h`,
   compute the matrix product  

```
out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
```

   which is equivalent to `out_h = left_h @ right_h.T`.  
   This O(N³) step is executed with a custom Triton kernel that tiles
   the `i` and `j` dimensions (BLOCK_M×BLOCK_N) and iterates over the
   reduction dimension `k` in blocks (BLOCK_K).  The kernel works on
   fp16 data for bandwidth, accumulates in fp32, and writes the result
   back in fp16.
6. **Layer‑norm on hidden** (`to_out_norm`) followed by the `out_gate`
   (elementwise multiplication).
7. **Final linear** (`to_out`) projecting back to the original channel
   dimension `C`.

Only the expensive triangular matrix multiplication (step 5) is
implemented in Triton; all surrounding ops use PyTorch for simplicity
and correctness.

The function signature follows the problem statement:

```python
def custom_kernel(data):
    # data = (input_tensor, mask, weights, config)
```

The returned tensor has shape `[B, N, N, C]` and dtype ``torch.float32``.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the O(N³) triangulation.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,            # const * fp16   [B,N,N,H]
    right_ptr,           # const * fp16   [B,N,N,H]
    out_ptr,             # * fp16         [B,N,N,H]
    B, N, H,             # int32 scalars
    stride_bl, stride_il, stride_kl, stride_hl,   # left strides
    stride_br, stride_ir, stride_kr, stride_hr,   # right strides
    stride_bo, stride_io, stride_jo, stride_ho,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using blocked matrix multiply for each (b,h) pair.
    """
    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_bh = tl.program_id(2)         # combined batch*hidden index

    # ---------------------------------------------------------------
    # Decode batch and hidden dimension from pid_bh
    # ---------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ---------------------------------------------------------------
    # Offsets for the output tile (i,j)
    # ---------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---------------------------------------------------------------
    # Base pointers for this (b,h) slice
    # ---------------------------------------------------------------
    base_left = left_ptr + b * stride_bl + h * stride_hl
    base_right = right_ptr + b * stride_br + h * stride_hr
    base_out = out_ptr + b * stride_bo + h * stride_ho

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over reduction dimension k in BLOCK_K chunks
    # ---------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(0, num_k):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K) -> (i,k)
        left_ptrs = base_left + \
                    offs_i[:, None] * stride_il + \
                    offs_k[None, :] * stride_kl
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed: we need right[b, j, k, h]
        # after transposition we access it as (k,j)
        right_ptrs = base_right + \
                     offs_k[:, None] * stride_ir + \
                     offs_j[None, :] * stride_kr
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # GEMM on the block
        acc += tl.dot(left, right)

    # ---------------------------------------------------------------
    # Store the result back (casting to fp16)
    # ---------------------------------------------------------------
    out_ptrs = base_out + \
               offs_i[:, None] * stride_io + \
               offs_j[None, :] * stride_jo
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with the following tensors (all on GPU):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]          # hidden dimension (often 128)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (float32) and gating
    # ------------------------------------------------------------------
    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(
        x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask may be bool or float; ensure matching dtype
        mask = mask.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4) Apply gating
    # ------------------------------------------------------------------
    left = left_proj * left_gate          # shape [B,N,N,H]
    right = right_proj * right_gate

    # Cast to fp16 for the Triton kernel (better bandwidth)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 5) Triangular multiplication via Triton
    # ------------------------------------------------------------------
    out_fp16 = torch.empty((B, N, N, H),
                           dtype=torch.float16,
                           device=device)

    # Strides for the tensors (already contiguous)
    stride_bl, stride_il, stride_kl, stride_hl = left_fp16.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out_fp16.stride()

    # Tunable block sizes (empirically reasonable for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),    # tiles along i
        triton.cdiv(N, BLOCK_N),    # tiles along j
        B * H                       # one program per (batch, hidden) pair
    )

    _tri_mul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 for the following LayerNorm & linear
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm + out gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate     # broadcasting over hidden dim

    # ------------------------------------------------------------------
    # 7) Final projection back to original dimension C
    # ------------------------------------------------------------------
    result = torch.nn.functional.linear(out, weights["to_out.weight"])

    return result
