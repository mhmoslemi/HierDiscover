    BLOCK_K: tl.constexpr, VEC: tl.constexpr

    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

                          B, M, N, K,
                          stride_qa_batch, stride_qa_m, stride_qa_k,
                          stride_qb_batch, stride_qb_n, stride_qb_k,
                          stride_qc_batch, stride_qc_m, stride_qc_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    batch_idx = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_idx = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    pid_m = block_idx // tl.cdiv(N, BLOCK_N)
    pid_n = block_idx % tl.cdiv(N, BLOCK_N)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    pid_n = block_idx % tl.cdiv(N, BLOCK_N)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core

The heavy part of the TriMul module is the contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of independent matrix multiplications for each channel *d*.
We reshape the tensors to a batch dimension (B*D) and launch a tiled
GEMM kernel written in Triton.  The kernel computes C = A @ Bᵀ for each
batch using FP16 inputs and FP32 accumulation.  The surrounding
layer‑norms, linear projections and gating are performed with PyTorch
operations (they are cheap compared with the O(N³·D) contraction).

The implementation works for any batch size (B∈{1,2}), sequence length
N∈{128,256,512,1024}, model dimension *dim* and hidden dimension
*hidden_dim* as required by the test suite.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication with transposed B.
# A : [B*D, M, K]   (row‑major, stride_m = K, stride_k = 1)
# B : [B*D, N, K]   (row‑major, but we treat it as transposed: strides
#                    for the K‑dimension are the original column stride,
#                    and for the N‑dimension are the original row stride)
# C : [B*D, M, N]   result of A @ Bᵀ
# -------------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    a_ptr,                # *float16  [batch, M, K]
    b_ptr,                # *float16  [batch, N, K]  (will be accessed as K×N)
    c_ptr,                # *float16  [batch, M, N]
    batch, M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # -----------------------------------------------------------------
    # program IDs  (batch, row tile, col tile)
    # -----------------------------------------------------------------
    pid_b = tl.program_id(0)                 # batch index (0 … batch-1)
    pid_m = tl.program_id(1)                 # tile row index
    pid_n = tl.program_id(2)                 # tile col index

    # tile offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # -----------------------------------------------------------------
    # base pointers for this batch
    # -----------------------------------------------------------------
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # -----------------------------------------------------------------
    # accumulator in FP32
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # main K loop (tiled)
    # -----------------------------------------------------------------
    for k in range(0, K, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, K - k)

        # ----- load A tile (M×cur_k) -----
        a_ptrs = a_base + \
                 offs_m[:, None] * stride_a_m + \
                 (k + tl.arange(0, cur_k))[None, :] * stride_a_k
        a_mask = (offs_m < M)[:, None] & ((k + tl.arange(0, cur_k)) < K)[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0).to(tl.float32)   # (BLOCK_M, cur_k)

        # ----- load Bᵀ tile (cur_k×N) -----
        # B is stored as [batch, N, K]; we interpret the K‑dimension as
        # the rows of the transposed matrix.
        b_ptrs = b_base + \
                 (k + tl.arange(0, cur_k))[:, None] * stride_b_k + \
                 offs_n[None, :] * stride_b_n
        b_mask = ((k + tl.arange(0, cur_k)) < K)[:, None] & (offs_n < N)[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0).to(tl.float32)   # (cur_k, BLOCK_N)

        # ----- update accumulator -----
        acc += tl.dot(a, b)    # (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------
    # write back C
    # -----------------------------------------------------------------
    c_ptrs = c_base + \
             offs_m[:, None] * stride_c_m + \
             offs_n[None, :] * stride_c_n
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N]  (may be all‑ones)
        - weights      : dict of model parameters (see docstring)
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim]
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 1) LayerNorm on the last dimension
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Gating scalars (sigmoid) – also linear without bias
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Apply mask (if provided) and gates
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)                     # [B, N, N, 1]
        left = left * mask
        right = right * mask
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Prepare tensors for the Triton GEMM kernel
    #    We compute: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    equivalently: for each channel d we perform a matrix multiply
    #    A = left[..., d]   (N×N) and   Bᵀ = right[..., d]ᵀ.
    #    By permuting to (B, D, N, N) we can treat (B*D) as the batch dim.
    # -----------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # (B, N, N, D) -> (B, D, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, D, N, _ = left_perm.shape                       # D == hidden_dim
    total_batch = B * D                                # batch of independent GEMMs

    # view as [total_batch, N, N]
    left_mat = left_perm.view(total_batch, N, N)
    right_mat = right_perm.view(total_batch, N, N)     # stored as (j, k)

    # Strides for the kernel (they are the same for left and right because both are row‑major)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()   # (N*N, N, 1)
    # For the transposed B we need: stride over K (row) = original column stride (=1)
    # and stride over N (col) = original row stride (=N)
    stride_b_batch, stride_b_n, stride_b_k = right_mat.stride()  # (N*N, N, 1)
    # Dispatch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    out_mat = torch.empty_like(left_mat)   # fp16 output

    trimul_gemm_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        total_batch,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,          # K‑stride for transposed B (original column stride)
        stride_b_n,          # N‑stride for transposed B (original row stride)
        out_mat.stride(0),
        out_mat.stride(1),
        out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim] (fp32 for following ops)
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 7) Post‑processing: LayerNorm, out‑gate, final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # final projection to dim

    return out
