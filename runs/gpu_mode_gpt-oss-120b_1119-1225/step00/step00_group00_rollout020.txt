def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # LayerNorm parameters
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    # Compute mean/var across dim channel using torch
    # shape [bs,N,N,dim]
    mu = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mu) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    
    # Apply mask if available (mask is included in data)
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1)  # [bs,N,N,1]
        left = left * mask_expanded
        right = right * mask_expanded

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    
    left = left * left_gate
    right = right * right_gate

    # Optionally cast to fp16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # allocate out buffer
    out_raw = torch.empty((bs, N, N, hidden_dim), device=input_tensor.device, dtype=torch.float32)

    # call triton kernel
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (bs * hidden_dim, (N + BLOCK_I - 1)//BLOCK_I, (N + BLOCK_J - 1)//BLOCK_J)

    trimul_kernel[grid](
        left, right, out_raw,
        bs, N, hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_raw.stride(0), out_raw.stride(1), out_raw.stride(2), out_raw.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        D=hidden_dim,
    )

    # now to_out_norm and gating
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out_raw, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    out = out_norm * out_gate

    # final projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward pass implementation.

The kernel computes
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where `left` and `right` are the linearly projected (and gated) pairwise
representations of shape (B, N, N, D_hidden).  The computation is performed
with a Triton kernel that treats each (batch, hidden) pair as an independent
program instance and carries out a blocked matrix‑multiply (i‑x‑k) × (k‑x‑j).

All other operations (LayerNorm, linear projections, sigmoid gates,
final LayerNorm and projection) are done with PyTorch for simplicity.
The kernel works with arbitrary batch size (1‑2), sequence length up to 1024
and hidden dimensions 128 / 384.  It uses fp32 arithmetic (inputs are
fp32) and stores the result also in fp32.
"""
import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the O(N³) tensor contraction
# ----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr,          # const float*   (B,N,N,D_hidden)
    right_ptr,         # const float*   (B,N,N,D_hidden)
    out_ptr,           # float*         (B,N,N,D_hidden)

    B, N, D_hidden,    # runtime scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    D: tl.constexpr,          # D_hidden (compile‑time constant)
    NUM_K: tl.constexpr,      # = ceil(N / BLOCK_K)
):
    """
    Each program instance handles a single (batch, hidden) pair.
    The i‑j plane is tiled with BLOCK_I × BLOCK_J blocks.
    """
    # --------------------------------------------------------------
    # decode program ids
    # --------------------------------------------------------------
    pid0 = tl.program_id(0)               # combined batch‑hidden axis
    pid_i = tl.program_id(1)               # i‑block
    pid_j = tl.program_id(2)               # j‑block

    b = pid0 // D                         # batch index
    d = pid0 % D                          # hidden‑channel index

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the i‑j tile (FP32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension k
    # --------------------------------------------------------------
    for k_iter in range(NUM_K):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (i, k) ---------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + d * stride_left_d
        )
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # (BLOCK_I, BLOCK_K)

        # ----- load right tile (k, j) --------------------------------
        # right is indexed as (b, j, k, d); we need it as (k, j)
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + d * stride_right_d
        )
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)          # (BLOCK_K, BLOCK_J)

        # ----- multiply‑accumulate ------------------------------------
        acc += tl.dot(left, right)                     # (BLOCK_I, BLOCK_J)

    # --------------------------------------------------------------
    # write the result
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + d * stride_out_d
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper: LayerNorm without bias (bias is supplied separately)
# ----------------------------------------------------------------------
def _layer_norm(x, weight, bias, eps=1e-5):
    # x : (..., C)
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    return x_norm * weight + bias

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
    )   # (B, N, N, dim)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape [B, N, N] -> broadcast to [..., 1]
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core O(N³) contraction via Triton
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous and on the same device
    left = left.contiguous()
    right = right.contiguous()
    out_raw = torch.empty((bs, seq_len, seq_len, hidden_dim),
                          device=input_tensor.device,
                          dtype=torch.float32)

    # Block sizes – tuned for H100, fp32
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Number of K‑tiles (compile‑time constant for the kernel)
    NUM_K = (seq_len + BLOCK_K - 1) // BLOCK_K

    # Grid: (batch * hidden), i‑tiles, j‑tiles
    grid = (
        bs * hidden_dim,
        (seq_len + BLOCK_I - 1) // BLOCK_I,
        (seq_len + BLOCK_J - 1) // BLOCK_J,
    )

    trimul_out_kernel[grid](
        left,
        right,
        out_raw,
        bs,
        seq_len,
        hidden_dim,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        out_raw.stride(0), out_raw.stride(1), out_raw.stride(2), out_raw.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        D=hidden_dim,
        NUM_K=NUM_K,
    )
    torch.cuda.synchronize()  # make sure the kernel is finished

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gate
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_raw,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
