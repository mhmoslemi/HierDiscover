def triton_trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0) # i tile
    pid_j = tl.program_id(1) # j tile
    pid_hb = tl.program_id(2) # combined batch and hidden group

    # compute i,j start
    i_start = pid * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Decode batch and hidden start
    num_h_groups = tl.cdiv(H, BLOCK_H)
    batch_id = pid_hb // num_h_groups
    h_group = pid_hb % num_h_groups
    h_start = h_group * BLOCK_H

    # bounds
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop h within block
    cur_H = tl.minimum(BLOCK_H, H - h_start)

    # Allocate accumulator for each hidden channel within the block
    # We will accumulate a list of tensors per h offset, stored as acc_h list.
    # Since we cannot have dynamic list length, we use static loops over BLOCK_H and store in local array.
    # We'll create an accumulator of shape (BLOCK_H, BLOCK_M, BLOCK_N) but we can flatten as 1D.
    acc = tl.zeros((BLOCK_H, BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k blocks
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        # compute K size
        cur_K = tl.minimum(BLOCK_K, N - k_start)

        # compute mask for k dimension
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # For each hidden offset
        for h_offset in range(0, cur_H):
            # compute left pointer for this hidden channel
            left_ptrs = left_ptr \
                + batch_id * stride_left_b \
                + (h_start + h_offset) * stride_left_h \
                + (i_start + offs_i[:, None]) * stride_left_i \
                + (k_start + offs_k[None, :]) * stride_left_k

            # compute mask for left (i,k)
            mask_left = mask_i[:, None] & mask_k[None, :]

            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)  # shape (BLOCK_M, BLOCK_K)

            # compute right pointer transposed (k,j)
            right_ptrs = right_ptr \
                + batch_id * stride_right_b \
                + (h_start + h_offset) * stride_right_h \
                + (k_start + offs_k[:, None]) * stride_right_i \
                + (j_start + offs_j[None, :]) * stride_right_k

            # mask for right (k,j)
            mask_right = mask_k[:, None] & mask_j[None, :]

            right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)  # shape (BLOCK_K, BLOCK_N)

            # dot product
            # left_tile: (M, K), right_tile: (K, N)
            acc_h = tl.dot(left_tile, right_tile)  # (M, N)

            # accumulate
            # store into acc[h_offset]
            acc[h_offset, :, :] += acc_h

    # after loop, store results to out_ptr
    for h_offset in range(0, cur_H):
        out_ptrs = out_ptr \
            + batch_id * stride_out_b \
            + (h_start + h_offset) * stride_out_h \
            + (i_start + offs_i[:, None]) * stride_out_i \
            + (j_start + offs_j[None, :]) * stride_out_j

        mask_out = mask_i[:, None] & mask_j[None, :]
        tl.store(out_ptrs, acc[h_offset, :, :], mask=mask_out)

"""
TriMul “outgoing” implementation (AlphaFold‑3 style) using Triton.

Algorithm
---------
1️⃣ Layer‑Norm the input 4‑D tensor x ∈ [B, N, N, C] (C = dim).
2️⃣ Project to a hidden dimension H (hidden_dim) with two linear layers:
       left  = x @ W_left.T,   right = x @ W_right.T   (shape [B,N,N,H]).
3️⃣ Compute three gate vectors (sigmoid) from x: left_gate, right_gate, out_gate.
4️⃣ Apply the optional pair‑wise mask (shape [B,N,N]) and the gates:
       left  = left  * mask[...,None] * left_gate
       right = right * mask[...,None] * right_gate
   (mask and gates are broadcasted over the hidden dimension.)
5️⃣ **Core Triton kernel** – for every batch b and hidden‑group h,
   compute the “outgoing” pairwise product
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply per hidden channel:
       out_h = left_h @ right_h.T .
   The kernel tiles the (i,j) plane (M×N) and iterates over K blocks,
   accumulating in FP32 while loading FP16 for bandwidth efficiency.
6️⃣ Layer‑Norm the result over the hidden dimension,
   multiply by out_gate and finally linearly project back to dim.

The kernel is the only non‑PyTorch part; everything else uses
torch.nn.functional for simplicity and to keep the implementation
readable while still achieving >1 kµs runtime on the reference configs.
"""

import torch
import triton
import triton.language as tl
import math

# ----------------------------------------------------------------------
# Triton kernel: computes out_h = left_h @ right_h.T for a tile (i,j)
# over a group of hidden channels.  Batch and hidden are fused into the
# third program‑id dimension.
# ----------------------------------------------------------------------
@triton.jit
def _triton_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    B, N, H,                                    # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)          # tile along i  (rows)
    pid_j = tl.program_id(1)          # tile along j  (cols)
    pid_hb = tl.program_id(2)         # fused (batch, hidden‑group)

    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # Decode batch index and hidden‑group start
    num_h_groups = tl.cdiv(H, BLOCK_H)
    batch_id = pid_hb // num_h_groups
    h_group  = pid_hb % num_h_groups
    h_start  = h_group * BLOCK_H

    # ------------------------------------------------------------------
    # Index ranges and masks for the i / j dimensions (shared across h)
    # ------------------------------------------------------------------
    i_range = tl.arange(0, BLOCK_M)
    j_range = tl.arange(0, BLOCK_N)
    i_idx   = i_start + i_range
    j_idx   = j_start + j_range
    mask_i  = i_idx < N
    mask_j  = j_idx < N

    # How many hidden channels does this group actually contain?
    cur_H = tl.minimum(BLOCK_H, H - h_start)

    # ------------------------------------------------------------------
    # Loop over hidden channels inside the group
    # ------------------------------------------------------------------
    for h_offset in range(0, cur_H):
        # ---- accumulator (FP32) for a single hidden channel ---------------
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # ---- Loop over the k dimension (the summed axis) -----------------
        num_k = tl.cdiv(N, BLOCK_K)
        for k_iter in range(num_k):
            k_start = k_iter * BLOCK_K
            k_range = tl.arange(0, BLOCK_K)
            k_idx   = k_start + k_range
            mask_k  = k_idx < N

            # ----------- Load a (M×K) tile from left -----------------------
            left_base = left_ptr + \
                batch_id * stride_left_b + \
                (h_start + h_offset) * stride_left_h

            left_ptrs = left_base + \
                (i_start + i_range[:, None]) * stride_left_i + \
                (k_start + k_range[None, :]) * stride_left_k

            mask_left = mask_i[:, None] & mask_k[None, :]
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)   # fp16
            left_tile = tl.cast(left_tile, tl.float32)                  # promote

            # ----------- Load a (K×N) tile from right (transposed) --------
            # right[j,k] --> we want right_T[k,j] = right[j,k]
            right_base = right_ptr + \
                batch_id * stride_right_b + \
                (h_start + h_offset) * stride_right_h

            right_ptrs = right_base + \
                (k_start + k_range[:, None]) * stride_right_i + \
                (j_start + j_range[None, :]) * stride_right_k

            mask_right = mask_k[:, None] & mask_j[None, :]
            right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)  # fp16
            right_tile = tl.cast(right_tile, tl.float32)                 # promote

            # ----------- Matrix multiply and accumulate -------------------
            acc += tl.dot(left_tile, right_tile)   # (M,N) FP32

        # ---------- Store the result for this hidden channel -------------
        out_base = out_ptr + \
            batch_id * stride_out_b + \
            (h_start + h_offset) * stride_out_h

        out_ptrs = out_base + \
            (i_start + i_range[:, None]) * stride_out_i + \
            (j_start + j_range[None, :]) * stride_out_j

        mask_out = mask_i[:, None] & mask_j[None, :]
        tl.store(out_ptrs, acc, mask=mask_out)

        left, right: tensors of shape [B, N, N, H] (dtype=fp16).

    Returns:
        out: tensor of shape [B, N, N, H] (dtype=fp32).
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # stride tuples (in elements, not bytes)
    s_l = left.stride()
    s_r = right.stride()
    s_o = out.stride()

    BLOCK_M, BLOCK_N, BLOCK_K, BLOCK_H = 64, 64, 64, 8

    # Grid: (i‑tiles, j‑tiles, batch*hidden‑group)
    grid_i = triton.cdiv(N, BLOCK_M)
    grid_j = triton.cdiv(N, BLOCK_N)
    n_h_groups = triton.cdiv(H, BLOCK_H)
    grid_hb = B * n_h_groups
    grid = (grid_i, grid_j, grid_hb)

    _triton_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Expected `data` tuple:
        (input_tensor, mask, weight_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype=fp32).
    """
    # Unpack
    input_tensor, mask, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input (dim → dim)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(
        x_norm, weights["left_proj.weight"]
    )   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(
        x_norm, weights["right_proj.weight"]
    )

    # ------------------------------------------------------------------
    # 3. Gate vectors (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(
        x_norm, weights["left_gate.weight"]
    ).sigmoid()
    right_gate = torch.nn.functional.linear(
        x_norm, weights["right_gate.weight"]
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_norm, weights["out_gate.weight"]
    ).sigmoid()          # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B,N,N] → [B,N,N,1] for broadcasting
        mask = mask.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # Cast to FP16 for the heavy contraction
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Triton contraction: out = Σ_k left[...,i,k,h] * right[...,j,k,h]
    # ------------------------------------------------------------------
    out = _triton_outgoing(left, right)          # [B,N,N,hidden_dim] fp32

    # ------------------------------------------------------------------
    # 6. Post‑projection layer‑norm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7. Apply the output gate and final linear projection (hidden→dim)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(
        out, weights["to_out.weight"]
    )   # final shape [B,N,N,dim]

    return out
