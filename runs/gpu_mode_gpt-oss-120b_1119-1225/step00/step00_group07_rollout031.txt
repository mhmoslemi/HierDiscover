"""
TriMul (outgoing) – fast forward pass

This implementation follows the reference PyTorch module but
accelerates the two heaviest phases:

*  Element‑wise masking & gating of the left/right projected tensors.
  A tiny Triton kernel computes `out = x * mask * gate` on a flattened
  1‑D view, keeping the whole operation on‑GPU without any Python loops.

*  The 4‑D contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is rewritten as a batched matrix multiplication.
  By reshaping the tensors to (B·H, N, N) we obtain
      out_reshaped = left_reshaped @ right_reshaped.T
  which `torch.bmm` computes with cuBLAS (Tensor‑Core) at near‑peak
  H100 bandwidth.  This yields the same result as the `einsum`
  in the reference code while being far more cache‑friendly.

The rest of the module (LayerNorms, Linear layers, sigmoids) stays in
pure PyTorch for clarity.  The final output dtype is float32, as required.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication x * mask * gate
# ----------------------------------------------------------------------
@triton.jit
def mul_mask_gate_kernel(
    x_ptr,                 # input tensor (flattened)
    mask_ptr,              # mask tensor (flattened, broadcasted)
    gate_ptr,              # gate tensor (flattened)
    out_ptr,               # output tensor (flattened)
    n_elems,               # total number of elements
    BLOCK: tl.constexpr,   # work‑group size
):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    valid = offs < n_elems

    x = tl.load(x_ptr + offs, mask=valid)
    m = tl.load(mask_ptr + offs, mask=valid)
    g = tl.load(gate_ptr + offs, mask=valid)

    tl.store(out_ptr + offs, x * m * g, mask=valid)


def apply_mask_and_gate(x: torch.Tensor, mask: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """
    Fuse `x = x * mask * gate` using the Triton kernel above.
    All tensors must be contiguous, have the same shape and dtype.
    """
    if mask is None:
        # mask is optional – just apply the gate
        return x * gate

    # Ensure contiguous layout for 1‑D flattening
    x_flat = x.contiguous().view(-1)
    mask_flat = mask.contiguous().view(-1)        # mask already has trailing dim = 1
    gate_flat = gate.contiguous().view(-1)

    out_flat = torch.empty_like(x_flat)
    total = x_flat.numel()
    BLOCK = 1024  # a good trade‑off for H100

    grid = lambda meta: ((total + meta["BLOCK"] - 1) // meta["BLOCK"],)
    mul_mask_gate_kernel[grid](
        x_flat, mask_flat, gate_flat, out_flat,
        total,
        BLOCK=BLOCK,
    )
    return out_flat.view_as(x)


# ----------------------------------------------------------------------
# Main entry point – custom Kernel
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  float32
        - mask         : [B, N, N]      (optional, can be all‑ones)
        - weights      : dict of model weights
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating networks (sigmoid) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # used later

    # ------------------------------------------------------------------
    # 4. Optional mask (shape [B,N,N]) → broadcasted to [B,N,N,1]
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype)  # ensure same dtype as tensors
        mask = mask.unsqueeze(-1)  # [B,N,N,1]

    # ------------------------------------------------------------------
    # 5. Fuse mask and gating using a tiny Triton kernel (still stays in fp16)
    # ------------------------------------------------------------------
    # Cast to half precision for the heavy part
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    if mask is not None:
        mask = mask.to(torch.float16)

    # left = left * mask * left_gate
    left = apply_mask_and_gate(left, mask, left_gate)
    # right = right * mask * right_gate
    right = apply_mask_and_gate(right, mask, right_gate)

    # ------------------------------------------------------------------
    # 6. Batched matrix multiplication that implements the einsum
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    We treat the hidden dimension as an extra batch dimension.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (BH, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (BH, N, N)

    # Perform batched GEMM: (BH, N, N) @ (BH, N, N)^T  -> (BH, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 7. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # apply outgoing gate (broadcasted over hidden_dim)
    out = out * out_gate

    # final linear projection back to `dim`
    output = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return output
