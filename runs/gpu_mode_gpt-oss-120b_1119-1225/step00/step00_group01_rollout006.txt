"""
custom_kernel – Triton‑accelerated outgoing TriMul (Triangular Multiplicative Update)

The reference PyTorch implementation consists of
  1. LayerNorm over the last dimension (dim)
  2. Two linear projections (left/right) and three gating projections
  3. A dense N×N×hidden contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
     which is mathematically a batch of GEMMs:
        out_{b,d} = L_{b,d} @ R_{b,d}^T   with L,R ∈ ℝ^{N×N}
  4. LayerNorm over the hidden dimension, an output gate and a final linear
     projection back to the original dimensionality.

Step‑3 dominates the runtime (O(N³·hidden_dim)).  We off‑load it to a
custom Triton kernel that treats each (batch, hidden) pair as an
independent GEMM.  The kernel performs block‑wise matrix multiplication
with accumulation in FP32 (inputs are FP32).  All other operations
(LayerNorm, Linear, sigmoid) are executed with PyTorch kernels.

The kernel works for any batch size B∈{1,2}, sequence length N∈{128…1024}
and hidden_dim (the “hidden” channel) = config[“hidden_dim”] (the tests use
128).  The implementation is fully self‑contained and returns a
float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel:  out = left @ right^T   (left/right : (B*H, N, N))
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    a_ptr,                     # const float*  left   (B·H, N, N)
    b_ptr,                     # const float*  right  (B·H, N, N)
    c_ptr,                     #       float*  out    (B·H, N, N)
    batch, M, N, K,            # sizes (int)
    stride_a_batch, stride_a_m, stride_a_k,   # strides for a
    stride_b_batch, stride_b_n, stride_b_k,   # strides for b (original N×K layout)
    stride_c_batch, stride_c_m, stride_c_n,   # strides for c
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C = A @ Bᵀ for many independent matrices.
    A : (batch, M, K)      (here M = K = N)
    B : (batch, N, K)      (the same shape as A, but we read it as (K, N) to apply the transpose)
    C : (batch, M, N)
    """
    pid = tl.program_id(0)

    # -------------------------------------------------
    # 1) Decode the 3‑D program index (batch, block_m, block_n)
    # -------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)

    batch_idx = pid // (num_m * num_n)                     # which (batch * hidden) matrix
    pid_mn = pid % (num_m * num_n)
    pid_m = pid_mn // num_n
    pid_n = pid_mn % num_n

    # -------------------------------------------------
    # 2) Compute the global offsets for this block
    # -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / columns
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to the start of the current block
    a_ptrs = a_ptr + batch_idx * stride_a_batch \
                     + offs_m[:, None] * stride_a_m \
                     + offs_k[None, :] * stride_a_k

    # NOTE: we load B as (K, N) => this implements the transpose on‑the‑fly
    b_ptrs = b_ptr + batch_idx * stride_b_batch \
                     + offs_k[:, None] * stride_b_k \
                     + offs_n[None, :] * stride_b_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # 3) Loop over the K dimension in BLOCK_K chunks
    # -------------------------------------------------
    for k_iter in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_iter * BLOCK_K
        k_valid = (cur_k + offs_k) < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & k_valid[None, :],
                    other=0.0)   # (BM, BK)
        b = tl.load(b_ptrs,
                    mask=mask_n[None, :] & k_valid[:, None],
                    other=0.0)   # (BK, BN)

        acc += tl.dot(a, b)          # (BM, BN)

        # advance the pointers to the next K‑chunk
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # -------------------------------------------------
    # 4) Write the result
    # -------------------------------------------------
    c_ptrs = c_ptr + batch_idx * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point used by the test harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N]  (may be None)
        - weights      : dict of torch.Tensor with the model parameters
        - config       : dict containing at least
                         {"dim": int, "hidden_dim": int, "nomask": bool (optional)}
    Returns
    -------
    torch.Tensor  [B, N, N, dim]  (float32)
    """
    # ------------------------------------------------------------------- #
    # 0) Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    B, N, _, D = input_tensor.shape
    assert D == dim, "Input channel dim does not match config['dim']"

    # ------------------------------------------------------------------- #
    # 1) LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------- #
    eps = 1e-5
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # 2) Linear projections + gating (performed with PyTorch)
    # ------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional mask (broadcast over hidden dimension)
    if not nomask and mask is not None:
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating values (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 3) Heavy contraction: out = Σ_k left[...,k,:] * right[...,k,:]
    #    Implemented as a batched GEMM with a Triton kernel.
    # ------------------------------------------------------------------- #
    # Re‑order to (B, hidden, N, N) so that the hidden dimension becomes the
    # second axis; then flatten B*hidden → batch for the kernel.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_h = B * hidden_dim
    # reshape to (batch_h, N, N)
    left_mat = left_perm.view(batch_h, N, N)
    right_mat = right_perm.view(batch_h, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat, device=device)

    # Strides (in elements, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_m = (N + BLOCK_M - 1) // BLOCK_M
    num_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = batch_h * num_m * num_n
    grid = (total_programs,)

    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_h,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_n,
        stride_b_k,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 4) Output LayerNorm, gating, and final linear projection
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                     # element‑wise gating
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
