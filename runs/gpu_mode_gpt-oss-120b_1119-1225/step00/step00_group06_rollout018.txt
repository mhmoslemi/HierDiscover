import torch
import triton
import triton.language as tl

@triton.jit
def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_ok,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = total_i * total_j
    b = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    i_tile = block_idx // total_j
    j_tile = block_idx % total_j

    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k]
        left_offset = b * stride_lb + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptr + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # right_T[k, j] = right[j, k]
        right_offset = b * stride_rb + offs_k[:, None] * stride_ri + offs_j[None, :] * stride_rk
        right_t = tl.load(right_ptr + right_offset, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(left, right_t)

    out = acc.to(tl.float16)
    out_offset = b * stride_ob + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_ok
    tl.store(out_ptr + out_offset, out, mask=mask_i[:, None] & mask_j[None, :])

    Triton-accelerated forward pass of the TriMul operator (outgoing version).
    Performs:
      - LayerNorm on input,
      - Linear projections left/right and gates,
      - Mask application,
      - Elementwise gating,
      - Batch matrix multiplication left @ right^T using a fused Triton kernel,
      - Output LayerNorm, out-gate, and final linear projection,
      returning a float32 tensor of shape [bs, seq_len, seq_len, dim].
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Cast input to fp16 for compute
    x = input_tensor.to(torch.float16)

    # LayerNorm
    norm_weight = weights["norm.weight"].to(x.dtype)
    norm_bias = weights["norm.bias"].to(x.dtype)
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # linear projections
    left_proj_weight = weights["left_proj.weight"].to(x.dtype)
    right_proj_weight = weights["right_proj.weight"].to(x.dtype)
    left = torch.nn.functional.linear(x, left_proj_weight)   # [b, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # optional mask
    if not nomask:
        # mask shape [b,N,N], broadcast over hidden dim
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # gates
    left_gate_weight = weights["left_gate.weight"].to(x.dtype)
    right_gate_weight = weights["right_gate.weight"].to(x.dtype)
    out_gate_weight = weights["out_gate.weight"].to(x.dtype)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Prepare flattened tensors for Triton kernel
    bsz, N, _, H = left.shape
    batch = bsz * H

    # Permute to [bsz, H, N, N] and view as [batch, N, N]
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(batch, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch, N, N)

    # Release original large tensors to save memory
    del left, right, x

    out_flat = torch.empty_like(left_flat)  # fp16

    # Get strides
    stride_l_batch = left_flat.stride(0)
    stride_l_i = left_flat.stride(1)
    stride_l_k = left_flat.stride(2)

    stride_r_batch = right_flat.stride(0)
    stride_r_i = right_flat.stride(1)
    stride_r_k = right_flat.stride(2)

    stride_o_batch = out_flat.stride(0)
    stride_o_i = out_flat.stride(1)
    stride_o_k = out_flat.stride(2)

    # Define block sizes (tune for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch * tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N),)

    triton_trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_i, stride_r_k,
        stride_o_batch, stride_o_i, stride_o_k,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # Reshape back to original layout [bsz, N, N, H]
    out = out_flat.view(bsz, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"].to(out.dtype)
    to_out_norm_bias = weights["to_out_norm.bias"].to(out.dtype)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5)

    # Apply out_gate (broadcast over last dim)
    out = out * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"].to(out.dtype)
    out = torch.nn.functional.linear(out, to_out_weight)

    # Cast to float32 for final output
    return out.to(torch.float32)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

The heavy part of the operation is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication
    left_d[b] @ right_d[b].T  for each hidden dimension d.
We fuse this across the hidden dimension by flattening (batch, d) into a
single batch dimension and compute the matrix multiplication with a custom
Triton kernel.  All other operations (LayerNorm, linear projections,
sigmoids, masking) are performed with PyTorch in fp16; the kernel
accumulates in fp32 and stores the result back in fp16.  The final
output is cast to fp32.

The kernel is tuned for H100 (BLOCK_M=N_BLOCK=64, BLOCK_K=32,
num_warps=8) and works for the shapes used in the test suite.
"""

import math
import torch
import triton
import triton.language as tl

@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N,
    stride_lb, stride_li, stride_lk,          # left: batch, i, k
    stride_rb, stride_ri, stride_rk,          # right: batch, i(=j), k
    stride_ob, stride_oi, stride_ok,          # out:   batch, i, j
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = Σ_k left[:,i,k] * right[:,j,k] for every batch."""
    pid = tl.program_id(0)

    # number of tiles in each dimension
    total_i = tl.cdiv(N, BLOCK_M)
    total_j = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = total_i * total_j

    # decode batch and tile indices
    b   = pid // blocks_per_batch
    bid = pid % blocks_per_batch
    i_tile = bid // total_j
    j_tile = bid % total_j

    # offsets for the output tile
    offs_i = i_tile * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_tile * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = kk * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i, k] -> shape (M, K)
        left_offset = (
            b * stride_lb
            + offs_i[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(
            left_ptr + left_offset,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )

        # right_T[k, j] = right[j, k] -> shape (K, N)
        right_offset = (
            b * stride_rb
            + offs_k[:, None] * stride_ri
            + offs_j[None, :] * stride_rk
        )
        right_T = tl.load(
            right_ptr + right_offset,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0),
        )

        # accumulate the dot product (M,K) * (K,N) -> (M,N)
        acc += tl.dot(left, right_T)

    # write the result in fp16
    out = acc.to(tl.float16)
    out_offset = (
        b * stride_ob
        + offs_i[:, None] * stride_oi
        + offs_j[None, :] * stride_ok
    )
    tl.store(out_ptr + out_offset, out, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Args:
        data: tuple (input, mask, weights, config)
            - input: torch.Tensor [bs, N, N, dim]
            - mask : torch.Tensor [bs, N, N] (may be all‑ones)
            - weights: dict with all linear / layer‑norm parameters
            - config: dict containing "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor [bs, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1. Normalisation and linear projections (all in fp16)
    # -----------------------------------------------------------------
    x = input_tensor.to(torch.float16)

    # LayerNorm over the last dimension
    norm_weight = weights["norm.weight"].to(x.dtype)
    norm_bias   = weights["norm.bias"].to(x.dtype)
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )  # [b, N, N, dim]

    # Projections
    left_proj_w = weights["left_proj.weight"].to(x.dtype)
    right_proj_w = weights["right_proj.weight"].to(x.dtype)
    left = torch.nn.functional.linear(x, left_proj_w)   # [b, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)

    # Optional mask (broadcast over the hidden dimension)
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [b, N, N, 1]
        left = left * mask
        right = right * mask

    # Gating values (sigmoid)
    left_gate_w  = weights["left_gate.weight"].to(x.dtype)
    right_gate_w = weights["right_gate.weight"].to(x.dtype)
    out_gate_w   = weights["out_gate.weight"].to(x.dtype)

    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 2. Triton kernel for the contraction Σ_k left * right
    # -----------------------------------------------------------------
    bsz, N, _, H = left.shape          # H == hidden_dim
    batch_flat = bsz * H                # (batch, hidden) merged

    # Move hidden dimension to the batch axis and make data contiguous
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(batch_flat, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(batch_flat, N, N)

    # Release tensors we no longer need to keep memory pressure low
    del left, right, x

    out_flat = torch.empty_like(left_flat)  # fp16 output

    # Strides for the flattened tensors (batch, i, k/j)
    stride_lb, stride_li, stride_lk = left_flat.stride()
    stride_rb, stride_ri, stride_rk = right_flat.stride()
    stride_ob, stride_oi, stride_ok = out_flat.stride()

    # Tile sizes tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid size: one program per (batch_flat, i_tile, j_tile)
    grid = (
        batch_flat
        * ((N + BLOCK_M - 1) // BLOCK_M)
        * ((N + BLOCK_N - 1) // BLOCK_N),
    )

    _trimul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        batch_flat,
        N,
        stride_lb, stride_li, stride_lk,
        stride_rb, stride_ri, stride_rk,
        stride_ob, stride_oi, stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # good utilisation on an H100
    )

    # Reshape back to [bs, N, N, H]
    out = (
        out_flat.view(bsz, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [b, N, N, H]

    # -----------------------------------------------------------------
    # 3. Output normalisation, gating and final projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(out.dtype)
    to_out_norm_b = weights["to_out_norm.bias"].to(out.dtype)
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5
    )
    out = out * out_gate  # element‑wise gate

    to_out_w = weights["to_out.weight"].to(out.dtype)  # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_w)    # [b, N, N, dim]

    # Return in float32 as required by the reference implementation
    return out.to(torch.float32)
