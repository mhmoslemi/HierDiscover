```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    
    # mask
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f
    
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    
    left = left * left_gate
    right = right * right_gate
    
    # Convert to fp16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)
    
    # Prepare for kernel: permute and reshape
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose k and j
    
    total_batch = B * hidden_dim
    left_reshaped = left_perm.view(total_batch, N, N)
    right_reshaped = right_perm.view(total_batch, N, N)
    
    # Allocate output
    out_reshaped = torch.empty_like(left_reshaped, dtype=torch.float32)
    
    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (total_batch, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_kernel[grid](
        left_reshaped, right_reshaped, out_reshaped,

        left_reshaped.stride(0), left_reshaped.stride(1), left_reshaped.stride(2),
        right_reshaped.stride(0), right_reshaped.stride(1), right_reshaped.stride(2),
        out_reshaped.stride(0), out_reshaped.stride(1), out_reshaped.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
```

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    total_batch, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # Use offsets for K dimension inside loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        k_idx = k_offset + tl.arange(0, BLOCK_K)

        # Compute pointer for left block (M x K)
        a_ptrs = left_ptr + pid_b * stride_left_batch + offs_m[:, None] * stride_left_m + k_idx[None, :] * stride_left_k
        a_mask = (offs_m[:, None] < M) & (k_idx[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Compute pointer for right block (K x N)
        b_ptrs = right_ptr + pid_b * stride_right_batch + k_idx[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        b_mask = (k_idx[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Store results
    c_ptrs = out_ptr + pid_b * stride_out_batch + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    total_batch, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # row and column offsets within the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over contraction dimension k
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = kb * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)

        # pointers for the left block (M x K)
        a_ptrs = left_ptr \
            + pid_b * stride_left_batch \
            + offs_m[:, None] * stride_left_m \
            + k_idx[None, :] * stride_left_k
        a_mask = (offs_m[:, None] < M) & (k_idx[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # pointers for the right block (K x N) – note the transpose (k as rows)
        b_ptrs = right_ptr \
            + pid_b * stride_right_batch \
            + k_idx[:, None] * stride_right_k \
            + offs_n[None, :] * stride_right_n
        b_mask = (k_idx[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # accumulate the product; tl.dot will promote float16 -> float32
        acc += tl.dot(a, b)

    # write back the result
    c_ptrs = out_ptr \
        + pid_b * stride_out_batch \
        + offs_m[:, None] * stride_out_m \
        + offs_n[None, :] * stride_out_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Triton‑based implementation of the outgoing TriMul operator.

    The function follows the reference PyTorch implementation:
    1. Layer‑normalize the input.
    2. Linear projections → left, right.
    3. Optional mask application.
    4. Gate values (sigmoid of linear projections) and element‑wise gating.
    5. Core N³ update computed by a custom Triton kernel that performs,
       for each hidden dimension, a batched matrix multiplication:
           out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    6. Layer‑norm on the hidden dimension, gating with out_gate,
       and a final linear projection back to the original channel dimension.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Apply optional mask (mask shape: [B, N, N])
    if mask is not None:
        # ensure mask has same dtype and broadcast over channel dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Compute gates (sigmoid of linear projections)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core N³ multiplicative update via Triton
    # Cast to half precision to reduce memory bandwidth.
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape  # left shape: [B, N, N, hidden_dim]

    # Move hidden dimension to the leading axis and make tensors contiguous.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, k, j)

    total_batch = B * hidden_dim
    left_reshaped = left_perm.view(total_batch, N, N)
    right_reshaped = right_perm.view(total_batch, N, N)

    # Allocate output buffer (float32 accumulator)
    out_reshaped = torch.empty_like(left_reshaped, dtype=torch.float32)

    # Block sizes – tuned for H100/FP16 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left_reshaped,          # pointer to left matrix
        right_reshaped,         # pointer to right (transposed) matrix
        out_reshaped,           # pointer to output
        total_batch,
        N, N, N,
        left_reshaped.stride(0),
        left_reshaped.stride(1),
        left_reshaped.stride(2),
        right_reshaped.stride(0),
        right_reshaped.stride(1),
        right_reshaped.stride(2),
        out_reshaped.stride(0),
        out_reshaped.stride(1),
        out_reshaped.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B, N, N, hidden_dim)
    out = out_reshaped.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # -----------------------------------------------------------------
    # 6. Post‑processing: hidden‑dim LayerNorm, gating, final Linear
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

# Triton implementation of the “outgoing” TriMul operator.
# -------------------------------------------------------
# The core operation is a three‑dimensional contraction:
#   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
# The implementation follows the PyTorch reference:
#   1. Layer‑norm the input.
#   2. Linear projections → left / right.
#   3. Optional mask application.
#   4. Gating (sigmoid of three additional linear layers) and element‑wise gating.
#   5. N³ update computed by a custom Triton kernel that treats each (batch, hidden)
#      pair as an independent matrix multiplication:
#          out_h = left_h @ (right_h)ᵀ
#   6. Hidden‑dim LayerNorm, out‑gate multiplication and final linear projection.
#
# The heavy N³ step is fused into a Triton kernel that operates on fp16 inputs
# (to cut memory traffic) while accumulating in fp32 for numerical safety.
# The kernel processes one (batch * hidden) “mini‑batch’’ at a time and
# tiles the M×N output matrix with BLOCK_M×BLOCK_N tiles, iterating over K
# with BLOCK_K tiles.  The kernel works for any sequence length N (including
# 1024) and any hidden dimension (128, 384 etc.).
#
# ----------------------------------------------------------------------
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    total_batch, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Fast batched matrix‑multiply for the TriMul contraction.

    Computes, for every batch index ``b`` (where a batch
    corresponds to a ``(original_batch, hidden)`` pair):
        out[b, i, j] = Σₖ left[b, i, k] * right[b, k, j]

    ``left`` is stored as (M×K) and ``right`` as (K×N) – i.e. ``right``
    is already transposed with respect to the original pairwise tensor.
    """
    pid_b = tl.program_id(0)          # which (batch, hidden) pair
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile column

    # Offsets inside the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the contraction dimension K
    for kb in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = kb * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)

        # ----- load left tile (M × K) -----
        a_ptrs = left_ptr \
            + pid_b * stride_left_batch \
            + offs_m[:, None] * stride_left_m \
            + k_idx[None, :] * stride_left_k
        a_mask = (offs_m[:, None] < M) & (k_idx[None, :] < K)
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)          # fp16

        # ----- load right tile (K × N) -----
        b_ptrs = right_ptr \
            + pid_b * stride_right_batch \
            + k_idx[:, None] * stride_right_k \
            + offs_n[None, :] * stride_right_n
        b_mask = (k_idx[:, None] < K) & (offs_n[None, :] < N)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)          # fp16

        # ----- multiply‑accumulate -----
        # tl.dot promotes fp16 → fp32 on H100 Tensor Cores
        acc += tl.dot(a, b)

    # Store the result tile
    c_ptrs = out_ptr \
        + pid_b * stride_out_batch \
        + offs_m[:, None] * stride_out_m \
        + offs_n[None, :] * stride_out_n
    c_mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=c_mask)

    out_mat = torch.empty_like(left_mat, dtype=torch.float32)
