"""
TriMul (outgoing) – Triton‑fused implementation.

The heavy part of TriMul is the Einstein‑summation:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batch of matrix multiplications
    out_d = left_d @ right_dᵀ      (for every hidden dimension d).

The kernel below computes this reduction in a fused
block‑wise fashion, handling the batch×hidden dimensions
as a single program‑id axis.  The rest of the module
(LayerNorms, Linear projections, gating, final projection)
is performed with PyTorch primitives for simplicity; the
key O(N³·H) work is off‑loaded to Triton.

The kernel works on fp16 inputs (left/right) and accumulates
in fp32 for numerical stability.  The final result is
returned as fp32 to match the original PyTorch module.
"""

import torch
import triton
import triton.language as tl
from torch import nn
import math


# ----------------------------------------------------------------------
# Triton kernel: batch‑hidden fused matmul (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *float16
    right_ptr,           # *float16
    out_ptr,             # *float16
    B, N, H,             # int32
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Program IDs:
        0 → i‑blocks (M dimension)
        1 → j‑blocks (N dimension)
        2 → batch * hidden index (P = B*H)
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_ph = tl.program_id(2)          # combined batch–hidden index

    # ------------------------------------------------------------------
    # Compute actual batch (b) and hidden (h) from the flattened index
    # ------------------------------------------------------------------
    b = pid_ph // H
    h = pid_ph % H

    # ------------------------------------------------------------------
    # Block offsets in i, j and k dimensions
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------
    for start_k in range(0, tl.cdiv(N, BLOCK_K)):
        k = start_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # --------------------------------------------------------------
        # Load left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        # --------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + i[:, None] * stride_l_i
            + k[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # --------------------------------------------------------------
        # Load right[b, j, k, h] as (k, j) for the dot product
        # --------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + k[:, None] * stride_r_k       # k dimension first
            + j[None, :] * stride_r_i       # then j dimension
            + h * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # --------------------------------------------------------------
        # Accumulate the matrix multiply (M×K)·(K×N) -> (M×N)
        # --------------------------------------------------------------
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    # Write the result back to out[b,i,j,h]
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + i[:, None] * stride_o_i
        + j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )

    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (dim → hidden_dim)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional external mask (broadcasted over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or numeric – convert to same dtype as x
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear transforms of x)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the heavy aggregation kernel
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Compute the core TriMul reduction with Triton
    # ------------------------------------------------------------------
    out_h = tri_mul_triton(left_h, right_h)          # fp16, shape [B,N,N,hidden_dim]

    # Bring back to fp32 for the following LayerNorms / linear
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 7️⃣ Output LayerNorm (hidden_dim)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply the output gate and final linear projection (hidden_dim → dim)
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # final [B,N,N,dim]

    return out
```
