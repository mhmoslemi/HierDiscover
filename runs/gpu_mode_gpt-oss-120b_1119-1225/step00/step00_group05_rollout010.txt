"""
TriMul (outgoing) forward implementation with Triton acceleration.

The heavy part of TriMul is the ternary einsum:
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
which is equivalent to a batch of matrix‑matrix multiplies
    out_d = left_d @ right_dᵀ   for each hidden channel d.

We fuse this N×N×N multiplication across the hidden dimension by
flattening (batch × hidden_dim) into a single batch axis and launching
a 3‑D Triton kernel (batch, block_m, block_n).  The kernel implements a
standard tiled GEMM (FP16 compute, FP32 accumulation) and writes the
result back as FP16, which is later cast to FP32 for the remaining
LayerNorm / linear steps (performed in PyTorch).

All other operations (LayerNorms, linear projections, gating, masking)
are carried out with PyTorch for simplicity and to keep memory layout
contiguous before the Triton call.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  where
#   A : [batch, M, K]  (float16)
#   B : [batch, N, K]  (float16)   – we pass B transposed as [batch, K, N]
#   C : [batch, M, N] (float16)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # pointers
    M, N, K,                                 # matrix dimensions
    stride_az, stride_am, stride_ak,         # A strides: batch, M, K
    stride_bz, stride_bk, stride_bn,         # B strides: batch, K, N  (B is transposed)
    stride_cz, stride_cm, stride_cn,         # C strides: batch, M, N
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched matmul kernel used by TriMul."""
    pid_z = tl.program_id(2)   # batch index (batch * hidden_dim)
    pid_m = tl.program_id(0)   # block row
    pid_n = tl.program_id(1)   # block col

    # ------------------------------------------------------------------
    # Compute offsets for this tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_ptr += pid_z * stride_az
    b_ptr += pid_z * stride_bz
    c_ptr += pid_z * stride_cz

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile (M x K) – row‑major
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile (K x N) – note B is already transposed, so we stride over K then N
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Cast to FP16 for the dot (dot operates on FP16 and accumulates FP32)
        a = a.to(tl.float16)
        b = b.to(tl.float16)

        acc += tl.dot(a, b)      # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write out the result tile
    # ------------------------------------------------------------------
    c = acc.to(tl.float16)
    c_ptrs = c_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data (Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]):
            - input_tensor: [B, N, N, dim] (float32)
            - mask:         [B, N, N] (bool or float, may be ignored if nomask)
            - weights:      model weight tensors
            - config:       contains "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on input (dim)
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype=left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triton accelerated ternary multiplication (TriMul)
    #    Compute out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]
    # ------------------------------------------------------------------
    # Reshape to (batch*hidden_dim, N, N) and cast to FP16 for the kernel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim

    left_flat = left_perm.view(batch_h, N, N).to(torch.float16)
    right_flat = right_perm.view(batch_h, N, N).to(torch.float16)

    # Right matrix needs to be transposed (K dimension first) for A @ Bᵀ
    right_T = right_flat.transpose(-2, -1)   # view with swapped strides

    # Output buffer (FP16)
    out_flat = torch.empty((batch_h, N, N), dtype=torch.float16, device=left_flat.device)

    # Compute strides for kernel (all in element counts, not bytes)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_T.stride(0)
    stride_b_k = right_T.stride(1)   # = 1 (since we transposed)
    stride_b_n = right_T.stride(2)   # = N (original K stride)

    stride_c_batch = out_flat.stride(0)
    stride_c_m = out_flat.stride(1)
    stride_c_n = out_flat.stride(2)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32   # a modest K tile works well for FP16 on H100

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    grid = (grid_m, grid_n, batch_h)   # (block_m, block_n, batch)

    _batched_matmul_kernel[grid](
        a_ptr=left_flat,
        b_ptr=right_T,
        c_ptr=out_flat,
        M=N,
        N=N,
        K=N,
        stride_az=stride_a_batch,
        stride_am=stride_a_m,
        stride_ak=stride_a_k,
        stride_bz=stride_b_batch,
        stride_bk=stride_b_k,
        stride_bn=stride_b_n,
        stride_cz=stride_c_batch,
        stride_cm=stride_c_m,
        stride_cn=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 and reshape to original layout [B, N, N, hidden_dim]
    out = out_flat.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # no bias as in reference

    return out
