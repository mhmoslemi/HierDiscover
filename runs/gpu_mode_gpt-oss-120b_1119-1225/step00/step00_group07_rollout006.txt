    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    nominal: B, N, N, _ = input_tensor.shape (assert shape)
    # LayerNorm: use torch.nn.functional.layer_norm
    norm_w = weights["norm.weight"]; norm_b = weights["norm.bias"]
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b, eps=eps)
    # Projections:
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight) 
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # mask
    if mask is not None:
        mask = mask.to(input_tensor.dtype).unsqueeze(-1)   # shape B,N,N,1
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    right = right_proj * right_gate

"""
TriMul (outgoing) – fast Triton implementation.

The forward pass follows the reference PyTorch version:

1. Layer‑norm over the last channel (dim).
2. Project the normalized tensor to a hidden dimension (left/right) and
   compute three sigmoid gates (left, right, out).
3. Apply optional pairwise mask, then element‑wise gate the projections.
4. Compute the triangular multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of `hidden_dim` matrix‑multiplications
   (left[b,d] @ right[b,d].T) for every batch element.
   The heavy N³ work is implemented in a fused Triton kernel
   (half‑precision loads, FP32 accumulation, FP32 output).
5. Layer‑norm over the hidden dimension, multiply by the out‑gate,
   and a final linear projection back to the original dimension.

Only the core O(N³) bilinear contraction is written in Triton; all other
operations use highly‑optimized PyTorch kernels.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel for the einsum: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
# -------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # Pointers
    left_ptr,       # [B, N, N, H]   (FP16)
    right_ptr,      # [B, N, N, H]   (FP16)
    out_ptr,        # [B, N, N, H]   (FP32)

    # Tensor meta‑data
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,   # left strides
    stride_rb, stride_rj, stride_rk, stride_rh,   # right strides
    stride_ob, stride_oi, stride_oj, stride_oh,   # out strides

    # Compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ---------------------------------------------------------------------
    # Program IDs
    #   axis 0 -> (batch, hidden) pair
    #   axis 1 -> i tile (M)
    #   axis 2 -> j tile (N)
    # ---------------------------------------------------------------------
    pid_bh = tl.program_id(0)
    pid_m  = tl.program_id(1)
    pid_n  = tl.program_id(2)

    total_bh = B * H
    if pid_bh >= total_bh:
        return

    # -------------------------------------------------------------
    # Translate pid_bh -> (b, h)
    # -------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # -------------------------------------------------------------
    # Tile offsets for the output matrix (i,j)
    # -------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -----------------------------------------------------------------
    # Base pointers for the current (b, h) slice
    # -----------------------------------------------------------------
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # -----------------------------------------------------------------
    # Accumulator in FP32
    # -----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k in tiles
    # -----------------------------------------------------------------
    num_tiles_k = tl.cdiv(N, BLOCK_K)
    for tile_k in range(0, num_tiles_k):
        offs_k = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load left tile: shape (BLOCK_M, BLOCK_K) -----
        # left[b,i,k,h]   -> stride_li (i), stride_lk (k)
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        a = tl.load(left_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16).to(tl.float32)

        # ----- Load right tile transposed: shape (BLOCK_K, BLOCK_N) -----
        # right[b,j,k,h] --> we need (k,j) order:
        #   stride_rk for k dimension, stride_rj for j dimension
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        b_tile = tl.load(right_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0,
                         dtype=tl.float16).to(tl.float32)

        # ----- Accumulate the dot product -----
        acc += tl.dot(a, b_tile)    # (M,K) @ (K,N) -> (M,N)

    # -----------------------------------------------------------------
    # Store the result (FP32)
    # -----------------------------------------------------------------
    out_ptrs = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# -------------------------------------------------------------------------
# Public entry point: custom_kernel
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor   [B, N, N, dim]  (float32)
        - mask         : torch.Tensor   [B, N, N]      (bool/float, may be None)
        - weights      : dict of named torch.Tensors
        - config       : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", False)

    device = input_tensor.device
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input layer‑norm over the channel dimension `dim`
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections to hidden dimension (no bias)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional pairwise mask
    # -----------------------------------------------------------------
    if mask is not None and not nomask:
        # mask is [B, N, N]; broadcast over hidden dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)        # [B,N,N,1]
        left  = left  * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating pathways (sigmoid)
    # -----------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply element‑wise gates
    left  = left  * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core TriMul contraction – Triton kernel
    #    Convert to half precision for the load, keep accumulation in FP32.
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # Cast to FP16 for bandwidth‑limited loads
    left_fp  = left.to(torch.float16)
    right_fp = right.to(torch.float16)

    # Output buffer (FP32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_lb, stride_li, stride_lk, stride_lh = left_fp.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right_fp.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Block sizes – tuned for H100 (64 is a good default)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    total_bh = B * hidden_dim
    grid = (total_bh,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_fp, right_fp, out,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk, stride_lh,
        stride_rb, stride_rj, stride_rk, stride_rh,
        stride_ob, stride_oi, stride_oj, stride_oh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8, num_stages=4
    )
    # Ensure kernel completion before we reuse tensors
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 6) Second layer‑norm over hidden dimension
    # -----------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 7) Apply out‑gate and final linear projection back to `dim`
    # -----------------------------------------------------------------
    out_norm = out_norm * out_gate          # point‑wise multiplication
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return output
```
