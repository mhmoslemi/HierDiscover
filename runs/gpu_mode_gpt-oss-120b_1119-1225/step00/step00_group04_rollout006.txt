    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", False)
    # Normalize input: use torch.nn.functional.layer_norm with weight/bias.
    # Note: weights might be of shape dim (weight) and dim (bias).
    # We compute normalized = (x - mean)/sqrt(var+eps) * weight + bias.
    # Use torch.nn.functional.layer_norm.
    eps = 1e-5  # typical
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [bs, N, N, hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if needed
    if not nomask:
        # mask shape [bs, N, N], unsqueeze last dim (for hidden) then convert to float
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates:
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # TriMul: compute out = left @ right^T for each batch and hidden dim.
    # Reshape to (B*hidden_dim, N, N)
    B, N, _, _ = left.shape
    left_reshape = left.permute(0,3,1,2).reshape(B * hidden_dim, N, N)  # shape (B*hd,N,N)
    right_reshape = right.permute(0,3,1,2).reshape(B * hidden_dim, N, N)  # shape (B*hd,N,N)
    out = torch.empty_like(left_reshape)  # allocate output shape (B*hd,N,N)

    # Launch Triton kernel
    # Determine strides
    # left stride: stride0 (batch*hidden), stride1 (M), stride2 (K)
    stride_left_b, stride_left_m, stride_left_k = left_reshape.stride()

@triton.jit
def tri_matmul_kernel(left_ptr, right_ptr, out_ptr,
                      total_batches, M, N, K,
                      stride_left_b, stride_left_m, stride_left_k,
                      stride_right_b, stride_right_n, stride_right_k,
                      stride_out_b, stride_out_m, stride_out_n,
                      BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # compute pid indices
    pid = tl.program_id(0)
    # total number of tiles per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # compute offsets for m and n within tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for valid indices
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k range offsets
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load left block: shape BLOCK_M x BLOCK_K
        left_ptrs = left_ptr + batch_id*stride_left_b + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right block transposed: we need right_T: shape BLOCK_K x BLOCK_N from right_ptr
        # right_ptr stored as (batch, N, K) (like right_reshape). We want values at (batch, j, k)
        # To get right_T's element at (k, j), we need offset: batch*stride + j*stride_right_n + k*stride_right_k
        # For tile we have j in offs_n, k in offs_k.
        right_ptrs = right_ptr + batch_id*stride_right_b + offs_n[:, None] * stride_right_n + offs_k[None, :] * stride_right_k
        # This loads shape BLOCK_N x BLOCK_K, but we need (BLOCK_K, BLOCK_N) orientation.
        # We'll load as shape (BLOCK_N, BLOCK_K) then transpose.
        right = tl.load(right_ptrs, mask=mask_n[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # Transpose right to shape (BLOCK_K, BLOCK_N)
        right_T = tl.transpose(right, (1,0))  # shape (BLOCK_K, BLOCK_N)
        # accumulate
        acc += tl.dot(left, right_T)

    # Write back
    out_ptrs = out_ptr + batch_id*stride_out_b + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

import triton
import triton.language as tl

@triton.jit
def tri_matmul_kernel(
    left_ptr,      # pointer to left matrix batch BxM xK (float16)
    right_ptr,     # pointer to right matrix batch BxN xK (float16)
    out_ptr,       # pointer to output matrix batch BxM xN (float16)
    batch, M, N, K,    # total_batch (B*C), dimensions
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # Compute number of tiles per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # Determine which batch and block we are processing
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer base for this batch
    batch_offset_left = batch_idx * stride_left_batch
    batch_offset_right = batch_idx * stride_right_batch
    batch_offset_out = batch_idx * stride_out_batch

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left block (M x K)
        left_ptrs = left_ptr + batch_offset_left + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed (K x N)
        # right stored as (batch, N, K)
        right_ptrs = right_ptr + batch_offset_right + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(left, right)  # (M,K) @ (K,N) -> (M,N)

    # Store result
    out_ptrs = out_ptr + batch_offset_out + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy part of the TriMul operation is the triple product

    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]

which can be expressed as a batched matrix multiplication:
    out_d = left_d @ right_dᵀ   for each hidden dimension d.

The kernel below performs a fused batched GEMM (B·hidden_dim independent
matmuls) in half‑precision while accumulating in fp32. The surrounding
logic (LayerNorms, linear projections, gating, masking, final projection)
is performed with regular PyTorch ops.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication left @ rightᵀ
# ----------------------------------------------------------------------
@triton.jit
def tri_matmul_kernel(
    left_ptr,            # fp16 *  (batch_h* N * K)
    right_ptr,           # fp16 *  (batch_h* N * K)  (original right, not transposed)
    out_ptr,             # fp16 *  (batch_h* N * N)
    batch, M, N, K,      # int32 scalars
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_n, stride_right_k,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # Number of tiles per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # Resolve batch index and tile coordinates
    batch_idx = pid // (num_blocks_m * num_blocks_n)
    tile_idx  = pid %  (num_blocks_m * num_blocks_n)
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx %  num_blocks_n

    # Offsets inside the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Offsets for the current batch
    batch_off_left  = batch_idx * stride_left_batch
    batch_off_right = batch_idx * stride_right_batch
    batch_off_out   = batch_idx * stride_out_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # -------- load left block (M x K) ----------
        left_ptrs = (left_ptr + batch_off_left
                     + offs_m[:, None] * stride_left_m
                     + offs_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)  # fp16 -> fp32

        # -------- load right block transposed (K x N) ----------
        # right is stored as (batch, N, K); we read it as (K, N)
        right_ptrs = (right_ptr + batch_off_right
                      + offs_k[:, None] * stride_right_k
                      + offs_n[None, :] * stride_right_n)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)  # fp16 -> fp32

        # -------- multiply‑accumulate ----------
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # Store back in fp16
    out_ptrs = (out_ptr + batch_off_out
                + offs_m[:, None] * stride_out_m
                + offs_n[None, :] * stride_out_n)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entrypoint used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data (tuple):
            - input_tensor : torch.Tensor of shape [B, N, N, D]
            - mask         : torch.Tensor of shape [B, N, N] (or None)
            - weights      : dict of model weight tensors
            - config       : dict containing at least 'dim' and 'hidden_dim'
    Returns:
        torch.Tensor of shape [B, N, N, D] (float32, on the same device)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    eps = 1e-5

    # --------------------------------------------------------------
    # Load weights
    # --------------------------------------------------------------
    norm_weight           = weights["norm.weight"]
    norm_bias             = weights["norm.bias"]
    left_proj_weight      = weights["left_proj.weight"]
    right_proj_weight     = weights["right_proj.weight"]
    left_gate_weight      = weights["left_gate.weight"]
    right_gate_weight     = weights["right_gate.weight"]
    out_gate_weight       = weights["out_gate.weight"]
    to_out_norm_weight    = weights["to_out_norm.weight"]
    to_out_norm_bias      = weights["to_out_norm.bias"]
    to_out_weight         = weights["to_out.weight"]

    # --------------------------------------------------------------
    # LayerNorm on the input
    # --------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight,
                     bias=norm_bias, eps=eps)

    # --------------------------------------------------------------
    # Linear projections (float32)
    # --------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hd]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # Masking (if required)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_f = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # Gates (float32) – sigmoid applied after linear
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # --------------------------------------------------------------
    # Apply per‑element gates (cast to fp16 for the heavy GEMM)
    # --------------------------------------------------------------
    left  = (left  * left_gate).to(torch.float16)
    right = (right * right_gate).to(torch.float16)

    # --------------------------------------------------------------
    # Reshape to (B*hd, N, N) for batched GEMM
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    left  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # --------------------------------------------------------------
    # Allocate output buffer (fp16)
    # --------------------------------------------------------------
    out_fp16 = torch.empty((B * hidden_dim, N, N), dtype=torch.float16,
                           device=device)

    # --------------------------------------------------------------
    # Launch the Triton kernel
    # --------------------------------------------------------------
    # Block sizes – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    stride_left_batch, stride_left_m, stride_left_k = left.stride()
    stride_right_batch, stride_right_n, stride_right_k = right.stride()
    stride_out_batch,  stride_out_m,  stride_out_n   = out_fp16.stride()

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_tiles  = B * hidden_dim * num_blocks_m * num_blocks_n

    grid = (total_tiles,)

    tri_matmul_kernel[grid](
        left, right, out_fp16,
        B * hidden_dim, N, N, N,
        stride_left_batch, stride_left_m, stride_left_k,
        stride_right_batch, stride_right_n, stride_right_k,
        stride_out_batch, stride_out_m, stride_out_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Convert back to fp32 and restore original layout
    # --------------------------------------------------------------
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N)
    out = out.permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # Post‑projection LayerNorm, gating and final linear
    # --------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # final projection to [B,N,N,dim]

    return out
