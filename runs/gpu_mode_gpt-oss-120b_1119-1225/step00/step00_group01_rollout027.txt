@triton.jit
def trmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lbn, stride_li, stride_lj,
    stride_rbn, stride_rj, stride_rk,
    stride_obn, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # batch index
    batch_idx = tl.program_id(0)
    # tile indices
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute base offsets
    # offsets for i and j within tile
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask for i,j within N
    mask_i = i_off < N
    mask_j = j_off < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # pointers
        left_ptrs = left_ptr + batch_idx * stride_lbn + i_off[:, None] * stride_li + k_off[None, :] * stride_lj
        right_ptrs = right_ptr + batch_idx * stride_rbn + k_off[:, None] * stride_rj + j_off[None, :] * stride_rk

        # load
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        # dot product
        acc += tl.dot(left, right)

    # store output
    out_ptrs = out_ptr + batch_idx * stride_obn + i_off[:, None] * stride_oi + j_off[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# Triton kernel performing batch matmul for TriMul
@triton.jit
def trmul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_lbatch, stride_li, stride_lk,
    stride_rbatch, stride_rk, stride_rj,
    stride_obatch, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul for TriMul: for each batch d (combined batch+channel) compute
       out[i,j] = sum_k left[i,k] * right[j,k] where left and right have shape (N,N)."""

    batch_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets within N
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers for this batch
    left_base = left_ptr + batch_idx * stride_lbatch
    right_base = right_ptr + batch_idx * stride_rbatch

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=tl.float16(0.0))

        # right[k,j] (notice the transposition)
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=tl.float16(0.0))

        acc += tl.dot(left, right)

    # write back
    out_ptrs = out_ptr + batch_idx * stride_obatch + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """Trifactor multiplication (outgoing) implemented with fused LayerNorm, linear gates
    and a Triton kernel for the core O(N³·C) contraction."""

    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1) layer‑norm over the last dimension
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # 2) linear projections (no bias)
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # 3) optional mask (broadcast on hidden dimension)
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    # 4) gating networks
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5) core TriMul contraction using Triton
    # ------------------------------------------------------------
    # cast to half precision for the heavy contraction
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # bring the hidden dimension next to batch for a contiguous layout
    # shape becomes [B, hidden_dim, N, N]
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    B, D, N, _ = left_perm.shape   # note N == N
    batch_channels = B * D

    # flatten batch and hidden dimension into a single axis (the kernel treats it as batch)
    left_flat = left_perm.view(batch_channels, N, N)
    right_flat = right_perm.view(batch_channels, N, N)

    # allocate output buffer in FP32 (higher precision for the accumulation)
    out_flat = torch.empty(batch_channels, N, N, dtype=torch.float32, device=input_tensor.device)

    # block sizes – tuned for H100; can be overridden by the caller if desired
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    grid = (
        batch_channels,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    # launch the kernel
    trmul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),  # note the transposed strides for right
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original layout: [B, hidden_dim, N, N] → [B, N, N, hidden_dim]
    out_perm = out_flat.view(B, D, N, N)
    out = out_perm.permute(0, 2, 3, 1)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------
    # 6) final normalisation, gating and linear projection back to `dim`
    # ------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    out = F.linear(out, to_out_weight)  # -> [B, N, N, dim]

    return out

"""
TriMul (outgoing) implementation for AlphaFold‑3 style pairwise updates.

The forward pass consists of:
1. Layer‑norm on the input pair tensor.
2. Linear projections + sigmoid gates (all done in PyTorch).
3. Optional binary mask broadcast over the hidden dimension.
4. Core O(N³·C) contraction
      out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
   performed with a fused Triton kernel that treats (batch, channel) as a
   single batch dimension and carries out a batched mat‑mul (left @ rightᵀ).
5. Post‑norm, output gate and final linear back to `dim`.

Only the heavy O(N³·C) contraction is executed on Triton; the surrounding
operations use highly‑optimized torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_matmul_kernel(
    left_ptr,            # *fp16   (batch*channel, N, N) = left[i,k]
    right_ptr,           # *fp16   (batch*channel, N, N) = right[j,k] (will be transposed)
    out_ptr,             # *fp32   (batch*channel, N, N)
    N,                   # scalar int, sequence length
    stride_lbatch, stride_li, stride_lk,
    stride_rbatch, stride_rk, stride_rj,
    stride_obatch, stride_oi, stride_oj,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batch matrix‑multiply kernel implementing:
        out[b,i,j] = Σ_k left[b,i,k] * right[b,j,k]
    where `b` indexes the combined (batch × channel) dimension.
    """
    batch_idx = tl.program_id(0)          # combined batch‑channel index
    pid_i = tl.program_id(1)              # tile along i dimension
    pid_j = tl.program_id(2)              # tile along j dimension

    # Offsets inside the N×N matrix
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this batch
    left_base  = left_ptr  + batch_idx * stride_lbatch
    right_base = right_ptr + batch_idx * stride_rbatch

    # Accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k] – row i, column k
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0)
        )

        # right[k,j] – note the transposition (original right[j,k])
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_rj
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0)
        )

        # Matrix‑multiply‑accumulate
        acc += tl.dot(left, right)   # result is FP32

    # Store the result
    out_ptrs = out_ptr + batch_idx * stride_obatch + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps
    )   # shape: [B, N, N, dim]

    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------

    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))

    right = right * right_gate

    # ------------------------------------------------------------------

    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    batch_chan = B * C                      # combined batch‑channel dimension

    left_flat  = left_perm.view(batch_chan, N, N)
    right_flat = right_perm.view(batch_chan, N, N)

        device=input_tensor.device

    BLOCK_J = 64
    BLOCK_K = 64

        batch_chan,

    )

        right_flat.stride(0), right_flat.stride(2), right_flat.stride(1),  # note swapped strides for transposition
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # Restore original layout: (B*C, N, N) -> [B, C, N, N] -> [B, N, N, C]
    out_perm = out_flat.view(B, C, N, N)                # [B, C, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()     # [B, N, N, C]

    # ------------------------------------------------------------------
    # 6) Post‑processing: norm, output gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to [B, N, N, dim]

    return out
```
