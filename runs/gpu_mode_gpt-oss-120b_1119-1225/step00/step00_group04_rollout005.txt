def triton_tri_mul(left, right, block_m=64, block_n=64, block_k=64):
    # left, right shapes [B, N, N, H], dtype torch.float16
    B, N, _, H = left.shape
    # permute to [B, H, N, N] contiguous
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()
    # reshape to [B*H, N, N]
    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_perm.view(B*H, N, N)
    # allocate output
    out = torch.empty_like(left_flat, dtype=torch.float32)  # shape [BH,N,N]
    # compute strides
    stride_ah = left_flat.stride(1)
    stride_ak = left_flat.stride(2)
    stride_ab = left_flat.stride(0)
    stride_bh = right_flat.stride(0)
    stride_bj = right_flat.stride(1)  # N dimension
    stride_bk = right_flat.stride(2)  # K dimension
    stride_ch = out.stride(0)
    stride_ci = out.stride(1)
    stride_cj = out.stride(2)

    grid = lambda meta: ( (B*H) * ((N + meta['BLOCK_M'] - 1)//meta['BLOCK_M']) * ((N + meta['BLOCK_N'] - 1)//meta['BLOCK_N']), )

@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # Compute number of blocks
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    batch = pid // (num_blocks_m * num_blocks_n)
    block_index = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_index // num_blocks_n
    pid_n = block_index % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for within bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for A and B initial position for this batch
    a_batch_ptr = A_ptr + batch * stride_a_batch
    b_batch_ptr = B_ptr + batch * stride_b_batch
    c_batch_ptr = C_ptr + batch * stride_c_batch

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A tile (M, K)
        a_ptrs = a_batch_ptr + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # load B tile (K, N) i.e., transposed version
        b_ptrs = b_batch_ptr + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # dot product: (M,K) x (K,N) -> (M,N)
        acc += tl.dot(a, b)

    # write back
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
tri_mul_outgoing_triton.py
---------------------------

Implementation of the “outgoing” Triangle Multiplicative Update (TriMul) used in
AlphaFold3 and related models.

Algorithm
~~~~~~~~~
1. **Layer‑norm** on the input tensor `x` (float32) using the supplied
   `norm.{weight,bias}`.
2. **Linear projections** from `dim` → `hidden_dim` (bias‑free) for the
   *left* and *right* branches and for the three gating branches.
   All linear ops are performed in float16 for speed.
3. Apply the optional **pairwise mask** (broadcast on the hidden dim) to the
   left/right projections.
4. Apply **gates**: multiply the left/right projections by the sigmoid of the
   corresponding gate tensors.
5. **Triangular multiplicative update** – the core O(N³) kernel:
   for each hidden channel `h` compute
       out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h] .
   This is a batched matrix multiplication of `left[b,h]` (N×N) with the
   transpose of `right[b,h]`.  The heavy reduction is done in a fused
   Triton kernel (`batch_matmul_kernel`) that processes many `(i,j)` tiles in
   parallel using Tensor‑cores (FP16×FP16→FP32 accumulation).
6. **Output layer‑norm** on the hidden result, followed by the output gate.
7. **Final linear** back to the original `dim` (float32).

Only the triangular multiplication (step 5) is written in Triton; all other
operations use native PyTorch (which already runs efficiently on the GPU).
The kernel works for arbitrary batch size (B≤2), sequence length
(N∈{128,…,1024}), `dim` ∈ {128,384} and `hidden_dim` = 128 as required by the
test suite.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def batch_matmul_kernel(
    A_ptr,               # ptr to left  [batch, M, K]  (M=N, K=N)
    B_ptr,               # ptr to right [batch, N, K]  (N=N, K=N) – we read its transpose
    C_ptr,               # ptr to output [batch, M, N]
    M, N, K,             # problem dimensions (all equal to seq_len)
    stride_a_batch, stride_a_m, stride_a_k,   # strides of A
    stride_b_batch, stride_b_n, stride_b_k,   # strides of B
    stride_c_batch, stride_c_m, stride_c_n,   # strides of C
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Batched matrix multiplication C = A @ Bᵀ
    where A : (batch, M, K)   and   B : (batch, N, K)

    All tensors are in row‐major order.  The kernel accumulates in FP32
    while the inputs are FP16, thus exploiting Tensor Cores on H100.
    """
    pid = tl.program_id(axis=0)

    # Number of tiles in each dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_blocks_m * num_blocks_n)
    block_idx = pid % (num_blocks_m * num_blocks_n)

    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_batch = A_ptr + batch * stride_a_batch
    b_batch = B_ptr + batch * stride_b_batch
    c_batch = C_ptr + batch * stride_c_batch

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)                      # FP16

        # Load Bᵀ tile: we want (BLOCK_K, BLOCK_N)
        b_ptrs = b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)                      # FP16

        # Matrix multiply‑accumulate: (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b)                       # FP16×FP16 → FP32

    # Write back result
    c_ptrs = c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def triton_tri_mul(left_fp16: torch.Tensor, right_fp16: torch.Tensor) -> torch.Tensor:
    """
    Triton‑accelerated implementation of the outgoing TriMul:
    out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left_fp16, right_fp16 : torch.Tensor
        Tensors of shape [B, N, N, H] in float16.

    Returns
    -------
    out : torch.Tensor
        Tensor of shape [B, N, N, H] in float32.
    """
    assert left_fp16.shape == right_fp16.shape
    B, N, _, H = left_fp16.shape

    # Move hidden dimension to the second axis to obtain a batch of H matrices per batch
    left = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right_fp16.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Collapse batch and hidden dimensions → [B*H, N, N]
    left_flat = left.view(B * H, N, N)
    right_flat = right.view(B * H, N, N)

    # Output buffer (FP32)
    out = torch.empty((B * H, N, N), dtype=torch.float32, device=left_fp16.device)

    # Extract strides (element strides, not byte strides)
    stride_a_batch = left_flat.stride(0)
    stride_a_m = left_flat.stride(1)
    stride_a_k = left_flat.stride(2)

    stride_b_batch = right_flat.stride(0)
    stride_b_n = right_flat.stride(1)   # N‑dimension of B
    stride_b_k = right_flat.stride(2)

    stride_c_batch = out.stride(0)
    stride_c_m = out.stride(1)
    stride_c_n = out.stride(2)

    # Tile sizes – chosen to fit H100 Tensor‑Core work‑groups
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: one program per (batch_hidden, tile_m, tile_n)
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = B * H * grid_m * grid_n
    grid = (total_programs,)

    batch_matmul_kernel[grid](
        left_flat,
        right_flat,
        out,
        N, N, N,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_n, stride_b_k,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module (no gradient computation).

    Parameters
    ----------
    data : Tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask_tensor   : [B, N, N]  (bool/float, may be None)
        - weights_dict  : mapping of parameter names to tensors on the same device
        - config_dict   : contains ``dim`` and ``hidden_dim`` values

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5               # LayerNorm epsilon – matches PyTorch defaults

    # ------------------------------------------------------------------
    # Gather weights – keep them on device
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)

    left_gate_weight  = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight   = weights["out_gate.weight"].to(torch.float16)

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]  # shape (dim, hidden_dim), kept FP32

    # ------------------------------------------------------------------
    # 1. Input layer‑norm (float32)
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B,N,N,dim] – FP32

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (cast to FP16 for speed)
    # ------------------------------------------------------------------
    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # FP16
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # Cast gates to FP16 so that subsequent element‑wise mul stays in FP16
    left_gate = left_gate.to(torch.float16)
    right_gate = right_gate.to(torch.float16)
    out_gate = out_gate.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Optional pairwise mask
    # ------------------------------------------------------------------
    if mask is not None:
        # mask shape: [B,N,N]; expand to hidden dim
        mask_exp = mask.to(torch.float16).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Apply gates (still FP16)
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Triangular multiplicative update – heavy O(N³) part
    # ------------------------------------------------------------------
    # left/right are [B,N,N,hidden_dim] FP16
    out_hidden = triton_tri_mul(left, right)   # [B,N,N,hidden_dim] FP32

    # ------------------------------------------------------------------
    # 6. Output layer‑norm (float32) and output gate
    # ------------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    # out_gate is FP16, up‑cast during multiplication
    out_hidden = out_hidden * out_gate

    # ------------------------------------------------------------------
    # 7. Final linear projection back to original dim (float32)
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out_hidden, to_out_weight)  # FP32, shape [B,N,N,dim]

    return output
