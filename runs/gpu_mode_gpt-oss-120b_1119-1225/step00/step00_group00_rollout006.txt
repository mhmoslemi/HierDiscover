```python
import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute program IDs.
    pid0 = tl.program_id(0)  # combined batch and hidden dimension
    pid_i = tl.program_id(1)  # i tile
    pid_j = tl.program_id(2)  # j tile

    # Decode batch and hidden indices.
    b = pid0 // H
    h = pid0 % H

    # Offsets for the i and j dimensions.
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointer to the output tile.
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    # Accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the summed axis).
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (M, K)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile transposed: shape (K, N)
        right_ptrs_T = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        right_T = tl.load(right_ptrs_T, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Multiply-accumulate.
        acc += tl.dot(left, right_T)

    # Write back the result.
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul (outgoing) forward pass accelerated with Triton.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor.
2. Four linear projections (no bias):
   * left  = Linear(dim → hidden_dim)
   * right = Linear(dim → hidden_dim)
   * left_gate  = Linear(dim → hidden_dim) → sigmoid
   * right_gate = Linear(dim → hidden_dim) → sigmoid
   * out_gate   = Linear(dim → hidden_dim) → sigmoid
3. Apply the (optional) pairwise mask to *left* and *right*.
4. Gate the projections:
   left  = left  * left_gate
   right = right * right_gate
5. Core operation (performed by a Triton kernel):
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix‑multiply L @ Rᵀ for every hidden channel *h*.
6. Layer‑norm the intermediate result (hidden_dim).
7. Multiply by *out_gate* and project back to *dim* with a final linear layer.

Only step 5 is executed on the GPU with Triton; all other steps use
standard PyTorch kernels (they are cheap compared to the O(N³) work).
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel for the O(N³) tensor contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, H,                           # batch, seq_len, hidden_dim
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute  out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].

    - The kernel is tiled with (BLOCK_M, BLOCK_N) output tiles.
    - The reduction over k is tiled with BLOCK_K.
    - Program ID 0 encodes a combined (batch, hidden) index.
    """
    pid0 = tl.program_id(0)   # combined batch & hidden dimension
    pid_i = tl.program_id(1)  # tile over i
    pid_j = tl.program_id(2)  # tile over j

    # ------------------------------------------------------
    # Decode batch (b) and hidden (h) from pid0
    # ------------------------------------------------------
    b = pid0 // H
    h = pid0 %  H

    # ------------------------------------------------------
    # Offsets for the i/j dimensions of the output tile
    # ------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------
    # Pointer to the output tile
    # ------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j

    # ------------------------------------------------------
    # Accumulator (float32 for numerical stability)
    # ------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------
    # Loop over the reduction dimension k
    # ------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h] → shape (M, K)
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)

        # right[b, j, k, h] transposed to (K, N) → shape (K, N)
        right_ptrs_T = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        right_T = tl.load(right_ptrs_T,
                          mask=mask_k[:, None] & mask_j[None, :],
                          other=0.0)

        # Accumulate the matrix product for this K‑tile
        acc += tl.dot(left, right_T)

    # ------------------------------------------------------
    # Store the completed output tile
    # ------------------------------------------------------
    tl.store(out_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Python entry point – the custom kernel used by the autograder
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (may be None)
        - weights      : dict of torch.Tensors (see skeleton)
        - config       : dict containing at least "dim", "hidden_dim",
                         "nomask" (bool) and optionally others.

    Returns
    -------
    torch.Tensor
        The processed tensor, shape [B, N, N, dim].
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    eps = 1e-5  # same epsilon as the reference implementation

    # ------------------------------------------------------------------
    # 1) Layer‑norm the input
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask is bool/float; ensure proper dtype and broadcast
        mask_f = mask.to(dtype=dtype, device=device).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid on a separate linear projection)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 5) Triton kernel: core tensor contraction
    # ------------------------------------------------------------------
    # Output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    out_s_b, out_s_i, out_s_j, out_s_h = out.stride()
    l_s_b,  l_s_i,  l_s_k,  l_s_h = left.stride()
    r_s_b,  r_s_i,  r_s_k,  r_s_h = right.stride()

    # Tiling parameters – choose values that keep shared memory within limits
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden, i_tiles, j_tiles)
    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    tri_mul_outgoing_kernel[grid](
        out,
        left,
        right,
        B,
        N,
        hidden_dim,
        out_s_b, out_s_i, out_s_j, out_s_h,
        l_s_b,  l_s_i,  l_s_k,  l_s_h,
        r_s_b,  r_s_i,  r_s_k,  r_s_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: norm → gate → final linear
    # ------------------------------------------------------------------
    # Layer‑norm over the hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # Apply output gate (same shape as out_norm)
    out_gated = out_norm * out_gate

    # Final projection back to the original dimension
    output = torch.nn.functional.linear(out_gated, to_out_weight)  # [B, N, N, dim]

    return output
