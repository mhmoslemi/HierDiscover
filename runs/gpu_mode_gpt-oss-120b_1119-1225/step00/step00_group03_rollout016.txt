    dim, hidden_dim = config['dim'], config['hidden_dim']

    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    # Use torch.einsum or F.linear for speed
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape [B,N,N,H]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply mask if present (mask is bool or float)
    if mask is not None and mask.numel()>0:

        left = left * mask
        right = right * mask

    right = right * right_gate

    # Convert to half for matmul
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, H = left_h.shape
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # shape [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()

    # Cast back to float32
    out = out_perm.to(torch.float32)

    # Permute back to original shape: [B,N,N,H]
    out = out.permute(0, 2, 3, 1)

    # Apply to_out_norm (layer norm on hidden dim)
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bm, stride_bn,
    stride_ch, stride_cm, stride_cn,
    M, N, K,

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1️⃣ Layer‑norm the input over the last dimension (`dim`).
2️⃣ Linear projections (no bias) → `left_proj` & `right_proj`.
3️⃣ Compute three sigmoid gates from the same normalized input:
   `left_gate`, `right_gate`, `out_gate`.
4️⃣ Apply the optional mask **before** gating.
5️⃣ Fuse gating:   left = left_proj * left_gate
                 right = right_proj * right_gate
   (both tensors have shape [B, N, N, H] – H = hidden_dim)
6️⃣ Convert to FP16 and permute to [B*H, N, N] (batch * channel together).
7️⃣ Compute the core tensor contraction
        out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   which is exactly a batched matrix‑multiply left @ rightᵀ.
   This is performed by a custom Triton kernel that processes many
   (B·H) independent matmuls in parallel, using FP16 for compute
   and FP32 accumulation.
8️⃣ Cast the result back to FP32 and reshape to [B, N, N, H].
9️⃣ Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to `dim`.

The kernel is deliberately small (BLOCK_M = BLOCK_N = BLOCK_K = 64)
and works for all test configurations (N up to 1024, B ≤ 2, H = 128).

Only the heavy N³‑like contraction is Triton‑accelerated;
all lighter ops (layer‑norm, linear layers, sigmoids) use
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
#   A : (batch, M, K)   – fp16
#   B : (batch, N, K)   – fp16 (we read it as transposed)
#   C : (batch, M, N)   – fp16 (accumulated in fp32)
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bm, stride_bn,
    stride_ch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_batch = tl.program_id(0)   # batch * hidden dimension
    pid_m = tl.program_id(1)       # tile over rows   (M)
    pid_n = tl.program_id(2)       # tile over cols   (N)

    # --- tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this tile
    a_tile_ptr = (
        a_ptr
        + pid_batch * stride_ah
        + offs_m[:, None] * stride_am
        + offs_k[None, :] * stride_ak
    )
    # NOTE: we need Bᵀ, i.e. we read B as (K, N)
    b_tile_ptr = (
        b_ptr
        + pid_batch * stride_bh
        + offs_k[:, None] * stride_bn      # original column stride (==1 for contiguous)
        + offs_n[None, :] * stride_bm      # original row stride (==K)
    )

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0)
        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to the next K‑tile
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bn

    # write result
    c_tile_ptr = (
        c_ptr
        + pid_batch * stride_ch
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_tile_ptr,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_fp16(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul C = A @ Bᵀ using the Triton kernel above.
    All tensors must be contiguous, FP16, and have matching batch size.
    Shapes:
        a : (B, M, K)   – fp16
        b : (B, N, K)   – fp16
    Returns:
        c : (B, M, N)   – fp16
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, N, K2 = b.shape
    assert K == K2

    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)

    # strides (in elements, not bytes)
    stride_ah, stride_am, stride_ak = a.stride()
    stride_bh, stride_bm, stride_bn = b.stride()
    stride_ch, stride_cm, stride_cn = c.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        triton.cdiv(M, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )
    _batched_matmul_kernel[grid](
        a,
        b,
        c,
        stride_ah,
        stride_am,
        stride_ak,
        stride_bh,
        stride_bm,
        stride_bn,
        stride_ch,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] or None
            - weights: dict of model parameters (see keys in the code)
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # layer‑norm over the last dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right_proj = F.linear(x, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B, N, N, H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))     # [B, N, N, H]

    # ------------------------------------------------------------------
    # optional mask (mask is broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # apply gates (still in FP32)
    # ------------------------------------------------------------------
    left = left_proj * left_gate      # [B, N, N, H]
    right = right_proj * right_gate   # [B, N, N, H]

    # ------------------------------------------------------------------
    # cast to FP16 and reshape for the Triton kernel
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # bring the hidden dimension next to the batch for a single fused batch
    left_h = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_h = right_h.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_h.shape
    a = left_h.reshape(B * H, N, N)   # (batch*H, M, K)
    b = right_h.reshape(B * H, N, N)  # (batch*H, N, K)

    # ------------------------------------------------------------------
    # core contraction: out = left @ rightᵀ (batch‑wise)
    # ------------------------------------------------------------------
    c_fp16 = _batched_matmul_fp16(a, b)  # (B*H, N, N) in fp16

    # ------------------------------------------------------------------
    # reshape back to original layout and convert to FP32
    # ------------------------------------------------------------------
    out = c_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # post‑processing: LN → gate → final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, H]

    out = out * out_gate                      # apply the output gate
    out = F.linear(out, weights["to_out.weight"])  # back to dim -> [B, N, N, dim]

    return out
