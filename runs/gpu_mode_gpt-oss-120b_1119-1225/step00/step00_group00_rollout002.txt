@triton.jit
def layernorm_fwd_kernel(
    input_ptr, output_ptr,
    weight_ptr, bias_ptr,
    B, N, C,
    stride_input_b, stride_input_i, stride_input_j, stride_input_c,
    stride_output_b, stride_output_i, stride_output_j, stride_output_c,
    stride_weight_c, stride_bias_c,
    eps: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    # compute position index
    pos = pid
    if pos >= total: return
    b = pos // (N*N)
    res = pos % (N*N)
    i = res // N
    j = res % N
    # compute base pointers
    input_base = input_ptr + b*stride_input_b + i*stride_input_i + j*stride_input_j
    # now iterate over C dimension
    sum = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)
    for c in range(0, C, BLOCK_C):
        c_offset = c
        size = min(BLOCK_C, C - c)
        # load
        x = tl.load(input_base + c_offset*stride_input_c,
                    mask=tl.arange(0, BLOCK_C) < size,
                    other=0.0)
        x_f = x.to(tl.float32)
        sum += tl.sum(x_f, axis=0)
        sum_sq += tl.sum(x_f * x_f, axis=0)
    mean = sum / C
    var = sum_sq / C - mean*mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # store normalized output
    for c in range(0, C, BLOCK_C):
        c_offset = c
        size = min(BLOCK_C, C - c)
        x = tl.load(input_base + c_offset*stride_input_c,
                    mask=tl.arange(0, BLOCK_C) < size,
                    other=0.0)
        x_f = x.to(tl.float32)
        y = (x_f - mean) * inv_std
        # apply affine
        w = tl.load(weight_ptr + c_offset*stride_weight_c,
                    mask=tl.arange(0, BLOCK_C) < size,
                    other=0.0)
        b_ = tl.load(bias_ptr + c_offset*stride_bias_c,
                     mask=tl.arange(0, BLOCK_C) < size,
                     other=0.0)
        y = y * w + b_
        # store
        tl.store(output_ptr + b*stride_output_b + i*stride_output_i + j*stride_output_j + c_offset*stride_output_c,
                 y.to(tl.float16) if output.dtype==tl.float16 else y,
                 mask=tl.arange(0, BLOCK_C) < size)

def layernorm_kernel(
    inp_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, C,
    stride_in_b, stride_in_i, stride_in_j, stride_in_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    stride_w_c, stride_b_c,
    eps,
    BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    # Compute indices
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N
    # Compute base pointer offsets
    inp_offset = b * stride_in_b + i * stride_in_i + j * stride_in_j
    # Reduce across C dimension
    sum_val = tl.zeros([], dtype=tl.float32)
    sum_sq = tl.zeros([], dtype=tl.float32)
    # iterate over C in BLOCK_C
    for off in range(0, C, BLOCK_C):
        c_offset = off
        cur_block = min(BLOCK_C, C - off)
        c_range = tl.arange(0, BLOCK_C)
        mask = c_range < cur_block
        # load
        x = tl.load(inp_ptr + inp_offset + c_offset * stride_in_c,
                    mask=mask, other=0.0)

        sum_val += tl.sum(x_f, axis=0, mask=mask)  # but tl.sum returns scalar across axes? Actually tl.sum(x_f) returns scalar sum across all elements; we need mask; not needed if loading only valid.

"""
custom_kernel for the “outgoing” TriMul operator (AlphaFold‑3 style).

Algorithm
---------
1. **Layer‑norm (input)** – a Triton kernel normalises the tensor over the
   channel dimension (`dim`).  The kernel works on one (batch, i, j) position
   at a time, accumulates mean/variance by looping over the channel axis and
   then writes the normalised + affine‑transformed result.

2. **Linear projections & gates** – ordinary PyTorch `torch.nn.functional.linear`
   (highly‑optimised GEMM) is used for:
   * left / right projections (`dim → hidden_dim`)
   * left / right / output gates (`dim → hidden_dim`)

3. **Masking** – if a mask is supplied it is broadcasted to the hidden
   dimension and applied to the projected tensors *before* gating.

4. **Tri‑multiplicative update** – the core operation
   `out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]`
   is performed as a batch of matrix‑multiplications:
   for every hidden channel `h` we compute
   `out_h = left_h @ right_h.T`.  This is realised with a single
   `torch.bmm` call on the reshaped tensors (equivalent to the `einsum`
   in the reference implementation).

5. **Layer‑norm (output)** – the same Triton kernel normalises the result
   over the hidden dimension.

6. **Output gating & final projection** – gated by `out_gate` and projected
   back to the original dimension with a final linear layer.

All operations keep `torch.float32` for numerical safety; the Triton kernels
are deliberately simple (one thread per (b,i,j) entry) and therefore do not
require intra‑block synchronisation.

The kernel is written for Triton 3.3.1 and targets an NVIDIA H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton layer‑norm kernel (works on a tensor of shape [B, N, N, C])
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    inp_ptr,                     # *Pointer* to input  (float32)
    out_ptr,                     # *Pointer* to output (float32)
    weight_ptr,                  # *Pointer* to gamma  (float32, shape [C])
    bias_ptr,                    # *Pointer* to beta   (float32, shape [C])
    B, N, C,                     # tensor dimensions
    stride_in_b, stride_in_i, stride_in_j, stride_in_c,   # input strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_c, # output strides
    stride_w_c, stride_b_c,      # weight / bias strides (normally 1)
    eps,                         # epsilon for numerical stability
    BLOCK_C: tl.constexpr,       # block size along the channel dim
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # --------------------------------------------------------------
    # Decode (b, i, j) from the linear pid
    # --------------------------------------------------------------
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Base pointer to the start of the C‑vector for this (b,i,j)
    inp_base = inp_ptr + b * stride_in_b + i * stride_in_i + j * stride_in_j

    # --------------------------------------------------------------
    # 1️⃣  Compute mean and variance across the C dimension
    # --------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    # Loop over C in chunks of BLOCK_C
    for c_off in range(0, C, BLOCK_C):
        cur = tl.minimum(BLOCK_C, C - c_off)               # number of valid elems
        c_range = tl.arange(0, BLOCK_C)
        mask = c_range < cur                                 # bool mask

        # Load a vector of up to BLOCK_C elements
        x = tl.load(inp_base + c_off * stride_in_c,
                    mask=mask, other=0.0).to(tl.float32)

        sum_val += tl.sum(x)                # scalar sum of the vector
        sum_sq  += tl.sum(x * x)             # sum of squares

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # --------------------------------------------------------------
    # 2️⃣  Normalise, apply affine transform and store the result
    # --------------------------------------------------------------
    for c_off in range(0, C, BLOCK_C):
        cur = tl.minimum(BLOCK_C, C - c_off)
        c_range = tl.arange(0, BLOCK_C)
        mask = c_range < cur

        x = tl.load(inp_base + c_off * stride_in_c,
                    mask=mask, other=0.0).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        y = (x - mean) * inv_std

        # affine: y = y * weight + bias
        w = tl.load(weight_ptr + c_off * stride_w_c,
                    mask=mask, other=1.0)
        b_ = tl.load(bias_ptr + c_off * stride_b_c,
                     mask=mask, other=0.0)
        y = y * w + b_

        # Store back
        out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j + c_off * stride_out_c
        tl.store(out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j + c_off * stride_out_c,
                 y,
                 mask=mask)

# ----------------------------------------------------------------------
# Helper that wraps the Triton layer‑norm kernel
# ----------------------------------------------------------------------
def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block_c: int = 32) -> torch.Tensor:
    """
    Normalise ``x`` over its last dimension using the supplied
    ``weight`` (γ) and ``bias`` (β).  The kernel processes one (b,i,j)
    entry per program instance.
    """
    assert x.is_contiguous()
    B, N, N2, C = x.shape
    assert N2 == N, "input must be square in the last two dims"
    out = torch.empty_like(x)

    # Strides (in elements, not bytes)
    stride_in_b, stride_in_i, stride_in_j, stride_in_c = x.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_c = out.stride()

    # Launch configuration: one program per (b,i,j)
    total = B * N * N
    grid = (total,)

    layernorm_kernel[grid](
        x, out,
        weight, bias,
        B, N, C,
        stride_in_b, stride_in_i, stride_in_j, stride_in_c,
        stride_out_b, stride_out_i, stride_out_j, stride_out_c,
        weight.stride(0), bias.stride(0),
        eps,
        BLOCK_C=block_c,
    )
    torch.cuda.synchronize()
    return out

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # LayerNorm epsilon

    # Ensure contiguity and float32
    x = input_tensor.contiguous().float()
    if mask is not None:
        mask = mask.contiguous().float()
    else:
        # a dummy mask of ones (no masking)
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                         device=x.device, dtype=x.dtype)

    B, N, N2, C = x.shape
    assert N2 == N and C == dim

    # ------------------------------------------------------------------
    # 1️⃣ Input layer‑norm (Triton)
    # ------------------------------------------------------------------
    x_norm = triton_layernorm(
        x,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block_c=32,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (PyTorch – fast GEMM)
    # ------------------------------------------------------------------
    # Shapes:
    #   left/right : [B, N, N, hidden_dim]
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if a real mask was provided)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask shape [B, N, N] → broadcast to hidden_dim
        mask_exp = mask.unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (linear + sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched matrix‑multiplies)
    # ------------------------------------------------------------------
    # Reshape to [B*hidden_dim, N, N] for a single bmm call.
    #   left  → (B, N, N, H) → (B, H, N, N) → (B*H, N, N)
    #   right → same, then transpose the last two dims.
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_flat = left_flat @ right_flat^T   (shape: B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm (Triton)
    # ------------------------------------------------------------------
    out_norm = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block_c=32,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Output gating & final projection
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final
```
