"""
custom_kernel
==============

Implementation of the “outgoing” TriMul operation (AlphaFold‑3) using a
Triton kernel for the cubic‑cost pairwise update:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

All other linear / LayerNorm / gating steps are performed with native
PyTorch for simplicity.  The heavy reduction over the third sequence
dimension (the N³ part) is fused in a Triton kernel that processes every
(b, d) slice independently, using a blocked matrix‑multiply with
FP16 inputs and FP32 accumulation.

The kernel works for any batch size ≤ 2, sequence length up to 1024 and
hidden dimension up to 128 (as required by the test suite).  It also
handles the optional mask (or the “nomask” configuration) and applies
the final LayerNorm, output gate and projection exactly as the reference
PyTorch implementation.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batch‑hidden‑dim (B×D) independent matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, D,                               # runtime shapes
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for a single (batch, hidden) slice, the matrix product
    left[b, :, :, d] @ right[b, :, :, d].T.
    The result is stored in out[b, :, :, d].

    Arguments:
        left_ptr, right_ptr, out_ptr : base pointers of the tensors.
        B, N, D                     : batch size, sequence length, hidden dim.
        stride_*                    : strides (in elements) of each tensor.
        BLOCK_M, BLOCK_N, BLOCK_K   : tiling sizes (compile‑time).
    """
    pid_m = tl.program_id(0)                # block row index (i)
    pid_n = tl.program_id(1)                # block col index (j)
    pid_bn = tl.program_id(2)               # combined batch & hidden dim index

    # ----- decode batch and hidden indices --------------------------------
    b = pid_bn // D                         # batch index
    d = pid_bn % D                          # hidden‑dim index

    # ----- block offsets ----------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # base offsets for this (b,d) slice
    b_off_l = b * stride_l_b
    b_off_r = b * stride_r_b
    b_off_o = b * stride_o_b
    d_off_l = d * stride_l_d
    d_off_r = d * stride_r_d
    d_off_o = d * stride_o_d

    # ---- accumulation ------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of k‑tiles
    num_tiles_k = tl.cdiv(N, BLOCK_K)

    for tile_k in range(0, num_tiles_k):
        offs_k = tile_k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left block (M x K) -----------------------------------------
        # shape: (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b_off_l + \
            offs_m[:, None] * stride_l_i + \
            offs_k[None, :] * stride_l_k + \
            d_off_l
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0)

        # ----- right block (K x N)  (transposed view) ----------------------
        # we need right[row=k, col=j] = right[b, j, k, d]
        # shape: (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b_off_r + \
            offs_n[None, :] * stride_r_i + \
            offs_k[:, None] * stride_r_k + \
            d_off_r
        right = tl.load(right_ptrs,
                        mask=mask_n[None, :] & mask_k[:, None],
                        other=0.0)

        # ----- block‑wise dot product (M x N) --------------------------------
        # left:  (M, K)  FP16, right: (K, N)  FP16  →  acc: (M, N)  FP32
        acc += tl.dot(left, right)

    # ----- store the result -------------------------------------------------
    out_ptrs = out_ptr + \
        b_off_o + \
        offs_m[:, None] * stride_o_i + \
        offs_n[None, :] * stride_o_j + \
        d_off_o
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module (no gradients).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool or float)
        - weights_dict : mapping from weight names to tensors
        - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32) – the result of the TriMul
        forward computation.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # LayerNorm on the input channel dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # ------------------------------------------------------------------
    # Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask ∈ {0,1}).  If nomask → skip masking.
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask_tensor is not None:
        mask = mask_tensor.to(dtype=dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the cubic update – reduces bandwidth, uses Tensor
    # Cores on H100.  Accumulation stays in FP32.
    # ------------------------------------------------------------------
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape  # N is sequence length

    # ------------------------------------------------------------------
    # Allocate output tensor (float32)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=device)

    # ------------------------------------------------------------------
    # Compute strides (in elements) – required by Triton kernel
    # ------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128   # rows per block
    BLOCK_N = 128   # cols per block
    BLOCK_K = 32    # reduction tile size

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch the kernel (use a few warps – 4 works well on H100)
    _trimul_kernel[grid](
        left_fp16, right_fp16, out,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_i, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm → output gate → final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=1e-5)

    out = out * out_gate  # broadcasted element‑wise multiplication

    # final linear projection back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
