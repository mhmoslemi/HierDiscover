@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, M, N, K,
    stride_left_bh, stride_left_m, stride_left_k,
    stride_right_bh, stride_right_k, stride_right_n,
    stride_out_bh, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    # maybe some other compile-time constants like dtype?
):
   # compute program ID
   pid = tl.program_id(0)
   # Compute total number of blocks: total = B*H * num_blocks_m * num_blocks_n
   num_blocks_m = tl.cdiv(M, BLOCK_M)
   num_blocks_n = tl.cdiv(N, BLOCK_N)
   blocks_per_batch = num_blocks_m * num_blocks_n
   bh = pid // blocks_per_batch   # B*H dimension
   block_idx = pid % blocks_per_batch
   pid_m = block_idx // num_blocks_n
   pid_n = block_idx % num_blocks_n

   # offset for this block
   offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   # offset for batch*head
   bh_stride = bh
   # Load out accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Loop over K dimension
   for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offset for k tile
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # compute pointers
        left_tile_ptr = left_ptr + bh_stride * stride_left_bh + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        right_tile_ptr = right_ptr + bh_stride * stride_right_bh + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n

        # load
        left_tile = tl.load(left_tile_ptr, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        right_tile = tl.load(right_tile_ptr, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        # cast to FP32
        left_tile = left_tile.to(tl.float32)
        right_tile = right_tile.to(tl.float32)

        # perform dot
        acc += tl.dot(left_tile, right_tile)

   # store result
   out_tile_ptr = out_ptr + bh_stride * stride_out_bh + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
   tl.store(out_tile_ptr, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, M, N, K,
    stride_left_b, stride_left_h, stride_left_m, stride_left_k,
    stride_right_b, stride_right_h, stride_right_k, stride_right_n,
    stride_out_b, stride_out_h, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute number of blocks
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_bh = num_blocks_m * num_blocks_n
    total_bh = B * H
    bh = pid // blocks_per_bh
    block_idx = pid % blocks_per_bh
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # compute indices for b and h
    b = bh // H
    h = bh % H

    # compute offsets within M,N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # compute base pointers
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load left tile
        left_ptrs = left_base + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # matrix multiply-accumulate
        acc += tl.dot(left, right)  # left: [BLOCK_M, BLOCK_K], right: [BLOCK_K, BLOCK_N]
    # Store result
    out_ptrs = out_base + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

The operation follows the reference implementation from AlphaFold3:
1. Layer‑norm over the channel dimension `dim`.
2. Linear projections (`left_proj`, `right_proj`) to a hidden dimension `hidden_dim`.
3. Optional pair‑wise mask (broadcasted on the hidden dimension).
4. Gating (sigmoid) on the projected tensors.
5. Pair‑wise inner‑product over the sequence axis `k`:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for every (batch, hidden) pair.
   It is implemented as a Triton kernel that performs a GEMM on the
   `(i,k)` and `(k,j)` slices for each hidden channel.
6. Layer‑norm over the hidden dimension, gating, and final linear projection
   back to `dim`.

The heavy tensor contraction (step 5) is the only part written in Triton;
all other operations stay in PyTorch (using FP16 for the projections to
reduce memory traffic). The final result is returned as a FP32 tensor.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the inner‑product over the `k` axis.
# Input tensors are expected as (batch, head, M, K) and (batch, head, K, N).
# Output is (batch, head, M, N).
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *fp16   left   [B, H, M, K]
    right_ptr,         # *fp16   right  [B, H, K, N]
    out_ptr,           # *fp16   out    [B, H, M, N]
    B, H, M, N, K,
    # strides for left  (B, H, M, K)
    stride_left_b, stride_left_h, stride_left_m, stride_left_k,
    # strides for right (B, H, K, N)
    stride_right_b, stride_right_h, stride_right_k, stride_right_n,
    # strides for out   (B, H, M, N)
    stride_out_b, stride_out_h, stride_out_m, stride_out_n,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # ------------------------------
    # 1) map 1‑D program id to (batch*head, block_m, block_n)
    # ------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_bh = num_blocks_m * num_blocks_n          # tiles for one (batch,head)

    bh_id      = pid // blocks_per_bh                     # flat index over B*H
    tile_index = pid % blocks_per_bh

    pid_m = tile_index // num_blocks_n                    # block row
    pid_n = tile_index %  num_blocks_n                    # block col

    # decode (b, h) from flat bh_id
    b = bh_id // H
    h = bh_id % H

    # ------------------------------
    # 2) tile offsets
    # ------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this (b, h) pair
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------
    # 3) loop over K dimension in tiles
    # ------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- left tile: shape (BLOCK_M, BLOCK_K) ----
        left_ptrs = left_base + offs_m[:, None] * stride_left_m + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ---- right tile: shape (BLOCK_K, BLOCK_N) ----
        right_ptrs = right_base + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_n
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # ---- matrix multiply‑accumulate ----
        acc += tl.dot(left, right)          # (BLOCK_M, BLOCK_K) · (BLOCK_K, BLOCK_N)

    # ------------------------------
    # 4) write result back (store as FP16)
    # ------------------------------
    out_ptrs = out_base + offs_m[:, None] * stride_out_m + offs_n[None, :] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of TriMul (outgoing) using a Triton kernel for the
    inner‑product contraction. `data` is a tuple:
        (input_tensor, mask, weights_dict, config_dict)
    The function returns a tensor of shape [bs, seq_len, seq_len, dim]
    (dtype=torch.float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = config.get("eps", 1e-5)

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16 for memory‑bandwidth reduction)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        # no masking – use a tensor of ones
        mask = torch.ones_like(input_tensor[..., 0], dtype=x.dtype, device=x.device)
    mask = mask.unsqueeze(-1).type_as(x)                # [B, N, N, 1]
    left  = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 and prepare tensors for the Triton kernel
    # ------------------------------------------------------------------
    left_f16  = left.to(torch.float16)
    right_f16 = right.to(torch.float16)

    # reshape to (B, H, N, N) – the kernel expects (batch, head, M, K)/(K,N)
    left_perm  = left_f16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]  (M=i, K=k)
    right_perm = right_f16.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]  (K=k, N=j)

    B, H, M, K = left_perm.shape   # M == N == K == seq_len
    N = K

    # ------------------------------------------------------------------
    # 6) Allocate output tensor for the kernel (FP16)
    # ------------------------------------------------------------------
    out_perm = torch.empty((B, H, M, N), dtype=torch.float16, device=left_perm.device)

    # ------------------------------------------------------------------
    # 7) Kernel launch configuration
    # ------------------------------------------------------------------
    # Tile sizes – chosen to fit into shared memory on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_bh    = B * H
    grid = (total_bh * num_blocks_m * num_blocks_n,)

    # launch
    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, H, M, N, K,
        # strides for left
        left_perm.stride(0), left_perm.stride(1), left_perm.stride(2), left_perm.stride(3),
        # strides for right
        right_perm.stride(0), right_perm.stride(1), right_perm.stride(2), right_perm.stride(3),
        # strides for out
        out_perm.stride(0), out_perm.stride(1), out_perm.stride(2), out_perm.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # enough warps for the chosen tile size
    )

    # ------------------------------------------------------------------
    # 8) Rearrange back to (B, N, N, hidden_dim) and cast to FP32
    # ------------------------------------------------------------------
    out = out_perm.permute(0, 2, 3, 1).contiguous().to(torch.float32)   # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 9) Final LayerNorm over hidden_dim, gating, and linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate                      # gating after to_out_norm
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
