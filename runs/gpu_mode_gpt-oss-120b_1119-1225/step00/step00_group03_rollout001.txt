                         stride_am, stride_ak, stride_ab, # maybe

                         stride_cm, stride_cn, stride_cb,
                         BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    # compute offsets for M,N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute ptrs for A and B
    a_ptrs = A_ptr + pid_batch * stride_ab + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = B_ptr + pid_batch * stride_bb + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

"""
custom_kernel
==============

Implementation of the “outgoing” Triangle Multiplicative Update (TriMul) operation
used in AlphaFold‑3 and related protein‑structure models.

The kernel consists of three stages:

1. **Pre‑processing (PyTorch)**
   * Layer‑norm of the input tensor.
   * Linear projections (`left_proj`, `right_proj`) and gating projections.
   * Application of the optional mask and the element‑wise sigmoid gates.

2. **Core Tri‑Mul (Triton)**
   * The heavy N³ reduction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
     is rewritten as a batched matrix‑multiply:
        left_d  = left[..., d]   ∈ ℝ^{N×N}
        right_d = right[..., d].T ∈ ℝ^{N×N}
        out_d   = left_d @ right_d
   * By reshaping the tensors we obtain a batch dimension
        B' = B × hidden_dim.
   * A custom Triton GEMM kernel computes all B' matrix products in parallel,
     using FP16 inputs (Tensor‑Core friendly) and FP32 accumulation for
     numerical stability.

3. **Post‑processing (PyTorch)**
   * Layer‑norm on the hidden dimension, multiplication by the output gate,
     and a final linear projection back to the original `dim`.

Only the N³ reduction is performed on the GPU via Triton, satisfying the
requirement that “part of the operations” be a custom kernel while keeping the
overall implementation clear and correct for all supported shapes.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton batched GEMM kernel used for the Tri‑Mul reduction.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    batch, m, n, k,                    # logical dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C[batch, m, n] = A[batch, m, k] @ B[batch, k, n] (FP16 inputs).

    The kernel is launched with a 3‑D grid:
        grid = (batch, ceil(m/BLOCK_M), ceil(n/BLOCK_N))
    """
    pid_batch = tl.program_id(0)   # which batch (B * hidden_dim)
    pid_m = tl.program_id(1)       # block row
    pid_n = tl.program_id(2)       # block column

    # Compute the absolute coordinates of the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds loads / stores
    mask_m = offs_m < m
    mask_n = offs_n < n

    # Initialize accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis K
    for kk in range(0, tl.cdiv(k, BLOCK_K)):
        cur_k = kk * BLOCK_K
        # Offsets for A (shape: batch, m, k)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_a_k
        )
        # Offsets for B (shape: batch, k, n)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        # Load with appropriate masks
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (cur_k + tl.arange(0, BLOCK_K) < k),
                    other=0.0)   # FP16 value -> promoted to FP32 for dot
        b = tl.load(b_ptrs,
                    mask=(cur_k + tl.arange(0, BLOCK_K) < k)[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate (FP16 * FP16 => FP32)
        acc += tl.dot(a, b)

    # Cast back to FP16 for storage
    c = acc.to(tl.float16)

    # Store result
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of model weights (see docstring)
            - config: dict with keys "dim", "hidden_dim" and optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Retrieve weights (all are stored as float32 tensors)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]         # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]    # (hidden_dim,)
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]              # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    # Use torch's fused layer_norm for simplicity and stability
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (mask is broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating pathways
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core Tri‑Mul via a batched GEMM kernel
    # ------------------------------------------------------------------
    # Permute to bring the hidden dimension next to the batch dimension:
    #   [B, N, N, H] -> [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Transpose the last two axes of right (required by the einsum formulation)
    right_t = right_perm.transpose(-1, -2).contiguous()   # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    batch_dim = B * H

    # Collapse the (B, H) dimensions into a single batch for the kernel
    left_mat = left_perm.reshape(batch_dim, N, N).to(torch.float16).contiguous()
    right_mat = right_t.reshape(batch_dim, N, N).to(torch.float16).contiguous()
    out_mat = torch.empty_like(left_mat, dtype=torch.float16)

    # Compute strides (in element units, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left_mat.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_mat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64
    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        batch_dim,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before moving on
    torch.cuda.synchronize()

    # Reshape back to [B, N, N, hidden_dim]
    out = (
        out_mat.to(torch.float32)
        .reshape(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # shape: [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim layer‑norm, out‑gate, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # back to [B, N, N, dim]

    return out
