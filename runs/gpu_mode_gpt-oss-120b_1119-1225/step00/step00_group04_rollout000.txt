Define custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # may also have config["nomask"].

    # Apply layernorm
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if not nomask:
    if not config.get("nomask", False):
        mask_overlap = mask.unsqueeze(-1).type_as(x_norm)
        left = left * mask_overlap
        right = right * mask_overlap

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute N^2 H output via Triton
    out_triton = triu_mul_triton(left, right)  # returns [B, N, N, hidden_dim]

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out_triton, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output

@triton.jit
def tr_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_bl, stride_bi, stride_bk, stride_bh,
        stride_rl, stride_ri, stride_rk, stride_rh,
        stride_ol, stride_oi, stride_ok, stride_oh,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    pid_i = tl.program_id(0)  # block row
    pid_j = tl.program_id(1)  # block col
    pid_bn = tl.program_id(2)  # batch * hidden

    batch_idx = pid_bn // H
    h_idx = pid_bn % H

    # Compute start offsets
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # offsets for i, j within block
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # mask for valid i/j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointers
    left_base = left_ptr + batch_idx * stride_bl + h_idx * stride_bh
    right_base = right_ptr + batch_idx * stride_rl + h_idx * stride_rh
    out_base = out_ptr + batch_idx * stride_ol + h_idx * stride_oh

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        # offsets for k
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_bi + cur_k[None, :] * stride_bk)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        # right element at (j, k)
        right_ptrs = right_base + (cur_k[:, None] * stride_rk + offs_j[None, :] * stride_ri)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # dot product accumulate
        acc += tl.dot(left, right)

    # Store results
    out_ptrs = out_base + (offs_i[:, None] * stride_oi + offs_j[None, :] * stride_ok)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul custom kernel (outgoing version) – AlphaFold3

This implementation follows the reference PyTorch module but off‑loads the
computationally heavy N³ “einsum” to a Triton kernel.  The steps are:

1. Layer‑norm + four linear projections (left, right, left_gate,
   right_gate, out_gate) – done with regular PyTorch ops.
2. Optional binary mask applied to the left/right tensors.
3. Gating: multiply left/right by their sigmoid gates.
4. **TriMul core** – for each batch b and hidden channel h we compute
   ``out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]``.
   This is a batched matrix multiplication of ``left_i_k`` with the
   transpose of ``right_j_k``.  The computation is tiled with
   BLOCK_M × BLOCK_N output tiles and a BLOCK_K reduction axis and
   accumulated in float32 for numerical stability.
5. Layer‑norm over the hidden dimension, gating by ``out_gate``,
   and a final linear projection back to the original feature dimension.

Only the N³ reduction (step 4) is executed in Triton; the rest
uses PyTorch which keeps the implementation simple while still meeting
the requirement of having a custom kernel.
"""

import torch
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton kernel for the “einsum” part:
#   out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
# ------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_bi, stride_bk, stride_bh,
    stride_rl, stride_ri, stride_rk, stride_rh,
    stride_ol, stride_oi, stride_ok, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # Program IDs
    # --------------------------------------------------------------
    pid_i = tl.program_id(0)          # block row   (i dimension)
    pid_j = tl.program_id(1)          # block col   (j dimension)
    pid_bn = tl.program_id(2)         # combined batch * hidden channel

    batch_idx = pid_bn // H
    h_idx     = pid_bn % H

    # --------------------------------------------------------------
    # Compute start offsets for this tile
    # --------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for the selected (batch, hidden) slice
    left_base  = left_ptr  + batch_idx * stride_bl + h_idx * stride_bh
    right_base = right_ptr + batch_idx * stride_rl + h_idx * stride_rh
    out_base   = out_ptr   + batch_idx * stride_ol + h_idx * stride_oh

    # --------------------------------------------------------------
    # Accumulator
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left[i, k]  → (M, K) block
        left_ptrs = left_base + (offs_i[:, None] * stride_bi + cur_k[None, :] * stride_bk)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right[j, k] (need the transpose) → (K, N) block
        #   we read right at (j, k) but store it as (k, j)
        right_ptrs = right_base + (cur_k[:, None] * stride_rk + offs_j[None, :] * stride_ri)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Dot‑product accumulate
        acc += tl.dot(left_tile, right_tile)   # (M, K) @ (K, N) → (M, N)

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    out_ptrs = out_base + (offs_i[:, None] * stride_oi + offs_j[None, :] * stride_ok)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                    BLOCK_M: int = 128, BLOCK_N: int = 128, BLOCK_K: int = 32):
    """
    Batched matrix multiplication using the Triton kernel above.
    left / right: shape [B, N, N, H] (float32)
    Returns: tensor of the same shape containing Σ_k left[..., i,k,h] * right[..., j,k,h]
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2, "pairwise tensors must be square"

    # Ensure contiguous layout (required for correct stride handling)
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # Strides (in elements, not bytes)
    stride_bl, stride_bi, stride_bk, stride_bh = left.stride()
    stride_rl, stride_ri, stride_rk, stride_rh = right.stride()
    stride_ol, stride_oi, stride_ok, stride_oh = out.stride()

    # Grid dimensions: (tiles over i, tiles over j, batch*hidden)
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_bn = B * H
    grid = (grid_i, grid_j, grid_bn)

    # Launch kernel
    _tri_mul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, H,
        stride_bl, stride_bi, stride_bk, stride_bh,
        stride_rl, stride_ri, stride_rk, stride_rh,
        stride_ol, stride_oi, stride_ok, stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # reasonable default for H100
    )
    return out


# ------------------------------------------------------------------
# Public entry point expected by the test harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Implements the outgoing TriMul operation using a Triton kernel for the
    cubic reduction.  Arguments:

      data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    # Config values
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # left and right projections to hidden dimension
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → [B,N,N,1] → broadcast to hidden dim
        mask_f = mask.to(dtype=left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul via Triton (Σ_k left[...,i,k,h] * right[...,j,k,h])
    # ------------------------------------------------------------------
    # The Triton kernel expects contiguous tensors; we have already ensured that.
    # It returns a tensor of shape [B, N, N, hidden_dim] in float32.
    out = _tri_mul_triton(left, right,
                          BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    # ------------------------------------------------------------------
    # 6) Output layer‑norm + gating + final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    # Final projection back to original dim
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
