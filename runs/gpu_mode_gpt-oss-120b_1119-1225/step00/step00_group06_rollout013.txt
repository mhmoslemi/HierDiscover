@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_k, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch_id = tl.program_id(0) # combined batch (b * H + h)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch indices
    b = batch_id // H
    h = batch_id % H

    # compute coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask out-of-range
    mask_m = offs_m < N
    mask_n = offs_n < N

    # load accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # compute pointers
        left_offset = b * stride_left_b + h * stride_left_h + offs_m[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left_ptrs = left_ptr + left_offset
        # load left tile, shape (BLOCK_M, BLOCK_K)
        left_tile = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        # cast to fp32
        left_tile = left_tile.to(tl.float32)

        # right tile (k, n)
        right_offset = b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_k + offs_n[None, :] * stride_right_j
        right_ptrs = right_ptr + right_offset
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        right_tile = right_tile.to(tl.float32)

        # dot
        acc += tl.dot(left_tile, right_tile)

        # increment pointers are not needed because we recompute addresses in each iteration
        
    # store result
    out_offset = b * stride_out_b + h * stride_out_h + offs_m[:, None] * stride_out_i + offs_n[None, :] * stride_out_j
    out_ptrs = out_ptr + out_offset
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype != tl.float32 else acc, mask=mask_m[:, None] & mask_n[None, :])

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # out shape is (B, H, N, N); need to permute back to (B,N,N,H)
    out = out.permute(0, 2, 3, 1)  # (B,N,N,H)

    # Final linear projection to dim
    out_flat = out.view(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_w).view(bs, N, N, dim)

    # Cast back to float32 if needed (original dtype)
    out = out.to(torch.float32)
    return out

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_k, stride_right_j,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute program IDs
    batch_id = tl.program_id(0)  # 0..(B*H-1)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # decompose batch id to batch and head index
    b = batch_id // H
    h = batch_id % H

    # offsets for the block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bound rows/cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute base offsets for left and right matrices per batch
    left_batch_offset = b * stride_left_b + h * stride_left_h
    right_batch_offset = b * stride_right_b + h * stride_right_h
    out_batch_offset = b * stride_out_b + h * stride_out_h

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M x K) -----
        # address = base + i*stride_i + k*stride_k
        left_offset = left_batch_offset \
            + offs_m[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left_ptrs = left_ptr + left_offset
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # ----- load right tile (K x N) -----
        # address = base + k*stride_k + j*stride_j
        right_offset = right_batch_offset \
            + offs_k[:, None] * stride_right_k \
            + offs_n[None, :] * stride_right_j
        right_ptrs = right_ptr + right_offset
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product: (M,K) @ (K,N) -> (M,N)
        acc += tl.dot(left, right)

    # store result
    out_offset = out_batch_offset \
        + offs_m[:, None] * stride_out_i \
        + offs_n[None, :] * stride_out_j
    out_ptrs = out_ptr + out_offset
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    The heavy O(N^3) pairwise interaction term is computed by a
    batched GEMM kernel where each head (batch * hidden_dim) performs
    a matrix product L @ R^T for the N×N pairwise maps.
    All other linear layers, LayerNorms and gating are performed
    in PyTorch (float16) for simplicity.
    """
    import torch
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, N2, d = input_tensor.shape
    assert N == N2, "Input must be square in the pairwise dimensions"
    device = input_tensor.device
    eps = 1e-5

    # Cast everything to float16 (except the final accumulator)
    x = input_tensor.to(torch.float16)

    # Load weights and cast
    norm_w = weights["norm.weight"].to(torch.float16)
    norm_b = weights["norm.bias"].to(torch.float16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # ---------- LayerNorm ----------
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ---------- Linear projections ----------
    # flatten for linear
    x_flat = x.view(-1, dim)  # shape (B*N*N, dim)
    left = torch.nn.functional.linear(x_flat, left_proj_w).view(bs, N, N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_w).view(bs, N, N, hidden_dim)

    # ---------- Mask ----------
    # config may contain 'nomask' key; if True, ignore mask
    apply_mask = not config.get("nomask", True)
    if apply_mask:
        # ensure mask is float16 and broadcastable
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- Gates ----------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, left_gate_w)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, right_gate_w)).view(bs, N, N, hidden_dim)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_flat, out_gate_w)).view(bs, N, N, hidden_dim)

    left = left * left_gate
    right = right * right_gate

    # ---------- Prepare tensors for Triton kernel ----------
    # We need left as (B, H, N, N) where last two dims correspond to (i,k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right must be transposed on the last two axes to become (k,j)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N)

    # Cast to half for kernel
    left_perm = left_perm.to(torch.float16)
    right_perm = right_perm.to(torch.float16)

    # Output accumulator in float32
    out_acc = torch.empty((bs, hidden_dim, N, N), dtype=torch.float32, device=device)

    # Get strides (in elements, not bytes)
    stride_left_b, stride_left_h, stride_left_i, stride_left_k = left_perm.stride()
    stride_right_b, stride_right_h, stride_right_k, stride_right_j = right_perm.stride()
    stride_out_b, stride_out_h, stride_out_i, stride_out_j = out_acc.stride()

    # Triton kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (bs * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_acc,
        bs, hidden_dim, N,
        stride_left_b, stride_left_h, stride_left_i, stride_left_k,
        stride_right_b, stride_right_h, stride_right_k, stride_right_j,
        stride_out_b, stride_out_h, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_acc.permute(0, 2, 3, 1)  # (B, N, N, H)

    # ---------- To_out LayerNorm ----------
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # ---------- Apply out_gate ----------
    out = out * out_gate

    # ---------- Final linear projection ----------
    out_flat = out.view(-1, hidden_dim)
    out = torch.nn.functional.linear(out_flat, to_out_w).view(bs, N, N, dim)

    # Cast final output to float32 as required
    out = out.to(torch.float32)
    return out

"""
TriMul (outgoing) forward pass with a custom Triton kernel.

The heavy O(N³) term
    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
is implemented as a batched GEMM where each (batch, head) pair computes
    L (N×N) @ Rᵀ (N×N)  →  (N×N)
The remaining linear layers, LayerNorms and gating are performed in PyTorch.
All tensor arguments are expected to reside on the CUDA device.
The final output is returned as float32.
"""
import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply L @ Rᵀ  (float16 → float32)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # pointers
    left_ptr,          # const half*   [B, H, N, N]   (i, k)
    right_ptr,         # const half*   [B, H, N, N]   (k, j)  (already transposed)
    out_ptr,           #       float*  [B, H, N, N]   (i, j)
    # sizes / strides
    B, H, N,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_k, stride_r_j,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # Thread / block identification
    # --------------------------------------------------------------
    batch_id = tl.program_id(0)          # combined batch*head   (0 … B*H‑1)
    pid_m    = tl.program_id(1)          # block‑row index      (i tiles)
    pid_n    = tl.program_id(2)          # block‑col index      (j tiles)

    # de‑compose batch_id → (b, h)
    b = batch_id // H
    h = batch_id % H

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑axis
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑axis
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointer offsets for the current (b, h) pair
    left_batch_offset  = b * stride_l_b + h * stride_l_h
    right_batch_offset = b * stride_r_b + h * stride_r_h
    out_batch_offset   = b * stride_o_b + h * stride_o_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # K‑loop (tile over the reduction dimension)
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load L (M x K) -------------------------------------------------
        # address: (..., i, k)
        left_offset = left_batch_offset \
                    + offs_m[:, None] * stride_l_i \
                    + offs_k[None, :] * stride_l_k
        left_ptrs = left_ptr + left_offset
        left_tile = tl.load(left_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)                     # half -> promoted to fp32 inside dot

        # ---- load R (K x N) -------------------------------------------------
        # address: (..., k, j)   (right is already transposed)
        right_offset = right_batch_offset \
                     + offs_k[:, None] * stride_r_k \
                     + offs_n[None, :] * stride_r_j
        right_ptrs = right_ptr + right_offset
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)

        # ---- dot product ----------------------------------------------------
        # left_tile: (M, K), right_tile: (K, N) → (M, N)
        acc += tl.dot(left_tile, right_tile)

    # --------------------------------------------------------------
    # Write back the result (still FP32)
    # --------------------------------------------------------------
    out_offset = out_batch_offset \
               + offs_m[:, None] * stride_o_i \
               + offs_n[None, :] * stride_o_j
    out_ptrs = out_ptr + out_offset
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape (bs, seq_len, seq_len, dim) in float32.
    """
    input_tensor, mask, weights, config = data

    # --------------------- configuration ---------------------
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    apply_mask = not config.get("nomask", True)        # False → ignore mask

    device = input_tensor.device
    eps = 1e-5

    # --------------------- cast to fp16 for compute ---------------------
    x = input_tensor.to(torch.float16)

    # --------------------- load and cast weights -----------------------
    norm_w          = weights["norm.weight"].to(torch.float16)
    norm_b          = weights["norm.bias"].to(torch.float16)
    left_proj_w     = weights["left_proj.weight"].to(torch.float16)
    right_proj_w    = weights["right_proj.weight"].to(torch.float16)
    left_gate_w     = weights["left_gate.weight"].to(torch.float16)
    right_gate_w    = weights["right_gate.weight"].to(torch.float16)
    out_gate_w      = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w   = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b   = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w        = weights["to_out.weight"].to(torch.float16)

    bs, N, N2, _ = x.shape
    assert N == N2, "Pairwise dimensions must be square"

    # --------------------- LayerNorm on input ------------------------
    x = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # --------------------- Linear projections -------------------------
    x_flat = x.view(-1, dim)                                 # (B*N*N, dim)

    left  = F.linear(x_flat, left_proj_w).view(bs, N, N, hidden_dim)   # (B,N,N,H)
    right = F.linear(x_flat, right_proj_w).view(bs, N, N, hidden_dim)  # (B,N,N,H)

    # --------------------- optional mask ------------------------------
    if apply_mask:
        # mask shape: (B, N, N) → (B, N, N, 1) → half
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left  = left * mask_f
        right = right * mask_f

    # --------------------- gating ------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(bs, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(bs, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(bs, N, N, hidden_dim)

    left  = left * left_gate
    right = right * right_gate

    # --------------------- prepare tensors for Triton -----------------
    # left: (B, N, N, H) → (B, H, N, N)  (i, k)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    # right must be (k, j) for the gemm → transpose last two axes
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    left_perm  = left_perm.to(torch.float16)
    right_perm = right_perm.to(torch.float16)

    # Output accumulator (float32) with shape (B, H, N, N)
    out_acc = torch.empty((bs, hidden_dim, N, N), dtype=torch.float32, device=device)

    # --------------------- strides (in elements) ---------------------
    stride_l_b, stride_l_h, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_h, stride_r_k, stride_r_j = right_perm.stride()
    stride_o_b, stride_o_h, stride_o_i, stride_o_j = out_acc.stride()

    # --------------------- launch Triton kernel ---------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (bs * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out_acc,
        bs, hidden_dim, N,
        stride_l_b, stride_l_h, stride_l_i, stride_l_k,
        stride_r_b, stride_r_h, stride_r_k, stride_r_j,
        stride_o_b, stride_o_h, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # --------------------- reshape back to (B, N, N, H) -------------
    out = out_acc.permute(0, 2, 3, 1)          # (B, N, N, H) still float32
    out = out.to(torch.float16)                # half for the remaining ops

    # --------------------- to_out LayerNorm -------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # --------------------- apply out_gate ---------------------------
    out = out * out_gate

    # --------------------- final linear projection ------------------
    out_flat = out.view(-1, hidden_dim)                     # (B*N*N, H)
    out = F.linear(out_flat, to_out_w).view(bs, N, N, dim) # (B, N, N, dim)

    # --------------------- cast to float32 (required) ---------------
    out = out.to(torch.float32)
    return out
