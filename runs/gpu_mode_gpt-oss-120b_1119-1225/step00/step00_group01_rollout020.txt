"""
custom_kernel
==============

Implementation of the “outgoing” TriMul operator from AlphaFold‑3 (paper Figure 2c).

The forward pass consists of:
1. Layer‑Norm over the channel dimension `dim`.
2. Two linear projections (`left_proj`, `right_proj`) + per‑position gating.
3. Optional binary mask that zeroes out masked pair‑wise entries.
4. A dense pair‑wise reduction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is an O(N³·C) operation (N = sequence length, C = hidden_dim).
   This reduction is fused into a custom Triton kernel that performs a batched
   matrix‑multiply for each (batch, hidden) slice, using tiling and `tl.dot`
   (Tensor‑core friendly when the data type is float16).
5. Layer‑Norm on the hidden dimension, multiplication by an output gate, and a
   final linear projection back to the original `dim`.

Only the heavy pair‑wise multiplication is executed on the GPU via Triton;
all surrounding operations are performed with regular PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the einsum('... i k d, ... j k d -> ... i j d')
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    a_ptr,               # *reinterpret<const float> left  [B, N, N, C]
    b_ptr,               # *reinterpret<const float> right [B, N, N, C]
    c_ptr,               # *reinterpret<float>       out   [B, N, N, C]
    N,                   # seq_len (int)
    stride_batch,        # stride to next batch element   (int)
    stride_i,            # stride to next i index (int)  (N*C)
    stride_k,            # stride to next k index (int)  (C)
    stride_d,            # stride to next channel (int)  (1)
    hidden_dim,          # C (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # block row (i)
    pid_j = tl.program_id(1)   # block col (j)
    pid_z = tl.program_id(2)   # combined batch * channel index

    # ------------------------------------------------------------------
    # 1) Resolve batch and channel from pid_z
    # ------------------------------------------------------------------
    b = pid_z // hidden_dim          # batch index
    d = pid_z % hidden_dim           # channel index

    # ------------------------------------------------------------------
    # 2) Global i / j coordinates for this block
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    # ------------------------------------------------------------------
    # 3) Accumulator (float32 for numerical safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # ------------------------------------------------------------------
        # Load a block of left  : shape (BLOCK_M, BLOCK_K)
        #   a[b, i, k, d]
        # ------------------------------------------------------------------
        a_ptrs = (
            a_ptr
            + b * stride_batch
            + i[:, None] * stride_i
            + k[None, :] * stride_k
            + d * stride_d
        )
        a = tl.load(a_ptrs,
                    mask=(i_mask[:, None] & k_mask[None, :]),
                    other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Load a block of right (transposed) : shape (BLOCK_K, BLOCK_N)
        #   b[b, j, k, d]  --> we need bᵀ[k, j]
        # ------------------------------------------------------------------
        b_ptrs = (
            b_ptr
            + b * stride_batch
            + k[:, None] * stride_k
            + j[None, :] * stride_i
            + d * stride_d
        )
        b = tl.load(b_ptrs,
                    mask=(k_mask[:, None] & j_mask[None, :]),
                    other=0.0).to(tl.float32)

        # ------------------------------------------------------------------
        # Block‑level matrix multiply and accumulate
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)                     # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # 4) Write the result back (output layout: [B, i, j, d])
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + b * stride_batch
        + i[:, None] * stride_i
        + j[None, :] * stride_k     # stride for the j‑dimension equals stride_k
        + d * stride_d
    )
    tl.store(c_ptrs, acc,
             mask=(i_mask[:, None] & j_mask[None, :]))

# Entry point exposed to the benchmarking harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load model weights (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) + optional masking
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    if mask is not None:
        # mask shape: [B, N, N] → broadcast over hidden_dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid of linear transforms) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Pair‑wise reduction via Triton (einsum over the middle axis)
    # ------------------------------------------------------------------
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # [B, N, N, hidden_dim]

    # Tensor strides (int64)
    stride_batch = left.stride(0)  # B‑axis stride
    stride_i = left.stride(1)      # i‑axis stride (N * hidden_dim)
    stride_k = left.stride(2)      # k‑axis stride (hidden_dim)
    stride_d = left.stride(3)      # channel stride (1)

    N = left.shape[1]          # seq_len
    B = left.shape[0]

    # Tunable tile size – chosen to fit well on H100 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        N,
        stride_batch,
        stride_i,
        stride_k,
        stride_d,
        hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 5) Final LayerNorm, gating, and output projection back to `dim`
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
