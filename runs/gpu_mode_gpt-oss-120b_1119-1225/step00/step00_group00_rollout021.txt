def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not nomask:
        # mask broadcast
        mask_ = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute for batched matmul
    # shape [B, H, N, N]
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape  # Actually left shape (B, H, N, N)
    # allocate output
    out = torch.empty((B, H, N, N), dtype=torch.float32, device=left.device)

    # define kernel
    @triton.jit
    def batched_matmul_kernel(
        left_ptr, right_ptr, out_ptr,
        BH, N,
        stride_lbh, stride_lm, stride_lk,
        stride_rbh, stride_rn, stride_rk,
        stride_obh, stride_om, stride_on,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        pid_z = tl.program_id(2)  # combined batch*hidden index
        pid_m = tl.program_id(0)  # row tile
        pid_n = tl.program_id(1)  # col tile

        # offsets
        m_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        n_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        # mask for bounds
        m_mask = m_offset < N
        n_mask = n_offset < N

        # accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        for kb in range(0, tl.cdiv(N, BLOCK_K)):
            k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
            k_mask = k_offset < N

            # load left block (M x K)
            left = tl.load(
                left_ptr
                + pid_z * stride_lbh
                + (m_offset[:, None] * stride_lm)
                + (k_offset[None, :] * stride_lk),
                mask=m_mask[:, None] & k_mask[None, :],
                other=0.0,
            )
            # load right block transposed (K x N)
            right = tl.load(
                right_ptr
                + pid_z * stride_rbh
                + (k_offset[:, None] * stride_rk)
                + (n_offset[None, :] * stride_rn),
                mask=k_mask[:, None] & n_mask[None, :],
                other=0.0,
            )
            # cast to fp32 for accumulation
            left_f32 = left.to(tl.float32)
            right_f32 = right.to(tl.float32)
            acc += tl.dot(left_f32, right_f32)

"""
TriMul outgoing kernel (AlphaFold3) implementation.
- LayerNorm over input channel dimension.
- Linear projections and gating (computed in PyTorch).
- Core pairwise interaction: for each hidden dimension h,
  compute out_h = left_h @ right_h^T using a custom Triton kernel that
  batch processes B*H independent matrix multiplications.
- Followed by output LayerNorm, gating and final linear projection.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    BH, N,
    stride_lbh, stride_lm, stride_lk,
    stride_rbh, stride_rn, stride_rk,
    stride_obh, stride_om, stride_on,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_z = tl.program_id(2)  # combined batch*hidden index
    pid_m = tl.program_id(0)  # row tile
    pid_n = tl.program_id(1)  # column tile

    # Offsets within the N dimension
    m_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds masks
    m_mask = m_offset < N
    n_mask = n_offset < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kb * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left: (M, K)
        left = tl.load(
            left_ptr
            + pid_z * stride_lbh
            + (m_offset[:, None] * stride_lm)
            + (k_offset[None, :] * stride_lk),
            mask=m_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # right (transposed): (K, N)
        right = tl.load(
            right_ptr
            + pid_z * stride_rbh
            + (k_offset[:, None] * stride_rk)
            + (n_offset[None, :] * stride_rn),
            mask=k_mask[:, None] & n_mask[None, :],
            other=0.0,
        )
        # Promote to fp32 and accumulate
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # write back
    c_ptr = out_ptr + pid_z * stride_obh + (m_offset[:, None] * stride_om) + (n_offset[None, :] * stride_on)
    tl.store(c_ptr, acc, mask=m_mask[:, None] & n_mask[None, :])

def custom_kernel(data):
    """
    Custom Triton implementation of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, dim]
        - mask: torch.Tensor of shape [B, N, N] (optional)
        - weights: dict of torch.Tensor containing module weights
        - config: dict containing keys 'dim', 'hidden_dim', 'nomask' (optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] resulting from the TriMul forward pass.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------- 1. LayerNorm on input ----------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------- 2. Linear projections ----------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ----------- 3. Optional mask ----------
    if not nomask:
        # mask broadcast to last dimension
        mask_ = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # ----------- 4. Gating ----------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ----------- 5. Cast to half and permute for batched gemm ----------
    left = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.to(torch.float16).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left.shape

    # Output buffer for the batched gemm (float32)
    out = torch.empty((B, H, N, N), dtype=torch.float32, device=left.device)

    # Strides (in elements)
    stride_lbh = left.stride(1)   # stride to next hidden (or batch*hidden) slice
    stride_lm = left.stride(2)    # row stride (N)
    stride_lk = left.stride(3)    # col stride (1)

    stride_rbh = right.stride(1)
    stride_rn = right.stride(2)   # row stride (N)
    stride_rk = right.stride(3)   # col stride (1)

    stride_obh = out.stride(1)
    stride_om = out.stride(2)
    stride_on = out.stride(3)

    BH = B * H
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        BH,
    )

    # Launch the kernel
    batched_matmul_kernel[grid](
        left,
        right,
        out,
        BH,
        N,
        stride_lbh,
        stride_lm,
        stride_lk,
        stride_rbh,
        stride_rn,
        stride_rk,
        stride_obh,
        stride_om,
        stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------- 6. Permute back to [B, N, N, H] ----------
    out = out.permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ----------- 7. Output LayerNorm ----------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ----------- 8. Apply out gate ----------
    out = out * out_gate

    # ----------- 9. Final linear projection ----------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing version) – Triton‑accelerated core.

Algorithm
---------
1. LayerNorm over the last channel dimension (dim).
2. Linear projections `left_proj`, `right_proj` (dim → hidden_dim) and optional
   masking.
3. Compute three sigmoid gates (left, right, out) from the same normalized input.
4. Apply left/right gates to the projected tensors.
5. Core pairwise operation:
   For each hidden dimension h compute
        out_h = left_h @ right_hᵀ
   i.e. a batched matrix multiplication of B·hidden_dim matrices of size N×N.
   This is performed with a custom Triton kernel that fuses the inner‐k loop.
6. Output LayerNorm (over hidden_dim), multiply by the out‑gate and a final
   linear projection back to `dim`.

Only the N³ inner product (step 5) is written in Triton; all other
operations use native PyTorch (which already runs on the GPU).  The kernel
operates on half‑precision inputs and accumulates in fp32 for numerical
stability.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    left_ptr,        # [BH, N, N] (fp16)
    right_ptr,       # [BH, N, N] (fp16)
    out_ptr,         # [BH, N, N] (fp32)
    BH, N,           # runtime scalars
    stride_lbh, stride_lm, stride_lk,   # left strides
    stride_rbh, stride_rn, stride_rk,   # right strides (row, col)
    stride_obh, stride_om, stride_on,   # out strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program ids
    pid_z = tl.program_id(2)                # combined batch * hidden index
    pid_m = tl.program_id(0)                # tile row
    pid_n = tl.program_id(1)                # tile col

    # tile offsets
    m_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    n_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks for rows/cols
    m_mask = m_offset < N
    n_mask = n_offset < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for kb in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = kb * BLOCK_K
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left block: shape (BLOCK_M, BLOCK_K)
        left = tl.load(
            left_ptr
            + pid_z * stride_lbh
            + (m_offset[:, None] * stride_lm)
            + (k_offset[None, :] * stride_lk),
            mask=m_mask[:, None] & k_mask[None, :],
            other=0.0,
        )
        # right block **transposed**: shape (BLOCK_K, BLOCK_N)
        # original right layout is (row = N, col = N); we need rightᵀ,
        # so we read original (col=k, row=n)
        right = tl.load(
            right_ptr
            + pid_z * stride_rbh
            + (k_offset[:, None] * stride_rk)        # original col stride
            + (n_offset[None, :] * stride_rn),       # original row stride
            mask=k_mask[:, None] & n_mask[None, :],
            other=0.0,
        )
        # promote to fp32 before dot
        acc += tl.dot(left.to(tl.float32), right.to(tl.float32))

    # store result with masks
    out = out_ptr + pid_z * stride_obh + (m_offset[:, None] * stride_om) + (n_offset[None, :] * stride_on)
    tl.store(out, acc, mask=m_mask[:, None] & n_mask[None, :])

    """
    Triton‑backed forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor, shape [B, N, N] (or None)
        - weights      : dict of torch.Tensor containing module parameters
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask`` (optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (float32)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # -----------------------------------------------------------------
    # 4) Gating (float32)
    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Batched matrix multiplication (N³) using Triton
    # -----------------------------------------------------------------
    # cast to fp16 and reorder to [B, hidden_dim, N, N]
    left = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()
    right = right.to(torch.float16).permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape  # shape now [B, hidden_dim, N, N]

    # allocate output buffer (float32)
    out = torch.empty((B, H, N, N), dtype=torch.float32, device=left.device)

    # strides in element counts
    stride_lbh = left.stride(1)   # stride to next hidden (or batch*hidden) slice = N*N
    stride_lm = left.stride(2)    # row stride N
    stride_lk = left.stride(3)    # col stride 1

    stride_rbh = right.stride(1)
    stride_rn = right.stride(2)   # original row stride N
    stride_rk = right.stride(3)   # original col stride 1

    stride_obh = out.stride(1)
    stride_om = out.stride(2)
    stride_on = out.stride(3)

    BH = B * H                     # total number of independent GEMMs
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),   # rows
        triton.cdiv(N, BLOCK_N),   # cols
        BH,                        # batch*hidden
    )

    # launch Triton kernel
    batched_matmul_kernel[grid](
        left,
        right,
        out,
        BH,
        N,
        stride_lbh,
        stride_lm,
        stride_lk,
        stride_rbh,
        stride_rn,
        stride_rk,
        stride_obh,
        stride_om,
        stride_on,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # bring result back to [B, N, N, hidden_dim]
    out = out.permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 6) Output LayerNorm over hidden_dim
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -----------------------------------------------------------------
    # 7) Apply out‑gate and final linear projection
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
