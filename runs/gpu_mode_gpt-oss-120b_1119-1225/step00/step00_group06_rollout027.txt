def fuse_gate_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_d,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * D
    mask = offs < total
    # compute indices
    b = offs // (N * N * D)
    off1 = offs % (N * N * D)
    i = off1 // (N * D)
    off2 = off1 % (N * D)
    k = off2 // D
    d = off2 % D

    mask_offset = b * stride_mask_b + i * stride_mask_i + k * stride_mask_k
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask).to(tl.float32)

    left_out = left_val * left_gate_val * mask_val
    tl.store(left_ptr + left_offset, left_out, mask=mask)

"""
TriMul (outgoing) implementation.

The core of the TriMul operator is the 4‑D Einstein‑sum
    out[b, i, j, d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
This is equivalent to a batched matrix multiplication per hidden
dimension d:   L_d @ R_dᵀ  (L_d = left[..., i, k, d],
                            R_d = right[..., j, k, d]).
We keep the heavy GEMM on CUDA (torch.matmul) and fuse the
mask‑gating step with a small Triton kernel to avoid extra
memory traffic. The kernel reads the mask, left‑gate and right‑gate
values and overwrites `left` and `right` in‑place:

    left  = left  * left_gate  * mask
    right = right * right_gate * mask

The rest of the module (layer‑norms, linear projections and the final
output gate) is expressed with standard torch operations, which are
highly optimized on H100.

The function `custom_kernel` receives a tuple
    (input_tensor, mask, weights, config)
and returns the processed tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_and_gates_kernel(
    left_ptr,               # *float32
    right_ptr,              # *float32
    left_gate_ptr,          # *float32
    right_gate_ptr,         # *float32
    mask_ptr,               # *float32 (broadcasted)
    B, N, D,                # compile‑time ints
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_d,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_d,
    stride_mask_b, stride_mask_i, stride_mask_k,
    BLOCK_SIZE: tl.constexpr,
):
    """In‑place: left  = left  * left_gate  * mask,
                right = right * right_gate * mask
       The mask does not have a channel dimension; it is broadcasted
       across D."""
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * D
    mask_oob = offs >= total
    # guard out‑of‑bounds threads
    offs = tl.where(mask_oob, tl.zeros_like(offs), offs)

    # ---- decode linear index into (b, i, k, d) -----------------
    # off = ((b * N + i) * N + k) * D + d
    b = offs // (N * N * D)
    off1 = offs % (N * N * D)
    i = off1 // (N * D)
    off2 = off1 % (N * D)
    k = off2 // D
    d = off2 % D

    # ---- load mask (broadcasted over d) -----------------------
    mask_offset = (
        b * stride_mask_b
        + i * stride_mask_i
        + k * stride_mask_k
    )
    mask_val = tl.load(mask_ptr + mask_offset, mask=~mask_oob).to(tl.float32)

    # ---- left ----------------------------------------------------------------
    left_offset = (
        b * stride_left_b
        + i * stride_left_i
        + k * stride_left_k
        + d * stride_left_d
    )
    left_val = tl.load(left_ptr + left_offset, mask=~mask_oob).to(tl.float32)

    lgate_offset = (
        b * stride_lgate_b
        + i * stride_lgate_i
        + k * stride_lgate_k
        + d * stride_lgate_d
    )
    lgate_val = tl.load(left_gate_ptr + lgate_offset, mask=~mask_oob).to(tl.float32)

    left_out = left_val * lgate_val * mask_val
    tl.store(left_ptr + left_offset, left_out, mask=~mask_oob)

    # ---- right ---------------------------------------------------------------
    right_offset = (
        b * stride_right_b
        + i * stride_right_i
        + k * stride_right_k
        + d * stride_right_d
    )
    right_val = tl.load(right_ptr + right_offset, mask=~mask_oob).to(tl.float32)

    rgate_offset = (
        b * stride_rgate_b
        + i * stride_rgate_i
        + k * stride_rgate_k
        + d * stride_rgate_d
    )
    rgate_val = tl.load(right_gate_ptr + rgate_offset, mask=~mask_oob).to(tl.float32)

    right_out = right_val * rgate_val * mask_val
    tl.store(right_ptr + right_offset, right_out, mask=~mask_oob)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be all‑ones when `nomask`)
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys `dim`, `hidden_dim`, `nomask` (optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------------- #
    # 1) unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 2) Input LayerNorm
    # --------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --------------------------------------------------------------------- #
    # 3) Linear projections (no bias) → left / right
    # --------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------- #
    # 4) Gating streams (sigmoid)
    # --------------------------------------------------------------------- #
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # --------------------------------------------------------------------- #
    # 5) Fuse mask + gates (Triton kernel) or simple element‑wise multiply
    # --------------------------------------------------------------------- #
    if nomask:
        # mask == 1 ⇒ only apply gating
        left = left * left_gate
        right = right * right_gate
    else:
        # make sure mask is floating point and on the same device
        mask_f = mask.to(input_tensor.dtype)
        total_elements = left.numel()
        BLOCK = 1024  # 1024 threads per program – good occupancy on H100

        # strides (in elements, not bytes)
        sL = left.stride()
        sR = right.stride()
        sLg = left_gate.stride()
        sRg = right_gate.stride()
        sM = mask_f.stride()

        grid = (triton.cdiv(total_elements, BLOCK),)

        _apply_mask_and_gates_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            mask_ptr=mask_f,
            B=B,
            N=N,
            D=hidden_dim,
            stride_left_b=sL[0],
            stride_left_i=sL[1],
            stride_left_k=sL[2],
            stride_left_d=sL[3],
            stride_right_b=sR[0],
            stride_right_i=sR[1],
            stride_right_k=sR[2],
            stride_right_d=sR[3],
            stride_lgate_b=sLg[0],
            stride_lgate_i=sLg[1],
            stride_lgate_k=sLg[2],
            stride_lgate_d=sLg[3],
            stride_rgate_b=sRg[0],
            stride_rgate_i=sRg[1],
            stride_rgate_k=sRg[2],
            stride_rgate_d=sRg[3],
            stride_mask_b=sM[0],
            stride_mask_i=sM[1],
            stride_mask_k=sM[2],
            BLOCK_SIZE=BLOCK,
        )
        # Gates are no longer needed; free memory immediately.
        del left_gate, right_gate

    # --------------------------------------------------------------------- #
    # 6) Core pairwise multiplication  (einsum) via batched GEMM
    # --------------------------------------------------------------------- #
    # left:  [B, N, N, D] -> [B, D, N, N]
    left_perm = left.permute(0, 3, 1, 2)
    # right: [B, N, N, D] -> [B, D, N, N] with (k, j) swapped
    right_perm = right.permute(0, 3, 2, 1)   # now (k, j)

    # Batched matrix multiplication across the last two dims:
    # (B, D, i, k) @ (B, D, k, j) -> (B, D, i, j)
    out_perm = torch.matmul(left_perm, right_perm)

    # Back to [B, N, N, D]
    out = out_perm.permute(0, 2, 3, 1)

    # --------------------------------------------------------------------- #
    # 7) Output LayerNorm (over hidden_dim)
    # --------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------------- #
    # 8) Output gate
    # --------------------------------------------------------------------- #
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 9) Final linear projection back to original dimension
    # --------------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"])

    return out
```
