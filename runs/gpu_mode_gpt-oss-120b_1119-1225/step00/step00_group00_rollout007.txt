def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data

    # Extract config params
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5 (from LayerNorm default presumably but maybe weights include eps? Not needed.)

    # x_norm
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5
    )

    # linear projections using torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask multiplication
    if mask_tensor is not None:
        # mask shape [B,N,N], broadcast
        mask = mask_tensor.unsqueeze(-1).type_as(left)  # float32
        left = left * mask
        right = right * mask

    # gate projections
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Shape for kernel
    B, N, _, _ = left.shape
    # hidden_dim = left.shape[-1] # check matches config hidden_dim

    # Reshape for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (K,J) as (K,J) -> (N,N)
    # Flatten group dimension B*H
    G = B * hidden_dim
    left_flat = left_perm.view(G, N, N)
    right_flat = right_perm.view(G, N, N)

    # Batched matmul using Triton
    out_flat = batched_matmul_triton(left_flat, right_flat)

    # Reshape back to [B,N,N,H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5
    )
    # gating
    out = out * out_gate

    # final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_g_a, stride_m_a, stride_k_a,
    stride_g_b, stride_k_b, stride_n_b,
    stride_g_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program IDs
    pid_g = tl.program_id(2)  # group id
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k in range(0, num_k_blocks):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute pointers
        a_ptrs = a_ptr + pid_g * stride_g_a + offs_m[:, None] * stride_m_a + offs_k[None, :] * stride_k_a
        b_ptrs = b_ptr + pid_g * stride_g_b + offs_k[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

        # masks
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # load
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # dot product, accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store result
    c_ptrs = c_ptr + pid_g * stride_g_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    # convert acc to same dtype as output (fp16)
    acc_cast = acc.to(tl.float16)
    tl.store(c_ptrs, acc_cast, mask=mask_c)

   right *= sigmoid(x @ W_right_gate)
5️⃣ Fuse the heavy N³ operation with a Triton batched‑matmul:
   out[b, i, j, h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   This is a batched GEMM over groups G = B·H:
       A₍g₎ = left_perm[g]  (M×K)   with M=N, K=N
       B₍g₎ = right_perm[g] (K×N)   (right already transposed)
   The kernel works in FP16 (inputs) with FP32 accumulation
   and writes FP16 results.
6️⃣ Convert result back to FP32, reshape to [B, N, N, H].
7️⃣ Layer‑norm over the hidden dimension, apply the output gate,
   and a final linear projection back to `dim`.
   The final tensor shape is [B, N, N, dim].

The Triton kernel (`batched_matmul_kernel`) processes each group
independently, using 2‑D tiling (BLOCK_M×BLOCK_N) and a K‑loop
with BLOCK_K.  Masks guard out‑of‑bounds loads.  The kernel is the
only non‑trivial part of the computation; everything else stays
in PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict, Any


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B) where A:[G,M,K], B:[G,K,N]
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dims
    stride_g_a, stride_m_a, stride_k_a,
    stride_g_b, stride_k_b, stride_n_b,
    stride_g_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    # program IDs
    # --------------------------------------------------------------
    pid_g = tl.program_id(2)          # group id  (0 … G-1)
    pid_m = tl.program_id(0)          # row block  (0 … ceil(M/BLOCK_M)-1)
    pid_n = tl.program_id(1)          # col block  (0 … ceil(N/BLOCK_N)-1)

    # --------------------------------------------------------------
    # offsets within the tile
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # accumulator (FP32)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(K, BLOCK_K)

    # --------------------------------------------------------------
    # main K‑loop
    # --------------------------------------------------------------
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # compute global pointers
        a_ptrs = a_ptr + pid_g * stride_g_a \
                        + offs_m[:, None] * stride_m_a \
                        + offs_k[None, :] * stride_k_a
        b_ptrs = b_ptr + pid_g * stride_g_b \
                        + offs_k[:, None] * stride_k_b \
                        + offs_n[None, :] * stride_n_b

        # out‑of‑bounds masks
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        # load FP16 values (other=0.0 for padding)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # dot product, accumulate in FP32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # write back result (convert to FP16 to save bandwidth)
    # --------------------------------------------------------------
    c_ptrs = c_ptr + pid_g * stride_g_c \
                   + offs_m[:, None] * stride_m_c \
                   + offs_n[None, :] * stride_n_c
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B using Triton.
    A: [G, M, K] (FP16)
    B: [G, K, N] (FP16)
    Returns C: [G, M, N] (FP16)
    """
    assert a.is_contiguous() and b.is_contiguous(), "inputs must be contiguous"
    G, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2, f"Incompatible inner dimensions {K} vs {K2}"

    # Allocate output (same dtype as A/B → FP16)
    c = torch.empty((G, M, N), dtype=a.dtype, device=a.device)

    # Strides in number of elements (not bytes)
    stride_g_a, stride_m_a, stride_k_a = a.stride()
    stride_g_b, stride_k_b, stride_n_b = b.stride()
    stride_g_c, stride_m_c, stride_n_c = c.stride()

    # Tile sizes – tuned for H100 (FP16 compute)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (M_tiles, N_tiles, G)
    grid = (triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N), G)

    batched_matmul_kernel[grid](
        a, b, c,
        M, N, K,
        stride_g_a, stride_m_a, stride_k_a,
        stride_g_b, stride_k_b, stride_n_b,
        stride_g_c, stride_m_c, stride_n_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return c


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict[str, Any]]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # extract weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (dim → hidden_dim), no bias
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).type_as(left)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) and apply
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to FP16 and prepare for Triton batched matmul
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to [B, H, N, N] where H = hidden_dim
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (transposed on the last two axes)

    # Collapse batch and hidden dimensions → groups G = B * H
    G = bs * hidden_dim
    left_flat  = left_perm.view(G, seq_len, seq_len)   # [G, N, N]
    right_flat = right_perm.view(G, seq_len, seq_len)  # [G, N, N]

    # ------------------------------------------------------------------
    # 5️⃣ Triton kernel: batched GEMM across the `k` dimension
    # ------------------------------------------------------------------
    out_flat_fp16 = batched_matmul_triton(left_flat, right_flat)  # [G, N, N] in FP16

    # Free memory that is no longer needed (optional, helps with large inputs)
    del left, right, left_perm, right_perm, left_flat, right_flat

    # ------------------------------------------------------------------
    # 6️⃣ Convert back to FP32 and reshape to the original layout
    # ------------------------------------------------------------------
    out = out_flat_fp16.to(torch.float32)
    out = out.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7️⃣ Final LayerNorm, gating, and output projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, H]

    out = out * out_gate  # broadcast over hidden dim

    out = F.linear(out, to_out_weight)  # [B, N, N, dim] (no bias)

    return out
```
