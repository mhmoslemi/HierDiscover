def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Move everything to contiguous on GPU
    x = input_tensor
    B, N, _, C = x.shape
    # asserts C == dim maybe?

    # LayerNorm via Triton
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = layernorm_triton(x, norm_weight, norm_bias, eps)

    # Linear projections (float32)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Compute projections
    left = F.linear(x_norm, left_proj_weight)  # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # Apply mask if present (unless nomask)
    if not config.get("nomask", False):
        # mask shape [B,N,N], broadcast to hidden_dim
        mask = mask_tensor.to(x_norm.dtype)  # ensure same dtype
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Compute gates
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Compute out via batched matrix multiplication
    # reshape for BMM
    left_bmm = left.permute(0, 3, 1, 2).contiguous()  # (B, hidden_dim, N, N)
    right_bmm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden_dim, N, N)
    # merge batch and hidden_dim
    left_flat = left_bmm.view(B * hidden_dim, N, N)
    right_flat = right_bmm.view(B * hidden_dim, N, N)
    out_flat = torch.bmm(left_flat, right_flat)  # (B*hidden_dim, N, N)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # back to (B,N,N,hidden_dim)

    # to_out_norm (layernorm over hidden_dim)
    # Use torch.nn.functional.layer_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out = out * out_gate

    # Final linear projection to dim
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul forward (outgoing) fused implementation.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the last dimension using a
   custom Triton kernel (computes mean/variance per (batch, i, j) entry).
2. Linear projections for the left/right branches and three gating
   linear layers using `torch.nn.functional.linear`.
3. (optional) Apply the pairwise mask.
4. Multiply each branch by its gate (sigmoid).
5. Compute the “tri‑multiplicative” update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is realized as a batched matrix‑multiply:
      - reshape left → (B·H, N, N) with axis (i,k)
      - reshape rightᵀ → (B·H, N, N) with axis (k,j)
      - torch.bmm on the fused batch dimension H (=hidden_dim).
6. Layer‑normalize the intermediate result over the hidden dimension
   (`to_out_norm`) with the standard PyTorch implementation.
7. Apply the output gate and final linear projection back to `dim`.

Only the LayerNorm in step 1 is written in Triton; all other
operations use highly‑optimised PyTorch kernels (including cuBLAS
for the batched GEMM).  The final output is float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton LayerNorm kernel (normalizes over the last dimension)
# --------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                         # const float*  input
    out_ptr,                       # float*        output
    weight_ptr,                    # const float*  weight (ln.weight)
    bias_ptr,                      # const float*  bias   (ln.bias)
    epsilon,                       # float         eps
    B, N, C,                       # int32         batch, seq_len, channels
    stride_xb, stride_xi, stride_xj, stride_xc,   # int64 strides for x
    stride_ob, stride_oi, stride_oj, stride_oc,   # int64 strides for out
    BLOCK_C: tl.constexpr,         # compile‑time block size for channel dim
):
    """Normalize each (b,i,j) slice of shape [C] (LayerNorm)."""
    pid = tl.program_id(0)                     # one program per (b,i,j)

    # total rows = B * N * N; grid is sized accordingly
    total = B * N * N
    if pid >= total:
        return

    # compute 3‑D indices from linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for this slice
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # ----------------------------------------------------------
    # 1) Compute mean and variance across the channel dimension
    # ----------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    offset = 0
    while offset < C:
        cur = offset + tl.arange(0, BLOCK_C)
        mask = cur < C                                 # mask out‑of‑range elements
        x = tl.load(x_base + cur * stride_xc, mask=mask, other=0.0)
        sum_val += tl.sum(x, axis=0)                  # scalar sum
        sum_sq  += tl.sum(x * x, axis=0)              # scalar sum of squares
        offset += BLOCK_C

    mean = sum_val / tl.float32(C)
    var  = sum_sq / tl.float32(C) - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # ----------------------------------------------------------
    # 2) Write the normalized output (with affine transform)
    # ----------------------------------------------------------
    offset = 0
    while offset < C:
        cur = offset + tl.arange(0, BLOCK_C)
        mask = cur < C
        x = tl.load(x_base + cur * stride_xc, mask=mask, other=0.0)

        w = tl.load(weight_ptr + cur, mask=mask, other=0.0)
        b = tl.load(bias_ptr   + cur, mask=mask, other=0.0)

        norm = (x - mean) * inv_std
        out = norm * w + b
        tl.store(out_base + cur * stride_oc, out, mask=mask)

        offset += BLOCK_C


def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """Convenience wrapper for the Triton LayerNorm."""
    B, N, N2, C = x.shape
    assert N == N2, "TriMul expects a square pairwise tensor."
    out = torch.empty_like(x)

    # strides (in elements, not bytes) – Triton expects int64
    stride_xb, stride_xi, stride_xj, stride_xc = x.stride()
    stride_ob, stride_oi, stride_oj, stride_oc = out.stride()

    total = B * N * N
    grid = (total,)

    BLOCK_C = 64                                            # tuneable; must divide C efficiently
    layernorm_kernel[grid](
        x_ptr=x,
        out_ptr=out,
        weight_ptr=weight,
        bias_ptr=bias,
        epsilon=eps,
        B=B,
        N=N,
        C=C,
        stride_xb=stride_xb,
        stride_xi=stride_xi,
        stride_xj=stride_xj,
        stride_xc=stride_xc,
        stride_ob=stride_ob,
        stride_oi=stride_oi,
        stride_oj=stride_oj,
        stride_oc=stride_oc,
        BLOCK_C=BLOCK_C,
        num_warps=4,
    )
    return out


# --------------------------------------------------------------
# Main entry point used by the test harness
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # input channel dimension
    hidden_dim = config["hidden_dim"]  # projection dimension
    eps = 1e-5

    B, N, N2, C = input_tensor.shape
    assert N == N2, "input must be square in the pairwise axes"
    assert C == dim, f"expected input dim {dim}, got {C}"

    # ------------------------------------------------------------------
    # LayerNorm on the input (Triton)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = triton_layernorm(input_tensor, norm_weight, norm_bias, eps)

    # ------------------------------------------------------------------
    # Linear projections & gating (pure PyTorch – uses cuBLAS)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]       # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]        # (dim, hidden_dim)

    # Projections
    left  = F.linear(x_norm, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask handling
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask shape: (B,N,N)  →  (B,N,N,1) broadcast over hidden_dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)    # (B,N,N,1)
        left  = left  * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Core TriMul contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # Implemented as a batched gemm (cuBLAS) over the hidden dimension.
    # ------------------------------------------------------------------
    # reshape for batched GEMM: (B*H, N, N) @ (B*H, N, N)
    left_bmm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_bmm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with (k,j)

    left_flat  = left_bmm.view(B * hidden_dim, N, N)
    right_flat = right_bmm.view(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat)         # (B*H, N, N)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # (B,N,N,H)

    # ------------------------------------------------------------------
    # Output LayerNorm over the hidden dimension (PyTorch)
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply out gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (B,N,N,dim)

    return out
