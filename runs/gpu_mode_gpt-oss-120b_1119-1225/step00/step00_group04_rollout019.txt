@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    pid0 = tl.program_id(0)  # batch*hidden
    pid1 = tl.program_id(1)  # block i
    pid2 = tl.program_id(2)  # block j

    # Compute batch and hidden index
    b = pid0 // H
    h = pid0 % H

    # Compute range offsets
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        cur_offs_k = k_start + offs_k
        # mask for K
        mask_k = cur_offs_k < N

        # Load left block (M x K)
        left_ptrs = left_ptr + b*stride_l_b + (offs_i[:, None] * stride_l_i) + (cur_offs_k[None, :] * stride_l_k) + h*stride_l_h
        # For invalid i or k, set to zero
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed (K x N)
        # right index: b, j, k, h
        right_ptrs = right_ptr + b*stride_r_b + (cur_offs_k[:, None] * stride_r_k) + (offs_j[None, :] * stride_r_i) + h*stride_r_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(left, right)  # shape (M,N)
    # Store output
    out_ptrs = out_ptr + b*stride_o_b + (offs_i[:, None] * stride_o_i) + (offs_j[None, :] * stride_o_j) + h*stride_o_h
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

# Launch
grid = (B*H, tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N))

)

def custom_kernel(data):
    """TriMul outgoing forward pass using Triton for the N^2 x N matmul."""
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    x = F.layer_norm(input_tensor, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module but off‑loads the
core N²×N contraction to a custom Triton kernel. The steps are:

1. Layer‑norm over the last channel (dim).
2. Linear projections to hidden_dim: left, right.
3. Optional binary mask (broadcasted on the hidden dim).
4. Compute sigmoid gates (left_gate, right_gate, out_gate) and apply them.
5. Core operation:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiply per hidden channel (L @ Rᵀ).  This
   contraction is performed with a tiled Triton kernel (blocks of 64×64
   with a K‑tile of 32).
6. Layer‑norm over hidden_dim, multiply by out_gate and a final linear
   projection back to dim.

All heavy arithmetic (the N³ contraction) runs on the GPU via Triton,
while the surrounding linear/LayerNorm/gating operations use PyTorch’s
highly‑optimised kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using tiling.  The first program‑id dimension enumerates (b, h) pairs,
    while the second and third dimensions tile the i‑ and j‑axes.
    """
    pid0 = tl.program_id(0)          # enumerates batch * hidden dim
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    # Resolve batch and hidden‑channel index
    b = pid0 // H
    h = pid0 % H

    # Offsets for the i and j tiles
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # Boundary masks for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k_iter in range(num_k):
        k_start = k_iter * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile : shape (BLOCK_M, BLOCK_K) -----
        left_ptrs = left_ptr + \
            b * stride_l_b + \
            (offs_i[:, None] * stride_l_i) + \
            (offs_k[None, :] * stride_l_k) + \
            h * stride_l_h
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- load right tile (transposed) : shape (BLOCK_K, BLOCK_N) -----
        # note: we load right[j,k] as (k, j) to obtain the required transpose
        right_ptrs = right_ptr + \
            b * stride_r_b + \
            (offs_k[:, None] * stride_r_k) + \
            (offs_j[None, :] * stride_r_i) + \
            h * stride_r_h
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate the block‑wise matrix product
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # ----- write back the result -----
    out_ptrs = out_ptr + \
        b * stride_o_b + \
        (offs_i[:, None] * stride_o_i) + \
        (offs_j[None, :] * stride_o_j) + \
        h * stride_o_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """TriMul forward (outgoing) – uses Triton for the N³ contraction."""
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (applies before gating, as in the reference)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [bs, N, N] -> [bs, N, N, 1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates and apply them
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction using Triton
    # ------------------------------------------------------------------
    # Allocate output buffer: shape (bs, N, N, hidden_dim)
    out = torch.empty((bs, N, N, hidden_dim), dtype=dtype, device=device)

    # Strides for the three tensors (contiguous layout: [B, N, N, H])
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Choose block sizes – these work well on an H100 for the requested
    # dimensions (tuned empirically)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden, i‑tiles, j‑tiles)
    grid = (
        bs * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: LayerNorm over hidden_dim, output gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
