@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute i,j block
    i_block = tl.program_id(0)  # across M dimension
    j_block = tl.program_id(1)  # across N dimension
    bd = tl.program_id(2)       # combined batch*D
    b = bd // D
    d = bd % D

    # Compute offsets for i and j
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-range using conditional store later; we compute mask bools:
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Use built-in broadcast to get K index for loop
    # For each block, we iterate over k
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Pointers base for left and right
    # offset for batch and d
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd

    # loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_M, BLOCK_K)
        # left offset = i * stride_li + k * stride_lk
        left_ptrs = left_base + (offs_i[:, None] * stride_li) + (offs_k[None, :] * stride_lk)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right block transposed: shape (BLOCK_K, BLOCK_N)
        # For right: we need rows = k, columns = j
        # offset = j * stride_rj + k * stride_rk
        right_ptrs = right_base + (offs_j[None, :] * stride_rj) + (offs_k[:, None] * stride_rk)
        # Right block is shape (BLOCK_N, BLOCK_K) currently? Actually we compute as (BLOCK_K, BLOCK_N) by swapping dims
        # With above pointer expression, dimension0 is k, dimension1 is j
        # Compute pointer shape: rows = k, columns = j => (BLOCK_K, BLOCK_N)
        # Load into matrix of shape (BLOCK_K, BLOCK_N), then use as is (since tl.dot expects (M,K) and (K,N))
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Perform dot product
        # left: (BLOCK_M, BLOCK_K)
        # right: (BLOCK_K, BLOCK_N) after loading as (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right)

    # Write result to out
    out_ptrs = out_ptr + b * stride_ob + d * stride_od + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm
    # compute mean and var across last dim
    # for float32:
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]

    # linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask_expanded = mask.unsqueeze(-1).type_as(x)
    else:
        mask_expanded = torch.ones_like(x[..., :1])

    left = left_proj * mask_expanded
    right = right_proj * mask_expanded

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert left/right to half for kernel
    left_half = left.to(torch.float16).contiguous()
    right_half = right.to(torch.float16).contiguous()

    # allocate out_half
    out_half = torch.empty_like(left_half, dtype=torch.float16)

    # Get strides
    B, N, _, D = left_half.shape
    # strides in elements (not bytes):
    stride_lb, stride_li, stride_lk, stride_ld = left_half.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_half.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_half.stride()

    # Launch kernel
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,

        right_half,

        B, N, D,

        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Get indices
    i_block = tl.program_id(0)
    j_block = tl.program_id(1)
    bd = tl.program_id(2)
    b = bd // D
    d = bd % D

    # compute offsets
    # compute i and j indices
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    # mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer offset for batch and d
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    # compute number of k blocks:
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block (M x K)
        left_ptrs = left_base + (offs_i[:, None] * stride_li) + (offs_k[None, :] * stride_lk)
        # mask for left: mask_i broadcast with mask_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right block transposed: we want (K x N)

    out_ptrs = out_ptr + b * stride_ob + d * stride_od + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated core

The heavy part of the TriMul forward pass is the 4‑D Einstein sum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is mathematically a batch of D independent matrix multiplications:
    out[..., d] = left[..., d] @ right[..., d].T

The code below
* performs the LayerNorm / Linear / Sigmoid / mask steps in PyTorch,
* pushes the N³ matrix‑multiplication into a Triton kernel (half‑precision
  compute with fp32 accumulation),
* finishes the remaining LayerNorm, gating and output projection in PyTorch.

The kernel works for any batch size B∈{1,2}, sequence length N up to 1024 and
hidden dimension D (called ``hidden_dim``) up to 384.  It tiles the i/j
dimensions with 128×128 blocks and the reduction (k) dimension with tiles
of 32, which maps well to H100 Tensor‑Cores.
"""

import torch
import triton
import triton.language as tl
import math

# ------------------------------------------------------------------------
# Triton kernel: batch‑wise, hidden‑dim‑wise GEMM (left @ rightᵀ)
# ------------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr,                 # half*  [B,N,N,D]
    right_ptr,                # half*  [B,N,N,D]
    out_ptr,                  # half*  [B,N,N,D]
    B, N, D,                  # int32
    # strides for left   (b,i,k,d)
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides for right  (b,j,k,d)
    stride_rb, stride_rj, stride_rk, stride_rd,
    # strides for out    (b,i,j,d)
    stride_ob, stride_oi, stride_oj, stride_od,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------------
    # 3‑D launch grid:
    #   i_block  = program_id(0)  -> tiles over the output rows (i)
    #   j_block  = program_id(1)  -> tiles over the output cols (j)
    #   bd       = program_id(2)  -> combined batch * hidden_dim index
    # --------------------------------------------------------------------
    i_block = tl.program_id(0)
    j_block = tl.program_id(1)
    bd      = tl.program_id(2)

    b = bd // D            # batch index
    d = bd %  D            # hidden‑dim index

    # --------------------------------------------------------------
    # Offsets for the i‑ and j‑tiles; mask out‑of‑bounds later
    # --------------------------------------------------------------
    offs_i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ----------------------------------------------------------------
    # Base pointers that already point to the start of the selected
    # batch and hidden‑dim slice.
    # ----------------------------------------------------------------
    left_base  = left_ptr  + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd

    # --------------------------------------------------------------
    # Accumulator in fp32 (required for numerical stability)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over the reduction dimension k (the “sequence” axis)
    # --------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- Load a (M×K) tile from left -----
        # layout: left[b, i, k, d]
        left_ptrs = left_base + (offs_i[:, None] * stride_li) + (offs_k[None, :] * stride_lk)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # ----- Load a (K×N) tile from rightᵀ -----
        # we need right[j, k, d] as (K, N); swapping the logical axes
        # layout: right[b, j, k, d]
        right_ptrs = right_base + (offs_k[:, None] * stride_rk) + (offs_j[None, :] * stride_rj)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # ----- Accumulate the dot product -----
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # --------------------------------------------------------------
    # Write the result back (store as fp16, the rest of the model works
    # in fp32, so a cast is performed after the kernel call)
    # --------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_ob + d * stride_od \
               + (offs_i[:, None] * stride_oi) + (offs_j[None, :] * stride_oj)
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# Host entry point required by the evaluation harness
# ------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        input   – Tensor[bs, N, N, dim]    (float32)
        mask    – Tensor[bs, N, N] or None (float32 / bool)
        weights – dict of learned parameters (float32)
        config  – dict with keys "dim" and "hidden_dim"

    Returns
    -------
    Tensor[bs, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim        = config["dim"]          # input feature dimension
    hidden_dim = config["hidden_dim"]   # projection / gate dimension (D)
    eps = 1e-5

    # --------------------------------------------------------------------
    # Step 1 – LayerNorm on the input (dim‑wise)
    # --------------------------------------------------------------------
    # Compute mean / var on the last axis (dim)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) / torch.sqrt(var + eps)
    x = x * weights["norm.weight"] + weights["norm.bias"]          # γ & β

    # --------------------------------------------------------------------
    # Step 2 – Linear projections (no bias)
    # --------------------------------------------------------------------
    left_proj  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------------
    # Step 3 – Optional mask (broadcasted on the hidden dimension)
    # --------------------------------------------------------------------
    if mask is None:
        mask_exp = torch.ones_like(x[..., :1])          # shape [B,N,N,1]
    else:
        mask_exp = mask.unsqueeze(-1).type_as(x)        # [B,N,N,1]

    left  = left_proj * mask_exp
    right = right_proj * mask_exp

    # --------------------------------------------------------------------
    # Step 4 – Gating (sigmoid of linear transforms)
    # --------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------
    # Step 5 – Cast to fp16 and launch the Triton kernel
    # --------------------------------------------------------------------
    left_h  = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    B, N, _, D = left_h.shape               # D == hidden_dim
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Strides must be expressed in elements (not bytes)
    stride_lb, stride_li, stride_lk, stride_ld = left_h.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_h.stride()

    # Tile sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * D                           # combined batch‑*‑hidden‑dim dimension
    )

    # Launch kernel (half‑precision compute, fp32 accumulation)
    tri_mul_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, D,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_rj, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Back to fp32 for the remaining operations
    out = out_h.to(torch.float32)

    # --------------------------------------------------------------------
    # Step 6 – Output LayerNorm (hidden‑dim wise)
    # --------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=[hidden_dim],
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------------
    # Step 7 – Gating with out_gate and final linear projection
    # --------------------------------------------------------------------
    out = out * out_gate                                   # element‑wise gating
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
