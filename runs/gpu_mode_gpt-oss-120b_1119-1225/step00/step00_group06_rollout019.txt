@triton.jit
def triton_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,
    stride_b_b, stride_b_i, stride_b_k, stride_b_h,
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_n = tl.program_id(0)  # column block index
    pid_m = tl.program_id(1)  # row block index
    pid_bh = tl.program_id(2) # batch*head index

    batch_id = pid_bh // H
    head_id = pid_bh % H

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # early exit for out-of-bounds? We'll mask loads and store.

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kp * BLOCK_K + tl.arange(0, BLOCK_K)

        # a block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_ptr + batch_id * stride_a_b + offs_m[:, None] * stride_a_i + offs_k[None, :] * stride_a_k + head_id * stride_a_h
        # b block: shape (BLOCK_K, BLOCK_N), where we load B transposed (k,j)
        b_ptrs = b_ptr + batch_id * stride_b_b + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_i + head_id * stride_b_h

        a = tl.load(a_ptrs, mask=(offs_m < N) & (offs_k < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < N) & (offs_n < N), other=0.0)

        # convert to float32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
    
    # store result
    c_ptrs = c_ptr + batch_id * stride_c_b + offs_m[:, None] * stride_c_i + offs_n[None, :] * stride_c_j + head_id * stride_c_h
    # apply mask for store (only store valid region)
    mask_store = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)

"""
TriMul forward (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ [B, N, N, D].
2. Compute half‑precision linear projections:
      left  = x_h @ W_left   (shape B,N,N,H)
      right = x_h @ W_right  (shape B,N,N,H)
   (x_h = x_norm cast to float16)
3. Apply optional mask, then element‑wise sigmoid‑gates
   (left_gate, right_gate) also produced by half‑precision linear layers.
4. Compute the high‑cost contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   for every hidden dimension h using a fused Triton GEMM kernel.
   The kernel runs a batched mat‑mul for each (batch, hidden) pair,
   accumulating in float32 for accuracy.
5. The result (still float32) undergoes a second layer‑norm,
   a sigmoid‑gate (out_gate) computed from the normalized input,
   and a final linear projection back to the original channel size D.
6. Return the tensor of shape [B, N, N, D].

Only the heavy O(N³·H) contraction is off‑loaded to Triton; all other
operations use fast PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_a_b, stride_a_i, stride_a_k, stride_a_h,
    stride_b_b, stride_b_i, stride_b_k, stride_b_h,
    stride_c_b, stride_c_i, stride_c_j, stride_c_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute for each (batch, head) pair:
        C[i, j] = Σ_k A[i, k] * B_T[k, j]
    where A = left[batch, :, :, head] and B_T is the transpose of
    right[batch, :, :, head] across the two sequence axes.
    """
    # 3‑D grid: (col‑block, row‑block, batch*head)
    pid_n = tl.program_id(0)      # column block index   (j)
    pid_m = tl.program_id(1)      # row    block index   (i)
    pid_bh = tl.program_id(2)     # linear index over batch*head

    batch_id = pid_bh // H
    head_id  = pid_bh % H

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # i indices
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # j indices

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K sized chunks
    for kp in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = kp * BLOCK_K + tl.arange(0, BLOCK_K)   # k indices

        # Pointers for A (shape M×K) and B_T (shape K×N)
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_b
            + offs_m[:, None] * stride_a_i
            + offs_k[None, :] * stride_a_k
            + head_id * stride_a_h
        )
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_b
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_i
            + head_id * stride_b_h
        )

        # Load tiles, mask out‑of‑bounds elements
        a = tl.load(a_ptrs, mask=(offs_m < N) & (offs_k < N), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < N) & (offs_n < N), other=0.0)

        # Promote to FP32 for precise accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # GEMM tile
        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_N)

    # Write back the result tile (C is float32)
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_b
        + offs_m[:, None] * stride_c_i
        + offs_n[None, :] * stride_c_j
        + head_id * stride_c_h
    )
    mask_store = (offs_m[:, None] < N) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask_store)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # 1. Layer‑norm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)               # [B, N, N, D]  (float32)

    # ------------------------------------------------------------------
    # 2. Cast to fp16 for the heavy linear work
    # ------------------------------------------------------------------
    x_h = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Linear projections (half‑precision)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left  = F.linear(x_h, left_proj_w)   # [B, N, N, H] (fp16)
    right = F.linear(x_h, right_proj_w)  # [B, N, N, H] (fp16)

    # ------------------------------------------------------------------
    # 4. Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5. Gates (half‑precision sigmoid)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w   = weights["out_gate.weight"].to(torch.float16)

    left_gate  = torch.sigmoid(F.linear(x_h, left_gate_w))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w)) # [B,N,N,H] fp16

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Batched contraction via Triton (result in fp32)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=device)

    # Strides (in elements, not bytes)
    stride_a_b, stride_a_i, stride_a_k, stride_a_h = left.stride()
    stride_b_b, stride_b_i, stride_b_k, stride_b_h = right.stride()
    stride_c_b, stride_c_i, stride_c_j, stride_c_h = out.stride()

    # Tile sizes – tuned for H100 (can be adjusted)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_N - 1) // BLOCK_N,     # blocks along j (columns)
        (N + BLOCK_M - 1) // BLOCK_M,     # blocks along i (rows)
        B * H                             # one program per (batch, hidden)
    )

    _tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_a_b, stride_a_i, stride_a_k, stride_a_h,
        stride_b_b, stride_b_i, stride_b_k, stride_b_h,
        stride_c_b, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7. Second layer‑norm over the hidden dimension (float32)
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_w,
                            bias=to_out_norm_b,
                            eps=1e-5)

    # ------------------------------------------------------------------
    # 8. Output gate (sigmoid) computed from the *normalized* input
    # ------------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # [B,N,N,H] (float32)

    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 9. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]                 # shape (dim, hidden_dim)
    result = F.linear(out_norm, to_out_w)               # [B,N,N,dim] (float32)

    return result
