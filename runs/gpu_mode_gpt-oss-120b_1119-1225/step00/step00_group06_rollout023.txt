@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    
    pid_batch_h = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    batch = pid_batch_h // H
    hid = pid_batch_h % H

    # Compute start offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for i,j boundaries
    mask_i = i_offset < N
    mask_j = j_offset < N

    # compute base pointers
    left_base = left_ptr + batch * stride_lb + hid * stride_lh
    right_base = right_ptr + batch * stride_rb + hid * stride_rh
    out_base = out_ptr + batch * stride_ob + hid * stride_oh

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # offsets for loading
        left_ptrs = left_base + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        right_ptrs = right_base + j_offset[:, None] * stride_rj + k_offset[None, :] * stride_rk

        # masked loads
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

        # Cast to fp32
        left = left.to(tl.float32)
        right = right.to(tl.float32)

        # Compute contribution: sum over k dimension
        # left: (M, K), right: (N, K)
        # Broadcasting to (M, N, K) -> sum over K -> (M, N)
        acc += tl.sum(left[:, None, :] * right[None, :, :], axis=2)

    # Store result
    out_ptrs = out_base + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

import torch
import triton
import triton.language as tl

# TriMul kernel

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # program ids
    pid_batch_h = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # resolve batch and hidden dim
    batch = pid_batch_h // H
    hid = pid_batch_h % H

    # compute offsets for i and j
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary masks
    mask_i = i_offset < N
    mask_j = j_offset < N

    # base pointers for this batch/hid slice
    left_base = left_ptr + batch * stride_lb + hid * stride_lh
    right_base = right_ptr + batch * stride_rb + hid * stride_rh
    out_base = out_ptr + batch * stride_ob + hid * stride_oh

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # load left tile: (M, K)
        left_ptrs = left_base + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float32)

        # load right tile transposed: (K, N)
        right_ptrs = right_base + k_offset[:, None] * stride_rk + j_offset[None, :] * stride_rj
        right_T = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0, dtype=tl.float32)

        # matrix multiply for this K block
        acc += tl.dot(left, right_T)

    # store result
    out_ptrs = out_base + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def trimul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the outgoing TriMul: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    using a fused Triton kernel.
    """
    assert left.is_contiguous() and right.is_contiguous()

    out = torch.empty_like(left)  # same dtype and shape

    # strides
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,

        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out

    tensor multiplication (implemented in Triton), a second layer‑norm,
    gating and a final linear projection.
    """

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm on the input (dim) -----

        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    left = torch.nn.functional.linear(
        x_norm, weights["left_proj.weight"]
    )  # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(
        x_norm, weights["right_proj.weight"]
    )

    # ----- Gating linear layers (dim -> hidden_dim) -----
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    # ----- Optional input mask (broadcast over hidden_dim) -----
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----- Apply per‑channel gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- Tricky cubic contract (B,N,N,hidden) -----
    # Use fused Triton kernel
    out = trimul_forward(left, right)  # shape [B,N,N,hidden_dim]

    # ----- Second layer‑norm over hidden_dim -----
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # ----- Out‑gate multiplication -----
    out = out * out_gate

    # ----- Final linear projection (hidden_dim -> dim) -----
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing variant) – fast forward pass.

Algorithm
---------
1. Layer‑norm over the last dimension (dim).
2. Linear projections   left = x·W_left   ,   right = x·W_right   (dim → hidden_dim).
3. Optional pair‑wise mask multiplication.
4. Gating networks:
        left  ← left  * sigmoid(x·W_left_gate)
        right ← right * sigmoid(x·W_right_gate)
        out_gate = sigmoid(x·W_out_gate)
5. Cubic contraction:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched mat‑mul `left @ rightᵀ` per hidden channel.
   The heavy O(N³·hidden_dim) work is performed by a fused Triton kernel.
6. Layer‑norm over the hidden dimension.
7. Multiply by out_gate.
8. Final linear projection (hidden_dim → dim).

The kernel works on a single hidden channel at a time (batch×hidden_dim is
flattened into the first program‑id axis).  Tile sizes (64×64×64) are chosen
to fit comfortably in H100 shared memory and to expose enough parallelism.
"""

import torch
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel that computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ------------------------------------------------------------------
    #  Program IDs:
    #    pid0 – flattened (batch, hidden) dimension
    #    pid1 – tile over i (rows)
    #    pid2 – tile over j (cols)
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Resolve batch index and hidden‑channel index from pid0
    batch = pid0 // H
    hid   = pid0 % H

    # ------------------------------------------------------------------
    #  Offsets for rows i and columns j
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # [M]
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # [N]

    mask_i = i_offset < N
    mask_j = j_offset < N

    # Base pointers for this (batch, hidden) slice
    left_base  = left_ptr  + batch * stride_lb + hid * stride_lh
    right_base = right_ptr + batch * stride_rb + hid * stride_rh
    out_base   = out_ptr   + batch * stride_ob + hid * stride_oh

    # Accumulator for the M×N tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(0, num_k):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)   # [K]
        mask_k   = k_offset < N

        # ------------------------------------------------------------------
        # Load left tile   shape (M, K)
        # ------------------------------------------------------------------
        left_ptrs = left_base \
            + i_offset[:, None] * stride_li \
            + k_offset[None, :] * stride_lk
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
            dtype=tl.float32
        )   # (M, K)

        # ------------------------------------------------------------------
        # Load right tile transposed   shape (K, N)
        #   right[b, j, k, h] → we need the K‑dimension first so we can
        #   use tl.dot(left, right_T)
        # ------------------------------------------------------------------
        right_ptrs = right_base \
            + k_offset[:, None] * stride_rk \
            + j_offset[None, :] * stride_rj
        right_T = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
            dtype=tl.float32
        )   # (K, N)

        # Matrix‑multiply this K‑block and accumulate
        acc += tl.dot(left, right_T)   # (M, N)

    # ------------------------------------------------------------------
    # Store the tile back to global memory
    # ------------------------------------------------------------------
    out_ptrs = out_base \
        + i_offset[:, None] * stride_oi \
        + j_offset[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def trimul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched cubic contraction (the core of TriMul) using the Triton kernel
    defined above.

    Args:
        left, right: tensors of shape [B, N, N, H] (float32, contiguous)

    Returns:
        out: tensor of shape [B, N, N, H] (float32)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape

    # Allocate output initialized to zero (required for edge tiles)
    out = torch.zeros_like(left)

    # Strides (int64)
    stride_lb, stride_li, stride_lk, stride_lh = left.stride()
    stride_rb, stride_rj, stride_rk, stride_rh = right.stride()
    stride_ob, stride_oi, stride_oj, stride_oh = out.stride()

    # Tile sizes – chosen to fit comfortably in H100 shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (B * H,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_lb,
        stride_li,
        stride_lk,
        stride_lh,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rh,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # 8 warps gives good occupancy on H100
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold‑3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor, shape [B, N, N]   (may be None)
        - weights_dict : {
              "norm.weight":      Tensor[dim],
              "norm.bias":        Tensor[dim],
              "left_proj.weight": Tensor[hidden_dim, dim],
              "right_proj.weight":Tensor[hidden_dim, dim],
              "left_gate.weight": Tensor[hidden_dim, dim],
              "right_gate.weight":Tensor[hidden_dim, dim],
              "out_gate.weight":  Tensor[hidden_dim, dim],
              "to_out_norm.weight": Tensor[hidden_dim],
              "to_out_norm.bias":   Tensor[hidden_dim],
              "to_out.weight":    Tensor[dim, hidden_dim],
          }
        - config_dict  : {"dim":int, "hidden_dim":int, "nomask":bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim].
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1) Input layer‑norm over the last dimension (dim)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (dim → hidden_dim)
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3) Gating networks (sigmoid on a linear projection of x)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate   = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # --------------------------------------------------------------
    # 4) Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 5) Apply per‑channel gates
    # --------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Cubic contraction via Triton (the O(N³·H) work)
    # --------------------------------------------------------------
    out = trimul_forward(left, right)   # shape [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7) Second layer‑norm over hidden_dim
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --------------------------------------------------------------
    # 8) Multiply by out‑gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 9) Final linear projection back to dim
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
