        # dot: a (M,K) ; b (N,K) ; want sum over K: a[i,k] * b[j,k]

        acc += tl.dot(a, b, trans_b=True)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bc

    # write back
    c_ptrs = c_ptr + pid_b * stride_cbatch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward implementation using Triton.

The forward pass consists of:
1️⃣ Layer‑norm over the channel dimension of the 4‑D input.
2️⃣ Linear projections (left / right) and multiplicative gates.
3️⃣ Core O(N³·Cₕ) “triangle” update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is executed by a custom Triton kernel that treats each
   (batch, hidden) pair as an independent matrix‑multiply:
       A = left  (N×N) , B = right (N×N)  →  A @ Bᵀ
   The kernel loops over the reduction dimension K in blocks and
   accumulates in FP32 while loading/storing in FP16.
4️⃣ A second Layer‑norm, an output‑gate and a final linear projection
   back to the original channel size.

All heavy work (the O(N³) reduction) lives in the Triton kernel,
while the surrounding linear / norm / sigmoid operations are done
with PyTorch for simplicity.
"""
import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batch of (B * hidden) matrix‑multiplications A @ Bᵀ
# A, B : [batch_h, N, N]  (FP16)
# C    : [batch_h, N, N]  (FP16) – result of Σₖ A[:,i,k] * B[:,j,k]
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,        # pointers
    M, N, K,                    # matrix sizes (all = seq_len)
    stride_ab, stride_am, stride_ak,   # A strides: batch, row (i), col (k)
    stride_bb, stride_br, stride_bc,   # B strides: batch, row (j), col (k)
    stride_cb, stride_cm, stride_cn,   # C strides: batch, row (i), col (j)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)               # tile index over M (i)
    pid_n = tl.program_id(1)               # tile index over N (j)
    pid_b = tl.program_id(2)               # batch * hidden index

    # ---- Offsets for the current tile ---------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # pointers to the first element of the tile
    a_tile_ptr = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_tile_ptr = b_ptr + pid_b * stride_bb + offs_n[:, None] * stride_br + offs_k[None, :] * stride_bc

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- Loop over K ---------------------------------------------------
    for _ in range(0, K, BLOCK_K):
        # Load a block of A (M×K) and B (N×K). B is loaded as (N,K) and
        # used with trans_b=True so that the multiplication uses Bᵀ (K,N).
        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)            # FP16
        b = tl.load(b_tile_ptr,
                    mask=mask_n[:, None] & mask_k[None, :],
                    other=0.0)            # FP16

        # Accumulate: a (M,K) • bᵀ (K,N) → (M,N)
        acc += tl.dot(a, b, trans_b=True)   # result in FP32

        # Move pointers to the next K‑block
        a_tile_ptr += BLOCK_K * stride_ak
        b_tile_ptr += BLOCK_K * stride_bc

    # ---- Store the result ---------------------------------------------
    c_tile_ptr = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_tile_ptr,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1️⃣ Input layer‑norm
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 2️⃣ Left / Right projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast to channel dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool or int – convert to float and broadcast
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections of the *normalized* input)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5️⃣ Core triangle multiplication – Triton kernel
    # --------------------------------------------------------------
    # Use half precision for the massive O(N³) work
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Rearrange to bring the hidden dimension into the batch axis:
    #   [B, N, N, H] -> [B, H, N, N] -> [B*H, N, N]
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()
    batch_h = bs * hidden_dim

    left_flat = left_perm.view(batch_h, seq_len, seq_len)   # (B*H, N, N)
    right_flat = right_perm.view(batch_h, seq_len, seq_len)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Strides for the flattened tensors (batch, row, col)
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_br, stride_bc = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    # Block sizes – tuned for H100 (shared‑mem friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(seq_len, BLOCK_M),   # rows (i)
        triton.cdiv(seq_len, BLOCK_N),   # cols (j)
        batch_h,                         # batch * hidden
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        seq_len,
        seq_len,
        seq_len,
        stride_ab, stride_am, stride_ak,
        stride_bb, stride_br, stride_bc,
        stride_cb, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Back to the original layout and to FP32 for the remaining layers
    out = out_flat.view(bs, hidden_dim, seq_len, seq_len)   # (B, H, N, N)
    out = out.permute(0, 2, 3, 1).contiguous()             # (B, N, N, H)
    out = out.to(dtype)                                    # FP32

    # --------------------------------------------------------------
    # 6️⃣ Second layer‑norm over the hidden dimension
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # --------------------------------------------------------------
    # 7️⃣ Output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 8️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])
    return out
