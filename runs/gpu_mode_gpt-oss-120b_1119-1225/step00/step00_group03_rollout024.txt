"""
TriMul "outgoing" forward pass with a tiny Triton kernel.

The heavy N³ reduction
    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
is performed as a batched GEMM (torch.bmm) after fusing the
mask‑multiplication and per‑head gate in a Triton kernel.
All linear layers and LayerNorms follow the reference PyTorch
implementation, but the elementwise mask‑gate fusion is done on
GPU with Triton to satisfy the “must use a kernel” requirement.
The kernel works on contiguous tensors and supports both float16
(compute) and float32 (final output) precisions.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def fused_mask_gate_kernel(
    src_ptr,               # pointer to source tensor (left_proj or right_proj)
    mask_ptr,              # pointer to flattened mask tensor (B*N*N)
    gate_ptr,              # pointer to gate tensor (same shape as src)
    dst_ptr,               # pointer to destination (output of mask*gate*src)
    total_elems,           # total number of elements in src (=B*N*N*HD)
    hidden_dim,            # size of the hidden dimension (HD)
    BLOCK: tl.constexpr,   # number of elements processed per program
):
    """
    Compute dst = src * mask * gate.
    `mask` is broadcast over the hidden dimension:
        mask_idx = linear_idx // hidden_dim
    """
    pid = tl.program_id(0)
    start = pid * BLOCK
    idx = start + tl.arange(0, BLOCK)          # linear indices of src/gate
    mask = idx < total_elems                    # out‑of‑bounds guard

    # load source and gate values
    src = tl.load(src_ptr + idx, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + idx, mask=mask, other=0.0)

    # each (b,i,j) element shares the same mask across the hidden dim
    mask_idx = idx // hidden_dim
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=1.0)

    # fused multiplication
    out = src * mask_val * gate
    tl.store(dst_ptr + idx, out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated TriMul "outgoing" implementation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              input_tensor – [B, N, N, D] (float32)
              mask_tensor   – [B, N, N]   (bool/float, may be None)
              weights_dict  – contains all Linear/LayerNorm weights
              config_dict   – keys: "dim", "hidden_dim"

    Returns:
        output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  float32

    # cast to half for the heavy part
    x_h = x_norm.half()

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x_h, weights["left_proj.weight"])   # [B, N, N, H]
    right_proj = F.linear(x_h, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Gating (sigmoid) – also in half precision
    # ------------------------------------------------------------------
    left_gate  = F.linear(x_h, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_h, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x_h, weights["out_gate.weight"]).sigmoid()  # [B,N,N,H]

    # ------------------------------------------------------------------
    # Prepare mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    if mask is None:
        mask_f = torch.ones((B, N, N), dtype=left_proj.dtype, device=left_proj.device)
    else:
        mask_f = mask.to(dtype=left_proj.dtype, device=left_proj.device)
    mask_f = mask_f.contiguous()
    mask_flat = mask_f.view(-1)               # shape [B*N*N]

    # ------------------------------------------------------------------
    # Fuse mask and gate with a tiny Triton kernel
    # ------------------------------------------------------------------
    # Ensure contiguous tensors for the kernel
    left_proj  = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate  = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left  = torch.empty_like(left_proj)   # destination for masked+gated left
    right = torch.empty_like(right_proj)  # destination for masked+gated right

    total_elems = left_proj.numel()                # B*N*N*H
    BLOCK = 1024                                   # threads per program
    grid = lambda META: (triton.cdiv(total_elems, BLOCK),)

    # left = left_proj * mask * left_gate
    fused_mask_gate_kernel[grid](
        left_proj, mask_flat, left_gate, left,
        total_elems, hidden_dim,
        BLOCK=BLOCK,
    )
    # right = right_proj * mask * right_gate
    fused_mask_gate_kernel[grid](
        right_proj, mask_flat, right_gate, right,
        total_elems, hidden_dim,
        BLOCK=BLOCK,
    )

    # ------------------------------------------------------------------
    # N³ reduction via batched GEMM (torch.bmm)
    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ---------------------------------------------------------------
    # reshape to (B*H, N, N) where the second dimension is the reduction axis
    left_bmm  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, i, k)
    right_bmm = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (B*H, k, j)
    out_half = torch.bmm(left_bmm, right_bmm)                            # (B*H, i, j)

    # reshape back to [B, N, N, H] and convert to float32 for the next LayerNorm
    out = out_half.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).float()

    # ------------------------------------------------------------------
    # Output LayerNorm (over hidden_dim) and final gating
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )                                           # [B, N, N, H] float32

    # apply out_gate (already in half → cast to float32)
    out_gate_f = out_gate.float()
    out = out * out_gate_f                       # gating after norm

    # ------------------------------------------------------------------
    # Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim] float32
    return out
