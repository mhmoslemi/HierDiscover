def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
    # Compute program id for batch-dim: each program processes a (b, h) tile.
    # We'll treat each combination as independent.
    pid = tl.program_id(axis=0)  # linear pid across B*H
    batch = pid // H
    hidden_idx = pid % H

    # Offsets for output tile (i,j)
    # Offsets for i-block row and j-block column
    offs_i = tl.arange(0, BLOCK_M) + tl.program_id(axis=1) * BLOCK_M
    offs_j = tl.arange(0, BLOCK_N) + tl.program_id(axis=2) * BLOCK_N

    # set mask for out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = tl.arange(0, BLOCK_K) + k_start
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        # left offset: left_ptr + batch*stride_left_b + hidden_idx*stride_left_h + i*stride_left_i + k*stride_left_k
        left_ptrs = left_ptr + batch * stride_left_b + hidden_idx * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

        # We will load right and transposed later? We need right[j,k] for each j and k. Actually for multiplication we need right transpose: we want right_transpose[i,k] = right[j,k] for j index? def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    device = x.device
    # LayerNorm x
    norm_weight = weights["norm.weight"]   # shape [dim]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )
    # Project left, right
    left_proj_weight = weights["left_proj.weight"]  # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    # Use torch.nn.functional.linear
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # (B,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Apply mask if needed
    if not nomask:
        # mask shape [B,N,N]; unsqueeze to [-1]
        mask = mask.to(left.dtype).unsqueeze(-1)   # broadcast to hidden dim
        left = left * mask
        right = right * mask
    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))
    left = left * left_gate
    right = right * right_gate
    # cast to fp16 for triton multiplication
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B,N,N,_ = left_fp16.shape # B,N,N,hidden_dim
    # Allocate output tensor
    out_fp16 = torch.empty_like(left_fp16)  # shape same (B,N,N,hidden_dim) but dtype fp16
    # Compute strides
    stride_left = left_fp16.stride()
    stride_right = right_fp16.stride()
    stride_out = out_fp16.stride()
    # left strides: (s0, s1, s2, s3)
    # Provide to kernel:
    # B, H = B, hidden_dim
    # call
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32 # maybe 32
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)
    trmul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, hidden_dim, N,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # Wait for kernel exec
    torch.cuda.synchronize()
    # Cast out back to float32
    out = out_fp16.to(torch.float32)
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )
    # Multiply with out_gate
    out_norm = out_norm * out_gate
    # to_out linear
    to_out_weight = weights["to_out.weight"]   # shape [dim, hidden_dim]
    # final output (B,N,N,dim)
    final = torch.nn.functional.linear(out_norm, to_out_weight)
    return final

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

The module computes, for an input tensor X ∈ ℝ^{B×N×N×C} :

1. LayerNorm over the channel dimension C.
2. Two linear projections (left/right) to a hidden dimension H.
3. Optional pairwise mask multiplication.
4. Gated versions of left/right via sigmoids of separate linear layers.
5. Core “triangle” multiplication:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
        out[:,d] = left[:,d] @ right[:,d].T
   for each hidden channel d.  This O(N³·H) step is performed
   by a custom Triton kernel that processes tiles of (i, j) and
   accumulates over k using FP16 inputs with FP32 accumulation.
6. LayerNorm on the hidden result, gating with a third sigmoid,
   and a final linear projection back to the original channel size C.

The kernel works for any batch size (B∈{1,2}), sequence length N,
channel dimension C (e.g. 128) and hidden dimension H (e.g. 128).
Masking is optional (controlled by config["nomask"]).

The implementation mixes PyTorch for the inexpensive linear /
normalisation steps and a Triton kernel for the heavy
tri‑multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out = left @ right^T  for each (batch, hidden) pair
# left, right:  [B, N, N, H]   (dtype = fp16)
# out: same shape, dtype = fp16 (later cast to fp32)
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr,          # *fp16   shape (B,N,N,H)
    right_ptr,         # *fp16   shape (B,N,N,H)
    out_ptr,           # *fp16   shape (B,N,N,H)

    B, H, N,                           # runtime constants
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    BLOCK_M: tl.constexpr,  # tile size over i
    BLOCK_N: tl.constexpr,  # tile size over j
    BLOCK_K: tl.constexpr   # tile size over k
):
    """
    Compute for each (b, h):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The kernel is launched on a 3‑D grid:
        grid[0] = B*H       (batch * hidden)
        grid[1] = ceil(N/BLOCK_M)   (i‑tiles)
        grid[2] = ceil(N/BLOCK_N)   (j‑tiles)
    """
    # ------------------------------------------------------------------
    # Identify which (batch, hidden) slice we are processing
    # ------------------------------------------------------------------
    pid = tl.program_id(axis=0)
    batch_id = pid // H
    hidden_id = pid % H

    # ------------------------------------------------------------------
    # Tile offsets for the output (i,j) block
    # ------------------------------------------------------------------
    pid_i = tl.program_id(axis=1)
    pid_j = tl.program_id(axis=2)

    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k dimension in BLOCK_K chunks
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---------- load left tile (M x K) ----------
        # ptr = base + batch*stride_b + hidden*stride_h + i*stride_i + k*stride_k
        left_ptrs = left_ptr + batch_id * stride_left_b \
                              + hidden_id * stride_left_h \
                              + offs_i[:, None] * stride_left_i \
                              + offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)          # fp16

        # ---------- load right tile (N x K) ----------
        # right dimensions are (j, k) – we load (j,k) and will transpose in dot
        right_ptrs = right_ptr + batch_id * stride_right_b \
                                + hidden_id * stride_right_h \
                                + offs_j[:, None] * stride_right_j \
                                + offs_k[None, :] * stride_right_k
        right_tile = tl.load(right_ptrs,
                             mask=mask_j[:, None] & mask_k[None, :],
                             other=0.0)         # fp16

        # ---------- dot product left (M,K) * right.T (K,N) ----------
        # tl.dot with trans_b=True treats right_tile as (N,K) and transposes it.
        acc += tl.dot(left_tile, right_tile, trans_b=True)   # FP32 accumulation

    # ------------------------------------------------------------------
    # Store the block result
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + batch_id * stride_out_b \
                        + hidden_id * stride_out_h \
                        + offs_i[:, None] * stride_out_i \
                        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc.to(tl.float16),                     # store as fp16
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Python entry point ----------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict of named parameters (torch.Tensors)
            - config: dict containing at least:
                * "dim":          C
                * "hidden_dim":   H
                * "nomask":       bool (True → ignore mask)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"]
    nomask = config.get("nomask", False)

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over channel dimension
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(C,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,C]  FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask application (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask is [B,N,N]; ensure float for multiplication
        mask_f = mask.to(x.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul – batched matrix multiplication via Triton
    #    left/right: [B,N,N,H] (FP16 for speed)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    out_fp16 = torch.empty_like(left_fp16)   # placeholder for result

    # Strides (in elements, not bytes) for each tensor
    stride_left = left_fp16.stride()   # (B, i, k, h)
    stride_right = right_fp16.stride()
    stride_out = out_fp16.stride()

    # Tile sizes – tuned for H100 FP16 tensor cores
    BLOCK_M = 64   # i‑tile
    BLOCK_N = 64   # j‑tile
    BLOCK_K = 32   # reduction tile over k

    grid = (
        B * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    trmul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        hidden_dim,
        N,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_out[0],
        stride_out[1],
        stride_out[2],
        stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel has completed before using the result
    torch.cuda.synchronize()

    # Cast back to FP32 for the remaining post‑processing
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Post‑TriMul LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H]

    out_norm = out_norm * out_gate                      # gated hidden representation

    # Final projection back to original channel dimension C
    final = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,C]

    return final

**Explanation of the key points**

* **LayerNorm & Linear layers** – performed in regular PyTorch because they are cheap relative to the O(N³) step and allow us to reuse the provided weight tensors directly.
* **Mask handling** – optional; if `config["nomask"]` is `True` the mask is ignored.
* **Gating** – implemented with simple sigmoid‑activated linear layers, exactly matching the reference implementation.
* **Triton kernel** – computes the triangular multiplicative update (`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`).  
  * The kernel processes a tile of the output `(i,j)` while accumulating over `k` in blocks.
  * Inputs are FP16 to exploit H100 tensor‑core throughput; accumulation is performed in FP32 for numerical stability.
  * By treating each `(batch, hidden)` pair as an independent program (`grid[0] = B*H`), we reuse the highly‑optimised matrix‑multiply pattern already present in Triton.
* **Post‑processing** – a second LayerNorm, the output gate, and the final linear projection bring the tensor back to shape `[B, N, N, C]` in FP32, matching the PyTorch reference.

The function `custom_kernel` can be called directly with the tuple `(input, mask, weights, config)` and will return the correct TriMul forward result while leveraging Triton for the dominant computation.
