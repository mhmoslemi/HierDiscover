def tri_mul_kernel_launcher(left, right, out):
    # left shape (B, N, N, hidden_dim). We'll permute to (B, hidden_dim, N, N), reshape batch.
    B, N, _, hidden_dim = left.shape   # left: B,N,N,hidden_dim
    # pre-permute
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # flatten batch and hidden_dim: batch_total = B * hidden_dim
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)
    # transpose right for matmul (k, n) needed
    right_t = right_flat.transpose(-1, -2).contiguous()  # shape (B*HD, N, N) but transposed.

    # allocate out tensor
    out_flat = torch.empty_like(left_flat) # shape batch_total,N,N.

    # compute striding
    left_ptr = left_flat
    right_ptr = right_t
    out_ptr = out_flat

```python
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # Allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile (M x K)
        a_ptrs = left_ptr + pid_batch * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B tile (K x N) (right transposed)
        b_ptrs = right_ptr + pid_batch * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # Compute dot product and accumulate
        acc += tl.dot(a, b)   # returns float32

    # Store result
    out_ptrs = out_ptr + pid_batch * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul “outgoing” implementation.

The kernel fuses the expensive N³ reduction:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
by viewing each (batch, channel) pair as an independent matrix‐multiply
(left: (N,K) , right⁺: (K,N)) and executing a batched GEMM in Triton.
All other operations (LayerNorm, Linear projections, gates, mask handling
and final LayerNorm + projection) are performed with PyTorch.
The Triton kernel works for arbitrary sequence length N and hidden dimension
by flattening the batch and channel dimensions (batch_total = B * hidden_dim).
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication for the N³ reduction.
# Input tensors are of shape (B*HD, N, N) where the last two dimensions are
#   left:  (i, k)   – row i, column k
#   right: (k, j)   – row k, column j  (obtained from the original right tensor
#                                      after swapping the two sequence axes)
# Output tensor has shape (B*HD, N, N) and stores the sum over k.
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program identifiers
    pid_batch = tl.program_id(0)          # 0 … total_batches-1
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block column

    # offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # (BLOCK_N,)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator (use float32 for numeric safety)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- load A = left[batch, i, k] -------------
        a_ptrs = left_ptr + \
                 pid_batch * stride_l_batch + \
                 offs_m[:, None] * stride_l_m + \
                 offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)   # promote to fp32 for the dot

        # -------- load B = right[batch, k, j] -------------
        b_ptrs = right_ptr + \
                 pid_batch * stride_r_batch + \
                 offs_k[:, None] * stride_r_k + \
                 offs_n[None, :] * stride_r_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)
        b = b.to(tl.float32)

        # -------- dot product and accumulation -----------
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # -------- store the result --------------------------
    out_ptrs = out_ptr + \
               pid_batch * stride_o_batch + \
               offs_m[:, None] * stride_o_m + \
               offs_n[None, :] * stride_o_n
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------------------- #
# Entry point used by the grader.
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” Triangle Multiplicative Update.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # LayerNorm on the input (LayerNorm over trailing dim = dim)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape: [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # Linear projections (no bias) and gate calculations
    # ------------------------------------------------------------------- #
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    # Project to hidden dimension
    left = F.linear(x, left_proj_w)   # [B, N, N, HD]
    right = F.linear(x, right_proj_w)   # [B, N, N, HD]

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B, N, N, HD]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    # ------------------------------------------------------------------- #
    # Apply mask (if present) – broadcast over hidden dimension
    # ------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # mask: [B, N, N] → [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------- #
    # Apply elementwise gating
    # ------------------------------------------------------------------- #
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # Triton batched GEMM for the N³ reduction
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape  # N is sequence length
    # Permute to (B, HD, N, N) where the two seq dims are (i,k) for left
    # and (k,j) for right (by swapping the last two dims of right)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, HD, N, N) → (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # (B, HD, N, N) → (k,j)

    # Collapse batch and hidden dimensions for the kernel
    total_batches = B * hidden_dim
    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat, dtype=torch.float32, device=device)

    # Strides
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # Choose block sizes (tuned for H100, FP32 accumulation)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch Triton kernel
    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, HD]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, HD]

    # ------------------------------------------------------------------- #
    # Final LayerNorm over hidden_dim, gating, and projection back to dim
    # ------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_w)   # shape: [B, N, N, dim]

    return out
