def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    # shapes
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # CuTensor: input_norm = layernorm(input)
    x_norm = layernorm_triton(input_tensor, weights["norm.weight"], weights["norm.bias"], eps)
    # Linear projections
    # left and right:
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])  # shape: (bs,N,N,hidden_dim)
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if given:
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).type_as(left) # convert to same dtype
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # compute out via batched matrix multiplication
    # reshape to (bs*hidden_dim, N, N)
    left_flat = left.permute(0,3,1,2).reshape(bs*hidden_dim, N, N)
    right_flat = right.permute(0,3,1,2).reshape(bs*hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))

    out_norm = out_norm * out_gate

    # final linear
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

import torch
import triton
import triton.language as tl

@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    M, D,
    stride_x_m, stride_x_d,
    stride_out_m, stride_out_d,
    eps,
    BLOCK_D: tl.constexpr,

```python
def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    Performs LayerNorm on input, linear projections, gating, and the
    triangular multiplicative update (matrix multiplication over the
    sequence dimension). The layernorms are implemented in Triton for
    row‑wise reduction, while the heavy batched matrix multiplication
    uses torch.bmm (cuBLAS). The function returns a tensor of shape
    [bs, seq_len, seq_len, dim].
    """
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    x_norm = layernorm_triton(input_tensor, weights["norm.weight"], weights["norm.bias"], eps)

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if provided (mask shape [bs, N, N])
    if mask_tensor is not None:
        mask_f = mask_tensor.to(x_norm.dtype).unsqueeze(-1)  # [bs, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Triangular multiplication: out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    # Cast to shape (bs*hidden_dim, N, N) and use batched GEMM.
    B, N, _, _ = left.shape
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_flat shape (B*hidden_dim, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # Output layer norm (over hidden_dim)
    out_norm = layernorm_triton(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps)

    # Apply output gate
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
```

"""
TriMul (outgoing) forward pass with Triton‑accelerated LayerNorm.
The heavy O(N³) update is performed by a batched GEMM (cuBLAS) while
both LayerNorms (input and output) are implemented in a custom Triton
kernel that reduces each row of the last dimension. No gradients are
computed.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel for row‑wise LayerNorm (no bias on mean/var)
# --------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,                # *float32   input   [M, D]
    out_ptr,              # *float32   output  [M, D]
    weight_ptr,           # *float32   gamma   [D]
    bias_ptr,             # *float32   beta    [D]
    M,                    # i32        number of rows
    D,                    # i32        hidden dimension
    stride_x_m, stride_x_d,
    stride_out_m, stride_out_d,
    eps,                  # f32        epsilon
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm.
    Each program processes one row (i.e. one (b,i,j) slice) of length D.
    """
    row = tl.program_id(0)
    if row >= M:
        return

    # ----------------------------------------------------------
    # load one full row (BLOCK_D >= max D)
    # ----------------------------------------------------------
    offs_d = tl.arange(0, BLOCK_D)
    mask_d = offs_d < D                     # mask out-of‑range cols

    x_row_ptr = x_ptr + row * stride_x_m
    x = tl.load(x_row_ptr + offs_d * stride_x_d,
                mask=mask_d,
                other=0.0,
                dtype=tl.float32)          # [BLOCK_D]

    # ----------------------------------------------------------
    # mean
    # ----------------------------------------------------------
    sum_x = tl.sum(x, axis=0)               # scalar
    mean = sum_x / D

    # ----------------------------------------------------------
    # variance
    # ----------------------------------------------------------
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / D
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ----------------------------------------------------------
    # normalized + affine
    # ----------------------------------------------------------
    y = diff * inv_std

    w = tl.load(weight_ptr + offs_d,
                mask=mask_d,
                other=1.0,
                dtype=tl.float32)          # gamma
    b = tl.load(bias_ptr + offs_d,
                mask=mask_d,
                other=0.0,
                dtype=tl.float32)          # beta

    y = y * w + b

    # ----------------------------------------------------------
    # store
    # ----------------------------------------------------------
    out_row_ptr = out_ptr + row * stride_out_m
    tl.store(out_row_ptr + offs_d * stride_out_d,
             y,
             mask=mask_d)


def _layernorm_triton(x: torch.Tensor,
                      weight: torch.Tensor,
                      bias: torch.Tensor,
                      eps: float = 1e-5) -> torch.Tensor:
    """
    Triton‑based LayerNorm over the last dimension.
    x – tensor of shape (..., D)   (any dtype, will be cast to fp32)
    weight, bias – 1‑D tensors of shape (D,)
    Returns a tensor of the same shape as x (dtype matches x).
    """
    assert weight.ndim == bias.ndim == 1
    D = weight.shape[0]

    # Flatten leading dimensions → (M, D)
    x_flat = x.contiguous().view(-1, D)
    out_flat = torch.empty_like(x_flat)

    # strides in elements (not bytes) – Triton expects strides in element units
    stride_x_m = x_flat.stride(0)
    stride_x_d = x_flat.stride(1)
    stride_out_m = out_flat.stride(0)
    stride_out_d = out_flat.stride(1)

    M = x_flat.shape[0]

    # launch one program per row
    grid = (M,)

    _layernorm_kernel[grid](
        x_flat,
        out_flat,
        weight,
        bias,
        M,
        D,
        stride_x_m,
        stride_x_d,
        stride_out_m,
        stride_out_d,
        eps,
        BLOCK_D=384,               # max D across all test cases
    )
    return out_flat.view_as(x)


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Performs the outgoing TriMul forward pass.
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (Triton)  →  x
    # ------------------------------------------------------------------
    x = _layernorm_triton(
        input_tensor,
        weights["norm.weight"].contiguous(),
        weights["norm.bias"].contiguous(),
        eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask  (mask shape [bs, N, N])
    # ------------------------------------------------------------------
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections on the *original* x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched GEMM (left @ rightᵀ) for each hidden channel.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape   # left shape = [B, N, N, hidden_dim]
    # reshape to (B*hidden_dim, N, N) for cuBLAS batched gemm
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out_flat shape: (B*hidden_dim, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (Triton) over hidden_dim
    # ------------------------------------------------------------------
    out_norm = _layernorm_triton(
        out,
        weights["to_out_norm.weight"].contiguous(),
        weights["to_out_norm.bias"].contiguous(),
        eps,
    )

    # ------------------------------------------------------------------
    # 7) Apply output gate
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to original dim
    # ------------------------------------------------------------------
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
