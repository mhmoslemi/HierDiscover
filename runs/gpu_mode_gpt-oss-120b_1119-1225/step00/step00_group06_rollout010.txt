def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,  # all ints
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids:
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)
    pid_z = tl.program_id(axis=2)  # combined batch-ch channel index

    # compute b and d from pid_z
    total_bd = B * D
    b = pid_z // D
    d = pid_z % D

    # tile start indices
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # load loop accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # compute pointer base offsets for left and right for given b,d:
    # left offset = left_ptr + b*stride_left_b + d*stride_left_d
    # Since stride_left_d = 1, we can add d directly.
    left_base = left_ptr + b * stride_left_b + d
    right_base = right_ptr + b * stride_right_b + d

    # compute offsets for i and k loops
    # left tile shape: M rows (i), K columns (k)
    # right tile shape: N rows (j), K columns (k) but we will access right transposed: need right[j, k].
    # We'll loop over k in chunks of BLOCK_K.

    for k_offset in range(0, N, BLOCK_K):
        # compute k range within block
        cur_k = tl.arange(0, BLOCK_K) + k_offset
        # mask for K boundary
        k_mask = cur_k < N

        # compute i offsets
        i_offsets = i0 + tl.arange(0, BLOCK_M)[:, None]  # shape (BLOCK_M,1)
        j_offsets = j0 + tl.arange(0, BLOCK_N)[None, :]  # shape (1,BLOCK_N)

        # compute offsets into left: base + i*stride_i + k*stride_k
        left_ptrs = left_base + i_offsets * stride_left_i + cur_k[None, :] * stride_left_k
        # similarly right: base + j*stride_j + k*stride_k
        right_ptrs = right_base + j_offsets * stride_right_j + cur_k[:, None] * stride_right_k

        # load
        left_vals = tl.load(left_ptrs, mask=(i_offsets[:, None] < N) & k_mask[None, :], other=0.0).to(tl.float32)
        right_vals = tl.load(right_ptrs, mask=(j_offsets[None, :] < N) & k_mask[:, None], other=0.0).to(tl.float32)

        # multiply-accumulate
        acc += tl.dot(left_vals, right_vals)  # shape (BLOCK_M, BLOCK_N)

    # store result
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + i0 * stride_out_i + j0 * stride_out_j
    # out shape: (BLOCK_M, BLOCK_N)
    tl.store(out_ptrs, acc.to(tl.float16) if output dtype is fp16 else acc, mask=(i0+tl.arange(0,BLOCK_M)[:, None] < N) & (j0+tl.arange(0,BLOCK_N)[None, :] < N))

import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

):

    j_range = j0 + tl.arange(0, BLOCK_N)

"""
TriMul (outgoing) fused kernel.

This implementation follows the reference ``TriMul`` forward pass but
off‑loads the cubic O(N³·C) tensor contraction

    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

to a custom Triton kernel.  For each batch‐head pair (b,c) the kernel
computes a matrix multiplication L @ Rᵀ where L and R have shape (N,N).
All other cheap operations (LayerNorm, Linear, sigmoid gates) are kept
in PyTorch for simplicity.  The kernel works with fp16 inputs (lower
memory traffic) and accumulates in fp32 for numerical stability.

Because the kernel is tiled it can handle the full range of sizes used
in the tests (seq_len up to 1024, hidden_dim≤128, batch≤2) while still
being a genuine Triton implementation.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # *fp16  [B,N,N,C]
    right_ptr,                    # *fp16  [B,N,N,C]
    out_ptr,                      # *fp32  [B,N,N,C]
    B, N, C,                      # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_i, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr,        # tile size in output i dimension
    BLOCK_N: tl.constexpr,        # tile size in output j dimension
    BLOCK_K: tl.constexpr,        # tile size in reduction k dimension
):
    pid_i = tl.program_id(0)      # tile index for i
    pid_j = tl.program_id(1)      # tile index for j
    pid_z = tl.program_id(2)      # combined batch*channel index

    # --------------------------------------------------------------
    # Decode batch (b) and channel (c) from pid_z
    # --------------------------------------------------------------
    b = pid_z // C                # batch index
    c = pid_z % C                 # channel index

    # --------------------------------------------------------------
    # Starting offsets of this tile
    # --------------------------------------------------------------
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # Index vectors for the current tile (will be broadcasted)
    # ------------------------------------------------------------------
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)

    mask_i = i_idx < N
    mask_j = j_idx < N

    # ------------------------------------------------------------------
    # Base pointers for the (b,c) slice of left / right
    # ------------------------------------------------------------------
    left_base  = left_ptr  + b * stride_left_b  + c * stride_left_c
    right_base = right_ptr + b * stride_right_b + c * stride_right_c

    # ------------------------------------------------------------------
    # Accumulator (fp32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction dimension k in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # left[i,k]  -> shape (BLOCK_M, BLOCK_K)
        left_offsets = (left_base
                        + i_idx[:, None] * stride_left_i
                        + k_idx[None, :] * stride_left_k)
        left_vals = tl.load(left_offsets,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # right[j,k] needs to be transposed to obtain rightᵀ (k,j)
        #   rightᵀ[k,j] = right[j,k]
        #   -> offset = k*stride_right_i + j*stride_right_k
        right_offsets = (right_base
                         + k_idx[:, None] * stride_right_i
                         + j_idx[None, :] * stride_right_k)
        right_vals = tl.load(right_offsets,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # accumulate the dot product of the two tiles
        acc += tl.dot(left_vals, right_vals)   # (BLOCK_M, BLOCK_N)

    # ------------------------------------------------------------------
    # Write the result back to global memory
    # ------------------------------------------------------------------
    out_base = out_ptr + b * stride_out_b + c * stride_out_c
    out_offsets = (out_base
                   + i_idx[:, None] * stride_out_i
                   + j_idx[None, :] * stride_out_j)
    tl.store(out_offsets, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input, mask, weights, config) where
        - input  : torch.Tensor [B, N, N, dim]
        - mask   : torch.Tensor [B, N, N]  (may be all‑ones)
        - weights: dict of torch.Tensor with the module's parameters
        - config : dict containing at least
                    {"dim":int, "hidden_dim":int, "nomask":bool}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load weight tensors
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the ``dim`` dimension
    # ------------------------------------------------------------------
    # Shape: [B,N,N,dim]
    x_norm = F.layer_norm(x, (dim,), weight=norm_w, bias=norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → shape [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    left = F.linear(x_norm, left_proj_w)      # (B,N,N,hidden_dim)
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Apply optional mask (broadcasted)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid on separate linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to fp16 for the heavy cubic contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, C = left.shape   # C == hidden_dim

    # ------------------------------------------------------------------
    # 6. Allocate output buffer for the contraction (fp32)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, C), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7. Compute strides (in elements) – torch tensors are contiguous
    # ------------------------------------------------------------------
    stride_left_b, stride_left_i, stride_left_k, stride_left_c = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_c = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_c = out.stride()

    # ------------------------------------------------------------------
    # 8. Launch the Triton kernel
    # ------------------------------------------------------------------
    # Tile sizes can be tuned; the choices below are a good compromise
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (
        triton.cdiv(N, BLOCK_M),          # i‑tiles
        triton.cdiv(N, BLOCK_N),          # j‑tiles
        B * C                             # one program per (batch, channel)
    )
    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        C=C,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_c=stride_left_c,
        stride_right_b=stride_right_b,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_c=stride_right_c,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_c=stride_out_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9. LayerNorm on the hidden dimension of the contracted tensor
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b, eps=eps)

    # ------------------------------------------------------------------
    # 10. Apply the output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate   # element‑wise gate (broadcasted)
    out = F.linear(out, to_out_w)   # shape [B,N,N,dim]

    return out
