@triton.jit

def trimul_kernel(

    left_ptr, right_ptr, out_ptr,

    B, N, H,

    stride_l_b, stride_l_i, stride_l_k, stride_l_d,

    stride_r_b, stride_r_j, stride_r_k, stride_r_d,

    stride_o_b, stride_o_i, stride_o_j, stride_o_d,

    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

):

    pid_bhd = tl.program_id(2) # batch*hidden

    pid_m = tl.program_id(0) # block row

    pid_n = tl.program_id(1) # block col

    # compute batch and hidden index

    b = pid_bhd // H

    d = pid_bhd % H

    # compute start offsets

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masked to within bounds

    mask_m = offs_m < N

    mask_n = offs_n < N

    # initialize accumulator

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k

    for k in range(0, tl.cdiv(N, BLOCK_K)):

        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K) # shape (BLOCK_K)

        # load left tile shape (BLOCK_M, BLOCK_K)

        left_ptrs = left_ptr + b * stride_l_b + (offs_m[:, None] * stride_l_i) + (offs_k[None, :] * stride_l_k) + d * stride_l_d

        left = tl.load(left_ptrs, mask = (offs_m[:, None] < N) & (offs_k[None, :] < N), other=0.0).to(tl.float32)

        # load right tile shape (BLOCK_N, BLOCK_K); note we need transpose (R_jk) -> (k,j) to multiply
        # Right shape is (b, j, k, d). To compute dot with left (i,k) * right (j,k), we need right tile transposed to (k, j).
        # We can load right as (offs_n[:, None] * stride_r_j) + (offs_k[None, :] * stride_r_k)
        # But our offsets want shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b * stride_r_b + (offs_n[:, None] * stride_r_j) + (offs_k[None, :] * stride_r_k) + d * stride_r_d

        # Actually this gives shape (BLOCK_N, BLOCK_K). We need transposed version. We can load into shape (BLOCK_K, BLOCK_N) by swapping axes: alt using indexing.
        # Simpler: load as (BLOCK_N, BLOCK_K) then transpose with tl.transpose? There's no transpose builtin, but we can compute dot via per-k sum across K dimension using elementwise multiplication:
        # For matmul, we usually compute left (M,K) * right (K,N). So we need right tile with shape (BLOCK_K, BLOCK_N). So we need to index right as (k, j).
        # So right_ptrs = right_ptr + b*stride_r_b + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_j) + d * stride_r_d.
        # Let's compute that.

        right_ptrs = right_ptr + b * stride_r_b + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_j) + d * stride_r_d

        right = tl.load(right_ptrs, mask = (offs_k[:, None] < N) & (offs_n[None, :] < N), other=0.0).to(tl.float32)

        # Multiply-accumulate

        acc += tl.dot(left, right)  # shape (BLOCK_M, BLOCK_N)

    # Store result

    out_ptrs = out_ptr + b * stride_o_b + (offs_m[:, None] * stride_o_i) + (offs_n[None, :] * stride_o_j) + d * stride_o_d

    tl.store(out_ptrs, acc, mask = mask_m[:, None] & mask_n[None, :])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_bhd = tl.program_id(2)  # batch * hidden idx
    pid_m = tl.program_id(0)    # block row (i)
    pid_n = tl.program_id(1)    # block col (j)

    # compute batch & hidden dim indices
    b = pid_bhd // H
    d = pid_bhd % H

    # compute offsets for tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M,K)
        left_ptrs = left_ptr + b * stride_lb + d * stride_ld + offs_m[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # right tile (K,N) we need right[j,k,d] but transpose to (k,j) as described
        # right shape (b, j, k, d) with strides (rb, rj, rk, rd)
        right_ptrs = right_ptr + b * stride_rb + d * stride_rd + offs_k[:, None] * stride_rk + offs_n[None, :] * stride_rj
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # dot product accumulate
        acc += tl.dot(left, right)  # (M,N)
    # write output
    out_ptrs = out_ptr + b * stride_ob + d * stride_od + offs_m[:, None] * stride_oi + offs_n[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_epsilon", 1e-5)
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    B, N, _, _ = input_tensor.shape

    # LayerNorm on last dim
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left = F.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # Apply mask if provided
    if mask is not None:
        # mask shape [B,N,N]; unsqueeze last dim for broadcast
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Core triton kernel
    # Allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # Compute strides in bytes for Triton
    def get_stride(t):
        return tuple(s * t.element_size() for s in t.stride())

    l_strides = get_stride(left) # returns tuple (b,i,k,d) stride in bytes
    r_strides = get_stride(right)
    out_strides = get_stride(out)

    # Dispatch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B * hidden_dim )
    trimul_einsum_kernel[grid](
        left_ptr = left,
        right_ptr = right,
        out_ptr = out,
        B = B,
        N = N,
        H = hidden_dim,
        stride_lb = l_strides[0],
        stride_li = l_strides[1],
        stride_lk = l_strides[2],
        stride_ld = l_strides[3],
        stride_rb = r_strides[0],
        stride_rj = r_strides[1],
        stride_rk = r_strides[2],
        stride_rd = r_strides[3],
        stride_ob = out_strides[0],
        stride_oi = out_strides[1],
        stride_oj = out_strides[2],
        stride_od = out_strides[3],
        BLOCK_M = BLOCK_M,
        BLOCK_N = BLOCK_N,
        BLOCK_K = BLOCK_K,
        num_warps = 4,
        # maybe specify num_stages = 1
    )

    # After kernel: apply output layernorm, gate, final linear
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    out = F.linear(out, to_out_weight)  # shape [B,N,N,dim]

    return out

"""
TriMul (outgoing) implementation with Triton fusion.

The heavy part of TriMul is the Einstein‐summation

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batch of `hidden_dim` matrix multiplications L_d @ R_dᵀ
with shapes (N,N).  This kernel fuses the inner‑product over the
sequence dimension `k` for every (batch, hidden) pair, using a standard
blocked GEMM pattern (block‑M × block‑N × block‑K).  All surrounding
operations (LayerNorm, Linear projections, sigmoid gates) are performed
with PyTorch for simplicity.

The kernel operates on tensors of shape [B, N, N, H] where
    B ∈ {1,2}, N ≤ 1024, H = hidden_dim (normally 128).

The output of the kernel is also [B, N, N, H] and is later normalised,
gated and projected back to the original channel dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    B, N, H,                                 # sizes
    stride_lb, stride_li, stride_lk, stride_ld,  # left strides (bytes)
    stride_rb, stride_rj, stride_rk, stride_rd,  # right strides (bytes)
    stride_ob, stride_oi, stride_oj, stride_od,  # out strides (bytes)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.

    The kernel is launched with a 3‑D grid:
        (num_blocks_M, num_blocks_N, B*H)

    Each program instance processes a tile of size (BLOCK_M × BLOCK_N)
    for a particular (batch, hidden) pair.
    """
    pid_bhd = tl.program_id(2)          # combined batch * hidden index
    pid_m   = tl.program_id(0)          # block row   (i)
    pid_n   = tl.program_id(1)          # block col   (j)

    # recover batch and hidden indices
    b = pid_bhd // H
    d = pid_bhd % H

    # ---------- tile offsets ----------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------- loop over K ----------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_lb
            + d * stride_ld
            + offs_m[:, None] * stride_li
            + offs_k[None, :] * stride_lk
        )
        left = tl.load(
            left_ptrs,
            mask=(mask_m[:, None] & mask_k[None, :]),
            other=0.0,
        ).to(tl.float32)

        # right tile: need (k, j) layout → shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_rb
            + d * stride_rd
            + offs_k[:, None] * stride_rk
            + offs_n[None, :] * stride_rj
        )
        right = tl.load(
            right_ptrs,
            mask=(mask_k[:, None] & mask_n[None, :]),
            other=0.0,
        ).to(tl.float32)

        # fused multiply‑add
        acc += tl.dot(left, right)      # (BLOCK_M, BLOCK_N)

    # ---------- store result ----------
    out_ptrs = (
        out_ptr
        + b * stride_ob
        + d * stride_od
        + offs_m[:, None] * stride_oi
        + offs_n[None, :] * stride_oj
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N] or None
            - weights      : dict of torch.Tensor
            - config       : dict with keys "dim", "hidden_dim"
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_epsilon", 1e-5)

    # ------------------------------------------------------------------
    # Load model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Optional mask handling
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid on linear of the normalized input)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the heavy einsum
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for stride calculations
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Helper to get stride in bytes
    def _stride_bytes(t):
        return tuple(s * t.element_size() for s in t.stride())

    l_s = _stride_bytes(left)
    r_s = _stride_bytes(right)
    o_s = _stride_bytes(out)

    # Block sizes – tuned for H100 and hidden_dim=128
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32   # 32 keeps register pressure modest

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,        # blocks over i
        (N + BLOCK_N - 1) // BLOCK_N,        # blocks over j
        B * hidden_dim,                      # one program per (batch, hidden)
    )

    _trimul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_lb=l_s[0],
        stride_li=l_s[1],
        stride_lk=l_s[2],
        stride_ld=l_s[3],
        stride_rb=r_s[0],
        stride_rj=r_s[1],
        stride_rk=r_s[2],
        stride_rd=r_s[3],
        stride_ob=o_s[0],
        stride_oi=o_s[1],
        stride_oj=o_s[2],
        stride_od=o_s[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # Output post‑processing (LayerNorm → gate → final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)    # back to original channel dim

    return out
