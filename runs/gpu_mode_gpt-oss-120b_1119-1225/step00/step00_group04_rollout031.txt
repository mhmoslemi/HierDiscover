def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(2)
    pid_m = tl.program_id(0)  # i tile
    pid_n = tl.program_id(1)  # j tile

    # offset base pointers for batch
    lo = left_ptr + pid_b * stride_left_b
    ro = right_ptr + pid_b * stride_right_b
    oo = out_ptr + pid_b * stride_out_b

    # offsets for i and j within tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for i/j within bounds
    mask_i = offs_m < N
    mask_j = offs_n < N

    # Loop over hidden dimension
    for d_off in range(0, hidden_dim, BLOCK_D):
        offs_d = d_off + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden_dim

        # allocate accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # Loop over k tiles
        for k_tile in range(0, N, BLOCK_K):
            cur_k = k_tile + tl.arange(0, BLOCK_K)
            mask_k = cur_k < N

            # compute pointers for left and right slices
            # shape (BLOCK_M, BLOCK_K, BLOCK_D)
            left_ptrs = lo + (offs_m[:, None, None] * stride_left_i
                               + cur_k[None, :, None] * stride_left_k
                               + offs_d[None, None, :] * stride_left_d)
            right_ptrs = ro + (offs_n[:, None, None] * stride_right_j
                               + cur_k[None, :, None] * stride_right_k
                               + offs_d[None, None, :] * stride_right_d)

            left = tl.load(left_ptrs, mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)
            right = tl.load(right_ptrs, mask = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0)

            # Compute outer product accumulation across K dimension
            # left: (M, K, D); right: (N, K, D)
            # We want acc += sum_k left[:,k,:] (outer) right[:,k,:]
            # Efficiently compute using broadcast and sum reduction.
            # Approach: expand left to (M, K, 1, D), right to (1, K, N, D)
            # Then multiply => (M, K, N, D), sum over K.
            # Can compute using a loop over k_small vectorized? Use tl.sum.
            # Use tl.dot across K dimension for each D? Not directly.
            # Use separate accumulation: for each k in K tile, perform vectorized outer product across M,N for all D.

            # Unroll k loop for speed:
            for kk in range(0, BLOCK_K):
                # mask per k
                k_valid = mask_k[kk]
                if not k_valid:
                    continue  # continue for out-of-bound k

                left_vec = left[:, kk, :]   # (M, D)
                right_vec = right[:, kk, :] # (N, D)

                # Broadcast: left_vec[:, None, :] * right_vec[None, :, :] -> (M, N, D)
                acc += left_vec[:, None, :] * right_vec[None, :, :]

        # End k loop

        # Store accumulator to out
        out_ptrs = oo + (offs_m[:, None, None] * stride_out_i
                         + offs_n[None, :, None] * stride_out_j
                         + offs_d[None, None, :] * stride_out_d)
        # mask for output indices
        mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_out)

@triton.jit
def layernorm_fwd_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_outb, stride_outi, stride_outj, stride_outc,
    stride_weight, stride_bias,
    epsilon: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # compute b,i,j
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # compute base pointers
    x_ptrs = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_ptrs = out_ptr + b * stride_outb + i * stride_outi + j * stride_outj

    # channel indices
    c = tl.arange(0, BLOCK_C)
    c_mask = c < C

    # load channels
    x = tl.load(x_ptrs + c * stride_xc, mask=c_mask, other=0.0).to(tl.float32)

    # Compute mean
    mean = tl.sum(x, axis=0) / C

    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) / C

    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # load weight and bias (scale and shift)
    w = tl.load(weight_ptr + c * stride_weight, mask=c_mask, other=1.0).to(tl.float32)
    b_bias = tl.load(bias_ptr + c * stride_bias, mask=c_mask, other=0.0).to(tl.float32)

    y = (x_centered * inv_std) * w + b_bias

    # store
    tl.store(out_ptrs + c * stride_outc, y, mask=c_mask)

"""
custom_kernel: TriMul (outgoing) forward pass with Triton‑fused LayerNorm.

Algorithm
---------
1. Fuse the first LayerNorm (over the input channel dimension `dim`) in a Triton
   kernel.  The kernel computes per‑position mean/variance, applies the learned
   scale (`weight`) and shift (`bias`) and writes the normalized tensor.
2. Apply the four linear projections (left, right, left_gate, right_gate,
   out_gate) with `torch.nn.functional.linear`.  The gating projections are
   followed by a sigmoid.
3. Multiply the two projected tensors by the optional mask and by their
   respective gates.
4. Reduce the large pairwise product using a batched matrix‑multiply:
   for every hidden channel `h`,
        out[..., :, :, h] = left[..., :, :, h] @ right[..., :, :, h].T
   This is implemented with a single `torch.bmm` on the reshaped tensors
   (highly optimised cuBLAS on H100).
5. Fuse the second LayerNorm (over the hidden dimension `hidden_dim`) in a
   second Triton kernel.
6. Apply the output gate, then the final projection back to `dim` with a
   bias‑free linear layer.

All heavy arithmetic (LayerNorm and the O(N³) inner product) stays on the GPU.
The final output is returned as a float32 tensor of shape
[bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_fwd_kernel(
    x_ptr,                 # *float32   input tensor
    out_ptr,               # *float32   output tensor
    weight_ptr,            # *float32   gamma (scale)
    bias_ptr,              # *float32   beta (shift)
    B, N, C,               # i32       batch, spatial, channel
    stride_xb, stride_xi, stride_xj, stride_xc,   # i32 strides for x
    stride_ob, stride_oi, stride_oj, stride_oc,   # i32 strides for out
    stride_weight, stride_bias,                    # i32 strides for weight/bias (1‑D)
    epsilon: tl.constexpr,                         # f32
    BLOCK_C: tl.constexpr,                         # i32, compile‑time channel tile size
):
    """LayerNorm forward for a 4‑D tensor [B, N, N, C] (norm over final dim)."""
    pid = tl.program_id(0)               # linear index over B*N*N
    total = B * N * N
    if pid >= total:
        return

    # decode (b, i, j) from the linear pid
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base pointers for this (b,i,j) element
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    out_base = out_ptr + b * stride_ob + i * stride_oi + j * stride_oj

    # channel indices (vectorized)
    c = tl.arange(0, BLOCK_C)
    c_mask = c < C

    # load the C values for this position
    x = tl.load(x_base + c * stride_xc, mask=c_mask, other=0.0).to(tl.float32)

    # compute mean and variance (unbiased=False)
    mean = tl.sum(x, axis=0) / C
    x_centered = x - mean
    var = tl.sum(x_centered * x_centered, axis=0) / C
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # load scale (weight) and shift (bias)
    w = tl.load(weight_ptr + c * stride_weight, mask=c_mask, other=1.0).to(tl.float32)
    b_shift = tl.load(bias_ptr + c * stride_bias, mask=c_mask, other=0.0).to(tl.float32)

    # normalize → scale → shift
    y = (x_centered * inv_std) * w + b_shift

    # write back
    tl.store(out_base + c * stride_oc, y, mask=c_mask)


def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim] (float32, GPU)
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # configuration & constants
    # ------------------------------------------------------------------
    dim = config["dim"]                # input channel dimension (C)
    hidden_dim = config["hidden_dim"]  # hidden dimension after projection
    eps = 1e-5

    B, N, _, _ = input_tensor.shape   # input shape = [B, N, N, dim]

    # ------------------------------------------------------------------
    # 1. First LayerNorm (over dim) – Triton kernel
    # ------------------------------------------------------------------
    x_norm = torch.empty_like(input_tensor)
    total_elements = B * N * N
    # compile‑time channel tile = dim (max 384, comfortably fits in registers)
    layernorm_fwd_kernel[(total_elements,)](
        x_ptr=input_tensor,
        out_ptr=x_norm,
        weight_ptr=weights["norm.weight"],
        bias_ptr=weights["norm.bias"],
        B=B,
        N=N,
        C=dim,
        stride_xb=input_tensor.stride(0),
        stride_xi=input_tensor.stride(1),
        stride_xj=input_tensor.stride(2),
        stride_xc=input_tensor.stride(3),
        stride_ob=x_norm.stride(0),
        stride_oi=x_norm.stride(1),
        stride_oj=x_norm.stride(2),
        stride_oc=x_norm.stride(3),
        stride_weight=weights["norm.weight"].stride(0),
        stride_bias=weights["norm.bias"].stride(0),
        epsilon=eps,
        BLOCK_C=dim,   # full channel dimension in one launch
    )

    # ------------------------------------------------------------------
    # 2. Linear projections & gates (pure PyTorch – already highly tuned)
    # ------------------------------------------------------------------
    # project to hidden dimension
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # optional mask (mask shape [B,N,N])
    if not config.get("nomask", False) and mask_tensor is not None:
        mask_f = mask_tensor.to(dtype).unsqueeze(-1)   # broadcast over hidden
        left = left * mask_f
        right = right * mask_f

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Pairwise multiplicative reduction (batched matrix multiply)
    #    out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    # ------------------------------------------------------------------
    # reshape to (B*hidden, N, N) for bmm
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # matrix multiply with transposed right (k dimension)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to [B, N, N, hidden_dim]
    out = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 4. Second LayerNorm (over hidden_dim) – Triton kernel
    # ------------------------------------------------------------------
    out_norm = torch.empty_like(out)
    total_elements_out = B * N * N
    layernorm_fwd_kernel[(total_elements_out,)](
        x_ptr=out,
        out_ptr=out_norm,
        weight_ptr=weights["to_out_norm.weight"],
        bias_ptr=weights["to_out_norm.bias"],
        B=B,
        N=N,
        C=hidden_dim,
        stride_xb=out.stride(0),
        stride_xi=out.stride(1),
        stride_xj=out.stride(2),
        stride_xc=out.stride(3),
        stride_ob=out_norm.stride(0),
        stride_oi=out_norm.stride(1),
        stride_oj=out_norm.stride(2),
        stride_oc=out_norm.stride(3),
        stride_weight=weights["to_out_norm.weight"].stride(0),
        stride_bias=weights["to_out_norm.bias"].stride(0),
        epsilon=eps,
        BLOCK_C=hidden_dim,   # hidden_dim ≤ 128 ⇒ one launch
    )

    # ------------------------------------------------------------------
    # 5. Output gating and final projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                       # broadcast over hidden
    final_out = F.linear(out_norm, weights["to_out.weight"])   # [B,N,N,dim]

    return final_out
