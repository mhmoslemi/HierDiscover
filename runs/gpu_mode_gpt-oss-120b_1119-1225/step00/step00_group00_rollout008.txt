@triton.jit
def batched_matmul_kernel(
    A, B, C,
    M, N, K,
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Compute pointers for this block
    a_ptrs = A + batch * stride_Abatch + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B + batch * stride_Bbatch + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

        # Increment pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write back result
    c_ptrs = C + batch * stride_Cbatch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask_c = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)

```python
import torch
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A, B, C,
                          M, N, K,
                          stride_Abatch, stride_Am, stride_Ak,
                          stride_Bbatch, stride_Bk, stride_Bn,
                          stride_Cbatch, stride_Cm, stride_Cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    batch = tl.program_id(1)

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for the current block
    a_ptrs = A + batch * stride_Abatch + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B + batch * stride_Bbatch + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # Accumulator in the compute type (float16 here)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
        acc += tl.dot(a, b)
        a_ptrs += BLOCK_K * stride_Ak           # move to next K‑tile of A
        b_ptrs += BLOCK_K * stride_Bk           # move to next K‑tile of B

    # Write out
    c_ptrs = C + batch * stride_Cbatch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module from AlphaFold3.
    The kernel fuses the heavy pairwise multiplication (the ``einsum`` over the
    sequence dimension) as a batched matrix multiplication in half‑precision.
    All other operations (layer‑norms, linear projections, gating, mask) are
    performed with PyTorch in single‑precision.
    """
    # Unpack arguments
    if len(data) == 4:
        input_tensor, mask, weights, config = data
    else:
        raise ValueError("Expected data tuple (input, mask, weights, config)")

    device = input_tensor.device
    dtype = input_tensor.dtype

    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) LayerNorm of the input
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections and gating weights
    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcasted on the channel dimension)
    if not nomask and mask is not None:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Prepare tensors for the batched GEMM
    #   left:  (B, N, N, hidden)  -> (B, hidden, N, N)  (i, k)
    #   right: (B, N, N, hidden)  -> (B, hidden, k, N)  (k, j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, k, j)

    B, H, N, _ = left_perm.shape
    # Flatten batch and hidden dimensions for the kernel
    left_flat = left_perm.view(-1, N, N).to(torch.float16)
    right_flat = right_perm.view(-1, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # 6) Launch Triton kernel: batched matmul A @ B   (A: i×k, B: k×j)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m * num_pid_n, left_flat.shape[0])   # (blocks over MxN, batch*hidden)

    # Strides
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to the original layout and cast to FP32 for the rest
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # 8) Final LayerNorm, out‑gate, and output projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

"""
custom_kernel
==============

Forward pass of the “outgoing” TriMul operator (AlphaFold‑3).

The heavy part of the computation is the Einstein sum

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication for every hidden channel *d*.
We implement this O(N³·C) step with a Triton kernel that performs a
half‑precision GEMM on the (batch*hidden) dimension:
    (i × k) @ (k × j) → (i × j)

All other operations (LayerNorm, Linear projections, sigmoid gates,
optional mask) are carried out with PyTorch (float‑32).  The result of the
kernel is cast back to float‑32 for the final LayerNorm, gating and output
projection.

The implementation supports variable batch size (1‑2), sequence length
N ∈ {128,256,512,1024}, model dimension *dim* (e.g. 128 or 384) and hidden
dimension *hidden_dim* (usually 128).  It works with or without a mask and
with any of the input distributions used in the tests.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched half‑precision matrix multiplication
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix sizes (M=i, N=j, K=k)
    stride_Abatch, stride_Am, stride_Ak,
    stride_Bbatch, stride_Bk, stride_Bn,
    stride_Cbatch, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B   with A : (batch, M, K), B : (batch, K, N), C : (batch, M, N)
    All tensors are in half‑precision; accumulation is also half‑precision
    (sufficient for the model because a final float‑32 cast follows).
    """
    pid = tl.program_id(0)                 # block index over the (M, N) plane
    batch = tl.program_id(1)               # batch * hidden index

    num_pid_m = tl.cdiv(M, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # Block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers for the current tile
    a_ptrs = A + batch * stride_Abatch + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
    b_ptrs = B + batch * stride_Bbatch + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

    # Accumulator (float16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)
        acc += tl.dot(a, b)                # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)
        a_ptrs += BLOCK_K * stride_Ak       # advance A by one K‑tile
        b_ptrs += BLOCK_K * stride_Bk       # advance B by one K‑tile

    # Store the result
    c_ptrs = C + batch * stride_Cbatch + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)


# ----------------------------------------------------------------------
# Entry point used by the judge
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args
    ----
    data : tuple
        (input_tensor, mask, weights, config)

        * input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        * mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
        * weights      : dict of model parameters (already on CUDA)
        * config       : dict containing
              - "dim"         (input feature dimension)
              - "hidden_dim"  (projection dimension)
              - "nomask"      (bool, optional)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32) – the forward output.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    if len(data) != 4:
        raise ValueError("custom_kernel expects a 4‑tuple: (input, mask, weights, config)")

    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Configuration values
    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over channel dimension)
    # ------------------------------------------------------------------
    if (mask is not None) and (not nomask):
        # mask: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate   = F.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Prepare for the batched GEMM (half‑precision)
    #    left : (B, N, N, hidden)  -> (B, hidden, i, k)
    #    right: (B, N, N, hidden)  -> (B, hidden, k, j)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, k, j)

    B, H, N, _ = left_perm.shape                    # B=batch, H=hidden, N=seq_len
    left_flat  = left_perm.view(-1, N, N).to(torch.float16)   # (B*H, N, N)
    right_flat = right_perm.view(-1, N, N).to(torch.float16)

    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 6) Triton launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Number of thread blocks covering the (M,N) plane
    num_pid_m = (N + BLOCK_M - 1) // BLOCK_M
    num_pid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_pid_m * num_pid_n, left_flat.shape[0])   # (blocks over M×N, batch*hidden)

    # Strides for the three tensors (in elements, not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bk, stride_bn = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # Launch the kernel
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch, stride_am, stride_ak,
        stride_b_batch, stride_bk, stride_bn,
        stride_c_batch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 7) Restore the original layout and convert back to FP32
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)
    # shape now: [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 8) Final LayerNorm, out‑gate and output projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # (dim, hidden)

    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # -> [B, N, N, dim] (float32)

    return out
