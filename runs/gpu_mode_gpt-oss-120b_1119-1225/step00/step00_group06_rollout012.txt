@triton.jit
def triangular_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)   # block row
    pid_j = tl.program_id(1)   # block col
    pid_z = tl.program_id(2)   # batch * hidden

    # compute batch index b and hidden index h
    b = pid_z // H
    h = pid_z % H

    # compute start indices for i and j
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets arrays
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    # bound masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_base in range(0, N, BLOCK_K):
        # compute k offsets
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointer offsets for left (i,k)
        # left index: (b, i, k, h)
        left_ptrs = left_ptr + b*stride_l_b + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k + h*stride_l_h
        # load left tile
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)  # dtype float16

        # pointer offsets for right (j,k) transposed as (k,j)
        # right index: (b, j, k, h) need transposed => (b, k, j, h)
        right_ptrs = right_ptr + b*stride_r_b + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j + h*stride_r_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # compute dot product and accumulate
        acc += tl.dot(left, right)  # left: (M,K), right: (K,N) -> (M,N). Both are float16; result is float32.

    # Store result
    out_ptrs = out_ptr + b*stride_o_b + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j + h*stride_o_h
    # store only within bounds
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated kernel implementation.

The forward pass follows the reference PyTorch code:

1. LayerNorm on the input channel dimension.
2. Four linear projections (left, right, left‑gate, right‑gate, out‑gate) without bias.
3. Optional mask (broadcasted on the channel dimension) applied to the left/right
   projections before gating.
4. Element‑wise gating (sigmoid) and multiplication.
5. Core O(N³·H) operation:
       out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   This is a batch of H independent N×N matrix multiplications (left @ rightᵀ) and
   is implemented in a fused Triton kernel for maximum bandwidth reuse.
6. LayerNorm over the hidden dimension, out‑gate gating and a final linear
   projection back to the original dimension.

Only the heavy pairwise multiplication (step 5) is executed on Triton; all
other operations use standard PyTorch kernels (they are memory‑bound and already
highly optimized).  The Triton kernel works on FP16 inputs, accumulates in
FP32 and writes the result as FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: per‑channel matrix multiplication (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr,                 # const half*   (B, N, N, H)
    right_ptr,                # const half*   (B, N, N, H)
    out_ptr,                  #       float*  (B, N, N, H)
    B, N, H,                  # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int32 strides for left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # int32 strides for right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # int32 strides for out
    BLOCK_M: tl.constexpr,    # tile rows   (i dimension)
    BLOCK_N: tl.constexpr,    # tile cols   (j dimension)
    BLOCK_K: tl.constexpr,    # reduction k dimension
):
    """
    Compute:
        out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
    for a tile (BLOCK_M × BLOCK_N) of (i, j) and a single channel h.
    """
    pid_i = tl.program_id(0)   # tile row index
    pid_j = tl.program_id(1)   # tile column index
    pid_z = tl.program_id(2)   # combined batch × channel index

    # --------------------------------------------------------------
    # Decode batch and hidden‑channel from pid_z
    # --------------------------------------------------------------
    b = pid_z // H
    h = pid_z % H

    # --------------------------------------------------------------
    # Offsets for the i and j dimensions
    # --------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # --------------------------------------------------------------
    # Accumulator in FP32
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Main k‑reduction loop
    # --------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + h * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # half → will be promoted on dot

        # right[b, j, k, h] transposed to (k, j) for dot
        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + offs_k[:, None] * stride_r_k
            + offs_j[None, :] * stride_r_j
            + h * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )   # half

        # Accumulate the inner product over k
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N), FP32 accumulation

    # --------------------------------------------------------------
    # Write the result
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + h * stride_o_h
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D]  (float32)
        - mask         : torch.Tensor [B, N, N]    (bool/float, optional)
        - weights      : dict of parameter tensors
        - config       : dict with keys "dim", "hidden_dim", "nomask"
    Returns
    -------
    torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-6,
    )  # shape [B, N, N, D], float32

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # Linear projections (all without bias)
    # ------------------------------------------------------------------
    # Flatten the first three dims to run a single GEMM per projection
    x_flat = x.view(-1, dim)                     # (B·N·N, D)

    left_proj = F.linear(x_flat, weights["left_proj.weight"])   # (B·N·N, H)
    right_proj = F.linear(x_flat, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"]))

    # Restore the original 4‑D layout
    left_proj = left_proj.view(B, N, N, hidden_dim)
    right_proj = right_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask may be bool, uint8, or float; convert to half and broadcast
        mask_f = mask.to(torch.float16).unsqueeze(-1)          # (B, N, N, 1)
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # ------------------------------------------------------------------
    # Apply gating
    # ------------------------------------------------------------------
    left = left_proj * left_gate   # float32
    right = right_proj * right_gate

    # Cast to half for the Triton kernel (memory bandwidth reduction)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # Triton kernel: per‑channel matmul left @ rightᵀ
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32)   # (B, N, N, H)

    # Strides in element units
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left.stride()
    r_stride_b, r_stride_j, r_stride_k, r_stride_h = right.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Tile sizes – chosen to fit in shared memory on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * hidden_dim,
    )
    _trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        l_stride_b,
        l_stride_i,
        l_stride_k,
        l_stride_h,
        r_stride_b,
        r_stride_j,
        r_stride_k,
        r_stride_h,
        o_stride_b,
        o_stride_i,
        o_stride_j,
        o_stride_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Output post‑processing (LayerNorm + out‑gate + final linear)
    # ------------------------------------------------------------------
    # Hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-6,
    )  # float32

    # Apply out‑gate (broadcasted)
    out = out * out_gate

    # Final linear projection back to `dim`
    out_flat = out.view(-1, hidden_dim)                # (B·N·N, H)
    out = F.linear(out_flat, weights["to_out.weight"])  # (B·N·N, D)
    out = out.view(B, N, N, dim)

    return out
