"""
custom_kernel for the outgoing Triangle Multiplicative Update (TriMul).

Algorithm
---------
1. Layer‑norm the input tensor `x` using the provided LN weights.
2. Linear projections:
   * `left  = x @ left_proj_weight.T`
   * `right = x @ right_proj_weight.T`
3. Optional mask (broadcasted on the hidden dimension) is applied to `left`
   and `right`.
4. Gated versions of the projections are obtained with three sigmoid
   gates (`left_gate`, `right_gate`, `out_gate`).  The two first gates are
   multiplied into `left`/`right`; the third gate will be applied after the
   multiplicative update.
5. **Outgoing multiplicative update** – compute, for every batch `b`,
   hidden channel `h` and pair of positions `(i, j)`:

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   This is a batched matrix‑multiply of shape `[B*H, N, N]` where each
   matrix is `left[...,h] @ right[...,h].T`.  The heavy O(N³·H) work is
   performed by a tiled Triton kernel that loads blocks of `left` and the
   transposed `right` in shared registers and accumulates with `tl.dot`.
6. Apply a second layer‑norm on the hidden dimension, multiply by `out_gate`,
   and finally a linear projection back to the original `dim` size.

The kernel works with `float16` for the large intermediate tensors
(`left`, `right`) and accumulates in `float32` to keep numerical stability.
All tensors are assumed to be on the same CUDA device.  The implementation
handles arbitrary batch sizes (1‑2), sequence lengths up to 1024,
and hidden dimensions up to 128 (as required by the test suite).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel for the outgoing multiplicative update (left @ right.T)
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,            # *const half   (B, N, N, H)
    right_ptr,           # *const half   (B, N, N, H)
    out_ptr,             # *mut   float32 (B, N, N, H)
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
    The kernel iterates over blocks of the `k` dimension and uses
    `tl.dot` for the inner multiplication / accumulation.
    """
    pid_i = tl.program_id(axis=0)      # block index in i dimension
    pid_j = tl.program_id(axis=1)      # block index in j dimension
    pid_z = tl.program_id(axis=2)      # combined batch * hidden index

    # decode batch and hidden channel from pid_z
    batch = pid_z // H
    hidden = pid_z % H

    # start offsets for the current tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range of i / j indices for this tile
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)

    # mask to ignore out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load a tile of `left` : shape (BLOCK_M, BLOCK_K) -----
        # left[b, i, k, h]
        off_l = (
            batch * stride_l_b
            + offs_i[:, None] * stride_l_i
            + offs_k[None, :] * stride_l_k
            + hidden * stride_l_h
        )
        left = tl.load(
            left_ptr + off_l,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # ----- load a tile of transposed `right` : shape (BLOCK_K, BLOCK_N) -----
        # we need right[b, j, k, h] but treat it as (k, j) for the dot
        off_r = (
            batch * stride_r_b
            + offs_k[:, None] * stride_r_k        # k index (row)
            + offs_j[None, :] * stride_r_i        # j index (col)
            + hidden * stride_r_h
        )
        right = tl.load(
            right_ptr + off_r,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )  # fp16

        # Inner product over k -> accumulate into fp32
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_N) fp32

    # ------------------------------------------------------------------
    # Store the result
    # ------------------------------------------------------------------
    off_o = (
        batch * stride_o_b
        + offs_i[:, None] * stride_o_i
        + offs_j[None, :] * stride_o_j
        + hidden * stride_o_h
    )
    tl.store(
        out_ptr + off_o,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def trimul_outgoing_matmul(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor):
    """
    Wrapper around the Triton kernel.  Assumes:
        left  : (B, N, N, H)  dtype=half, contiguous
        right : (B, N, N, H)  dtype=half, contiguous
        out   : (B, N, N, H)  dtype=float32, contiguous
    """
    B, N, _, H = left.shape

    # Extract strides (in element counts, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Choose tiling parameters.  These are empirically good for N up to 1024.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid: (i_tiles, j_tiles, batch*hidden)
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        B * H,
    )
    trimul_outgoing_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    # No need to return; `out` is written in‑place.


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module implemented with a
    Triton kernel for the costly multiplicative update.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (float32 or bool)
        - weights      : dict with all layer‑norm and linear weights
        - config       : dict containing `dim` and `hidden_dim`
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5   # layer‑norm epsilon (same as PyTorch default)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Linear projections (float16 for speed)
    # ------------------------------------------------------------------
    # Linear layers without bias: out = x @ weight.T
    left_proj_w = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]    # (hidden_dim, dim)

    # Project, keep FP16 for the massive tensors
    left = F.linear(x, left_proj_w).to(torch.float16)
    right = F.linear(x, right_proj_w).to(torch.float16)

    # ------------------------------------------------------------------
    # 3. Mask (if present)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask: [B, N, N] → [B, N, N, 1] and broadcast
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (computed in FP32 for numerical stability)
    # ------------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))  # [B,N,N,hidden]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))    # [B,N,N,hidden]

    # Apply the first two gates (still FP16 after multiplication)
    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    # ------------------------------------------------------------------
    # 5. Outgoing multiplicative update (Trilinear product)
    # ------------------------------------------------------------------
    # Allocate output buffer (fp32) and run Triton kernel
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left.device)
    trimul_outgoing_matmul(left, right, out)   # out now in fp32

    # ------------------------------------------------------------------
    # 6. Second LayerNorm on hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )  # still fp32

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate  # out_gate is fp32, broadcast over hidden dim

    to_out_w = weights["to_out.weight"]   # (dim, hidden_dim)
    out = F.linear(out, to_out_w)        # [B, N, N, dim]  fp32

    return out
