"""
TriMul kernel (outgoing update) – fast batched matmul implementation with Triton.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor `x` (shape [B, N, N, C]).
2. Linear projections to hidden dimension:
        left  = x_norm @ W_left  (no bias)
        right = x_norm @ W_right (no bias)
3. Optional pairwise mask (broadcast to the hidden dim) is applied to `left`
   and `right`.
4. Compute three gating vectors from `x_norm` via sigmoid‑gated linear layers
   and fuse them:
        left  = left  * sigmoid(x_norm @ W_left_gate)
        right = right * sigmoid(x_norm @ W_right_gate)
        out_gate = sigmoid(x_norm @ W_out_gate)
5. Heavy cubic reduction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is exactly a batched matrix‑multiply:
        out[b,:, :,d] = left[b,:, :,d] @ right[b,:, :,d].T
   A custom Triton kernel computes the N×N matrix products for all
   (batch, hidden) pairs in parallel.
6. Apply a second LayerNorm on the hidden dimension, multiply the result
   by `out_gate` and finish with a linear projection back to `C`.

The Triton kernel treats each `(batch, head)` pair as an independent program
instance and tiles the matrices with (BLOCK_M, BLOCK_N, BLOCK_K).  The kernel
uses fp32 accumulation (the inputs are fp32) and respects arbitrary strides,
so it works for any contiguous layout produced by the surrounding PyTorch
code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul for the outgoing TriMul update
# ----------------------------------------------------------------------
@triton.jit
def _triton_trimul_outgoing_kernel(
    # tensor pointers
    left_ptr, right_ptr, out_ptr,
    # static sizes
    BATCH, SEQLEN, HIDDEN,
    # strides (int64)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each (batch, head) pair:
        out = left @ right.T
    where `left` and `right` have shape [N, N] and `out` also [N, N].
    The combined batch dimension is BATCH * HIDDEN.
    """
    # --------------------------------------------------------------
    # 1) Identify which (batch, head) we are processing.
    # --------------------------------------------------------------
    pid = tl.program_id(0)                     # 0 … BATCH*HIDDEN-1
    batch_idx = pid // HIDDEN
    head_idx  = pid % HIDDEN

    # Tile indices for the output matrix
    pid_m = tl.program_id(1)                   # row‑tile
    pid_n = tl.program_id(2)                   # col‑tile

    # ------------------------------------------------------------------
    # 2) Compute base pointers for this (batch, head) pair.
    # ------------------------------------------------------------------
    left_base  = left_ptr  + batch_idx * stride_left_batch  + head_idx * stride_left_h
    right_base = right_ptr + batch_idx * stride_right_batch + head_idx * stride_right_h
    out_base   = out_ptr   + batch_idx * stride_out_batch   + head_idx * stride_out_h

    # ------------------------------------------------------------------
    # 3) Offsets for the current tile.
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # `k` will be iterated in the inner loop
    offs_k = tl.arange(0, BLOCK_K)

    # Mask for out‑of‑bounds rows/cols
    mask_i = offs_i < SEQLEN
    mask_j = offs_j < SEQLEN

    # Accumulator (fp32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) Loop over the reduction dimension `k`.
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(SEQLEN, BLOCK_K)
    for k in range(0, num_k_blocks):
        k_start = k * BLOCK_K
        cur_k = k_start + offs_k
        mask_k = cur_k < SEQLEN

        # ------- load A = left[b,i,k] --------------------------------
        a_ptrs = left_base + \
                 (offs_i[:, None] * stride_left_i) + \
                 (cur_k[None, :] * stride_left_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)

        # ------- load B = right[b,j,k]  (we need right.T, so we load
        #         as [k, j] which matches (K,N) for the dot product) ---
        b_ptrs = right_base + \
                 (cur_k[:, None] * stride_right_k) + \
                 (offs_j[None, :] * stride_right_j)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # ------- dot product for this k‑block ------------------------
        acc += tl.dot(a, b)        # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------------------------------------------
    # 5) Write the result back to global memory.
    # ------------------------------------------------------------------
    c_ptrs = out_base + \
             (offs_i[:, None] * stride_out_i) + \
             (offs_j[None, :] * stride_out_j)
    tl.store(c_ptrs, acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator implemented with a
    Triton‑accelerated batched matrix multiplication.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C]
        - mask         : torch.Tensor, shape [B, N, N] (binary or float)
        - weights      : dict of torch.Tensor, model parameters
        - config       : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32 on the same device).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    # Compute mean / var over the channel axis
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = ((input_tensor - mean) ** 2).mean(dim=-1, keepdim=True)
    inv_std = torch.rsqrt(var + eps)

    x_norm = (input_tensor - mean) * inv_std
    x_norm = x_norm * norm_weight + norm_bias   # broadcast over leading dims

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)  # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # Apply pairwise mask (if present).  `nomask` can be True → mask is all ones.
    # --------------------------------------------------------------
    if config.get("nomask", False):
        mask_tensor = torch.ones_like(mask, dtype=left.dtype, device=device)
    else:
        mask_tensor = mask.to(left.dtype)
    mask_unsq = mask_tensor.unsqueeze(-1)          # [B,N,N,1]

    left  = left  * mask_unsq
    right = right * mask_unsq

    # --------------------------------------------------------------
    # Gating vectors (sigmoid of linear transforms of the *normalized* input)
    # --------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]   # (hidden_dim, dim)
    right_gate_weight = weights["right_gate.weight"]  # (hidden_dim, dim)
    out_gate_weight   = weights["out_gate.weight"]    # (hidden_dim, dim)

    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Heavy cubic reduction – batched matmul via Triton
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # Ensure tensors are contiguous for stride calculations
    left  = left.contiguous()
    right = right.contiguous()
    out   = out.contiguous()

    # Strides (int64) – note the order is (B, N, N, H)
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – tuned for H100 (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: one program per (batch * hidden_dim) pair, plus tiles for M and N.
    grid = (B * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _triton_trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        BATCH=B,
        SEQLEN=N,
        HIDDEN=hidden_dim,
        stride_left_batch=stride_left_batch,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_batch=stride_right_batch,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_batch=stride_out_batch,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # Second LayerNorm on hidden dimension, gating, final linear
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]      # (dim, hidden_dim)

    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # final projection back to `dim`

    return out
