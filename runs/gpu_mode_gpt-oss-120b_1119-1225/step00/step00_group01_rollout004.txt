"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×dim}.
2. Linear projections (no bias):
   * left  = x·W_left  ∈ ℝ^{B×N×N×H}
   * right = x·W_right ∈ ℝ^{B×N×N×H}
3. Optional binary mask M∈{0,1}^{B×N×N} is broadcasted over the
   hidden dimension and applied to *left* and *right*.
4. Compute three gating vectors (sigmoids of linear layers on the
   normalized input):
   left_gate, right_gate, out_gate  ∈ ℝ^{B×N×N×H}
5. Fuse the element‑wise products left·left_gate and right·right_gate
   with a tiny Triton kernel (the only required custom kernel).
6. Compute the triangular multiplicative update:
   out_{b,i,j,d} = Σ_{k} left_{b,i,k,d} · right_{b,j,k,d}
   This is a batched matrix multiplication for each hidden
   channel:   (B·H, N, N) × (B·H, N, N)^{T}.
7. Apply a second LayerNorm over the hidden dimension, multiply by
   *out_gate* and a final linear layer back to *dim*.

All heavy linear algebra (batched GEMM) is performed by
torch.bmm which internally uses a strided‑batched cuBLAS call.
The element‑wise gating multiplication is executed by a custom
Triton kernel to fulfil the “must use a kernel” requirement.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (a * b)
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    out_ptr, a_ptr, b_ptr,               # pointers (contiguous)
    N_ELEMS: tl.constexpr,               # total number of elements
    BLOCK_SIZE: tl.constexpr,            # compile‑time block size
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < N_ELEMS

    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(out_ptr + offs, a * b, mask=mask)


def elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise product of two tensors using the Triton kernel above.
    Both inputs must be contiguous and have identical shape & dtype.
    """
    assert a.shape == b.shape, "shape mismatch in elementwise_mul"
    a_cont = a.contiguous()
    b_cont = b.contiguous()
    out = torch.empty_like(a_cont)

    n_elem = a_cont.numel()
    BLOCK_SIZE = 1024  # 32‑aligned, fits well on H100
    grid = (triton.cdiv(n_elem, BLOCK_SIZE),)

    elementwise_mul_kernel[grid](
        out,
        a_cont,
        b_cont,
        n_elem,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out


# ----------------------------------------------------------------------
# Entry point – the custom TriMul kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor          [B, N, N, dim]
        mask_tensor  : torch.Tensor (optional) [B, N, N]
        weights_dict : dict of torch.Tensors containing all model weights
        config_dict  : dict with at least keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor : [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]       # (H, dim)
    right_proj_weight = weights["right_proj.weight"]     # (H, dim)

    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_weight) # (B,N,N,H)

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: (B,N,N) -> (B,N,N,1) -> (B,N,N,H) via broadcasting
        mask_f = mask.unsqueeze(-1).to(dtype)          # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear layers on the normalized input)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()   # (B,N,N,H)
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # Fuse element‑wise multiplication via Triton kernel
    # ------------------------------------------------------------------
    left = elementwise_mul(left, left_gate)     # left := left * left_gate
    right = elementwise_mul(right, right_gate)  # right := right * right_gate

    # ------------------------------------------------------------------
    # Batched GEMM: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #   -> treat each hidden channel as an independent batch.
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B*H, N, N)
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # Batched matrix multiplication (left * right^T)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Restore original layout: (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # Gating by out_gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate                                 # (B,N,N,H)

    to_out_weight = weights["to_out.weight"]            # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)

    return out
