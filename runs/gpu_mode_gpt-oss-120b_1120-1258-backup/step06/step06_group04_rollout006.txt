def custom_kernel(data):
    x, mask, weights, cfg = data
    B,N,_,dim = x.shape
    hidden_dim = cfg['hidden_dim']
    eps = 1e-5
    nomask = cfg.get('nomask', True)
    device = x.device
    total_rows = B*N*N

    # 1. LayerNorm: fused fp32 -> fp16
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty_like(x_flat, dtype=torch.float16)
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights['norm.weight'],
        weights['norm.bias'],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # shape
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # 2. Combine left/right projection+gate as single linear steps
    rows = total_rows
    # concat left/proj+gate weight
    left_w = torch.cat([weights['left_proj.weight'], weights['left_gate.weight']], dim=0).to(dtype=torch.float16, device=device)
    right_w = torch.cat([weights['right_proj.weight'], weights['right_gate.weight']], dim=0).to(dtype=torch.float16, device=device)

    # linear for all rows: produces (rows, 2*hidden_dim)
    left_cat = torch.nn.functional.linear(x_norm.view(rows, dim), left_w)
    right_cat = torch.nn.functional.linear(x_norm.view(rows, dim), right_w)

    # split result
    left_proj = left_cat[:, :hidden_dim]
    left_gate_pre = left_cat[:, hidden_dim:]
    right_proj = right_cat[:, :hidden_dim]
    right_gate_pre = right_cat[:, hidden_dim:]

    # output gate
    out_gate = torch.nn.functional.linear(
        x_norm.view(rows, dim), 
        weights['out_gate.weight'].to(dtype=torch.float16, device=device)
    )  # shape (rows, hidden_dim)
    out_gate = torch.sigmoid(out_gate)

    # 3. optional mask - mask shape (B,N,N) -> (rows, 1)
    if not nomask:
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # 4. apply sigmoid gates
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate
    right_h = right_proj * right_gate

    # free unneeded variables?
    del left_proj, left_gate_pre, right_proj, right_gate_pre, left_cat, right_cat

    # 5. reshape for batched matmul
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)
    # permute to (B, hidden_dim, N, N) and reshape as (B*hidden_dim, N, N)
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # 6. batched GEMM: out_mat = left @ right^T (both (B*H,N,N))
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # 7. reshape to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 8. second LayerNorm + multiplication with out_gate + final projection
    # We compute layernorm using Triton kernel

"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm over the last dimension (`dim`).  A custom Triton
    kernel (`layernorm_fp16_kernel`) reads FP32 input, computes mean &
    variance, writes the normalized result as FP16.

2️⃣  Fuse the left/right projection + gate linear layers.  The two
    linear maps for each side are concatenated (2·hidden_dim × dim) and
    evaluated with a single `torch.nn.functional.linear`.  The output
    is split into the projection and pre‑gate tensors, the gate values
    are activated with `sigmoid`, and an (optional) mask is applied.

3️⃣  Compute the tri‑multiplicative update
        out[b,i,j,h] = Σₖ left[b,i,k,h] · right[b,j,k,h]
    as a batched GEMM.  By reshaping the tensors we obtain a batch of
    size B·hidden_dim and call `torch.bmm`, which runs on the H100
    Tensor‑Core units in FP16.

4️⃣  Second LayerNorm (over the hidden dimension) – the same Triton
    kernel as in step 1.

5️⃣  Multiply by the output gate, then a final linear projection
    (hidden → dim).  All heavy arithmetic stays in FP16; the final
    result is cast back to FP32.

Only the LayerNorms are written in Triton; the remaining ops use
high‑throughput PyTorch kernels (cuBLAS / Tensor‑Core).  The design
reduces memory traffic (fused linear‑gate, in‑place masking) and
launches the Triton kernel with a larger row block (`BLOCK_M=256`) to
lower launch overhead.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var+eps)

    # ---------- Pass 2: write normalized values (γ·x̂+β) ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be all‑ones)
        - weights_dict : mapping the weight names used in the reference
                         PyTorch implementation to torch.Tensors on the same device.
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor[float32] shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) using Triton
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()               # [R, dim] FP32
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Tuning: a larger row block reduces launch overhead.
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # Shape back to [B, N, N, dim] (still FP16)
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left/right branches
    # --------------------------------------------------------------
    rows = total_rows  # B·N·N

    # Concatenate (proj, gate) weights → (2·hidden_dim, dim)
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    # Linear maps (rows, dim) @ (2·H, dim).T → (rows, 2·H)
    left_cat = F.linear(x_norm.view(rows, dim), left_w)   # FP16
    right_cat = F.linear(x_norm.view(rows, dim), right_w)

    # Split the concatenated results
    left_proj = left_cat[:, :hidden_dim]          # (rows, H)
    left_gate_pre = left_cat[:, hidden_dim:]      # (rows, H)
    right_proj = right_cat[:, :hidden_dim]
    right_gate_pre = right_cat[:, hidden_dim:]

    # Output gate (shared across positions)
    out_gate = torch.sigmoid(
        F.linear(
            x_norm.view(rows, dim),
            weights["out_gate.weight"].to(dtype=torch.float16, device=device),
        )
    )   # (rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate          # (rows, H)
    right_h = right_proj * right_gate       # (rows, H)

    # Release temporary tensors that are no longer needed
    del left_proj, left_gate_pre, right_proj, right_gate_pre, left_cat, right_cat

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via batched GEMM
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # Merge H into the batch dimension → [B·H, N, N]
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_mat = left_mat @ right_matᵀ   (FP16, tensor‑core accelerated)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) → FP16
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection
    # --------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, hidden_dim)       # FP16
    gated = out_norm * out_gate                         # FP16

    final_fp16 = F.linear(
        gated.view(rows_h, hidden_dim),
        weights["to_out.weight"].to(dtype=torch.float16, device=device),
    )   # (rows_h, dim) FP16

    # Cast back to FP32 for compatibility with the reference implementation
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
```

**Explanation of key optimisations**

* **Triton LayerNorm** – replaces the PyTorch `nn.LayerNorm` which would need
  two full‑tensor passes (mean/variance then normalisation).  The kernel
  works directly on the flattened `[B·N·N, dim]` tensor and writes the
  normalised result in FP16, halving memory traffic.

* **Fused Linear‑Gate** – left/right projection and their corresponding
  gates are computed with a single `torch.nn.functional.linear` each
  (`2·hidden_dim × dim` weight) and then split.  This removes four separate
  matmul calls.

* **In‑place mask & gating** – the optional mask is applied after the
  projection,‑gate activation is performed in‑place, reducing temporary
  allocations.

* **Batched GEMM** – the core `einsum` is expressed as a batched matrix
  multiplication (`torch.bmm`).  On H100 this maps to the Tensor‑Core
  implementation and is far faster than a naïve loop or `einsum`.

* **Larger Triton block size** – `BLOCK_M = 256` cuts the number of kernel
  launches for the layer‑norms roughly in half, lowering launch overhead.

All heavy arithmetic stays in FP16; only the final result is cast back to
FP32, matching the original PyTorch module’s output type while achieving
significant speed‑ups.
