"""
TriMul (outgoing) – fast forward implementation.

The kernel follows the reference PyTorch module but:
1. Row‑wise LayerNorm (both normalisations) are fused into a custom Triton kernel
   that computes mean/variance in fp32 while keeping the output in fp16.
2. The cubic‑time update `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`
   is realised as a batched matrix multiplication `L @ Rᵀ` (B·H independent GEMMs)
   instead of a slow `torch.einsum`.  This lets the H100’s Tensor Cores handle the
   ≈ N³·H FLOPs efficiently.
3. All heavy linear work stays in fp16, only the LayerNorm statistics are in fp32.
   The final result is cast back to float32 to match the reference.

The function `custom_kernel` expects a tuple
    (input_tensor, mask_tensor, weights_dict, config_dict)
and returns a float32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp16 output, fp32 statistics)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const  * [total_rows, dim]   (fp16 or fp32)
    y_ptr,                # mutable* [total_rows, dim]   (fp16)
    weight_ptr,           # const  * [dim]               (fp32)
    bias_ptr,             # const  * [dim]               (fp32)
    total_rows,           # int (runtime)
    dim,                  # int (runtime)
    eps,                  # fp32 scalar (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    Computes μ, σ² in fp32 and writes   y = ((x-μ)/√(σ²+eps))*γ + β   in fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # 1️⃣  Compute sum and sum‑of‑squares for mean/variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # Load as the type of x_ptr (fp16 or fp32) and promote to fp32
        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1/√(var+eps)

    # --------------------------------------------------------------
    # 2️⃣  Write normalised values (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ  (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β  (fp32)

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (first) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] float32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256
    BLOCK_D = 128                     # works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate linear layers
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices (left/right proj + left/right gate)
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"], # [H, dim]
            weights["left_gate.weight"],  # [H, dim]
            weights["right_gate.weight"], # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # shape [4*H, dim]

    # Linear mapping (fp16) → [R, 4*H]
    proj_gate = F.linear(x_norm, w_4h)      # fp16

    # Slice the result
    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2*H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]

    # Output gate (independent weight)
    out_gate_raw = F.linear(x_norm,
                            weights["out_gate.weight"].to(torch.float16))   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    # mask handling – applied after projection, before gating
    if not config.get("nomask", True):
        # mask shape: [B, N, N] → flatten to [R, 1]
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for gate tensors
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Batched matrix multiplication (the cubic update)
    # ------------------------------------------------------------------
    # Permute to [B, H, N, N] and fuse B·H as batch dimension
    left_bh = left.permute(0, 3, 1, 2).reshape(B * H, N, N)          # fp16
    right_bh = right.permute(0, 3, 1, 2).reshape(B * H, N, N)        # fp16

    # out_bh[b*h + h, i, j] = Σ_k left[i,k] * right[j,k]
    out_bh = torch.bmm(left_bh, right_bh.transpose(1, 2))           # fp16, shape [B*H, N, N]

    # Restore shape [B, N, N, H]
    out = out_bh.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)       # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate               # [B, N, N, H] fp16
    final = F.linear(out_norm.reshape(total_rows, H),
                     weights["to_out.weight"].to(torch.float16))   # fp16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
```
