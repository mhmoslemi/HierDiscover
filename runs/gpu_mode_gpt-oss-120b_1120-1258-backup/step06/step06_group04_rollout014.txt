"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton + fused Linear

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor  [B,N,N,D]  (FP32 → FP16)
    – custom Triton kernel (no extra Python loops).

2️⃣  Fuse the three Linear projections (left, right, out‑gate) into **one**
    matrix‑multiply:
        (B·N·N, D)  @  (5·H, D).T  →  (B·N·N, 5·H)
    The 5·H channels are split into
        left_proj, left_gate, right_proj, right_gate, out_gate.
    This reduces the number of passes over the normalized tensor
    (3 × linear → 1 × linear) and lowers memory traffic.

3️⃣  Apply the sigmoid gates and the optional mask, then obtain the
    gated tensors `left_h` and `right_h` of shape (B,N,N,H).

4️⃣  Reshape to (B·H, N, N) and perform the batched GEMM that implements
    the einsum   out_h(i,j) = Σₖ left_h(i,k)·right_h(j,k).
    (torch.bmm on FP16, fully Tensor‑Core accelerated.)

5️⃣  Second LayerNorm over the hidden dimension (FP16) – same Triton kernel.

6️⃣  Multiply by the output gate, then a final Linear (hidden → D).

All heavy arithmetic is performed in FP16; the final result matches the
reference module (float32).  At least one Triton kernel (LayerNorm) is
present, satisfying the problem requirement while substantially cutting
runtime by fusing the three Linear layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input, FP32)
    y_ptr,          # half*    [total_rows, dim]   (output, FP16)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)         # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                   # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+eps)

    # ---------- Pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - μ) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point: custom Triton‑accelerated TriMul (outgoing)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32] shape [B, N, N, D]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be ignored)
        weights_dict : {
            "norm.weight":          Tensor[D],
            "norm.bias":            Tensor[D],
            "left_proj.weight":     Tensor[H, D],
            "right_proj.weight":    Tensor[H, D],
            "left_gate.weight":     Tensor[H, D],
            "right_gate.weight":    Tensor[H, D],
            "out_gate.weight":      Tensor[H, D],
            "to_out_norm.weight":   Tensor[H],
            "to_out_norm.bias":     Tensor[H],
            "to_out.weight":        Tensor[D, H],
        }
        config_dict  : {"dim": int,
                       "hidden_dim": int,
                       "nomask": bool (optional, default True)}

    Returns:
        torch.Tensor[float32] shape [B, N, N, D]
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.reshape(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Block sizes tuned for dim ≤ 384
    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gate (single GEMM)
    # ------------------------------------------------------------------
    # Concatenate all five linear weight matrices: L_proj, L_gate,
    # R_proj, R_gate, Out_gate   → shape (5*H, dim)
    fused_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    fused_out = F.linear(x_norm_fp16, fused_weight)   # (rows, 5*H), FP16

    # Split the concatenated output
    left_proj = fused_out[:, :H]                      # (rows, H)
    left_gate_pre = fused_out[:, H:2 * H]             # (rows, H)
    right_proj = fused_out[:, 2 * H:3 * H]            # (rows, H)
    right_gate_pre = fused_out[:, 3 * H:4 * H]        # (rows, H)
    out_gate_pre = fused_out[:, 4 * H:]               # (rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.reshape(total_rows, 1).to(dtype=torch.float16,
                                                  device=device,
                                                  non_blocking=True)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)            # (rows, H)

    left_h = left_proj * left_gate                    # (rows, H)
    right_h = right_proj * right_gate                 # (rows, H)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left_h = left_h.view(B, N, N, H)
    right_h = right_h.view(B, N, N, H)

    # Bring hidden dim to batch axis → [B*H, N, N]
    left_mat = left_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication (FP16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # Back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H), FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dimension
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)          # (B,N,N,H), FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # --------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, H)               # (B,N,N,H), FP16
    gated = out_norm * out_gate                        # (B,N,N,H), FP16

    # Final Linear (hidden → dim)
    to_out_weight = weights["to_out.weight"].to(dtype=torch.float16,
                                                device=device,
                                                non_blocking=True)
    final_fp16 = F.linear(gated.view(rows_h, H), to_out_weight)  # (rows_h, dim)

    # Cast back to FP32 as required by the reference implementation
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
