"""
Fast outgoing TriMul (AlphaFold‑3) forward pass – PyTorch + Triton.

The implementation fuses the two LayerNorms into a custom Triton
kernel (row‑wise, fp32 reduction → fp16 output).  All linear projections,
gating, mask handling, the N³ multiplicative update and the final
linear are kept in half‑precision and executed with Torch’s highly
optimised kernels (batched GEMM).  The cubic update is performed by a
single batched matrix‑multiply (`torch.bmm`) after a cheap reshape, which
is considerably faster than `torch.einsum`.  The final tensor is
cast back to float32 as required by the reference model.

Supported shapes:
    input  : [B, N, N, dim]    (float32)
    mask   : [B, N, N]         (bool/float, optional)
    dim ∈ {128, 384}, hidden_dim = 128
    B ∈ {1,2},   N ≤ 1024
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (float32)
    y_ptr,               # mutable * [total_rows, dim]   (float16)
    weight_ptr,          # const   * [dim]               (float32)
    bias_ptr,            # const   * [dim]               (float32)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # fp32 (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm.
    Compute μ,σ² in fp32 and write the normalised y in fp16:
        y = ((x-μ)/√(σ²+eps))*weight + bias
    """
    pid = tl.program_id(0)                     # each program processes a block of rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute sum and sum‑of‑squares (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                 # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalised values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out_fp32 = norm * w[None, :] + b[None, :]                # fp32

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp32.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    # 6️⃣ Second LayerNorm over hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()   # (R, H)
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    _layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)               # (B, N, N, H) fp16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate and final linear projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # (B, N, N, H) fp16
    final = F.linear(out_norm.view(total_rows, H),
                     weights["to_out.weight"].to(torch.float16))   # (R, dim) fp16
    final = final.view(B, N, N, dim).to(torch.float32)                # (B,N,N,dim) float32

    return final
```
