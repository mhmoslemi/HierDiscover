def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N
    H = hidden_dim

    # 1. first layernorm via triton
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    x_norm = torch.empty_like(x_flat, dtype=torch.float16)
    BLOCK_M = 256
    BLOCK_D = 32 if dim <= 64 else 64 if dim <= 128 else 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    # 2. fused projection and gating weights
    w_proj_gate = torch.cat([
        weights["left_proj.weight"],
        weights["right_proj.weight"],
        weights["left_gate.weight"],
        weights["right_gate.weight"]
    ], dim=0).to(torch.float16)
    proj_gate = F.linear(x_norm, w_proj_gate)  # [R, 4*H] fp16

    left_proj_raw = proj_gate[:, :H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    out_gate_raw = F.linear(x_norm, weights["out_gate.weight"].to(torch.float16))

    # 3. apply mask if not nomask
    if not config.get("nomask", True):
        mask = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask
        right_proj_raw = right_proj_raw * mask
    # 4. sigmoid gates
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate
    right = right_proj_raw * right_gate

    # 5. reshape
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # 6. batched GEMM for cubic multiplication
    # permute to (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # right transpose on last two dims
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    # 7. second LayerNorm over hidden dim
    out_flat = out.view(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # out_norm shape [total_rows, H] half
    out_norm = out_norm * out_gate.view(total_rows, H)  # gating

    # 8. final linear projection
    final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))
    final = final.view(B, N, N, dim).to(torch.float32)

    return final

"""
Fast outgoing TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) – Triton kernel.
2️⃣ Fused linear projection + gating:
   * Concatenate left/right projection and gate weights → one FP16 GEMM.
   * Separate the 4×hidden_dim result into left_proj, right_proj,
     left_gate, right_gate.
   * Compute out_gate with a second GEMM.
3️⃣ Optional mask (broadcast over the hidden dim) applied to the
   *raw* projected values before gating.
4️⃣ Sigmoid gates → element‑wise gating of the projected tensors.
5️⃣ Reshape to 4‑D tensors `[B,N,N,H]`.
6️⃣ Cubic multiplicative update:
   `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   Implemented as a batched matrix‑multiply (B·H independent GEMMs)
   using `torch.bmm`, i.e. `out = left @ rightᵀ`.
7️⃣ Second row‑wise LayerNorm (fp16) – Triton kernel.
8️⃣ Apply the output gate, a final FP16 linear projection back to `dim`,
   and cast the result to float32.

All heavy arithmetic (projections, gates, GEMM, layer‑norm) is performed
in half‑precision to fully exploit H100 Tensor‑Core throughput.
Only the two LayerNorms are written in Triton; the remaining
operations use highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim] (float32)
    y_ptr,               # mutable * [total_rows, dim] (float16)
    weight_ptr,          # const   * [dim]           (float32)
    bias_ptr,            # const   * [dim]           (float32)
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per program
):
    """
    Compute per‑row mean/variance in fp32 and write the normalised
    result in fp16:  y = ((x-μ)/√(σ²+eps))*γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute sum and sum‑of‑squares for mean/var
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalised values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : model parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()   # [R, dim] fp32
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)                       # [R, dim] fp16

    BLOCK_M = 256
    BLOCK_D = 64                     # works for dim up to 384
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gates (one GEMM for the four matrices)
    # ------------------------------------------------------------------
    # concat left/right projection + left/right gate weight matrices
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [4*H, dim]

    proj_gate = F.linear(x_norm, w_proj_gate)            # [R, 4*H] fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2*H]
    left_gate_raw  = proj_gate[:, 2*H:3*H]
    right_gate_raw = proj_gate[:, 3*H:4*H]

    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)      # [H, dim]
    )                                                    # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)  # [R,1]
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate          # [R, H] fp16
    right = right_proj_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Reshape to 4‑D tensors
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6️⃣ Cubic multiplicative update (batched GEMM)
    #     out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #     ⇒ for each (b,h) perform MatMul( left_{bh}, right_{bh}ᵀ )
    # ------------------------------------------------------------------
    # bring hidden dim to the batch dimension
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)               # [B*H, N, N]
    right_mat = right_perm.view(B * H, N, N)              # [B*H, N, N]

    # Batched matmul: left @ rightᵀ  →  [B*H, N, N]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()        # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.reshape(total_rows, H)    # gating (fp16)

    final = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16)          # [dim, H]
    )                                                        # [R, dim] fp16

    final = final.reshape(B, N, N, dim).to(torch.float32)   # cast back to fp32

    return final
