    # gating

    right_gate = torch.sigmoid(right_gate_pre)

    rows_h = B * N * N
    out_flat = out.view(rows_h, h).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)
    grid2 = (triton.cdiv(rows_h, BLOCK_M),)

        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        h,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, h)

    # Apply output gate
    out_gate = out_gate.view(B, N, N, h)   # shape same as out_norm
    gated = out_norm * out_gate

    final_fp16 = torch.nn.functional.linear(
        gated.view(rows_h, h),
        weights["to_out.weight"].to(dtype=torch.float16, device=device)
    )
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)
    return final

"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑accelerated.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the last dimension (FP32 → FP16) using a custom
    Triton kernel.
2️⃣ Fuse the four linear projections (left_proj, left_gate, right_proj,
    right_gate) **and** the output gate into a **single** FP16 GEMM.
   Split the result, apply the optional mask, compute sigmoid gates and
   obtain the gated left/right hidden tensors.
3️⃣ Tri‑multiplicative update:
        out_h(b,i,j) = Σ_k left_h(b,i,k)·right_h(b,j,k)
   implemented as a batched GEMM on FP16 tensors
   (shape [B*H, N, N] → torch.bmm).
4️⃣ Second LayerNorm over the hidden dimension (FP16) via the same Triton
   kernel.
5️⃣ Apply the output gate, a final FP16 linear projection (hidden → dim) and
   cast the result back to float32.

All heavy arithmetic is performed in FP16; the custom Triton kernel is used
for both LayerNorm passes, satisfying the requirement to implement a part of
the computation in Triton while keeping the overall runtime well below the
target.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per block
):
    """Row‑wise LayerNorm on a 2‑D view of the tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                 # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var + eps)

    # ---------- Pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul (outgoing) forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        mask_tensor  : torch.Tensor[float32] shape [B, N, N] (may be ignored)
        weights_dict : {
            "norm.weight":          Tensor[dim],
            "norm.bias":            Tensor[dim],
            "left_proj.weight":     Tensor[hidden_dim, dim],
            "right_proj.weight":    Tensor[hidden_dim, dim],
            "left_gate.weight":     Tensor[hidden_dim, dim],
            "right_gate.weight":    Tensor[hidden_dim, dim],
            "out_gate.weight":      Tensor[hidden_dim, dim],
            "to_out_norm.weight":   Tensor[hidden_dim],
            "to_out_norm.bias":     Tensor[hidden_dim],
            "to_out.weight":        Tensor[dim, hidden_dim],
        }
        config_dict  : {"dim": int,
                       "hidden_dim": int,
                       "nomask": bool (optional, default True)}

    Returns:
        torch.Tensor[float32] shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )
    # Tuned block sizes (dim ≤ 384 → 1–3 column blocks)
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating + output‑gate (single GEMM)
    # ------------------------------------------------------------------
    # Assemble a single weight matrix of shape (5·H, dim)
    w_left_proj = weights["left_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate   = weights["out_gate.weight"]
    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    # Linear: (rows, dim) @ (5·H, dim).T → (rows, 5·H)
    combined = F.linear(x_norm_fp16, combined_weight)  # FP16

    h = hidden_dim
    left_proj      = combined[:, 0 * h : 1 * h]   # (rows, H)
    left_gate_pre  = combined[:, 1 * h : 2 * h]
    right_proj     = combined[:, 2 * h : 3 * h]
    right_gate_pre = combined[:, 3 * h : 4 * h]
    out_gate_pre   = combined[:, 4 * h : 5 * h]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(total_rows, 1).to(dtype=torch.float16, device=device)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)          # (rows, H)

    left_h  = left_proj * left_gate   # (rows, H)
    right_h = right_proj * right_gate # (rows, H)

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left_h  = left_h.view(B, N, N, h)
    right_h = right_h.view(B, N, N, h)

    # Fuse hidden dimension into the batch axis: [B·H, N, N]
    left_mat  = left_h.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)
    right_mat = right_h.permute(0, 3, 1, 2).contiguous().view(B * h, N, N)

    # out_mat[b·h, i, j] = Σ_k left_mat[b·h, i, k] * right_mat[b·h, j, k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16

    # Reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, h, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden_dim (FP16)
    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, h).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, h)   # FP16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection
    # ------------------------------------------------------------------
    out_gate = out_gate.view(B, N, N, h)       # FP16
    gated = out_norm * out_gate                 # FP16

    final_fp16 = F.linear(
        gated.view(rows_h, h),
        weights["to_out.weight"]
        .to(dtype=torch.float16, device=device, non_blocking=True)
    )                                            # (rows_h, dim) FP16

    # Cast back to float32 as required by the reference module
    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
