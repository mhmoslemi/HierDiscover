"""
Optimized outgoing TriMul forward pass (AlphaFold‑3) using Triton kernels.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (float32 → float16) on the input tensor.
2️⃣ Fuse the four linear projections (left/right + left‑gate/right‑gate) into a
   single half‑precision GEMM:   X_norm @ [W_left;W_right;W_lgate;W_rgate].T
   → raw left/right tensors and gate tensors.
3️⃣ Compute the output‑gate projection (another half‑precision GEMM).
4️⃣ Apply optional mask to left/right, then sigmoid on gates and element‑wise
   gating.
5️⃣ Triangular multiplicative update (batched half‑precision GEMM):
   (B·H, N, N) @ (B·H, N, N)ᵀ   (H = hidden_dim).
6️⃣ Fuse the second LayerNorm (over the hidden dimension) with the
   multiplication by the output‑gate in a single Triton kernel.
7️⃣ Final linear projection back to `dim` (half‑precision) and cast to float32.

All heavy per‑row normalisations are performed in custom Triton kernels,
while the large matrix multiplications rely on cuBLAS Tensor‑Core kernels.
The implementation works for both masked and unmasked inputs and supports
`dim ∈ {128,384}`, `hidden_dim = 128`, batch size 1‑2 and sequence lengths up
to 1024.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim] (float32)
    y_ptr,               # mutable * [total_rows, dim] (float16)
    weight_ptr,          # const   * [dim]           (float32) γ
    bias_ptr,            # const   * [dim]           (float32) β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- compute mean & variance ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)        # 1 / sqrt(var + eps)

    # ---- write normalized values γ·(x−μ)·rstd + β (fp16) -----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                       # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm + element‑wise multiplication by a gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,                  # input & output, [total_rows, dim] (fp16)
    weight_ptr, bias_ptr,          # LayerNorm γ,β (float32)
    gate_ptr,                      # gate tensor, same shape as x (fp16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- mean & variance (fp32) ------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, add bias, multiply by gate ---------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)                     # gate (fp32)

        out = norm * w[None, :] + b[None, :]    # fp32
        out = out * g                           # apply gate (fp32)

        out_fp16 = out.to(tl.float16)
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all linear / layer‑norm parameters
        - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack & prepare
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()          # [R, dim] fp32
    x_norm_half = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)                     # [R, dim] fp16

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_half,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Project to left/right + gate tensors (fused GEMM)
    # --------------------------------------------------------------
    # Convert linear weights to half‑precision once.
    w_left = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right = weights["right_proj.weight"].to(torch.float16)  # [H, dim]
    w_lgate = weights["left_gate.weight"].to(torch.float16)   # [H, dim]
    w_rgate = weights["right_gate.weight"].to(torch.float16)  # [H, dim]

    # Concatenate along output dimension: shape [4*H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)  # [4H, dim]

    # Half‑precision linear: X_norm @ W_concat.T → [R, 4H]
    proj_gate = F.linear(x_norm_half, w_concat)   # fp16

    left_raw   = proj_gate[:, 0:H]           # [R, H]
    right_raw  = proj_gate[:, H:2*H]         # [R, H]
    lgate_raw  = proj_gate[:, 2*H:3*H]       # [R, H]
    rgate_raw  = proj_gate[:, 3*H:4*H]       # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate projection (separate GEMM)
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_norm_half, w_out_gate)           # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Optional mask + sigmoid gates
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [R, 1] (float16)
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_h
        right_raw = right_raw * mask_h

    lgate = lgate_raw.sigmoid_()          # in‑place sigmoid (fp16)
    rgate = rgate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()    # fp16, to be used later

    left  = left_raw * lgate               # [R, H] fp16
    right = right_raw * rgate              # [R, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H]
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Bring hidden dim into the batch axis: [B, H, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch & head: [B·H, N, N]
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Batched half‑precision matrix multiplication (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] fp16

    # Restore shape [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(total_rows, H)                                # [R, H]

    # --------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + multiplication by output gate
    # --------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],    # γ (float32)
        weights["to_out_norm.bias"],      # β (float32)
        out_gate.view(total_rows, H),     # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)   # back to fp32
    return final
