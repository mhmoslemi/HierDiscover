"""
custom_kernel
=============
Highly‑optimized forward pass for the *outgoing* TriMul operator (AlphaFold‑3).

Algorithm (all heavy per‑row work is done in custom Triton kernels):
1️⃣ Row‑wise LayerNorm on the input tensor (FP32 → FP16).
2️⃣ Fuse the four linear projections (left, right, left‑gate, right‑gate) into a
   single half‑precision GEMM:  [R, dim] @ [4·H, dim]ᵀ → [R, 4·H].
3️⃣ Compute the output‑gate projection (another half‑precision GEMM).
4️⃣ Apply optional mask, then in‑place sigmoid on the gate tensors and element‑wise
   gating of left/right projections.
5️⃣ Batched matrix multiplication (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N)
   implements the `einsum('... i k d, ... j k d -> ... i j d')`.
6️⃣ Fused LayerNorm over the hidden dimension **and** multiplication by the
   previously‑computed output‑gate (single Triton kernel).
7️⃣ Final linear projection back to `dim` (half‑precision GEMM) and cast to FP32.

The implementation works for masks present or absent, supports `dim` ∈ {128,384},
`hidden_dim` = 128, batch size 1‑2 and sequence lengths up to 1024.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   * [total_rows, dim]   (FP32)
    y_ptr,               # mutable * [total_rows, dim]   (FP16)
    weight_ptr,          # const   * [dim]               (FP32) γ
    bias_ptr,            # const   * [dim]               (FP32) β
    total_rows,          # int (runtime)
    dim,                 # int (runtime)
    eps,                 # float (runtime)
    BLOCK_M: tl.constexpr,  # rows processed per program
    BLOCK_D: tl.constexpr,  # cols processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean / variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # ---------- write normalized (γ·(x-μ)·rstd + β) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Fused LayerNorm + multiplication by a gate (hidden dim)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,                # input / output [total_rows, dim] (FP16)
    weight_ptr, bias_ptr,        # LN parameters γ,β (FP32)
    gate_ptr,                    # gate tensor, same shape as x (FP16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------- mean / variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------- normalized + bias + gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load input
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # load gate (FP16 → FP32)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate = tl.load(g_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # FP32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * gate          # FP32
        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or float)
        - weights_dict : mapping from string names to torch.Tensor (all parameters)
        - config_dict  : must contain "dim" (int) and "hidden_dim" (int);

    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (torch.float32)
    """

    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = input_tensor.view(total_rows, dim).contiguous()   # [R, dim] FP32

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)

        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (FP32)
        weights["norm.bias"],            # β (FP32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates
    # --------------------------------------------------------------
    # Convert linear weights to FP16 once
    w_left   = weights["left_proj.weight"].to(torch.float16)   # [H, dim]
    w_right  = weights["right_proj.weight"].to(torch.float16)  # [H, dim]
    w_lgate  = weights["left_gate.weight"].to(torch.float16)  # [H, dim]
    w_rgate  = weights["right_gate.weight"].to(torch.float16)  # [H, dim]

    # Concatenate along output dimension -> [4*H, dim]
    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate], dim=0)

    # Linear (R, dim) @ (4*H, dim).T  → (R, 4*H)   (FP16)
    proj_gate = F.linear(x_norm, w_concat)          # [R, 4*H] FP16

    left_proj_raw  = proj_gate[:, 0:H]               # [R, H]
    right_proj_raw = proj_gate[:, H:2 * H]           # [R, H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]       # [R, H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]       # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate linear projection
    # --------------------------------------------------------------
    w_out_gate = weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    out_gate_raw = F.linear(x_norm, w_out_gate)                # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣ Apply mask (if present) and sigmoid gates
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → [R, 1] (FP16) for broadcasting
        mask_h = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Element‑wise gating
    left = left_proj_raw * left_gate      # [R, H] FP16
    right = right_proj_raw * right_gate   # [R, H] FP16

    # --------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]

    # Move hidden dim into batch dimension for a simple bmm

    right = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Reshape to original layout
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(total_rows, H)                                 # [R, H]

    # --------------------------------------------------------------
    # 6️⃣ Fused LayerNorm (hidden dim) + multiplication by out‑gate
    # --------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (FP32)
        weights["to_out_norm.bias"],     # β (FP32)
        out_gate.view(total_rows, H),    # gate (FP16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] FP16

    # Cast back to FP32 as required by the reference implementation
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
