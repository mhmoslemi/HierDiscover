"""
Optimized **outgoing** TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the [B·N·N, dim] input (Triton kernel) → fp16.
2️⃣ Fuse the four linear projections (`left_proj`, `right_proj`,
   `left_gate`, `right_gate`) **and** the output‑gate projection
   into a *single* half‑precision GEMM:  [R, dim] @ [5·H, dim]ᵀ → [R,5·H].
   The result is split into:
        left_proj, right_proj, left_gate, right_gate, out_gate.
3️⃣ (Optional) apply the pair‑wise mask to `left_proj`/`right_proj`.
   Apply `sigmoid` to the three gate tensors (in‑place) and obtain
   gated tensors `left` and `right`.
4️⃣ Reshape to [B, N, N, H] and perform the expensive N³ update:
   for every head `h` compute (B·H) batched matrix‑multiplications
        out[h] = left[h] @ right[h]ᵀ
   using `torch.bmm` (cuBLAS/Tensor‑Core on H100).
5️⃣ Second Row‑wise LayerNorm on the hidden tensor (again Triton) → fp16.
6️⃣ Multiply by the output‑gate, project back to the original `dim`
   with a final half‑precision linear layer and cast the result to `float32`.

Only the two LayerNorms are written in Triton; all dense GEMMs use
cuBLAS which is already optimal for half‑precision on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 accumulation, FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (float32)
    y_ptr,          # mutable* [total_rows, dim] (float16)
    w_ptr,          # const  * [dim] (float32)   – gamma
    b_ptr,          # const  * [dim] (float32)   – beta
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The reduction (mean / variance) is performed in FP32,
    the final result is stored as FP16.
    """
    pid = tl.program_id(0)                     # one program per tile of rows
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean and variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # loop over column blocks
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)     # [BLOCK_D]
        mask_d = offs_d < dim

        # read a tile of shape [BLOCK_M, BLOCK_D]
        x_tile = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(x_tile, axis=1)
        sum_sq  += tl.sum(x_tile * x_tile, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                       # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized result γ·(x−μ)·rstd + β  (FP16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_tile = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (x_tile - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                # FP32
        out_fp16 = out.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – may be omitted
        - weights_dict : mapping from string to torch.Tensor containing all
                         Linear / LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim" and optional "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣  Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device
    B, N, _, _ = input_tensor.shape
    R = B * N * N                         # rows for the first LN

    # --------------------------------------------------------------
    # 1️⃣  First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = input_tensor.view(R, dim).contiguous()          # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    # launch Triton LN kernel
    BLOCK_M = 256          # can be tuned; 256 gives good occupancy on H100
    BLOCK_D = 32           # dim is multiple of 32 for all test cases
    grid = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],            # γ (FP32)
        weights["norm.bias"],              # β (FP32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣  Fuse left/right projections + the three gate projections
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate all five weight matrices: left_proj, right_proj,
    # left_gate, right_gate, out_gate   → shape [5·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                     # FP16

    # Single GEMM: [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]  (FP16)
    proj = F.linear(x_norm, w_fused)                        # FP16

    # Slice the fused result
    left_proj_raw   = proj[:, 0:H]          # [R, H]
    right_proj_raw  = proj[:, H:2*H]        # [R, H]
    left_gate_raw   = proj[:, 2*H:3*H]      # [R, H]
    right_gate_raw  = proj[:, 3*H:4*H]      # [R, H]
    out_gate_raw    = proj[:, 4*H:5*H]      # [R, H]

    # --------------------------------------------------------------
    # 3️⃣  Mask (if present), sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → flatten to [R, 1] and cast to half
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid (half‑precision)
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    # Apply the three gates
    left  = left_proj_raw * left_gate            # [R, H] FP16
    right = right_proj_raw * right_gate          # [R, H] FP16

    # --------------------------------------------------------------
    # 4️⃣  N³ multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape back to the original 4‑D layout
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # bring the head dimension into the batch axis for a single bmm call
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)  # [B·H, N, N]

    # Batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # Restore the original layout: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣  Second LayerNorm over the hidden dimension → FP16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],     # γ (FP32)
        weights["to_out_norm.bias"],       # β (FP32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)   # FP16

    # --------------------------------------------------------------
    # 6️⃣  Apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                         # [B,N,N,H] FP16

    final = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),        # [dim, H]
    )                                                       # FP16
    final = final.view(B, N, N, dim).to(torch.float32)

    return final
