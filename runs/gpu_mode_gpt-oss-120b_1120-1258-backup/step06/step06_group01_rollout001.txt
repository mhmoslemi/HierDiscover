"""
TriMul (outgoing) – fast forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input tensor (B·N·N × dim).  
    Implemented in Triton; reduction is in fp32 while the
    normalized output is stored in fp16.

2️⃣  Fuse the four projection/gate linear layers *and* the
    output‑gate linear into a *single* GEMM:
        [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   (R = B·N·N, H = hidden_dim)
    The result is split into
        left_proj, right_proj, left_gate, right_gate, out_gate.

3️⃣  (Optional) apply the pair‑mask to left/right proj,
    then apply the three sigmoid gates (in‑place, fp16).

4️⃣  Reshape to [B, N, N, H] and perform the N³ multiplicative
    update with a batched half‑precision GEMM:
        out_mat = left_mat @ right_matᵀ   →   [B·H, N, N].

5️⃣  Second row‑wise LayerNorm over the hidden dimension (again Triton).

6️⃣  Apply the output‑gate, project back to `dim` with a final
    linear layer (cuBLAS), and return fp32 output.

Only the two LayerNorm steps are written in Triton – everything
else uses highly‑optimized cuBLAS kernels via `torch.nn.functional.linear`
and `torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# --------------------------------------------------------------
@triton.jit
def rowwise_layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]    (float32)
    y_ptr,          # mutable * [total_rows, dim]    (float16)
    w_ptr,          # const   * [dim]                (float32) γ
    b_ptr,          # const   * [dim]                (float32) β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    Mean/variance are accumulated in fp32, the final result is stored
    in fp16 to save bandwidth.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over column blocks
    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(0, num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # [BLOCK_M, BLOCK_D] tile
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # reduction across columns
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized result (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

        dim=0,
    ).to(torch.float16)                # fp16 for cuBLAS Tensor‑Cores

    # fused GEMM: [R, dim] @ [5*H, dim]ᵀ → [R, 5*H] (fp16)
    proj = F.linear(x_norm, w_fused)   # fp16

    # split
    left_proj_raw   = proj[:, 0          : H]          # [R, H]
    right_proj_raw  = proj[:, H          : 2*H]       # [R, H]
    left_gate_raw   = proj[:, 2*H        : 3*H]       # [R, H]
    right_gate_raw  = proj[:, 3*H        : 4*H]       # [R, H]
    out_gate_raw    = proj[:, 4*H        : 5*H]       # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and element‑wise gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B, N, N] → flatten to [R, 1] (fp16)
        mask_h = mask_tensor.to(torch.float16).reshape(R, 1)
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # in‑place sigmoid (half precision)
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate    # [R, H] fp16
    right = right_proj_raw * right_gate  # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Reshape for the N³ multiplicative update
    # --------------------------------------------------------------
    # [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # bring head dimension into the batch axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # batched half‑precision GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    left_mat  = left_perm.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right_perm.view(B * H, N, N)         # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension → fp16
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    rowwise_layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate, final projection and cast to fp32
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                     # [B,N,N,H] fp16
    final_half = F.linear(
        out_norm.view(R, H),          # input
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                 # fp16, shape [R, dim]

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
