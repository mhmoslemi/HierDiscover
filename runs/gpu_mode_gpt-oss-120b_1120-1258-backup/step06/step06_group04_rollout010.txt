@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr, y_ptr,  # input and output [total_rows, dim]
    gamma_ptr, beta_ptr, gate_ptr,
    total_rows,
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # Pass1 sum and sumsq
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # x is fp16
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # Pass2: normalize, apply gamma,beta and gate
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptr_tile = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        gate_ptr_tile = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptr_tile, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        gate_vals = tl.load(gate_ptr_tile, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float16)

        norm = (vals - mean[:, None]) * rstd[:, None]  # fp32

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]  # fp32
        out_fp16 = out.to(tl.float16)

        # multiply by gate
        out_fp16 = out_fp16 * gate_vals

        y_ptr_tile = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptr_tile, out_fp16, mask=mask_m[:, None] & mask_d[None, :])
```

"""
TriMul “outgoing” forward (AlphaFold‑3) – fast Triton‑fused LayerNorm+gate.

Algorithm
---------
1️⃣  First LayerNorm (over the last channel) → fp16.  Implemented with
    torch.nn.functional.layer_norm (highly tuned native kernel).

2️⃣  Fuse *all* linear projections and gates (left_proj, left_gate,
    right_proj, right_gate, out_gate) into a single matmul:
        (B·N·N, dim) @ (5·H, dim).T → (B·N·N, 5·H)
    where H = hidden_dim.  Simple slicing yields the five tensors.

3️⃣  Apply the mask (if present) to left/right projections, then the
    sigmoid gates.  This yields `left_h` and `right_h` in fp16.

4️⃣  Tri‑multiplicative update:
        out(i,j) = Σ_k left_h(i,k)·right_h(j,k)
    performed as a batched GEMM (torch.bmm) on tensors of shape
    (B·H, N, N).  All arithmetic stays in fp16.

5️⃣  Second LayerNorm (over the hidden dimension) **and** multiplication
    by the output‑gate are fused into a custom Triton kernel
    `layernorm_gate_fp16_kernel`.  The kernel computes per‑row mean /
    variance, applies γ/β, then multiplies by the pre‑computed
    `out_gate`.

6️⃣  Final linear projection (hidden → dim) with a regular cuBLAS matmul,
    cast back to float32.

Only the fused LayerNorm+gate kernel is written in Triton, satisfying the
requirement that *some* work be performed on the GPU with Triton while
keeping the overall pipeline fully GPU‑accelerated.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_gate_fp16_kernel(
    x_ptr,          # fp16*   [total_rows, dim]   input tensor
    y_ptr,          # fp16*   [total_rows, dim]   normalized & gated output
    gamma_ptr,      # fp32*   [dim]               scale (γ)
    beta_ptr,       # fp32*   [dim]               bias (β)
    gate_ptr,       # fp16*   [total_rows, dim]   elementwise gate to apply after LN
    total_rows,     # int
    dim: tl.constexpr,   # hidden dimension (H)
    eps: tl.constexpr,   # epsilon for stability
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- Pass 1: compute mean and variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                                      # fp32

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)      # 1 / sqrt(var + eps)

    # ---------- Pass 2: normalize, apply γ/β and gate ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr   = x_ptr   + offs_m[:, None] * dim + offs_d[None, :]
        gate_tile_ptr = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)               # fp32

        gate = tl.load(
            gate_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float16)               # fp16

        # (x - μ) / σ
        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)  # fp32
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)  # fp32

        out = norm * w[None, :] + b[None, :]                    # fp32
        out_fp16 = out.to(tl.float16)                          # fp16

        # multiply by the element‑wise output gate
        out_fp16 = out_fp16 * gate

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    Arguments

    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)

        mask_tensor  : torch.Tensor            shape [B, N, N] (may be ignored)
        weights_dict : mapping of parameter names → torch.Tensor

                       - "norm.weight", "norm.bias"

                       - "left_gate.weight", "right_gate.weight"
                       - "out_gate.weight"
                       - "to_out_norm.weight", "to_out_norm.bias"
                       - "to_out.weight"
        config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor[float32] shape [B, N, N, D]
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    H = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # 1️⃣ First LayerNorm (native PyTorch – very fast)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)          # → fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all linear projections + gates into a single matmul
    # ------------------------------------------------------------------
    # concatenate weights: (5·H, dim)
    w_left_proj  = weights["left_proj.weight"]
    w_right_proj = weights["right_proj.weight"]
    w_left_gate  = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate   = weights["out_gate.weight"]

    combined_weight = torch.cat(
        [w_left_proj, w_left_gate, w_right_proj, w_right_gate, w_out_gate],
        dim=0,
    ).to(dtype=torch.float16, device=device).contiguous()   # (5·H, dim)

    rows = B * N * N
    x_flat = x_norm.view(rows, dim)          # (B·N·N, dim)

    # (rows, 5·H) = (rows, dim) @ (5·H, dim).T
    proj_concat = F.linear(x_flat, combined_weight)    # fp16

    # ------------------------------------------------------------------
    # Split the concatenated result
    # ------------------------------------------------------------------
    left_proj      = proj_concat[:, :H]
    left_gate_pre  = proj_concat[:, H:2*H]
    right_proj     = proj_concat[:, 2*H:3*H]
    right_gate_pre = proj_concat[:, 3*H:4*H]
    out_gate_pre   = proj_concat[:, 4*H:5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_fp = mask.view(rows, 1).to(dtype=torch.float16, device=device)
        left_proj  = left_proj * mask_fp
        right_proj = right_proj * mask_fp

    # ------------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)          # (rows, H)

    left_h  = left_proj * left_gate
    right_h = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # reshape to [B·H, N, N] for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # out_mat : (B·H, N, N) = left_mat @ right_matᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to [B, N, N, H]
    out = (
        out_mat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()

    # ------------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()               # fp16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(rows_h, BLOCK_M),)

    layernorm_gate_fp16_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                                 # fp16 gate
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(dtype=torch.float16, device=device),
    )   # (rows_h, dim) fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # cast to fp32
    return final
```
