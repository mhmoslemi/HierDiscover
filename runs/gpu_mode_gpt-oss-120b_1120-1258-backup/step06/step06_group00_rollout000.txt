"""
custom_kernel – Outgoing TriMul (AlphaFold‑3) forward pass
--------------------------------------------------------

Algorithm
~~~~~~~~~
1️⃣ Row‑wise LayerNorm on the input 4‑D tensor (B,N,N,dim) → fp16.
   Implemented with a small Triton kernel that computes mean/var per row.

2️⃣ Fuse the four linear projections (left,right and their gates) into a
   single half‑precision GEMM:
       [R, dim] @ [4·H, dim].T  →  [R, 4·H]   with  R = B·N·N
   The result is split into left_proj, right_proj, left_gate, right_gate.

3️⃣ Compute the output‑gate with a second half‑precision GEMM.

4️⃣ (Optional) Apply the pair‑wise mask only to left/right projections.
   Apply sigmoid in‑place to the three gate tensors and combine:
        left  = left_proj * sigmoid(left_gate)
        right = right_proj * sigmoid(right_gate)

5️⃣ Triangular multiplicative update (the core “einsum”):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched FP16 GEMM:
        (B·H, N, N) @ (B·H, N, N)ᵀ   using torch.bmm (Tensor‑Cores).

6️⃣ Second LayerNorm over the hidden dimension H → fp16 (same Triton kernel).

7️⃣ Multiply by the output‑gate, apply the final linear projection
   (hidden → dim) and cast the result back to float32.

All heavy normalisations are done with Triton, while the large matrix
multiplications are delegated to cuBLAS (torch.bmm / F.linear) for
maximum throughput on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ---------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,                # const   * [total_rows, dim]   (float32)
    y_ptr,                # mutable * [total_rows, dim]   (float16)
    weight_ptr,           # const   * [dim]               (float32)  γ
    bias_ptr,             # const   * [dim]               (float32)  β
    total_rows,           # int (runtime)
    dim,                  # int (runtime)
    eps,                  # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    """Row‑wise LayerNorm:
       y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # ----------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ----------------------------------------------------------------
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                 # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ---------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            weights_dict : dict of model parameters (LayerNorm & Linear weights)
            config_dict  : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------
    x, mask, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", True)   # True → no mask
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim

    # -----------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # -----------------------------------------------------------
    # flatten input over the row dimension
    x_flat = x.view(total_rows, dim).contiguous()          # [R, dim] fp32
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 256          # rows per program – tuned for H100
    BLOCK_D = 32           # cols per inner iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,                     # H100: 4 warps ≈ 128 threads per block
    )

    # -----------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (4·H outputs)
    # -----------------------------------------------------------
    # concatenate weight matrices for a single GEMM
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # half‑precision linear: [R, dim] @ [4·H, dim].T → [R, 4·H]
    fused = F.linear(x_norm_fp16, w_4h)     # fp16

    # split the fused tensor
    left_raw   = fused[:, 0:H]                     # [R, H]
    right_raw  = fused[:, H:2 * H]                 # [R, H]
    left_gate_raw  = fused[:, 2 * H:3 * H]         # [R, H]
    right_gate_raw = fused[:, 3 * H:4 * H]         # [R, H]

    # output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # -----------------------------------------------------------
    # 3️⃣ Mask, sigmoid gates, and combine
    # -----------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → flatten to [R, 1] (fp16)
        mask_fp16 = mask.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # in‑place sigmoid to reduce memory pressure
    left_gate   = left_gate_raw.sigmoid_()
    right_gate  = right_gate_raw.sigmoid_()
    out_gate    = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate          # [R, H] fp16
    right = right_raw * right_gate        # [R, H] fp16

    # -----------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # -----------------------------------------------------------
    # reshape to 4‑D tensors
    left  = left.view(B, N, N, H)              # [B, N, N, H]
    right = right.view(B, N, N, H)             # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)       # [B, N, N, H]

    # bring head dimension into the batch axis → [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # combine B and H → (B·H, N, N)
    left_mat  = left_perm.view(B * H, N, N)        # fp16
    right_mat = right_perm.view(B * H, N, N)       # fp16

    # batched matrix multiplication on Tensor‑Cores
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N] fp16

    # back to 4‑D: [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension → fp16
    # -----------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    out_norm = out_norm_fp16.view(B, N, N, H)   # fp16

    # -----------------------------------------------------------
    # 6️⃣ Apply output‑gate and final linear projection
    # -----------------------------------------------------------
    out_norm = out_norm * out_gate               # fp16

    final_fp16 = F.linear(
        out_norm.view(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                   # [R, dim] fp16

    # cast back to float32 for the model output
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
