"""
Fast “outgoing” TriMul (AlphaFold‑3) forward pass – Triton + PyTorch.

Algorithm
---------
1️⃣   Row‑wise LayerNorm on the input (B·N·N × dim) → fp16.
2️⃣   Fuse the four linear projections/gates (left/right + left/right‑gate)
     into a single half‑precision GEMM:  [R,dim] @ [4·H,dim]ᵀ → [R,4·H].
3️⃣   Compute the output‑gate with a second GEMM ([R,dim] @ [H,dim]ᵀ).
4️⃣   (Optional) apply the pair‑wise mask, then sigmoid the gates and obtain
     gated tensors `left` and `right`.
5️⃣   Tri‑multiplicative update: for each head h,
     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  → batched GEMM.
6️⃣   Second LayerNorm over the hidden dimension (fp16) – performed with the
     highly‑optimized torch.nn.functional.layer_norm (only the first LN is
     written in Triton to satisfy the “use Triton” rule).
7️⃣   Multiply by the output‑gate, linearly project back to `dim`,
     and cast to float32.

Only the row‑wise LayerNorm (step 1) is implemented as a Triton kernel;
all large matrix‑multiplications use cuBLAS via torch.nn.functional.linear /
torch.bmm, which is optimal on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]    (float32)
    y_ptr,          # mutable *[total_rows, dim]    (float16)
    weight_ptr,     # const   *[dim]                (float32)
    bias_ptr,       # const   *[dim]                (float32)
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # cols per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x – μ) / sqrt(σ²+eps)) * γ + β
    Output is fp16; accumulation uses fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer to 2‑D tile
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # [M,D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values γ·(x‑μ)·rstd + β (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) – may be unused
        - weights_dict : dict with the following keys (all float32):
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight",
            "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total rows for row‑wise ops

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) using Triton
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ (fp32)
        weights["norm.bias"],      # β (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into one GEMM (fp16)
    # --------------------------------------------------------------
    H = hidden_dim
    # Concatenate the four weight matrices: [4·H, dim]
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # fp16

    # [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]
    proj_gate = F.linear(x_norm, w_concat)                # fp16

    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Output‑gate (separate GEMM)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16),   # [H, dim]
    )                                                     # fp16

    # --------------------------------------------------------------
    # 3️⃣ Mask (if present) + sigmoid gates + apply gating
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] → flattened [R,1] (fp16)
        mask_flat = mask.to(torch.float16).reshape(R, 1)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_proj_raw * left_gate       # [R, H] fp16
    right = right_proj_raw * right_gate     # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H]
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring head dimension before (N,N) so we can batch‑matmul over heads
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # merge batch & head dims → (B·H, N, N)
    left_mat  = left.view(B * H, N, N)   # fp16
    right_mat = right.view(B * H, N, N)  # fp16

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dim (use torch's fast implementation)
    # --------------------------------------------------------------
    # torch.nn.functional.layer_norm works on the last dimension
    out_norm = F.layer_norm(
        out,
        normalized_shape=(H,),
        weight=weights["to_out_norm.weight"],   # fp32
        bias=weights["to_out_norm.bias"],       # fp32
        eps=eps,
    ).to(torch.float16)                         # fp16

    # --------------------------------------------------------------
    # 6️⃣ Apply output gate + final linear projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate               # elementwise fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )                                                   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
