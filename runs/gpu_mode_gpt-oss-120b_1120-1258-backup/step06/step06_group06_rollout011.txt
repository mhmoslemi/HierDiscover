@triton.jit
def layernorm_proj_kernel(
    x_ptr,           # [R, dim] fp32
    w_ptr,           # [dim, total_o] fp16
    sum_w_ptr,       # [total_o] fp32
    out_ptr,         # [R, total_o] fp16
    total_rows, dim, total_o,
    eps,
    stride_xm, stride_xd,
    stride_wd, stride_wo,
    stride_om, stride_od,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
   pid_m = tl.program_id(0)
   pid_o = tl.program_id(1)

   row_start = pid_m * BLOCK_M
   offs_m = row_start + tl.arange(0, BLOCK_M)
   mask_m = offs_m < total_rows

   out_start = pid_o * BLOCK_O
   offs_o = out_start + tl.arange(0, BLOCK_O)
   mask_o = offs_o < total_o

   # compute per-row mean/var across dim
   sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
   sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
       d_start = d_off * BLOCK_D
       offs_d = d_start + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim
       x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
       x = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

       sum_val += tl.sum(x, axis=1)
       sum_sq += tl.sum(x * x, axis=1)

   mean = sum_val / dim
   var = sum_sq / dim - mean * mean
   rstd = tl.rsqrt(var + eps)   # fp32

   # compute raw dot product for output tile
   # accumulator
   raw = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

   for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
       d_start = d_off * BLOCK_D
       offs_d = d_start + tl.arange(0, BLOCK_D)
       mask_d = offs_d < dim

       # load x slice
       x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
       x_vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float16) # maybe keep fp16

       # load weight slice: w_ptr is dim x total_o (fp16) stored row-major.
       w_ptrs = w_ptr + offs_d[:, None] * stride_wd + offs_o[None, :] * stride_wo
       w_vals = tl.load(w_ptrs, mask=mask_d[:, None] & mask_o[None, :], other=0.0).to(tl.float16)

       # compute dot: x_vals (M, D) * w_vals (D, O) -> (M, O)
       # tl.dot expects shapes (M, K) and (K, N)
       raw += tl.dot(x_vals, w_vals)  # outcome fp32 (or maybe automatic casting)

   # apply layernorm scaling and shifting using sum_w per output
   sum_w = tl.load(sum_w_ptr + offs_o, mask=mask_o, other=0.0)  # fp32, shape [O]
   # broadcast sum_w to M rows
   # compute normalized output: (raw - mean[:,None] * sum_w) * rstd[:,None]
   mean_b = mean[:, None]   # shape [M,1]
   rstd_b = rstd[:, None]
   shift = mean_b * sum_w[None, :] # broadcasting
   normalized = (raw - shift) * rstd_b

   # cast to fp16 and store
   out_ptrs = out_ptr + offs_m[:, None] * stride_om + offs_o[None, :] * stride_od
   tl.store(out_ptrs, normalized.to(tl.float16), mask=mask_m[:, None] & mask_o[None, :])

@triton.jit
def layernorm_proj_kernel(
    x_ptr,                 # const * [total_rows, dim]  float32
    w_ptr,                 # const * [dim, total_o]     fp16 (scaled)
    sum_w_ptr,             # const * [total_o]          fp32
    bias_ptr,              # const * [total_o]          fp32
    out_ptr,               # mutable * [total_rows, total_o] fp16
    total_rows, dim, total_o,
    eps,
    stride_xm, stride_xd,
    stride_wd, stride_wo,
    stride_om, stride_od,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_m = tl.program_id(0)
    pid_o = tl.program_id(1)

    row_start = pid_m * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    out_start = pid_o * BLOCK_O
    offs_o = out_start + tl.arange(0, BLOCK_O)
    mask_o = offs_o < total_o

    # Compute mean/var over dim
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d_off * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
        x = tl.load(x_ptrs,
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1/sqrt(var+eps)

    # Accumulate raw dot product with scaled weights
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d_off * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # Load (M, D) from x
        x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float16)

        # Load (D, O) weight slice
        w_ptrs = w_ptr + offs_d[:, None] * stride_wd + offs_o[None, :] * stride_wo
        w_vals = tl.load(w_ptrs,
                         mask=mask_d[:, None] & mask_o[None, :],
                         other=0.0).to(tl.float16)

        acc += tl.dot(x_vals, w_vals)   # result fp32

    # Load sum_w and bias for O
    sum_w = tl.load(sum_w_ptr + offs_o,
                    mask=mask_o,
                    other=0.0).to(tl.float32)   # [O]
    bias = tl.load(bias_ptr + offs_o,
                    mask=mask_o,
                    other=0.0).to(tl.float32)   # [O]

"""
Highly‑optimized outgoing TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Fuse the first LayerNorm (µ,σ) with the four linear projections
    (left/right + their gates) and the output‑gate projection.
    For each row r of the input (shape [dim]) we compute
        mean, var  →  rstd = 1/√(var+ε)
        raw = Σ_d x_{r,d}·(γ_d·W_{o,d})            (γ = norm.weight)
    The final output of the fused kernel is
        out_{r,o} = (raw_{r,o} – mean_r·Σ_d γ_d·W_{o,d})·rstd_r
                    + β·W_{o}                     (β = norm.bias)

    This replaces five separate GEMMs and a large temporary
    [R,dim] tensor (R = B·N·N) with a single Triton kernel that
    does a per‑row LayerNorm + a large concatenated GEMM.

2️⃣  Apply the optional mask, sigmoid the gate vectors and fuse the
    element‑wise gating (left/right).

3️⃣  Compute the N³ multiplicative update with a single batched
    half‑precision GEMM:
        out = (left @ rightᵀ)        (left,right: [B·H,N,N])

4️⃣  Fuse the second LayerNorm with the output‑gate multiplication
    (still using a small Triton kernel).

5️⃣  Final linear projection back to `dim` (half‑precision, cuBLAS).

All heavy per‑row normalisations are performed in custom Triton kernels,
while the large matrix multiplications rely on Tensor‑Core‑enabled cuBLAS
for maximum throughput.  The implementation works for both masked and
unmasked inputs and supports `dim` ∈ {128,384}, `hidden_dim` = 128,
batch size 1‑2 and sequence lengths up to 1024.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + 5 linear projections
# ----------------------------------------------------------------------
@triton.jit
def layernorm_proj_kernel(
    x_ptr,               # const  * [total_rows, dim]          (float32)
    w_ptr,               # const  * [dim, total_o]            (float16)   scaled by norm.weight
    sum_w_ptr,           # const  * [total_o]                 (float32)   Σ_d γ·W
    bias_ptr,            # const  * [total_o]                 (float32)   β·W
    out_ptr,             # mutable* [total_rows, total_o]    (float16)
    total_rows, dim, total_o,
    eps,
    stride_xm, stride_xd,          # strides of x
    stride_wd, stride_wo,          # strides of w (dim × total_o)
    stride_om, stride_od,          # strides of out
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    BLOCK_O: tl.constexpr,
):
    pid_m = tl.program_id(0)          # row‑tile
    pid_o = tl.program_id(1)          # output‑tile

    row_start = pid_m * BLOCK_M
    out_start = pid_o * BLOCK_O

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_o = out_start + tl.arange(0, BLOCK_O)

    mask_m = offs_m < total_rows
    mask_o = offs_o < total_o

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean / variance   (single pass over dim)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)
    # accumulator for the raw GEMM (scaled weights)
    acc = tl.zeros([BLOCK_M, BLOCK_O], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d_off * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # ---- load x (float32) -------------------------------------------------
        x_ptrs = x_ptr + offs_m[:, None] * stride_xm + offs_d[None, :] * stride_xd
        x_fp32 = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        sum_val += tl.sum(x_fp32, axis=1)
        sum_sq  += tl.sum(x_fp32 * x_fp32, axis=1)

        # ---- also use x as fp16 for the GEMM ---------------------------------
        x_fp16 = x_fp32.to(tl.float16)

        # ---- load the corresponding slice of the scaled weight matrix ----------
        w_ptrs = w_ptr + offs_d[:, None] * stride_wd + offs_o[None, :] * stride_wo
        w_fp16 = tl.load(w_ptrs,
                         mask=mask_d[:, None] & mask_o[None, :],
                         other=0.0).to(tl.float16)

        # ---- GEMM accumulation (M×D)·(D×O) ------------------------------------
        acc += tl.dot(x_fp16, w_fp16)     # fp32 accumulation

    # ------------------------------------------------------------------
    # 2️⃣ Compute normalised output (apply μ,σ and the bias term)
    # ------------------------------------------------------------------
    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var+ε)

    # load Σ_d γ·W   and   β·W   for this output tile
    sum_w = tl.load(sum_w_ptr + offs_o,
                    mask=mask_o,
                    other=0.0).to(tl.float32)   # [BLOCK_O]
    bias  = tl.load(bias_ptr + offs_o,
                    mask=mask_o,
                    other=0.0).to(tl.float32)   # [BLOCK_O]

    # (raw - mean * ΣγW) * rstd + bias
    mean_b = mean[:, None]                # [BLOCK_M,1]
    rstd_b = rstd[:, None]                # [BLOCK_M,1]
    shifted = acc - mean_b * sum_w[None, :]    # [BLOCK_M,BLOCK_O]
    out_fp32 = shifted * rstd_b + bias[None, :]   # [BLOCK_M,BLOCK_O]

    # ------------------------------------------------------------------
    # 3️⃣ Store result (fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + offs_m[:, None] * stride_om + offs_o[None, :] * stride_od
    tl.store(out_ptrs,
             out_fp32.to(tl.float16),
             mask=mask_m[:, None] & mask_o[None, :])



# ----------------------------------------------------------------------
# Triton kernel: LayerNorm + element‑wise multiplication by a gate
# (used for the second LayerNorm + out‑gate)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr, y_ptr,            # input / output  [total_rows, dim] (fp16)
    weight_ptr, bias_ptr,    # LayerNorm γ,β (float32)
    gate_ptr,                # gate tensor, same shape as x (fp16)
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---- mean / variance -------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d_off * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalisation + gate --------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        d_start = d_off * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]
        out = out * g

        out_fp16 = out.to(tl.float16)

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict  : model parameters (see skeleton)
        - config_dict   : must contain "dim", "hidden_dim" and optionally "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim
    total_o = 5 * H                 # left, right, left_gate, right_gate, out_gate

    # --------------------------------------------------------------
    # 1️⃣ Fuse first LayerNorm + five linear projections
    # --------------------------------------------------------------
    # flatten input to [R, dim] (float32)
    x_flat = input_tensor.view(total_rows, dim).contiguous()

    # ---------- prepare concatenated weights ----------
    # original (out, dim) weight tensors (float32)
    w_left  = weights["left_proj.weight"]       # [H, dim]
    w_right = weights["right_proj.weight"]      # [H, dim]
    w_lgate = weights["left_gate.weight"]       # [H, dim]
    w_rgate = weights["right_gate.weight"]      # [H, dim]
    w_outg  = weights["out_gate.weight"]        # [H, dim]

    w_concat = torch.cat([w_left, w_right, w_lgate, w_rgate, w_outg], dim=0)  # [5*H, dim]

    # LayerNorm parameters
    norm_w = weights["norm.weight"]   # γ, shape [dim]
    norm_b = weights["norm.bias"]     # β, shape [dim]

    # ---- scaled weights (γ·W) -------------------------------------------------
    #   w_scaled = γ * W   (broadcast over dim)
    w_scaled = (w_concat * norm_w).to(torch.float16)          # [5*H, dim]  fp16
    w_scaled_T = w_scaled.t().contiguous()                  # [dim, 5*H]  fp16

    # ---- pre‑compute Σ_d (γ·W)  and  β·W  per output -------------------------
    sum_w_scaled = (w_concat * norm_w).sum(dim=1).float()    # [5*H] float32
    bias_term    = (norm_b * w_concat).sum(dim=1).float()   # [5*H] float32

    # allocate output buffer for the five fused projections
    out_concat = torch.empty(total_rows, total_o,
                             dtype=torch.float16, device=device)

    # --------------------------------------------------------------
    # Triton launch parameters
    # --------------------------------------------------------------
    BLOCK_M = 64          # rows per program
    BLOCK_D = 64          # inner‑dim tile
    BLOCK_O = 64          # outputs per program

    grid = (triton.cdiv(total_rows, BLOCK_M),
            triton.cdiv(total_o, BLOCK_O))

    # strides (all tensors are 2‑D C‑contiguous)
    stride_xm = dim
    stride_xd = 1
    stride_wd = total_o
    stride_wo = 1
    stride_om = total_o
    stride_od = 1

    layernorm_proj_kernel[grid](
        x_flat,
        w_scaled_T,
        sum_w_scaled,
        bias_term,
        out_concat,
        total_rows,
        dim,
        total_o,
        eps,
        stride_xm, stride_xd,
        stride_wd, stride_wo,
        stride_om, stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        BLOCK_O=BLOCK_O,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # Split fused output into the five tensors
    # --------------------------------------------------------------
    left_raw      = out_concat[:, 0:H]                # [R, H]
    right_raw     = out_concat[:, H:2*H]               # [R, H]
    left_gate_raw = out_concat[:, 2*H:3*H]             # [R, H]
    right_gate_raw= out_concat[:, 3*H:4*H]             # [R, H]
    out_gate_raw  = out_concat[:, 4*H:5*H]             # [R, H]

    # --------------------------------------------------------------
    # 2️⃣ Apply optional mask and sigmoid gates
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        mask_fp16 = mask_tensor.to(torch.float16).reshape(total_rows, 1)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw  * left_gate     # [R, H] fp16
    right = right_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B,H,N,N]

    left_mat  = left.view(B * H, N, N)       # [B·H, N, N]
    right_mat = right.view(B * H, N, N)      # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]
    out_flat = out.view(total_rows, H)                                 # [R, H]

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm + multiplication by out‑gate (fused)
    # --------------------------------------------------------------
    out_norm_gated = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_norm_gated,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        out_gate.view(total_rows, H),    # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    w_to_out = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_norm_gated, w_to_out)        # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
