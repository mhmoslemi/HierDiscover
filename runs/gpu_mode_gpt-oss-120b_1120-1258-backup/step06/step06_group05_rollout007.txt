"""
custom_kernel – Outgoing TriMul (AlphaFold‑3) forward pass

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input (B·N·N × dim) → fp16.
2️⃣ Fuse the four projection matrices (left/right) and the four gate
   matrices (left‑gate/right‑gate/out‑gate) into a single half‑precision GEMM:
   [R, dim] @ [5·H, dim]ᵀ → [R, 5·H].
3️⃣ Slice the result, apply the optional pairwise mask, and the sigmoid
   gates.  This yields ``left`` and ``right`` tensors of shape [R, H] (fp16).
4️⃣ Reshape to [B, H, N, N] and perform a batched GEMM:
   out_mat = left_mat @ right_matᵀ    (B·H, N, N) × (B·H, N, N)ᵀ
   → out      of shape [B, N, N, H] (fp16).
5️⃣ Row‑wise LayerNorm over the hidden dimension.
6️⃣ Multiply by the output‑gate and a final half‑precision linear
   projection back to ``dim``.
7️⃣ Cast the result to float32 (the public API dtype).

Only the two row‑wise LayerNorms are written in Triton;
all large dense matrix multiplications use cuBLAS via `torch.nn.functional.linear`
and `torch.bmm`, which on an H100 are already near optimal.
The implementation works for any of the test configurations
(e.g. dim∈{128,384}, hidden_dim=128, N up to 1024, B∈{1,2}).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]    (float32)
    y_ptr,          # mutable *[rows, dim]    (float16)
    w_ptr,          # const   *[dim]         (float32)  – γ
    b_ptr,          # const   *[dim]         (float32)  – β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # columns processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ----------- mean & variance (fp32 accumulator) ---------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        # x_ptr[offs_m, offs_d]
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]    # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)        # sum over D
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                # 1 / sqrt(var+eps)

    # ----------- write normalized values (fp16) -------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]    # [M, D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)      # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)      # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool or 0/1 float)
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (all provided as float32 tensors)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]                # input channel dim (e.g. 128 or 384)
    hidden_dim = cfg["hidden_dim"]  # H = 128 (always in the spec)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                         # total (batch * pairwise) rows

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_norm1 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm1](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projections + all gates into ONE linear
    #    Weight matrix shape: [5·H, dim]  (left, right, left_gate,
    #    right_gate, out_gate)
    # ------------------------------------------------------------------
    H = hidden_dim
    # Concatenate the five weight matrices (all are [H, dim] in fp32)
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                 # [5·H, dim] fp16

    # Linear: [R, dim] @ [5·H, dim]ᵀ → [R, 5·H] (half‑precision)
    proj = F.linear(x_norm, w_concat)   # fp16

    # Slice the combined result
    left_raw      = proj[:, 0 * H : 1 * H]          # [R, H]
    right_raw     = proj[:, 1 * H : 2 * H]          # [R, H]
    left_gate_raw = proj[:, 2 * H : 3 * H]          # [R, H]
    right_gate_raw= proj[:, 3 * H : 4 * H]          # [R, H]
    out_gate_raw  = proj[:, 4 * H : 5 * H]          # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates, and element‑wise gating
    # ------------------------------------------------------------------
    if not cfg.get("nomask", True):
        # mask: [B,N,N] → [R,1] (fp16)
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # Sigmoid gates (in‑place for speed)
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Gated projections
    left  = left_raw * left_gate          # [R, H] fp16
    right = right_raw * right_gate        # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] → permute to [B, H, N, N] for bmm
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right= right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]

    # Collapse batch & head dimensions
    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension (fp16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid_norm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm.view(B, N, N, H)   # fp16

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate.view(B, N, N, H)   # fp16 elementwise

    # Final linear: [R, H] @ [dim, H]ᵀ → [R, dim] (still fp16)
    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )  # fp16

    # Cast to the required float32 output
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
