"""
Highly‑optimized forward pass for the **outgoing** TriMul operator
(AlphaFold‑3 style).  

Algorithm
---------
1️⃣ Input LayerNorm – use the fast native PyTorch implementation and
    cast the result to FP16 (the rest of the pipeline works in half‑precision).

2️⃣ Fused projection + gating for the *left* and *right* streams:
   * Concatenate projection‑ and gate‑weight matrices → single `torch.nn.functional.linear`.
   * Split the result, apply a sigmoid on the gate half and multiply.
   * Optional binary mask is applied after the gates (still in FP16).

3️⃣ Output‑gate:
   * Linear on the (already) normalised tensor → sigmoid.

4️⃣ Triangular multiplicative update (batched GEMM):
   * `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
   * Implemented as a single batched FP16 matmul:  
     `left  : (B·H, N, N) ,  right : (B·H, N, N)  →  out : (B·H, N, N)`.

5️⃣ Fused hidden‑dim LayerNorm + output‑gate (custom Triton kernel).

6️⃣ Final projection back to the model dimension (`dim`) – a regular FP16 GEMM.

All heavy tensor‑core matrix operations stay in half‑precision,
the only Triton kernels are the two LayerNorms (input norm handled by
PyTorch, hidden‑dim norm + gate fused in one kernel).  This reduces
kernel‑launch overhead while keeping the memory‑bound parts fused.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Kernel 1 – fused hidden‑dim LayerNorm + output‑gate (FP16)
# ----------------------------------------------------------------------
@triton.jit
def fused_layernorm_gate_kernel(
    x_ptr,          # const half*  [total_rows, hidden]
    y_ptr,          # mut   half*  [total_rows, hidden]
    weight_ptr,    # const float* [hidden]   (γ)
    bias_ptr,      # const float* [hidden]   (β)
    gate_ptr,      # const half*  [total_rows, hidden]   (sigmoid(out_gate))
    total_rows: tl.int32,
    hidden: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / hidden
    var  = sum_sq / hidden - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1/√(var+eps)

    # ---------- normalise, apply γ/β, multiply by gate ----------
    for d in range(0, tl.cdiv(hidden, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < hidden
        ptr = x_ptr + offs_m[:, None] * hidden + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]

        # gate (already sigmoid’ed)
        gate = tl.load(gate_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out = out * gate
        tl.store(y_ptr + offs_m[:, None] * hidden + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (inp, mask, weights, cfg)
        - inp    : torch.Tensor [B, N, N, C]   (float32)
        - mask   : torch.Tensor [B, N, N]      (float32/bool) – ignored if cfg["nomask"]=True
        - weights: dict of named parameters (float32)
        - cfg    : {"dim":C, "hidden_dim":H, "nomask":bool, "eps":float (opt.)}

    Returns
    -------
    torch.Tensor [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dim     = cfg["dim"]                     # C
    hidden  = cfg["hidden_dim"]              # H
    eps     = cfg.get("eps", 1e-5)
    nomask  = cfg.get("nomask", True)

    B, N, _, _ = inp.shape
    total_rows = B * N * N                      # rows after flattening

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (float32 → fp16) – leverage torch's fast kernel
    # ------------------------------------------------------------------
    # layer_norm returns float32; we immediately cast to half to keep
    # everything in low precision for the rest of the pipeline.
    x_norm_fp16 = F.layer_norm(
        inp,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).half()                                   # [B,N,N,dim] → half

    # flatten to (R, dim) for the linear layers
    x_norm_fp16 = x_norm_fp16.view(total_rows, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + gating for *left* and *right*
    # ------------------------------------------------------------------
    # Cast all projection / gate weight matrices to fp16 once
    left_proj_w  = weights["left_proj.weight"].half()
    left_gate_w  = weights["left_gate.weight"].half()
    right_proj_w = weights["right_proj.weight"].half()
    right_gate_w = weights["right_gate.weight"].half()
    out_gate_w   = weights["out_gate.weight"].half()

    # ---- left side ---------------------------------------------------
    left_comb_w = torch.cat([left_proj_w, left_gate_w], dim=0)   # (2H, C)
    left_comb   = F.linear(x_norm_fp16, left_comb_w)            # (R, 2H)
    left_proj_raw, left_gate_raw = torch.split(left_comb, hidden, dim=1)
    left_gate = torch.sigmoid(left_gate_raw)                    # (R, H)
    left = left_proj_raw * left_gate                            # (R, H)

    # ---- right side --------------------------------------------------
    right_comb_w = torch.cat([right_proj_w, right_gate_w], dim=0)   # (2H, C)
    right_comb   = F.linear(x_norm_fp16, right_comb_w)              # (R, 2H)
    right_proj_raw, right_gate_raw = torch.split(right_comb, hidden, dim=1)
    right_gate = torch.sigmoid(right_gate_raw)                      # (R, H)
    right = right_proj_raw * right_gate                            # (R, H)

    # ---- optional binary mask ----------------------------------------
    if not nomask:
        # mask: (B,N,N) → (R,1) broadcast on hidden dim
        mask_flat = mask.view(total_rows, 1).to(torch.float16)
        left  = left  * mask_flat
        right = right * mask_flat

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H)
    left  = left.view(B, N, N, hidden)
    right = right.view(B, N, N, hidden)

    # (B·H, N, N) with the proper transpose on the right operand
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * hidden, N, N)   # (i,k)
    right_mat = right.permute(0, 3, 2, 1).reshape(B * hidden, N, N)  # (k,j)

    # Batched half‑precision matmul → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat)

    # back to (B, N, N, H)
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Output gate (layer‑norm → sigmoid)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(x_norm_fp16, out_gate_w)      # (R, H) half
    out_gate = torch.sigmoid(out_gate_raw)                # (R, H) half

    # ------------------------------------------------------------------
    # 5️⃣ Fused hidden‑dim LayerNorm + out‑gate (Triton)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, hidden).contiguous()   # (R, H) half
    out_norm_gate = torch.empty_like(out_flat)             # (R, H) half

    # Choose block sizes that fit hidden ≤ 384
    BLOCK_M = 256
    BLOCK_D = 32 if hidden <= 128 else 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    fused_layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate,                         # already sigmoid’ed, half
        total_rows,
        hidden,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (FP16 GEMM)
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].half()            # (dim, H)
    final_fp16 = F.linear(out_norm_gate, to_out_w)        # (R, dim) half
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
