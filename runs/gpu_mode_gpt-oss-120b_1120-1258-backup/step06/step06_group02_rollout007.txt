"""
fast_outgoing_trimul.py

Implementation of the **outgoing** TriMul operator (AlphaFold‑3) that
* applies a fused row‑wise LayerNorm in fp16 (custom Triton kernel)
* fuses the four linear projections (left/right proj + left/right gate)
  into a single GEMM
* computes the cubic‑time multiplicative update with a batched fp16
  matrix‑multiply (torch.bmm) – far faster than the reference einsum
* applies a second fused LayerNorm and the final output projection.

All heavy arithmetic stays in fp16 so that Tensor‑Cores on an H100 are
fully utilised.  The public entry point is `custom_kernel(data)` whose
signature matches the specification.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # cols processed per inner loop
):
    """
    Compute per‑row mean/variance in fp32 and write the normalised result
    in fp16:
        y = ((x - μ) / sqrt(σ²+eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute sum and sum‑of‑squares for mean/var (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim

        # load a tile (BLOCK_M x BLOCK_D)
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Normalise and apply γ,β (fp16 output)
    # ------------------------------------------------------------------
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # fp32 load

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        # γ (weight) and β (bias) are fp32 → keep computation in fp32
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]               # fp32

        # store as fp16
        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool or float)
        - weights_dict : model weights (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack & basic shapes
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N                # R = B·N·N
    H = hidden_dim                        # shortcut

    # --------------------------------------------------------------
    # 1️⃣ first LayerNorm (row‑wise, fp16)
    # --------------------------------------------------------------
    # flatten to [R, dim] (contiguous fp32)
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    # Triton launch parameters – good for dim up to 384
    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ  (float32)
        weights["norm.bias"],            # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # --------------------------------------------------------------
    # 2️⃣ fused projection + gate linear (one GEMM)
    # --------------------------------------------------------------
    # concat the four weight matrices (shape: [4*H, dim])
    w_proj_gate = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [4*H, dim]

    proj_gate = F.linear(x_norm, w_proj_gate)   # [R, 4*H] fp16

    # split the result
    left_proj_raw   = proj_gate[:, 0:H]
    right_proj_raw  = proj_gate[:, H:2*H]
    left_gate_raw   = proj_gate[:, 2*H:3*H]
    right_gate_raw  = proj_gate[:, 3*H:4*H]

    # output‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                            # [R, H] fp16

    # --------------------------------------------------------------
    # 3️⃣ optional mask, sigmoid gates and gating
    # --------------------------------------------------------------
    if not config.get("nomask", True):
        # mask is broadcast over the hidden dimension
        mask_flat = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    # inplace sigmoid for the three gate tensors (fp16)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gates
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 4️⃣ reshape to 4‑D tensors
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ triangular multiplicative update (batched GEMM)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    → out = left @ rightᵀ   (per head)
    # --------------------------------------------------------------
    # move hidden dim to batch dimension: (B, H, N, N)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # reshape to a single batch dimension for torch.bmm
    left_mat  = left_perm.view(B * H, N, N)               # [B*H, N, N]
    right_mat = right_perm.view(B * H, N, N)              # [B*H, N, N]

    # batched GEMM: left @ rightᵀ
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B*H, N, N]

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ second LayerNorm over hidden dimension (row‑wise)
    # --------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 7️⃣ apply output gate and final projection back to `dim`
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate                # fp16
    # final linear: hidden_dim -> dim
    final = F.linear(
        out_norm.reshape(total_rows, H),
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] fp16
    final = final.reshape(B, N, N, dim).to(torch.float32)

    return final
```
