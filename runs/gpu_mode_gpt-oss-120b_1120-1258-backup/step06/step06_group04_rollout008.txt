    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

"""
TriMul (outgoing) – fast Triton implementation.
================================================

The kernel pipeline follows the AlphaFold‑3 TriMul forward pass:

1️⃣  Row‑wise LayerNorm of the input 4‑D tensor  [B, N, N, C]  (FP32 → FP16).
2️⃣  Fuse proj+gate for *left* and *right* branches:
        (C → H) + (C → H) → (C → 2·H) via a single `torch.nn.functional.linear`.
3️⃣  Optional masking (broadcast over H).
4️⃣  Apply sigmoid‑gates to the projected tensors.
5️⃣  Tri‑multiplicative update  out[i,j] = Σₖ left[i,k]·right[j,k]   (batched GEMM).
6️⃣  **Fused** LayerNorm (over hidden dim) + output‑gate + final linear
    (hidden → C) in a single Triton kernel:
        out_norm = LayerNorm(out)                # γ·(x‑μ)/σ+β
        gated   = out_norm * out_gate
        out     = gated @ W_outᵀ                 # no bias in the reference model
   The kernel both normalises each row and performs the final matrix‑multiply,
   saving two full‑tensor passes.

All heavy arithmetic is performed in FP16, the final result is cast back
to FP32 to match the reference implementation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Kernel 1: row‑wise LayerNorm (FP32 input → FP16 output)
# --------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [rows, dim]    (input, FP32)
    y_ptr,          # half*    [rows, dim]    (output, FP16)
    gamma_ptr,      # float*   [dim]          (scale)
    beta_ptr,       # float*   [dim]          (bias)
    rows,           # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- Pass 1: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_D]
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- Pass 2: write normalized values ----------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# --------------------------------------------------------------
# Kernel 2: fused LayerNorm (hidden) + output‑gate + final linear
# --------------------------------------------------------------
@triton.jit
def fused_norm_gate_linear_kernel(
    inp_ptr,           # half*   [rows, K]               (raw out)
    gate_ptr,          # half*   [rows, K]               (out_gate)
    gamma_ptr,         # half*   [K]                     (to_out_norm.weight)
    beta_ptr,          # half*   [K]                     (to_out_norm.bias)
    weight_T_ptr,      # half*   [K, N]                  (to_out.weight^T)
    bias_ptr,          # half*   [N]                     (to_out.bias – zero if not used)
    out_ptr,           # half*   [rows, N]               (final result)
    rows, K, N,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_K: tl.constexpr,   # reduction dim tile
    BLOCK_N: tl.constexpr,   # output feature tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- mean / var over K (first pass) ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K
        inp_tile = tl.load(
            inp_ptr + offs_m[:, None] * K + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [BLOCK_M, BLOCK_K]
        sum_val += tl.sum(inp_tile, axis=1)
        sum_sq  += tl.sum(inp_tile * inp_tile, axis=1)

    mean = sum_val / K
    var  = sum_sq / K - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: normalise, apply gamma/beta, gating, GEMM ----
    # accumulator for output block (M x N)
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load raw tensor and gate
        inp_tile = tl.load(
            inp_ptr + offs_m[:, None] * K + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]
        gate_tile = tl.load(
            gate_ptr + offs_m[:, None] * K + offs_k[None, :],
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        # layer‑norm (γ·x̂ + β)
        norm = (inp_tile - mean[:, None]) * rstd[:, None]

        gamma = tl.load(gamma_ptr + offs_k, mask=mask_k, other=0.0).to(tl.float32)
        beta  = tl.load(beta_ptr  + offs_k, mask=mask_k, other=0.0).to(tl.float32)

        normed = norm * gamma[None, :] + beta[None, :]  # [M, K]

        # apply output gate
        gated = normed * gate_tile                       # [M, K]

        # now accumulate GEMM with weight_T (K x N)
        for nn in range(0, tl.cdiv(N, BLOCK_N)):
            offs_n = nn * BLOCK_N + tl.arange(0, BLOCK_N)
            mask_n = offs_n < N

            w_tile = tl.load(
                weight_T_ptr + offs_k[:, None] * N + offs_n[None, :],
                mask=mask_k[:, None] & mask_n[None, :],
                other=0.0,
            ).to(tl.float32)                     # [K, N]

            acc += tl.dot(gated, w_tile)          # (M,N) += (M,K)·(K,N)

    # add bias (if any)
    for nn in range(0, tl.cdiv(N, BLOCK_N)):
        offs_n = nn * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        b = tl.load(bias_ptr + offs_n, mask=mask_n, other=0.0).to(tl.float32)
        acc[:, :] += b[None, :]                     # broadcast over M

    # write result
    for nn in range(0, tl.cdiv(N, BLOCK_N)):
        offs_n = nn * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_n = offs_n < N
        out_tile_ptr = out_ptr + offs_m[:, None] * N + offs_n[None, :]
        tl.store(out_tile_ptr, acc.to(tl.float16),
                 mask=mask_m[:, None] & mask_n[None, :])


# --------------------------------------------------------------
# Entry point
# --------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the TriMul “outgoing” module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor[float32]   shape [B, N, N, C]
        - mask_tensor  : torch.Tensor[float32]   shape [B, N, N] (may be ignored)
        - weights_dict : dict mapping weight names to torch.Tensors
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor[float32] shape [B, N, N, C]
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    rows = B * N * N
    x_flat = x.view(rows, dim).contiguous()  # FP32
    x_norm_fp16 = torch.empty((rows, dim), dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 128
    grid = (triton.cdiv(rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    x_norm = x_norm_fp16.view(B, N, N, dim)          # FP16

    # --------------------------------------------------------------
    # 2️⃣ Fuse proj+gate for left/right branches
    # --------------------------------------------------------------
    # concatenate proj & gate weights (2·H × dim)
    left_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)
    right_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    rows_flat = rows
    # Linear: (rows, dim) @ (2H, dim).T  → (rows, 2H)
    left_concat = F.linear(x_norm.view(rows_flat, dim), left_w)   # FP16
    right_concat = F.linear(x_norm.view(rows_flat, dim), right_w)

    # split to proj / pre‑gate parts
    left_proj = left_concat[:, :hidden]          # (rows, H)
    left_gate_pre = left_concat[:, hidden:]      # (rows, H)
    right_proj = right_concat[:, :hidden]
    right_gate_pre = right_concat[:, hidden:]

    # Output gate (shared across i,j)
    out_gate_w = weights["out_gate.weight"].to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    out_gate = torch.sigmoid(
        F.linear(x_norm.view(rows_flat, dim), out_gate_w)
    )   # (rows, H)

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over H)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(rows_flat, 1).to(
            dtype=torch.float16, device=device, non_blocking=True
        )
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h = left_proj * left_gate          # (rows, H)
    right_h = right_proj * right_gate       # (rows, H)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via batched GEMM
    # --------------------------------------------------------------
    left = left_h.view(B, N, N, hidden)      # (B,N,N,H)
    right = right_h.view(B, N, N, hidden)

    # reshape to [B·H, N, N] for a single batched GEMM
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)

    # out = left_mat @ right_matᵀ    (shape [B·H, N, N])
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # FP16
    out = (
        out_mat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )   # (B, N, N, H)   FP16

    # --------------------------------------------------------------
    # 6️⃣ Fused LayerNorm (hidden) + out‑gate + final linear
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, hidden).contiguous()
    out_gate_flat = out_gate.view(rows_h, hidden).contiguous()

    # Prepare weights for fused kernel (all in FP16)
    to_out_norm_w = weights["to_out_norm.weight"].to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    to_out_norm_b = weights["to_out_norm.bias"].to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    # final linear weight transposed for easier GEMM (K×N)
    to_out_w_T = weights["to_out.weight"].t().contiguous().to(
        dtype=torch.float16, device=device, non_blocking=True
    )
    # Bias of the final linear – the reference model uses bias=False,
    # but we keep a placeholder for completeness.
    to_out_b = torch.zeros(dim, dtype=torch.float16, device=device)

    out_fp16 = torch.empty((rows_h, dim), dtype=torch.float16, device=device)

    # tile sizes – chosen to keep shared memory < 64 KB on H100
    BLOCK_M = 128
    BLOCK_K = 32
    BLOCK_N = 64
    grid = (triton.cdiv(rows_h, BLOCK_M),)
    fused_norm_gate_linear_kernel[grid](
        out_flat,
        out_gate_flat,
        to_out_norm_w,
        to_out_norm_b,
        to_out_w_T,
        to_out_b,
        out_fp16,
        rows_h,
        hidden,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_K=BLOCK_K,
        BLOCK_N=BLOCK_N,
    )

    # --------------------------------------------------------------
    # 7️⃣ Cast back to FP32
    # --------------------------------------------------------------
    final = out_fp16.view(B, N, N, dim).to(dtype=torch.float32)
    return final
