```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def layernorm_fp16_kernel(x_ptr, y_ptr, gamma_ptr, beta_ptr,
                          total_rows, dim: tl.constexpr, eps: tl.constexpr,
                          BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # first pass: compute mean and var per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_iters = tl.cdiv(dim, BLOCK_D)
    for i in range(num_iters):
        offs_d = i * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: write normalized values
    for i in range(num_iters):
        offs_d = i * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(x_ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]  # FP32
        out_fp16 = out.to(tl.float16)
        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16, mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the “outgoing” TriMul operator.

    Algorithm
    ---------
    1. Row‑wise LayerNorm over the channel dimension (`dim`) using a
       custom Triton kernel, output in FP16.
    2. Fuse *all* linear projections (left / right projections, their
       gates and the output gate) into a single GEMM: one matrix‑multiply
       of shape  [total_rows, dim] × [(5·hidden), dim]^T.
       The result is split into the five required tensors.
    3. Optional pairwise mask applied to the projected values.
    4. Apply sigmoid gates and obtain ``left`` and ``right`` tensors.
    5. Perform the tri‑multiplicative update by reshaping the tensors
       to [B·hidden, N, N] and running a batched FP16 matrix‑multiply
       (torch.bmm) – equivalent to the einsum in the reference code.
    6. Second LayerNorm (over the hidden dimension) via the same Triton
       kernel.
    7. Multiply by the output gate, apply the final linear projection
       (hidden → dim) and cast the result back to float32.

    All heavy arithmetic is performed in FP16; the final result matches
    the reference implementation (float32).
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    total_rows = B * N * N
    # Flatten to (total_rows, dim) – assume contiguous layout
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Choose block sizes, fine‑tuned for dim ≤ 384
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # (B, N, N, dim) in fp16
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse all linear projections + output gate into a single GEMM
    # --------------------------------------------------------------
    rows = total_rows
    # Concatenate the five weight matrices: Lproj, Lgate, Rproj, Rgate, OutGate
    w_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)

    # One fused linear pass
    fused_out = F.linear(x_norm.view(rows, dim), w_all)  # (rows, 5*hidden)
    # Split
    H = hidden_dim
    left_proj = fused_out[:, 0:H]
    left_gate_pre = fused_out[:, H:2*H]
    right_proj = fused_out[:, 2*H:3*H]
    right_gate_pre = fused_out[:, 3*H:4*H]
    out_gate_pre = fused_out[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [rows, 1]
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16, device=device, non_blocking=True)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    left_h = left_proj * left_gate      # (rows, H)
    right_h = right_proj * right_gate   # (rows, H)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left = left_h.view(B, N, N, H)
    right = right_h.view(B, N, N, H)

    # Merge the hidden dimension into the batch axis → [B·H, N, N]
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # GEMM: out[i] = left[i] @ right[i].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # FP16, shape [B·H, N, N]

    # Back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden_dim) → FP16
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate, final projection and cast back to FP32
    # --------------------------------------------------------------

# -*- coding: utf-8 -*-

"""
TriMul “outgoing” forward pass (AlphaFold‑3) – Triton‑accelerated.

Key ideas
---------
1️⃣  Row‑wise LayerNorm (FP32 → FP16) via a tiny Triton kernel.
2️⃣  Fuse **all** linear maps (left/right projections, their gates and the
    output gate) into a *single* GEMM.  This cuts the number of large
    matrix‑multiplications from three to one and dramatically reduces
    memory traffic (the large input tensor is read only once).
3️⃣  Optional mask is applied after the projection step (as in the reference).
4️⃣  Left/right tensors are reshaped to [B·H, N, N] and the tri‑multiplicative
    update is realised as a batched FP16 matrix‑multiply (`torch.bmm`), i.e.
    ``out[i] = left[i] @ right[i].T``.
5️⃣  A second LayerNorm over the hidden dimension (again Triton).
6️⃣  Multiply by the output gate, apply the final linear projection
    (hidden → dim) and cast back to ``float32``.  All heavy arithmetic
    stays in FP16, while the result matches the PyTorch reference.

The implementation works for all shapes listed in the prompt and
requires only the Triton kernel for LayerNorm – the rest leverages
highly‑optimised CuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # float*   [total_rows, dim]   (input)
    y_ptr,          # half*    [total_rows, dim]   (output)
    gamma_ptr,      # float*   [dim]               (scale)
    beta_ptr,       # float*   [dim]               (bias)
    total_rows,     # int
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per block
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)         # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣  Compute mean and variance for each row (pass 1)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_iters = tl.cdiv(dim, BLOCK_D)
    for d in range(num_d_iters):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # sum over D
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣  Write normalized values (γ·x̂+β) (pass 2)
    # ------------------------------------------------------------------
    for d in range(num_d_iters):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x_tile_ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_tile_ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]    # (x-μ)/σ

        w = tl.load(gamma_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(beta_ptr  + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]            # still FP32
        out_fp16 = out.to(tl.float16)

        y_tile_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_tile_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the “outgoing” TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor[float32] shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor[float32] shape [B, N, N]
                          (may be ignored when cfg[\"nomask\"] is True)
        - weights_dict : mapping of all model weights (float32)
        - config_dict  : {
                "dim"        : int,
                "hidden_dim" : int,
                "nomask"     : bool (optional, default True)
          }

    Returns
    -------
    torch.Tensor[float32] : shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    device = x.device
    B, N, _, dim = x.shape
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    nomask = cfg.get("nomask", True)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (FP32 → FP16) using Triton
    # --------------------------------------------------------------
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(
        (total_rows, dim), dtype=torch.float16, device=device
    )

    # Tuned block sizes: dim ≤ 384 ⇒ at most 3 iterations over D
    BLOCK_M = 256          # #rows processed per program
    BLOCK_D = 128          # #cols processed per iteration
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # (B, N, N, dim) in fp16 – ready for the remaining ops
    x_norm = x_norm_fp16.view(B, N, N, dim)

    # --------------------------------------------------------------
    # 2️⃣ Fuse *all* linear maps (proj + gate + out‑gate) into one GEMM
    # --------------------------------------------------------------
    rows = total_rows
    # Concatenate the five weight matrices along the output‑feature axis
    w_all = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
            weights["out_gate.weight"],    # (H, dim)
        ],
        dim=0,
    ).to(dtype=torch.float16, device=device, non_blocking=True)  # (5·H, dim)

    # One large Linear pass → (rows, 5·H)
    fused_out = F.linear(x_norm.view(rows, dim), w_all)

    H = hidden_dim
    left_proj      = fused_out[:, 0:H]
    left_gate_pre  = fused_out[:, H:2*H]
    right_proj     = fused_out[:, 2*H:3*H]
    right_gate_pre = fused_out[:, 3*H:4*H]
    out_gate_pre   = fused_out[:, 4*H:5*H]

    # --------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(rows, 1).to(dtype=torch.float16, device=device, non_blocking=True)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 4️⃣ Apply sigmoid gates
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)

    left_h  = left_proj * left_gate      # (rows, H)
    right_h = right_proj * right_gate    # (rows, H)

    # --------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via batched GEMM
    # --------------------------------------------------------------
    left = left_h.view(B, N, N, H)      # (B, N, N, H)
    right = right_h.view(B, N, N, H)

    # Merge hidden dim into batch dim → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # out_mat[i] = left_mat[i] @ right_mat[i].T   (FP16)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # FP16

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm (over hidden) – Triton again
    # --------------------------------------------------------------
    rows_h = B * N * N
    out_flat = out.view(rows_h, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # FP16

    # --------------------------------------------------------------
    # 7️⃣ Apply output gate, final projection & cast back to FP32
    # --------------------------------------------------------------
    gated = out_norm * out_gate.view(B, N, N, H)   # FP16

    final_fp16 = F.linear(
        gated.view(rows_h, H),
        weights["to_out.weight"].to(dtype=torch.float16, device=device, non_blocking=True)
    )   # (rows_h, dim)  FP16

    final = final_fp16.view(B, N, N, dim).to(dtype=torch.float32)

    return final
