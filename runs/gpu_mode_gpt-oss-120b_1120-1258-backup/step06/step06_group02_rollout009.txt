"""
TriMul (outgoing) forward pass – highly‑optimized implementation.

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] (C = dim).
1. Row‑wise LayerNorm (fp32 → fp16) is performed by a custom Triton kernel.
2. Four linear projections (left/right projection + left/right gating) are
   fused into a single fp16 GEMM.
3. An optional pairwise mask is applied to the projections before the
   sigmoid gates.
4. The cubic “tri‑multiplicative” update  
      out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]  
   is realised as a batched fp16 matrix multiplication:
      out[b,:, :,h] = left[b,:, :,h] @ right[b,:, :,h].T
5. A second fp16 LayerNorm (over the hidden dimension) is applied by the
   same Triton kernel, followed by element‑wise gating.
6. The final linear projection back to `dim` is performed with a single
   fp16 GEMM; the result is cast to float32 to match the reference model.

Only the two LayerNorms are written in Triton; the remaining work uses
high‑throughput torch operations (fp16 GEMMs, matmul) that fully exploit
Tensor‑Core acceleration on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]               (float32)
    bias_ptr,       # const   * [dim]               (float32)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Per‑row (row = total_rows) mean/variance in fp32, write normalized
    result in fp16:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ compute sum and sum‑of‑squares for mean/var (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d_blocks = tl.cdiv(dim, BLOCK_D)
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a block of the row
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)           # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized values (fp16)
    # ------------------------------------------------------------------
    for d_off in range(num_d_blocks):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β

        out = norm * w[None, :] + b[None, :]   # still fp32
        out_fp16 = out.to(tl.float16)

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor of shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor of shape [B, N, N] (bool/float) or None
        - weights_dict : dict with the following keys (all float32):
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight"
            * "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : must contain
            * "dim" (int)
            * "hidden_dim" (int)
            * "nomask" (bool, optional, default=True)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = input_tensor.device

    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N               # R = B·N·N
    H = hidden_dim                       # hidden dimension

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    # Flatten to [R, dim] (float32) and allocate fp16 buffer for the result
    x_flat = input_tensor.reshape(total_rows, dim).contiguous()
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 128          # dims per block (covers dim ≤ 384)
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],          # γ (fp32)
        weights["norm.bias"],            # β (fp32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection & gate linear maps (fp16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate weight matrices for left/right proj + left/right gate
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                 # [4*H, dim]

    proj_gate = F.linear(x_norm, w_4h)  # [R, 4*H] fp16

    # Slice the concatenated result
    left_proj_raw  = proj_gate[:, 0:H]
    right_proj_raw = proj_gate[:, H:2 * H]
    left_gate_raw  = proj_gate[:, 2 * H:3 * H]
    right_gate_raw = proj_gate[:, 3 * H:4 * H]

    # Separate out_gate linear map
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim] → fp16
    )                                                   # [R, H] fp16

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → [R, 1] and fp16
        mask_h = mask_tensor.reshape(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # In‑place sigmoid for the three gate tensors
    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate    # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Reshape to 4‑D tensors [B, N, N, H]
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, H)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update via batched fp16 matmul
    # ------------------------------------------------------------------
    #   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #   ⇔ out[b,:, :,h] = left[b,:, :,h] @ right[b,:, :,h].T
    left_4d  = left.permute(0, 3, 1, 2)    # [B, H, N, K]  (K == N)
    right_4d = right.permute(0, 3, 2, 1)   # [B, H, K, N]  (K == N, transposed)
    out_4d   = torch.matmul(left_4d, right_4d)   # [B, H, N, N] fp16
    out      = out_4d.permute(0, 2, 3, 1)        # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over hidden dim H → fp16
    # ------------------------------------------------------------------
    out_flat = out.reshape(total_rows, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    out_norm = out_norm.view(B, N, N, H)          # [B, N, N, H] fp16

    # ------------------------------------------------------------------
    # 7️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate                 # element‑wise gate (fp16)
    final = F.linear(
        out_norm.reshape(total_rows, H),
        weights["to_out.weight"].to(torch.float16)  # [dim, H] → fp16
    )                                                # [R, dim] fp16

    final = final.reshape(B, N, N, dim).to(torch.float32)

    return final
