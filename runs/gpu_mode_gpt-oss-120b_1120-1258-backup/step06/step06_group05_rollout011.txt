"""
Fast outgoing TriMul (AlphaFold‑3) forward pass – Triton + PyTorch.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16) – Triton kernel.
2️⃣ Fuse *all* linear projections/gates (left/right + left/right‑gate + out‑gate)
   into a single half‑precision GEMM:
        [R, dim] @ [5·H, dim]ᵀ → [R, 5·H]   (R = B·N·N, H = hidden_dim)
3️⃣ Split the result, apply the optional pair‑wise mask, sigmoid‑gates and
   obtain gated tensors `left` and `right` (fp16).
4️⃣ Batched GEMM for the multiplicative update:
        (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   (Tensor‑Core matmul)
5️⃣ Row‑wise LayerNorm over the hidden dimension – Triton kernel.
6️⃣ Multiply by the output‑gate, project back to `dim` with a final
   half‑precision linear layer, and return fp32.

Only the two LayerNorms are implemented in Triton; all large matrix
multiplications use cuBLAS via `torch.nn.functional.linear` / `torch.bmm`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[total_rows, dim]    (float32)
    y_ptr,          # mutable *[total_rows, dim]    (float16)
    weight_ptr,     # const   *[dim]                (float32)
    bias_ptr,       # const   *[dim]                (float32)
    total_rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # cols processed per iteration
):
    """
    Row‑wise LayerNorm:
        y = γ·(x−μ)/√(σ²+ε) + β
    The output is fp16; accumulation is fp32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------ 1️⃣ mean & variance ---------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------ 2️⃣ write normalized values -------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr,
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float)
        - weights_dict : mapping with all Linear/LayerNorm parameters
                         (float32 tensors)
        - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                       # total rows for the row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16) – Triton
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 1024               # rows per program (tune for occupancy)
    BLOCK_D = 64 if dim >= 64 else 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse ALL linear projections & gates into ONE GEMM
    # ------------------------------------------------------------------
    # concatenate weight matrices: [left_proj, right_proj, left_gate,
    # right_gate, out_gate]   → shape (5·H, dim)
    w_all = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # fp16 weight matrix

    # (R, dim) @ (5·H, dim)ᵀ → (R, 5·H)
    proj_all = F.linear(x_norm, w_all)        # fp16

    # split into the five parts
    left_proj_raw  = proj_all[:, 0:H]
    right_proj_raw = proj_all[:, H:2 * H]
    left_gate_raw  = proj_all[:, 2 * H:3 * H]
    right_gate_raw = proj_all[:, 3 * H:4 * H]
    out_gate_raw   = proj_all[:, 4 * H:5 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply optional mask, sigmoid gates and gating
    # ------------------------------------------------------------------
    if not nomask:
        mask_flat = mask.view(R, 1).to(torch.float16)   # [R,1] fp16
        left_proj_raw = left_proj_raw * mask_flat
        right_proj_raw = right_proj_raw * mask_flat

    left_gate = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate = out_gate_raw.sigmoid_()

    left = left_proj_raw * left_gate      # [R, H] fp16
    right = right_proj_raw * right_gate   # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left.view(B, N, N, H)          # [B, N, N, H]
    right = right.view(B, N, N, H)        # [B, N, N, H]

    # bring the head dimension into the batch dimension
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # [B·H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   – Tensor‑Core half‑matmul
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape to (R, H) for the second LayerNorm
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous().view(R, H)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over hidden dimension – Triton
    # ------------------------------------------------------------------
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output gate and final projection back to `dim`
    # ------------------------------------------------------------------
    out_norm = out_norm.view(B, N, N, H)            # [B,N,N,H] fp16
    out_norm = out_norm * out_gate.view(B, N, N, H) # gated, fp16

    final_half = F.linear(
        out_norm.view(R, H),
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )  # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
