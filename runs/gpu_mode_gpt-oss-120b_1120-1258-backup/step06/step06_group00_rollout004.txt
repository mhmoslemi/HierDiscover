"""
TriMul (outgoing) forward implementation with Triton LayerNorm.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (float32 → float16).
2️⃣ Fuse the four linear projections (left/right + their gates) into a single
   GEMM (R × dim) @ (4·H × dim)ᵀ.  Split the result into left/right and the two
   gate tensors.
3️⃣ Compute the output‑gate with a second GEMM.
4️⃣ (Optional) Apply the mask to the left/right projections.
5️⃣ Apply sigmoid to the three gate tensors and fuse the gates:
       left  = left_proj  * sigmoid(left_gate)
       right = right_proj * sigmoid(right_gate)
6️⃣ Reshape to (B, N, N, H) and perform the triangular multiplicative update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched fp16 GEMM
       (B·H, N, N) @ (B·H, N, N)ᵀ  via torch.bmm (cuBLAS/Tensor‑Cores).
7️⃣ Second LayerNorm over the hidden dimension (fp16) using the same Triton
   kernel.
8️⃣ Fuse the element‑wise output‑gate multiplication with the final
   linear projection:
       out = ((out_norm * out_gate) @ to_out.weightᵀ)
   (fp16 GEMM) and cast the result back to float32.

All heavy element‑wise work (layer‑norm) is done with Triton; the large
matrix‑multiplications use cuBLAS which is optimal on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,            # const   * [total_rows, dim]   (float32)
    y_ptr,            # mutable * [total_rows, dim]   (float16)
    weight_ptr,       # const   * [dim]               (float32) γ
    bias_ptr,         # const   * [dim]               (float32) β
    total_rows,       # int (runtime)
    dim,              # int (runtime)
    eps,              # float (compile‑time constexpr)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    """
    Row‑wise LayerNorm:
        y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute per‑row mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M, D]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Write normalized values γ·(x−μ)·rstd + β  (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β
        out = norm * w[None, :] + b[None, :]                     # fp32

        out_fp16 = out.to(tl.float16)
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Helper: flatten batch‑seq dims to a single dimension
# ----------------------------------------------------------------------
def _flatten_batch_seq(x):
    """Flatten the first three dims (B,N,N) → R."""
    B, N, _, C = x.shape
    return x.view(B * N * N, C)

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        weights_dict : dict with all linear / layer‑norm parameters
        config_dict  : {"dim": int,
                        "hidden_dim": int,
                        "nomask": bool (optional, default True)}
    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    x, mask, weights, config = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # ------------------------------------------------------------------
    x_flat = _flatten_batch_seq(x).contiguous()               # [R, dim] fp32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ (float32)
        weights["norm.bias"],        # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates (4·H outputs)
    # ------------------------------------------------------------------
    # Concatenate projection / gate weight matrices (all have shape [H, dim])
    w_4h = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # GEMM: (R, dim) @ (4·H, dim)ᵀ → (R, 4·H)
    fused_4h = F.linear(x_norm_fp16, w_4h)   # fp16

    # Reshape to (B, N, N, 4·H) and split
    fused_4h = fused_4h.view(B, N, N, 4 * H)
    left_raw   = fused_4h[..., 0:H]               # (B,N,N,H)
    right_raw  = fused_4h[..., H:2*H]
    left_gate_raw  = fused_4h[..., 2*H:3*H]
    right_gate_raw = fused_4h[..., 3*H:4*H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate linear (separate GEMM)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # (R, H)
    out_gate = out_gate_raw.view(B, N, N, H)          # (B,N,N,H)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask & sigmoid gating
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B,N,N) → (B,N,N,1) and broadcast to H
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left_raw  = left_raw  * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate.sigmoid_()

    left  = left_raw  * left_gate    # (B,N,N,H) fp16
    right = right_raw * right_gate   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update:
    #     out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Transform to batched matrices of shape (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (BH,N,N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (BH,N,N)

    # Batched GEMM: (BH,N,N) @ (BH,N,N)ᵀ  → (BH,N,N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to (B,N,N,H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension → fp16
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    out_norm = out_norm_fp16.view(B, N, N, H)   # (B,N,N,H) fp16

    # ------------------------------------------------------------------
    # 7️⃣ Fuse output‑gate multiplication with final linear projection
    # ------------------------------------------------------------------
    # (out_norm * out_gate) @ to_out.weightᵀ
    gated = (out_norm * out_gate).view(R, H)   # (R,H) fp16
    final_fp16 = F.linear(
        gated,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # (R, dim) fp16

    # ------------------------------------------------------------------
    # 8️⃣ Cast back to float32 and return
    # ------------------------------------------------------------------
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
