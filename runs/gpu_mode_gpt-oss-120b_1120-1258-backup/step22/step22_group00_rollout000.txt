"""
TriMul (outgoing) forward pass – pure‑PyTorch implementation.

The operator works on a 4‑D tensor `x` of shape [B, N, N, dim] and computes:

1. Row‑wise LayerNorm over the last dimension (dim).
2. Five linear projections (left, right, left_gate, right_gate, out_gate)
   sharing the same input `x_norm`.  All linear layers have no bias.
3. Optional per‑pair mask is applied to the left/right tensors.
4. Element‑wise sigmoid gates are applied to the left/right tensors.
5. A batched matrix multiplication (Einstein sum) contracts over the
   third axis `k` → `out[b,i,j,:] = Σₖ left[b,i,k,:] * right[b,j,k,:]`.
6. A second LayerNorm over the hidden dimension (hidden_dim).
7. The output‑gate (out_gate) is multiplied element‑wise.
8. A final linear projection back to the original `dim`.

All weight tensors are taken from the `weights` dictionary, and
configuration parameters (`dim`, `hidden_dim`, `nomask`) are read from
`config`.  The result is a float‑32 tensor of shape [B, N, N, dim].
"""

import torch
import torch.nn.functional as F

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping of weight names to torch.Tensor (float32)
                required keys:
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm over the feature dimension `dim`
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=w["norm.weight"],
        bias=w["norm.bias"],
        eps=eps,
    )  # shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (left/right) and gates (left_gate/right_gate/out_gate)
    # ------------------------------------------------------------------
    left  = F.linear(x_norm, w["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, w["right_proj.weight"])

    # Optional per‑pair mask (broadcasted on the feature dimension)
    if not nomask and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    left_gate  = F.linear(x_norm, w["left_gate.weight"]).sigmoid()   # [B, N, N, hidden_dim]
    right_gate = F.linear(x_norm, w["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x_norm, w["out_gate.weight"]).sigmoid()

    # Apply sigmoid gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3️⃣ Contract over the third axis `k` (Einstein summation)
    # ------------------------------------------------------------------
    # out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 5️⃣ Apply the output gate and final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, w["to_out.weight"])  # [B, N, N, dim]

    return out
