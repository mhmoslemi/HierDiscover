"""
Outgoing TriMul – optimized forward pass (no backward).

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened pairwise tensor.
2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into one FP16 GEMM:  proj = x_norm @ W_fusedᵀ   → shape [R, 5·H].
3️⃣ Split the fused projection, apply the optional pair‑wise mask,
   compute sigmoid gates and store three tensors `left`, `right`,
   `out_gate` in layout [B, H, N, N] (FP16).  `left`/`right` are already masked.
4️⃣ Batched half‑precision GEMM (cuBLAS):
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]
   (the classic “tri‑mul” contraction,  O(N³) work).
5️⃣ Fuse the second LayerNorm, the out‑gate multiplication and the final
   projection into a single Triton kernel.  The kernel first reduces the
   hidden dimension to obtain the per‑position mean/variance, then
   normalises, applies the `out_gate`, and finally performs the
   hidden‑dim → feature‑dim projection.  This eliminates an extra kernel
   launch and the intermediate μ/σ buffers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)

    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---- first pass: compute mean & variance ----
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)      # [M, D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- second pass: write normalised values ----
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        gamma = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # FP32
        beta  = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # FP32

        out = norm * gamma[None, :] + beta[None, :]
        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out.to(tl.float16),
            mask=mask_m[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr,                     # [R] half (ignored if HAS_MASK==0)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # Decode flat index r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # Optional per‑row mask (broadcast over hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]            # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # stride of one hidden‑dim plane inside the (B, H, N, N) layout
    hidden_stride = N * N
    # start of hidden‑dim 0 for this (b,i,k)
    base_offset = b * (H * N * N) + i * N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R, H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)

        # sigmoid (FP32 -> FP16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_offset,
                 left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset,
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset,
                 outg, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out‑gate and final projection
# ----------------------------------------------------------------------
@triton.jit
def fused_output_kernel(
    out_ptr,               # [B, H, N, N]   FP16   (result of the tri‑mul)
    gate_ptr,              # [B, H, N, N]   FP16   (out_gate)
    ln_w_ptr,              # [H]            FP32   (to_out_norm.weight)
    ln_b_ptr,              # [H]            FP32   (to_out_norm.bias)
    w_out_T_ptr,           # [H, dim]       FP16   (to_out.weightᵀ)
    out_final_ptr,         # [B, N, N, dim] FP32   (final result)
    total_rows, H, N, dim,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_N: tl.constexpr,
):
    """
    Each program works on a tile of `BLOCK_M` rows (flattened B·N·N)
    and a tile of `BLOCK_N` output features.
    """
    pid_m = tl.program_id(0)   # rows
    pid_n = tl.program_id(1)   # output‑dim tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)
    offs_n = col_start + tl.arange(0, BLOCK_N)

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode b,i,j from the flattened row index
    nn = N * N
    b = offs_m // nn
    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    # stride to step the hidden dimension
    hidden_stride = N * N
    base_offset = b * (H * N * N) + i * N + j        # offset of hidden‑dim 0

    # -----------------------------------------------------------------
    # 1️⃣  Compute μ and σ per row (first reduction over H)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        ptr = out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)                     # [M]

    # -----------------------------------------------------------------
    # 2️⃣  Normalise, apply out‑gate, LayerNormγ/β and project to dim
    # -----------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for k_start in range(0, H, BLOCK_K):
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < H

        # load out and gate (FP16 → FP32)
        out_vals  = tl.load(
            out_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        gate_vals = tl.load(
            gate_ptr + base_offset[:, None] + offs_k[None, :] * hidden_stride,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # [M, K]

        # LayerNorm parameters (γ,β) – float32 scalars per hidden dim
        gamma = tl.load(ln_w_ptr + offs_k, mask=mask_k, other=0.0)   # [K]
        beta  = tl.load(ln_b_ptr + offs_k, mask=mask_k, other=0.0)   # [K]

        # ((out - μ) * rstd) → normalised hidden activation
        diff = (out_vals - mu[:, None]) * rstd[:, None]            # [M, K]
        norm = diff * gamma[None, :] + beta[None, :]                # [M, K]
        term = norm * gate_vals                                      # [M, K]
        term_fp16 = term.to(tl.float16)                              # FP16 for dot

        # weight tile (H, dim) → (K, TILE_N)   FP16
        w_ptr  = w_out_T_ptr + offs_k[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(
            w_ptr,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )                                                             # [K, TILE_N]

        # Accumulate – dot works on FP16 → FP32
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)             # [M, TILE_N]

    # -----------------------------------------------------------------
    # 3️⃣  Write result
    # -----------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(
        dst_ptr,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:
        data (tuple): (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with the following float32 keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
            - config_dict  must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # unpack inputs
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total pairwise positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.reshape(R, dim).contiguous()                     # [R, dim] FP32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],
        w["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (5·hidden_dim)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5·H, dim]

    proj = F.linear(x_norm, fused_w)   # (R, 5·H) FP16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, optional mask, sigmoid gates → left/right/out_gate
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        # flatten mask to [R] and cast to half. 1.0 means keep, 0.0 means mask‑out.
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    BLOCK_R = 64
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS)
    # --------------------------------------------------------------
    # left/right are [B, H, N, N] (FP16).  Collapse B and H for bmm.
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    # Matrix product over the N dimension (i,k)·(j,k)ᵀ  →  [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out.view(B, H, N, N)          # [B, H, N, N]  FP16

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate and final projection
    # --------------------------------------------------------------
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # [H, dim]

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 128   # rows per program (flattened B·N·N)
    BLOCK_K_OUT = 32    # hidden‑dim tile
    BLOCK_N_OUT = 128   # output‑dim tile

    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    fused_output_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    return final
