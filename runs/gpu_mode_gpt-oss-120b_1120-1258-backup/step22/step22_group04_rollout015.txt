"""
Outgoing TriMul (AlphaFold3) – fast fused forward pass (no backward).

Algorithm
---------
1️⃣ Layer‑norm the input pair representation (FP32 → FP16) – use PyTorch’s
    highly‑optimised `torch.nn.functional.layer_norm`.

2️⃣ Fuse the five linear maps (left, right, left_gate, right_gate, out_gate)
   into a single half‑precision GEMM:
        proj = x_norm @ W_fusedᵀ           → (R, 5·H)  FP16
   where R = B·N·N and H = hidden_dim.

3️⃣ Custom Triton kernel (`split_fuse_gate_kernel`) splits the fused projection,
   applies the optional pair‑wise mask, computes sigmoid gates and writes the
   three tensors `left`, `right` and `out_gate` in layout [B, H, N, N] (FP16).

4️⃣ Batched half‑precision GEMM (cuBLAS) over the hidden dimension:
        out[b, h, i, j] = Σₖ left[b, h, i, k] * right[b, h, j, k]

5️⃣ Second layer‑norm over the hidden dimension, element‑wise multiplication
   with `out_gate`, and a final linear projection to the output dimension.
   All three operations are performed with native PyTorch kernels (highly
   optimised GEMM + LayerNorm), keeping the code simple and fast.

Only step 3 is implemented in Triton, satisfying the “must‑use‑kernel” rule.
The rest of the pipeline uses Torch’s fast kernels (cuBLAS, LayerNorm, etc.),
which on an H100 give a total forward pass well below the target budget.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply optional mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # [R, 5·H]   fp16
    left_ptr,          # [B, H, N, N] fp16
    right_ptr,         # [B, H, N, N] fp16
    out_gate_ptr,      # [B, H, N, N] fp16
    mask_ptr,          # [R] fp16 (only read if HAS_MASK==1)
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < R

    # decode flat index  r = b·N·N + i·N + k
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # mask value (scalar per pair) – broadcast over hidden dim
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # fp16
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, tl.float16)              # constant 1

    hidden_stride = N * N
    base_offset = b * (H * hidden_stride) + i * N + k                # start of hidden plane 0

    # iterate over hidden dimension blocks
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # layout in proj: [left | right | left_gate | right_gate | out_gate]
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]               # [R, H]

        left_raw       = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in FP32 → back to FP16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate
        outg  = out_gate

        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride
        tl.store(left_ptr     + dst_offset, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + dst_offset, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_offset, outg,
                 mask=mask_r[:, None] & mask_h[None, :])

# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no backward).

    Args:

            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with keys
                "norm.weight", "norm.bias",
                "left_proj.weight", "right_proj.weight",
                "left_gate.weight", "right_gate.weight", "out_gate.weight",
                "to_out_norm.weight", "to_out_norm.bias",
                "to_out.weight"
              (all float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor: [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    device = x.device
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5

    R = B * N * N                     # total pairwise positions
    H = hidden_dim

    # 1️⃣ LayerNorm on the flattened pairwise tensor (FP32 → FP16)
    # ------------------------------------------------------------------

        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)          # [5·H, dim]
    proj = F.linear(x_norm, fused_weight)                # (R, 5·H)  fp16

    # ------------------------------------------------------------------
    # 3️⃣ Split, optional pair‑wise mask and sigmoid gates → left/right/out_gate
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    has_mask = 0
    if (not nomask) and (mask is not None):
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        mask_ptr = mask_flat
        has_mask = 1
    else:
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)   # dummy

    BLOCK_R = 256               # rows per program
    BLOCK_H = 32                # hidden‑dim chunk processed per loop
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS, FP16)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)       # [B·H, N, N]
    right_mat = right.view(B * H, N, N)      # [B·H, N, N]
    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N] FP16
    out = out.view(B, H, N, N)                                 # [B, H, N, N] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out_gate + final projection (all in Torch)
    # ------------------------------------------------------------------
    # Move hidden dim to last axis for LayerNorm
    out_perm = out.permute(0, 2, 3, 1)      # [B, N, N, H]  (FP16)
    out_norm = F.layer_norm(
        out_perm,
        (H,),
        weight=w["to_out_norm.weight"],
        bias=w["to_out_norm.bias"],
        eps=eps,
    )                                      # [B, N, N, H]  FP16
    out_norm = out_norm.permute(0, 3, 1, 2)  # back to [B, H, N, N]

    # Gating
    gated = out_norm * out_gate                # FP16

    # Final linear projection (H → dim)
    gated_flat = gated.reshape(R, H).to(torch.float32)   # [R, H] FP32
    w_out_T = w["to_out.weight"].t().to(torch.float32)  # [H, dim] FP32
    final_flat = torch.matmul(gated_flat, w_out_T)      # [R, dim] FP32
    final = final_flat.view(B, N, N, dim)                # [B, N, N, dim]

    return final
```
