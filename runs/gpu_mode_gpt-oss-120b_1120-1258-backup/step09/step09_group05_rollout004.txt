"""
TriMul (outgoing) – mixed‑precision forward pass accelerated with Triton.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) on the flattened [B·N·N, C] tensor.
   Implemented as a Triton kernel (mean / variance per row, γ/β applied).

2️⃣ Fuse the four “projection + gating” linear layers into a single FP16 GEMM:
        X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ  →  [R, 4·H]
   where R = B·N·N and H = hidden_dim (always 128 in the tests).
   Split the result into (left_raw, right_raw, left_gate_raw, right_gate_raw).

3️⃣ Compute the output‑gate with a second FP16 GEMM.

4️⃣ (Optional) apply the binary mask to the raw left/right tensors,
   then apply sigmoid in‑place to the three gate tensors.

5️⃣ Form the “cubic” multiplicative update:
        out[h] = left[h] @ right[h]ᵀ   for every head h = 0…H‑1.
   Implemented by reshaping → batched fp16 matrix‑multiply (torch.bmm).

6️⃣ Second LayerNorm over the hidden dimension (fp16) – same Triton kernel.

7️⃣ Multiply by the output‑gate, apply the final FP16 linear projection,
   and cast the result back to float32.

The implementation works for any B∈{1,2}, N∈{128…1024},
dim∈{128,384,768}, hidden_dim=128 and supports both masked and un‑masked inputs.
Only the LayerNorms are written in Triton; all dense GEMMs use cuBLAS
through `torch.nn.functional.linear` / `torch.bmm`, which already exploit
Tensor‑Cores on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 input → fp16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim]   (fp16 or fp32)
    y_ptr,          # mutable* [total_rows, dim]   (fp16)
    w_ptr,          # const  * [dim]               (fp32) – gamma
    b_ptr,          # const  * [dim]               (fp32) – beta
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    #   1. Compute mean & variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                     # cast to fp32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # 1 / sqrt(var + eps)

    # ------------------------------------------------------------------
    #   2. Write normalized + affine transformed result (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        # γ,β are fp32 → keep computation in fp32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]            # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        - mask_tensor  : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights_dict : mapping name → torch.nn.Parameter (float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool}
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ unpack & constants
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim          = cfg["dim"]
    hidden_dim   = cfg["hidden_dim"]      # = H
    nomask       = cfg.get("nomask", True)
    eps          = 1e-5
    device       = x.device

    B, N, _, _ = x.shape
    R          = B * N * N               # total rows = B·N·N
    H          = hidden_dim              # hidden dimension (always 128)

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → fp16
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim]  fp32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128        # rows per program (tunable)
    BLOCK_D = 64         # channel tile size
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],    # gamma (fp32)
        weights["norm.bias"],      # beta  (fp32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fuse left/right projections & their gates into one GEMM
    # --------------------------------------------------------------
    # Concatenate the four weight matrices -> shape [4·H, dim] (fp16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                 # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H]  (fp16)

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # --------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # --------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H]  fp16

    # --------------------------------------------------------------
    # 4️⃣ Mask + sigmoid gates (in‑place to keep memory low)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] → [R, 1] → broadcast over H
        mask_fp16 = mask.view(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H] fp16
    right = right_raw * right_gate    # [R, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update: batched matrix‑multiply
    # --------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)
    out_gate = out_gate.view(B, N, N, H)

    # Move the head dimension into the batch axis → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (N×N) @ (N×N)ᵀ  →  [B·H, N, N]  (fp16, Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (fp16)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                # [R, H]
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # gamma (fp32)
        weights["to_out_norm.bias"],     # beta  (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast back to fp32
    # --------------------------------------------------------------
    # Element‑wise gate still fp16
    out_norm_fp16 = out_norm_fp16 * out_gate.view(R, H)

    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
