"""
TriMul (outgoing) – fast Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm (FP32 → FP16) on the flattened [B·N·N, C] tensor.
   Implemented as a tiny Triton kernel that processes BLOCK_M rows
   and loops over the channel dimension in BLOCK_D‑wide tiles.

2️⃣ Fuse the four linear projections/gates (left/right projection + left/right
   gate) into a single FP16 GEMM:
      X_norm @ [W_left; W_right; W_left_gate; W_right_gate].T  →  [R, 4·H].
   Split the result into raw left/right tensors and raw gate tensors.

3️⃣ Compute the output‑gate with a second FP16 GEMM.

4️⃣ (Optional) Apply the binary mask to the raw left/right tensors,
   then apply the sigmoid to the three gate tensors (in‑place).

5️⃣ Reshape left/right to [B, N, N, H] (contiguous) and evaluate the cubic
   multiplicative update with a single torch.einsum:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This avoids the costly permute‑and‑contiguous copy required for an
   explicit batched matmul.

6️⃣ Second LayerNorm over the hidden dimension (H) – again the Triton kernel.

7️⃣ Multiply by the output‑gate, apply the final FP16 linear projection,
   and cast back to float32.

The implementation respects the original PyTorch semantics while keeping
the heavy reductions in custom Triton kernels and the large matrix
multiplications in cuBLAS (via torch.nn.functional.linear / torch.einsum),
which gives a good trade‑off between speed and memory usage.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (any dtype, we read as fp16 and cast)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – gamma
    b_ptr,          # const  * [dim]          (fp32) – beta
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # float (runtime)
    BLOCK_M: tl.constexpr,   # rows processed per program
    BLOCK_D: tl.constexpr,   # channel tiles
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel processes BLOCK_M rows at a time and iterates over the
    channel dimension in BLOCK_D‑wide tiles.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1. Compute mean and variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        # Load a tile of shape [BLOCK_M, BLOCK_D]
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)          # fp16→fp32 if the source was fp16
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                     # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2. Write normalized result (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # (x-μ)/σ

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # gamma (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # beta  (fp32)
        out = norm * w[None, :] + b[None, :]                # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom Triton‑accelerated TriMul entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, C]  (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]  (bool/float) or None
        - weights_dict : mapping of parameter names → torch.Tensor (float32)
        - config_dict  : {
              "dim"         : input channel dimension C,
              "hidden_dim" : hidden dimension H,
              "nomask"     : bool (True ⇒ ignore mask)
          }

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N          # R = B·N·N
    H = hidden_dim                  # usually 128 in the provided tests

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # ------------------------------------------------------------------
    # Flatten to [R, dim] (contiguous)
    x_flat = x.view(total_rows, dim).contiguous()
    x_norm_fp16 = torch.empty(total_rows, dim,
                              dtype=torch.float16,
                              device=device)

    # Tuned block sizes (works well on H100 for dim ≤ 768)
    BLOCK_M = 256
    BLOCK_D = 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],      # γ  (float32)
        weights["norm.bias"],        # β  (float32)
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM (FP16)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # FP16 for the GEMM

    fused = F.linear(x_norm_fp16, w_fused)   # [R, 4·H]  (FP16)

    # Split into the four needed tensors
    left_raw, right_raw, left_gate_raw, right_gate_raw = torch.split(fused, H, dim=1)

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM, also FP16)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)
    )   # [R, H]  (FP16)

    # ------------------------------------------------------------------
    # 4️⃣ (Optional) mask & sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # Broadcast mask over the hidden dimension
        mask_fp16 = mask.view(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid to keep memory pressure low
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # Apply gates
    left  = left_raw * left_gate    # [R, H]  FP16
    right = right_raw * right_gate  # [R, H]  FP16

    # ------------------------------------------------------------------
    # 5️⃣ Cubic multiplicative update (no extra permute)
    # ------------------------------------------------------------------
    # Reshape to the original 4‑D layout – this view is cheap (no copy)
    left  = left.view(B, N, N, H)   # [B, N, N, H]
    right = right.view(B, N, N, H)  # [B, N, N, H]

    # Einstein summation implements   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # It works directly on the contiguously‑stored tensors and avoids the
    # expensive permute+contiguous+view dance required for a batched matmul.
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)   # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ  (float32)
        weights["to_out_norm.bias"],     # β  (float32)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Apply output‑gate, final projection, cast to float32
    # ------------------------------------------------------------------
    # Element‑wise gate (still FP16)
    out_norm_fp16 = out_norm_fp16 * out_gate.view(total_rows, H)

    # Final linear projection back to the original channel dimension
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim]  FP16

    # Reshape and cast to FP32 for the API contract
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
