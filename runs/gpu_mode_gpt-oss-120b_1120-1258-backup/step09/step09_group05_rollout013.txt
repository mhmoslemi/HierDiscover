"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

Algorithm (forward only)
------------------------
1️⃣  Row‑wise LayerNorm on the flattened [B·N·N, C] input (fp32 → fp16) with a
    custom Triton kernel.

2️⃣  Fuse *all* linear projections (left/right) and gates (left/right/out) into a
    single FP16 GEMM:
        X_norm @ [W_left ; W_right ; W_left_gate ; W_right_gate ; W_out_gate]ᵀ
    → tensor of shape [R, 5·H] (R = B·N·N, H = hidden_dim).
    Split the result and immediately apply the (optional) binary mask,
    the sigmoid gates and the element‑wise left/right gating:
        left  = (proj_left  * mask) * sigmoid(gate_left)
        right = (proj_right * mask) * sigmoid(gate_right)
        out_gate = sigmoid(gate_out)

3️⃣  Cubic multiplicative update (the expensive N³ part) using
    `torch.einsum('b i k h, b j k h -> b i j h', left, right)`.  
    This avoids the costly `permute + view + torch.bmm` copy‑pattern of the
    reference implementation.

4️⃣  Second row‑wise LayerNorm over the hidden dimension (H) with the same
    Triton kernel.

5️⃣  Apply the output‑gate (element‑wise) and perform the final FP16 linear
    projection back to the original channel dimension:
        y = (LN(out) * out_gate) @ W_outᵀ

All tensors are kept in fp16 after the first LayerNorm, which lets the
large matrix‑multiplications (steps 2, 3 and 5) run on Tensor‑Cores.
The implementation only uses Triton for the two LayerNorm passes – the rest
leverages highly‑optimised cuBLAS kernels (`torch.nn.functional.linear`,
`torch.einsum`).  This mixture gives a large speed‑up over the pure PyTorch
reference while keeping the code clear and easy to maintain.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const  * [total_rows, dim] (fp32)
    y_ptr,          # mutable* [total_rows, dim] (fp16)
    w_ptr,          # const  * [dim]          (fp32) – γ
    b_ptr,          # const  * [dim]          (fp32) – β
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """
    Row‑wise LayerNorm.
    y = ((x - μ) / sqrt(σ² + eps)) * γ + β
    The kernel processes BLOCK_M rows at a time and iterates over the
    channel dimension in BLOCK_D‑wide tiles.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------- 1. mean / variance --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # -------------------- 2. write normalized values (fp16) --------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]     # (x-μ)/σ

        # γ and β are fp32 – keep computation in fp32, cast at the end
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]              # fp32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, C], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.nn.Parameter (float32) containing:
              * "norm.weight", "norm.bias"
              * "left_proj.weight", "right_proj.weight"
              * "left_gate.weight", "right_gate.weight"
              * "out_gate.weight"
              * "to_out_norm.weight", "to_out_norm.bias"
              * "to_out.weight"
        - config_dict  : dict with keys
              * "dim"         – input channel dimension  C
              * "hidden_dim" – hidden dimension        H
              * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor of shape [B, N, N, C], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]               # C
    hidden_dim = cfg["hidden_dim"] # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – fp32 → fp16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, C] (fp32)
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128      # rows per program – empirically good on H100
    BLOCK_D = 64       # channel tile size
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],    # γ (float32)
        weights["norm.bias"],      # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection + all gates (single GEMM, FP16)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices: 4*H for proj/gates + 1*H for out_gate
    fused_weight = torch.cat(
        [
            weights["left_proj"].weight,   # [H, C]
            weights["right_proj"].weight,  # [H, C]
            weights["left_gate"].weight,   # [H, C]
            weights["right_gate"].weight,  # [H, C]
            weights["out_gate"].weight,    # [H, C]
        ],
        dim=0,
    ).to(torch.float16)                # shape [5*H, C]

    fused = F.linear(x_norm_fp16, fused_weight)   # [R, 5*H]  (FP16)

    # split
    left_raw        = fused[:, 0:H]
    right_raw       = fused[:, H:2 * H]
    left_gate_raw   = fused[:, 2 * H:3 * H]
    right_gate_raw  = fused[:, 3 * H:4 * H]
    out_gate_raw    = fused[:, 4 * H:5 * H]

    # ------------------------------------------------------------------
    # 3️⃣ (optional) mask & sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] broadcastable to [R, H]
        mask_fp16 = mask.view(R, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    # apply gating
    left  = left_raw * left_gate      # [R, H]
    right = right_raw * right_gate    # [R, H]

    # ------------------------------------------------------------------
    # 4️⃣ Cubic multiplicative update – einsum (no extra permutation)
    # ------------------------------------------------------------------
    # reshape to 4‑D tensors
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                 # [R, H]
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Apply output‑gate, final linear projection (FP16 → FP32)
    # ------------------------------------------------------------------
    # element‑wise gating
    out_norm_fp16 = out_norm_fp16 * out_gate.view(R, H)

    # final projection back to C channels
    final_fp16 = F.linear(
        out_norm_fp16,
        weights["to_out"].weight.to(torch.float16)   # [C, H]
    )   # [R, C]  (FP16)

    # reshape to original 4‑D layout and cast back to float32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
