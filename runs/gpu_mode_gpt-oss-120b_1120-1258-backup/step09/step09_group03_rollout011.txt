All heavy per‑element work (LayerNorm, mask handling, sigmoid) is fused in
Triton kernels; the large dense matrix multiplications use cuBLAS (torch.bmm)
which is already optimal on an H100. The implementation works for any batch
size ≤ 2, sequence length up to 1024 and arbitrary `dim` / `hidden_dim` used
in the tests.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   *[rows, dim]   (float32 or float16)
    y_ptr,               # mutable *[rows, dim]   (float16)
    weight_ptr,          # const   *[dim]        (float32) γ
    bias_ptr,            # const   *[dim]        (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1. compute mean & variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        # Load as fp32 regardless of original dtype
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2. write normalized values ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                      # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """For each row:
           left  = (proj_left  * mask) * sigmoid(proj_left_gate)
           right = (proj_right * mask) * sigmoid(proj_right_gate)
           out_gate = sigmoid(proj_out_gate)
    All tensors are FP16; sigmoid is computed in FP32.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ---- optional mask broadcast ----
    if HAS_MASK:
        # mask is 0/1 per row; broadcast across hidden dim
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0).to(tl.float16)      # [BLOCK_R]
        mask_val = mask_val[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full((BLOCK_R, 1), 1.0, dtype=tl.float16)

    # ---- loop over hidden dimension ----
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)      # [BLOCK_H]
        mask_h = offs_h < H

        # base offset for the row segment (first projection)
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]    # [R,H]

        # raw projection values
        left_raw       = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw      = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw  = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw   = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # sigmoid in fp32 for stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        dst_off = offs_r[:, None] * H + offs_h[None, :]      # [R,H]
        tl.store(left_ptr + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + dst_off,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Outgoing TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : mapping with LayerNorm and Linear parameters (float32)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                      # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm on input (FP32 → FP16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256      # rows per program
    BLOCK_D = 32       # columns per tile
    grid_norm = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection for left/right + all gates (5·H outputs)
    # --------------------------------------------------------------
    # Concatenate weight matrices (shape: [5*H, dim]) and cast to FP16
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)   # [5*H, dim]

    # (R, dim) @ (5*H, dim).T → (R, 5*H)   (FP16)
    proj = F.linear(x_norm, w_concat)      # fp16
    del w_concat, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split fused projection, apply mask, compute sigmoid gates
    # --------------------------------------------------------------
    left = torch.empty(R, H, dtype=torch.float16, device=device)
    right = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True) or mask is None:
        HAS_MASK = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        # Flatten mask to [R] (values 0/1) and keep as FP16
        mask_fp16 = mask.to(torch.float16).reshape(R)
        HAS_MASK = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)

    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    # Reshape to [B, N, N, H] → permute to bring H into batch axis
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back to [B, N, N, H] (fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + output‑gate multiplication
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    # Re‑use the same row‑wise LayerNorm kernel
    grid_norm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out

    # Element‑wise multiply by the output gate (both FP16)
    out_norm = out_norm * out_gate
    del out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
