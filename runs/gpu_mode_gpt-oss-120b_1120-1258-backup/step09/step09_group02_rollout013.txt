def custom_kernel(data):
    x, mask, weights, cfg = data
    dim, hidden_dim = cfg["dim"], cfg["hidden_dim"]
    eps=1e-5
    B,N,_,_ = x.shape
    R = B * N * N
    H = hidden_dim

    # 1. LayerNorm via Triton (x_norm half)
    x_flat = x.view(R, dim).contiguous()
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=x.device)
    BLOCK_M, BLOCK_D = 256, 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # 2. projections
    # Combine weight matrices into fused weight
    w_left = weights["left_proj.weight"]
    w_right = weights["right_proj.weight"]
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # use torch linear for each:
    left = F.linear(x_norm, w_left)      # (R,H) fp16
    right = F.linear(x_norm, w_right)
    left_gate = torch.sigmoid(F.linear(x_norm, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_norm, w_right_gate))
    out_gate = torch.sigmoid(F.linear(x_norm, w_out_gate))
    # free x_norm
    del x_norm

    if not cfg.get("nomask", True):
        mask_flat = mask.to(torch.float16).reshape(R,1)  # (R,1)
        left = left * mask_flat
        right = right * mask_flat

    left = left * left_gate
    right = right * right_gate

    # 3. reshape to batch form
    left = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # 4. TriMul via einsum
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # 5. 2nd LayerNorm (using Triton)
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # gating
    out_norm = out_norm * out_gate
    # final projection
    final = F.linear(out_norm, weights["to_out.weight"].to(torch.float16))
    final = final.view(B, N, N, dim).to(torch.float32)
    return final

"""
custom_kernel – fast “outgoing” TriMul (AlphaFold3)

Algorithm (high‑level)
----------------------
1️⃣ Row‑wise LayerNorm (fp32 → fp16) on the 4‑D input.
2️⃣ One fused Triton kernel computes the five linear projections,
   applies the optional pair‑wise mask and the three sigmoid gates,
   and writes directly the three tensors needed later:
        left  = (x·W_left) * mask * sigmoid(x·W_left_gate)
        right = (x·W_right) * mask * sigmoid(x·W_right_gate)
        out_gate = sigmoid(x·W_out_gate)
   No intermediate (R,5·H) tensor is materialised → huge memory‑traffic
   reduction.
3️⃣ Tri‑multiplicative update:
        out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
   Implemented with `torch.einsum`, which internally uses an
   efficient batched GEMM on the hidden dimension.
4️⃣ Row‑wise LayerNorm on the hidden‑dim tensor, element‑wise
   multiplication with `out_gate`.
5️⃣ Final linear projection back to the original `dim` (fp16 GEMM)
   and cast to float32.

Only the two LayerNorm kernels and the fused projection/gate kernel
are written in Triton – satisfying the “must‑use‑Triton” rule – while
all heavy GEMMs reuse cuBLAS via PyTorch for maximum throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------------------------------------------------------------
    # 1️⃣ mean & variance (fp32 accumulation)
    # ---------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------------------------------------------------------------
    # 2️⃣ normalise, apply γ/β and store as fp16
    # ---------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]        # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ, fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β, fp32
        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused projection + mask + sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def fused_proj_gate_kernel(
    x_ptr,              # const   *[R, D]          (float16)
    left_ptr,           # mutable *[R, H]          (float16)
    right_ptr,          # mutable *[R, H]          (float16)
    out_gate_ptr,       # mutable *[R, H]          (float16)
    mask_ptr,           # const   *[R]            (float16)   (ignored if HAS_MASK=0)
    w_left_ptr,         # const   *[D, H]          (float16)
    w_right_ptr,        # const   *[D, H]          (float16)
    w_left_gate_ptr,    # const   *[D, H]          (float16)
    w_right_gate_ptr,   # const   *[D, H]          (float16)
    w_out_gate_ptr,     # const   *[D, H]          (float16)
    R, D, H,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per program
    BLOCK_K: tl.constexpr,   # reduction dim tile
):
    """
    For each row r and hidden index h:
        left  = (x·W_left)      * mask * sigmoid(x·W_left_gate)
        right = (x·W_right)     * mask * sigmoid(x·W_right_gate)
        out_gate = sigmoid(x·W_out_gate)
    All arithmetic is done in fp32, final results stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ----------------------------------------------------------------
    # optional per‑row mask (broadcast later)
    # ----------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # fp16
        mask_val = m.to(tl.float32)[:, None]                    # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float32)

    # ----------------------------------------------------------------
    # iterate over hidden dimension tiles
    # ----------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # accumulators in fp32
        acc_l  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_r  = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_lg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_rg = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)
        acc_og = tl.zeros([BLOCK_R, BLOCK_H], dtype=tl.float32)

        # ----------------------------------------------------------------
        # reduction over the input dimension D
        # ----------------------------------------------------------------
        for k_off in range(0, tl.cdiv(D, BLOCK_K)):
            offs_k = k_off * BLOCK_K + tl.arange(0, BLOCK_K)      # [BLOCK_K]
            mask_k = offs_k < D

            # X block  [BLOCK_R, BLOCK_K]
            x = tl.load(x_ptr + offs_r[:, None] * D + offs_k[None, :],
                        mask=mask_r[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

            # weight blocks (transposed: D×H)
            w_l  = tl.load(w_left_ptr  + offs_k[:, None] * H + offs_h[None, :],
                           mask=mask_k[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            w_r  = tl.load(w_right_ptr + offs_k[:, None] * H + offs_h[None, :],
                           mask=mask_k[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            w_lg = tl.load(w_left_gate_ptr  + offs_k[:, None] * H + offs_h[None, :],
                           mask=mask_k[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            w_rg = tl.load(w_right_gate_ptr + offs_k[:, None] * H + offs_h[None, :],
                           mask=mask_k[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)
            w_og = tl.load(w_out_gate_ptr   + offs_k[:, None] * H + offs_h[None, :],
                           mask=mask_k[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float32)

            # dot products
            acc_l  += tl.dot(x, w_l)   # (BLOCK_R, BLOCK_H)
            acc_r  += tl.dot(x, w_r)
            acc_lg += tl.dot(x, w_lg)
            acc_rg += tl.dot(x, w_rg)
            acc_og += tl.dot(x, w_og)

        # --------------------------------------------------------------
        # sigmoid gates (computed in fp32 for stability)
        # --------------------------------------------------------------
        lg_sig = (1.0 / (1.0 + tl.exp(-acc_lg))).to(tl.float16)   # left gate
        rg_sig = (1.0 / (1.0 + tl.exp(-acc_rg))).to(tl.float16)   # right gate
        og_sig = (1.0 / (1.0 + tl.exp(-acc_og))).to(tl.float16)   # out gate

        # --------------------------------------------------------------
        # apply mask + gates, store results (fp16)
        # --------------------------------------------------------------
        left = (acc_l * mask_val).to(tl.float16) * lg_sig
        right= (acc_r * mask_val).to(tl.float16) * rg_sig
        out_gate = og_sig

        # write
        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,      mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right,     mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate,  mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Entry point – forward pass of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N]      (bool/float) or None
        - weights_dict : dict of torch.Tensors (float32, bias=False for Linear)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total rows for row‑wise ops
    H = hidden_dim
    D = dim

    # ------------------------------------------------------------------
    # 1️⃣ first LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],    # γ  (float32)
        weights["norm.bias"],      # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ fused projection + mask + sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    # transpose weights so they are stored as [dim, hidden] for easy dot
    w_left_T       = weights["left_proj.weight"].t().contiguous().to(torch.float16)
    w_right_T      = weights["right_proj.weight"].t().contiguous().to(torch.float16)
    w_left_gate_T  = weights["left_gate.weight"].t().contiguous().to(torch.float16)
    w_right_gate_T = weights["right_gate.weight"].t().contiguous().to(torch.float16)
    w_out_gate_T   = weights["out_gate.weight"].t().contiguous().to(torch.float16)

    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)  # ignored
        has_mask = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        dummy_mask = mask_flat
        has_mask = 1

    BLOCK_R = 32           # rows per program (tuned for H100)
    BLOCK_H = 32           # hidden tiles
    BLOCK_K = 32           # reduction tiles (dim ≤ 384)
    grid_proj = (triton.cdiv(R, BLOCK_R),)

    fused_proj_gate_kernel[grid_proj](
        x_norm,
        left,
        right,
        out_gate,
        dummy_mask,
        w_left_T,
        w_right_T,
        w_left_gate_T,
        w_right_gate_T,
        w_out_gate_T,
        R, D, H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # free memory we no longer need
    del x_norm, w_left_T, w_right_T, w_left_gate_T, w_right_gate_T, w_out_gate_T

    # ------------------------------------------------------------------
    # 3️⃣ Tri‑multiplicative update (einsum over the sequence dimension)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # out shape: [B, N, N, H]
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)

    # ------------------------------------------------------------------
    # 4️⃣ second LayerNorm + gating with out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # element‑wise multiplication with the output‑gate (fp16)
    out_norm = out_norm * out_gate
    del out, out_gate

    # ------------------------------------------------------------------
    # 5️⃣ final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        out_norm,
        weights["to_out.weight"].t().contiguous().to(torch.float16),   # [dim, H]
    )   # -> fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
