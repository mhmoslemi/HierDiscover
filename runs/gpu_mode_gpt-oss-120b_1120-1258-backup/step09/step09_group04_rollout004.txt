"""
TriMul (outgoing) – high‑performance forward implementation (H100, Triton 3.3.1)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the trailing channel dimension (dim) – use the
    highly‑optimised torch.nn.functional.layer_norm (FP32 → FP16).

2️⃣ Fuse the four linear projections (left_proj, right_proj, left_gate, right_gate)
    into one FP16 GEMM:
        X_norm @ W_fused.T → [R, 4·H]   (R = B·N·N, H = hidden_dim)
    Split the result and compute the output‑gate with a second linear.

3️⃣ Apply optional binary mask (broadcasted) and sigmoid gates (in‑place).

4️⃣ Perform the N³ multiplicative update with a batched matrix‑multiply:
        (B·H, N, N) @ (B·H, N, N)ᵀ  →  (B·H, N, N)
    The operation is executed with torch.bmm (Tensor‑Core‑enabled FP16).

5️⃣ Fuse the second LayerNorm (on the hidden dimension) with the element‑wise
    multiplication by the output‑gate into a single Triton kernel.  The kernel
    reduces each row of FP16 data to FP32 mean/variance, normalises, applies
    γ/β and finally multiplies by the pre‑computed out‑gate.

6️⃣ Final linear projection back to the original dimension (FP16 GEMM) and cast
    to FP32 for the output.

Only the fused layer‑norm + gate kernel is written in Triton, satisfying the
requirement that “part of the operations” run on a custom kernel while keeping
the overall runtime well below the target.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float16)
    gate_ptr,       # const   * [total_rows, dim]   (float16)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean & variance (FP32) for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # cast FP16→FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise, apply γ/β and multiply by the gate
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # raw activation (FP16 → FP32)
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # gate (already FP16, cast to FP32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (FP32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (FP32)

        out = (norm * w[None, :] + b[None, :]) * g          # FP32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape                 # [B, N, N, dim]
    R = B * N * N                         # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) – torch implementation (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()            # [R, dim] FP32
    # torch.layer_norm works on the last dimension; it returns FP32 → cast to FP16
    x_norm_fp16 = F.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single FP16 GEMM
    # ------------------------------------------------------------------
    # Build fused weight: [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)  # [R, 4·H] (FP16)

    left_raw   = fused[:, 0:H]                # [R, H]
    right_raw  = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]        # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]        # [R, H]

    # Separate linear for the output‑gate (still FP16)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Mask (if required) and sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape -> [R, 1] broadcast to left/right
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()               # [R, H] (FP16)

    # Apply gates to the projections
    left  = left_raw * left_gate    # [R, H] FP16
    right = right_raw * right_gate  # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) → (B·H, N, N)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H,N,N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)   # (B·H,N,N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)   (FP16 Tensor‑Core GEMM)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H] FP16

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + out‑gate (over hidden dim H) – Triton kernel
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()               # [R, H] FP16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    # Triton launch configuration (tuned for H100)
    BLOCK_M = 128          # rows per program
    BLOCK_D = min(64, H)   # cols per iteration (no larger than dim)
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate,                              # gate tensor (FP16)
        out_norm_gate,
        weights["to_out_norm.weight"],         # γ (FP32)
        weights["to_out_norm.bias"],           # β (FP32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)   # [B,N,N,dim] FP32
    return final
