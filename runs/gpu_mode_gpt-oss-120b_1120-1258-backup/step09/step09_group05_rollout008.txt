"""
TriMul – outgoing update (AlphaFold‑3 style) – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input `x` (float32 → float16) using
    :func:`torch.nn.functional.layer_norm` (CuDNN) – this is the fastest
    implementation for the “norm” step.

2️⃣  Fuse the four linear layers (left/right projection + left/right gate) into a
    single FP16 GEMM:
            X_norm @ [W_left; W_right; W_left_gate; W_right_gate]ᵀ  →  [R,4·H]
    where R = B·N·N and H = hidden_dim.  The result is split into raw
    left/right tensors and raw gate tensors.

3️⃣  Compute the output‑gate with a second FP16 GEMM.

4️⃣  (Optional) apply the binary mask to the raw left/right tensors and apply
    sigmoid to all three gate tensors (in‑place).

5️⃣  Reshape ``left``/``right`` to `[B, H, N, N]`, flatten the head dimension and
    perform a batched matrix multiplication:
            out_mat = left_mat @ right_matᵀ          (Tensor‑Core FP16)
    which implements the Einstein sum
        `einsum('... i k d, ... j k d -> ... i j d')`.

6️⃣  Fuse the second LayerNorm **and** the element‑wise multiplication with the
    output‑gate into a single Triton kernel (`layernorm_gate_kernel`).  This removes
    an extra kernel launch and one full‑tensor element‑wise pass.

7️⃣  Final linear projection (`to_out`) and cast back to float32.

The implementation keeps at least one Triton kernel (step 6) as required,
while the rest of the pipeline uses highly‑optimized PyTorch/CuBLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm + optional point‑wise gate multiplication
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const  * [total_rows, dim]   (fp16)
    y_ptr,          # mutable* [total_rows, dim]   (fp16) – normalized * gate
    w_ptr,          # const  * [dim]               (fp32) – γ
    b_ptr,          # const  * [dim]               (fp32) – β
    gate_ptr,       # const  * [total_rows, dim]   (fp16) – sigmoid(out_gate)
    total_rows,     # int (runtime)
    dim,            # int (runtime)
    eps,            # compile‑time float
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per tile
):
    """Row‑wise LayerNorm followed by element‑wise multiplication with a gate."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)            # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # 1. Compute mean and variance for each row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp32 load
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                            # 1 / sqrt(var+eps)

    # ------------------------------------------------------------------
    # 2. Write normalized result (fp16) and multiply by the gate
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]     # (x‑μ)/σ   (fp32)

        # γ and β are fp32 – keep computation in fp32, cast at the end
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        out = norm * w[None, :] + b[None, :]              # fp32

        # ---- gate multiplication (gate is fp16, cast to fp32) ----
        gate = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # fp16 → fp32
        out = out * gate

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of torch.Parameters (float32)
        - config_dict  : dict with keys
            * "dim"         – input channel dimension (C)
            * "hidden_dim" – hidden dimension (H)
            * "nomask"      – bool, True → ignore mask

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim], dtype float32
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]           # H
    nomask = cfg.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    total_rows = B * N * N                  # R = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (use CuDNN, then cast to fp16)
    # ------------------------------------------------------------------
    # `torch.nn.functional.layer_norm` normalises over the last dimension.
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                     # [B, N, N, dim] fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (single FP16 GEMM)
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices: [H, dim] × 4 → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    # Flatten input for the GEMM: [R, dim]
    x_flat = x_norm.view(total_rows, dim)
    fused = F.linear(x_flat, w_fused)       # [R, 4·H]  (fp16)

    # Split fused result
    left_raw        = fused[:, 0:H]
    right_raw       = fused[:, H:2 * H]
    left_gate_raw   = fused[:, 2 * H:3 * H]
    right_gate_raw  = fused[:, 3 * H:4 * H]

    # ------------------------------------------------------------------
    # 3️⃣ Output‑gate (separate GEMM)
    # ------------------------------------------------------------------
    out_gate_raw = F.linear(
        x_flat,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H] fp16

    # ------------------------------------------------------------------
    # 4️⃣ (Optional) mask & sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1]  (broadcast to [R, H])
        mask_fp16 = mask.reshape(total_rows, 1).to(torch.float16)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid → keep memory low
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate    # [R, H] fp16
    right = right_raw * right_gate  # [R, H] fp16

    # ------------------------------------------------------------------
    # 5️⃣ Cubic update (batched matmul over the hidden dimension)
    # ------------------------------------------------------------------
    # Reshape to [B, N, N, H] then bring H into the batch dimension
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Permute to [B, H, N, N] and flatten batch×head → [B·H, N, N]
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # Batched matrix multiplication: (N×N) @ (N×N)ᵀ  →  [B·H, N, N]  (FP16 Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout → [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + multiplication with output‑gate (single Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(total_rows, H).contiguous()
    out_gate_flat = out_gate.view(total_rows, H).contiguous()
    out_norm_gate_fp16 = torch.empty_like(out_flat, dtype=torch.float16)

    # Choose tile sizes that cover the hidden dimension efficiently.
    BLOCK_M = 128          # rows per program
    BLOCK_D = 64           # hidden‑dim tiles (covers H=128 nicely)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_gate_kernel[grid](
        out_flat,
        out_norm_gate_fp16,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        out_gate_flat,                   # gate (fp16)
        total_rows,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection & cast back to float32
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] fp16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
