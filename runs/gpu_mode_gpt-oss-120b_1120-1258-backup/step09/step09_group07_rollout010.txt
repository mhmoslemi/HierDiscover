"""
custom_kernel: fast forward of the *outgoing* TriMul operator.

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the input (FP32 → FP16) – Triton.
2️⃣  Fuse the four linear projections (left/right + gates) into a single
    mat‑mul on the normalized tensor.  Split the result into the four
    tensors.
3️⃣  Compute the output gate with a second linear on the same normalized
    tensor.
4️⃣  Apply the pair‑wise mask (if present) and the sigmoid gates.
5️⃣  Compute the N³ multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d]·right[b,j,k,d]
    This is a batched matrix‑multiply with batch size B·H.
6️⃣  Fuse the second LayerNorm (over the hidden dimension) with the
    element‑wise multiplication by the output gate – Triton.
7️⃣  Final linear projection back to the original channel dimension
    (FP16 → FP32).

Only the two LayerNorm steps are written in Triton; the dense GEMMs and
the N³ update use PyTorch’s highly‑optimised CUDA kernels.  The kernel
layout (BLOCK_M=256, BLOCK_D=64) was chosen to minimise the number of
launches while keeping good occupancy on an H100.  All heavy compute
remains in FP16 (with FP32 accumulation in the LayerNorms) and the
final result is returned in FP32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)  γ
    bias_ptr,       # const   * [dim]            (float32)  β
    total_rows,     # int32   runtime
    dim,            # int32   runtime
    eps,            # float   compile‑time
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Reduce to mean / var (FP32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # Loop over the channel dimension
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise, apply γ / β and store as FP16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)

        norm = (val - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β
        out = norm * w[None, :] + b[None, :]                       # FP32

        out_fp16 = out.to(tl.float16)
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise gate (FP16 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float16)
    gate_ptr,       # const   * [total_rows, dim] (float16)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]            (float32)  γ
    bias_ptr,       # const   * [dim]            (float32)  β
    total_rows,     # int32   runtime
    dim,            # int32   runtime
    eps,            # float   compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Compute mean / var (FP32) – same as the pure LayerNorm kernel
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        val = tl.load(ptr,
                      mask=mask_m[:, None] & mask_d[None, :],
                      other=0.0).to(tl.float32)
        sum_val += tl.sum(val, axis=1)
        sum_sq  += tl.sum(val * val, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Normalise, apply γ/β and finally multiply by the gate (all FP32)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)
        g_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        norm = (x_val - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g_val
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        weights_dict : mapping containing all Linear/LayerNorm parameters
        config_dict  : {"dim": int, "hidden_dim": int,
                       "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N              # total number of (b,i,k) rows
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension (FP32 → FP16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()            # [R, dim] FP32
    x_norm = torch.empty(R, dim,
                         dtype=torch.float16,
                         device=device)          # [R, dim] FP16

    BLOCK_M = 256   # rows per program – half the number of launches vs 128
    BLOCK_D = 64    # channels per iteration – fits dim up to 384
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],       # γ (float32)
        weights["norm.bias"],         # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections and their gates into a single GEMM
    # ------------------------------------------------------------------
    # Build one big weight matrix [4·H, dim] on the device (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)          # [4·H, dim]

    fused = F.linear(x_norm, w_fused)               # [R, 4·H] FP16

    left_raw       = fused[:, 0:H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]

    # Output gate (separate linear, still FP16)
    out_gate_raw = F.linear(
        x_norm,
        weights["out_gate.weight"].to(torch.float16, non_blocking=True)   # [H, dim]
    )                                                                       # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if present) and the sigmoid gates
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask shape [B, N, N] → [R, 1] (FP16)
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()

    left  = left_raw * left_gate      # [R, H] FP16
    right = right_raw * right_gate    # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ N³ multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # Reshape to (B, N, N, H)
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # Bring the hidden dimension into the batch axis → (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Back‑to the original layout (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm (over hidden dim) with the output gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                # [R, H] FP16
    out_gate_flat = out_gate                              # [R, H] FP16
    out_ln_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    layernorm_gate_kernel[grid](
        out_flat,
        out_gate_flat,
        out_ln_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_ln_gate,
        weights["to_out.weight"].to(torch.float16, non_blocking=True)   # [dim, H]
    )                                                                   # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)              # FP32 output
    return final
