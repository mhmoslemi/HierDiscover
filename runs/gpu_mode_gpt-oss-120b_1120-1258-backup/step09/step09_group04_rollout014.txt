@triton.jit
def gate_mask_kernel(
    left_raw_ptr, right_raw_ptr,
    left_gate_raw_ptr, right_gate_raw_ptr,
    out_gate_raw_ptr,
    mask_ptr, # can be null pointer if has_mask=0
    left_out_ptr, right_out_ptr,
    out_gate_out_ptr,
    total_rows, hidden_dim,
    has_mask: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < hidden_dim

        # pointers with offsets
        left_raw = tl.load(left_raw_ptr + offs_m[:, None] * hidden_dim + offs_h[None,:],
                           mask=mask_m[:, None] & mask_h[None,:],
                           other=0.0).to(tl.float32)
        right_raw = tl.load(right_raw_ptr + offs_m[:, None] * hidden_dim + offs_h[None,:],
                            mask=mask_m[:, None] & mask_h[None,:],
                            other=0.0).to(tl.float32)
        left_gate_raw = tl.load(left_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_h[None,:],
                                mask=mask_m[:, None] & mask_h[None,:],
                                other=0.0).to(tl.float32)
        right_gate_raw = tl.load(right_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_h[None,:],
                                 mask=mask_m[:, None] & mask_h[None,:],
                                 other=0.0).to(tl.float32)
        out_gate_raw = tl.load(out_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_h[None,:],
                               mask=mask_m[:, None] & mask_h[None,:],
                               other=0.0).to(tl.float32)
        # compute sigmoids
        left_gate = 1.0 / (1.0 + tl.exp(-left_gate_raw))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate_raw))
        out_gate = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        # mask
        if has_mask:
            mask_val = tl.load(mask_ptr + offs_m, mask=mask_m, other=0.0).to(tl.float32)
            mask_val = mask_val[:, None] # broadcast
            left = left_raw * left_gate * mask_val
            right = right_raw * right_gate * mask_val
        else:
            left = left_raw * left_gate
            right = right_raw * right_gate

        # store
        tl.store(left_out_ptr + offs_m[:, None] * hidden_dim + offs_h[None,:],
                 left.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None,:])
        tl.store(right_out_ptr + offs_m[:, None] * hidden_dim + offs_h[None,:],
                 right.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None,:])
        tl.store(out_gate_out_ptr + offs_m[:, None] * hidden_dim + offs_h[None,:],
                 out_gate.to(tl.float16),
                 mask=mask_m[:, None] & mask_h[None,:])

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; nomask = config.get("nomask", True)

    device = x.device
    B,N,_,_ = x.shape
    R = B*N*N
    H = hidden_dim
    eps = 1e-5

    # First LayerNorm: using Triton kernel
    x_flat = x.view(R, dim).contiguous()
    x_norm_fp16 = torch.empty_like(x_flat, dtype=torch.float16)
    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat, x_norm_fp16,
        weights["norm.weight"], weights["norm.bias"],
        R, dim, eps,
        BLOCK_M=BLOCK_M, BLOCK_D=BLOCK_D,
        num_warps=4, num_stages=2
    )

    # fused linear projection
    w_fused = torch.cat([weights["left_proj.weight"],
                         weights["right_proj.weight"],
                         weights["left_gate.weight"],
                         weights["right_gate.weight"]], dim=0).to(torch.float16)
    fused = F.linear(x_norm_fp16, w_fused)  # shape [R,4*H]

    # split raw tensors
    left_raw = fused[:, 0:H]
    right_raw = fused[:, H:2*H]
    left_gate_raw = fused[:, 2*H:3*H]
    right_gate_raw = fused[:, 3*H:4*H]

    # out_gate raw
    out_gate_raw = F.linear(x_norm_fp16, weights["out_gate.weight"].to(torch.float16))

    # Gate and mask kernel
    left_gated = torch.empty_like(left_raw)
    right_gated = torch.empty_like(right_raw)
    out_gate = torch.empty_like(out_gate_raw)

    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)  # [R,1]
    else:
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    # gating kernel
    BLOCK_H = 32
    grid_gate = (triton.cdiv(R, BLOCK_M),)
    gate_mask_kernel[grid_gate](
        left_raw, right_raw,
        left_gate_raw, right_gate_raw,
        out_gate_raw,
        mask_fp16,
        left_gated, right_gated,
        out_gate,
        R, H,
        0 if (nomask or mask is None) else 1,
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )

    # Now left_gated,right_gated have shape (R,H)
    # reshape to (B,N,N,H)
    left = left_gated.view(B, N, N, H)
    right = right_gated.view(B, N, N, H)

    # Batched GEMM
    left_mat = left.permute(0, 3, 1, 2).reshape(B*H, N, N)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B*H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()   # (B,N,N,H)

    # second layernorm + gate multiplication
    out_flat = out.view(R, H).contiguous()
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,
        out_norm_gate,
        weights["to_out_norm.weight"], weights["to_out_norm.bias"],
        R, H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

    # final linear
    final_fp16 = F.linear(out_norm_gate, weights["to_out.weight"].to(torch.float16))
    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final

"""
TriMul (outgoing) – high‑performance forward pass.

The implementation follows the PyTorch reference but moves the two
LayerNorms and the three sigmoid‑gate + optional mask operations to
hand‑written Triton kernels.  All matrix multiplications are performed
with torch’s Tensor‑Core‑enabled FP16 GEMMs, which gives the best
throughput on an H100.  The overall algorithm is:

1️⃣ Row‑wise LayerNorm on the input (FP32 → FP16) – Triton.
2️⃣ One fused linear projection that produces four tensors
   (left_raw, right_raw, left_gate_raw, right_gate_raw) of shape
   [R, hidden] (R = B·N·N).  The projection is a regular torch FP16 GEMM.
3️⃣ Compute the output‑gate (out_gate_raw) with a second FP16 GEMM.
4️⃣ Triton kernel that
   – applies the binary mask (if present),
   – computes the three sigmoids,
   – multiplies left/right by their gates,
   – stores the gated tensors and the output‑gate.
5️⃣ Batched N³ multiplicative update:
   left_gated/right_gated are reshaped to (B·H, N, N) and multiplied
   with a transposed version of right using torch.bmm.
6️⃣ Triton kernel that fuses the second LayerNorm (normalises the
   result of the multiplicative update) and the element‑wise product
   with the pre‑computed out_gate.
7️⃣ Final FP16 linear projection back to the original dimensionality.

Only the two LayerNorms and the gate+mask kernel are written in
Triton; everything else uses highly‑optimised torch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 input → FP16 output)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim] (float32)
    y_ptr,          # mutable * [total_rows, dim] (float16)
    weight_ptr,     # const   * [dim]           (float32)
    bias_ptr,       # const   * [dim]           (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ---------- second pass: normalise and apply γ,β ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                     # still FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton kernel: fused sigmoid‑gate, optional mask and out‑gate
# ----------------------------------------------------------------------
@triton.jit
def gate_mask_kernel(
    left_raw_ptr, right_raw_ptr,           # const * [R, H] (fp16)
    left_gate_ptr, right_gate_ptr,         # const * [R, H] (fp16)
    out_gate_raw_ptr,                     # const * [R, H] (fp16)
    mask_ptr,                             # const * [R]    (fp16) – may be dummy
    left_out_ptr, right_out_ptr,          # mutable * [R, H] (fp16)
    out_gate_out_ptr,                     # mutable * [R, H] (fp16)
    total_rows, hidden_dim,
    HAS_MASK: tl.constexpr,                # 0 → no mask, 1 → mask present
    BLOCK_M: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_row = offs_m < total_rows

    for h_off in range(0, tl.cdiv(hidden_dim, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < hidden_dim

        # load the five tensors
        left_raw   = tl.load(left_raw_ptr   + offs_m[:, None] * hidden_dim + offs_h[None, :],
                             mask=mask_row[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
        right_raw  = tl.load(right_raw_ptr  + offs_m[:, None] * hidden_dim + offs_h[None, :],
                             mask=mask_row[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
        left_gate  = tl.load(left_gate_ptr  + offs_m[:, None] * hidden_dim + offs_h[None, :],
                             mask=mask_row[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
        right_gate = tl.load(right_gate_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                             mask=mask_row[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)
        out_gate_raw = tl.load(out_gate_raw_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                               mask=mask_row[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float32)

        # sigmoid(x) = 1 / (1 + exp(-x))
        left_gate  = 1.0 / (1.0 + tl.exp(-left_gate))
        right_gate = 1.0 / (1.0 + tl.exp(-right_gate))
        out_gate   = 1.0 / (1.0 + tl.exp(-out_gate_raw))

        if HAS_MASK:
            # mask is a scalar per (b,i,k) → broadcast over H
            mask_val = tl.load(mask_ptr + offs_m,
                               mask=mask_row,
                               other=0.0).to(tl.float32)[:, None]   # [BLOCK_M,1]
            left  = left_raw  * left_gate  * mask_val
            right = right_raw * right_gate * mask_val
        else:
            left  = left_raw  * left_gate
            right = right_raw * right_gate

        # store gated tensors
        tl.store(left_out_ptr  + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 left .to(tl.float16),
                 mask=mask_row[:, None] & mask_h[None, :])
        tl.store(right_out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 right.to(tl.float16),
                 mask=mask_row[:, None] & mask_h[None, :])
        tl.store(out_gate_out_ptr + offs_m[:, None] * hidden_dim + offs_h[None, :],
                 out_gate.to(tl.float16),
                 mask=mask_row[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise multiplication by out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    gate_ptr,       # const   * [total_rows, dim]   (float16)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ---------- first pass: mean & variance ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- second pass: normalise, apply γ/β and multiply by gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g              # FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# Custom Triton‑accelerated forward of the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of all Linear/LayerNorm parameters
        - config_dict  : {"dim": int, "hidden_dim": int,
                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack and constants
    # --------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape                # [B, N, N, dim]
    R = B * N * N                        # total number of (b,i,k) rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection that yields left/right + their gates
    # --------------------------------------------------------------
    #   [R, dim] @ [4·H, dim]ᵀ  →  [R, 4·H]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # [H, dim]
            weights["right_proj.weight"],  # [H, dim]
            weights["left_gate.weight"],   # [H, dim]
            weights["right_gate.weight"],  # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # shape [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)  # FP16, shape [R, 4·H]

    left_raw   = fused[:, 0:H]                # [R, H]
    right_raw  = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]        # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]        # [R, H]

    # out‑gate (separate linear)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )                                                   # [R, H]

    # --------------------------------------------------------------
    # 3️⃣ Sigmoid gates + optional mask (fused Triton kernel)
    # --------------------------------------------------------------
    left_gated   = torch.empty_like(left_raw)
    right_gated  = torch.empty_like(right_raw)
    out_gate    = torch.empty_like(out_gate_raw)

    if not nomask and mask is not None:
        # flatten mask to [R] and keep it in fp16
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)   # [R,1]
    else:
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)  # dummy

    # launch
    BLOCK_H = 32
    grid_gate = (triton.cdiv(R, BLOCK_M),)
    gate_mask_kernel[grid_gate](
        left_raw, right_raw,
        left_gate_raw, right_gate_raw,
        out_gate_raw,
        mask_fp16,
        left_gated, right_gated,
        out_gate,
        R,
        H,
        0 if (nomask or mask is None) else 1,   # HAS_MASK constexpr
        BLOCK_M=BLOCK_M,
        BLOCK_H=BLOCK_H,
        num_warps=4,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # --------------------------------------------------------------
    # reshape to (B, N, N, H) → (B·H, N, N)
    left = left_gated.view(B, N, N, H)
    right = right_gated.view(B, N, N, H)

    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise out‑gate (fused)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()        # [R, H] FP16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)

    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
        num_stages=2,
    )

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # --------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )                                                # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
