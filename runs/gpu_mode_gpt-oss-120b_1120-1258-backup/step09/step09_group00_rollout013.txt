@triton.jit
def split_fuse_gate_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr, mask_ptr,
    rows: tl.int32, H: tl.int32, N: tl.int32,
    HAS_MASK: tl.constexpr, BLOCK_R: tl.constexpr, BLOCK_H: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]  # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    seq_sq = N * N

    # compute batch index = b = offs_r // seq_sq
    b_idx = offs_r // seq_sq                # [BLOCK_R]
    offset_plane = offs_r - b_idx * seq_sq   # i*N + k

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # base pointer offsets for proj segments
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]
        left_raw_ptr = proj_ptr + base
        right_raw_ptr = proj_ptr + base + H
        left_gate_ptr = proj_ptr + base + 2 * H
        right_gate_ptr = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        # load raw data
        left_raw = tl.load(left_raw_ptr,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        right_raw = tl.load(right_raw_ptr,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        left_gate_raw = tl.load(left_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw = tl.load(right_gate_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw = tl.load(out_gate_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # sigmoid gates (in fp32)
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # compute target bh indices: bh = b_idx * H + (h_off*BLOCK_H + offs_h)
        bh = b_idx[:, None] * H + offs_h[None, :]  # [BLOCK_R, BLOCK_H]

        # compute offsets for left_ptr and right_ptr
        # flat offset = bh * seq_sq + offset_plane
        left_ptr_off = left_ptr + bh * seq_sq + offset_plane[:, None]
        right_ptr_off = right_ptr + bh * seq_sq + offset_plane[:, None]

        tl.store(left_ptr_off, left, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr_off, right, mask=mask_r[:, None] & mask_h[None, :])

        # store out_gate in out_gate_ptr (R,H) layout
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst_ptr, out_gate, mask=mask_r[:, None] & mask_h[None, :])
```

@triton.jit
def fused_ln_gate_linear_kernel(
    x_ptr, gate_ptr, y_ptr,
    ln_weight_ptr, ln_bias_ptr,
    out_weight_ptr,   # transposed weight shape [H, D] (fp16)
    rows: tl.int32, H: tl.int32, D: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # First pass: compute mean & var over H dimension
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        x_ptr_off = x_ptr + offs_r[:, None] * H + offs_h[None, :]
        x_vals = tl.load(x_ptr_off,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
        sum_val += tl.sum(x_vals, axis=1)
        sum_sq += tl.sum(x_vals * x_vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # accumulator for output (R, D)
    acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

    # Second pass: compute activation and gemm with out_weight_T (H, D)
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Load x and gate
        x_ptr_off = x_ptr + offs_r[:, None] * H + offs_h[None, :]
        x_vals = tl.load(x_ptr_off,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)
        gate_ptr_off = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        g_vals = tl.load(gate_ptr_off,
                         mask=mask_r[:, None] & mask_h[None, :],
                         other=0.0).to(tl.float32)

        # Load LN weight & bias
        w_ln = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)  # float32
        b_ln = tl.load(ln_bias_ptr + offs_h, mask=mask_h, other=0.0)    # float32

        # Normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]  # fp32

        # LN scaling + bias, then apply gate
        act = (norm * w_ln[None, :] + b_ln[None, :]) * g_vals  # fp32

        # Cast to fp16 for dot
        act_fp16 = act.to(tl.float16)

        # For each d block compute dot
        for d_off in range(0, tl.cdiv(D, BLOCK_D)):
            offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
            mask_d = offs_d < D

            # Load out_weight_T slice: shape (BLOCK_H, BLOCK_D)
            w_out_ptr = out_weight_ptr + offs_h[:, None] * D + offs_d[None, :]
            w_out = tl.load(w_out_ptr,
                            mask=mask_h[:, None] & mask_d[None, :],
                            other=0.0)   # fp16

            # Accumulate
            acc += tl.dot(act_fp16, w_out)   # (R,D)
    # Store result
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        y_ptr_off = y_ptr + offs_r[:, None] * D + offs_d[None, :]
        tl.store(y_ptr_off,
                 acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])
```

"""
TriMul (outgoing) – Triton implementation
========================================

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16.
2️⃣ One fused linear projection to obtain the concatenated tensors
   `[left, right, left_gate, right_gate, out_gate]` (size 5·H).
3️⃣ Triton kernel `split_fuse_gate_kernel`  
   • splits the concatenated projection, applies the pair‑wise mask,  
   • evaluates the three sigmoid gates,  
   • writes the gated *left* and *right* tensors directly into the layout
     required by the batched GEMM: `[B·H, N, N]` (no extra permutes),  
   • stores the `out_gate` tensor in the original `[R, H]` layout
     (R = B·N·N) for the second LayerNorm.
4️⃣ Batched matrix multiplication
   `(B·H, N, N) @ (B·H, N, N)ᵀ`  →  `(B·H, N, N)`.
5️⃣ Triton kernel `fused_ln_gate_linear_kernel`  
   • row‑wise LayerNorm on the GEMM result,  
   • multiplies by `out_gate`,  
   • immediately projects back to `dim` with the transposed output
     weight `W_outᵀ` (shape H×dim).  This fuses the second LayerNorm,
     the out‑gate multiplication and the final linear projection, saving
     one large intermediate tensor and one extra kernel launch.
6️⃣ Reshape to `[B, N, N, dim]` and cast to float32.

The implementation keeps the heavy O(N³) batched GEMM in
`torch.bmm` (cuBLAS Tensor‑cores) and offloads only the
LayerNorm / gating / reshaping work to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # write normalized values  y = γ·(x‑μ)·rstd + β  (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]                # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Split the 5·H projection, apply mask, sigmoid gates,
#     and store left/right directly in [B·H, N, N] layout.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[B*H, N, N] (float16)
    right_ptr,         # mutable *[B*H, N, N] (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # mask broadcast (1.0 if no mask)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]                         # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    seq_sq = tl.int64(N) * tl.int64(N)                       # N·N

    # batch index b = r // (N·N)  and intra‑plane offset = r % (N·N)
    b_idx = offs_r // (seq_sq)                               # [BLOCK_R]
    offset_in_plane = offs_r - b_idx * seq_sq                 # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)    # [BLOCK_H]
        mask_h = offs_h < H

        # --- pointers to the five consecutive segments -----------------
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [R,Hblk]
        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        # --- load raw tensors -----------------------------------------
        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # --- sigmoid (fp32) -------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # --- apply mask and gates --------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # --- compute target row indices in the [B*H, N, N] layout -------
        # target row index = b*H + h
        bh = b_idx[:, None] * H + offs_h[None, :]           # [BLOCK_R, BLOCK_H]

        # flat offset for left/right tensors: bh * (N·N) + offset_in_plane
        dst_off = bh * seq_sq + offset_in_plane[:, None]    # [BLOCK_R, BLOCK_H]

        tl.store(left_ptr  + dst_off, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + dst_off, right,
                 mask=mask_r[:, None] & mask_h[None, :])

        # --- store out‑gate (still in [rows, H] layout) ---------------
        out_gate_dst = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]
        tl.store(out_gate_dst, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse second LayerNorm, out‑gate multiplication and final linear
# ----------------------------------------------------------------------
@triton.jit
def fused_ln_gate_linear_kernel(
    x_ptr,          # const   *[rows, H]   (float16)   – GEMM result
    gate_ptr,       # const   *[rows, H]   (float16)   – out_gate
    y_ptr,          # mutable *[rows, D]   (float16)   – final output
    ln_weight_ptr,  # const   *[H]        (float32)   – LN γ
    ln_bias_ptr,    # const   *[H]        (float32)   – LN β
    out_w_ptr,      # const   *[H, D]     (float16)   – W_outᵀ (transposed)
    rows: tl.int32,
    H: tl.int32,
    D: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # 1) compute mean & variance over H (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H
        x_ptr_off = x_ptr + offs_r[:, None] * H + offs_h[None, :]
        vals = tl.load(x_ptr_off,
                       mask=mask_r[:, None] & mask_h[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) For each output‑dim block, accumulate (LN(x)·gate)·W_outᵀ
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(D, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D
        acc = tl.zeros([BLOCK_R, BLOCK_D], dtype=tl.float32)

        for h_off in range(0, tl.cdiv(H, BLOCK_H)):
            offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
            mask_h = offs_h < H

            # load x and gate
            x_ptr_off = x_ptr + offs_r[:, None] * H + offs_h[None, :]
            x_vals = tl.load(x_ptr_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            g_ptr_off = gate_ptr + offs_r[:, None] * H + offs_h[None, :]
            g_vals = tl.load(g_ptr_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float32)

            # LN parameters (γ,β)
            w_ln = tl.load(ln_weight_ptr + offs_h, mask=mask_h, other=0.0)   # fp32
            b_ln = tl.load(ln_bias_ptr   + offs_h, mask=mask_h, other=0.0)   # fp32

            # normalize, apply LN scale/bias and out‑gate
            norm = (x_vals - mean[:, None]) * rstd[:, None]                # fp32
            act  = (norm * w_ln[None, :] + b_ln[None, :]) * g_vals        # fp32
            act_fp16 = act.to(tl.float16)                                  # (R, Hblock)

            # load slice of W_outᵀ  – shape (Hblock, Dblock)
            w_ptr = out_w_ptr + offs_h[:, None] * D + offs_d[None, :]       # [Hblk, Dblk]
            w_slice = tl.load(w_ptr,
                              mask=mask_h[:, None] & mask_d[None, :],
                              other=0.0)                                    # fp16

            # accumulate
            acc += tl.dot(act_fp16, w_slice)                               # (R, Dblk)

        # store the block result
        y_ptr_off = y_ptr + offs_r[:, None] * D + offs_d[None, :]
        tl.store(y_ptr_off,
                 acc.to(tl.float16),
                 mask=mask_r[:, None] & mask_d[None, :])

    """
    Forward pass of the outgoing Triangle‑Multiplicative (TriMul) update.

    Arguments
    ---------
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, [B, N, N, dim], float32
        - mask_tensor  : torch.Tensor, [B, N, N]  (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """

    # unpack arguments
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N            # total number of (b,i,k) rows
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()               # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_D = 32

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],       # γ  (float32)
        weights["norm.bias"],         # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # --------------------------------------------------------------
    # 2️⃣ Fused projection → 5·H channels (fp16)
    # --------------------------------------------------------------
    w_concat = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # [5·H, dim]
    proj = F.linear(x_norm, w_concat)       # [R, 5·H] fp16
    del x_norm, w_concat                    # free early

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates, store left/right in
    #     [B·H, N, N] layout and out_gate in [R, H]
    # --------------------------------------------------------------
    left_mat  = torch.empty((B * H, N, N), dtype=torch.float16, device=device)
    right_mat = torch.empty((B * H, N, N), dtype=torch.float16, device=device)
    out_gate  = torch.empty((R, H), dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)   # flatten mask
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left_mat,
        right_mat,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM   (B·H, N, N) @ (B·H, N, N)ᵀ
    # --------------------------------------------------------------
    # left_mat/right_mat are already in the required shape
    out_mat = torch.bmm(left_mat.view(B * H, N, N),
                        right_mat.view(B * H, N, N).transpose(1, 2))

    # reshape to [B, N, N, H]  for the second LayerNorm
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    out_flat = out.view(R, H).contiguous()

    # --------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out‑gate multiplication and final
    #     projection back to `dim`.
    # --------------------------------------------------------------
    # transposed output weight for efficient GEMM inside the kernel
    out_w_T = weights["to_out.weight"].t().contiguous().to(torch.float16)  # [H, dim]

    final_half = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_D = 32
    grid_fuse = (triton.cdiv(R, BLOCK_R),)
    fused_ln_gate_linear_kernel[grid_fuse](
        out_flat,                         # x_ptr
        out_gate,                         # gate_ptr
        final_half,                       # y_ptr
        weights["to_out_norm.weight"],    # LN γ  (float32)
        weights["to_out_norm.bias"],      # LN β  (float32)
        out_w_T,                          # W_outᵀ (float16) shape [H, dim]
        R,
        H,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # free intermediates
    del out_flat, out_gate, out, out_mat, left_mat, right_mat, out_w_T

    # --------------------------------------------------------------
    # 6️⃣ Cast back to fp32 and restore original layout
    # --------------------------------------------------------------
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
