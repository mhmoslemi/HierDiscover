        weights['right_proj.weight'],

        weights['right_gate.weight'],
        weights['out_gate.weight'],

    proj_flat = proj.view(R, 5*H)

    # Reshape left/right to shape (B,N,N,H)
    # Use view without permute:
    left_ = left.view(B, N, N, H)  # shape (B,N,N,H)
    right_ = right.view(B, N, N, H)

    # Need right perm = swap axes 1 and 2 (i <-> j) for multiplication.
    right_perm = right_.permute(0, 2, 1, 3)  # (B,N,N,H) -> (B,N,N,H) with i<->j

    # Now compute einsum: sum over k dimension (third axis of left_, second axis of right_perm). Using torch.einsum.
    # This will produce shape (B,N,N,H)
    out = torch.einsum('b i k h, b j k h -> b i j h', left_, right_perm)

    del left, right, left_, right_, right_perm

    # 5. Second LN + out_gate gating, fused.

    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=x.device)

    grid2 = (triton.cdiv(R, 128),) # using BLOCK_M=128 for second LN? Might unify.
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,   # gate per row H
        gated_norm,
        weights['to_out_norm.weight'],
        weights['to_out_norm.bias'],
        R,
        H,
        eps,
        BLOCK_M=128,
        BLOCK_D=32,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # 6. Final linear projection: from hidden H to dim

        gated_norm,
        weights['to_out.weight'].to(torch.float16),
        bias=None,
    )
    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final

"""
Optimized Triton implementation of the *outgoing* TriMul operator (AlphaFold‑3).
Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – performed with the
    highly‑optimized torch.nn.functional.layer_norm (fp16).

2️⃣  Single fused Linear that produces the five tensors required for the
    update: left‑proj, right‑proj, left‑gate, right‑gate, out‑gate
    (shape [R,5·H] where R = B·N·N).

3️⃣  Triton kernel `split_fuse_gate_kernel`  
    – splits the fused projection, applies the optional pair‑wise mask,  
      applies sigmoid to the three gate tensors, and writes the
      masked × gated *left*, *right* and *out_gate* tensors (fp16).

4️⃣  Tri‑multiplicative update:  
    left and right are reshaped to (B,N,N,H).  The right tensor is
    transposed on the two sequence axes and the contraction
    `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]` is performed
    with `torch.einsum` (batched GEMM on H100 tensor‑cores).

5️⃣  Triton kernel `layernorm_gate_kernel`  
    – row‑wise LayerNorm on the hidden‑dim tensor *out* and simultaneous
      element‑wise multiplication by *out_gate* (fused, fp16).

6️⃣  Final projection back to `dim` with a half‑precision Linear (cuBLAS).

Only the split‑gate, the second LayerNorm+gate, and (optionally) the
first LayerNorm are written in Triton – satisfying the requirement to
use custom kernels while letting cuBLAS handle the heavy matrix
multiplications.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 3️⃣ Split‑fused projection, mask, and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)           # [BLOCK_R]
    mask_r = offs_r < rows

    # Load optional mask (scalar per row) and broadcast over hidden dim
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension in chunks
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout of the fused projection:
        # [ left , right , left_gate , right_gate , out_gate ]   each ≡ H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]       # [R,H] offset for segment 0

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)                     # fp16
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid computed in fp32 for numerical stability
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left   = left_raw  * mask_val * left_gate
        right  = right_raw * mask_val * right_gate

        # store results
        left_dst_ptr     = left_ptr  + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr    = right_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,     left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,    right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,out_gate, mask=mask_r[:, None] & mask_h[None, :])

# ----------------------------------------------------------------------
# 5️⃣ Row‑wise LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (fp16)
    gate_ptr,       # const   *[rows, dim]   (fp16)
    y_ptr,          # mutable *[rows, dim]   (fp16)
    weight_ptr,     # const   *[dim]        (fp32)   γ
    bias_ptr,       # const   *[dim]        (fp32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # Compute mean & variance in fp32
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, apply γ, β and multiply by gate (all fp32 → fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load x and gate
        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]          # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    R = B * N * N                # total rows for row‑wise operations
    H = hidden_dim

            weights["right_gate.weight"],

    BLOCK_D = 32
    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # (dim, H)
        bias=None,
    )   # → [R, dim]  fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
