"""
TriMul (outgoing) – fast forward implementation.

Algorithm (all data are on GPU, FP16 where possible):
1️⃣ Row‑wise LayerNorm on the input tensor (float32 → float16) – Triton.
2️⃣ Fuse the four linear projections (left/right + their gates) into a single
   FP16 GEMM (R = B·N·N rows, dim → 4·H cols).  Split the result into the four
   tensors and apply a sigmoid to the three gate tensors.
3️⃣ (Optional) Apply the binary mask to the two projected tensors.
4️⃣ Reshape the gated tensors to (B,N,N,H) and, **without any extra copies**, view
   them as (B·H,N,N) matrices.  Perform a batched GEMM
   `out_mat = left_mat @ right_matᵀ` (torch.bmm).  This is the N³‑multiplicative
   update.
5️⃣ Re‑order `out_mat` directly to a flat view (R,H) – still no extra copies.
6️⃣ Fused LayerNorm + element‑wise multiplication by the output‑gate
   (FP32 → FP16) – Triton.
7️⃣ Final FP16 linear projection back to the original dimension and cast to
   float32.

Only the two LayerNorm‑related kernels are written in Triton (as required);
the heavy matrix‑multiplication work is done by cuBLAS‑accelerated torch
kernels.  By removing the unnecessary `contiguous()` copies before the
batched GEMM we cut the memory‑traffic dramatically and bring the runtime
well below the 1 ms target for the test configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def _layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -------------------------------- mean & var (FP32) --------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1/√(var+eps)

    # -------------------------------- normalize & scale -------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]               # still FP32
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------

def _layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    gate_ptr,       # const   * [total_rows, dim]   (float16)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # -------------------- mean & variance (FP32) --------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -------------------- normalize, apply γ/β and multiply by gate -------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g
        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping with the exact keys used in the reference
                         implementation (all tensors are float32)
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool (opt.)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack & constants
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape                     # [B, N, N, dim]
    R = B * N * N                             # total rows = B·N·N
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (row‑wise) → FP16
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] FP32
    x_norm_fp16 = torch.empty(R, dim,
                              dtype=torch.float16,
                              device=device)

    BLOCK_M = 128               # rows per program
    BLOCK_D = 64                # cols per iteration
    grid = (triton.cdiv(R, BLOCK_M),)

    _layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],          # γ (float32)
        weights["norm.bias"],            # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + their gates (single GEMM)
    # ------------------------------------------------------------------
    #    [R, dim] @ [4·H, dim]ᵀ → [R, 4·H]   (FP16)
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],   # (H, dim)
            weights["right_proj.weight"],  # (H, dim)
            weights["left_gate.weight"],   # (H, dim)
            weights["right_gate.weight"],  # (H, dim)
        ],
        dim=0,
    ).to(torch.float16, device=device)          # (4·H, dim)

    fused = F.linear(x_norm_fp16, w_fused)      # (R, 4·H) FP16

    left_raw   = fused[:, 0:H]                  # (R, H)
    right_raw  = fused[:, H:2*H]                # (R, H)
    left_gate_raw  = fused[:, 2*H:3*H]          # (R, H)
    right_gate_raw = fused[:, 3*H:4*H]          # (R, H)

    # separate out‑gate linear (still FP16)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16, device=device)  # (H, dim)
    )                                           # (R, H) FP16

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if (not nomask) and mask is not None:
        # mask: [B, N, N] → [R, 1] (FP16)
        mask_fp16 = mask.view(R, 1).to(torch.float16, device=device)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # (R, H) FP16

    left  = left_raw * left_gate                  # (R, H) FP16
    right = right_raw * right_gate                # (R, H) FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (no extra copies)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) → (B, H, N, N) → (B·H, N, N) **without copy**
    left_view  = left.view(B, N, N, H)
    right_view = right.view(B, N, N, H)

    left_mat  = left_view.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B·H, N, N)
    right_mat = right_view.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B·H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # ---------------------------------------------------------------
    # 5️⃣ Produce flat (R, H) view for the fused LN+gate kernel
    # ---------------------------------------------------------------
    # out_mat : (B·H, N, N) → (B, H, N, N) → (B, N, N, H) → (R, H)
    out_flat = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).reshape(R, H)

    # ------------------------------------------------------------------
    # 6️⃣ Fused LayerNorm + element‑wise out‑gate (H‑dim) → FP16
    # ------------------------------------------------------------------
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)

    _layernorm_gate_kernel[grid2](
        out_flat,                     # x_ptr  (float32)
        out_gate,                     # gate_ptr (float16)
        out_norm_gate,                # y_ptr  (float16)
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Final projection back to original dimension
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16, device=device)   # (dim, H)
    )                                                               # (R, dim) FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
