Algorithm (all heavy compute in FP16)

@triton.jit

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # FP32 load
        sum_val += tl.sum(vals,  axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalisation + γ/β -----------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                 # FP32
        tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Fuse LayerNorm (hidden dim) + element‑wise out‑gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim] (float16)
    gate_ptr,       # const   * [total_rows, dim] (float16)
    out_ptr,        # mutable * [total_rows, dim] (float16)
    w_ptr,          # const   * [dim]            (float32)  γ
    b_ptr,          # const   * [dim]            (float32)  β
    total_rows,     # int32   runtime
    dim,            # int32   runtime
    eps,            # float   compile‑time
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ---- reduction ----------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalise, γ/β and multiply by gate -------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[One, :]) * g_vals
        tl.store(out_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 0️⃣ Helper – launch a 1‑D grid with sensible block sizes
# ----------------------------------------------------------------------
def _launch_layernorm(x, weight, bias, eps=1e-5):
    """Row‑wise LayerNorm (FP32→FP16). Returns a FP16 tensor."""
    B, N, _, dim = x.shape
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim).contiguous()

    out = torch.empty_like(x_flat, dtype=torch.float16)

    # heuristics – dimensions are small (≤384) → use 64‑wide D‑tiles
    BLOCK_D = 64
    BLOCK_M = 256 if total_rows > 4096 else 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_kernel[grid](
        x_flat,
        out,
        weight,
        bias,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    return out


def _launch_layernorm_gate(x, gate, weight, bias, eps=1e-5):
    """Fused (LayerNorm + element‑wise gate) on FP16 tensors."""
    total_rows, dim = x.shape
    out = torch.empty_like(x, dtype=torch.float16)

    BLOCK_D = 64
    BLOCK_M = 256 if total_rows > 4096 else 128
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_gate_kernel[grid](
        x,
        gate,
        out,
        weight,
        bias,
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    return out


# ----------------------------------------------------------------------
# 2️⃣ Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of all linear / layer‑norm parameters
          (all stored as float32 tensors)
        - config_dict  : {"dim": int, "hidden_dim": int,
                          "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor : shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask     = cfg.get("nomask", True)
    eps        = 1e-5
    device     = x.device
    B, N, _, _ = x.shape
    R = B * N * N               # total rows for the flattened view
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over `dim`  (FP32 → FP16)
    # ------------------------------------------------------------------
    x_norm_fp16 = _launch_layernorm(x, weights["norm.weight"], weights["norm.bias"], eps)

    # ------------------------------------------------------------------
    # 2️⃣  Fuse the four projections + out‑gate into one GEMM
    # ------------------------------------------------------------------
    #   W_fused shape → [5·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
            weights["out_gate.weight"],    # [H, dim]
        ],
        dim=0,
    ).to(torch.float16, non_blocking=True)

    # Linear on normalized input (FP16)
    fused = F.linear(x_norm_fp16, w_fused)          # [R, 5·H]

    left_raw       = fused[:, 0:H]                     # [R, H]
    right_raw      = fused[:, H:2 * H]
    left_gate_raw  = fused[:, 2 * H:3 * H]
    right_gate_raw = fused[:, 3 * H:4 * H]
    out_gate_raw   = fused[:, 4 * H:5 * H]

    # ------------------------------------------------------------------
    # 3️⃣  Mask (if any) & sigmoid gates (in‑place)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16, non_blocking=True).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()          # already FP16

    left  = left_raw  * left_gate     # [R, H] FP16
    right = right_raw * right_gate    # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣  N³ multiplicative update (batched matmul)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) – already contiguous
    left  = left.view(B, N, N, H)
    right = right.view(B, N, N, H)

    # bring hidden dim into the batch axis for cuBLAS (B·H, N, N)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N)

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 5️⃣  Fuse second LayerNorm with out‑gate multiplication
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()       # [R, H] FP16
    out_gate_flat = out_gate                     # [R, H] FP16

    out_norm_gate_fp16 = _launch_layernorm_gate(
        out_flat,
        out_gate_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps,
    )                                            # [R, H] FP16

    # ------------------------------------------------------------------
    # 6️⃣  Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate_fp16,
        weights["to_out.weight"].to(torch.float16, non_blocking=True),
    )                                            # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
