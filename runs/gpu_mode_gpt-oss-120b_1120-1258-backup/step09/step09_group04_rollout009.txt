"""
TriMul (outgoing) – high‑performance forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the last dimension (dim) → FP16.
2️⃣ Fuse the four linear layers (left_proj, right_proj, left_gate, right_gate) into
   a single GEMM:  X_norm @ W_fused.T → [R, 4·H]  (R = B·N·N, H = hidden_dim).
   Split the result into the four tensors.
3️⃣ Compute the output‑gate with a second linear (still FP16) and apply sigmoid.
4️⃣ (Optional) Apply binary mask to the two projected tensors.
   Apply sigmoid to the three gate tensors (in‑place) and multiply:
        left  = left_raw  * sigmoid(left_gate)
        right = right_raw * sigmoid(right_gate)
5️⃣ Batched multiplicative update  out = left @ rightᵀ  (B·H independent GEMMs).
   This is performed with a single torch.matmul that internally uses Tensor‑Core
   FP16 GEMMs.
6️⃣ Fuse the second LayerNorm and the element‑wise multiplication by the out‑gate
   into one Triton kernel (FP32 → FP16).
7️⃣ Final linear projection back to the original dimensionality (FP16 GEMM) and
   cast to float32.

All heavy element‑wise work (both LayerNorms) is written in Triton; the matrix
multiplications use the highly‑optimised cuBLAS kernels exposed through
torch.nn.functional.linear / torch.matmul.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (input FP32 → output FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # cols per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ----------- pass 1: compute mean & variance (FP32) -------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # [M,D]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)               # FP32
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                               # 1 / sqrt(var+eps)

    # ----------- pass 2: normalise, apply γ/β (still FP32) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]          # (x-μ)/σ

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]                   # still FP32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: fused LayerNorm + element‑wise out‑gate (FP32 → FP16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   * [total_rows, dim]   (float32)
    gate_ptr,       # const   * [total_rows, dim]   (float16)
    y_ptr,          # mutable * [total_rows, dim]   (float16)
    weight_ptr,     # const   * [dim]              (float32)
    bias_ptr,       # const   * [dim]              (float32)
    total_rows,     # int32   (runtime)
    dim,            # int32   (runtime)
    eps,            # fp32    (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ----------- pass 1: mean & variance (ignore gate) -------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ----------- pass 2: normalise, apply γ/β and multiply gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # raw activation (FP32)
        x = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        # gate value (FP16 → FP32)
        g = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                    mask=mask_m[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        norm = (x - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = (norm * w[None, :] + b[None, :]) * g              # FP32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fuse left/right projections + gates into a single GEMM
    # ------------------------------------------------------------------
    # Concatenate the four weight matrices: [H, dim] each → [4·H, dim]
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],    # [H, dim]
            weights["right_proj.weight"],   # [H, dim]
            weights["left_gate.weight"],    # [H, dim]
            weights["right_gate.weight"],   # [H, dim]
        ],
        dim=0,
    ).to(torch.float16)                     # [4·H, dim]

    fused = F.linear(x_norm_fp16, w_fused)   # FP16, shape [R, 4·H]

    left_raw   = fused[:, 0:H]                # [R, H]
    right_raw  = fused[:, H:2*H]              # [R, H]
    left_gate_raw  = fused[:, 2*H:3*H]        # [R, H]
    right_gate_raw = fused[:, 3*H:4*H]        # [R, H]

    # output‑gate linear (still FP16)
    out_gate_raw = F.linear(
        x_norm_fp16,
        weights["out_gate.weight"].to(torch.float16)   # [H, dim]
    )   # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Apply mask (if any) and sigmoid gates
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] → [R, 1] (broadcast over hidden dim)
        mask_fp16 = mask.to(torch.float16).reshape(R, 1)
        left_raw  = left_raw * mask_fp16
        right_raw = right_raw * mask_fp16

    # In‑place sigmoid (FP16 → FP16)
    left_gate  = left_gate_raw.sigmoid_()
    right_gate = right_gate_raw.sigmoid_()
    out_gate   = out_gate_raw.sigmoid_()               # [R, H] (FP16)

    # gated projections
    left  = left_raw * left_gate       # [R, H] FP16
    right = right_raw * right_gate     # [R, H] FP16

    # ------------------------------------------------------------------
    # 4️⃣ Batched N³ multiplicative update (B·H independent GEMMs)
    # ------------------------------------------------------------------
    # reshape to (B, N, N, H) → (B, H, N, N) for batched matmul
    left = left.view(B, N, N, H).permute(0, 3, 1, 2)   # (B, H, N, N)
    right = right.view(B, N, N, H).permute(0, 3, 1, 2) # (B, H, N, N)

    # out = left @ rightᵀ   (batch‑wise GEMM, FP16 Tensor‑Core)
    out = torch.matmul(left, right.transpose(-2, -1))  # (B, H, N, N)

    # back to (B, N, N, H)
    out = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Fused LayerNorm + element‑wise out‑gate over hidden dim H
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()          # [R, H] FP16
    out_gate_flat = out_gate                         # already [R, H] FP16
    out_norm_gate = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)

    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_norm_gate,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # 6️⃣ Final linear projection back to original dim
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gate,
        weights["to_out.weight"].to(torch.float16)   # [dim, H]
    )   # [R, dim] FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)
    return final
```

This implementation follows the reference `TriMul` forward pass while:

* Fusing four linear projections into one GEMM.
* Performing both LayerNorms (the first and the second combined with the out‑gate) in custom Triton kernels to keep the data in FP16 and avoid extra memory traffic.
* Using a single `torch.matmul` for the massive `(B·H, N, N)` batched matrix multiplication; on an H100 this call maps to highly‑optimised Tensor‑Core kernels.
* Keeping all intermediate tensors in FP16 and only converting back to float32 for the final output, meeting the required API and precision constraints.
