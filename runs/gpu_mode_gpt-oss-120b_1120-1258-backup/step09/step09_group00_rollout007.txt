"""
TriMul (outgoing) – fast Triton‑accelerated forward pass
=======================================================

The operator works on a 4‑D tensor `x` of shape [B, N, N, C] (C == dim).
The forward pass consists of

1️⃣ Row‑wise LayerNorm on the last dimension (fp32 → fp16).
2️⃣ A single fused linear projection that simultaneously computes
   * left   = x_norm @ W_left
   * right  = x_norm @ W_right
   * left_gate  = sigmoid(x_norm @ W_left_gate)
   * right_gate = sigmoid(x_norm @ W_right_gate)
   * out_gate   = sigmoid(x_norm @ W_out_gate)

   The five projected tensors are packed into one matrix and processed
   by a custom Triton kernel that also applies the optional pair‑wise mask
   and the three sigmoid gates.

3️⃣ Batched half‑precision matrix multiplication on the hidden dimension:
   For every batch b and hidden channel h we compute
       out[b, h, i, j] = Σ_k left[b, h, i, k] * right[b, h, j, k]
   This is exactly a batch of (B·H) GEMMs of size (N, N) which cuBLAS
   (via `torch.matmul`) executes with Tensor‑cores.

4️⃣ Row‑wise LayerNorm on the hidden‑dim tensor, multiplication by `out_gate`,
   and a final linear projection back to `dim`.  The LayerNorm+gate step
   is fused in a second Triton kernel; the linear projection is performed
   with `torch.nn.functional.linear` (again FP16).

All heavy per‑row work (LayerNorms, masking, sigmoids) is done in Triton,
while the large N³ GEMM is delegated to cuBLAS, which is already optimal
on an H100.  Mixed‑precision (FP16) is used everywhere except the
LayerNorm parameters (stored in FP32) to keep numerical stability.

The implementation below follows the algorithm described above and has
been tuned (block sizes, warps) for the target shapes (B∈{1,2},
N∈{128,256,512,1024}, dim∈{128,384}, hidden_dim=128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)  γ
    bias_ptr,       # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- mean / variance (accumulate in fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- write normalized y = γ·(x‑μ)·rstd + β (fp16) ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)  # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)  # β fp32

        out_fp16 = (norm * w[None, :] + b[None, :]).to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    left_ptr,          # mutable *[rows, H]   (float16)
    right_ptr,         # mutable *[rows, H]   (float16)
    out_gate_ptr,      # mutable *[rows, H]   (float16)
    mask_ptr,          # const   *[rows]    (float16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # optional mask broadcast
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                 # [BLOCK_R, 1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R,H] offsets for seg0

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw    = tl.load(left_raw_ptr,    mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_raw   = tl.load(right_raw_ptr,   mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        left_g_raw  = tl.load(left_gate_ptr,   mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        right_g_raw = tl.load(right_gate_ptr, mask=mask_r[:, None] & mask_h[None, :], other=0.0)
        out_g_raw   = tl.load(out_gate_raw_ptr,mask=mask_r[:, None] & mask_h[None, :], other=0.0)

        # sigmoid in fp32 for better accuracy
        lg_f32 = left_g_raw.to(tl.float32)
        left_gate = (1.0 / (1.0 + tl.exp(-lg_f32))).to(tl.float16)

        rg_f32 = right_g_raw.to(tl.float32)
        right_gate = (1.0 / (1.0 + tl.exp(-rg_f32))).to(tl.float16)

        og_f32 = out_g_raw.to(tl.float32)
        out_gate = (1.0 / (1.0 + tl.exp(-og_f32))).to(tl.float16)

        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        left_ptr  = left_ptr  + offs_r[:, None] * H + offs_h[None, :]
        right_ptr = right_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_gate_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_ptr,     left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr,    right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Row‑wise LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (float16)
    gate_ptr,       # const   *[rows, dim]   (float16)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # ---------- mean / variance (fp32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- normalized * weight + bias, then *gate ----------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x and gate
        x_vals = tl.load(x_ptr   + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr+ offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]    # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals
        y_fp16 = y_fp32.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with all Linear/LayerNorm parameters (float32)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0️⃣ unpack
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim         = cfg["dim"]
    hidden_dim  = cfg["hidden_dim"]
    nomask      = cfg.get("nomask", True)
    eps         = 1e-5
    device      = x.device

    B, N, _, _ = x.shape               # x: [B, N, N, dim]
    R = B * N * N                       # total rows seen by row‑wise kernels
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 -> fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256          # rows per program
    BLOCK_D = 32           # channels per program
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],   # γ (float32)
        w["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused linear projection (5 * H outputs)
    # ------------------------------------------------------------------
    # concatenate the five weight matrices (shape: [5*H, dim])
    w_concat = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                       # [5*H, dim]

    proj = F.linear(x_norm, w_concat)         # [R, 5*H] in fp16
    del w_concat, x_norm                      # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, apply mask and sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if nomask:
        # Kernel will ignore mask pointer
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        HAS_MASK = 0
    else:
        # mask: bool/float → fp16, flatten to [R]
        mask_fp16 = mask.to(torch.float16).reshape(R)
        mask_ptr = mask_fp16
        HAS_MASK = 1

    BLOCK_R = 256
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    #    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]
    #    Implemented as a batch of (B·H) GEMMs of size (N,N)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H] then bring H into batch dimension
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2)   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2)  # [B, H, N, N]

    # torch.matmul on 4‑D tensors performs a batched GEMM over the
    # last two dimensions (including the transpose on the RHS)
    out = torch.matmul(left, right.transpose(-1, -2))   # [B, H, N, N]

    # back to [B, N, N, H] for the second LayerNorm
    out = out.permute(0, 2, 3, 1).contiguous()          # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,
        gated_norm,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
