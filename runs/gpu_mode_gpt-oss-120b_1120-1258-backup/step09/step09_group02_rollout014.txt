@triton.jit
def norm_mul_kernel(
    inp_ptr,   # fp16 [R, H]
    gate_ptr,  # fp16 [R, H] (out_gate)
    out_ptr,   # fp16 [R, H]
    weight_ptr, # fp32 [H] gamma
    bias_ptr,   # fp32 [H] beta
    rows: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    # compute mean & var per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H
        ptr = inp_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / H
    var = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # normalize and multiply gate, then apply gamma/beta
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H
        ptr = inp_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32
        gate = tl.load(gate_ptr + offs_m[:, None] * H + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        out = norm * gate   # fp32

        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        beta = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)     # fp32
        out = out * gamma[None, :] + beta[None, :]

        out_fp16 = out.to(tl.float16)
        out_ptr = out_ptr + offs_m[:, None] * H + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])
```

"""
high‑performance "outgoing" TriMul (AlphaFold3)

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the flattened input (B·N·N × dim) – use
   torch.nn.functional.layer_norm (fast fused CUDA kernel).

2️⃣ Fused linear projection that simultaneously computes the five
   weight matrices (left, right, left_gate, right_gate, out_gate):
        X_norm @ W_fusedᵀ → (R, 5·H)   with   R = B·N·N.
   The heavy GEMM is done by cuBLAS (FP16) which is optimal on H100.

3️⃣ Triton kernel `split_fuse_gate_kernel` reads the fused tensor,
   applies the optional pair‑wise mask, evaluates the three sigmoid
   gates (in FP32 for numerical stability) and writes the three
   result tensors `left`, `right` and `out_gate` (all FP16).

4️⃣ Batched matrix‑multiplication over the hidden dimension:
        left_mat  = left .view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
        right_mat = right.view(...).permute(...).contiguous()
        out_mat   = torch.bmm(left_mat, right_mat.transpose(1, 2))
   (B·H independent GEMMs of size N×N, executed by cuBLAS).

5️⃣ Triton kernel `norm_mul_kernel` performs the second
   row‑wise LayerNorm on `out_mat`, applies the learned γ/β,
   and finally multiplies element‑wise by `out_gate` (gate after
   normalization).  This replaces two separate kernels (norm + mul).

6️⃣ Final linear projection back to the original `dim` (FP16 GEMM) and
   cast to FP32 for the returned tensor.

Only the two kernels (steps 3 & 5) are custom – the rest uses
highly tuned cuBLAS/cuDNN kernels, giving sub‑millisecond runtimes
on the target configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Kernel 1: split fused projection, apply mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const  *[rows, 5*H]  (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const  *[rows]      (fp16) – ignored if HAS_MASK==0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,      # rows per program
    BLOCK_H: tl.constexpr,      # hidden dim per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # optional mask (scalar per row, broadcast over H)
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)      # [BLOCK_R]
        m = m[:, None]                                              # [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # iterate over hidden dimension
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)            # [BLOCK_H]
        mask_h = offs_h < H

        # layout of the fused weight matrix:
        # 0 .. H-1               -> left
        # H .. 2H-1              -> right
        # 2H .. 3H-1             -> left_gate
        # 3H .. 4H-1             -> right_gate
        # 4H .. 5H-1             -> out_gate
        row_off = offs_r[:, None] * (5 * H)                         # [R,1]
        col_off = offs_h[None, :]                                   # [1,H]

        left_raw   = tl.load(proj_ptr + row_off + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)                            # fp16
        right_raw  = tl.load(proj_ptr + row_off + H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        left_g_raw = tl.load(proj_ptr + row_off + 2 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        right_g_raw= tl.load(proj_ptr + row_off + 3 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)
        out_g_raw  = tl.load(proj_ptr + row_off + 4 * H + col_off,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0)

        # sigmoid in FP32 (more stable)
        lg = left_g_raw.to(tl.float32)
        left_gate  = (1.0 / (1.0 + tl.exp(-lg))).to(tl.float16)

        rg = right_g_raw.to(tl.float32)
        right_gate = (1.0 / (1.0 + tl.exp(-rg))).to(tl.float16)

        og = out_g_raw.to(tl.float32)
        out_gate = (1.0 / (1.0 + tl.exp(-og))).to(tl.float16)

        # apply mask & gates
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # store
        tl.store(left_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr    + offs_r[:, None] * H + offs_h[None, :],
                 right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + offs_r[:, None] * H + offs_h[None, :],
                 out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Kernel 2: second LayerNorm + multiplication by out_gate
# ----------------------------------------------------------------------
@triton.jit
def norm_mul_kernel(
    inp_ptr,          # const  *[rows, H]   (fp16)
    gate_ptr,         # const  *[rows, H]   (fp16) – out_gate
    out_ptr,          # mutable *[rows, H]  (fp16)
    weight_ptr,       # const  *[H]        (fp32) – γ
    bias_ptr,         # const  *[H]        (fp32) – β
    rows: tl.int32,
    H: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # hidden dim per program
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---------- 1) compute mean & variance (FP32) ----------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < H
        ptr = inp_ptr + offs_m[:, None] * H + offs_d[None, :]   # [M,D]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)                # fp32
        sum_val += tl.sum(vals, axis=1)            # [M]
        sum_sq  += tl.sum(vals * vals, axis=1)     # [M]

    mean = sum_val / H
    var  = sum_sq / H - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---------- 2) normalize, apply γ/β, then multiply by out_gate ----------
    for d_off in range(0, tl.cdiv(H, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < H
        ptr = inp_ptr + offs_m[:, None] * H + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x-mean) * rstd  -> normalized value (FP32)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # affine transform γ,β (FP32)
        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        beta  = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)
        out_fp32 = norm * gamma[None, :] + beta[None, :]

        # multiply by out_gate (FP16) – gate is still FP16
        gate = tl.load(gate_ptr + offs_m[:, None] * H + offs_d[None, :],
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        out_fp32 = out_fp32 * gate

        # store back as FP16
        out_ptr + offs_m[:, None] * H + offs_d[None, :]  # address
        tl.store(out_ptr + offs_m[:, None] * H + offs_d[None, :],
                 out_fp32.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (no gradients).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
            - weights_dict : dict with all linear / layer‑norm parameters
                            (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (torch implementation, fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                       # [R, dim] fp32
    x_norm = torch.nn.functional.layer_norm(
        x_flat,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    ).to(torch.float16)                                        # [R, dim] fp16

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection into five heads (fp16 GEMM)
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                         # (5*H, dim)

    proj = F.linear(x_norm, w_fused)                            # [R, 5*H] fp16
    del w_fused, x_norm                                         # free early

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask & sigmoid gates (custom Triton kernel)
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        # no mask → kernel will ignore the mask pointer
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
    else:
        mask_flat = mask.to(torch.float16).reshape(R)          # [R]
        mask_ptr = mask_flat
        has_mask = 1

    BLOCK_R = 256                     # rows per program (256*8 = 2048 threads → OK)
    BLOCK_H = 4                       # hidden dim per program (256*4 = 1024 threads)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over hidden dimension (B·H independent N×N mul)
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] then move H into batch dim
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to [B, N, N, H]  (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + element‑wise multiplication with out_gate
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()        # [R, H] fp16
    out_norm = torch.empty_like(out_flat, dtype=torch.float16)

    BLOCK_M = 256
    BLOCK_D = 4
    grid2 = (triton.cdiv(R, BLOCK_M),)
    norm_mul_kernel[grid2](
        out_flat,
        out_gate,
        out_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del out_flat, out, out_gate

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_norm,                                   # [R, H] fp16
        weights["to_out.weight"].to(torch.float16) # [dim, H] fp16
    )                                                # [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```
