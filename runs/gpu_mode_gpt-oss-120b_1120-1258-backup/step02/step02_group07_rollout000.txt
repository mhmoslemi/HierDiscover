"""
TriMul (outgoing) – GPU‑accelerated forward pass.

The heavy part of the TriMul operator is the N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which can be expressed as a batched matrix multiplication
    out_d = left_d @ right_dᵀ   (for each batch and hidden channel d).

This implementation:
* Performs LayerNorms, Linear projections and gating with AMP (fp16) for speed.
* Packs the batch and hidden dimensions together and launches a custom Triton
  kernel that computes a batched GEMM in fp16 with fp32 accumulation.
* The kernel tiles the matrices (default 128×128×32) and writes the result
  in fp16; the rest of the network (norm, gating, final linear) runs in
  fp16 and the final tensor is returned as fp32.

Only the GEMM is implemented in Triton; the rest uses highly‑optimised
PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched GEMM: C = A @ B   (A: (M,K), B: (K,N)) for many batches.
    A, B, C are stored as (batch, M, K), (batch, K, N) and (batch, M, N).
    All tensors are fp16, the accumulator is fp32.
    """
    pid_m = tl.program_id(0)      # block row index
    pid_n = tl.program_id(1)      # block column index
    pid_b = tl.program_id(2)      # batch index (batch*hidden)

    #--- offsets within a block -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    #--- accumulator -----------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    #--- main loop over K dimension --------------------------------------------
    num_k_tiles = tl.cdiv(K, BLOCK_K)
    for tile_k in range(num_k_tiles):
        k_base = tile_k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        # pointers for the current tile
        a_ptrs = (A_ptr
                  + pid_b * batch_stride_a
                  + offs_m[:, None] * stride_am
                  + offs_k[None, :] * stride_ak)
        b_ptrs = (B_ptr
                  + pid_b * batch_stride_b
                  + offs_k[:, None] * stride_bk
                  + offs_n[None, :] * stride_bn)

        # load with masks for partial tiles
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0, dtype=tl.float16)

        # accumulate
        acc += tl.dot(a, b)

    #--- write back -------------------------------------------------------------
    c_ptrs = (C_ptr
              + pid_b * batch_stride_c
              + offs_m[:, None] * stride_cm
              + offs_n[None, :] * stride_cn)

    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32) with keys
                         ["norm.weight", "norm.bias",
                          "left_proj.weight", "right_proj.weight",
                          "left_gate.weight", "right_gate.weight",
                          "out_gate.weight",
                          "to_out_norm.weight", "to_out_norm.bias",
                          "to_out.weight"]
        - config       : dict containing at least {"dim": int,
                                                 "hidden_dim": int,
                                                 "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    # ---------------------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------------------------------------------------------
    # Load model weights (already on the correct device)
    # ---------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---------------------------------------------------------------------
    # Mixed‑precision (fp16) forward up to the heavy contraction
    # ---------------------------------------------------------------------
    with torch.autocast(device_type="cuda", dtype=torch.float16):
        # LayerNorm over the channel dimension `dim`
        x = F.layer_norm(input_tensor, (dim,),
                         weight=norm_weight, bias=norm_bias)

        # Linear projections (no bias)
        left = F.linear(x, left_proj_weight)       # [B, N, N, hidden_dim]
        right = F.linear(x, right_proj_weight)

        # Optional mask

            left = left * mask_f
            right = right * mask_f

        # Gating scalars (sigmoid)
        left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
        right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
        out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

        # Apply gates before the contraction
        left = left * left_gate
        right = right * right_gate

    # ---------------------------------------------------------------------
    # Prepare tensors for the batched GEMM kernel
    # ---------------------------------------------------------------------
    # left/right are now fp16 tensors (because of autocast)
    # We need right transposed on the last two axes for the matmul.
    left_mat = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_mat = right.permute(0, 3, 2, 1).contiguous()  # [B, hidden_dim, N, N] (k,j)

    B, H, N, _ = left_mat.shape
    batch_dim = B * H                         # flattened batch (batch * hidden)

    # Flatten the leading two dimensions so that each GEMM receives a (N, N) matrix
    left_flat = left_mat.view(batch_dim, N, N)
    right_flat = right_mat.view(batch_dim, N, N)

    # Allocate output buffer (fp16)
    out_flat = torch.empty_like(left_flat)

    # Strides for the flattened tensors (they are contiguous)
    stride_am = left_flat.stride(1)   # stride for moving along M (row) = N
    stride_ak = left_flat.stride(2)   # stride for moving along K (col) = 1
    stride_bk = right_flat.stride(1)  # also N
    stride_bn = right_flat.stride(2)  # 1
    stride_cm = out_flat.stride(1)    # N
    stride_cn = out_flat.stride(2)    # 1

    batch_stride_a = left_flat.stride(0)   # N * N
    batch_stride_b = right_flat.stride(0)  # N * N
    batch_stride_c = out_flat.stride(0)    # N * N

    # ---------------------------------------------------------------------
    # Launch the Triton batched GEMM kernel
    # ---------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_dim,
    )
    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,                     # M, N, K (all equal to seq_len)
        stride_am, stride_ak,
        stride_bk, stride_bn,
        stride_cm, stride_cn,
        batch_stride_a, batch_stride_b, batch_stride_c,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------------
    # Remaining post‑processing (LayerNorm, gating, final linear)
    # ---------------------------------------------------------------------
    with torch.autocast(device_type="cuda", dtype=torch.float16):
        out = F.layer_norm(out, (hidden_dim,),
                           weight=to_out_norm_weight,
                           bias=to_out_norm_bias)
        out = out * out_gate
        out = F.linear(out, to_out_weight)   # back to `dim`

    # Return FP32 as required by the API
    return out.to(torch.float32)
```

This implementation respects the required signature, uses a custom Triton kernel for the N³ contraction, and keeps the rest of the pipeline in mixed‑precision PyTorch for maximal throughput on an H100.
