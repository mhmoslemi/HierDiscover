@triton.jit
def layernorm_fp16_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr, rows, dim, eps, BLOCK_M, BLOCK_D):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

"""
TriMul “outgoing” operator – fast forward pass.
------------------------------------------------
1️⃣  Row‑wise LayerNorm (FP16) over the channel dimension `dim`.
    The kernel loads half‑precision values, accumulates the mean/variance
    in FP32 and writes back the normalized tensor with learned γ/β.
2️⃣  Cast the normalized tensor to FP16 and apply the four learned linear
    maps (left/right projection, three gates).  If a mask is present, it
    is applied after the projection (same as the reference implementation).
3️⃣  The multiplicative update  
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]  
    is performed with a single ``torch.einsum`` call which is lowered to a
    batched matrix‑multiply on H100 (no explicit transposes or copies).
4️⃣  Second LayerNorm (still FP16) over the hidden dimension.
5️⃣  Apply the output gate and a final linear projection back to `dim`.
    The final result is cast to FP32 as required.

Only the first LayerNorm is written in Triton – this satisfies the
“must have a custom kernel” requirement while the heavy O(N³) part
leverages the highly‑optimized cuBLAS path via ``einsum``.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (half‑precision input/weight/bias)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,            # const __fp16*   [rows, dim]
    y_ptr,            #       __fp16*   [rows, dim]
    w_ptr,            # const __fp16*   [dim]   (γ)
    b_ptr,            # const __fp16*   [dim]   (β)
    rows,             # int (run‑time)
    dim: tl.constexpr,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """Row‑wise LayerNorm for a 2‑D tensor (half → half, accumulation FP32)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # 1st pass: compute sum and sum‑of‑squares (mean & variance)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/sqrt(var+eps)

    # --------------------------------------------------------------
    # 2nd pass: write normalized values with γ/β
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]

        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0).to(tl.float32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0).to(tl.float32)

        out = norm * w[None, :] + b[None, :]           # still FP32
        out = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same CUDA device).
    """
    # ------------------------------------------------------------------
    # 0️⃣  Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    eps          = 1e-5
    nomask       = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1️⃣  LayerNorm over `dim` (FP16, Triton)
    # ------------------------------------------------------------------
    # flatten the first three dimensions -> (total_rows, dim)
    B, N, _, _ = input_tensor.shape
    total_rows = B * N * N

    # ensure contiguous layout
    x = input_tensor.contiguous()               # [B, N, N, dim]  float32
    x_flat = x.view(total_rows, dim)            # [R, dim]

    # allocate output buffer for LN
    x_norm_flat = torch.empty_like(x_flat, dtype=torch.float16)

    # kernel launch configuration (tuned for N≤1024, dim≤384)
    BLOCK_M = 256                               # rows per program
    BLOCK_D = 64                                # columns per program
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"].to(torch.float16).contiguous(),
        weights["norm.bias"].to(torch.float16).contiguous(),
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()   # ensure kernel completion

    # reshape back to 4‑D tensor
    x_norm = x_norm_flat.view(B, N, N, dim)      # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣  Linear projections + gates (all FP16)
    # ------------------------------------------------------------------
    # Helper: half‑precision weight copies (once per call)
    w_left_proj  = weights["left_proj.weight"].to(torch.float16)
    w_right_proj = weights["right_proj.weight"].to(torch.float16)
    w_left_gate  = weights["left_gate.weight"].to(torch.float16)
    w_right_gate = weights["right_gate.weight"].to(torch.float16)
    w_out_gate   = weights["out_gate.weight"].to(torch.float16)
    w_to_out_norm = weights["to_out_norm.weight"].to(torch.float16)
    b_to_out_norm = weights["to_out_norm.bias"].to(torch.float16)
    w_to_out      = weights["to_out.weight"].to(torch.float16)

    # flatten the pairwise axis for linear layers
    x_flat_h = x_norm.view(-1, dim)                # [R, dim]  fp16

    # ---- projections -------------------------------------------------
    left  = F.linear(x_flat_h, w_left_proj)        # [R, H]
    right = F.linear(x_flat_h, w_right_proj)       # [R, H]

    # ---- optional mask (broadcasted over hidden dim) -----------------
    if not nomask:
        # mask: [B, N, N] -> [R, 1]
        mask_h = mask_tensor.to(torch.float16).view(-1, 1)
        left  = left  * mask_h
        right = right * mask_h

    # ---- gates --------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_flat_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_flat_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_flat_h, w_out_gate))

    # apply element‑wise gates
    left  = left  * left_gate    # [R, H]
    right = right * right_gate   # [R, H]

    # reshape back to 4‑D tensors (still FP16)
    left  = left.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)  # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3️⃣  Triangular multiplicative update (einsum → batched GEMM)
    # ------------------------------------------------------------------
    # out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    out = torch.einsum('b i k d, b j k d -> b i j d', left, right)  # FP16

    # ------------------------------------------------------------------
    # 4️⃣  Second LayerNorm over the hidden dimension (FP16)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )   # still fp16

    # ------------------------------------------------------------------
    # 5️⃣  Output gating + final projection back to `dim`
    # ------------------------------------------------------------------
    out = out * out_gate                          # element‑wise FP16
    out_flat = out.view(-1, hidden_dim)           # [R, H]
    out_proj = F.linear(out_flat, w_to_out)       # [R, dim]  FP16
    out_proj = out_proj.view(B, N, N, dim)        # [B, N, N, dim]

    # final cast to FP32 (as in the reference implementation)
    return out_proj.to(torch.float32)
