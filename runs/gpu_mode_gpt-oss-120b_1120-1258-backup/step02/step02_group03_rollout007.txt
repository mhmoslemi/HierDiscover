"""
TriMul (outgoing) – high‑performance Triton implementation.

Algorithm
---------
1️⃣  Layer‑norm on the channel dimension (`dim`).
2️⃣  Fuse the five linear projections (left, right, left‑gate, right‑gate,
    out‑gate) into a *single* half‑precision GEMM:
       combined = x_h @ [W_left;W_right;W_left_gate;W_right_gate;W_out_gate]ᵀ
   and then slice the result.
3️⃣  Apply the optional pairwise mask (broadcast on the hidden dimension) *before*
    the element‑wise gates, then multiply by the sigmoid gates.
4️⃣  Compute the heavy 𝑁³ update with a custom Triton kernel:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   The kernel performs a batched FP16 mat‑mul (A @ Bᵀ) with FP32 accumulation.
   Autotuning selects block sizes that best exploit H100 tensor‑cores.
5️⃣  Reshape back to `[B,N,N,hidden]`, promote to FP32 and apply a second
    layer‑norm over `hidden`.
6️⃣  Multiply by the out‑gate, then a final half‑precision linear projection
    back to the original `dim`.  The final tensor is returned as FP32.

All heavy arithmetic (linear maps + N³ update + final projection) runs in
FP16, while the two layer‑norms run in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton kernel for the batched mat‑mul  C = A @ Bᵀ
# ----------------------------------------------------------------------
@triton.autotune(
    configs=[
        triton.Config({"BLOCK_M": 64, "BLOCK_N": 64, "BLOCK_K": 32}, num_warps=4),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 32}, num_warps=8),
        triton.Config({"BLOCK_M": 128, "BLOCK_N": 128, "BLOCK_K": 64}, num_warps=8),
        triton.Config({"BLOCK_M": 256, "BLOCK_N": 64, "BLOCK_K": 32}, num_warps=8),
    ],
    key=["M", "N", "K"],
)
@triton.jit
def _batched_matmul_kernel(
    A,
    B,
    C,
    M,
    N,
    K,
    stride_Ab,
    stride_Am,
    stride_Ak,
    stride_Bb,
    stride_Bk,
    stride_Bn,
    stride_Cb,
    stride_Cm,
    stride_Cn,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched fp16 GEMM (C = A @ B) with FP32 accumulation.

    A : (batch, M, K)   – row‑major
    B : (batch, K, N)   – row‑major (already transposed)
    C : (batch, M, N)
    """
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_b = tl.program_id(2)  # batch (merged B·hidden)

    # Offsets for the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers for this block (row‑major layout)
    a_ptrs = (
        A
        + pid_b * stride_Ab
        + offs_m[:, None] * stride_Am
        + offs_k[None, :] * stride_Ak
    )
    b_ptrs = (
        B
        + pid_b * stride_Bb
        + offs_k[:, None] * stride_Bk
        + offs_n[None, :] * stride_Bn
    )

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k = tl.cdiv(K, BLOCK_K)
    for _ in range(num_k):
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & (offs_k[None, :] < K),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k[:, None] < K) & mask_n[None, :],
            other=0.0,
        )
        acc += tl.dot(a, b)  # dot promotes to FP32 automatically
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Write back (FP16) with proper masking
    c_ptrs = (
        C
        + pid_b * stride_Cb
        + offs_m[:, None] * stride_Cm
        + offs_n[None, :] * stride_Cn
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no backward).

    Args
    ----
    data : Tuple[torch.Tensor, torch.Tensor|None, dict, dict]
        input_tensor : [B, N, N, dim]   (float32)
        mask         : [B, N, N]        (bool/float) or None
        weights      : dict of model weights
        config       : dict containing keys
                       "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ---------------------- unpack -------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ---------------------- 1️⃣ Layer‑norm over `dim` -------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]  (FP32)

    # ---------------------- 2️⃣ fused linear projections ---------------------------
    # Cast to FP16 once
    x_h = x.to(torch.float16)

    # Concatenate the five projection matrices (out_features stacked)
    combined_weight = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden, dim)
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,  # (5*hidden, dim)
    ).contiguous().to(torch.float16)

    # Single half‑precision GEMM
    combined = F.linear(x_h, combined_weight)  # [B, N, N, 5*hidden]

    # Slice back the five tensors
    left, right, left_gate, right_gate, out_gate = torch.split(
        combined, hidden, dim=-1
    )  # each: [B,N,N,hidden]

    # Gate activations (sigmoid) – stay in FP16
    left_gate = left_gate.sigmoid()
    right_gate = right_gate.sigmoid()
    out_gate = out_gate.sigmoid()

    # ---------------------- 3️⃣ optional mask ------------------------------------
    if not nomask and mask is not None:
        # mask shape -> [B, N, N, 1]  (FP16)
        mask_h = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_h
        right = right * mask_h

    # ---------------------- 4️⃣ apply element‑wise gates -------------------------
    left = left * left_gate
    right = right * right_gate

    # ---------------------- 5️⃣ heavy 𝑁³ update (batched mat‑mul) ----------------
    B, N, _, _ = left.shape  # B, N, N, hidden
    # Rearrange to (B, hidden, N, N) and (B, hidden, N, N)ᵀ
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) – transposed on the last two axes

    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)   # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)  # (BH, K=N, N)

    out_flat = torch.empty_like(left_flat)  # (BH, N, N), FP16

    # Choose block sizes (autotuned)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        BH,
    )

    _batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0),
        left_flat.stride(1),
        left_flat.stride(2),
        right_flat.stride(0),
        right_flat.stride(1),
        right_flat.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden) – still FP16
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------- 6️⃣ post‑processing ---------------------------------
    out_fp32 = out.to(torch.float32)

    # Layer‑norm over the hidden dimension
    out_norm = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply the out‑gate (broadcasted)
    out_scaled = out_norm * out_gate.to(torch.float32)

    # Final linear projection back to `dim` (half‑precision for speed)
    to_out_weight_h = weights["to_out.weight"].to(torch.float16)
    out_scaled_h = out_scaled.to(torch.float16)

    final = F.linear(out_scaled_h, to_out_weight_h).to(torch.float32)  # [B,N,N,dim]

    return final
