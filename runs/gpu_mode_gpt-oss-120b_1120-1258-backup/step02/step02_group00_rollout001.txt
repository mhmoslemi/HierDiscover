"""
Optimized "outgoing" TriMul operator (AlphaFold‑3) using Triton.

Algorithm
---------
1️⃣  Layer‑norm over the last channel (dim) – Triton kernel that
    accumulates in fp32 but writes the normalized result in fp16.
2️⃣  Fuse the five linear projections (left/right & three gates) into a
    single GEMM (`F.linear`) by concatenating their weight matrices.
    All computations are performed in fp16 (Tensor‑Core friendly).
3️⃣  (Optional) apply the pairwise mask to the left/right projections.
4️⃣  Apply sigmoid gates and element‑wise gating (still in fp16).
5️⃣  Triangular multiplicative update:
      out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    realized as a batched matrix‑multiply (`torch.bmm`) on tensors of
    shape (B·H, N, N) in fp16.
6️⃣  Second Layer‑norm over the hidden dimension (H) – same Triton kernel.
7️⃣  Multiply by the output gate and project back to `dim` with a final
    linear layer (fp16). Cast to fp32 before returning.

All heavy reductions (the two LayerNorms) are performed by a custom
Triton kernel; dense linear algebra uses fused fp16 GEMMs (cuBLAS /
Tensor‑Core). No explicit synchronisation barriers are inserted,
relying on the automatic stream dependencies between kernels and
PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm kernel – writes fp16 output, keeps fp32 accumulation
# ----------------------------------------------------------------------
@triton.jit
def layernorm_fp16_kernel(
    x_ptr,          # *const float*   [total_rows, dim]   (fp32 input)
    y_ptr,          # *mut   half*    [total_rows, dim]   (fp16 output)
    weight_ptr,     # *const float*   [dim]              (γ)
    bias_ptr,       # *const float*   [dim]              (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # int (compile‑time)
    eps: tl.constexpr,          # float (compile‑time)
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per tile
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # Pass 1: compute row‑wise mean & variance (fp32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # -----------------------------------------------------------------
    # Pass 2: write normalized values (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)       # fp32 raw data

        # (x - mean) * rstd  → fp32
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β are fp32; promote to fp16 right before store
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out_fp32 = norm * w + b            # fp32
        out_fp16 = out_fp32.to(tl.float16) # down‑cast

        out_ptrs = y_ptr + (offs_m[:, None] * dim + offs_d[None, :])
        tl.store(out_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5
    nomask = config.get("nomask", True)   # True → no mask applied

    # --------------------------------------------------------------
    # Prepare shapes
    # --------------------------------------------------------------
    B, N, _, _ = input_tensor.shape             # [B, N, N, C]
    total_rows = B * N * N                       # rows for the flattened view

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp16 output)
    # --------------------------------------------------------------
    # Flatten to 2‑D tensor [total_rows, dim]
    x_flat = input_tensor.view(total_rows, dim).contiguous()
    # Allocate fp16 buffer for normalized result
    x_norm_fp16 = torch.empty((total_rows, dim), dtype=torch.float16, device=device)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)

    layernorm_fp16_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # No explicit sync – PyTorch ops that follow will
    # automatically wait for the kernel to finish.

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projections + gates (all in fp16)
    # --------------------------------------------------------------
    # Concatenate weight matrices (shape: (5*H, C)), cast to fp16 once
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    concat_weight = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate,
         w_right_gate, w_out_gate],
        dim=0,
    ).half().contiguous()                     # (5*H, C)

    # Single GEMM: [total_rows, C] @ [C, 5*H]^T → [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_weight)   # fp16

    # Split the concatenated result
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )  # each: [total_rows, H]

    # --------------------------------------------------------------
    # 2a️⃣ Optional mask (applied before gating)
    # --------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → flatten → [total_rows, 1]
        mask_flat = mask_tensor.view(total_rows, 1).to(dtype=torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 2b️⃣ Apply sigmoid gates (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)   # [total_rows, H]
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    # Element‑wise gating
    left_fp16 = left_proj * left_gate      # [total_rows, H]
    right_fp16 = right_proj * right_gate   # [total_rows, H]

    # Reshape to 4‑D tensors for the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)          # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)      # [B, N, N, H]

    # --------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # --------------------------------------------------------------
    # Permute to gather the 'k' dimension contiguously for GEMM:
    # left:  (B, H, N, N)   = left.permute(0,3,1,2)
    # right: (B, H, N, N)   = right.permute(0,3,2,1)   – note the transposition on k
    left_perm = left.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)      # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)    # [B*H, N, N]

    # Batched matrix multiply in fp16 (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N] fp16

    # Reshape back to 4‑D tensor: [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # fp16

    # --------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden_dim (fp16 output)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim).contiguous()
    out_norm_fp16 = torch.empty_like(out_flat)  # fp16 buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_fp16_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)  # [B, N, N, H] fp16

    # --------------------------------------------------------------
    # 5️⃣ Apply final output gate and linear projection back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate          # fp16   (element‑wise)

    # Final linear projection H → C (fp16)
    to_out_weight_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_weight_fp16)  # fp16, shape [total_rows_h, C]

    # Cast back to fp32 as required by the spec
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
