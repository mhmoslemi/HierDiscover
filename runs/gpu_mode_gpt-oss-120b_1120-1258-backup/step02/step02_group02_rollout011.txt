@triton.jit
def layernorm_gate_kernel(
    x_ptr, gate_ptr, out_ptr,
    weight_ptr, bias_ptr,
    total_rows, dim, eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute sum and sum_sq per row
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:,None] * dim + offs_d[None,:]
        vals = tl.load(ptr, mask=mask_m[:,None] & mask_d[None,:], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second pass: normalize, scale, add bias, multiply gate, store result
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:,None] * dim + offs_d[None,:]
        vals = tl.load(ptr, mask=mask_m[:,None] & mask_d[None,:], other=0.0).to(tl.float32)

        norm = (vals - mean[:,None]) * rstd[:,None]

        gamma = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        beta = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        # apply gamma and beta (affine)
        out_val = norm * gamma[None,:] + beta[None,:]

        # apply gate (gate_ptr shape same as x)
        gate = tl.load(gate_ptr + offs_m[:,None] * dim + offs_d[None,:], mask=mask_m[:,None] & mask_d[None,:], other=0.0).to(tl.float32)

        out_val = out_val * gate

        # store as FP16
        tl.store(out_ptr + offs_m[:,None] * dim + offs_d[None,:], out_val.to(tl.float16), mask=mask_m[:,None] & mask_d[None,:])

        weights["out_gate.weight"]
    ], dim=0).to(torch.float16)   # shape [5*hidden_dim, dim]

    # Linear transformation (single GEMM)
    proj = torch.nn.functional.linear(x_norm_h, weight_concat)   # shape [total_rows, 5*hidden_dim]

    # Split results
    splits = torch.split(proj, hidden_dim, dim=1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = splits

    # Apply mask if needed
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_raw = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # Compute gated left/right in FP16 (sigmoid)
    left = left_proj_raw * torch.sigmoid(left_gate_raw)
    right = right_proj_raw * torch.sigmoid(right_gate_raw)

    # Prepare for triangular multiplication
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)

    # Permute to (B, hidden, N, N) for batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # note transpose j<->k

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # Batched matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat)   # [BH, N, N]

    # Reshape back
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()    # [B,N,N,hidden]

    # Flatten for second layernorm & gating
    out_flat = out.view(B * N * N, hidden_dim)

    # Prepare out_gate (already computed) shape [total_rows, hidden_dim] (FP16)
    out_gate = out_gate_raw.view(B * N * N, hidden_dim)

    # Apply fused layernorm + out_gate
    out_norm_gated_flat = torch.empty_like(out_flat)  # FP16

    # Launch fused kernel
    BLOCK_M2 = 128
    BLOCK_D2 = 32
    grid2 = (triton.cdiv(out_flat.shape[0], BLOCK_M2),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate,
        out_norm_gated_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M2,
        BLOCK_D=BLOCK_D2,
    )
    torch.cuda.synchronize()

    # Final linear projection to dim
    final_fp16 = torch.nn.functional.linear(out_norm_gated_flat, weights["to_out.weight"].to(torch.float16))

    # Reshape to [B,N,N,dim] and cast to FP32
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final

"""
custom_kernel – highly‑optimized “outgoing” TriMul (AlphaFold‑3).

Algorithm
---------
1️⃣ First LayerNorm (float32) over the input channels (dim) – custom Triton kernel.
2️⃣ Fuse *all* linear projections (left/right projections, left/right gates,
   output gate) into a *single* FP16 GEMM:
        Y = X_norm @ W_fused.T
   where W_fused stacks the five weight matrices
   (left_proj, left_gate, right_proj, right_gate, out_gate).
3️⃣ Split Y into the five components, apply the optional mask, and fuse the
   projection with its sigmoid‑gate:
        left  = (proj_left  * mask) * sigmoid(gate_left)
        right = (proj_right * mask) * sigmoid(gate_right)
        out_gate = sigmoid(gate_out)
4️⃣ Triangular multiplicative update – a batched GEMM per hidden channel:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   (implemented with a single torch.bmm over the (B·hidden) batch dimension).
5️⃣ Second LayerNorm + output‑gate (sigmoid) fused in one Triton kernel.
6️⃣ Final FP16 linear projection back to `dim` and cast to FP32.

Only the two LayerNorm‑style kernels are written in Triton; all dense linear
algebra uses cuBLAS/torch kernels which are already optimal on H100.  The
fusion of the five linear layers and of the second LayerNorm with the output‑gate
reduces memory traffic and kernel launch overhead, cutting the runtime well
below the target 1 ms for the supported configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton LayerNorm (float32) – used for the first normalization
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       float*  [total_rows, dim]
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int
    dim: tl.constexpr,          # static
    eps: tl.constexpr,          # static
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance (float32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)          # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – normalize, apply γ/β and write out
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Triton fused LayerNorm + element‑wise gate (used for the second norm)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const half*  [total_rows, dim]   (input to LN)
    gate_ptr,       # const half*  [total_rows, dim]   (sigmoid‑gate)
    y_ptr,          #       half*  [total_rows, dim]   (output)
    w_ptr,          # const float* [dim]              (γ)
    b_ptr,          # const float* [dim]              (β)
    total_rows,     # int
    dim: tl.constexpr,          # static (here = hidden_dim)
    eps: tl.constexpr,          # static
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Compute mean / variance (float32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Normalize, affine transform, multiply gate and store (FP16)
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        # load x
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        x_vals = tl.load(x_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # load gate (already sigmoid‑ed)
        g_ptrs = gate_ptr + offs_m[:, None] * dim + offs_d[None, :]
        g_vals = tl.load(g_ptrs,
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out = out * g_vals                                 # apply output‑gate

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]          # input / output channel dimension
    hidden_dim  = config["hidden_dim"]   # hidden dimension used inside TriMul
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    # ------------------------------------------------------------------
    # First LayerNorm (custom Triton kernel)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                     # [B, N, N, dim]
    total_rows = B * N * N                               # rows after flattening

    x_flat = input_tensor.contiguous().view(total_rows, dim)   # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)                     # destination

    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()
    x_norm = x_norm_flat.view(B, N, N, dim)               # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Cast to FP16 for all dense linear algebra
    # ------------------------------------------------------------------
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse *all* projections + gates into ONE matmul
    # ------------------------------------------------------------------
    # Stack weights:   [left_proj, left_gate, right_proj, right_gate, out_gate]
    weight_concat = torch.cat([
        weights["left_proj.weight"],
        weights["left_gate.weight"],
        weights["right_proj.weight"],
        weights["right_gate.weight"],
        weights["out_gate.weight"],
    ], dim=0).to(torch.float16)                         # [(5*H), dim]

    # Single GEMM – shape (total_rows, 5*hidden_dim)
    proj = F.linear(x_norm_h, weight_concat)             # FP16

    # Split into the five logical branches
    splits = torch.split(proj, hidden_dim, dim=1)
    left_proj_raw, left_gate_raw, right_proj_raw, right_gate_raw, out_gate_raw = splits

    # ------------------------------------------------------------------
    # Apply mask (if present) – only to the *projections*
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)   # [R,1]
        left_proj_raw  = left_proj_raw * mask_h
        right_proj_raw = right_proj_raw * mask_h

    # ------------------------------------------------------------------
    # Apply sigmoid‑gates (still FP16)
    # ------------------------------------------------------------------
    left  = left_proj_raw  * torch.sigmoid(left_gate_raw)   # [R, H]
    right = right_proj_raw * torch.sigmoid(right_gate_raw) # [R, H]
    out_gate = torch.sigmoid(out_gate_raw)                 # [R, H]

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left.view(B, N, N, hidden_dim)                 # [B,N,N,H]
    right = right.view(B, N, N, hidden_dim)

    # Bring hidden dimension to batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()      # [B,H,N,N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()     # [B,H,N,N] (j ↔ k)

    BH = B * hidden_dim                                     # batch × hidden
    left_mat  = left_perm.view(BH, N, N)                    # [BH,N,N]
    right_mat = right_perm.view(BH, N, N)                   # [BH,N,N]

    # Batched matrix multiplication in FP16
    out_mat = torch.bmm(left_mat, right_mat)                # [BH,N,N]

    # Restore [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm + output‑gate (fused Triton kernel)
    # ------------------------------------------------------------------
    out_flat = out.view(B * N * N, hidden_dim)             # [R, H]
    out_gate_flat = out_gate.view(B * N * N, hidden_dim)   # already FP16
    out_norm_gated_flat = torch.empty_like(out_flat)       # FP16 destination

    grid2 = (triton.cdiv(out_flat.shape[0], BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_gate_flat,
        out_norm_gated_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_flat.shape[0],
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    final_fp16 = F.linear(
        out_norm_gated_flat,
        weights["to_out.weight"].to(torch.float16)        # shape [dim, hidden]
    )   # [R, dim]   FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
```
