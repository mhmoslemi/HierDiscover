"""
Highly‑optimized Triton implementation of the *outgoing* TriMul operator
(AlphaFold‑3 style).

Algorithm
---------
1. Layer‑norm the 4‑D input (float32 → float16).
2. Fuse the two linear projections (left/right) into a single GEMM
   `proj = x @ [W_left; W_right]`.  The same trick is used for the
   two gating linear layers.
3. Optional mask is applied as a broadcasted element‑wise product.
4. Gating scalars are computed with a sigmoid and fused onto the
   projected tensors (`left = proj_left * left_gate`, `right = proj_right * right_gate`).
5. The N³ contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is a batched matrix multiplication `C = A @ Bᵀ`.  It is realised with a
   custom Triton kernel that streams K‑tiles through shared memory, uses
   FP16 inputs, FP32 accumulation and Tensor‑core `tl.dot`.  The kernel
   auto‑tunes its block sizes (128/64) and the number of warps.
6. The result is reshaped back to `[B,N,N,hidden]`, Layer‑normed,
   multiplied by the out‑gate, and projected back to the original `dim`.
   The final tensor is returned in float32.

All heavy work (projections, gating, N³ contraction) is performed in
half‑precision; only the lightweight normalisation and the final linear
layer stay in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Batched GEMM: C[b] = A[b] @ B[b]ᵀ   (A: M×K, B: N×K  →  B is read as K×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,  # fp16 *  (batch, M, K)
    b_ptr,  # fp16 *  (batch, K, N)   read as (K,N)
    c_ptr,  # fp16 *  (batch, M, N)
    M, N, K,
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bk, stride_bn,   # strides for B
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM with FP32 accumulation.
    ``batch`` dimension is folded into the first program‑id.
    """
    pid0 = tl.program_id(0)   # batch_id * tiles_M + tile_m
    pid1 = tl.program_id(1)   # tile_n

    # ----- decode batch id and tile coordinates ---------------------------------
    tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id   = pid0 // tiles_m
    tile_m     = pid0 %  tiles_m
    tile_n     = pid1

    # ----- offsets inside the current tile ---------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ----- pointers to the start of the tile (k‑loop offset = 0) ------------------
    a_ptrs = a_ptr + batch_id * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # ----- accumulator (FP32) ----------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- K‑loop ---------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # masks for out‑of‑bounds loads
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)   # fp16·fp16 → fp32

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ----- store result ---------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.

    Parameters
    ----------
    data : tuple

        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32)
        - config       : dict with keys {"dim": int,
                                         "hidden_dim": int,
                                         "nomask": bool (optional)}

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    B, N, _, dim = inp.shape
    hidden = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input layer‑norm (float32 for numerical stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Cast to fp16 – the rest of the pipeline works in half‑precision
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Fuse projection + gating linear layers
    # -----------------------------------------------------------------
    #   a) concatenate left/right projection weights (out_features stacked)
    proj_w = torch.cat(
        [weights["left_proj.weight"], weights["right_proj.weight"]], dim=0
    ).to(torch.float16)          # (2*hidden, dim)

    gate_w = torch.cat(
        [weights["left_gate.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)          # (2*hidden, dim)

    #   b) flatten the 4‑D tensor for a single GEMM
    x_flat = x_fp16.view(-1, dim)                 # (B*N*N, dim)

    #   c) projection (left & right) – shape (B*N*N, 2*hidden)
    proj = F.linear(x_flat, proj_w)                # fp16

    #   d) gating – shape (B*N*N, 2*hidden)
    gate = torch.sigmoid(F.linear(x_flat, gate_w))

        # mask: (B,N,N) → (B*N*N,1)

    right_proj = proj[:, hidden:]

    right_gate = gate[:, hidden:]

    right = right_proj * right_gate

    # -----------------------------------------------------------------

    right = right.view(B, N, N, hidden).permute(0, 3, 2, 1).contiguous()
    # left : (B, hidden, N, N)   – A matrix layout  (M=N, K=N)
    # right: (B, hidden, N, N)   – B matrix layout  (K=N, N)

    left_mat = left.view(B * hidden, N, N)
    right_mat = right.view(B * hidden, N, N)

    # -----------------------------------------------------------------
    # 5) Out‑gate (computed once, kept in fp32 for the final stage)
    # -----------------------------------------------------------------
    out_gate = torch.sigmoid(F.linear(x_fp16, weights["out_gate.weight"].to(torch.float16)))
    # shape (B, N, N, hidden)
    out_gate = out_gate.view(B, N, N, hidden)

    # -----------------------------------------------------------------
    # 6) Batched GEMM (A @ Bᵀ) via Triton
    # -----------------------------------------------------------------
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # Adaptive tile size – tuned for the range of N used in the tests
    if N <= 256:
        BLOCK_M = BLOCK_N = 64
        BLOCK_K = 64
        num_warps = 4
    elif N <= 512:
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 64
        num_warps = 8
    else:               # N up to 1024
        BLOCK_M = BLOCK_N = 128
        BLOCK_K = 128
        num_warps = 8

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)
    # The first launch dimension packs (batch*hidden, tile_m)
    grid = (B * hidden * grid_m, grid_n)

    batched_gemm_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bk=right_mat.stride(1),
        stride_bn=right_mat.stride(2),
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
        num_stages=4,
    )

    # -----------------------------------------------------------------
    # 7) Reshape back to [B, N, N, hidden] (float32)
    # -----------------------------------------------------------------
    out = (
        out_mat.view(B, hidden, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
        .to(torch.float32)
    )

    # -----------------------------------------------------------------
    # 8) Output LayerNorm, out‑gate and final projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate  # broadcasting over the hidden dimension

    # final linear projection back to original `dim`
    out = F.linear(out, weights["to_out.weight"])  # bias=False, kept in float32

    return out
```
