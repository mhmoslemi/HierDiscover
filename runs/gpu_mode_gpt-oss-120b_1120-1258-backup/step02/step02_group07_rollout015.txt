"""
TriMul (outgoing) – fused Triton kernel for the forward pass.

The kernel fuses the LayerNorm‑normalised input with **all** linear
projections, sigmoid gates and the optional mask.  The result is two
tensors ``left`` and ``right`` of shape ``[B, H, N, N]`` (half precision)
and an ``out_gate`` tensor of the same shape.  These tensors are then
combined with a single batched GEMM (torch.bmm) to produce the
tri‑multiplicative update, after which the remaining LayerNorm,
gate and output projection are performed with regular PyTorch ops.

By reading the normalised input only once per element and by avoiding the
``permute‑contiguous`` copies that the reference implementation performs,
memory traffic is drastically reduced, giving a large speed‑up on H100
GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: fused projection + gating (outgoing TriMul)
# --------------------------------------------------------------
@triton.jit
def tri_mul_fused_kernel(
    # pointers
    x_ptr, mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_ptr,
    # weight pointers (shape [H, D])
    w_left_proj_ptr, w_left_gate_ptr,
    w_right_proj_ptr, w_right_gate_ptr,
    w_out_gate_ptr,
    # strides (int64)
    stride_x_batch, stride_x_i, stride_x_j, stride_x_dim,
    stride_mask_batch, stride_mask_i, stride_mask_j,
    stride_left_batch, stride_left_h, stride_left_i, stride_left_j,
    stride_right_batch, stride_right_h, stride_right_i, stride_right_j,
    stride_out_gate_batch, stride_out_gate_h, stride_out_gate_i, stride_out_gate_j,
    # sizes (int)
    B, N, D, H,
    epsilon,          # layer‑norm epsilon (float32)
    nomask,           # 1 -> no mask, 0 -> use mask
    # compile‑time constants
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute:
        left  = (x @ W_left_proj^T) * sigmoid(x @ W_left_gate^T) * mask
        right = (x @ W_right_proj^T) * sigmoid(x @ W_right_gate^T) * mask
        out_gate = sigmoid(x @ W_out_gate^T)

    All operations are performed on the layer‑normalised input ``x``.
    The outputs have shape [B, H, N, N] (half‑precision).
    """
    pid_m = tl.program_id(0)          # tile over rows (B * N * N)
    pid_n = tl.program_id(1)          # tile over hidden dimension H

    # -----------------------------------------------------------------
    # 1) Compute the flattened row indices for this tile.
    #    total_rows = B * N * N
    # -----------------------------------------------------------------
    total_rows = B * N * N
    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    # row indices for this block
    rows = row_start + tl.arange(0, BLOCK_M)
    row_mask = rows < total_rows

    # map flat row index -> (b, i, j)
    bn = N * N
    b = rows // bn
    rem = rows % bn
    i = rem // N
    j = rem % N

    # -----------------------------------------------------------------
    # 2) Offsets for the input tensors.
    # -----------------------------------------------------------------
    # x (B, N, N, D) – last dimension is contiguous (stride_x_dim = 1)
    x_base = (
        b * stride_x_batch
        + i * stride_x_i
        + j * stride_x_j
    )                                   # shape [BLOCK_M]

    # optional mask (B, N, N)
    mask_base = (
        b * stride_mask_batch
        + i * stride_mask_i
        + j * stride_mask_j
    )                                   # shape [BLOCK_M]

    # left / right / out_gate output layout: (B, H, N, N)
    left_base = (
        b * stride_left_batch
        + i * stride_left_i
        + j * stride_left_j
    )                                   # shape [BLOCK_M]
    right_base = (
        b * stride_right_batch
        + i * stride_right_i
        + j * stride_right_j
    )                                   # shape [BLOCK_M]
    out_gate_base = (
        b * stride_out_gate_batch
        + i * stride_out_gate_i
        + j * stride_out_gate_j
    )                                   # shape [BLOCK_M]

    # hidden‑dim offsets for this tile
    cols = col_start + tl.arange(0, BLOCK_N)
    col_mask = cols < H

    # -----------------------------------------------------------------
    # 3) Accumulators (FP32)
    # -----------------------------------------------------------------
    acc_left_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_left_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_right_proj = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_right_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    acc_out_gate = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # 4) Loop over the feature dimension D in tiles of size BLOCK_K
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(D, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_valid = k_offset < D

        # ----- load a tile of the normalised input -----------------
        x_ptrs = x_ptr + x_base[:, None] + k_offset[None, :] * stride_x_dim
        x = tl.load(
            x_ptrs,
            mask=row_mask[:, None] & k_valid[None, :],
            other=0.0
        ).to(tl.float32)                 # (BLOCK_M, BLOCK_K)

        # ----- load weight tiles ------------------------------------
        # each weight matrix is stored as (H, D) with stride_h = D, stride_k = 1
        w_stride_h = D

        # helper to load a weight tile
        def load_w(w_ptr):
            w_ptrs = w_ptr + k_offset[:, None] + cols[None, :] * w_stride_h
            return tl.load(
                w_ptrs,
                mask=k_valid[:, None] & col_mask[None, :],
                other=0.0
            )                               # (BLOCK_K, BLOCK_N)

        w_lp = load_w(w_left_proj_ptr)     # left projection
        w_lg = load_w(w_left_gate_ptr)     # left gate
        w_rp = load_w(w_right_proj_ptr)    # right projection
        w_rg = load_w(w_right_gate_ptr)    # right gate
        w_og = load_w(w_out_gate_ptr)      # out gate

        # ----- accumulate dot products -------------------------------
        acc_left_proj += tl.dot(x, w_lp)   # (BLOCK_M, BLOCK_N)
        acc_left_gate += tl.dot(x, w_lg)
        acc_right_proj += tl.dot(x, w_rp)
        acc_right_gate += tl.dot(x, w_rg)
        acc_out_gate += tl.dot(x, w_og)

    # -----------------------------------------------------------------
    # 5) Apply sigmoid activations (gate) and optional mask
    # -----------------------------------------------------------------
    # sigmoid(x) = 1/(1+exp(-x))
    left_gate_sig = 1.0 / (1.0 + tl.exp(-acc_left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-acc_right_gate))
    out_gate_sig = 1.0 / (1.0 + tl.exp(-acc_out_gate))

    # mask handling
    if nomask:
        mask_val = tl.full((BLOCK_M, 1), 1.0, dtype=tl.float32)
    else:
        m = tl.load(
            mask_ptr + mask_base,
            mask=row_mask,
            other=0.0
        ).to(tl.float32)                # (BLOCK_M)
        mask_val = m[:, None]            # broadcast over hidden dim

    # left and right after gating and mask
    left_tile = (acc_left_proj * left_gate_sig) * mask_val
    right_tile = (acc_right_proj * right_gate_sig) * mask_val

    # cast to fp16 before writing
    left_tile = left_tile.to(tl.float16)
    right_tile = right_tile.to(tl.float16)
    out_gate_tile = out_gate_sig.to(tl.float16)

    # -----------------------------------------------------------------
    # 6) Store results
    # -----------------------------------------------------------------
    left_ptrs = left_out_ptr + left_base[:, None] + cols[None, :] * stride_left_h
    right_ptrs = right_out_ptr + right_base[:, None] + cols[None, :] * stride_right_h
    out_gate_ptrs = out_gate_ptr + out_gate_base[:, None] + cols[None, :] * stride_out_gate_h

    store_mask = row_mask[:, None] & col_mask[None, :]

    tl.store(left_ptrs, left_tile, mask=store_mask)
    tl.store(right_ptrs, right_tile, mask=store_mask)
    tl.store(out_gate_ptrs, out_gate_tile, mask=store_mask)


# --------------------------------------------------------------
# Python entry point
# --------------------------------------------------------------

    """

    Parameters
    ----------

        (input_tensor, mask, weights, config)

                               "left_gate.weight", "right_gate.weight",

                               "to_out_norm.weight", "to_out_norm.bias",
                               "to_out.weight"

    -------
    torch.Tensor

    """

    # -----------------------------------------------------------------

        x = F.layer_norm(
            input_tensor,
            (dim,),
            weight=norm_weight,
            bias=norm_bias,
        )  # [B, N, N, dim]  (float16 inside autocast)

        # -------------------------------------------------------------
        # 2) Prepare tensors for the fused Triton kernel
        # -------------------------------------------------------------
        # allocate output buffers (FP16)
        left = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
        right = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)
        out_gate = torch.empty((B, hidden_dim, N, N), dtype=torch.float16, device=device)

        # optional mask (broadcasted to FP16)
        if not nomask and mask is not None:
            mask_f = mask.to(torch.float16)
        else:
            mask_f = torch.empty(0, device=device, dtype=torch.float16)  # dummy

        # -----------------------------------------------------------------
        # 3) Launch the fused kernel
        # -----------------------------------------------------------------
        # block sizes (tuned for H100, float16)
        BLOCK_M = 64
        BLOCK_N = 32
        BLOCK_K = 32

        # Strides for the input tensors
        stride_x = x.stride()               # (B, N, N, D)
        stride_mask = mask_f.stride() if mask_f.numel() else (0, 0, 0)

        # Strides for the output tensors (B, H, N, N)
        stride_left = left.stride()
        stride_right = right.stride()
        stride_out_gate = out_gate.stride()

        # Launch grid
        total_rows = B * N * N
        grid = (triton.cdiv(total_rows, BLOCK_M), triton.cdiv(hidden_dim, BLOCK_N))

        tri_mul_fused_kernel[grid](
            # pointers
            x, mask_f,
            left, right, out_gate,
            # weight pointers (all are FP16, shape [H, D])
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
            # strides
            stride_x[0], stride_x[1], stride_x[2], stride_x[3],
            stride_mask[0] if mask_f.numel() else 0,
            stride_mask[1] if mask_f.numel() else 0,
            stride_mask[2] if mask_f.numel() else 0,
            stride_left[0], stride_left[1], stride_left[2], stride_left[3],
            stride_right[0], stride_right[1], stride_right[2], stride_right[3],
            stride_out_gate[0], stride_out_gate[1], stride_out_gate[2], stride_out_gate[3],
            # sizes
            B, N, dim, hidden_dim,
            # layer‑norm epsilon (same as PyTorch default)
            1e-5,
            int(not nomask),   # 0 -> use mask, 1 -> no mask
            # compile‑time constants
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        )

        # -------------------------------------------------------------
        # 4) Batched GEMM: out = left @ rightᵀ   (FP16, FP32 acc)
        # -------------------------------------------------------------
        # reshape to (B*H, N, N)
        left_mat = left.view(B * hidden_dim, N, N)
        right_mat = right.view(B * hidden_dim, N, N)
        # rightᵀ via transpose (no copy)
        out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

        # restore to (B, N, N, H)
        out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

        # -------------------------------------------------------------
        # 5) Final normalisation, gating and projection back to dim
        # -------------------------------------------------------------
        # to_out_norm (LayerNorm over hidden_dim)
        out = F.layer_norm(
            out,
            (hidden_dim,),
            weight=weights["to_out_norm.weight"],
            bias=weights["to_out_norm.bias"],
        )

        # out_gate has shape (B, H, N, N) → permute to (B,N,N,H) to match out
        out_gate = out_gate.permute(0, 2, 3, 1).contiguous()
        out = out * out_gate

        # final linear back to original dim
        out = F.linear(out, weights["to_out.weight"])

        # ensure output is FP32 as required by the API
        out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # The problem statement demands that a Triton kernel is launched.
    # The fused kernel above already does the heavy work, so we do not
    # need a dummy kernel any more.
    # -----------------------------------------------------------------
    return out
```
