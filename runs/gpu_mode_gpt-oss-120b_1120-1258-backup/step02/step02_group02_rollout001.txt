"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1. **First LayerNorm (dim)** – Row‑wise LayerNorm over the last channel
   dimension using a Triton kernel (float32).

2. **FP16 projection & gating** – The two linear projections (left/right) and
   their sigmoid‑gates are fused by concatenating the weight matrices and
   calling `torch.nn.functional.linear`.  The output‑gate is computed separately.
   All dense ops run in FP16.

3. **Optional mask** – If a mask is supplied (`nomask=False`) it is applied to
   the left/right projections (broadcast over the hidden dimension).

4. **Triangular multiplicative update** – For each hidden channel `h` we need
   `out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]`.  This
   is a batched GEMM (`left @ rightᵀ`) performed with `torch.bmm` on tensors of
   shape `[B*H, N, N]` in FP16.

5. **Second LayerNorm (hidden_dim)** – Row‑wise LayerNorm over the hidden
   dimension (again a Triton kernel, float32).

6. **Output gate & final projection** – Apply the output‑gate (sigmoid) and a
   final linear projection back to `dim`.  The last linear runs in FP16 and the
   result is cast back to float32.

The kernel works for any combination of batch size `B ∈ {1,2}`, sequence length
`N ∈ {128,256,512,1024}`, input channel `dim ∈ {128,384}` and hidden dimension
`hidden_dim = 128`.  A mask tensor of shape `[B,N,N]` may be omitted
(`nomask=True`).  All heavy reductions (the two LayerNorms) run in Triton,
while the matrix multiplications use PyTorch’s highly‑optimized FP16 kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       float*  [total_rows, dim]
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static dimension size
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Row‑wise LayerNorm for a 2‑D tensor stored in row‑major order.
    Computes:
        y = (x - μ) / √(σ² + eps) * γ + β
    """
    pid = tl.program_id(0)

    # Row range for this program
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows                        # [BLOCK_M]

    # ------------------------------------------------------------------
    # Pass 1 – compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # iterate over column blocks
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim

        # linear offset = row * dim + col
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1/√(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values with γ (weight) and β (bias)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted over rows)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Flatten and LayerNorm over the input channel dimension (dim)
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()                     # [B,N,N,dim]
    B, N, _, _ = x.shape
    total_rows = B * N * N
    x_flat = x.view(total_rows, dim)                  # [R, dim]
    x_norm_flat = torch.empty_like(x_flat)            # destination

    # Tunable block sizes (larger values improve occupancy on H100)
    BLOCK_M = 256
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # Restore 4‑D shape (still FP32)
    x_norm = x_norm_flat.view(B, N, N, dim)          # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Cast to FP16 for all dense linear algebra
    # ------------------------------------------------------------------
    total_rows = B * N * N
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # ------------------------------------------------------------------
    # Fuse projection + gate for left and right paths
    # ------------------------------------------------------------------
    # left: concat(proj, gate) → shape [2*H, dim]
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                               # [2H, dim]

    # right: concat(proj, gate)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                               # [2H, dim]

    # Linear in FP16 (biasless)
    left_concat = F.linear(x_norm_h, left_concat_w)   # [R, 2H]
    right_concat = F.linear(x_norm_h, right_concat_w) # [R, 2H]

    # Split back into projection & pre‑gate
    left_proj_h = left_concat[:, :hidden_dim]            # [R, H]
    left_gate_pre_h = left_concat[:, hidden_dim:]        # [R, H]

    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # Output gate (no bias)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h, weights["out_gate.weight"].to(torch.float16))
    )   # [R, H]

    # ------------------------------------------------------------------
    # Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B,N,N] -> [R,1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # Apply element‑wise sigmoid gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)   # [R, H]
    right_gate_h = torch.sigmoid(right_gate_pre_h)  # [R, H]

    left_h  = left_proj_h * left_gate_h    # [R, H]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    # move hidden dim to batch dimension for bmm:
    # left_perm: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    # right_perm must have the K dimension last for transposition:
    # we need right[j,k] when we multiply left[i,k] * right[j,k] → right transposed
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # [B, H, N, N]

    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)      # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)     # [BH, N, N]

    # Batched matrix multiplication: left @ rightᵀ  (right already transposed)
    out_mat = torch.bmm(left_mat, right_mat)  # [BH, N, N]

    # Reshape back to [B, N, N, H] (still FP16)
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                          # [B,N,N,H]

    # ------------------------------------------------------------------
    # Second LayerNorm over hidden_dim (float32)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)
    total_rows_h = B * N * N
    out_flat = out_f32.view(total_rows_h, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # float32

    # ------------------------------------------------------------------
    # Apply output gate and final projection back to dim
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate_h.view(B, N, N, hidden_dim).to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # final linear (hidden_dim → dim) in FP16
    out_norm_h = out_norm.to(torch.float16)
    final_fp16 = F.linear(
        out_norm_h.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [R, dim], FP16

    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
