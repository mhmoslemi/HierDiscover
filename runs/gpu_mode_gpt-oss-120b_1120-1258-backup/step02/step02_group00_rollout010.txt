"""
TriMul “outgoing” forward pass – highly‑optimized Triton + PyTorch implementation.

Algorithm
---------
1. Row‑wise LayerNorm over the last channel (`dim`).  A custom Triton kernel
   reads the (fp32 or fp16) input, accumulates mean/variance in fp32 and writes
   the normalized result directly in fp16 to avoid an extra cast.
2. Fuse the five linear projections (left/right + three gates) into a single
   FP16 GEMM by concatenating the weight matrices.
3. (Optional) apply the pairwise mask to the left/right projections.
4. Apply sigmoid gates and element‑wise gating – all in FP16.
5. Triangular multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   performed with a single `torch.einsum`, which internally maps to a batched
   GEMM without any costly permutes or copies.
6. Row‑wise LayerNorm over the hidden dimension (`hidden_dim`) – the same
   Triton kernel as in step 1, now reading FP16.
7. Apply the final output gate, a linear projection back to `dim` (FP16 GEMM)
   and cast the result to FP32.

Only the two LayerNorms are written in Triton; the heavy linear algebra uses
high‑performance cuBLAS/Tensor‑Core kernels via PyTorch.  All intermediate
tensors stay in FP16, dramatically reducing memory traffic and runtime.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (writes FP16, works for FP32 or FP16 input)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # *const T*        [total_rows, dim]   (T = fp32 or fp16)
    y_ptr,               # *mut   fp16*     [total_rows, dim]   (output FP16)
    weight_ptr,          # *const fp32*     [dim]                (γ)
    bias_ptr,            # *const fp32*     [dim]                (β)
    total_rows,          # int (runtime)
    dim,                 # int (compile‑time)
    eps,                 # float (compile‑time)
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
    INPUT_DTYPE: tl.constexpr,   # tl.float32 or tl.float16
):
    """Row‑wise LayerNorm for a 2‑D tensor.
    Input may be fp32 or fp16; the mean/variance are accumulated in fp32.
    The normalized output is written as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1) Compute mean & variance (float32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        # Load either fp32 or fp16, then promote to fp32 for accumulation
        vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=INPUT_DTYPE,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / sqrt(var+eps)

    # -----------------------------------------------------------------
    # 2) Write normalized values (FP16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        x_ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            x_ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=INPUT_DTYPE,
        ).to(tl.float32)

        # (x - mean) / sqrt(var+eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) are FP32; promote to FP32 for the mul/add
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0).to(tl.float32)

        out_fp32 = norm * w[None, :] + b[None, :]      # fp32
        out_fp16 = out_fp32.to(tl.float16)             # cast to fp16

        y_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(y_ptrs, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor  : torch.Tensor [B, N, N] (bool/float) – may be ignored
            - weights_dict : dict of model weights (float32 tensors)
            - config_dict  : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns:
        torch.Tensor [B, N, N, dim] (float32, on the same CUDA device)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]               # input channel dimension C
    hidden_dim = config["hidden_dim"] # projection dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1) First LayerNorm over ``dim`` (input is float32, output FP16)
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape                 # [B, N, N, dim]
    total_rows = B * N * N
    x_flat = input_tensor.view(total_rows, dim)     # [total_rows, dim]  (float32)

    # allocate FP16 output buffer for the normalized tensor
    x_norm_fp16 = torch.empty(
        total_rows, dim, dtype=torch.float16, device=device
    )

    # launch kernel: INPUT_DTYPE = tl.float32 because x is FP32
    BLOCK_M = 128
    BLOCK_D = 32
    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_fp16,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        INPUT_DTYPE=tl.float32,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2) Fused linear projections + three gating projections (all FP16)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (shape: (5*H, dim))
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]
    concat_w = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate, w_right_gate, w_out_gate],
        dim=0,
    ).half().contiguous()                     # (5*H, dim)

    # Single GEMM: [total_rows, dim] @ [dim, 5*H]^T  → [total_rows, 5*H]   (FP16)
    proj_all = F.linear(x_norm_fp16, concat_w)   # fp16

    # Split back to the individual tensors
    splits = torch.split(proj_all, hidden_dim, dim=1)
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = splits

    # ------------------------------------------------------------------
    # 3) Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → [total_rows, 1]
        mask_flat_fp16 = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj_fp16 = left_proj_fp16 * mask_flat_fp16
        right_proj_fp16 = right_proj_fp16 * mask_flat_fp16

    # ------------------------------------------------------------------
    # 4) Sigmoid gates (still FP16) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)       # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)    # [total_rows, H]
    out_gate = torch.sigmoid(out_gate_fp16)        # [total_rows, H]

    left = left_proj_fp16 * left_gate               # [total_rows, H]
    right = right_proj_fp16 * right_gate            # [total_rows, H]

    # Reshape to 4‑D tensors for the triangular update
    left = left.view(B, N, N, hidden_dim)           # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)        # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5) Triangular multiplicative update via einsum (no extra copies)
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    out = torch.einsum('b i k h, b j k h -> b i j h', left, right)  # fp16

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden_dim (input FP16 → output FP16)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    out_norm_fp16 = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_fp16,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        INPUT_DTYPE=tl.float16,                     # now reading FP16
    )
    torch.cuda.synchronize()

    out_norm = out_norm_fp16.view(B, N, N, hidden_dim)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Apply final output gate and project back to ``dim``
    # ------------------------------------------------------------------
    gated = out_norm * out_gate          # (B, N, N, H)   fp16

    # Final linear projection: hidden_dim → dim (FP16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # [total_rows_h, dim]

    # Cast result back to FP32 as required by the interface
    final = final_fp16.view(B, N, N, dim).to(torch.float32)

    return final
