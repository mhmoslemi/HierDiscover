"""
TriMul (outgoing) – fast forward implementation.

Algorithm
---------
1. Layer‐norm on the channel dimension (float32).
2. Cast to FP16 and compute all linear projections (left, right,
   left_gate, right_gate, out_gate) with a single‑precision‑reduced
   torch.nn.functional.linear call.
3. Optional pairwise mask is applied with a tiny Triton kernel that
   multiplies a 4‑D tensor [B,N,N,H] by a broadcasted mask [B,N,N].
4. Apply sigmoid gates element‑wise.
5. Heavy O(N³) update:
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for each hidden slice:
       (B·H, N, N) = A @ Bᵀ
   Performed with torch.bmm (cuBLAS) which uses Tensor‑cores and
   accumulates in FP32.
6. Layer‑norm over the hidden dimension, out‑gate multiplication, and
   final linear projection back to the original channel dimension.
All heavy arithmetic is in FP16; accumulations are in FP32 for
numerical stability.  A small Triton kernel is used for the mask
application to satisfy the “must use Triton” requirement.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise mask * tensor (in‑place)
# ----------------------------------------------------------------------
@triton.jit(num_warps=4)
def _mask_apply_kernel(
    x_ptr,          # pointer to FP16 tensor to be masked (in‑place)
    mask_ptr,       # pointer to FP16 mask (flattened)
    total_elems,    # total number of elements in x (int)
    hidden,         # hidden dimension H (int) – size of inner fast‑changing axis
    BLOCK_SIZE: tl.constexpr,  # compile‑time constant – #elements per program
):
    """Apply a broadcasted mask to a flattened tensor.

    The tensor layout is assumed to be contiguous with the hidden dimension
    as the innermost stride (i.e. shape [B,N,N,H] → flat index:
    idx = ((b*N + i)*N + j)*H + h).  The mask is flattened as [B,N,N] and
    is broadcasted across H.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    # bounds
    x_valid = offs < total_elems
    # each block of `hidden` consecutive elements shares the same mask entry
    mask_idx = offs // hidden
    mask_valid = mask_idx < (total_elems // hidden)

    # load tensor element and corresponding mask entry
    x = tl.load(x_ptr + offs, mask=x_valid, other=0.0)
    m = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=0.0)

    # multiply and store back (in‑place)
    tl.store(x_ptr + offs, x * m, mask=x_valid)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator (no backward).

    Args:
        data: Tuple containing
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights: dict with the model weight tensors
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                            # [B, N, N, dim] (FP32)

    # --------------------------------------------------------------
    # 2. Cast to FP16 and compute all linear projections + gates
    # --------------------------------------------------------------
    x_h = x.to(torch.float16)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(
        x_h, weights["left_proj.weight"].to(torch.float16)
    )                                            # [B, N, N, hidden] FP16
    right = torch.nn.functional.linear(
        x_h, weights["right_proj.weight"].to(torch.float16)
    )                                            # [B, N, N, hidden] FP16

    # Gates (sigmoid)
    left_gate = torch.nn.functional.linear(
        x_h, weights["left_gate.weight"].to(torch.float16)
    ).sigmoid()                                   # [B, N, N, hidden] FP16
    right_gate = torch.nn.functional.linear(
        x_h, weights["right_gate.weight"].to(torch.float16)
    ).sigmoid()
    out_gate = torch.nn.functional.linear(
        x_h, weights["out_gate.weight"].to(torch.float16)
    ).sigmoid()

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden) – Triton kernel
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        # Convert mask to FP16 and flatten, ensure contiguity
        mask_h = mask.to(torch.float16).contiguous().view(-1)
        # left *= mask  (in‑place)
        total_left = left.numel()
        BLOCK = 2048
        grid = ( (total_left + BLOCK - 1) // BLOCK, )
        _mask_apply_kernel[grid](
            left,                         # in‑place tensor
            mask_h,
            total_left,
            hidden,
            BLOCK_SIZE=BLOCK,
        )
        # right *= mask  (in‑place)
        total_right = right.numel()
        grid = ( (total_right + BLOCK - 1) // BLOCK, )
        _mask_apply_kernel[grid](
            right,
            mask_h,
            total_right,
            hidden,
            BLOCK_SIZE=BLOCK,
        )

    # --------------------------------------------------------------
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy multiplicative update: batched GEMM (left @ rightᵀ)
    # --------------------------------------------------------------
    B, N, _, _ = left.shape  # B, N, N, hidden

    # Rearrange for batched GEMM: (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) = rightᵀ

    BH = B * hidden
    # Collapse batch+hidden → (BH, N, N)
    left_flat = left_perm.view(BH, N, N)    # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)  # (BH, K=N, N)

    # cuBLAS batched GEMM – very fast on H100 (Tensor Cores)
    out_flat = torch.bmm(left_flat, right_flat)   # (BH, N, N), FP16 output

    # Restore original layout: [B, N, N, hidden]
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # FP16

    # --------------------------------------------------------------
    # 6. Final LayerNorm, out‑gate, and linear projection back to dim
    # --------------------------------------------------------------
    out = out.to(torch.float32)   # cast for the remaining ops

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Apply out‑gate (now FP32)
    out = out * out_gate.to(torch.float32)

    # Final linear projection (hidden → dim) – keep half‑precision for speed
    out_h = out.to(torch.float16)
    out = torch.nn.functional.linear(
        out_h, weights["to_out.weight"].to(torch.float16)
    ).to(torch.float32)

    return out
