import torch
from torch import nn, einsum
import math

# Reference code in PyTorch
class TriMul(nn.Module):
    def __init__(
        self,
        dim: int,
        hidden_dim: int,
    ):
        super().__init__()

        self.norm = nn.LayerNorm(dim)

        self.left_proj = nn.Linear(dim, hidden_dim, bias=False)
        self.right_proj = nn.Linear(dim, hidden_dim, bias=False)

        self.left_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.right_gate = nn.Linear(dim, hidden_dim, bias=False)
        self.out_gate = nn.Linear(dim, hidden_dim, bias=False)

        self.to_out_norm = nn.LayerNorm(hidden_dim)
        self.to_out = nn.Linear(hidden_dim, dim, bias=False)

    def forward(self, x: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
        """
        x: [bs, seq_len, seq_len, dim]
        mask: [bs, seq_len, seq_len]

        Returns:
            output: [bs, seq_len, seq_len, dim]
        """
        batch_size, seq_len, _, dim = x.shape

        x = self.norm(x)

        left = self.left_proj(x)
        right = self.right_proj(x)

        mask = mask.unsqueeze(-1)
        left = left * mask
        right = right * mask

        left_gate = self.left_gate(x).sigmoid()
        right_gate = self.right_gate(x).sigmoid()
        out_gate = self.out_gate(x).sigmoid()

        left = left * left_gate
        right = right * right_gate

        out = einsum('... i k d, ... j k d -> ... i j d', left, right)
        # This einsum is the same as the following:
        # out = torch.zeros(batch_size, seq_len, seq_len, dim, device=x.device)
        
        # # Compute using nested loops
        # for b in range(batch_size):
        #     for i in range(seq_len):
        #         for j in range(seq_len):
        #             # Compute each output element
        #             for k in range(seq_len):
        #                 out[b, i, j] += left[b, i, k, :] * right[b, j, k, :]

        out = self.to_out_norm(out)
        out = out * out_gate
        return self.to_out(out)

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the channel dimension (dim) – Triton kernel.
2️⃣ Cast the normalized tensor to FP16 and fuse the two projection + gate
   linear layers for the *left* and *right* paths (weights concatenated),
   reducing the number of GEMMs from 6 to 4.
3️⃣ Apply the optional mask to the projections, then element‑wise multiply
   each projection with its sigmoid‑gate.
4️⃣ Triangular multiplicative update:
        out_h[i,j] = Σ_k left_h[i,k] · right_h[j,k]   (h = hidden channel)
   This is a batched GEMM (`left @ rightᵀ`) performed in FP16.
5️⃣ Second LayerNorm over the hidden dimension – Triton kernel.
6️⃣ Apply the output‑gate (sigmoid) and a final FP16 linear projection back
   to `dim`.  The final result is cast back to FP32.

All heavy reductions (the two LayerNorms) run in Triton, while the
dense linear algebra is executed with PyTorch’s highly‑optimised FP16
kernels.  The implementation works for every configuration described
in the prompt (different batch sizes, sequence lengths, dimensions,
and with/without a mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       float*  [total_rows, dim]
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static dimension size
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    """
    Row‑wise LayerNorm for a 2‑D tensor stored in row‑major order.
    Computes:
        y = (x - μ) / √(σ² + eps) * γ + β
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean and variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # linear offset = row * dim + col
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values with γ (weight) and β (bias)
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x - mean) / sqrt(var + eps)
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ and β (broadcasted over rows)
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Prepare input
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                     # [B, N, N, dim]
    total_rows = B * N * N                    # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)          # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)    # destination

    BLOCK_M = 128      # tune‑able: rows per program
    BLOCK_D = 32       # tune‑able: columns per inner loop

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # Restore 4‑D shape (still FP32)
    x_norm = x_norm_flat.view(B, N, N, dim)   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Cast to FP16 for all dense linear algebra
    # ------------------------------------------------------------------
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse projection + gate for left and right paths
    # ------------------------------------------------------------------
    # concat weight matrices (out_features, in_features)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                        # [2*H, dim]
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)                        # [2*H, dim]

    # left: [total_rows, 2*hidden_dim]
    left_concat = F.linear(x_norm_h, left_concat_w)
    left_proj_h = left_concat[:, :hidden_dim]          # projection
    left_gate_pre_h = left_concat[:, hidden_dim:]      # pre‑sigmoid gate

    # right: same pattern
    right_concat = F.linear(x_norm_h, right_concat_w)
    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # out‑gate (no bias)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h, weights["out_gate.weight"].to(torch.float16))
    )   # [total_rows, hidden_dim]

    # ------------------------------------------------------------------
    # Apply optional mask (broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape [B, N, N] → flatten → [total_rows, 1]
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # Apply element‑wise gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h  = left_proj_h * left_gate_h    # [total_rows, hidden_dim]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 3️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left  = left_h.view(B, N, N, hidden_dim)   # [B, N, N, H]
    right = right_h.view(B, N, N, hidden_dim)

    # bring hidden dim to the second axis: [B, H, N, N]
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()   # transpose k ↔ j

    BH = B * hidden_dim                         # batch × hidden
    left_mat  = left_perm.view(BH, N, N)        # [BH, N, N] (FP16)
    right_mat = right_perm.view(BH, N, N)       # [BH, N, N]

    # batched matrix multiplication in FP16
    out_mat = torch.bmm(left_mat, right_mat)   # [BH, N, N]

    # reshape back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                           # FP16

    # ------------------------------------------------------------------
    # 4️⃣ Second LayerNorm over hidden_dim (float32)
    # ------------------------------------------------------------------
    # Cast to float32 for the kernel
    out_f32 = out.to(torch.float32)
    total_rows_h = B * N * N
    out_flat = out_f32.view(total_rows_h, hidden_dim)
    out_norm_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)   # float32

    # ------------------------------------------------------------------
    # 5️⃣ Apply output gate (sigmoid) and final projection
    # ------------------------------------------------------------------
    # out_gate was computed in FP16 → cast to float32 for multiplication
    out_gate_f32 = out_gate_h.view(B, N, N, hidden_dim).to(torch.float32)
    out_norm = out_norm * out_gate_f32

    # final linear projection (hidden_dim → dim) in FP16, then back to FP32
    out_norm_h = out_norm.to(torch.float16)
    final_h = F.linear(
        out_norm_h.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim]   (FP16)
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
