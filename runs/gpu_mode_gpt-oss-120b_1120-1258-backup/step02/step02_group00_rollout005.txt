    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # shape (BLOCK_M,)
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]  # each row stride * dim + offset
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0, dtype=tl.float16).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # second loop to write out
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs, mask=mask_m[:, None] & mask_d[None, :], other=0.0, dtype=tl.float16).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_half = out.to(tl.float16)  # cast back

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_half, mask=mask_m[:, None] & mask_d[None, :])
```

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) forward pass.
-----------------------------------------------------------------
Algorithm (all tensors stay on the same CUDA device):
1. Row‑wise LayerNorm over the input channel dim (fp16 I/O, fp32
   accumulation). Implemented with a Triton kernel.
2. Fuse the five linear projections (left/right + three gates) by
   concatenating their weight matrices and applying a single GEMM
   in fp16 (torch.nn.functional.linear).  The result is split back
   into the five tensors.
3. (Optional) apply the pairwise mask to the left/right projections.
4. Apply sigmoid gates (still fp16) and multiply element‑wise.
5. Triangular multiplicative update:
       out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
   This is a batched GEMM L @ Rᵀ with shape (B·H, N, N) performed by
   torch.bmm in fp16 (Tensor‑Core).
6. Second row‑wise LayerNorm over the hidden dimension (fp16 I/O,
   fp32 accumulation) – another Triton kernel.
7. Multiply by the final output gate and project back to `dim` with a
   fused fp16 linear layer.  Cast the result to float32 before returning.
The heavy LayerNorm reductions are done by custom Triton kernels, while
the large matrix‑multiplications use cuBLAS/Tensor‑Core for maximum
throughput.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (row‑wise, fp16 I/O, fp32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const half*   [total_rows, dim]
    y_ptr,          # *mut   half*   [total_rows, dim]
    weight_ptr,     # *const float*  [dim]
    bias_ptr,       # *const float*  [dim]
    total_rows,     # int (runtime)
    *,
    dim: tl.constexpr,      # compile‑time channel dimension
    eps: tl.constexpr,      # compile‑time epsilon
    BLOCK_M: tl.constexpr,  # rows per program
    BLOCK_D: tl.constexpr,  # channels per iteration
):
    """Row‑wise LayerNorm on a 2‑D tensor (half → half output)."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # First pass: mean & variance (float32 accumulation)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # pointer arithmetic: each row has stride = dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)    # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # Second pass: write normalized values (fp16)
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
            dtype=tl.float16,
        ).to(tl.float32)

        # (x - mean) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]          # fp32
        out_half = out.to(tl.float16)                # back to fp16

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out_half,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : [bs, seq_len, seq_len, dim] (float32, CUDA)
            - mask_tensor  : [bs, seq_len, seq_len] (float32 or bool)
            - weights_dict : model parameters (float32 tensors)
            - config_dict  : {"dim": ..., "hidden_dim": ..., "nomask": bool}

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, CUDA)
    """
    # --------------------------------------------------------------
    # Unpack
    # --------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]                # input channel dimension C
    hidden_dim = config["hidden_dim"]  # projection dimension H
    eps = 1e-5
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Convert input to the compute datatype (fp16)
    # --------------------------------------------------------------
    x_fp16 = input_tensor.half()                     # [B, N, N, C]
    B, N, _, _ = x_fp16.shape
    total_rows = B * N * N                           # rows for flattened view

    # --------------------------------------------------------------
    # 1) First LayerNorm over `dim` (fp16 I/O)
    # --------------------------------------------------------------
    x_flat = x_fp16.view(total_rows, dim)            # [total_rows, C]
    x_norm = torch.empty(total_rows, dim,
                         dtype=torch.float16,
                         device=device)           # result buffer

    # Block sizes – tuned for H100, works for dim up to 384
    BLOCK_M = 256          # rows per program
    BLOCK_D = 64           # channels per tile

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim=dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # ------------------------------------------------------------------
    # 2) Fused linear projections + gates (single GEMM, fp16)
    # ------------------------------------------------------------------
    # Cast all projection / gate weights once to fp16
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    concat_weight = torch.cat(
        [w_left_proj, w_right_proj,
         w_left_gate, w_right_gate,
         w_out_gate],
        dim=0,
    ).half().contiguous()                              # (5*H, C)

    # Single fused linear
    proj_all = F.linear(x_norm, concat_weight)          # (total_rows, 5*H), fp16

    # Split back
    left_proj, right_proj, left_gate, right_gate, out_gate = torch.split(
        proj_all, hidden_dim, dim=1
    )                                                    # each (total_rows, H)

    # --------------------------------------------------------------
    # 2a) Optional mask (broadcast over hidden dimension)
    # --------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → [total_rows, 1] → fp16
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_flat
        right_proj = right_proj * mask_flat

    # --------------------------------------------------------------
    # 2b) Apply sigmoid gates (still fp16)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate)    # (total_rows, H)
    right_gate = torch.sigmoid(right_gate)
    out_gate = torch.sigmoid(out_gate)

    left = left_proj * left_gate            # (total_rows, H)
    right = right_proj * right_gate

    # --------------------------------------------------------------
    # Reshape for the triangular update
    # --------------------------------------------------------------
    left = left.view(B, N, N, hidden_dim)               # [B, N, N, H]
    right = right.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)       # gate for later

    # Permute to (B, H, N, N) layout required by batched GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    # right needs to be transposed on the K dimension → (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)                # (BH, N, N)
    right_mat = right_perm.view(BH, N, N)              # (BH, N, N)

    # --------------------------------------------------------------
    # 3) Triangular multiplicative update (batched GEMM, fp16)
    # --------------------------------------------------------------
    out_mat = torch.bmm(left_mat, right_mat)           # (BH, N, N), fp16

    # Reshape back to 4‑D tensor [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 4) Second LayerNorm over hidden_dim (fp16 I/O)
    # --------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)      # (total_rows_h, H)
    out_norm = torch.empty_like(out_flat)              # half buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        dim=hidden_dim,
        eps=eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    out_norm = out_norm.view(B, N, N, hidden_dim)       # [B, N, N, H]

    # --------------------------------------------------------------
    # 5) Apply final output gate and project back to `dim`
    # --------------------------------------------------------------
    gated = out_norm * out_gate                         # fp16 element‑wise

    # Final linear projection: hidden_dim → dim (fp16)
    to_out_weight = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # (total_rows_h, H)
    final_half = F.linear(gated_flat, to_out_weight)   # (total_rows_h, dim), fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)  # cast to fp32

    return final
