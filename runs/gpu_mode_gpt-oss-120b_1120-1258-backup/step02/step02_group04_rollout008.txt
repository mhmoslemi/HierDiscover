1. **Fusion of the five linear projections** (left_proj, right_proj,
   left_gate, right_gate, out_gate).  All five are concatenated into a single
   weight matrix of shape (5·hidden, dim) and performed with a single
   half‑precision GEMM (`F.linear`).  This reduces the memory traffic on the
   huge input tensor `[B, N, N, dim]` from five passes to one.

2. **Batched GEMM for the N³ contraction**.  After the projections the
   contraction
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   is a batch of `B·hidden` matrix multiplications `A @ Bᵀ`.  We launch a
   custom Triton kernel that streams K‑tiles of size 128 through shared
   memory and uses Tensor‑cores (`tl.dot`) with FP16→FP32 accumulation.

3. **All remaining element‑wise work (masking, sigmoid gating, LayerNorm,
   final projection) stays in PyTorch** – these kernels are well‑optimised
   and their cost is negligible compared with the O(N³·hidden) work.

The routine returns a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ  (A : M×K , B : N×K)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr,                     # fp16 * (batch, M, K)
    b_ptr,                     # fp16 * (batch, N, K)  (will be read as K×N)
    c_ptr,                     # fp16 * (batch, M, N)
    M, N, K,                   # int32 sizes
    stride_az, stride_am, stride_ak,   # strides for A
    stride_bz, stride_bn, stride_bk,   # strides for B (note: K dimension is stride_bk)
    stride_cz, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched half‑precision GEMM with Tensor‑core acceleration.
    The first program‑id packs (batch, tile_m) and the second program‑id
    enumerates tile_n.
    """
    pid0 = tl.program_id(0)          # batch_id * tiles_M + tile_m
    pid1 = tl.program_id(1)          # tile_n

    # ------------------------------------------------------------------
    # Decode batch index and tile coordinates
    # ------------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    batch_id   = pid0 // num_tiles_m
    tile_m     = pid0 %  num_tiles_m
    tile_n     = pid1

    # ------------------------------------------------------------------
    # Offsets inside the current tile
    # ------------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # ------------------------------------------------------------------
    # Pointers to the start of the tile (no k‑loop offset yet)
    # ------------------------------------------------------------------
    a_ptrs = a_ptr + batch_id * stride_az \
                  + offs_m[:, None] * stride_am \
                  + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + batch_id * stride_bz \
                  + offs_k[:, None] * stride_bk \
                  + offs_n[None, :] * stride_bn

    # ------------------------------------------------------------------
    # Accumulator (FP32 for numerical safety)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # out‑of‑bounds handling for the very last tiles
        mask_a = (offs_m[:, None] < M) & (offs_k[None, :] < K)
        mask_b = (offs_k[:, None] < K) & (offs_n[None, :] < N)

        a = tl.load(a_ptrs, mask=mask_a, other=0.0)   # [BLOCK_M, BLOCK_K]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)   # [BLOCK_K, BLOCK_N]

        # fp16·fp16 → fp32 accumulation (Tensor‑cores)
        acc += tl.dot(a, b)

        # advance K‑tiles
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # Store result (back to fp16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_cz \
                    + offs_m[:, None] * stride_cm \
                    + offs_n[None, :] * stride_cn
    mask_c = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_c)


def custom_kernel(data):
    """
    Forward pass of the *outgoing* TriMul operator.
    -------------------------------------------------
    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        input   – torch.Tensor [B, N, N, dim] (float32)
        mask    – torch.Tensor [B, N, N] (bool/float) or None
        weights – dict of parameters (float32)
        config  – dict with keys {"dim", "hidden_dim", "nomask"}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # 0) unpack
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (FP32 for numerical stability)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2) Cast to fp16 – heavy work will use half‑precision
    # -----------------------------------------------------------------
    x_fp16 = x.to(torch.float16)

    # -----------------------------------------------------------------
    # 3) Fuse the five linear projections into one GEMM
    #    weight_all : (5*hidden, dim)
    # -----------------------------------------------------------------
    # Concatenate weights on the output‑feature dimension (dim is the input dim)
    weight_all = torch.cat(
        [
            weights["left_proj.weight"],   # (hidden, dim)
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # (5*hidden, dim)

    # flatten input to (B*N*N, dim)
    B, N, _, _ = x_fp16.shape
    x_flat = x_fp16.view(-1, dim)                # (B·N·N, dim)

    # single fused linear pass
    proj_all = F.linear(x_flat, weight_all)      # (B·N·N, 5*hidden)
    proj_all = proj_all.view(B, N, N, 5 * hidden)

    # split the concatenated projection
    left      = proj_all[..., 0                : hidden]          # (B,N,N,hidden)
    right     = proj_all[..., hidden          : 2 * hidden]      # (B,N,N,hidden)
    left_gate = proj_all[..., 2 * hidden      : 3 * hidden]      # (B,N,N,hidden)
    right_gate= proj_all[..., 3 * hidden      : 4 * hidden]      # (B,N,N,hidden)
    out_gate  = proj_all[..., 4 * hidden      : 5 * hidden]      # (B,N,N,hidden)

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid) – still fp16, then broadcast cast to fp32 later
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate)
    right_gate = torch.sigmoid(right_gate)
    out_gate   = torch.sigmoid(out_gate)

    # -----------------------------------------------------------------
    # 5) Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask_fp16
        right = right * mask_fp16

    # -----------------------------------------------------------------
    # 6) Apply the element‑wise gates
    # -----------------------------------------------------------------
    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 7) N³ contraction:  out = Σₖ left[...,k,:] * right[...,k,:]
    #    Implemented as a batched GEMM (A @ Bᵀ) with a custom Triton kernel.
    # -----------------------------------------------------------------
    # reshape to (batch*hidden, M, K) and (batch*hidden, N, K)
    #   left  -> (B*hidden, N, N)   (row‑major)
    #   right -> (B*hidden, N, N)   (will be accessed as K×N inside the kernel)
    left_mat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden, N, N)
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden, N, N)

    # output buffer (fp16, accumulation is fp32 inside the kernel)
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------
    # Triton launch configuration – tuned for H100
    # -----------------------------------------------------------------
    BLOCK_M = 128          # rows per tile
    BLOCK_N = 128          # cols per tile
    BLOCK_K = 128          # depth tile (bigger K‑tile reduces loop count)

    grid_m = triton.cdiv(N, BLOCK_M)
    grid_n = triton.cdiv(N, BLOCK_N)

    # First program‑id packs (batch*hidden) * tiles_M, second is tile_N
    grid = (B * hidden * grid_m, grid_n)

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_az=left_mat.stride(0),
        stride_am=left_mat.stride(1),
        stride_ak=left_mat.stride(2),
        stride_bz=right_mat.stride(0),
        stride_bn=right_mat.stride(1),   # B's N‑dim stride
        stride_bk=right_mat.stride(2),   # B's K‑dim stride
        stride_cz=out_mat.stride(0),
        stride_cm=out_mat.stride(1),
        stride_cn=out_mat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # best for Tensor‑cores on H100
        num_stages=3,
    )

    # -----------------------------------------------------------------
    # 8) Reshape back to [B, N, N, hidden] and cast to FP32 for the rest
    # -----------------------------------------------------------------
    out = out_mat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 9) Output LayerNorm (hidden dim) + out‑gate scaling
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate.to(torch.float32)

    # -----------------------------------------------------------------
    # 10) Final projection back to the original `dim`
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # (dim, hidden), bias=False

    return out
```
