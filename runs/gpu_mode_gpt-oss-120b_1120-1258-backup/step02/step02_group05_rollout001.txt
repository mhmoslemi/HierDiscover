"""
TriMul (outgoing) – Triton‑accelerated cubic contraction.

Algorithm
---------
1. Layer‑norm the 4‑D input over the feature dimension (float32).
2. Cast to FP16 and apply bias‑free linear projections for the left/right
   tensors together with three sigmoid gates.
3. (Optional) mask the projected tensors – the mask is broadcasted over the
   hidden dimension.
4. Fuse the hidden dimension into the batch dimension and run a custom
   Triton kernel that computes, for every (batch, hidden) pair

        out[b,h,i,j] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   i.e. a batched GEMM with A,B in FP16 and accumulation in FP32.
   The kernel tiles the i‑ and j‑axes (BLOCK_M × BLOCK_N) and the reduction
   k‑axis (BLOCK_K) and uses `tl.dot` (Tensor‑core friendly) for the inner
   product.
5. Reshape the result back to [B, N, N, hidden] (float32), apply a second
   Layer‑norm, multiply by the output gate and finally a bias‑free linear
   projection back to the original feature dimension.

The heavy O(N³·H) contraction is performed entirely in Triton; the remaining
operations stay in PyTorch for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_tri_gemm_kernel(
    A,               # left  tensor (M, N, N)  – FP16
    B,               # right tensor (M, N, N)  – FP16 (read as Bᵀ)
    C,               # output (M, N, N)       – FP32
    N,               # sequence length
    stride_am, stride_ai, stride_ak,   # strides of A: batch, row(i), col(k)
    stride_bm, stride_bk, stride_bj,   # strides of B: batch, row(k), col(j)
    stride_cm, stride_ci, stride_cj,   # strides of C: batch, row(i), col(j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM C = A @ Bᵀ for each entry in the first dimension.
    All tensors are contiguous, A/B are FP16, C is FP32.
    """
    batch_id = tl.program_id(0)          # 0‑th axis → (batch * hidden_dim)
    pid_m   = tl.program_id(1)          # tile‑row index
    pid_n   = tl.program_id(2)          # tile‑col index

    # start coordinates of the tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # row / col indices within the tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32 (Tensor‑core dot returns FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over reduction dimension k in BLOCK_K steps
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---- load tile of A (shape BLOCK_M × BLOCK_K) ----
        # A[batch_id, i, k]
        a_ptr = A + batch_id * stride_am + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(
            a_ptr,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float16)

        # ---- load tile of Bᵀ (shape BLOCK_K × BLOCK_N) ----
        # Bᵀ is read as B[batch_id, k, j]
        b_ptr = B + batch_id * stride_bm + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(
            b_ptr,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float16)

        # ---- inner product (Tensor‑core) ----
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write the tile back to C
    c_ptr = C + batch_id * stride_cm + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(
        c_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for the
    cubic contraction.  All tensors are assumed to reside on the CUDA device.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: [B, N, N, dim] (float32)
            - mask:          [B, N, N]   (any dtype) – may be ignored if config["nomask"]=True
            - weights: dict with model parameters (tensors on the same device)
            - config:  dict with keys:
                "dim"        – input feature dimension,
                "hidden_dim" – hidden dimension after projection,
                "nomask"     – bool, whether to ignore the mask.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------ #
    # 1. Input LayerNorm (float32, stable)
    # ------------------------------------------------------------------ #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim] – float32

    # ------------------------------------------------------------------ #
    # 2. FP16 projections + sigmoid gates
    # ------------------------------------------------------------------ #
    # cast once to fp16
    x_h = x.half()                     # [B, N, N, dim] – fp16

    # pre‑cast weights to fp16 to avoid repeated conversion
    w_left_proj   = weights["left_proj.weight"].half()
    w_right_proj  = weights["right_proj.weight"].half()
    w_left_gate   = weights["left_gate.weight"].half()
    w_right_gate  = weights["right_gate.weight"].half()
    w_out_gate    = weights["out_gate.weight"].half()
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    w_to_out      = weights["to_out.weight"]

    # bias‑free linear projections
    left  = F.linear(x_h, w_left_proj)   # [B, N, N, H] – fp16
    right = F.linear(x_h, w_right_proj)  # [B, N, N, H] – fp16

    # sigmoid gates (still fp16)
    left_gate  = torch.sigmoid(F.linear(x_h, w_left_gate))
    right_gate = torch.sigmoid(F.linear(x_h, w_right_gate))
    out_gate   = torch.sigmoid(F.linear(x_h, w_out_gate))

    # apply element‑wise gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------ #
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] → fp16
        mask_exp = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------ #
    # 4. Batched GEMM: out[b, h, i, j] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    # ------------------------------------------------------------------ #
    B, N, _, _ = left.shape  # N is the sequence length

    # bring hidden dim to the second axis and make the tensor contiguous
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # collapse (B, H) → M_total so the kernel sees a single batch dimension
    M_total = B * hidden_dim
    left_flat  = left_perm.view(M_total, N, N)   # [M, N, N] – fp16
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N] – fp16
    out_flat   = torch.empty((M_total, N, N), dtype=torch.float32, device=left_flat.device)

    # ------------------------------------------------------------------ #
    # 5. Triton kernel launch
    # ------------------------------------------------------------------ #
    # choose tile sizes (tuned for H100 Tensor‑cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    # compute strides (int64 scalars, as required by Triton)
    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bk = right_flat.stride(1)   # row (k) stride
    stride_bj = right_flat.stride(2)   # col (j) stride

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    # launch grid: (M_total, tiles_i, tiles_j)
    grid = (
        M_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_tri_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bk,
        stride_bj,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # H100 prefers 4‑warps per block for this shape
        num_stages=3,         # pipelined prefetch
    )

    # reshape back to [B, N, N, H] (float32)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 6. Output LayerNorm, output‑gate multiplication, final linear
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
        eps=eps,
    )
    # out_gate is fp16 – promote to fp32 before multiplication
    out = out * out_gate.float()

    # final bias‑free linear projection back to dim
    output = F.linear(out, w_to_out)   # [B, N, N, dim] – float32
    return output
```

This implementation follows the reference PyTorch logic but offloads the dominant cubic contraction to a custom Triton kernel. The kernel tiles the `i`‑ and `j`‑axes (BLOCK_M × BLOCK_N) and the reduction axis `k` (BLOCK_K), uses `tl.dot` for Tensor‑core‑accelerated inner products, and accumulates in FP32 for numerical stability. Optional masking is handled in PyTorch (broadcasted over the hidden dimension) to keep the kernel simple and fast. The overall design respects all constraints, supports both masked and un‑masked inputs, and keeps the final output in `float32`.
