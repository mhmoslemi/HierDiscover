@triton.jit
def layernorm_gate_kernel(x_ptr, y_ptr, w_ptr, b_ptr, gate_ptr, total_rows, dim, eps, BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)
    mask_m = offs_m < total_rows

    # compute sums
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_M], dtype=tl.float32)
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr, mask=mask_m[:, None] & mask_d[None,:], other=0.0).to(tl.float32)
        norm = (vals - mean[:, None]) * rstd[:, None]
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)
        gate_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :], mask=mask_m[:, None] & mask_d[None, :], other=0.0).to(tl.float32)
        out = (norm * w[None, :] + b[None, :]) * gate_vals
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out, mask=mask_m[:, None] & mask_d[None, :])
```

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    x_norm_flat = torch.empty_like(x_flat)
    BLOCK_M = 512
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M), )
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # no explicit sync

    x_norm = x_norm_flat.view(B, N, N, dim)

    # Cast to fp16 for dense ops
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # Fused linear for left and left_gate
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    left_concat = F.linear(x_norm_h, left_concat_w)  # [total_rows, 2*hidden]
    left_proj_h = left_concat[:, :hidden_dim]
    left_gate_pre_h = left_concat[:, hidden_dim:]

    right_concat = F.linear(x_norm_h, right_concat_w)
    right_proj_h = right_concat[:, :hidden_dim]
    right_gate_pre_h = right_concat[:, hidden_dim:]

    # out gate
    out_gate_h = torch.sigmoid(F.linear(x_norm_h, weights["out_gate.weight"].to(torch.float16)))
    out_gate_f32 = out_gate_h.to(torch.float32)

    # Apply mask if needed
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # Apply gates
    left_gate_h = torch.sigmoid(left_gate_pre_h)
    right_gate_h = torch.sigmoid(right_gate_pre_h)

    left_h = left_proj_h * left_gate_h
    right_h = right_proj_h * right_gate_h

    # Triangular multiplicative update (batched GEMM)
    left = left_h.view(B, N, N, hidden_dim)
    right = right_h.view(B, N, N, hidden_dim)

    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] where k,j

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    out_mat = torch.bmm(left_mat, right_mat)  # [BH, N, N]

    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]
    out_f32 = out.to(torch.float32)

    # Second LayerNorm + gate fusion
    total_rows_h = B * N * N
    out_flat = out_f32.view(total_rows_h, hidden_dim)
    out_ln_gate_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_kernel[grid2](

        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )

    out_ln_gate = out_ln_gate_flat.view(B, N, N, hidden_dim)
    out_final_h = F.linear(out_ln_gate.view(total_rows_h, hidden_dim), weights["to_out.weight"].to(torch.float16))

    return out_final

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm over the input channel dimension (`dim`).  Implemented in
   pure Triton to avoid the overhead of torch.nn.LayerNorm.
2️⃣ Cast the normalized tensor to FP16 and fuse the two linear projections
   (`left_proj` & `left_gate`, `right_proj` & `right_gate`) by concatenating the
   weight matrices.  This reduces the number of GEMMs from 6 to 4.
3️⃣ (Optional) apply the pair‑wise mask and the sigmoid‑gates to the
   projections.
4️⃣ Tri‑multiplicative update:
        out[b,i,j,d] = Σₖ left[b,i,k,d] · right[b,j,k,d]
   This is a batched matrix multiplication (`left @ rightᵀ`) performed with
   torch.bmm on FP16 tensors.
5️⃣ Second LayerNorm over the hidden dimension (`hidden_dim`) **and**
   multiplication by the output‑gate are fused in a single Triton kernel.
6️⃣ Final FP16 linear projection back to `dim` and cast to FP32.

All heavy reductions (the two LayerNorms) run in Triton, while dense linear
algebra uses PyTorch’s highly‑optimised cuBLAS kernels.  The implementation
works for every shape listed in the prompt (different batch sizes, sequence
lengths, hidden dimensions, with or without a mask).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       float*  [total_rows, dim]
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # ------------------------------------------------------------------
    # Pass 1 – compute mean & variance per row
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)                # row‑wise sum
        sum_sq  += tl.sum(vals * vals, axis=1)         # row‑wise sum of squares

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)                         # 1 / √(var+eps)

    # ------------------------------------------------------------------
    # Pass 2 – write normalized values (γ, β) and store
    # ------------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton LayerNorm + gate fusion (float32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const float*  [total_rows, dim]
    y_ptr,          #       float*  [total_rows, dim] (LN(x) * gate)
    w_ptr,          # const float*  [dim]   (γ)
    b_ptr,          # const float*  [dim]   (β)
    gate_ptr,       # const float*  [total_rows, dim] (pre‑computed gate)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static number of columns
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # columns per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # --------------------------------------------------------------
    # Pass 1 – mean & variance
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # Pass 2 – write LN(x) * gate
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        gate = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = (norm * w[None, :] + b[None, :]) * gate
        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: [B, N, N, dim]   (float32, CUDA)
            - mask_tensor : [B, N, N]        (any dtype, CUDA)
            - weights_dict: model parameters as torch Tensors
            - config_dict : {'dim': ..., 'hidden_dim': ..., 'nomask': bool, ...}

    Returns:
        Tensor of shape [B, N, N, dim] (float32, CUDA)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 0️⃣ Clean view & basic shapes
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                     # [B, N, N, dim]
    total_rows = B * N * N                    # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32) – Triton
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)          # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)    # destination

    BLOCK_M = 512      # rows per program (tunable, ≤65536/BLOCK_D)
    BLOCK_D = 64       # columns per inner loop (tunable, divides dim)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # (no explicit sync – launch is asynchronous)

    # Restore the 4‑D shape (still FP32)
    x_norm = x_norm_flat.view(B, N, N, dim)

    # ------------------------------------------------------------------
    # 2️⃣ Cast to FP16 and fuse left/right projection + gate
    # ------------------------------------------------------------------
    x_norm_h = x_norm.view(total_rows, dim).to(torch.float16)

    # concatenate proj & gate weights (out_features = 2*hidden_dim)
    left_concat_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).to(torch.float16)                        # [2*H, dim]
    right_concat_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).to(torch.float16)

    # fused linear (proj + gate) for left side
    left_concat = F.linear(x_norm_h, left_concat_w)                # [total_rows, 2*H]
    left_proj_h   = left_concat[:, :hidden_dim]                     # projection
    left_gate_pre = left_concat[:, hidden_dim:]                     # pre‑sigmoid gate

    # fused linear for right side
    right_concat = F.linear(x_norm_h, right_concat_w)              # [total_rows, 2*H]
    right_proj_h   = right_concat[:, :hidden_dim]
    right_gate_pre = right_concat[:, hidden_dim:]

    # out‑gate (no bias)
    out_gate_h = torch.sigmoid(
        F.linear(x_norm_h, weights["out_gate.weight"].to(torch.float16))
    )   # [total_rows, hidden_dim]
    out_gate_f32 = out_gate_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj_h  = left_proj_h * mask_h
        right_proj_h = right_proj_h * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply element‑wise sigmoid‑gates
    # ------------------------------------------------------------------
    left_gate_h  = torch.sigmoid(left_gate_pre)
    right_gate_h = torch.sigmoid(right_gate_pre)

    left_h  = left_proj_h * left_gate_h      # [total_rows, hidden_dim]
    right_h = right_proj_h * right_gate_h

    # ------------------------------------------------------------------
    # 5️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    left = left_h.view(B, N, N, hidden_dim)    # [B, N, N, H]
    right = right_h.view(B, N, N, hidden_dim)

    # shape → [B, H, N, N] where axes are (batch, hidden, i, k) / (batch, hidden, k, j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()

    BH = B * hidden_dim                        # batch × hidden
    left_mat  = left_perm.view(BH, N, N)       # [BH, N, N]
    right_mat = right_perm.view(BH, N, N)      # [BH, N, N]

    # batched half‑precision matrix multiplication
    out_mat = torch.bmm(left_mat, right_mat)   # [BH, N, N]

    # back to [B, N, N, H]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )                                           # FP16

    # ------------------------------------------------------------------
    # 6️⃣ Second LayerNorm + output‑gate (fused in Triton)
    # ------------------------------------------------------------------
    out_f32 = out.to(torch.float32)
    total_rows_h = B * N * N
    out_flat = out_f32.view(total_rows_h, hidden_dim)            # [rows, hidden]
    out_ln_gate_flat = torch.empty_like(out_flat)

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_gate_kernel[grid2](
        out_flat,
        out_ln_gate_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        out_gate_f32.view(total_rows_h, hidden_dim),
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    # (still asynchronous)

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out_ln_gate = out_ln_gate_flat.view(B, N, N, hidden_dim).to(torch.float16)
    final_h = F.linear(
        out_ln_gate.view(total_rows_h, hidden_dim),
        weights["to_out.weight"].to(torch.float16)
    )   # [total_rows_h, dim]   (FP16)
    final = final_h.view(B, N, N, dim).to(torch.float32)

    return final
