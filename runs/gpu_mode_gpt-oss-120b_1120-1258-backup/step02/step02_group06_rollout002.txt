@triton.jit
def tri_mul_gemm_kernel(a_ptr, b_ptr, c_ptr,
                        N,
                        stride_am, stride_ai, stride_ak,
                        stride_bm, stride_bj, stride_bk,
                        stride_cm, stride_ci, stride_cj,
                        BLOCK_M: tl.constexpr,
                        BLOCK_N: tl.constexpr,
                        BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # offsets
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    # mask for out-of-range indices
    mask_i = i_range < N
    mask_j = j_range < N

    # compute pointers
    c_ptrs = c_ptr + pid_bh * stride_cm + i_range[:, None] * stride_ci + j_range[None, :] * stride_cj

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over tiles of k dimension
    for k_base in range(0, N, BLOCK_K):
        k_range = k_base + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        a_ptrs = a_ptr + pid_bh * stride_am + i_range[:, None] * stride_ai + k_range[None, :] * stride_ak
        b_ptrs = b_ptr + pid_bh * stride_bm + k_range[:, None] * stride_bk + j_range[None, :] * stride_bj

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        acc += tl.dot(a, b, out_dtype=tl.float32)

    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul‑Outgoing (AlphaFold3) – Triton‑accelerated implementation
================================================================
The function `custom_kernel` implements the forward pass of the
“outgoing” TriMul operation:

1. Layer‑norm over the input feature dimension (dim) – PyTorch.
2. Bias‑free linear projections → `left` and `right` (fp16).
3. Optional pair‑wise mask (broadcasted on the hidden dim).
4. Three sigmoid gates (left, right, out) – fp16.
5. Cubic contraction
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   realised as a batched GEMM `C = A @ Bᵀ` with `A,B ∈ ℝ^{N×N}`
   for each `(batch, hidden)` pair.  The heavy O(N³·H) work is
   executed in a custom Triton kernel that:
   * loads tiles of `A` and the transposed `B` in fp16,
   * multiplies them with `tl.dot` (Tensor‑Core), accumulating in fp32.
6. Layer‑norm over the hidden dimension, element‑wise multiplication
   with the output gate, and a final bias‑free linear projection back
   to the original dimension.

All expensive arithmetic (steps 2‑5) is performed in half‑precision
to maximise bandwidth and Tensor‑Core utilisation; the final result
is returned as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM C = A @ Bᵀ
#   A, B  : (M, N, N)   fp16
#   C     : (M, N, N)   fp32
#   M = batch * hidden_dim
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,               # left  matrix  (M, N, N)  – fp16
    b_ptr,               # right matrix  (M, N, N)  – fp16 (read transposed)
    c_ptr,               # output matrix (M, N, N) – fp32
    N,                   # sequence length (int)
    stride_am, stride_ai, stride_ak,   # strides of A: (M, i, k)
    stride_bm, stride_bj, stride_bk,   # strides of B: (M, j, k) – we index as (k, j)
    stride_cm, stride_ci, stride_cj,   # strides of C: (M, i, j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Tile identifiers
    # ------------------------------------------------------------------
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_bh = tl.program_id(2)  # batch * hidden index

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # Index vectors for the current tile
    # ------------------------------------------------------------------
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # Pointers to the output tile C[i, j]
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_bh * stride_cm
        + i[:, None] * stride_ci
        + j[None, :] * stride_cj
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Reduction over k (the contracted dimension)
    # ------------------------------------------------------------------
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i, k] – fp16 tile
        a_ptrs = (
            a_ptr
            + pid_bh * stride_am
            + i[:, None] * stride_ai
            + k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=tl.float16(0.0),
        )

        # B[k, j] – we read the transposed view of right (fp16)
        b_ptrs = (
            b_ptr
            + pid_bh * stride_bm
            + k[:, None] * stride_bk
            + j[None, :] * stride_bj
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=tl.float16(0.0),
        )

        # Tensor‑core dot, accumulate in fp32
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ------------------------------------------------------------------
    # Write the output tile
    # ------------------------------------------------------------------
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(
    data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim]    (float32)
            - mask         : [B, N, N]         (float32 / bool) – may be ignored
            - weights      : dict of model parameters (already on device)
            - config       : dict with keys
                * "dim"        – input feature dimension
                * "hidden_dim" – hidden dimension of the projection
                * "nomask"     – bool, True ⇒ ignore mask

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # Unpack weights (all are bias‑free linear layers except the LayerNorms)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the input feature dim
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # Work in half‑precision for the heavy path
    x_h = x.half()

    # ------------------------------------------------------------------
    # 2) Linear projections (bias‑free) → left, right
    # ------------------------------------------------------------------
    left_proj_w_h = left_proj_weight.half()
    right_proj_w_h = right_proj_weight.half()
    left = F.linear(x_h, left_proj_w_h)   # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w_h)  # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 3) Optional pair‑wise mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; cast to the same dtype as x_h
        mask_exp = mask.to(x_h.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating – still in fp16
    # ------------------------------------------------------------------
    left_gate_w_h = left_gate_weight.half()
    right_gate_w_h = right_gate_weight.half()
    out_gate_w_h = out_gate_weight.half()

    left_gate = torch.sigmoid(F.linear(x_h, left_gate_w_h))   # [B,N,N,H] fp16
    right_gate = torch.sigmoid(F.linear(x_h, right_gate_w_h))  # [B,N,N,H] fp16
    out_gate = torch.sigmoid(F.linear(x_h, out_gate_w_h))    # [B,N,N,H] fp16

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Batched GEMM C = A @ Bᵀ  (the O(N³·H) core)
    # ------------------------------------------------------------------
    #   We treat each (batch, hidden) pair as an independent GEMM.
    #   Shape transformation: (B, N, N, H) → (B, H, N, N) → (M, N, N)
    #   where M = B * H.
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] fp16
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N] fp16

    B, H, N, _ = left_perm.shape
    M_total = B * H

    left_flat = left_perm.view(M_total, N, N)    # [M, N, N] fp16
    right_flat = right_perm.view(M_total, N, N)  # [M, N, N] fp16
    out_flat = torch.empty(
        (M_total, N, N), dtype=torch.float32, device=left_flat.device
    )  # [M, N, N] fp32

    # ----- Triton launch configuration ---------------------------------
    # tile size 64×64 works well for all supported N (≤1024) on H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    stride_am = left_flat.stride(0)
    stride_ai = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bm = right_flat.stride(0)
    stride_bj = right_flat.stride(1)   # stride over the “j” dimension
    stride_bk = right_flat.stride(2)   # stride over the “k” dimension

    stride_cm = out_flat.stride(0)
    stride_ci = out_flat.stride(1)
    stride_cj = out_flat.stride(2)

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles over i
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles over j
        M_total,                        # one program per (batch, hidden)
    )

    # Launch kernel. 8 warps give good occupancy for the 64×64 tile.
    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bj,
        stride_bk,
        stride_cm,
        stride_ci,
        stride_cj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
    )

    # Reshape back to the original 4‑D layout: [B, N, N, H]
    out = (
        out_flat.view(B, H, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # [B, N, N, H]  fp32

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm, output gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # fp32

    # Apply output gate (broadcast from fp16 to fp32)
    out = out * out_gate.to(out.dtype)

    # Final linear projection back to original dim (bias‑free)
    output = F.linear(out, to_out_weight)   # [B, N, N, dim]  float32

    return output
