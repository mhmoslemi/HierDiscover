import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr,
                           N_ELEMENTS,
                           stride_a, stride_b, stride_c,
                           BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N_ELEMENTS
    a = tl.load(a_ptr + offs * stride_a, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs * stride_b, mask=mask, other=0.0)
    c = a * b
    tl.store(c_ptr + offs * stride_c, c, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # cast to half
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape

    # concatenate all projection weights
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    W_all = torch.cat([left_proj_w, right_proj_w,
                       left_gate_w, right_gate_w,
                       out_gate_w], dim=0)  # (5*hidden, dim)

    # fused linear projection
    x_flat = x_h.view(-1, dim)  # (B*N*N, dim)
    out_all = torch.nn.functional.linear(x_flat, W_all)  # (M, 5*hidden)

    # split and reshape
    # split into 5 tensors of shape (M, hidden)
    splits = torch.split(out_all, hidden, dim=1)
    left = splits[0].view(B, N, N, hidden)
    right = splits[1].view(B, N, N, hidden)
    left_gate = splits[2].view(B, N, N, hidden).sigmoid()
    right_gate = splits[3].view(B, N, N, hidden).sigmoid()
    out_gate = splits[4].view(B, N, N, hidden).sigmoid()

    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # (B,N,N,1)
        left = left * mask_h
        right = right * mask_h

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Heavy multiplicative update (batched GEMM via torch)
    # reshape for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) with J,K swapped
    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    # compute batched matmul, fp16 inputs, fp32 accumulation automatically
    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden) FP16

    # Convert to float32 for layernorm and final steps
    out = out.to(torch.float32)

    # Second layer norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # Multiply with out_gate (elementwise) using Triton kernel
    # note out_gate currently half; convert to float32
    out_gate_f32 = out_gate.to(torch.float32)

    # allocate result buffer
    out_mul = torch.empty_like(out)
    total_elements = out.numel()
    # compute grid
    BLOCK_SIZE = 1024  # choose 1024 threads per block for elementwise kernel
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    elementwise_mul_kernel[grid](
        out,
        out_gate_f32,
        out_mul,
        total_elements,
        out.stride(0), out_gate_f32.stride(0), out_mul.stride(0),
        BLOCK=BLOCK_SIZE,
    )
    # after kernel, out_mul holds out * out_gate

    # Final linear projection

    out = torch.nn.functional.linear(out_h, to_out_weight_h).to(torch.float32)

    return out

import torch
import triton
import triton.language as tl

@triton.jit
def _elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, N,
                            BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK + tl.arange(0, BLOCK)
    mask = offs < N
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    tl.store(c_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Optimized forward pass of the “outgoing” TriMul operator.

    The implementation fuses the five linear projections into a single
    matrix multiplication, uses torch’s highly‑optimized batched GEMM for
    the O(N³) update, and only a tiny Triton kernel for the final
    element‑wise multiplication with the output gate.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # 1) LayerNorm on channel dimension
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )                                   # [B, N, N, dim]  FP32

    # 2) Cast to FP16 for all heavy linear work
    x_h = x.to(torch.float16)

    B, N, _, _ = x_h.shape

    # 3) Fuse all five Linear layers (left, right, left_gate, right_gate, out_gate)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)   # (hidden, dim)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # concatenate along the output‑feature dimension
    W_all = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    )                                   # (5*hidden, dim)

    # Perform the single big matmul
    x_flat = x_h.view(-1, dim)          # (B*N*N, dim)
    out_all = torch.nn.functional.linear(x_flat, W_all)   # (B*N*N, 5*hidden)

    # Split back and reshape
    splits = torch.split(out_all, hidden, dim=1)
    left = splits[0].view(B, N, N, hidden)
    right = splits[1].view(B, N, N, hidden)
    left_gate = splits[2].view(B, N, N, hidden).sigmoid()
    right_gate = splits[3].view(B, N, N, hidden).sigmoid()
    out_gate = splits[4].view(B, N, N, hidden).sigmoid()

    # 4) Optional pairwise mask (broadcast on hidden)
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # 5) Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # 6) Heavy O(N³) multiplicative update – batched GEMM via torch (FP16 inputs, FP32 accumulation)
    #   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, hidden, N, N) -> (B, hidden, K, J)

    BH = B * hidden
    left_flat = left_perm.view(BH, N, N)                # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)              # (BH, K=N, N)

    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))  # (BH, N, N)

    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden) FP16

    # 7) Second LayerNorm (hidden dim)
    out = out.to(torch.float32)                         # convert for LayerNorm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )                                                    # FP32

    # 8) Multiply with out‑gate – performed with a tiny Triton kernel
    out_gate_f32 = out_gate.to(torch.float32)
    out_flat = out.view(-1)
    gate_flat = out_gate_f32.view(-1)
    result_flat = torch.empty_like(out_flat)

    total = out_flat.numel()
    BLOCK = 1024
    grid = ((total + BLOCK - 1) // BLOCK, )
    _elementwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        result_flat,
        total,
        BLOCK=BLOCK,
    )

    out = result_flat.view_as(out)                     # FP32

    # 9) Final linear projection back to original dim
    out_h = out.to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out_h, to_out_w).to(torch.float32)

    return out

    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # concatenate along the output‑feature axis → (5*hidden, dim)
    W_all = torch.cat(
        [left_proj_w, right_proj_w,
         left_gate_w, right_gate_w,
         out_gate_w],
        dim=0,
    )                                              # (5*hidden, dim)

    # single matmul: (B·N·N, dim) @ (dim, 5*hidden).T  → (B·N·N, 5*hidden)
    x_flat = x_h.view(-1, dim)                     # (M, dim)   where M = B·N·N
    out_all = torch.nn.functional.linear(x_flat, W_all)  # (M, 5*hidden) FP16

    # split back into the five tensors and reshape
    split = torch.split(out_all, hidden, dim=1)
    left       = split[0].view(B, N, N, hidden)
    right      = split[1].view(B, N, N, hidden)
    left_gate  = split[2].view(B, N, N, hidden).sigmoid()
    right_gate = split[3].view(B, N, N, hidden).sigmoid()
    out_gate   = split[4].view(B, N, N, hidden).sigmoid()

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask_h
        right = right * mask_h

    # --------------------------------------------------------------
    # 4. Apply element‑wise gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Heavy O(N³) multiplicative update – batched GEMM via cuBLAS
    # --------------------------------------------------------------
    #   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #   we reshape to (B·hidden, N, N) @ (B·hidden, N, N)ᵀ
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, hidden, N, N) with (k,j)

    BH = B * hidden
    left_flat  = left_perm.view(BH, N, N)               # (BH, M=N, K=N)
    right_flat = right_perm.view(BH, N, N)              # (BH, K=N, N)

    out_flat = torch.matmul(left_flat, right_flat.transpose(-2, -1))  # (BH, N, N)
    out = out_flat.view(B, hidden, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,hidden) FP16

    # --------------------------------------------------------------
    # 6. Second LayerNorm (hidden dim)
    # --------------------------------------------------------------
    out = out.to(torch.float32)
    out = torch.nn.functional.layer_norm(
        out,
        (hidden,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )                                              # FP32

    # --------------------------------------------------------------
    # 7. Multiply with out‑gate – tiny Triton kernel (mandatory Triton use)
    # --------------------------------------------------------------
    out_gate_f32 = out_gate.to(torch.float32)

    # flatten tensors to 1‑D contiguous views
    out_flat = out.view(-1)
    gate_flat = out_gate_f32.view(-1)
    result_flat = torch.empty_like(out_flat)

    total_elems = out_flat.numel()
    BLOCK_SZ = 1024
    grid = ((total_elems + BLOCK_SZ - 1) // BLOCK_SZ,)

    _elementwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        result_flat,
        total_elems,
        BLOCK=BLOCK_SZ,
    )
    out = result_flat.view_as(out)               # FP32

    # --------------------------------------------------------------
    # 8. Final linear projection back to original dim
    # --------------------------------------------------------------
    out_h = out.to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden)
    out = torch.nn.functional.linear(out_h, to_out_w).to(torch.float32)

    return out
