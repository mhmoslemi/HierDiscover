"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) implementation

Algorithm
---------
1. **Row‑wise LayerNorm** over the last channel (`dim`).  Implemented in Triton
   (`layernorm_kernel`).  The kernel accumulates in FP32, works for FP16/FP32
   input and writes the normalized tensor in the same dtype as the input.

2. **Fused linear projections & gates** – concatenate the five weight
   matrices (`left_proj`, `right_proj`, `left_gate`, `right_gate`, `out_gate`)
   and perform a single GEMM (`F.linear`).  The result is split back into the
   five tensors and cast to FP16 for the remaining computation.

3. **Masking** (optional).  The mask is broadcasted to the projection
   dimension and multiplied before the gates.

4. **Sigmoid gates** and element‑wise gating of the left/right projections.

5. **Triangular multiplicative update** (the core `einsum`).  After reshaping
   the tensors to `[B, H, N, N]` a batched matrix‑multiply (`torch.bmm`) on
   `BH = B·H` independent `N×N` matrices computes  
   `out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]`.

6. **Second LayerNorm** over the hidden dimension (`hidden_dim`).  Same Triton
   kernel as step 1.

7. **Output gate + linear projection** back to `dim`.  The gated tensor is
   multiplied by the final weight matrix with a second `F.linear`.  The final
   result is cast back to FP32.

All heavy reductions (the two LayerNorms) are performed by a custom Triton
kernel, while the dense linear algebra (projections, batched GEMM, final
projection) leverages Tensor‑Core‑accelerated FP16 GEMMs from cuBLAS.
The output is a float‑32 tensor of shape `[B, N, N, dim]`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (FP32 accumulation, works for FP16 data)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # *const T*   [total_rows, dim]   (T = fp16 or fp32)
    y_ptr,          # *mut   T*   [total_rows, dim]
    w_ptr,          # *const float* [dim]   (γ)
    b_ptr,          # *const float* [dim]   (β)
    total_rows,     # int32
    dim: tl.constexpr,       # compile‑time
    eps: tl.constexpr,       # compile‑time
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # columns per iteration
):
    """Row‑wise LayerNorm for a 2‑D tensor."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    # -----------------------------------------------------------------
    # 1. Compute mean & variance (FP32)
    # -----------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        # address = row * dim + col
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)      # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # 1 / sqrt(var + eps)

    # -----------------------------------------------------------------
    # 2. Normalize, apply γ/β and write back
    # -----------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptrs,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        # (x - mean) * rstd
        norm = (vals - mean[:, None]) * rstd[:, None]

        # γ (weight) and β (bias) – stored as FP32
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]

        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs,
                 out,
                 mask=mask_m[:, None] & mask_d[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        input_tensor : torch.Tensor, shape [B, N, N, dim]   (float32)
        mask_tensor  : torch.Tensor, shape [B, N, N]      (float32 or bool)
        weights_dict : dict with all linear / norm parameters
        config_dict  : {
                         "dim"          : int,
                         "hidden_dim"   : int,
                         "nomask"       : bool (optional)
                       }

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on the same CUDA device)
    """
    # ------------------------------------------------------------------
    # 0. Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)   # if True → ignore mask

    # ------------------------------------------------------------------
    # 1. First LayerNorm over `dim`
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                      # x: [B, N, N, dim]
    total_rows = B * N * N                     # rows for flatten view

    x_flat = x.view(total_rows, dim)           # [total_rows, dim]
    # output buffer – keep same dtype as input (float32)
    x_norm_flat = torch.empty_like(x_flat)

    BLOCK_M = 256
    BLOCK_D = 64

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 2. Fused linear projections + gates (FP16 for Tensor‑Core speed)
    # ------------------------------------------------------------------
    # Convert to FP16 after norm
    x_norm_fp16 = x_norm_flat.half()                     # [total_rows, dim]

    # Concatenate the five weight matrices (shape: (5*hidden_dim, dim))
    w_left_proj   = weights["left_proj.weight"]
    w_right_proj  = weights["right_proj.weight"]
    w_left_gate   = weights["left_gate.weight"]
    w_right_gate  = weights["right_gate.weight"]
    w_out_gate    = weights["out_gate.weight"]

    concat_w = torch.cat(
        [w_left_proj, w_right_proj, w_left_gate,
         w_right_gate, w_out_gate],
        dim=0,
    ).half().contiguous()                     # (5*H, dim)

    # Single GEMM: [total_rows, dim] @ [dim, 5*H]^T  ->  [total_rows, 5*H]
    proj_all = F.linear(x_norm_fp16, concat_w)         # FP16

    # Split back to the five tensors
    left_proj_fp16, right_proj_fp16, left_gate_fp16, right_gate_fp16, out_gate_fp16 = \
        torch.split(proj_all, hidden_dim, dim=1)

    # ------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] → flatten → [total_rows, 1]
        mask_flat = mask_tensor.contiguous().view(total_rows, 1).to(x.dtype)
        mask_fp16 = mask_flat.half()
        left_proj_fp16 = left_proj_fp16 * mask_fp16
        right_proj_fp16 = right_proj_fp16 * mask_fp16

    # ------------------------------------------------------------------
    # 4. Apply sigmoid gates and element‑wise gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(left_gate_fp16)   # [total_rows, H]
    right_gate = torch.sigmoid(right_gate_fp16)
    out_gate = torch.sigmoid(out_gate_fp16)

    left_fp16 = left_proj_fp16 * left_gate
    right_fp16 = right_proj_fp16 * right_gate

    # Reshape to 4‑D tensors for the triangular update
    left = left_fp16.view(B, N, N, hidden_dim)           # [B, N, N, H]
    right = right_fp16.view(B, N, N, hidden_dim)         # [B, N, N, H]
    out_gate = out_gate.view(B, N, N, hidden_dim)       # [B, N, N, H]

    # ------------------------------------------------------------------
    # 5. Triangular multiplicative update (batched GEMM)
    #    out_h[i, j] = Σ_k left_h[i, k] * right_h[j, k]
    # ------------------------------------------------------------------
    # Permute to [B, H, N, N] and make left/right contiguous for bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k transposed)

    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)     # [B*H, N, N]
    right_mat = right_perm.view(BH, N, N)   # [B*H, N, N]

    # Batched GEMM in FP16 (Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat)   # [B*H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Second LayerNorm over hidden_dim (input FP16)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat = out.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    out_norm_flat = torch.empty_like(out_flat)      # FP16 output buffer

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat,
        out_norm_flat,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        hidden_dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat.view(B, N, N, hidden_dim)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7. Final gating + linear projection back to `dim`
    # ------------------------------------------------------------------
    gated = out_norm * out_gate            # FP16 element‑wise

    # Linear projection: H -> dim (FP16)
    to_out_w_fp16 = weights["to_out.weight"].half()
    gated_flat = gated.view(total_rows_h, hidden_dim)   # [total_rows_h, H]
    final_fp16 = F.linear(gated_flat, to_out_w_fp16)   # [total_rows_h, dim]

    final = final_fp16.view(B, N, N, dim).to(torch.float32)  # cast to FP32

    return final
