def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    batch_stride_a, batch_stride_b, batch_stride_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute block index
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    batch_idx = pid // (num_m * num_n)
    pid = pid % (num_m * num_n)
    block_m = pid // num_n
    block_n = pid % num_n

    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    a_ptrs = a_ptr + batch_idx * batch_stride_a + offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak
    b_ptrs = b_ptr + batch_idx * batch_stride_b + tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=..., other=0.0)
        b = tl.load(b_ptrs, mask=..., other=0.0)

        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

def batched_trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N: tl.constexpr,
    hidden_dim: tl.constexpr,
    batch_dim: tl.constexpr,   # B
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0) # tile i dimension
    pid_n = tl.program_id(1) # tile j dimension
    pid_bh = tl.program_id(2) # combined batch+hidden
    # compute actual batch and hidden indices
    B = batch_dim
    H = hidden_dim
    b = pid_bh // H
    h = pid_bh % H

    # offsets for i and j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch and hidden
    a_base = a_ptr + b * stride_batch_a + h * stride_h_a   # need stride_h_a = 1
    b_base = b_ptr + b * stride_batch_b + h * stride_h_b
    c_base = c_ptr + b * stride_batch_c + h * stride_h_c

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

"""
custom_kernel – “outgoing” TriMul (AlphaFold‑3) implementation.

Key ideas
---------
1️⃣ First LayerNorm over the channel dimension (dim) – Triton kernel.
2️⃣ Fuse all linear projections (left/right proj & gates and out‑gate) into a
   single GEMM:   y = x @ Wᵀ   where   W = [left_proj; left_gate; right_proj;
   right_gate; out_gate]  (shape (5·hidden, dim)).  This removes two extra
   reads of the large 4‑D tensor.
3️⃣ Apply the optional mask, then the sigmoid‑gates, obtaining
   tensors *left* and *right* of shape [B,N,N,hidden] in FP16.
4️⃣ Triangular multiplicative update:
        out[i,j,:] = Σₖ left[i,k,:] * right[j,k,:]
   Implemented as a custom batched GEMM (no permute/copy) where each
   (batch, hidden) pair is a separate matrix‑multiply of size N×N.
   The kernel works on the native layout (stride_i = N·H, stride_k = H,
   stride_j = H) and uses Tensor‑core friendly FP16 → FP32 accumulation.
5️⃣ Second LayerNorm over the hidden dimension – Triton kernel (input FP16,
   output FP32).
6️⃣ Apply the output‑gate (sigmoid) and a final FP16 linear projection
   back to *dim*.  The result is cast to FP32 before return.

All heavy reductions run in Triton, while the tall‑skinny GEMMs and the
final projection use PyTorch’s fast FP16 kernels.  The implementation
covers all configurations listed in the prompt (different batch sizes,
sequence lengths, dimensions and optional masks).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton row‑wise LayerNorm (generic dtype, accumulates in FP32)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const *  [rows, dim]   (FP16 or FP32)
    y_ptr,          #       *  [rows, dim]   (output dtype)
    w_ptr,          # const *  [dim]        (γ, FP32)
    b_ptr,          # const *  [dim]        (β, FP32)
    total_rows,     # int (runtime)
    dim: tl.constexpr,          # static dimension size
    eps: tl.constexpr,          # static epsilon
    BLOCK_M: tl.constexpr,      # rows per program
    BLOCK_D: tl.constexpr,      # cols per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < total_rows

    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    # --------------------------------------------------------------
    # 1) compute per‑row mean / variance
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)    # [BLOCK_D]
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)                               # [BLOCK_M, BLOCK_D]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # 2) write normalized values
    # --------------------------------------------------------------
    for d in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptrs = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptrs,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)

        out = norm * w[None, :] + b[None, :]
        out_ptrs = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptrs, out, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Batched Tri‑multiplicative update (A @ Bᵀ) over the hidden channel.
# Input layout: [B, N, N, H]   (row‑major, strides: (N*N*H, N*H, H, 1))
# ----------------------------------------------------------------------
@triton.jit
def batched_trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # FP16 tensors  [B,N,N,H]
    N, H, B,                      # runtime dimensions
    stride_am, stride_ak,         # A: row‑stride (i), col‑stride (k)
    stride_bk, stride_bn,         # B: row‑stride (k), col‑stride (j)
    stride_cm, stride_cn,         # C: row‑stride (i), col‑stride (j)
    stride_batch_a, stride_batch_b, stride_batch_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs – 3‑D grid: (tile_i, tile_j, batch×hidden)
    pid_i = tl.program_id(0)          # tile along i (output rows)
    pid_j = tl.program_id(1)          # tile along j (output cols)
    pid_bh = tl.program_id(2)         # combined batch & hidden index

    # --------------------------------------------------------------
    # derive batch index b and hidden index h
    # --------------------------------------------------------------
    hidden = H
    b = pid_bh // hidden
    h = pid_bh % hidden

    # base pointers for this (b,h) pair (hidden stride = 1)
    a_base = a_ptr + b * stride_batch_a + h
    b_base = b_ptr + b * stride_batch_b + h
    c_base = c_ptr + b * stride_batch_c + h

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # k‑dimension loop
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A[i,k]  – shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_base + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )   # FP16 → FP32 in dot

        # B[k,j] = right[j,k]  – shape (BLOCK_K, BLOCK_N)
        b_ptrs = b_base + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # dot accumulates in FP32
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # write back C[i,j]
    # --------------------------------------------------------------
    c_ptrs = c_base + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    store_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=store_mask)

# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on CUDA)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5
    nomask      = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Prepare input tensor
    # ------------------------------------------------------------------
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape                     # [B, N, N, dim]
    total_rows = B * N * N                    # rows for the flattened view

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 accumulator, input can be fp32)
    # ------------------------------------------------------------------
    x_flat = x.view(total_rows, dim)                  # [total_rows, dim]
    x_norm_flat = torch.empty_like(x_flat)            # output (float32)

    BLOCK_M = 128      # rows per program (tuneable)
    BLOCK_D = 32       # columns per inner loop (tuneable)

    grid = (triton.cdiv(total_rows, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm_flat,
        weights["norm.weight"],
        weights["norm.bias"],
        total_rows,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # Cast to FP16 for the dense linear algebra
    # ------------------------------------------------------------------
    x_norm_h = x_norm_flat.view(total_rows, dim).to(torch.float16)

    # ------------------------------------------------------------------
    # 2️⃣ Fuse all projections & gates into one GEMM
    #    Weight shapes:
    #      left_proj, left_gate, right_proj, right_gate, out_gate  → (H, dim)
    #    Concatenated weight → (5*H, dim)
    # ------------------------------------------------------------------
    proj_weight = torch.cat(
        [
            weights["left_proj.weight"],
            weights["left_gate.weight"],
            weights["right_proj.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                        # (5*H, dim)

    # single GEMM: [total_rows, dim] @ [dim, 5*H]ᵀ → [total_rows, 5*H]
    proj_out = F.linear(x_norm_h, proj_weight)   # fp16

    H = hidden_dim
    left_proj    = proj_out[:, 0:H]               # (total_rows, H)
    left_gate_pre = proj_out[:, H:2*H]
    right_proj   = proj_out[:, 2*H:3*H]
    right_gate_pre = proj_out[:, 3*H:4*H]
    out_gate_pre = proj_out[:, 4*H:5*H]

    # out‑gate needs a sigmoid
    out_gate = torch.sigmoid(out_gate_pre)        # (total_rows, H)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask_h = mask_tensor.view(total_rows, 1).to(torch.float16)
        left_proj = left_proj * mask_h
        right_proj = right_proj * mask_h

    # ------------------------------------------------------------------
    # 4️⃣ Apply element‑wise gates
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)

    left_h  = left_proj * left_gate      # (total_rows, H)
    right_h = right_proj * right_gate    # (total_rows, H)

    # ------------------------------------------------------------------
    # 5️⃣ Reshape to 4‑D tensors for the triangular update
    # ------------------------------------------------------------------
    left_h  = left_h.view(B, N, N, H)     # [B, N, N, H]  fp16
    right_h = right_h.view(B, N, N, H)    # [B, N, N, H]  fp16

    # ------------------------------------------------------------------
    # 6️⃣ Triangular multiplicative update (batched GEMM)
    #    out_h[i,j] = Σ_k left_h[i,k]·right_h[j,k]   (per hidden channel)
    # ------------------------------------------------------------------
    out_h = torch.empty_like(left_h)      # fp16 output

    # strides of the 4‑D tensors (contiguous layout)
    stride_b = left_h.stride(0)   # B‑dim stride = N*N*H
    stride_i = left_h.stride(1)   # i‑dim stride = N*H
    stride_k = left_h.stride(2)   # k‑dim stride = H
    stride_h = left_h.stride(3)   # = 1

    # tiling parameters (tuneable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    num_tile_i = triton.cdiv(N, BLOCK_M)
    num_tile_j = triton.cdiv(N, BLOCK_N)
    total_batches = B * H                       # (batch × hidden)

    grid = (num_tile_i, num_tile_j, total_batches)
    batched_trimul_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        N,               # sequence length
        H,               # hidden dimension
        B,               # batch size
        stride_i,        # a row stride (i)
        stride_k,        # a col stride (k)
        stride_k,        # b row stride (k)
        stride_i,        # b col stride (j)
        stride_i,        # c row stride (i)
        H,               # c col stride (j) == hidden_dim
        stride_b,        # batch stride for A
        stride_b,        # batch stride for B
        stride_b,        # batch stride for C
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 7️⃣ Second LayerNorm over hidden_dim (input FP16, output FP32)
    # ------------------------------------------------------------------
    total_rows_h = B * N * N
    out_flat_fp16 = out_h.view(total_rows_h, H)          # fp16
    out_norm_flat_fp32 = torch.empty(
        total_rows_h, H, dtype=torch.float32, device=out_h.device
    )                                                    # fp32

    grid2 = (triton.cdiv(total_rows_h, BLOCK_M),)
    layernorm_kernel[grid2](
        out_flat_fp16,
        out_norm_flat_fp32,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        total_rows_h,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
    )
    torch.cuda.synchronize()

    out_norm = out_norm_flat_fp32.view(B, N, N, H)       # fp32

    # ------------------------------------------------------------------
    # 8️⃣ Apply output gate and final projection
    # ------------------------------------------------------------------
    out_gate_f32 = out_gate.view(B, N, N, H).to(torch.float32)

    final_fp16 = F.linear(
