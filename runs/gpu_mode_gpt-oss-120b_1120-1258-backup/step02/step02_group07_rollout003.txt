@triton.jit
def trimul_outgoing_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # block row index for i dimension
    pid_n = tl.program_id(1)  # block col index for j dimension
    pid_g = tl.program_id(2)  # group index (b * H + h)

    # decode batch and hidden
    b = pid_g // H
    h = pid_g % H

    # Compute offsets for rows i and cols j

"""
TriMul (outgoing) – fused Triton kernel for the N³ contraction.

The heavy contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched GEMM in a custom Triton kernel.
All other operations (LayerNorm, linear projections, gating) are performed
under torch.cuda.amp.autocast() in FP16 (with FP32 accumulation) and the
kernel processes each hidden channel independently, avoiding the costly
permutes and views present in the reference PyTorch implementation.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_gemm_kernel(
    left_ptr,          # const fp16 *
    right_ptr,         # const fp16 *
    out_ptr,           # fp16 *
    B, N, H,           # sizes (int64)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for each (batch, hidden) pair.

    left  shape: (B, N, N, H) indexed as (b, i, k, h)
    right shape: (B, N, N, H) indexed as (b, j, k, h)
    out   shape: (B, N, N, H) indexed as (b, i, j, h)
    """
    pid_m = tl.program_id(0)   # block row (i)
    pid_n = tl.program_id(1)   # block col (j)
    pid_g = tl.program_id(2)   # group = b * H + h

    # Decode batch (b) and hidden channel (h)
    b = pid_g // H
    h = pid_g % H

    # Offsets within the tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Compute base pointers (without the K dimension)
    # left: (b, i, k, h)
    base_left = (
        left_ptr
        + b * stride_left_b
        + h * stride_left_h
        + offs_i[:, None] * stride_left_i
    )
    # right: (b, j, k, h) – we will address it as (k, j) later
    base_right = (
        right_ptr
        + b * stride_right_b
        + h * stride_right_h
        + offs_j[None, :] * stride_right_j
    )
    # output: (b, i, j, h)
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + h * stride_out_h
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
    )

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + offs_k                     # absolute k indices
        mask_k = cur_k < N

        # Load a tile of left  (M x K)
        left_tile = tl.load(
            base_left + cur_k[None, :] * stride_left_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load a tile of right (K x N) – note the order (k, j)
        right_tile = tl.load(
            base_right + cur_k[:, None] * stride_right_k,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Matrix‑multiply‑accumulate (FP16 inputs, FP32 accumulation)
        acc += tl.dot(left_tile, right_tile)

    # Store the result (convert back to FP16)
    out_tile = acc.to(tl.float16)
    tl.store(out_ptrs, out_tile, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of model parameters (float32) with keys
                         ["norm.weight", "norm.bias",
                          "left_proj.weight", "right_proj.weight",
                          "left_gate.weight", "right_gate.weight",
                          "out_gate.weight",
                          "to_out_norm.weight", "to_out_norm.bias",
                          "to_out.weight"]
        - config       : dict containing at least
                         {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor
        Output tensor [B, N, N, dim] (float32)
    """
    torch.set_grad_enabled(False)  # no autograd needed

    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load weights (already on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # Mixed‑precision forward pass up to the heavy N³ contraction
    # -----------------------------------------------------------------
    # LayerNorm on the input channel dimension
    with torch.cuda.amp.autocast():
        x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

        # Linear projections (no bias)
        left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
        right = F.linear(x, right_proj_weight)

        # Optional mask (broadcast over hidden dimension)
        if not nomask and mask is not None:
            mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
            left = left * mask_f
            right = right * mask_f

        # Gating scalars (sigmoid)
        left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
        right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
        out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

        # Apply gates
        left = left * left_gate
        right = right * right_gate

        # -------------------------------------------------------------
        # Batched GEMM via Triton: out = left @ rightᵀ   (per hidden channel)
        # -------------------------------------------------------------
        B, N, _, H = left.shape  # H == hidden_dim

        # Allocate output buffer (FP16)
        out = torch.empty_like(left, dtype=torch.float16, device=device)

        # Strides (int64) – needed by the Triton kernel
        stride_left = left.stride()
        stride_right = right.stride()
        stride_out = out.stride()

        # GEMM tile sizes – tuned for H100 tensor‑cores
        BLOCK_M = 64
        BLOCK_N = 64
        BLOCK_K = 16

        # Grid: (tiles_i, tiles_j, batch*hidden)
        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B * H,
        )

        trimul_outgoing_gemm_kernel[grid](
            left,
            right,
            out,
            B,
            N,
            H,
            stride_left[0],
            stride_left[1],
            stride_left[2],
            stride_left[3],
            stride_right[0],
            stride_right[1],
            stride_right[2],
            stride_right[3],
            stride_out[0],
            stride_out[1],
            stride_out[2],
            stride_out[3],
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            num_warps=4,
        )

        # -------------------------------------------------------------
        # Post‑processing (LayerNorm, final gating, output projection)
        # -------------------------------------------------------------
        # LayerNorm over hidden dimension
        out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

        # Apply the final out‑gate
        out = out * out_gate

        # Linear projection back to the original channel dimension
        out = F.linear(out, to_out_weight)

    # The API expects float32 output
    out = out.to(torch.float32)
    return out
