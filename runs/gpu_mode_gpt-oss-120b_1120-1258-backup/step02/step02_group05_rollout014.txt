Pseudo:

@triton.jit
def batched_tri_mul_kernel(
    A, B, C,
    M, N,
    stride_am, stride_ai, stride_ak,  # A strides: batch*hidden dimension (M total) stride, i stride, k stride
    stride_bm, stride_bk, stride_bj,
    stride_cm, stride_ci, stride_cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):

   pid0 = tl.program_id(0)  # batch index (M_total)
   pid1 = tl.program_id(1)  # row tile
   pid2 = tl.program_id(2)  # col tile

   # Compute offsets for row and col
   row_offset = pid1 * BLOCK_M
   col_offset = pid2 * BLOCK_N

   # Compute number of k tiles
   num_k = tl.cdiv(N, BLOCK_K)

   # Initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # For each k tile
   for k_tile in range(num_k):
       # Load A tile
       a_ptr = A + pid0 * stride_am + (row_offset + tl.arange(0,BLOCK_M))[:,None] * stride_ai + (k_tile*BLOCK_K + tl.arange(0,BLOCK_K))[None,:] * stride_ak
       a = tl.load(a_ptr, mask=(row_offset + tl.arange(0,BLOCK_M) < N)[:,None] & (k_tile*BLOCK_K + tl.arange(0,BLOCK_K) < N)[None,:], other=0.0)

@triton.jit
def batched_tri_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    N, hidden_dim,
    stride_am, stride_ai, stride_ak, stride_ah,
    stride_bm, stride_bk, stride_bj, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid0 = tl.program_id(0) # (batch*hidden)
    pid1 = tl.program_id(1) # row tile i
    pid2 = tl.program_id(2) # col tile j

    # compute batch and hidden index
    batch = pid0 // hidden_dim
    hid = pid0 % hidden_dim

    # compute offsets
    i0 = pid1 * BLOCK_M
    j0 = pid2 * BLOCK_N

    # ranges
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # compute base offsets
    a_base = A_ptr + batch * stride_am + hid * stride_ah
    b_base = B_ptr + batch * stride_bm + hid * stride_bh
    c_base = C_ptr + batch * stride_cm + hid * stride_ch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop K
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[i,k] tile
        a_ptr = a_base + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr, mask=mask_i[:, None] & mask_k[None,:], other=0.0).to(tl.float16)

        # B[k,j] tile (transposed)
        b_ptr = b_base + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_j[None,:], other=0.0).to(tl.float16)

        acc += tl.dot(a, b)

    # store
    c_ptr = c_base + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptr, acc, mask=mask_i[:, None] & mask_j[None,:])

"""
TriMul (outgoing) forward pass – Triton‑accelerated cubic contraction.

Algorithm
---------
1. Layer‑norm the input on the feature dimension (float32).
2. Cast to FP16 and compute the two projection + gate linear layers in a
   *single* matmul each (proj || gate concatenated) → half‑precision.
3. Fuse elementwise gate (sigmoid) and produce gated tensors `left` and `right`.
   (If a mask is supplied we *do not* materialise a masked copy – the mask is
   applied inside the Triton kernel.)
4. Batched GEMM:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a GEMM for every (batch,head) pair.  The kernel works directly on
   the original `[B,N,N,H]` layout, avoiding an expensive permute+copy.
   For the masked case we load mask[i,k] / mask[j,k] and multiply on‑the‑fly.
5. Layer‑norm on the hidden dimension, multiply by the output‑gate and a final
   bias‑free linear projection back to the original `dim`.

All heavy work (step 4) is done in Triton, the remaining ops use
torch.nn.functional for simplicity.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernels
# --------------------------------------------------------------------------- #

@triton.jit
def batched_tri_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    N, H,                     # sequence length, hidden dimension
    stride_am, stride_ai, stride_ak, stride_ah,
    stride_bm, stride_bj, stride_bk, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    C = A @ Bᵀ  for every (batch, head) pair.
    A, B : [B, N, N, H]  (fp16)
    C    : [B, N, N, H]  (fp32)
    """
    pid0 = tl.program_id(0)   # batch * hidden
    pid_m = tl.program_id(1)  # tile row i
    pid_n = tl.program_id(2)  # tile col j

    # decode batch and head from pid0
    batch = pid0 // H
    head  = pid0 %  H

    # offsets for this (batch,head)
    off_A = batch * stride_am + head * stride_ah
    off_B = batch * stride_bm + head * stride_bh
    off_C = batch * stride_cm + head * stride_ch

    # tile coordinates
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A[batch,head,i,k]
        a_ptr = A_ptr + off_A + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Bᵀ → B[batch,head,k,j]
        b_ptr = B_ptr + off_B + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16)

        acc += tl.dot(a, b)

    # write back
    c_ptr = C_ptr + off_C + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

@triton.jit
def batched_tri_gemm_masked_kernel(
    A_ptr, B_ptr, C_ptr, M_ptr,
    N, H,
    stride_am, stride_ai, stride_ak, stride_ah,
    stride_bm, stride_bj, stride_bk, stride_bh,
    stride_cm, stride_ci, stride_cj, stride_ch,
    stride_mm, stride_mi, stride_mk,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Same as `batched_tri_gemm_kernel` but applies a binary mask
    M ∈ {0,1} broadcasted over the hidden dimension.
    """
    pid0 = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    batch = pid0 // H
    head  = pid0 %  H

    off_A = batch * stride_am + head * stride_ah
    off_B = batch * stride_bm + head * stride_bh
    off_C = batch * stride_cm + head * stride_ch
    off_M = batch * stride_mm                     # mask has no head dim

    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k0 in range(0, N, BLOCK_K):
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # A
        a_ptr = A_ptr + off_A + i[:, None] * stride_ai + k[None, :] * stride_ak
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # Bᵀ
        b_ptr = B_ptr + off_B + k[:, None] * stride_bk + j[None, :] * stride_bj
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0,
                    dtype=tl.float16)

        # mask[i,k]  (shape BLOCK_M x BLOCK_K)
        m_i_ptr = M_ptr + off_M + i[:, None] * stride_mi + k[None, :] * stride_mk
        m_i = tl.load(m_i_ptr,
                      mask=mask_i[:, None] & mask_k[None, :],
                      other=0.0,
                      dtype=tl.float16)

        # mask[j,k] → needed as (BLOCK_K x BLOCK_N)
        m_j_ptr = M_ptr + off_M + k[:, None] * stride_mk + j[None, :] * stride_mi
        m_j = tl.load(m_j_ptr,
                      mask=mask_k[:, None] & mask_j[None, :],
                      other=0.0,
                      dtype=tl.float16)

        a = a * m_i          # apply mask on the left side
        b = b * m_j          # apply mask on the right side

        acc += tl.dot(a, b)

    c_ptr = C_ptr + off_C + i[:, None] * stride_ci + j[None, :] * stride_cj
    tl.store(c_ptr,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])

# --------------------------------------------------------------------------- #
# Entry point
# --------------------------------------------------------------------------- #

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : [B, N, N, dim]  (float32)
        mask         : [B, N, N]       (any dtype) – may be omitted when
                                          config["nomask"] is True
        weights      : dict of model parameters (torch.Tensor)
        config       : dict with keys "dim", "hidden_dim", "nomask" (bool)
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", True)
    eps        = 1e-5

    # ------------------------------------------------------------------- #
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------- #
    # 2. Cast to fp16 and perform fused projection + gate matmuls
    # ------------------------------------------------------------------- #
    x_h = x.half()   # [B, N, N, dim] (fp16)

    # concatenate proj & gate weights to halve the number of matmuls
    left_combined_w = torch.cat(
        [weights["left_proj.weight"], weights["left_gate.weight"]], dim=0
    ).half()  # [(2*H), dim]
    right_combined_w = torch.cat(
        [weights["right_proj.weight"], weights["right_gate.weight"]], dim=0
    ).half()  # [(2*H), dim]

    # (B,N,N,2*H)
    left_combined = F.linear(x_h, left_combined_w)   # fp16
    right_combined = F.linear(x_h, right_combined_w) # fp16

    # split proj / gate
    left_proj = left_combined[..., :hidden_dim]
    left_gate = torch.sigmoid(left_combined[..., hidden_dim:])   # fp16

    right_proj = right_combined[..., :hidden_dim]
    right_gate = torch.sigmoid(right_combined[..., hidden_dim:]) # fp16

    # gated tensors
    left  = left_proj * left_gate   # [B,N,N,H] fp16
    right = right_proj * right_gate # [B,N,N,H] fp16

    # ------------------------------------------------------------------- #
    # 3. Optional mask – we *do not* materialise a masked copy.
    #    The mask will be applied inside the Triton kernel.
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask is broadcast over the hidden dimension, convert to fp16 once
        mask_fp16 = mask.to(left.dtype)      # [B,N,N] fp16
    else:
        mask_fp16 = None  # placeholder

    # ------------------------------------------------------------------- #
    # 4. Batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Operates on the original [B,N,N,H] layout – no permute/contiguous copy.
    # ------------------------------------------------------------------- #
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # strides for A / left
    stride_am, stride_ai, stride_ak, stride_ah = left.stride()
    # strides for B / right
    stride_bm, stride_bj, stride_bk, stride_bh = right.stride()
    # strides for C / out
    stride_cm, stride_ci, stride_cj, stride_ch = out.stride()

    # block sizes (tuned for H100 Tensor‑cores)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # program_id(0)
        (N + BLOCK_M - 1) // BLOCK_M,       # program_id(1) – row tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # program_id(2) – col tiles
    )

    if nomask:
        batched_tri_gemm_kernel[grid](
            left,
            right,
            out,
            N,
            hidden_dim,
            stride_am, stride_ai, stride_ak, stride_ah,
            stride_bm, stride_bj, stride_bk, stride_bh,
            stride_cm, stride_ci, stride_cj, stride_ch,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )
    else:
        # mask strides
        stride_mm, stride_mi, stride_mk = mask_fp16.stride()
        batched_tri_gemm_masked_kernel[grid](
            left,
            right,
            out,
            mask_fp16,
            N,
            hidden_dim,
            stride_am, stride_ai, stride_ak, stride_ah,
            stride_bm, stride_bj, stride_bk, stride_bh,
            stride_cm, stride_ci, stride_cj, stride_ch,
            stride_mm, stride_mi, stride_mk,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
        )

    # ------------------------------------------------------------------- #
    # 5. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H] float32

    # output gate (computed from the original x)
    out_gate = torch.sigmoid(
        F.linear(x_h, weights["out_gate.weight"].half())
    )  # [B,N,N,H] fp16

    out = out * out_gate.float()

    # final linear back to `dim` (half precision for speed)
    out = F.linear(out.half(), weights["to_out.weight"].half()).float()

    return out
