"""
TriMul (outgoing) – Triton + torch fused implementation
------------------------------------------------------
Algorithm
~~~~~~~~~
1️⃣ Row‑wise LayerNorm over the last dimension (float32 → float16) using a
   tiny Triton kernel (no synchronisation across the hidden axis).

2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single GEMM:  X_norm @ W_fusedᵀ → [R, 5·H]   (R = B·N·N).

3️⃣ Split the fused tensor, apply the optional pair‑wise mask,
   evaluate sigmoid gates and reshape to   [B, H, N, N].

4️⃣ Batched half‑precision GEMM (cuBLAS) over the hidden axis:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ   (half‑precision).

5️⃣ Compute per‑position mean / variance of ``out`` (float32 reduction).

6️⃣ Second LayerNorm, element‑wise out‑gate and final Linear
   (hidden → model dimension) are performed with dense torch kernels:
        out_norm = ((out−μ)·rstd)·γ + β
        out_norm = out_norm * out_gate
        output   = out_norm @ W_outᵀ                (float16 → float32)

Only the first LayerNorm is a custom Triton kernel – all heavy matrix
multiplications are delegated to highly‑optimised PyTorch/cuBLAS kernels,
which gives a fast end‑to‑end implementation on H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm kernel (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,         # const  *[rows, dim]  (float32)
    y_ptr,         # mutable* [rows, dim] (float16)
    w_ptr,         # const  *[dim]       (float32) γ
    b_ptr,         # const  *[dim]       (float32) β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalized values: γ·(x‑μ)·rstd + β
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]            # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)      # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)      # β (fp32)

        out = norm * w[None, :] + b[None, :]                     # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict  : mapping with all Linear/LayerNorm parameters
                              (float32 tensors, bias=False for Linear)
            - config_dict   : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                           # total (b,i,k) positions
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16) using Triton
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],            # γ (float32)
        w["norm.bias"],              # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat  # free memory

    # --------------------------------------------------------------
    # 2️⃣ Fuse five linear maps into a single GEMM (FP16)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                 # [5·H, dim]

    # Linear: (R, dim) @ (5·H, dim).T -> (R, 5·H)
    proj = F.linear(x_norm, fused_w)                    # fp16
    del fused_w, x_norm

    # --------------------------------------------------------------
    # 3️⃣ Split, apply mask, sigmoid gates and reshape to [B, H, N, N]
    # --------------------------------------------------------------
    proj = proj.view(R, 5, H)                           # [R, 5, H]

    left_raw  = proj[:, 0, :]   # (R, H)
    right_raw = proj[:, 1, :]   # (R, H)
    left_gate_raw  = proj[:, 2, :]
    right_gate_raw = proj[:, 3, :]
    out_gate_raw   = proj[:, 4, :]

    # Optional pair‑wise mask (broadcast over hidden dim)
    if cfg.get("nomask", True) or mask is None:
        mask_flat = None
    else:
        # mask shape [B, N, N] → [R] (float16 0/1)
        mask_flat = mask.to(torch.float16).reshape(R, 1)

    if mask_flat is not None:
        left_raw  = left_raw  * mask_flat
        right_raw = right_raw * mask_flat

    # Sigmoid gates (computed in FP16, sufficient for the model)
    left_gate  = torch.sigmoid(left_gate_raw)
    right_gate = torch.sigmoid(right_gate_raw)
    out_gate   = torch.sigmoid(out_gate_raw)

    # Apply gates
    left  = left_raw  * left_gate
    right = right_raw * right_gate

    # Reshape to (B, H, N, N)
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    out_gate = out_gate.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    del proj, left_raw, right_raw, left_gate_raw, right_gate_raw, out_gate_raw

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM over the hidden dimension (cuBLAS, FP16)
    # --------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)          # [B·H, N, N]
    right_mat = right.view(B * H, N, N)         # [B·H, N, N]

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16
    out = out.view(B, H, N, N)                             # (B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Per‑position mean & rsqrt‑variance (float32)
    # --------------------------------------------------------------
    out_fp32 = out.float()
    mu   = out_fp32.mean(dim=1)                     # (B, N, N)
    var  = out_fp32.var(dim=1, unbiased=False)     # (B, N, N)
    rstd = torch.rsqrt(var + eps)                  # (B, N, N)

    # --------------------------------------------------------------
    # 6️⃣ Second LayerNorm, out_gate, and final Linear (FP16 → FP32)
    # --------------------------------------------------------------
    # Normalise over hidden dimension
    mu   = mu.unsqueeze(1)          # (B, 1, N, N)
    rstd = rstd.unsqueeze(1)        # (B, 1, N, N)

    # (out - μ) * rstd
    norm = (out_fp32 - mu) * rstd                     # (B, H, N, N) fp32

    # Apply to_out_norm γ/β (per‑hidden feature)
    gamma = w["to_out_norm.weight"].view(1, H, 1, 1)   # (1, H, 1, 1) fp32
    beta  = w["to_out_norm.bias" ].view(1, H, 1, 1)   # (1, H, 1, 1) fp32
    norm = norm * gamma + beta                       # (B, H, N, N) fp32

    # Multiply with out_gate (cast gate to fp32)
    norm = norm * out_gate.float()                   # (B, H, N, N) fp32

    # Final projection: hidden -> model dim
    # flatten to (R, H), cast to FP16 for the GEMM with W_out (dim, H)
    hidden_flat = norm.view(R, H).half()             # (R, H) fp16
    w_out = w["to_out.weight"].to(torch.float16)    # (dim, H)

    # Linear (R, H) @ (dim, H).T → (R, dim)  (still fp16)
    out_final_fp16 = F.linear(hidden_flat, w_out)    # (R, dim) fp16

    # Cast back to FP32 and reshape to (B, N, N, dim)
    out_final = out_final_fp16.float().view(B, N, N, dim)

    return out_final
