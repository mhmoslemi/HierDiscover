"""
TriMul (outgoing) – highly‑optimized Triton implementation

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the flattened input (B·N·N × dim) – fp32 → fp16.
2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:  X_norm @ W_fusedᵀ  →  (R, 5·H)   (R = B·N·N, H = hidden_dim).
3️⃣  Split the fused tensor, apply the optional pair‑wise mask, evaluate the
    three sigmoid gates and store the three tensors directly in the layout
    (B, H, N, N) required for the batched GEMM.
4️⃣  Batched half‑precision GEMM (cuBLAS Tensor‑Core):
        out[b, h] = left[b, h] @ right[b, h]ᵀ   → (B·H, N, N)  fp16.
5️⃣  Fuse the *second* LayerNorm, the out‑gate, and the final linear projection
    into a single Triton kernel.  The kernel first computes the per‑position
    mean / rsqrt‑variance of `out`, then normalises, applies the out‑gate,
    performs the hidden‑dim → dim projection (H × dim weight) and writes the
    result as a contiguous [B, N, N, dim] float‑32 tensor.

Only the three kernels above are launched – all heavy matrix multiplications
are delegated to cuBLAS, which on H100 delivers peak TFLOP performance.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # const   *[rows, dim]  (float32)
    y_ptr,               # mutable *[rows, dim]  (float16)
    w_ptr,               # const   *[dim]       (float32)   γ
    b_ptr,               # const   *[dim]       (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # Compute mean and variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_d = tl.cdiv(dim, BLOCK_D)
    for d_idx in range(0, num_d):
        d_off = d_idx * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = d_off < dim
        ptr = x_ptr + offs_m[:, None] * dim + d_off[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # Write normalised values:  γ·(x‑μ)·rstd + β  (convert to fp16)
    # ------------------------------------------------------------------
    for d_idx in range(0, num_d):
        d_off = d_idx * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = d_off < dim
        ptr = x_ptr + offs_m[:, None] * dim + d_off[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]

        w = tl.load(w_ptr + d_off, mask=mask_d, other=0.0)   # γ
        b = tl.load(b_ptr + d_off, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]
        tl.store(y_ptr + offs_m[:, None] * dim + d_off[None, :],
                 out.to(tl.float16),
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply optional mask & sigmoid gates.
#    Output layout: [B, H, N, N] (fp16) for left/right/out_gate.
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]        (fp16) – ignored if HAS_MASK=0
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)            # [BLOCK_R]
    mask_r = offs_r < R

    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # optional mask (broadcasted over H)
    if HAS_MASK:
        m_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m_val = m_val[:, None]                                      # [BLOCK_R,1]
    else:
        m_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    hidden_stride = N * N                     # distance between consecutive h
    base_off = b * (H * N * N) + i * N + k    # offset of the first hidden slice

    # iterate over hidden dimension
    num_h = tl.cdiv(H, BLOCK_H)
    for h_idx in range(0, num_h):
        h_off = h_idx * BLOCK_H + tl.arange(0, BLOCK_H)       # [BLOCK_H]
        mask_h = h_off < H

        # pointer into the fused projection tensor
        proj_base = offs_r[:, None] * (5 * H) + h_off[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw      = tl.load(proj_ptr + proj_base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw     = tl.load(proj_ptr + proj_base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw = tl.load(proj_ptr + proj_base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw= tl.load(proj_ptr + proj_base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw  = tl.load(proj_ptr + proj_base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid in fp32 for stability, then back to fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left  = left_raw  * m_val * left_gate
        right = right_raw * m_val * right_gate
        outg  = out_gate   * m_val

        dst_off = base_off[:, None] + h_off[None, :] * hidden_stride

        tl.store(left_ptr    + dst_off,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr   + dst_off,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr+ dst_off,
                 outg,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Fuse second LayerNorm, out_gate and final linear projection.
#    The kernel also computes the per‑position mean / rsqrt‑variance of `out`.
# ----------------------------------------------------------------------
@triton.jit
def final_fused_kernel(
    out_ptr,            # const *[B, H, N, N] (fp16)
    gate_ptr,           # const *[B, H, N, N] (fp16)   (out_gate)
    ln_w_ptr,           # const *[H]            (float32)   γ of to_out_norm
    ln_b_ptr,           # const *[H]            (float32)   β of to_out_norm
    w_out_T_ptr,        # const *[H, dim]       (fp16)      weight transposed
    out_final_ptr,      # mutable *[B, N, N, dim] (float32)
    total_rows: tl.int32,   # = B * N * N
    H: tl.int32,
    N: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_N: tl.constexpr,
):
    pid_m = tl.program_id(0)          # rows‑tile
    pid_n = tl.program_id(1)          # output‑dim‑tile

    row_start = pid_m * BLOCK_M
    col_start = pid_n * BLOCK_N

    offs_m = row_start + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = col_start + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_m = offs_m < total_rows
    mask_n = offs_n < dim

    # decode (b,i,j) from flat row index

    rem = offs_m - b * nn
    i = rem // N
    j = rem - i * N

    base_off = b * (H * N * N) + i * N + j          # offset of hidden‑slice h = 0

    # ------------------------------------------------------------------
    # 1) First pass: mean & rsqrt‑variance of `out` over the hidden dim.
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    num_k = tl.cdiv(H, BLOCK_K)
    for kt in range(0, num_k):
        k_off = kt * BLOCK_K + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = k_off < H

        ptr = out_ptr + base_off[:, None] + k_off[None, :] * hidden_stride
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)          # [M, K]

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mu   = sum_val / H
    var  = sum_sq / H - mu * mu
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2) Second pass: LayerNorm + out_gate + projection to `dim`.
    # ------------------------------------------------------------------
    acc = tl.zeros([BLOCK_M, BLOCK_N], dtype=tl.float32)

    for kt in range(0, num_k):
        k_off = kt * BLOCK_K + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = k_off < H

        # load raw `out` and `gate`
        out_vals = tl.load(out_ptr + base_off[:, None] + k_off[None, :] * hidden_stride,
                           mask=mask_m[:, None] & mask_k[None, :],
                           other=0.0).to(tl.float32)          # [M, K]
        gate_vals = tl.load(gate_ptr + base_off[:, None] + k_off[None, :] * hidden_stride,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)          # [M, K]

        # LayerNorm parameters (γ,β) for this hidden slice
        gamma = tl.load(ln_w_ptr + k_off, mask=mask_k, other=0.0)   # [K] fp32
        beta  = tl.load(ln_b_ptr + k_off, mask=mask_k, other=0.0)   # [K] fp32

        # (out - μ) * rstd  →  normalise
        diff = (out_vals - mu[:, None]) * rstd[:, None]           # [M, K]

        # apply second LayerNorm (γ·x + β)
        norm = diff * gamma[None, :] + beta[None, :]               # [M, K]

        # multiply by out_gate
        term = norm * gate_vals                                   # [M, K]
        term_fp16 = term.to(tl.float16)

        # load weight tile (K, dim_tile)
        w_ptr = w_out_T_ptr + k_off[:, None] * dim + offs_n[None, :]
        w_tile = tl.load(w_ptr,
                         mask=mask_k[:, None] & mask_n[None, :],
                         other=0.0)                                 # [K, N]

        # accumulate mat‑mul (M×K)·(K×N) → (M×N)
        acc += tl.dot(term_fp16, w_tile).to(tl.float32)

    # ------------------------------------------------------------------
    # Write final result  → [B, N, N, dim] (float32)
    # ------------------------------------------------------------------
    dst_ptr = out_final_ptr + offs_m[:, None] * dim + offs_n[None, :]
    tl.store(dst_ptr, acc,
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : mapping with all linear / layer‑norm parameters
                         (all stored as float32, Linear layers have no bias)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total number of (b,i,j) positions
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim]  fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128
    BLOCK_D = 64
    grid_ln = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_ln](
        x_flat,
        x_norm,
        w["norm.weight"],      # γ  (float32)
        w["norm.bias"],        # β  (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=8,
    )
    del x_flat

    # ------------------------------------------------------------------
    # 2️⃣ Fuse the five linear maps into a single GEMM
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # (5·H, dim)

    proj = F.linear(x_norm, fused_w)        # (R, 5·H) fp16
    del fused_w, x_norm

    # ------------------------------------------------------------------
    # 3️⃣ Split, (optional) mask & sigmoid gates → layout (B,H,N,N)
    # ------------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # flatten mask to [R] and cast to fp16
        mask_flat = mask.to(torch.float16).reshape(R).contiguous()
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 256
    BLOCK_H = 64
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=8,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Batched half‑precision GEMM via cuBLAS (Tensor‑Core)
    # ------------------------------------------------------------------
    left_mat  = left.view(B * H, N, N)            # (B·H, N, N) fp16
    right_mat = right.view(B * H, N, N)           # (B·H, N, N) fp16

    out = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N) fp16
    out = out.view(B, H, N, N)                              # (B, H, N, N) fp16
    # `out` will be consumed by the final fused kernel
    del left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Fuse second LayerNorm, out_gate & projection to `dim`
    # ------------------------------------------------------------------
    # Transpose final linear weight to (H, dim) and cast to fp16
    w_out_T = w["to_out.weight"].to(torch.float16).t().contiguous()   # (H, dim)

    final = torch.empty(B, N, N, dim, dtype=torch.float32, device=device)

    BLOCK_M_OUT = 64                     # rows per program
    BLOCK_K_OUT = 32                     # hidden‑dim chunk
    BLOCK_N_OUT = 64                     # output‑dim chunk
    grid_out = (
        triton.cdiv(R, BLOCK_M_OUT),
        triton.cdiv(dim, BLOCK_N_OUT),
    )
    final_fused_kernel[grid_out](
        out,
        out_gate,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        w_out_T,
        final,
        R,
        H,
        N,
        dim,
        eps,
        BLOCK_M=BLOCK_M_OUT,
        BLOCK_K=BLOCK_K_OUT,
        BLOCK_N=BLOCK_N_OUT,
        num_warps=8,
    )

    # ------------------------------------------------------------------
    # Return the float‑32 result
    # ------------------------------------------------------------------
    return final
