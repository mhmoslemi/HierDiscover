"""
Tri‑Multiplicative (outgoing) update – Triton‑accelerated forward pass.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) → fp16.
2️⃣ One fused linear projection that creates the five tensors needed for the
   update (left, right, left‑gate, right‑gate, out‑gate).  This uses a regular
   cuBLAS GEMM (torch.nn.functional.linear) on the fp16 normalised data.
3️⃣ A single Triton kernel splits the fused projection, applies the optional
   pair‑wise mask, and applies the sigmoid gates, producing three fp16 tensors:
   `left`, `right` and `out_gate`.
4️⃣ Batched GEMM over the hidden dimension:
   (B·H, N, N) @ (B·H, N, N)ᵀ   →   (B·H, N, N)
   which implements the `einsum('... i k d, ... j k d -> ... i j d')`.
   torch.bmm (cuBLAS) is used because it already exploits Tensor‑cores.
5️⃣ A fused Triton kernel performs a second row‑wise LayerNorm on the hidden‑
   dimension tensor and multiplies the result by the `out_gate` tensor.
6️⃣ Final linear projection from hidden_dim → dim (fp16 GEMM) and cast to
   float32 for the output.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)      # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # mean & variance (fp32 accumulator)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # write normalised values (γ·(x‑μ)·rstd + β) as fp16
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]      # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        out = norm * w[None, :] + b[None, :]               # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16) – ignored if HAS_MASK=0
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # optional unary mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        m = m[:, None]                                          # broadcast to [BLOCK_R,1]
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # loop over hidden dimension in tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Layout in the fused projection:
        #   [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_raw_ptr= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # sigmoid in fp32 for numerical stability
        lg = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        rg = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        og = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # apply mask & gates (mask only on left/right)
        left  = left_raw  * m * lg
        right = right_raw * m * rg

        # store results
        left_dst_ptr    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr= out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,    left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr,   right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr,og,
                 mask=mask_r[:, None] & mask_h[None, :])

# 5️⃣ Row‑wise LayerNorm + out‑gate multiplication (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (float16)
    gate_ptr,       # const   *[rows, dim]   (float16)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # --------------------------------------------------------------
    # mean & variance (fp32)
    # --------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # --------------------------------------------------------------
    # normalized * weight + bias, then multiply by gate (fp16)
    # --------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim

        # load data and gate
        x_vals = tl.load(x_ptr   + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr+ offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        # normalize
        norm = (x_vals - mean[:, None]) * rstd[:, None]   # fp32

        # layer‑norm parameters
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β fp32

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals
        y_fp16 = y_fp32.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of all Linear/LayerNorm parameters (float32)
                         Keys must match the names used in the reference module:
                         "norm.weight", "norm.bias",
                         "left_proj.weight", "right_proj.weight",
                         "left_gate.weight", "right_gate.weight", "out_gate.weight",
                         "to_out_norm.weight", "to_out_norm.bias",
                         "to_out.weight"
        - config_dict  : must contain
                         "dim" (int), "hidden_dim" (int), "nomask" (bool, optional)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device
    B, N, _, _ = x.shape
    R = B * N * N                # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # ------------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()          # [R, dim] fp32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 32
    grid_norm = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid_norm](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ (float32)
        weights["norm.bias"],     # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection producing 5*H channels (left, right, gates)
    # ------------------------------------------------------------------
    # Concatenate the five weight matrices (each [H, dim]) → [5*H, dim]
    fused_w = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5*H, dim]

    # Linear (no bias) :  x_norm [R, dim] @ fused_w.T [dim, 5*H] → [R, 5*H] (fp16)
    proj = F.linear(x_norm, fused_w)                   # fp16
    del fused_w, x_norm                                 # free memory early

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # Determine if mask should be applied

        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        HAS_MASK = 1
        # Convert to float16; 1.0 => keep, 0.0 => drop
        mask_ptr = mask.to(torch.float16).reshape(R)

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=HAS_MASK,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # ------------------------------------------------------------------
    # reshape for batch GEMM:  [B, N, N, H] → [B, H, N, N]
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)     # [B·H, N, N]
    right_mat = right.view(B * H, N, N)    # [B·H, N, N]

    # (B·H, N, N) @ (B·H, N, N)ᵀ → (B·H, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused)
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()                     # [R, H]

    grid_norm2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid_norm2](

        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (float32)
        weights["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 GEMM) and cast to fp32
    # ------------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
