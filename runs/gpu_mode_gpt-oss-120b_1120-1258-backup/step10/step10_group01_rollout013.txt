@triton.jit
def split_fuse_gate_write_kernel(
    proj_ptr, left_ptr, right_ptr, out_gate_ptr,
    mask_ptr, # maybe dummy
    R, H, N,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < R

    # compute b,i,k per row
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # reshape into broadcast shape for hidden
    # compute base offsets for left/right/out_gate write
    # compute mask per row
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)
        mask_val = mask_val[:, None]  # broadcast
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # compute pointers for sections in proj: offset = offs_r * (5*H) + offs_h
        # left_raw: base + offs_h
        # right_raw: base + H + offs_h
        # left_gate: base + 2*H + offs_h
        # right_gate: base + 3*H + offs_h
        # out_gate_raw: base + 4*H + offs_h

        base_proj = offs_r[:, None] * (5 * H) + offs_h[None, :]
        left_raw = tl.load(proj_ptr + base_proj,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)   # fp16
        right_raw = tl.load(proj_ptr + base_proj + H,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0)
        left_gate_raw = tl.load(proj_ptr + base_proj + 2*H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate_raw = tl.load(proj_ptr + base_proj + 3*H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        out_gate_raw = tl.load(proj_ptr + base_proj + 4*H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # compute sigmoid
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        left = left_raw * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # compute write offsets for left/right/out_gate
        # idx_base = b*H*N*N + i*N + k  ; shape [BLOCK_R]
        # left_ptr_ptr = left_ptr + (b*H*N*N) + offs_h*N*N + i*N + k
        # So compute base_offset = b * (H*N*N) + i*N + k   # shape [BLOCK_R]
        # Then offset = base_offset[:, None] + offs_h[None,:] * (N*N)

        base_offset = b * (H * N * N) + i * N + k  # [BLOCK_R]
        # hidden stride
        hidden_stride = N * N  # per increase of hidden idx
        out_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        # store
        tl.store(left_ptr + out_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr + out_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr + out_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])

"""
TriMul (outgoing) – fast Triton+cuBLAS implementation
====================================================

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the 4‑D input (B·N·N × dim) – fp32 → fp16.
2️⃣ Fuse the five linear maps (left/right projections + three gates) into a
   single fp16 GEMM:  X_norm @ W_fusedᵀ → (R, 5·H)    (R = B·N·N).
3️⃣ Split the fused result, apply the (optional) pair‑wise mask, evaluate the
   sigmoid gates and write the three tensors *directly* in the layout
   (B, H, N, N) that is required by the upcoming batched GEMM.
   (All arithmetic in fp16, sigmoid computed in fp32 for stability.)
4️⃣ Batched half‑precision GEMM over the hidden dimension:
        out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ
   – performed with ``torch.bmm`` (cuBLAS Tensor‑Core kernels).
5️⃣ Permute the result to (B, N, N, H) and apply a second LayerNorm + 
   element‑wise multiplication with ``out_gate`` (fused Triton kernel).
6️⃣ Final linear projection back to the original ``dim`` (fp16 → fp32).

Only three Triton kernels are used – all heavy matrix‑multiplications are
handled by cuBLAS, which gives near‑peak performance on an H100.  The
expensive transposes required by a naïve implementation are eliminated by
writing the projected tensors directly in the (B, H, N, N) layout (step 3)."""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Row‑wise LayerNorm (float32 → float16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,        # const   *[rows, dim]      (float32)
    y_ptr,        # mutable *[rows, dim]      (float16)
    w_ptr,        # const   *[dim]           (float32)   γ
    b_ptr,        # const   *[dim]           (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,    # rows processed per program
    BLOCK_D: tl.constexpr,    # dim processed per inner loop
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)          # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- write normalized values (γ·(x‑μ)·rstd + β) -----------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        out = norm * w[None, :] + b[None, :]                 # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# 2️⃣ Split fused projection, apply mask & sigmoid gates,
#     write directly in (B, H, N, N) layout
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[R, 5*H]   (fp16)
    left_ptr,          # mutable *[B, H, N, N] (fp16)
    right_ptr,         # mutable *[B, H, N, N] (fp16)
    out_gate_ptr,      # mutable *[B, H, N, N] (fp16)
    mask_ptr,          # const   *[R]       (fp16) (ignored if HAS_MASK=0)
    R: tl.int32,
    H: tl.int32,
    N: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row r = b·N·N + i·N + k (r ∈ [0,R)):
        left[b,h,i,k]  = (proj_left  * mask[r]) * sigmoid(proj_left_gate)
        right[b,h,i,k] = (proj_right * mask[r]) * sigmoid(proj_right_gate)
        out_gate[b,h,i,k] = sigmoid(proj_out_gate)
    All arithmetic in fp16 except sigmoid (computed in fp32).
    The results are stored directly in the (B,H,N,N) layout required by the
    batched GEMM, thus avoiding an extra transpose.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < R

    # ---- decode (b,i,k) from flat row index ---------------------------------
    nn = N * N
    b = offs_r // nn
    rem = offs_r - b * nn
    i = rem // N
    k = rem - i * N

    # broadcast mask (if any)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # broadcast over H
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # constant stride for hidden dimension in the output layout
    hidden_stride = N * N                     # distance between consecutive h

    # base offset (b·H·N·N + i·N + k) – common to all hidden slices of a row
    base_offset = b * (H * N * N) + i * N + k          # [BLOCK_R]

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # ----- load the five blocks from the fused projection ----------------
        proj_base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # [BLOCK_R, BLOCK_H]

        left_raw     = tl.load(proj_ptr + proj_base,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_raw    = tl.load(proj_ptr + proj_base + H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        left_gate    = tl.load(proj_ptr + proj_base + 2 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        right_gate   = tl.load(proj_ptr + proj_base + 3 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)
        out_gate_raw = tl.load(proj_ptr + proj_base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # ----- sigmoid (stable fp32) ---------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # ----- apply mask and gates ----------------------------------------
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ----- compute destination offsets -----------------------------------
        # offset = base_offset + h * hidden_stride
        dst_offset = base_offset[:, None] + offs_h[None, :] * hidden_stride

        # store
        tl.store(left_ptr      + dst_offset,
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + dst_offset,
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + dst_offset,
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 3️⃣ Second LayerNorm + element‑wise multiplication with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (fp16)
    gate_ptr,         # const   *[rows, dim]   (fp16)
    y_ptr,            # mutable *[rows, dim]   (fp16)
    w_ptr,            # const   *[dim]        (fp32)  γ
    b_ptr,            # const   *[dim]        (fp32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """

    The final result is stored as fp16.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)                 # [BLOCK_M]
    mask_m = offs_m < rows

    # ---- compute mean & variance (fp32) ---------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ---- normalize, apply γ/β, multiply with gate, write -----------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_val = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                        mask=mask_m[:, None] & mask_d[None, :],
                        other=0.0).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_val - mean[:, None]) * rstd[:, None]           # fp32

        # γ, β
        w = tl.load(w_ptr + offs_d, mask=mask_d, other=0.0)      # fp32
        b = tl.load(b_ptr + offs_d, mask=mask_d, other=0.0)      # fp32
        norm = norm * w[None, :] + b[None, :]                    # fp32

        # gate value
        gate_val = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                           mask=mask_m[:, None] & mask_d[None, :],
                           other=0.0).to(tl.float32)          # fp32

        out = norm * gate_val                                    # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 out_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward kernel for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
            - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
            - weights_dict : mapping with all Linear/LayerNorm parameters
                             (float32 tensors, bias=False for Linear)
            - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack arguments
    # --------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] float32
    x_norm = torch.empty(R, dim, dtype=torch.float16, device=device)

    BLOCK_M = 128      # rows per program (tuned for H100)
    BLOCK_D = 64       # dim chunk size
    grid = (triton.cdiv(R, BLOCK_M),)
    layernorm_kernel[grid](
        x_flat,
        x_norm,
        w["norm.weight"],     # γ (float32)
        w["norm.bias"],       # β (float32)
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # --------------------------------------------------------------
    # 2️⃣ Fused linear: left/right proj + all gates → (R, 5·H)
    # --------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                               # [5·H, dim]

    proj = F.linear(x_norm, fused_w)                   # fp16, shape [R, 5·H]
    del fused_w, x_norm                                # free intermediates

    # --------------------------------------------------------------
    # 3️⃣ Split, mask & apply sigmoid gates (write directly in B·H·N·N layout)
    # --------------------------------------------------------------
    left      = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    right     = torch.empty(B, H, N, N, dtype=torch.float16, device=device)
    out_gate  = torch.empty(B, H, N, N, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        has_mask = 0
        mask_ptr = dummy_mask
    else:
        mask_flat = mask.to(torch.float16).reshape(R)   # [R]
        has_mask = 1
        mask_ptr = mask_flat

    BLOCK_R = 128          # rows per program
    BLOCK_H = 32           # hidden dim chunk
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        N,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dim)
    # --------------------------------------------------------------
    # shape (B·H, N, N) – contiguous for cuBLAS
    left_mat  = left.view(B * H, N, N)
    right_mat = right.view(B * H, N, N)

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # reshape back to (B, H, N, N) (still contiguous)
    out = out_mat.view(B, H, N, N)

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication by out_gate (fused kernel)
    # --------------------------------------------------------------
    # Need layout (B·N·N, H) → permute + contiguous
    out_perm = out.permute(0, 2, 3, 1).contiguous()            # [B, N, N, H]
    gate_perm = out_gate.permute(0, 2, 3, 1).contiguous()     # [B, N, N, H]

    out_flat = out_perm.view(R, H).contiguous()
    gate_flat = gate_perm.view(R, H).contiguous()
    out_fused = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    grid2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_mul_kernel[grid2](
        out_flat,
        gate_flat,
        out_fused,
        w["to_out_norm.weight"],   # γ (float32)
        w["to_out_norm.bias"],     # β (float32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, gate_flat, out, out_gate, out_perm, gate_perm

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        w["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
