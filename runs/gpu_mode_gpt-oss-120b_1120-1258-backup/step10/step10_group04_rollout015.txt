@triton.jit
def proj_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (float16)
    rstd_ptr,          # const   *[rows]    (float32)
    bias_ptr,          # const   *[5*H]    (float32)
    mask_ptr,          # const   *[rows]   (float16) (optional)
    left_ptr,          # mutable *[rows, H] (float16)
    right_ptr,         # mutable *[rows, H]
    out_gate_ptr,      # mutable *[rows, H]
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # load mask values if needed
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)[:, None]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # load rstd
    rstd_vals = tl.load(rstd_ptr + offs_r, mask=mask_r, other=1.0)  # fp32

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # Compute offsets for the five slices
        # left slice start = 0*H, right slice start = 1*H, left_gate =2*H, right_gate =3*H, out_gate =4*H
        # base pointer per row: proj_ptr + offs_r[:,None] * (5*H) + offs_h[None,:] + slice_offset
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]

        left_raw = tl.load(proj_ptr + base,
                           mask=mask_r[:, None] & mask_h[None, :],
                           other=0.0).to(tl.float16)
        right_raw = tl.load(proj_ptr + base + H,
                            mask=mask_r[:, None] & mask_h[None, :],
                            other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0).to(tl.float16)
        right_gate_raw = tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw = tl.load(proj_ptr + base + 4 * H,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0).to(tl.float16)

        # Load bias slices for each of the five components
        # bias shape [5*H] fp32
        bias_left = tl.load(bias_ptr + offs_h, mask=mask_h, other=0.0)  # fp32
        bias_right = tl.load(bias_ptr + H + offs_h, mask=mask_h, other=0.0)
        bias_left_gate = tl.load(bias_ptr + 2*H + offs_h, mask=mask_h, other=0.0)
        bias_right_gate = tl.load(bias_ptr + 3*H + offs_h, mask=mask_h, other=0.0)
        bias_out_gate = tl.load(bias_ptr + 4*H + offs_h, mask=mask_h, other=0.0)

        rstd_tile = rstd_vals[:, None]  # [BLOCK_R,1] fp32

        # Apply rstd scaling and bias, then convert to fp16 for further ops
        left = (left_raw.to(tl.float32) * rstd_tile + bias_left[None, :]).to(tl.float16)
        right = (right_raw.to(tl.float32) * rstd_tile + bias_right[None, :]).to(tl.float16)
        left_gate = (left_gate_raw.to(tl.float32) * rstd_tile + bias_left_gate[None, :]).to(tl.float16)
        right_gate = (right_gate_raw.to(tl.float32) * rstd_tile + bias_right_gate[None, :]).to(tl.float16)
        out_gate = (out_gate_raw.to(tl.float32) * rstd_tile + bias_out_gate[None, :]).to(tl.float16)

        # sigmoid gates
        left_gate = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate = (1.0 / (1.0 + tl.exp(-out_gate.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left = left * mask_val * left_gate
        right = right * mask_val * right_gate

        # store
        left_ptr + offs_r[:, None] * H + offs_h[None, :]
        etc.

"""
Outgoing TriMul (AlphaFold‑3) – Triton-accelerated forward pass
-----------------------------------------------------------------
We fuse as many element‑wise operations as possible into three Triton kernels:

1️⃣ *center_kernel* – row‑wise LayerNorm “mean & variance” step only.
   It writes:
       • centered = (x - μ) * γ               (FP16)
       • rstd    = rsqrt(σ² + eps)            (FP32, one scalar per row)
   The bias β is *not* applied here – it will be added later together with the
   linear projection bias.

2️⃣ *proj_gate_kernel* – consumes the projected tensor (R,5·H), the per‑row
   ``rstd`` and the pre‑computed bias   β·W_fusedᵀ.
   It performs, per hidden‑dim block:
        y = (proj_block * rstd + bias_block)            # FP32 → FP16
        left  = y_left  * mask * sigmoid(y_left_gate)
        right = y_right * mask * sigmoid(y_right_gate)
        out_gate = sigmoid(y_out_gate)
   The three outputs (left, right, out_gate) have shape (R,H) and are stored
   directly in FP16.

3️⃣ *layernorm_gate_kernel* – the second LayerNorm that follows the
   batched GEMM.  It normalises ``out`` (R,H), multiplies by ``out_gate`` and
   finally applies the learned LN weight/bias.  The result is fed to a
   tiny FP16 linear layer (H → dim) using PyTorch’s highly‑optimised cuBLAS.

All heavy matrix multiplications (center → projection, final projection and
the N×N batched GEMM) are delegated to cuBLAS/torch.bmm, which already
exploits Tensor‑Cores on an H100.  Only the element‑wise parts are realised
in Triton, dramatically reducing kernel‑launch overhead and memory traffic.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 1️⃣ Centering + row‑wise mean/var (no bias, γ applied)
# ----------------------------------------------------------------------
@triton.jit
def center_kernel(
    x_ptr,            # const   *[rows, dim]   (fp32)
    centered_ptr,     # mutable *[rows, dim]   (fp16)
    rstd_ptr,         # mutable *[rows]       (fp32)
    norm_weight_ptr,  # const   *[dim]        (fp32)   γ of the first LN
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim tiles
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # 1) Compute mean and variance (float32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_R], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_R], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)               # fp32, one per row

    # write rstd for the later projection kernel
    tl.store(rstd_ptr + offs_r, rstd, mask=mask_r)

    # ------------------------------------------------------------------
    # 2) Write centred and γ‑scaled tensor (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x = tl.load(x_ptr + offs_r[:, None] * dim + offs_d[None, :],
                    mask=mask_r[:, None] & mask_d[None, :],
                    other=0.0).to(tl.float32)

        gamma = tl.load(norm_weight_ptr + offs_d,
                        mask=mask_d,
                        other=0.0)                 # fp32

        centered = (x - mean[:, None]) * gamma[None, :]   # fp32
        centered_fp16 = centered.to(tl.float16)

        tl.store(
            centered_ptr + offs_r[:, None] * dim + offs_d[None, :],
            centered_fp16,
            mask=mask_r[:, None] & mask_d[None, :],
        )


# ----------------------------------------------------------------------
# 2️⃣ Project, add bias (β·Wᵀ), apply rstd, mask and sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def proj_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)   – result of linear
    rstd_ptr,          # const   *[rows]     (fp32)
    bias_ptr,          # const   *[5*H]      (fp32)   – β·Wᵀ
    mask_ptr,          # const   *[rows]     (fp16)   (ignored if HAS_MASK=0)
    left_ptr,          # mutable *[rows, H] (fp16)
    right_ptr,         # mutable *[rows, H] (fp16)
    out_gate_ptr,      # mutable *[rows, H] (fp16)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)          # [BLOCK_R]
    mask_r = offs_r < rows

    # --------------------------------------------------------------
    # Load optional pair‑wise mask (broadcasted over H)
    # --------------------------------------------------------------
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r,
                           mask=mask_r,
                           other=1.0)[:, None]                 # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # rstd per row (fp32)
    rstd = tl.load(rstd_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]

    # --------------------------------------------------------------
    # Process hidden dimension in tiles
    # --------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for the five contiguous slices: L,R,Lg,Rg,Og
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # [R, H]

        left_raw      = tl.load(proj_ptr + base,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_raw     = tl.load(proj_ptr + base + H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        left_gate_raw = tl.load(proj_ptr + base + 2 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        right_gate_raw= tl.load(proj_ptr + base + 3 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)
        out_gate_raw  = tl.load(proj_ptr + base + 4 * H,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0).to(tl.float16)

        # --------------------------------------------------------------
        # Load row‑wise bias slices (β·Wᵀ)  – one slice per component
        # --------------------------------------------------------------
        bias_l = tl.load(bias_ptr + offs_h,
                         mask=mask_h,
                         other=0.0)                       # fp32
        bias_r = tl.load(bias_ptr + H + offs_h,
                         mask=mask_h,
                         other=0.0)
        bias_lg = tl.load(bias_ptr + 2 * H + offs_h,
                         mask=mask_h,
                         other=0.0)
        bias_rg = tl.load(bias_ptr + 3 * H + offs_h,
                         mask=mask_h,
                         other=0.0)
        bias_og = tl.load(bias_ptr + 4 * H + offs_h,
                         mask=mask_h,
                         other=0.0)

        # --------------------------------------------------------------
        # Apply rstd scaling, add bias and cast back to fp16
        # --------------------------------------------------------------
        rstd_tile = rstd[:, None]          # [BLOCK_R,1] fp32

        left      = (left_raw.to(tl.float32)      * rstd_tile + bias_l[None, :]).to(tl.float16)
        right     = (right_raw.to(tl.float32)     * rstd_tile + bias_r[None, :]).to(tl.float16)
        left_gate = (left_gate_raw.to(tl.float32) * rstd_tile + bias_lg[None, :]).to(tl.float16)
        right_gate= (right_gate_raw.to(tl.float32)* rstd_tile + bias_rg[None, :]).to(tl.float16)
        out_gate  = (out_gate_raw.to(tl.float32)  * rstd_tile + bias_og[None, :]).to(tl.float16)

        # --------------------------------------------------------------
        # Sigmoid gates (computed in fp32 for stability)
        # --------------------------------------------------------------
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate.to(tl.float32)))).to(tl.float16)

        # --------------------------------------------------------------
        # Apply pair‑wise mask and store
        # --------------------------------------------------------------
        left  = left * mask_val * left_gate
        right = right * mask_val * right_gate

        tl.store(left_ptr      + offs_r[:, None] * H + offs_h[None, :],
                 left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_ptr     + offs_r[:, None] * H + offs_h[None, :],
                 right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :],
                 out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# 5️⃣ Second LayerNorm + out‑gate multiplication  (fused)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_gate_kernel(
    x_ptr,          # const   *[rows, dim]   (fp16)  – output of the GEMM
    gate_ptr,       # const   *[rows, dim]   (fp16)  – out_gate from step 2
    y_ptr,          # mutable *[rows, dim]   (fp16)  – LN‑gated result
    weight_ptr,     # const   *[dim]        (fp32)  – γ of second LN
    bias_ptr,       # const   *[dim]        (fp32)  – β of second LN
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)           # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1) Compute mean & variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)   # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(ptr,
                       mask=mask_m[:, None] & mask_d[None, :],
                       other=0.0).to(tl.float32)
        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)          # fp32

    # ------------------------------------------------------------------
    # 2) Normalise, apply LN weight/bias and multiply by out‑gate
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        x_vals = tl.load(x_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)
        g_vals = tl.load(gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
                         mask=mask_m[:, None] & mask_d[None, :],
                         other=0.0).to(tl.float32)

        norm = (x_vals - mean[:, None]) * rstd[:, None]       # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ (fp32)
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β (fp32)

        y_fp32 = (norm * w[None, :] + b[None, :]) * g_vals       # fp32
        y_fp16 = y_fp32.to(tl.float16)

        tl.store(y_ptr + offs_m[:, None] * dim + offs_d[None, :],
                 y_fp16,
                 mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the *outgoing* TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple (input_tensor, mask_tensor, weights_dict, config_dict)
        input_tensor : torch.Tensor, shape [B, N, N, dim], dtype torch.float32
        mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        weights_dict : model weights (all fp32)
        config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # 0️⃣ Unpack ----------------------------------------------------
    # --------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]          # = H
    eps = 1e-5
    device = x.device
    B, N, _, _ = x.shape
    R = B * N * N                         # total rows for row‑wise ops
    H = hidden_dim

    # --------------------------------------------------------------
    # 1️⃣ Compute centred input and per‑row rstd
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                 # [R, dim] fp32
    x_centered = torch.empty_like(x_flat, dtype=torch.float16)  # fp16
    rstd = torch.empty(R, dtype=torch.float32, device=device)  # fp32

    BLOCK_R = 256
    BLOCK_D = 64
    grid_center = (triton.cdiv(R, BLOCK_R),)
    center_kernel[grid_center](
        x_flat,
        x_centered,
        rstd,
        weights["norm.weight"],          # γ (fp32)
        R,
        dim,
        eps,
        BLOCK_R=BLOCK_R,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat                          # free memory early

    # --------------------------------------------------------------
    # 2️⃣ Fused linear projection (no bias) → [R, 5*H]
    # --------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                     # shape (5*H, dim)

    # bias for the projection (β·Wᵀ) – computed once on the host
    bias_out = torch.nn.functional.linear(
        weights["norm.bias"].unsqueeze(0),   # (1, dim)
        w_fused,                            # (5*H, dim)
    ).squeeze(0).contiguous()              # (5*H,)  fp32

    proj = F.linear(x_centered, w_fused, bias=None)   # [R, 5*H] fp16
    del x_centered, w_fused

    # --------------------------------------------------------------
    # 3️⃣ Split, apply rstd, bias, mask and sigmoid gates
    # --------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        mask_fp16 = torch.empty(0, dtype=torch.float16, device=device)  # dummy
    else:
        mask_fp16 = mask.to(torch.float16).reshape(R)        # flatten
        has_mask = 1

    BLOCK_R = 256
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    proj_gate_kernel[grid_split](
        proj,
        rstd,
        bias_out,
        mask_fp16,
        left,
        right,
        out_gate,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, rstd, bias_out, mask_fp16

    # --------------------------------------------------------------
    # 4️⃣ Batched GEMM: (B·H, N, N) @ (B·H, N, N)ᵀ
    # --------------------------------------------------------------
    # reshape to [B, N, N, H] → bring H into batch dimension
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    left_mat = left.reshape(B * H, N, N)      # [B·H, N, N]
    right_mat = right.reshape(B * H, N, N)    # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]
    del left_mat, right_mat, left, right, out_mat

    # --------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused)
    # --------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    gated_norm = torch.empty_like(out_flat, dtype=torch.float16, device=device)

    BLOCK_M = 256
    BLOCK_D = 64
    grid_ln2 = (triton.cdiv(R, BLOCK_M),)
    layernorm_gate_kernel[grid_ln2](
        out_flat,
        out_gate,
        gated_norm,
        weights["to_out_norm.weight"],   # γ (fp32)
        weights["to_out_norm.bias"],     # β (fp32)
        R,
        H,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del out_flat, out_gate, out

    # --------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 GEMM) + cast to fp32
    # --------------------------------------------------------------
    final_half = F.linear(
        gated_norm,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
        bias=None,
    )   # → [R, dim] fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
