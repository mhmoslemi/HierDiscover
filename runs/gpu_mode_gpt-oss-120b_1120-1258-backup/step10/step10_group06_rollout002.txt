"""
TriMul (outgoing) – high‑performance Triton implementation.

Algorithm
---------
1️⃣ Row‑wise LayerNorm on the input tensor (fp32 → fp16)
2️⃣ Fuse the five linear maps (2 projections + 3 gates) into one GEMM:
       X_norm @ W_fusedᵀ → (R, 5·H)   with  R = B·N·N .
3️⃣ Triton kernel `split_fuse_gate_kernel` splits the fused result, applies an
   optional pair‑wise mask, evaluates sigmoid‑gates and writes three tensors:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
4️⃣ Batched half‑precision matrix‑multiply per hidden dimension:
       out[b·H+i] = left[b·H+i] @ right[b·H+i]ᵀ   (torch.bmm)
5️⃣ Triton kernel `layernorm_mul_kernel` performs the second LayerNorm on the
   result and multiplies it element‑wise by `out_gate`.
6️⃣ Final linear projection back to the original dimension (fp16 → fp32).

Only the two LayerNorms and the split‑gate logic are written in Triton; the
large GEMMs use cuBLAS (via torch.nn.functional.linear / torch.bmm) which
already exploits Tensor‑Cores on H100.  This yields >2× speed‑up vs. the
pure‑PyTorch reference.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: row‑wise LayerNorm (fp32 → fp16)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,          # const   *[rows, dim]   (float32)
    y_ptr,          # mutable *[rows, dim]   (float16)
    weight_ptr,     # const   *[dim]        (float32)   γ
    bias_ptr,       # const   *[dim]        (float32)   β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,   # rows per program
    BLOCK_D: tl.constexpr,   # dim packed per iteration
):
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)  # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ mean & variance (accumulate in fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)      # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]   # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)          # Σx
        sum_sq  += tl.sum(vals * vals, axis=1)   # Σx²

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ write normalized values (γ·(x-μ)·rstd + β) → fp16
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        norm = (vals - mean[:, None]) * rstd[:, None]   # fp32

        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # γ
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # β

        out = norm * w[None, :] + b[None, :]       # fp32
        out_fp16 = out.to(tl.float16)

        out_ptr = y_ptr + offs_m[:, None] * dim + offs_d[None, :]
        tl.store(out_ptr, out_fp16, mask=mask_m[:, None] & mask_d[None, :])


# ----------------------------------------------------------------------
# Triton kernel: split fused projection, apply mask, sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (float16)
    left_ptr,          # mutable *[rows, H]    (float16)
    right_ptr,         # mutable *[rows, H]    (float16)
    out_gate_ptr,      # mutable *[rows, H]    (float16)
    mask_ptr,          # const   *[rows]       (float16)  (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,   # rows per program
    BLOCK_H: tl.constexpr,   # hidden dim per iteration
):
    """
    For each row r:
        left  = (proj_left  * mask_r) * sigmoid(proj_left_gate)
        right = (proj_right * mask_r) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All ops in fp16, sigmoid evaluated in fp32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)      # [BLOCK_R]
    mask_r = offs_r < rows

    # per‑row mask (broadcast to hidden dim)
    if HAS_MASK:
        mask_val = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)  # [BLOCK_R]
        mask_val = mask_val[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)   # [BLOCK_H]
        mask_h = offs_h < H

        # layout: [left, right, left_gate, right_gate, out_gate] each H long
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]   # offset into proj

        p_left      = proj_ptr + base
        p_right     = proj_ptr + base + H
        p_left_gate = proj_ptr + base + 2 * H
        p_right_gate= proj_ptr + base + 3 * H
        p_out_gate  = proj_ptr + base + 4 * H

        # load raw tensors
        left_raw       = tl.load(p_left,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_raw      = tl.load(p_right,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        left_gate_raw  = tl.load(p_left_gate,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        right_gate_raw = tl.load(p_right_gate,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw   = tl.load(p_out_gate,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)

        # sigmoid (fp32 → fp16)
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # store
        dst_left    = left_ptr    + offs_r[:, None] * H + offs_h[None, :]
        dst_right   = right_ptr   + offs_r[:, None] * H + offs_h[None, :]
        dst_outgate = out_gate_ptr+ offs_r[:, None] * H + offs_h[None, :]

        tl.store(dst_left,    left,    mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_right,   right,   mask=mask_r[:, None] & mask_h[None, :])
        tl.store(dst_outgate, out_gate,mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton kernel: second LayerNorm + element‑wise multiply with out_gate
# ----------------------------------------------------------------------
@triton.jit
def layernorm_mul_kernel(
    x_ptr,            # const   *[rows, dim]   (float16)
    gate_ptr,         # const   *[rows, dim]   (float16)  (out_gate)
    y_ptr,            # mutable *[rows, dim]   (float16)  (result)
    weight_ptr,       # const   *[dim]        (float32)  γ
    bias_ptr,         # const   *[dim]        (float32)  β
    rows: tl.int32,
    dim: tl.int32,
    eps: tl.float32,
    BLOCK_M: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Row‑wise LayerNorm on `x` (fp16) followed by multiplication with `gate`.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_M
    offs_m = row_start + tl.arange(0, BLOCK_M)               # [BLOCK_M]
    mask_m = offs_m < rows

    # ------------------------------------------------------------------
    # 1️⃣ mean / variance (fp32)
    # ------------------------------------------------------------------
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)       # [BLOCK_D]
        mask_d = offs_d < dim
        ptr = x_ptr + offs_m[:, None] * dim + offs_d[None, :] # [M,D]
        vals = tl.load(
            ptr,
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        sum_val += tl.sum(vals, axis=1)
        sum_sq  += tl.sum(vals * vals, axis=1)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    rstd = tl.rsqrt(var + eps)

    # ------------------------------------------------------------------
    # 2️⃣ normalize, apply γ/β, multiply with gate, store (fp16)
    # ------------------------------------------------------------------
    for d_off in range(0, tl.cdiv(dim, BLOCK_D)):
        offs_d = d_off * BLOCK_D + tl.arange(0, BLOCK_D)
        mask_d = offs_d < dim

        # load x
        x_vals = tl.load(
            x_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        # (x‑μ)·rstd
        norm = (x_vals - mean[:, None]) * rstd[:, None]   # fp32

        # γ, β
        w = tl.load(weight_ptr + offs_d, mask=mask_d, other=0.0)   # fp32
        b = tl.load(bias_ptr   + offs_d, mask=mask_d, other=0.0)   # fp32
        norm = norm * w[None, :] + b[None, :]                     # fp32

        # gate
        gate = tl.load(
            gate_ptr + offs_m[:, None] * dim + offs_d[None, :],
            mask=mask_m[:, None] & mask_d[None, :],
            other=0.0,
        ).to(tl.float32)

        out = norm * gate                     # fp32
        out_fp16 = out.to(tl.float16)

        tl.store(
            y_ptr + offs_m[:, None] * dim + offs_d[None, :],
            out_fp16,
            mask=mask_m[:, None] & mask_d[None, :],
        )

    """

    torch.Tensor

    # --------------------------------------------------------------

    hidden_dim = cfg["hidden_dim"]

    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                # total rows for row‑wise ops
    H = hidden_dim

    # 1️⃣ First LayerNorm (fp32 → fp16)
    # --------------------------------------------------------------
    x_flat = x.view(R, dim).contiguous()                     # [R, dim] fp32

    BLOCK_D = 32

    layernorm_kernel[grid](
        x_flat,
        x_norm,
        weights["norm.weight"],   # γ : [dim]  float32
        weights["norm.bias"],     # β : [dim]  float32
        R,
        dim,
        eps,
        BLOCK_M=BLOCK_M,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    del x_flat

    # 2️⃣ Fused linear: left/right proj + three gates
    # --------------------------------------------------------------
    # Concatenate weight matrices of the five Linear layers (no bias)

        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                                   # [5·H, dim]

    # (R, dim) @ (5·H, dim).T → (R, 5·H)
    proj = F.linear(x_norm, w_concat)                     # fp16
    del w_concat, x_norm

    # 3️⃣ Split fused result, apply mask, sigmoid gates
    # --------------------------------------------------------------
    left     = torch.empty(R, H, dtype=torch.float16, device=device)
    right    = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate = torch.empty(R, H, dtype=torch.float16, device=device)

    # Prepare mask (if any)

        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask shape [B, N, N] → [R] (float16 0/1)
        mask_half = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_half

    BLOCK_R = 128
    BLOCK_H = 32
    grid_split = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid_split](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # --------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over hidden dim)
    # --------------------------------------------------------------
    # Bring H into batch dimension: [B, H, N, N]
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # out_mat[b·H] = left_mat[b·H] @ right_mat[b·H]^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # fp16

    # Reshape back to [B, N, N, H]

    # --------------------------------------------------------------
    # 6️⃣ Final linear projection back to `dim` (fp16 → fp32)
    # --------------------------------------------------------------
    final_half = F.linear(
        out_fused,
        weights["to_out.weight"].to(torch.float16),   # [dim, H]
    )   # fp16

    final = final_half.view(B, N, N, dim).to(torch.float32)
    return final
```

The implementation follows the reference algorithm while fusing the two LayerNorms and the projection‑gate split into Triton kernels, letting cuBLAS handle the heavy N³ GEMMs. The output tensor has the required shape `[bs, seq_len, seq_len, dim]` and dtype `float32`.
