"""
Outgoing TriMul (AlphaFold3) – Triton‑accelerated forward pass

Algorithm
---------
1️⃣  Row‑wise LayerNorm on the 4‑D input (B,N,N,D) via the highly‑optimised
    torch.nn.functional.layer_norm (fp32 → fp16).

2️⃣  Fuse the 5 linear projections (left/right + 3 gates) into a single GEMM
    (x_norm @ W_fused.T) producing a tensor of shape [B,N,N,5*H] in fp16.

3️⃣  One custom Triton kernel (`split_fuse_gate_kernel`) splits the fused tensor
    into `left`, `right` and `out_gate`, applies the optional pair‑wise mask,
    and evaluates the sigmoid gates – all in half‑precision.

4️⃣  Batched GEMM (cuBLAS) carries out the true tri‑multiplicative update:
    (B·H,N,N) @ (B·H,N,N)ᵀ  →  (B·H,N,N).

5️⃣  A second LayerNorm (torch.nn.functional.layer_norm) is applied to the
    GEMM result, then multiplied element‑wise with `out_gate` (still fp16).

6️⃣  Final linear projection back to the original feature dimension (fp16 GEMM)
    and a cast to fp32 yields the module output.

Only step 3 is written in Triton; the two LayerNorms use the highly‑optimised
PyTorch implementation. This reduces kernel‑launch overhead while keeping the
compute‑intensive parts fused in a custom kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# 3️⃣ Split fused projection, apply mask & sigmoid gates (single kernel)
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H] (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]    (fp16)   (ignored if HAS_MASK=0)
    rows: tl.int32,
    H: tl.int32,
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """Split a [rows,5*H] tensor into left/right/out_gate, mask & gate them."""
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)               # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # Load optional pair‑wise mask (broadcasted over H)
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)   # [BLOCK_R]
        mask_val = m[:, None]                                   # [BLOCK_R,1]
    else:
        mask_val = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Process hidden dimension tiles
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)        # [BLOCK_H]
        mask_h = offs_h < H

        # Offsets for the five contiguous slices: [L,R,Lg,Rg,Og]
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]      # [R,H]

        left_raw_ptr    = proj_ptr + base
        right_raw_ptr   = proj_ptr + base + H
        left_gate_ptr   = proj_ptr + base + 2 * H
        right_gate_ptr  = proj_ptr + base + 3 * H
        out_gate_ptr_raw= proj_ptr + base + 4 * H

        left_raw   = tl.load(left_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_raw  = tl.load(right_raw_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        left_g_raw = tl.load(left_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        right_g_raw= tl.load(right_gate_ptr,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)
        out_g_raw  = tl.load(out_gate_ptr_raw,
                             mask=mask_r[:, None] & mask_h[None, :],
                             other=0.0).to(tl.float16)

        # ---- sigmoid (fp32 for stability) ----
        left_gate  = (1.0 / (1.0 + tl.exp(-left_g_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_g_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_g_raw.to(tl.float32)))).to(tl.float16)

        # ---- apply mask & gates ----
        left  = left_raw  * mask_val * left_gate
        right = right_raw * mask_val * right_gate

        # ---- write results ----
        left_dst_ptr      = left_ptr      + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr     = right_ptr     + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr  = out_gate_ptr  + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr, left,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr, right,
                 mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate,
                 mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Custom forward pass for the outgoing TriMul operator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 “outgoing” version).

    Args:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict with keys
            "norm.weight", "norm.bias",
            "left_proj.weight", "right_proj.weight",
            "left_gate.weight", "right_gate.weight", "out_gate.weight",
            "to_out_norm.weight", "to_out_norm.bias",
            "to_out.weight"
          All tensors are float32 on the same device as the inputs.
        - config_dict  : must contain
            "dim" (int), "hidden_dim" (int), "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, cfg = data
    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps = 1e-5
    device = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # total rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ First LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # torch.nn.functional.layer_norm is highly‑optimised on GPU.
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )                                 # [B,N,N,dim] float32
    x_norm_h = x_norm.to(torch.float16)   # half‑precision for the rest

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (produces [B,N,N,5*H] in fp16)
    # ------------------------------------------------------------------
    w_fused = torch.cat(
        [
            weights["left_proj.weight"],
            weights["right_proj.weight"],
            weights["left_gate.weight"],
            weights["right_gate.weight"],
            weights["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)                # [5*H, dim]

    proj = F.linear(x_norm_h, w_fused)   # [B,N,N,5*H] fp16
    proj_flat = proj.view(-1, 5 * H).contiguous()   # [R,5*H]

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask, apply sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    if cfg.get("nomask", True):
        has_mask = 0
        dummy_mask = torch.empty(0, dtype=torch.float16, device=device)
        mask_ptr = dummy_mask
    else:
        # mask values are 0/1 → float16, flattened to [R]
        mask_fp16 = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_fp16

    BLOCK_R = 256   # rows per program
    BLOCK_H = 64    # hidden dim tiles
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj_flat,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, proj_flat, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to [B, H, N, N] for batched GEMM
    left = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B,H,N,N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous() # [B,H,N,N]

    left_mat = left.view(B * H, N, N)    # [B·H, N, N]
    right_mat = right.view(B * H, N, N)  # [B·H, N, N]

    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]
    del left_mat, right_mat, left, right, out_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + out‑gate multiplication (fused in Python)
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )                                 # [B,N,N,H] float32
    out_norm_h = out_norm.to(torch.float16)

    # reshape out_gate to match out shape
    out_gate = out_gate.view(B, N, N, H)

    gated = out_norm_h * out_gate      # element‑wise gating (fp16)

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim` (fp16 GEMM) + cast to fp32
    # ------------------------------------------------------------------
    to_out_w_fp16 = weights["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(gated, to_out_w_fp16, bias=None)       # [B,N,N,dim] fp16

    final = final_half.to(torch.float32)
    return final
