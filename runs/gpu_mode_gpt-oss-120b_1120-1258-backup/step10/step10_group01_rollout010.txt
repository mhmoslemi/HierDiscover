"""
Optimized “outgoing” TriMul forward pass (AlphaFold‑3).

Algorithm
---------
1️⃣  Layer‑norm on the input tensor (B,N,N,dim) – performed with
    torch.nn.functional.layer_norm (GPU‑native, FP16 accumulation).

2️⃣  Fuse the five linear maps (left/right projections + three gates) into a
    single GEMM:
        X_norm @ W_fusedᵀ → (R, 5·H)   where R = B·N·N, H = hidden_dim.
    The fused weight matrix is built once from the supplied model weights
    and kept in FP16.

3️⃣  Triton kernel (`split_fuse_gate_kernel`) splits the fused projection,
    applies the optional pair‑wise mask, evaluates the three sigmoid gates,
    and writes three FP16 tensors:
        left  = (proj_left  * mask) * sigmoid(proj_left_gate)
        right = (proj_right * mask) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)

4️⃣  Tri‑multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] · right[b,j,k,h]
    Implemented as a batched GEMM with torch.bmm on tensors of shape
    (B·H, N, N).  This uses cuBLAS Tensor‑Core kernels and is the dominant
    compute (≈ N³·H FLOPs).

5️⃣  Second Layer‑norm on the intermediate result followed by element‑wise
    multiplication with `out_gate`.  Both operations are fused in a single
    torch.layer_norm + broadcasted multiplication (the data movement is tiny).

6️⃣  Final linear projection back to `dim` – a standard FP16 GEMM.

Only the mask handling & gate computation is written in Triton (the
heaviest memory‑bound step).  All large matrix multiplications are delegated
to cuBLAS/Tensor‑Core for peak performance on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel 1: split fused projection, apply optional mask & sigmoid gates
# ----------------------------------------------------------------------
@triton.jit
def split_fuse_gate_kernel(
    proj_ptr,          # const   *[rows, 5*H]   (fp16)
    left_ptr,          # mutable *[rows, H]   (fp16)
    right_ptr,         # mutable *[rows, H]   (fp16)
    out_gate_ptr,      # mutable *[rows, H]   (fp16)
    mask_ptr,          # const   *[rows]      (fp16) – ignored if HAS_MASK==0
    rows: tl.int32,    # total rows = B·N·N
    H: tl.int32,       # hidden dimension
    HAS_MASK: tl.constexpr,
    BLOCK_R: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    """
    For each row i (i indexes B·N·N):
        left  = (proj_left  * mask[i]) * sigmoid(proj_left_gate)
        right = (proj_right * mask[i]) * sigmoid(proj_right_gate)
        out_gate = sigmoid(proj_out_gate)
    All arithmetic in fp16, sigmoid evaluated in fp32 for stability.
    """
    pid = tl.program_id(0)
    row_start = pid * BLOCK_R
    offs_r = row_start + tl.arange(0, BLOCK_R)                # [BLOCK_R]
    mask_r = offs_r < rows

    # ------------------------------------------------------------------
    # mask value (scalar per row) – broadcast to hidden dimension
    # ------------------------------------------------------------------
    if HAS_MASK:
        m = tl.load(mask_ptr + offs_r, mask=mask_r, other=1.0)      # [BLOCK_R]
        m = m[:, None]                                            # broadcast
    else:
        m = tl.full([BLOCK_R, 1], 1.0, dtype=tl.float16)

    # ------------------------------------------------------------------
    # Loop over hidden dimension
    # ------------------------------------------------------------------
    for h_off in range(0, tl.cdiv(H, BLOCK_H)):
        offs_h = h_off * BLOCK_H + tl.arange(0, BLOCK_H)          # [BLOCK_H]
        mask_h = offs_h < H

        # layout in the fused projection:
        # [left, right, left_gate, right_gate, out_gate] each of size H
        base = offs_r[:, None] * (5 * H) + offs_h[None, :]        # byte offset base

        left_raw_ptr      = proj_ptr + base
        right_raw_ptr     = proj_ptr + base + H
        left_gate_ptr     = proj_ptr + base + 2 * H
        right_gate_ptr    = proj_ptr + base + 3 * H
        out_gate_raw_ptr = proj_ptr + base + 4 * H

        left_raw      = tl.load(left_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_raw     = tl.load(right_raw_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        left_gate_raw = tl.load(left_gate_ptr,
                                mask=mask_r[:, None] & mask_h[None, :],
                                other=0.0)
        right_gate_raw = tl.load(right_gate_ptr,
                                 mask=mask_r[:, None] & mask_h[None, :],
                                 other=0.0)
        out_gate_raw = tl.load(out_gate_raw_ptr,
                               mask=mask_r[:, None] & mask_h[None, :],
                               other=0.0)

        # sigmoid in fp32 → fp16
        left_gate  = (1.0 / (1.0 + tl.exp(-left_gate_raw.to(tl.float32)))).to(tl.float16)
        right_gate = (1.0 / (1.0 + tl.exp(-right_gate_raw.to(tl.float32)))).to(tl.float16)
        out_gate   = (1.0 / (1.0 + tl.exp(-out_gate_raw.to(tl.float32)))).to(tl.float16)

        # apply mask and gates
        left  = left_raw  * m * left_gate
        right = right_raw * m * right_gate

        # store results
        left_dst_ptr  = left_ptr  + offs_r[:, None] * H + offs_h[None, :]
        right_dst_ptr = right_ptr + offs_r[:, None] * H + offs_h[None, :]
        out_gate_dst_ptr = out_gate_ptr + offs_r[:, None] * H + offs_h[None, :]

        tl.store(left_dst_ptr,  left,  mask=mask_r[:, None] & mask_h[None, :])
        tl.store(right_dst_ptr, right, mask=mask_r[:, None] & mask_h[None, :])
        tl.store(out_gate_dst_ptr, out_gate, mask=mask_r[:, None] & mask_h[None, :])


# ----------------------------------------------------------------------
# Triton‑backed custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], dtype float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/float) or None
        - weights_dict : dict of model parameters (float32 tensors, bias=False
                         for Linear layers)
        - config_dict  : must contain "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, w, cfg = data
    dim        = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    eps        = 1e-5
    device     = x.device

    B, N, _, _ = x.shape
    R = B * N * N                     # rows for row‑wise ops
    H = hidden_dim

    # ------------------------------------------------------------------
    # 1️⃣ Row‑wise LayerNorm (use PyTorch builtin – highly tuned)
    # ------------------------------------------------------------------
    # cast input to half, convert LN parameters to half as well
    x_half = x.to(torch.float16)
    norm_w = w["norm.weight"].to(torch.float16)
    norm_b = w["norm.bias"].to(torch.float16)
    x_norm = F.layer_norm(
        x_half,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )                                 # [B, N, N, dim]  fp16

    # ------------------------------------------------------------------
    # 2️⃣ Fused projection (5 linear maps in one GEMM)
    # ------------------------------------------------------------------
    fused_w = torch.cat(
        [
            w["left_proj.weight"],
            w["right_proj.weight"],
            w["left_gate.weight"],
            w["right_gate.weight"],
            w["out_gate.weight"],
        ],
        dim=0,
    ).to(torch.float16)               # shape [5·H, dim]

    # flatten batch & seq dimensions to treat each pair (b,i,k) as a row
    x_flat = x_norm.view(R, dim).contiguous()
    proj   = F.linear(x_flat, fused_w)   # [R, 5·H]  fp16
    del fused_w, x_flat, x_norm          # free intermediates

    # ------------------------------------------------------------------
    # 3️⃣ Split, mask & sigmoid gates (single Triton kernel)
    # ------------------------------------------------------------------
    left      = torch.empty(R, H, dtype=torch.float16, device=device)
    right     = torch.empty(R, H, dtype=torch.float16, device=device)
    out_gate  = torch.empty(R, H, dtype=torch.float16, device=device)

    # mask handling – if “nomask” is True we skip the mask completely
    if cfg.get("nomask", True):
        has_mask = 0
        mask_ptr = torch.empty(0, dtype=torch.float16, device=device)
    else:
        # mask is shape [B, N, N]; flatten to [R] and cast to fp16
        mask_half = mask.to(torch.float16).reshape(R)
        has_mask = 1
        mask_ptr = mask_half

    BLOCK_R = 128               # rows per program instance
    BLOCK_H = 64                # hidden‑dim chunk (covers hidden_dim ≤ 128 efficiently)
    grid = (triton.cdiv(R, BLOCK_R),)
    split_fuse_gate_kernel[grid](
        proj,
        left,
        right,
        out_gate,
        mask_ptr,
        R,
        H,
        HAS_MASK=has_mask,
        BLOCK_R=BLOCK_R,
        BLOCK_H=BLOCK_H,
        num_warps=4,
    )
    del proj, mask_ptr

    # ------------------------------------------------------------------
    # 4️⃣ Tri‑multiplicative update (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    # reshape to [B, N, N, H]  →  permute to bring H in front for batched GEMM
    left  = left.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.view(B, N, N, H).permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left.view(B * H, N, N)      # [B·H, N, N]
    right_mat = right.view(B * H, N, N)     # [B·H, N, N]

    # left @ rightᵀ   (half‑precision Tensor‑Core)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # [B·H, N, N]

    # restore original layout  →  [B, N, N, H]  (still fp16)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    del out_mat, left, right, left_mat, right_mat

    # ------------------------------------------------------------------
    # 5️⃣ Second LayerNorm + multiplication with out_gate
    # ------------------------------------------------------------------
    out_flat = out.view(R, H).contiguous()
    # LayerNorm (GPU native) – parameters are float16
    to_out_norm_w = w["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = w["to_out_norm.bias"].to(torch.float16)
    out_norm = F.layer_norm(
        out_flat,
        (H,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )                                 # [R, H] fp16
    # element‑wise gating
    out_fused = out_norm * out_gate     # [R, H] fp16
    del out_flat, out_norm, out_gate, out

    # ------------------------------------------------------------------
    # 6️⃣ Final projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = w["to_out.weight"].to(torch.float16)   # [dim, H]
    final_half = F.linear(out_fused, to_out_w)        # [R, dim]  fp16
    final = final_half.view(B, N, N, dim).to(torch.float32)

    return final
```
